def active_lag_descent(h0, time_limit):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    current_eval = evaluate_c5(h)
    best_h = h.copy()
    best_eval = current_eval
    k_active = min(n, 20)  # number of active lags
    eta = 0.05
    
    t_start = time.time()
    no_improve = 0
    max_iter = 5000
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        # Compute all lag values
        f = dx * np.correlate(h, 1.0 - h, mode="full")
        current_eval = np.max(f)
        
        # Select active lags
        top_k = np.argsort(f)[-k_active:]
        
        # Compute average gradient over active lags
        grad = np.zeros(n)
        for m in top_k:
            k = m - (n - 1)
            if k >= 0:
                L = n - k
                grad[k:n] += dx * (1.0 - h[:L])
                grad[:L] += dx * (-h[k:n])
            else:
                r = -k
                L = n - r
                grad[:L] += dx * (1.0 - h[r:n])
                grad[r:n] += dx * (-h[:L])
        grad /= k_active
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        direction = -grad  # minimize
        
        # Backtracking line search
        step_eta = eta
        improved = False
        for _ in range(20):
            h_trial = project_feasible(h + step_eta * direction, n)
            eval_trial = evaluate_c5(h_trial)
            if eval_trial < current_eval - 1e-12:

                current_eval = eval_trial
                improved = True
                break

            no_improve = 0
            if current_eval < best_eval:
                best_eval = current_eval
                best_h = h.copy()
        else:
            no_improve += 1
            k_active = min(k_active + 2, n)  # increase active set
            if no_improve > 50:
                break

def resample(h_old, n_new):

    x_old = np.arange(n_old) + 0.5  # center of each bin
    # Map new bins to old positions

def project_feasible(y, n):
    """Project y onto {h: 0<=h[i]<=1, sum(h)=n/2} using bisection for shift."""
    target = n / 2.0
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    # Bisection for alpha such that sum(clip(y+alpha, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                step = min(float(room[i]), residual)
                h[i] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                step = min(float(h[i]), -residual)
                h[i] -= step
                residual += step

                    break
    return h

def project_feasible(y, n):
    target = n / 2.0
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    # Find alpha such that sum(clip(y + alpha, 0, 1)) = target
    # The function S(alpha) = sum(clip(y + alpha, 0, 1)) is monotonically increasing
    # We can find the alpha by sorting
    
    # S(alpha) = sum of (y_i + alpha) for those where 0 <= y_i + alpha <= 1
    #           + 1 for those where y_i + alpha > 1
    #           + 0 for those where y_i + alpha < 0
    
    # This is piecewise linear in alpha, with breakpoints at -y_i and 1-y_i
    # We can binary search over these breakpoints
    
    # Simpler: just use bisection but with fewer iterations
    # Since y is in [0,1] (or close to it), alpha is in [-1, 1] typically
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # Quick fix
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # Simple fix: distribute to non-saturated elements
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if residual > 0:
            count = mask.sum()
            if count > 0:
                h[mask] += residual / count
                h = np.clip(h, 0, 1)
        else:
            count = mask.sum()
            if count > 0:
                h[mask] += residual / count  # residual is negative
                h = np.clip(h, 0, 1)
    return h

```python
import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (all top-level)

def project_feasible(y, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    y = np.where(np.isfinite(y), y, 0.5)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(y + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        # Distribute residual to non-saturated elements
        if res > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += res / len(idx)
            h = np.clip(h, 0.0, 1.0)
            # Second pass for any remaining
            res2 = target - h.sum()
            if abs(res2) > 1e-14:
                if res2 > 0:
                    idx2 = np.where(h < 1.0)[0]
                else:
                    idx2 = np.where(h > 0.0)[0]
                if len(idx2) > 0:
                    h[idx2] += res2 / len(idx2)
                    h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    """Return (c5_value, full_correlation_array)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def compute_all_gradients(h, n, dx, weights):
    """Compute weighted sum of lag gradients."""
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-8:
            continue
        k = m - (n - 1)
        if k >= 0:
            L = n - k
            grad[k:] += w * dx * (1.0 - h[:L])
            grad[:L] += w * dx * (-h[k:])
        else:
            r = -k
            L = n - r
            grad[:L] += w * dx * (1.0 - h[r:])
            grad[r:] += w * dx * (-h[:L])
    return grad

def soft_descent(h0, time_limit, tau_init=0.003, tau_min=0.0005):
    """Soft gradient descent with annealing."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    tau = tau_init
    eta = 0.1
    best_h = h.copy()

    t0 = time.time()
    stagnant = 0
    
    while time.time() - t0 < time_limit:

        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
        
        # Compute softmax weights
        fmax = np.max(f)
        z = (f - fmax) / tau
        z = np.clip(z, -40, 40)
        e = np.exp(z)
        w = e / e.sum()
        
        # Gradient
        grad = compute_all_gradients(h, n, dx, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
