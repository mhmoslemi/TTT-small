import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h, dx):
    """Compute C5 bound = max of full correlation * dx."""
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def project_feasible(h, n):
    """Project h onto {x : sum(x)=n/2, 0<=x<=1}."""
    target = n / 2.0
    if abs(h.sum() - target) < 1e-12 and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def coordinate_sweep(h, n, rng=None):
    """One full sweep of coordinate descent. Returns updated h and c5 value."""
    c = np.correlate(h, 1.0 - h, mode="full")
    shifts = np.arange(2 * n - 1) - (n - 1)
    order = np.arange(n) if rng is None else rng.permutation(n)
    
    for j in order:
        x0 = h[j]
        idx1 = j - shifts
        idx2 = j + shifts
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        
        a = np.zeros(2 * n - 1)
        a[mask1] += (1.0 - h[idx1[mask1]])
        a[mask2] -= h[idx2[mask2]]
        
        # Define objective
        c_local = c  # reference (will be updated below)
        x0_local = x0
        
        def obj(x):
            # Compute new max correlation value
            linear = c_local + (x - x0_local) * a
            quad = c_local[n - 1] + x * (1.0 - x) - x0_local * (1.0 - x0_local)
            return max(linear.max(), quad)
        
        res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        new_x = float(res.x)
        h[j] = new_x
        c = np.correlate(h, 1.0 - h, mode="full")
    
    return h, c.max()

def pnorm_gradient(h, n, p):
    """Compute gradient of (sum c_k^p)^(1/p)."""
    c = np.correlate(h, 1.0 - h, mode="full")
    # Avoid issues with negative c values (shouldn't happen but safeguard)
    c_clipped = np.maximum(c, 1e-30)
    cp = c_clipped ** (p - 1)
    norm = (c_clipped ** p).sum() ** (1.0 / p)
    
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    
    for j in range(n):
        idx1 = j - shifts
        idx2 = j + shifts
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        
        a = np.zeros(2 * n - 1)
        a[mask1] += (1.0 - h[idx1[mask1]])
        a[mask2] -= h[idx2[mask2]]
        # For the quadratic term (m = n-1):
        a[n-1] = 1.0 - 2.0 * h[j]  # derivative of x(1-x)
        
        grad[j] = (cp @ a) / norm
    
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h, n)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Phase 1: Coordinate descent (40% of budget)
    t1_end = start_time + budget_s * 0.4
    sweep_count = 0
    no_improve_count = 0
    
    while time.time() < t1_end:
        h, c_max = coordinate_sweep(h, n, rng)
        h = project_feasible(h, n)
        c5 = compute_c5(h, dx)
        sweep_count += 1
        
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > 50:
            # Try a perturbation
            n_pert = rng.integers(5, 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            h[idx] = np.clip(h[idx] + rng.normal(0, 0.03, n_pert), 0, 1)
            h = project_feasible(h, n)
            no_improve_count = 0
    
    # Phase 2: Perturbation + re-optimization (35% of budget)
    t2_end = start_time + budget_s * 0.75
    attempt = 0
    while time.time() < t2_end:
        attempt += 1
        # Create perturbed copy
        h_trial = best_h.copy()
        n_pert = rng.integers(5, 30)
        idx = rng.choice(n, size=n_pert, replace=False)
        sigma = 0.02 + 0.05 * (attempt % 5) / 5.0
        h_trial[idx] = np.clip(h_trial[idx] + rng.normal(0, sigma, n_pert), 0, 1)
        h_trial = project_feasible(h_trial, n)
        
        # Optimize

            h_trial, _ = coordinate_sweep(h_trial, n, rng)
            h_trial = project_feasible(h_trial, n)

            best_c5 = c5
