def compute_c5_all(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    """Project h onto {x: 0 <= x <= 1, sum(x) = target_sum}"""
    N = len(h)
    # Bisection on shift
    lo, hi = -max(h) - 1, 1 - min(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        shifted = np.clip(h + mid, 0, 1)
        s = np.sum(shifted)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix any residual
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute diff among non-boundary elements
        mask = (result > 1e-6) & (result < 1 - 1e-6)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0, 1)
    return result

    # Log-sum-exp
    m = np.max(f)
    return tau * np.log(np.sum(np.exp((f - m) / tau))) + m

def grad_c5_smoothed(h, tau, dx):
    N = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    f = corr * dx
    
    m = np.max(f)
    weights = np.exp((f - m) / tau)
    total = np.sum(weights)
    weights /= total

        if weights[r] < 1e-15:

        w = weights[r] * dx

        # Contribution from -h[i+k] term: valid when 0 <= i+k < N, i.e., -k <= i <= N-1-k

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    N = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_correlations(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smoothed_max(f, tau):
    m = np.max(f)
    return m + tau * np.log(np.mean(np.exp((f - m) / tau)))

def grad_smoothed_max(h, tau, dx):
    N = len(h)
    f = compute_correlations(h, dx)
    m = np.max(f)
    w = np.exp((f - m) / tau)
    w /= np.sum(w)
    
    grad = np.zeros(N)
    for r in range(2 * N - 1):
        if w[r] < 1e-14:
            continue
        k = r - (N - 1)
        wr = w[r] * dx
        # First term: (1-h[i-k]) for valid i-k
        s1 = max(0, k)
        e1 = min(N - 1, k + N - 1)
        if s1 <= e1:
            grad[s1:e1+1] += wr * (1.0 - h[s1-k:e1-k+1])
        # Second term: -h[i+k] for valid i+k
        s2 = max(0, -k)
        e2 = min(N - 1, N - 1 - k)
        if s2 <= e2:
            grad[s2:e2+1] += wr * (-h[s2+k:e2+k+1])
    return grad

def active_set_optimize(h, dx, target_sum, K=10, max_rounds=50):

    best = h.copy()
    
    for round in range(max_rounds):
        f = compute_correlations(best, dx)
        c5_val = np.max(f)
        
        # Get top K lags
        top_k = np.argsort(f)[-K:]

        def obj(x):
            return x[-1]  # t is the last variable
