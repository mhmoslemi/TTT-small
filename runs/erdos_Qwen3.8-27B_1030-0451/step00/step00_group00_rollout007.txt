def pair_line_params(h, u, v):
    n = len(h)
    L = 2 * n - 1
    A = np.zeros(L)
    m = v - u  # difference
    
    for k in range(L):
        d = k - (n - 1)
        # Partial w.r.t. h[u]
        # A_u: 1 if 0 <= u+d < n
        au = 1.0 if (0 <= u + d < n) else 0.0
        bu = 1.0 if (0 <= u - d < n) else 0.0
        du = au * (1.0 - h[u + d] if 0 <= u + d < n else 0.0) - bu * h[u - d] if 0 <= u - d < n else 0.0
        
        # Partial w.r.t. h[v]
        av = 1.0 if (0 <= v + d < n) else 0.0
        bv = 1.0 if (0 <= v - d < n) else 0.0
        dv = av * (1.0 - h[v + d] if 0 <= v + d < n else 0.0) - bv * h[v - d] if 0 <= v - d < n else 0.0
        
        A[k] = du - dv
    
    # Quadratic correction
    G = np.zeros(L)
    k1 = (n - 1) + m
    k2 = (n - 1) - m
    if 0 <= k1 < L:
        G[k1] = 1.0
    if 0 <= k2 < L and k2 != k1:
        G[k2] = 1.0
    
    return A, G

            h_new[u] += s_star
            h_new[v] -= s_star
            # Clip (shouldn't be needed if interval is correct, but safety)
            h_new[u] = np.clip(h_new[u], 0, 1)
            h_new[v] = np.clip(h_new[v], 0, 1)

                continue
            # Exact evaluation

                h = h_new
                Q = Q_new
                best_c = c_new
                improved = True
                break  # restart with new h

    r2_end = v + n
    A[r2_start:r2_end] -= neg_h_rev

    n_grid = 200
    grid = np.linspace(s_lo, s_hi, n_grid)
    vals = np.zeros(n_grid)
    for i, s in enumerate(grid):
        vals[i] = np.max(Q + A*s + G*s*s)
    
    best_idx = np.argmin(vals)
    
    # Refine with ternary search in local neighborhood
    lo = grid[max(0, best_idx-1)]
    hi = grid[min(n_grid-1, best_idx+1)]
    
    for _ in range(40):

        m2 = hi - (hi-lo)/3
        f1 = np.max(Q + A*m1 + G*m1*m1)
        f2 = np.max(Q + A*m2 + G*m2*m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    s_star = (lo + hi) / 2
    return s_star
