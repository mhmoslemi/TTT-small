def exact_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full')
    return np.max(full * dx)

def compute_active_gradient(h, top_k=20, tau=0.01):
    """Compute weighted gradient from top K active lags."""
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full')
    scores = full * dx
    
    # Find top K lags
    M = len(full)  # 2n-1
    if M <= top_k:
        top_idx = np.arange(M)
        top_vals = scores
    else:
        # Use softmax weights over all, but focus on top
        # Threshold: keep lags within tau of max
        max_val = np.max(scores)
        threshold = max_val - tau
        active = scores >= threshold
        if np.sum(active) > top_k:
            # Take top K
            all_idx = np.argsort(scores)[::-1][:top_k]
            top_idx = all_idx
            top_vals = scores[all_idx]
        else:
            top_idx = np.where(active)[0]
            top_vals = scores[top_idx]
    
    if len(top_idx) == 0:
        return None, 0.0
    
    # Compute weights (softmax over active)
    # Scale to make weights meaningful
    scaled = (top_vals - np.max(top_vals)) / max(tau, 1e-10)
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    center = n - 1
    
    for idx, w in zip(top_idx, weights):
        k = idx - center  # lag
        if k >= 0:
            # grad[m] += (1-h[m-k]) for m in [k, n-1]
            if k < n:
                grad[k:] += (1.0 - h[:n-k]) * w

    """Project h to feasible set [0,1]^n with sum = n/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    s = np.sum(h)
    diff = target - s
    if abs(diff) < tol:
        return h
    # Distribute adjustment
    # Find elements not at bounds
    interior = (h > tol) & (h < 1.0 - tol)
    if np.sum(interior) == 0:
        # All at bounds, try to adjust
        if diff > 0:
            # Need to increase, find elements at 0
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                per = diff / len(zeros)
                h[zeros] = np.minimum(1.0, h[zeros] + per)
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                per = -diff / len(ones)
                h[ones] = np.maximum(0.0, h[ones] - per)
        return h
    # Spread adjustment over interior
    adj = diff / np.sum(interior)
    h[interior] += adj
    h = np.clip(h, 0.0, 1.0)
    # Final correction on the largest interior element
    s = np.sum(h)
    residual = target - s
    if abs(residual) > 1e-12:

        interior = (h > 1e-12) & (h < 1.0 - 1e-12)

            idx = np.where(interior)[0][0]
            h[idx] += residual

    return h

def pairwise_exchange(h, i, j, delta):
    """Apply pairwise exchange h[i]+=delta, h[j]-=delta if feasible."""
    new_hi = h[i] + delta
    new_hj = h[j] - delta

        h[i] = np.clip(new_hi, 0, 1)
        h[j] = np.clip(new_hj, 0, 1)
        return True
    return False
