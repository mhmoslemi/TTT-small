import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    lo = float(np.min(h)) - 0.1
    hi = float(np.max(h)) + 0.1
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def thue_morse_seq(N):
    t = np.zeros(N, dtype=np.float64)
    for i in range(1, N):
        if i & 1:
            t[i] = 1.0 - t[i >> 1]
        else:
            t[i] = t[i >> 1]
    return t

def resample_linear(h, n_new):
    N = len(h)
    if n_new == N:
        return h.copy()
    x_old = (np.arange(N) + 0.5) * (2.0 / N)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)
