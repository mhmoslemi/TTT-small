import numpy as np
from scipy.optimize import minimize
import time

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lse_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    obj = m + tau * np.log(sum_exp)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        shift = idx - (n - 1)
        wi = w[idx] * dx
        js = max(0, -shift)
        je = min(n, n - shift)
        if js < je:
            grad[js:je] += wi * (1.0 - h[js + shift:je + shift])
        js2 = max(0, shift)
        je2 = min(n, n + shift)
        if js2 < je2:
            grad[js2:je2] -= wi * h[js2 - shift:je2 - shift]
    return obj, grad

def project_feasible(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def interpolate_h(h_src, n_tgt):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_tgt) + 0.5) / n_tgt
    h_new = np.interp(x_tgt, x_src, h_src)
    return project_feasible(h_new, n_tgt / 2.0)

def compute_active_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(corr)
    tol = max(1e-10, 1e-6 * max_val)
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for idx in active:
        shift = idx - (n - 1)
        js = max(0, -shift)
        je = min(n, n - shift)
        if js < je:
            grad[js:je] += (1.0 - h[js + shift:je + shift])
        js2 = max(0, shift)
        je2 = min(n, n + shift)
        if js2 < je2:
            grad[js2:je2] -= h[js2 - shift:je2 - shift]
    grad /= len(active)
    return grad * dx, corr

def local_search(h, max_time=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    target = n / 2.0
    c = c5_exact(h)
    t0 = time.time()
    no_improve = 0
    stick_count = 0

    while time.time() - t0 < max_time:
        improved = False
        grad, corr = compute_active_grad(h)

        # Gradient-informed pair moves
        order = np.argsort(grad)
        k = min(8, n // 2)
        tried = set()
        for a in range(k):
            for b in range(n - k, n):
                i, j = int(order[a]), int(order[b])
                if i == j or (i, j) in tried:
                    continue
                tried.add((i, j))
                max_d = min(1.0 - h[i], h[j])
                if max_d < 1e-10:
                    continue
                for frac in [0.99, 0.7, 0.4, 0.2, 0.08, 0.02]:
                    d = frac * max_d
                    h[i] += d
                    h[j] -= d
                    c_new = c5_exact(h)
                    if c_new < c - 1e-14:
                        c = c_new
                        improved = True
                        break
                    else:
                        h[i] -= d
                        h[j] += d
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random pair moves
            for _ in range(40):
                i, j = int(rng.randint(n)), int(rng.randint(n))
                if i == j:
                    continue
                max_d = min(1.0 - h[i], h[j])
                if max_d < 1e-10:
                    continue
                d = 0.1 * max_d * rng.uniform(0.5, 2.0)
                d = min(d, max_d)
                h[i] += d
                h[j] -= d
                c_new = c5_exact(h)
                if c_new < c - 1e-14:
                    c = c_new
                    improved = True
                    break
                else:
                    h[i] -= d
                    h[j] += d

        if not improved:
            no_improve += 1
            if no_improve > 80:
                stick_count += 1
                scale = 0.01 * min(1.0, stick_count * 0.3)
                h = project_feasible(h + scale * rng.randn(n), target)
                c = c5_exact(h)
                no_improve = 0
        else:
            no_improve = 0
            stick_count = 0

    return h, c

def slsqp_optimize(h, tau, n, maxiter=300):
    target = n / 2.0

    def obj_func(x):
        return lse_obj_grad(x, tau)[0]

    def grad_func(x):
        return lse_obj_grad(x, tau)[1]

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    res = minimize(
        obj_func, h, jac=grad_func, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return project_feasible(res.x, target)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s * 0.95
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h_init = np.full(81, 0.5)

    n0 = len(h_init)
    target0 = n0 / 2.0
    h_init = project_feasible(h_init, target0)
    best_h = h_init.copy()
    best_c5 = c5_exact(best_h)

    def time_ok(min_remaining=5):
        return (deadline - time.time()) > min_remaining

    # Phase 1: Progressive LSE SLSQP optimization
    h = h_init.copy()
    tau_seq = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for tau in tau_seq:
        if not time_ok(40):
            break
        try:
            h_new = slsqp_optimize(h, tau, n0, maxiter=200)
            c_new = c5_exact(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass

    # Phase 2: Intensive local search on best
    if time_ok(60):
        h_ls, c_ls = local_search(best_h.copy(), min(180, deadline - time.time() - 60), rng)
        if c_ls < best_c5:
            best_c5 = c_ls
            best_h = h_ls.copy()

    # Phase 3: Try other resolutions
    for n_try in [120, 96, 160, 80, 200]:
        if not time_ok(50):
            break
        if n_try > 2500:
            continue
        h_try = interpolate_h(best_h, n_try)
        # Quick LSE at new resolution
        try:
            h_try = slsqp_optimize(h_try, 0.01, n_try, maxiter=100)
        except Exception:
            pass
        h_try, c_try = local_search(h_try, min(100, deadline - time.time() - 40), rng)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()

    # Phase 4: Final polish
    if time_ok(30):
        h_fin, c_fin = local_search(best_h.copy(), min(90, deadline - time.time() - 10), rng)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = c5_exact(best_h)

    return best_h, best_c5, len(best_h)
