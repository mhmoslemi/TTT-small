import numpy as np

_IDX_CACHE = {}


def remaining_time(t_start, budget_s, reserve=0.0):
    return max(0.0, float(budget_s) - (time.monotonic() - t_start) - float(reserve))


def candidate_better(score, size, best_score, best_size):
    if not np.isfinite(score):
        return False
    if score < best_score - 1e-12:
        return True
    if abs(score - best_score) <= 1e-12 and size < best_size:
        return True
    return False


def get_idx(n, mode):
    key = (int(n), int(mode))
    idx = _IDX_CACHE.get(key)
    if idx is None:
        r = np.arange(n, dtype=np.int64)[:, None]
        c = np.arange(n, dtype=np.int64)[None, :]
        if mode == 0:
            idx = c - r + (n - 1)
        else:
            idx = r - c + (n - 1)
        idx = idx.astype(np.int64, copy=False)
        _IDX_CACHE[key] = idx
    return idx


def calibrate_mode():
    n = 9
    rng = np.random.default_rng(20240518)
    h = 0.1 + 0.8 * rng.random(n)
    b = 1.0 - h
    z = np.correlate(h, b, "full")
    center = n - 1
    best_mode = 1
    best_err = np.inf
    for mode in (0, 1):
        zm = np.zeros(2 * n - 1, dtype=float)
        if mode == 0:
            for i in range(n):
                hi = h[i]
                for j in range(n):
                    zm[center + j - i] += hi * b[j]
        else:
            for i in range(n):
                hi = h[i]
                for j in range(n):
                    zm[center + i - j] += hi * b[j]
        scale = max(1.0, float(np.max(np.abs(z))))
        err = float(np.max(np.abs(z - zm))) / scale
        if err < best_err:
            best_err = err
            best_mode = mode
    return best_mode


def project_to_box_sum(x, target):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x.copy()
    if not np.all(np.isfinite(x)):
        x = np.full(n, target / max(1, n), dtype=float)
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        lam = 0.5 * (lo + hi)
        s = float(np.clip(x - lam, 0.0, 1.0).sum())
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0).astype(float, copy=False)

    for _ in range(30):
        err = target - float(y.sum())
        if abs(err) <= 1e-12:
            break
        if err > 0.0:
            free = np.where(y < 1.0 - 1e-14)[0]
            if free.size == 0:
                break
            room = 1.0 - y[free]
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            if err >= tot:
                y[free] = 1.0
            else:
                y[free] += err * room / tot
        else:
            free = np.where(y > 1e-14)[0]
            if free.size == 0:
                break
            room = y[free]
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            if -err >= tot:
                y[free] = 0.0
            else:
                y[free] += err * room / tot

    y = np.clip(y, 0.0, 1.0)
    err = target - float(y.sum())
    if abs(err) > 1e-10:
        if err > 0.0:
            free = np.where(y < 1.0)[0]
        else:
            free = np.where(y > 0.0)[0]

                room = 1.0 - y[free]
            else:

            else:
                y[j] = max(0.0, y[j] + max(err, -float(room.max())))
    return y


def corr_full_scaled(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return np.zeros(0, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(2 * h.size - 1, dtype=float)
    return np.correlate(h, 1.0 - h, "full") * (2.0 / float(h.size))


def exact_score(h):
    z = corr_full_scaled(h)
    if z.size == 0:
        return 0.0
    val = float(np.max(z))
    if not np.isfinite(val):
        return np.inf
    return val


def grad_z_rows(h, q_indices, mode):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    center = n - 1
    klen = int(np.size(q_indices))
    out = np.empty((klen, n), dtype=float)
    for k in range(klen):
        s = int(q_indices[k]) - center
        g = np.zeros(n, dtype=float)
        if mode == 1:
            if s >= 0:
                L = n - s
                g[s:] += 1.0 - h[:L]
                g[:L] -= h[s:]
            else:
                a = -s
                L = n - a
                g[:L] += 1.0 - h[a:]
                g[a:] -= h[:L]
        else:
            if s >= 0:
                L = n - s
                g[:L] += 1.0 - h[s:]
                g[s:] -= h[:L]
            else:
                a = -s
                L = n - a
                g[a:] += 1.0 - h[:L]
                g[:L] -= h[a:]
        out[k] = dx * g
    return out


def grad_weighted(h, weights, q_indices, mode):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    center = n - 1
    g = np.zeros(n, dtype=float)
    for k in range(int(np.size(weights))):
        w = float(weights[k])
        if abs(w) <= 1e-14:
            continue
        s = int(q_indices[k]) - center
        if mode == 1:
            if s >= 0:
                L = n - s
                g[s:] += (w * dx) * (1.0 - h[:L])
                g[:L] -= (w * dx) * h[s:]
            else:
                a = -s
                L = n - a
                g[:L] += (w * dx) * (1.0 - h[a:])
                g[a:] -= (w * dx) * h[:L]
        else:
            if s >= 0:
                L = n - s
                g[:L] += (w * dx) * (1.0 - h[s:])
                g[s:] -= (w * dx) * h[:L]
            else:
                a = -s
                L = n - a
                g[a:] += (w * dx) * (1.0 - h[:L])
                g[:L] -= (w * dx) * h[a:]
    return g


def lse_smooth(h, tau, mode, need_grad=False):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0, 0.0, None, None
    dx = 2.0 / float(n)
    z = np.correlate(h, 1.0 - h, "full") * dx
    if not np.all(np.isfinite(z)):
        return np.inf, np.inf, None, None
    zmax = float(np.max(z))
    tau = float(tau)
    if tau <= 1e-8:
        tau = 1e-8
    shifted = (z - zmax) / tau
    shifted = np.clip(shifted, -745.0, 0.0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0.0:
        e = np.zeros_like(z)
        e[int(np.argmax(z))] = 1.0
        s = 1.0
    F = zmax + tau * math.log(s)
    if need_grad:
        p = e / s
        nz = np.where(p > 1e-9)[0]
        if nz.size == 0:
            nz = np.array([int(np.argmax(p))], dtype=int)
        if nz.size < max(1, int(0.35 * n)):
            grad = grad_weighted(h, p[nz], nz, mode)
        else:
            idx = get_idx(n, mode)
            M = p[idx]
            b = 1.0 - h
            grad = dx * (M.dot(b) - M.T.dot(h))
        return F, zmax, p, grad
    return F, zmax, None, None


def pgd_optimize(h0, time_limit, mode, tau0=None, tau_min=1e-5, seed=0):
    ts = time.monotonic()
    deadline = ts + max(1.0, float(time_limit))
    n = int(np.size(h0))
    if n <= 0:
        return np.asarray(h0, dtype=float).ravel(), np.inf

    h = project_to_box_sum(h0, target)

        z = corr_full_scaled(h)
        if z.size == 0:
            ivar = 0.02
        else:
            spread = float(np.max(z) - np.min(z))
            ivar = float(np.clip(0.12 * spread + 0.005, 0.003, 0.05))
    else:
        ivar = float(tau0)

    tau = ivar
    F, exact, p, grad = lse_smooth(h, tau, mode, True)
    best_h = h.copy()
    best_exact = float(exact)

    alpha = 0.05
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        if not np.isfinite(F) or grad is None or not np.all(np.isfinite(grad)):
            F, exact, p, grad = lse_smooth(h, tau, mode, True)
            if not np.isfinite(F) or grad is None:
                break
        gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-13:
            if tau <= float(tau_min) * 1.5:
                break
            tau = max(tau * 0.7, float(tau_min))
            it += 1
            continue

        direction = -grad / max(1.0, gmax)
        alphas = (alpha, alpha * 2.0, alpha * 0.5, alpha * 4.0,
                  alpha * 0.25, alpha * 0.1, alpha * 8.0,
                  0.005, 0.02, 0.1, 0.5, 1.0)
        accepted = False
        for a in alphas:
            if a <= 0.0 or a > 100.0:
                continue
            h_try = project_to_box_sum(h + a * direction, target)
            F_try, exact_try, _, _ = lse_smooth(h_try, tau, mode, False)
            if (np.isfinite(exact_try) and exact_try < best_exact - 1e-11) or \
               (np.isfinite(F_try) and F_try < F - 1e-11):
                h = h_try
                F = float(F_try)
                if np.isfinite(exact_try) and exact_try < best_exact:
                    best_exact = float(exact_try)
                    best_h = h.copy()
                alpha = min(max(float(a), 1e-5), 10.0) * 1.25
                stall = 0
                accepted = True
                break

        if not accepted:
            alpha *= 0.5
            stall += 1
            if stall > 25:
                alpha = 0.05
                stall = 0
                if time.monotonic() < deadline - 5.0:
                    rngp = np.random.default_rng((seed + it) % 1000003)
                    scale = 0.01 if tau < 0.01 else 0.03
                    h = project_to_box_sum(h + scale * rngp.standard_normal(n), target)
                    F, exact, p, grad = lse_smooth(h, tau, mode, True)
                    if np.isfinite(exact) and exact < best_exact:
                        best_exact = float(exact)
                        best_h = h.copy()

        it += 1
        if it % 45 == 0:
            if tau > float(tau_min):
                tau = max(tau * 0.78, float(tau_min))
                if alpha < 0.02:
                    alpha = 0.05
        if time.monotonic() > deadline - 4.0:
            break

    return best_h, best_exact


class SmoothMaxObj:
    def __init__(self, tau, mode, target):
        self.tau = tau
        self.mode = mode
        self.target = target
        self.last_x = None
        self.last_res = None

    def _res(self, x):
        x = np.asarray(x, dtype=float)
        if self.last_x is not None and self.last_x.shape == x.shape and np.array_equal(self.last_x, x):
            return self.last_res
        res = lse_smooth(x, self.tau, self.mode, True)
        self.last_x = x.copy()
        self.last_res = res
        return res

    def fun(self, x):
        return self._res(x)[0]

    def jac(self, x):
        return self._res(x)[3]

    def eq(self, x):
        return float(np.sum(x) - self.target)

    def eq_jac(self, x):
        return np.ones(int(np.size(x)), dtype=float)


def polish_smooth_slsqp(h, mode, tau=1e-4, maxiter=25, ftol=1e-12):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    target = 0.5 * n
    h0 = project_to_box_sum(h, target)
    if n > 600 or maxiter <= 0:
        return h0, exact_score(h0)

    obj = SmoothMaxObj(tau, mode, target)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": obj.eq, "jac": obj.eq_jac}]
    try:
        sol = opt.minimize(obj.fun, h0, jac=obj.jac, method="SLSQP",
                           bounds=bounds, constraints=constraints,
                           options={"maxiter": int(maxiter), "ftol": ftol})
        h2 = project_to_box_sum(sol.x, target)
        return h2, exact_score(h2)
    except Exception:
        return h0, exact_score(h0)


class EpiObj:
    def __init__(self, mode, dim, active, target):
        self.mode = int(mode)
        self.dim = int(dim)
        self.active = np.asarray(active, dtype=int)
        self.target = float(target)

    def fun(self, x):
        return float(x[-1])

    def fun_jac(self, x):
        g = np.zeros(self.dim + 1, dtype=float)
        g[-1] = 1.0
        return g

    def ineq(self, x):
        h = x[:self.dim]
        t = float(x[-1])
        z = corr_full_scaled(h)
        return t - z[self.active]

    def ineq_jac(self, x):
        h = x[:self.dim]
        g = grad_z_rows(h, self.active, self.mode)
        out = np.empty((g.shape[0], self.dim + 1), dtype=float)
        out[:, :self.dim] = -g
        out[:, self.dim] = 1.0
        return out

    def eq(self, x):
        return float(np.sum(x[:self.dim]) - self.target)

    def eq_jac(self, x):
        g = np.zeros(self.dim + 1, dtype=float)
        g[:self.dim] = 1.0
        return g


def polish_epigraph(h, mode, top_k=60, maxiter=25, passes=2, ftol=1e-12):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    target = 0.5 * n
    best_h = project_to_box_sum(h, target)
    best_score = exact_score(best_h)
    if n > 800 or maxiter <= 0 or passes <= 0:
        return best_h, best_score

    top_k = max(1, min(int(top_k), 2 * n - 1))
    for _ in range(int(passes)):
        z = corr_full_scaled(best_h)
        active = np.argsort(z)[-top_k:]
        zactive = z[active]
        t0 = float(np.max(zactive)) + 1e-4
        t0 = min(1.0 - 1e-7, max(1e-6, t0))

        x0 = np.empty(n + 1, dtype=float)
        x0[:n] = best_h
        x0[n] = t0

        obj = EpiObj(mode, n, active, target)
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        constraints = [
            {"type": "ineq", "fun": obj.ineq, "jac": obj.ineq_jac},
            {"type": "eq", "fun": obj.eq, "jac": obj.eq_jac},
        ]
        try:
            sol = opt.minimize(obj.fun, x0, jac=obj.fun_jac, method="SLSQP",
                               bounds=bounds, constraints=constraints,
                               options={"maxiter": int(maxiter), "ftol": ftol})
            h2 = project_to_box_sum(sol.x[:n], target)
            s2 = exact_score(h2)
            if candidate_better(s2, h2.size, best_score, best_h.size):
                best_h = h2.copy()
                best_score = s2
        except Exception:
            pass
    return best_h, best_score


def interp_to(h, target_n):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    target_n = int(max(1, target_n))
    if n == 0:
        return np.zeros(0, dtype=float)
    if target_n == n:
        return project_to_box_sum(h, 0.5 * n)
    if n == 1:
        return np.full(target_n, float(h[0]), dtype=float)
    old = np.linspace(0.0, 1.0, n, endpoint=False)
    new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    y = np.interp(new, old, h, left=h[0], right=h[-1])
    return project_to_box_sum(y, 0.5 * target_n)


def resize_to(h, target_n):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    target_n = int(max(1, target_n))
    if n == 0:
        return np.zeros(0, dtype=float)
    if target_n == n:
        return project_to_box_sum(h, 0.5 * n)
    if target_n > n and target_n % n == 0:
        y = np.repeat(h, target_n // n)
        return project_to_box_sum(y, 0.5 * target_n)
    return interp_to(h, target_n)


def downsample_by2(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n < 100 or n % 2 != 0:
        return project_to_box_sum(h, 0.5 * n)
    m = n // 2
    y = np.mean(h.reshape(2, m), axis=0)
    return project_to_box_sum(y, 0.5 * m)


def gaussian_perturb(h, scale, seed):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    rng = np.random.default_rng(int(seed) % 1000003)
    return project_to_box_sum(h + float(scale) * rng.standard_normal(n), 0.5 * n)


def sine_candidate(n, seed, base=None, scale=0.04):
    n = int(max(1, n))
    rng = np.random.default_rng(int(seed) % 1000003)
    if base is not None:
        h = np.asarray(base, dtype=float).ravel().copy()
        scale = float(scale) * 0.35
    else:
        h = np.full(n, 0.5, dtype=float)
    t = np.arange(n, dtype=float)
    for k in range(1, max(2, int(np.clip(n // 40, 1, 10)) + 1)):
        amp = rng.uniform(0.0, max(1e-6, float(scale)))
        if amp > 0.0:
            phase = rng.uniform(0.0, 2.0 * np.pi)
            h += amp * np.sin((2.0 * np.pi * k * t) / float(n) + phase)
    return project_to_box_sum(h, 0.5 * n)


def binary_candidate(n, seed, scale=0.0):
    n = int(max(1, n))
    rng = np.random.default_rng(int(seed) % 1000003)
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n, dtype=float)
    h[idx[: n // 2]] = 1.0
    if scale > 0.0:
        h += float(scale) * rng.standard_normal(n)
    return project_to_box_sum(h, 0.5 * n)

    # Try simple symmetry/carrier variants.
    rem = remaining_time(t_start, budget_s, 60.0)
    if best_n <= 1000 and rem > 120.0:
        cands = (best_h, 1.0 - best_h, best_h[::-1], (1.0 - best_h)[::-1])
        for c in cands:
            hcv = project_to_box_sum(c, 0.5 * best_n)
            scv = exact_score(hcv)
            if candidate_better(scv, hcv.size, best_score, best_n):
                best_h, best_score, best_n = hcv, scv, int(hcv.size)
            rem = remaining_time(t_start, budget_s, 90.0)
            if rem < 80.0:
                break

    # Upsample to 500 points.
    rem = remaining_time(t_start, budget_s, 120.0)
    if best_n < 500 and rem > 260.0:
        h2a = resize_to(best_h, 500)
        h2b = interp_to(best_h, 500)
        s2a = exact_score(h2a)
        s2b = exact_score(h2b)
        if s2a <= s2b:
            h2_main, h2_other, init2 = h2a, h2b, s2a
        else:
            h2_main, h2_other, init2 = h2b, h2a, s2b

        if init2 < best_score + 0.05:
            t_alloc = min(rem, max(90.0, 0.30 * budget_s))
            h2o, s2 = pgd_optimize(h2_main, t_alloc, mode, seed=seed + 101)
            if candidate_better(s2, h2o.size, best_score, best_n):
                best_h, best_score, best_n = h2o, s2, int(h2o.size)

            rem = remaining_time(t_start, budget_s, 150.0)
            if rem > 120.0:
                h2o2, s22 = pgd_optimize(h2_other, min(rem - 30.0, 100.0), mode, seed=seed + 102)
                if candidate_better(s22, h2o2.size, best_score, best_n):
                    best_h, best_score, best_n = h2o2, s22, int(h2o2.size)

    # Smooth-max SLSQP polish.
    rem = remaining_time(t_start, budget_s, 180.0)
    if best_n <= 600 and rem > 260.0:
        maxiter = 30 if best_n <= 300 else (20 if best_n <= 500 else 12)
        hp, sp = polish_smooth_slsqp(best_h, mode, tau=1e-4, maxiter=maxiter)
        if candidate_better(sp, hp.size, best_score, best_n):
            best_h, best_score, best_n = hp, sp, int(hp.size)

    # Active epigraph polish.
    rem = remaining_time(t_start, budget_s, 220.0)
    if best_n <= 800 and rem > 320.0:
        top_k = min(90, max(30, 2 * best_n - 1))
        maxiter = 22 if best_n <= 300 else 14
        hp, sp = polish_epigraph(best_h, mode, top_k=top_k, maxiter=maxiter, passes=2)
        if candidate_better(sp, hp.size, best_score, best_n):
            best_h, best_score, best_n = hp, sp, int(hp.size)

    # Try a 1000-point refinement only if still wanted and time allows.
    rem = remaining_time(t_start, budget_s, 150.0)
    if best_n < 1000 and rem > 420.0 and best_score > 0.38080:
        h3a = resize_to(best_h, 1000)
        h3b = interp_to(best_h, 1000)
        s3a = exact_score(h3a)
        s3b = exact_score(h3b)
        if s3a <= s3b:
            h3_main, h3_other, init3 = h3a, h3b, s3a
        else:
            h3_main, h3_other, init3 = h3b, h3a, s3b

        if init3 < best_score + 0.04:
            t_alloc = min(rem - 60.0, max(90.0, 0.24 * budget_s))
            h3o, s3 = pgd_optimize(h3_main, t_alloc, mode, seed=seed + 202, tau_min=8e-6)
            if candidate_better(s3, h3o.size, best_score, best_n):
                best_h, best_score, best_n = h3o, s3, int(h3o.size)

            rem = remaining_time(t_start, budget_s, 180.0)
            if rem > 120.0:
                h3o2, s32 = pgd_optimize(h3_other, min(rem - 40.0, 90.0),
                                         mode, seed=seed + 203, tau_min=8e-6)
                if candidate_better(s32, h3o2.size, best_score, best_n):
                    best_h, best_score, best_n = h3o2, s32, int(h3o2.size)

    # Small perturbations around the best candidate.
    tries = 0
    for tries in range(4):
        rem = remaining_time(t_start, budget_s, 160.0)
        if rem < 110.0:
            break
        scale = 0.012 + 0.008 * tries
        hstart = gaussian_perturb(best_h, scale, seed + 1000 + 17 * tries)
        hstart2 = sine_candidate(best_h.size, seed + 3000 + 29 * tries, base=best_h, scale=0.03)
        t_alloc = min(rem - 60.0, 70.0)
        ho, so = pgd_optimize(hstart, max(1.0, t_alloc * 0.55), mode, seed=seed + 4000 + tries)
        if candidate_better(so, ho.size, best_score, best_n):
            best_h, best_score, best_n = ho, so, int(ho.size)
        rem = remaining_time(t_start, budget_s, 160.0)
        if rem > 70.0:
            ho2, so2 = pgd_optimize(hstart2, max(1.0, min(rem - 60.0, 60.0)), mode, seed=seed + 5000 + tries)
            if candidate_better(so2, ho2.size, best_score, best_n):
                best_h, best_score, best_n = ho2, so2, int(ho2.size)

    # One final epigraph pass if enough time remains and dimension is moderate.
    rem = remaining_time(t_start, budget_s, 220.0)
    if best_n <= 800 and rem > 300.0:
        top_k = min(80, max(28, 2 * best_n - 1))
        maxiter = 18 if best_n <= 300 else 10
        hp, sp = polish_epigraph(best_h, mode, top_k=top_k, maxiter=maxiter, passes=1)
        if candidate_better(sp, hp.size, best_score, best_n):
            best_h, best_score, best_n = hp, sp, int(hp.size)

    # Downsample only if it does not hurt the score.
    if best_h.size >= 500:
        hd = downsample_by2(best_h)
        sd = exact_score(hd)
        if candidate_better(sd, hd.size, best_score, best_n):
            best_h, best_score, best_n = hd, sd, int(hd.size)

    # Final repair and validation.
    if best_h.size == 0 or not np.all(np.isfinite(best_h)):
        best_h = np.full(250, 0.5, dtype=float)
    best_h = project_to_box_sum(best_h, 0.5 * best_h.size)
    best_score = exact_score(best_h)
    if not np.isfinite(best_score):
        best_h = project_to_box_sum(parent, 0.5 * parent.size)
        best_score = exact_score(best_h)

    return best_h, float(best_score), int(best_h.size)
