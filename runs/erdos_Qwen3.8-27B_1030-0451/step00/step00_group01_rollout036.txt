def local_optimize(h, n, max_iters=1000, lr_decay=0.999):
    """Local optimization using subgradient method with projection."""
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/n)
    
    lr = 0.01
    n_active = min(5, n)
    
    for it in range(max_iters):
        corr = compute_correlation(h)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        current_c5 = max_val * 2.0 / n
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find active set (top n_active)
        top_idx = np.argsort(corr)[-n_active:]
        
        # Compute subgradient
        subgrad = np.zeros(n)
        for k in top_idx:
            m = k - (n - 1)
            if m == 0:
                g = 1 - 2*h
            elif m > 0:
                # A_m gradient
                g = np.zeros(n)
                g[m:] += (1 - h[:n-m])  # (1-h[j-m]) for j >= m
                g[:n-m] -= h[m:]          # -h[j+m] for j < n-m
            else:
                # B_{|m|} gradient, m < 0
                am = -m
                g = np.zeros(n)
                g[:n-am] += (1 - h[am:])  # (1-h[j+am]) for j < n-am
                g[am:] -= h[:n-am]         # -h[j-am] for j >= am
            subgrad += g
        
        subgrad /= len(top_idx)
        
        # Step
        h_new = h - lr * subgrad
        h_new = project_feasible(h_new, n)
        
        # Check if improvement
        new_c5 = compute_c5(h_new, 2.0/n)
        if new_c5 < current_c5:
            h = h_new
        else:
            lr *= lr_decay
        
        if lr < 1e-8:
            break
    
    return best_h, best_c5

        corr = np.correlate(h, 1-h, mode='full')
        max_val = corr.max()
        current_c5 = max_val * dx
        
        if current_c5 < best_c5 - 1e-10:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 50:
                lr *= 0.9
                if lr < 1e-8:
                    break
        
        # Active set: all lags within 0.01 of max
        active = np.where(corr >= max_val - 0.01)[0]
        
        # Subgradient: average of active gradients
        subgrad = np.zeros(N)
        for k in active:
            _, g = _corr_val_grad(k, h, N)
            subgrad += g
        subgrad /= len(active)
        
        # Step
        h_new = _project(h - lr * subgrad, N)
        h = h_new
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def _corr_val_grad(k, h, N):
    m = k - (N - 1)
    if m == 0:
        val = np.dot(h, 1.0 - h)
        grad = 1.0 - 2.0 * h
    elif m > 0:
        if m >= N:
            return 0.0, np.zeros(N)
        val = np.dot(h[m:], 1.0 - h[:N - m])
        grad = np.zeros(N)
        grad[m:] += (1.0 - h[:N - m])
        grad[:N - m] -= h[m:]
    else:
        am = -m
        if am >= N:
            return 0.0, np.zeros(N)
        val = np.dot(h[:N - am], 1.0 - h[am:])
        grad = np.zeros(N)
        grad[:N - am] += (1.0 - h[am:])
        grad[am:] -= h[:N - am]
    return val, grad

def _project(h, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _obj_fn(x, N, beta, mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = corr.max()
    es = np.exp(beta * (corr - mc))
    se = es.sum()
    obj = (mc + np.log(se)) / beta
    s = x.sum() - N / 2.0
    obj += 0.5 * mu * s * s
    return obj

def _jac_fn(x, N, beta, mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = corr.max()
    es = np.exp(beta * (corr - mc))
    se = es.sum()
    w = es / se
    grad = np.zeros(N)
    for k in range(len(corr)):
        _, g = _corr_val_grad(k, x, N)
        grad += w[k] * g
    s = x.sum() - N / 2.0
    grad += mu * s
    return grad

def _lbfgs_optimize(h, N, deadline, n_rounds=5):
    h = _project(h.astype(float).copy(), N)
    dx = 2.0 / N
    best_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * N

    for r in range(n_rounds):
        if time.time() > deadline:
            break
        beta = 5.0 * (3.0 ** r)
        mu = 20.0 * (4.0 ** r)
        res = minimize(_obj_fn, h, jac=_jac_fn, args=(N, beta, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10})
        h = _project(res.x, N)
        c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def _subgrad_polish(h, N, deadline):
    h = _project(h.astype(float).copy(), N)
    dx = 2.0 / N
    best_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    best_h = h.copy()
    lr = 0.005
    stall = 0

    for it in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        c5 = max_val * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.92
                if lr < 1e-9:
                    break
                stall = 0

        active = np.where(corr >= max_val - 0.05)[0]
        subgrad = np.zeros(N)
        for k in active:
            _, g = _corr_val_grad(k, h, N)
            subgrad += g
        subgrad /= len(active)
        h = _project(h - lr * subgrad, N)

    return best_h, best_c5

def _upsample(h, N):
    old_N = len(h)
    if old_N == N:
        return h.copy()
    if old_N == 1:
        return np.full(N, h[0])
    old_x = np.arange(old_N, dtype=float)
    new_x = np.linspace(0, old_N - 1, N)
    return np.interp(new_x, old_x, h)

def _random_binary(N, rng):
    h = np.zeros(N)
    n_ones = N // 2
    idx = rng.permutation(N)
    h[idx[:n_ones]] = 1.0
    if N % 2 == 1:
        h[idx[n_ones]] = 1.0
        h = _project(h, N)
    return h

def _block_start(N, rng):
    for bs in [7, 11, 17, 5]:
        if bs < N:
            h = np.tile([1.0, 0.0], (N // 2) + 1)[:N]
            h = np.reshape(h, (-1, 2))
            new_h = np.zeros(N)
            for i in range(h.shape[0]):
                start = i * bs
                end = min(start + bs, N)
                if i % 2 == 0:
                    new_h[start:end] = 1.0
                else:
                    new_h[start:end] = 0.0
            return _project(new_h, N)
    return _random_binary(N, rng)

def _sine_start(N, rng):
    freq = rng.integers(2, 8)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, N)
    h = (np.sin(2 * np.pi * freq * x + phase) > 0).astype(float)
    return _project(h, N)

    t0 = time.time()
    deadline = t0 + budget_s * 0.93

    def _update(h, c5, n):
