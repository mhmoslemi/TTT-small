def project_mass(y, target):
    y = np.clip(np.asarray(y, dtype=np.float64), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + 0.5 * (lo + hi), 0, 1)
    res = math.fsum(z) - target
    if abs(res) > 1e-12:
        idx = np.arange(len(z))
        if res > 0:
            for i in idx[np.argsort(-z)]:
                d = min(z[i], res)
                z[i] -= d; res -= d
                if abs(res) < 1e-14: break
        else:
            res = -res
            for i in idx[np.argsort(1 - z)]:
                d = min(1 - z[i], res)
                z[i] += d; res -= d

def resample_step(h_old, n_new):
    """Resample a step function from n_old to n_new points."""
    n_old = len(h_old)
    # Build cumulative sum (area under step function)
    # The step function on [0,2] with n_old steps of width dx_old
    # Cell i covers [2*i/n_old, 2*(i+1)/n_old] with height h_old[i]
    # For n_new cells, cell j covers [2*j/n_new, 2*(j+1)/n_new]
    # Area in cell j = sum of h_old[i] * overlap_length(cell_i, cell_j)
    
    # Use interpolation approach
    # Old cell centers at x_old = 2*(arange(n_old)+0.5)/n_old
    # New cell centers at x_new = 2*(arange(n_new)+0.5)/n_new
    # But for step functions, we should use the actual overlap
    
    # Simpler: use np.interp on the step function
    # Define step function breakpoints
    x_old_edges = np.linspace(0, 2, n_old + 1)
    x_new_edges = np.linspace(0, 2, n_new + 1)
    
    new_h = np.zeros(n_new)
    for j in range(n_new):
        # Cell j spans [x_new_edges[j], x_new_edges[j+1]]
        lo, hi = x_new_edges[j], x_new_edges[j+1]
        total_area = 0.0
        for i in range(n_old):
            cell_lo = x_old_edges[i]
            cell_hi = x_old_edges[i + 1]
            overlap = min(hi, cell_hi) - max(lo, cell_lo)
            if overlap > 0:
                total_area += h_old[i] * overlap
        new_h[j] = total_area / (hi - lo)
    
    return new_h
