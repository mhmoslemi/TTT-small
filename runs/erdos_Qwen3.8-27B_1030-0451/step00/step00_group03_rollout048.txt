        else:
            order = np.argsort(-h)
            for i in order:
                step = min(h[i], -r)
                if step <= 0:
                    break
                h[i] -= step
                r += step
                if abs(r) < 1e-14:
                    break
    return h

def interp_to(h, nc):
    n = len(h)
    tp = (np.arange(n) + 0.5) * (2.0 / n)
    tc = (np.arange(nc) + 0.5) * (2.0 / nc)
    hc = np.interp(tc, tp, h)
    return project(hc, 0.5 * nc)

def lag_grad(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        g[k:k+L] += 1.0 - h[:L]
        g[:L] -= h[k:k+L]
    else:
        r = -k
        L = n - r
        g[:L] += 1.0 - h[r:]
        g[r:] -= h[:L]
    return g

def lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    f = compute_f(h)
    fm = float(np.max(f))
    z = (f - fm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = e / float(np.sum(e))
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if s[m] > 1e-7:
            g += s[m] * lag_grad(h, m - (n - 1))
    return g * dx, fm

def proj_desc(h0, target, tlim, tau0=0.004):
    t0 = time.time()
    h = h0.copy()
    bh = h.copy()
    bc = evaluate_c5(h)
    tau = tau0
    stall = 0
    while time.time() - t0 < tlim:
        f = compute_f(h)
        cm = float(np.max(f))
        if cm < bc - 1e-13:
            bc = cm
            bh = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 60 == 0:
                tau = min(tau * 1.2, 0.012)
                if tau > 0.011:
                    break
        g, _ = lse_grad(h, tau)
        gn = float(np.linalg.norm(g))
        if gn < 1e-11:
            break
        ok = False
        for i in range(20):
            eta = 0.03 / gn * (0.5 ** i)
            hn = project(h - eta * g, target)
            if np.array_equal(hn, h):
                continue
            cn = evaluate_c5(hn)
            if cn < cm - 1e-13:
                h = hn
                ok = True
                break
        if not ok and stall > 40:
            rng = np.random.default_rng((stall * 7919 + 31) % (2**31))
            pert = rng.normal(0.0, 0.01, len(h))
            hn = project(h + pert, target)
            if not np.array_equal(hn, h):
                cn = evaluate_c5(hn)
                h = hn
                if cn < bc - 1e-13:
                    bc = cn
                    bh = hn.copy()
                    stall = 0
    return bh, bc

def _slsqp_obj(x):
    return x[-1]

def _slsqp_obj_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def _slsqp_eq(x):
    return float(np.sum(x[:_G['n']]) - _G['target'])

def _slsqp_eq_grad(x):
    g = np.zeros(_G['n'] + 1)
    g[:_G['n']] = 1.0
    return g

def _slsqp_ineq(x):
    n = _G['n']
    dx = 2.0 / n
    h = x[:n]
    t = x[-1]
    lags = _G['lags']
    vals = np.zeros(len(lags))
    for i in range(len(lags)):
        k = lags[i] - (n - 1)
        if k >= 0:
            L = n - k
            vals[i] = t - float(np.sum(h[k:k+L] * (1.0 - h[:L]))) * dx
        else:
            r = -k
            L = n - r
            vals[i] = t - float(np.sum(h[:L] * (1.0 - h[r:]))) * dx
    return vals

def _slsqp_ineq_jac(x):
    n = _G['n']
    dx = 2.0 / n
    h = x[:n]
    lags = _G['lags']
    na = len(lags)
    J = np.zeros((na, n + 1))
    for i in range(na):
        k = lags[i] - (n - 1)
        g = lag_grad(h, k) * dx
        J[i, :n] = -g
        J[i, -1] = 1.0
    return J

def slsqp_polish(h0, tlim):
    n = len(h0)
    target = 0.5 * n
    f = compute_f(h0)
    c0 = float(np.max(f))
    n_act = min(35, 2 * n - 1)
    order = np.argsort(-f)[:n_act]
    _G['n'] = n
    _G['target'] = target
    _G['lags'] = order
    x0 = np.concatenate([h0, [c0 + 0.003]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_grad},
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac}
    ]
    best_h = h0.copy()
    best_c = c0
    try:
        res = minimize(_slsqp_obj, x0, jac=_slsqp_obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-14})
        if res.x is not None:
            h_sol = project(res.x[:n], target)
            c_sol = evaluate_c5(h_sol)
            if c_sol < best_c:
                best_h = h_sol
                best_c = c_sol
    except Exception:
        pass
    if time.time() < t0_slsqp_end():
        return best_h, best_c
    return best_h, best_c

def t0_slsqp_end():
    return 0.0

def _make_slsqp_polish(h0, tlim):
    n = len(h0)
    target = 0.5 * n
    f = compute_f(h0)
    c0 = float(np.max(f))
    n_act = min(35, 2 * n - 1)
    order = np.argsort(-f)[:n_act]
    _G['n'] = n
    _G['target'] = target
    _G['lags'] = order
    x0 = np.concatenate([h0, [c0 + 0.003]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_grad},
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac}
    ]
    best_h = h0.copy()
    best_c = c0
    t_start = time.time()
    max_iter = 200
    while time.time() - t_start < tlim:
        try:
            res = minimize(_slsqp_obj, x0, jac=_slsqp_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.x is not None:
                h_sol = project(res.x[:n], target)
                c_sol = evaluate_c5(h_sol)
                if c_sol < best_c - 1e-14:
                    best_c = c_sol
                    best_h = h_sol.copy()
                    x0 = np.concatenate([h_sol, [c_sol + 0.001]])
                    f_new = compute_f(h_sol)
                    order2 = np.argsort(-f_new)[:n_act]
                    _G['lags'] = order2
                    max_iter = max(100, max_iter // 2)
                else:
                    break
            else:
                break
        except Exception:
            break
    return best_h, best_c

def binary_local(h0, target, tlim):
    n = len(h0)
    h = h0.copy()
    bh = h.copy()
    bc = evaluate_c5(h)
    t0 = time.time()
    while time.time() - t0 < tlim:
        improved = False
        for p in range(n):
            best_val = h[p]
            best_c = evaluate_c5(bh)
            for trial in [0.0, 1.0, 0.25, 0.5, 0.75]:
                if abs(trial - h[p]) < 1e-6:
                    continue
                h[p] = trial
                hp = project(h, target)
                c = evaluate_c5(hp)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = hp[p]
                    improved = True
                    bh = hp.copy()
            h = bh.copy()
            h[p] = best_val
            h = project(h, target)
    return bh, bc

def random_restart(h_ref, n, target, n_trials, tlim):
    bh = h_ref.copy()
    bc = evaluate_c5(bh)
    t0 = time.time()
    rng = np.random.default_rng(12345)
    for _ in range(n_trials):
        if time.time() - t0 > tlim:
            break
        if rng.random() < 0.5:
            noise = rng.normal(0, 0.1, n)
            y = h_ref + noise
        else:
            y = rng.uniform(0, 1, n)
        h = project(y, target)
        c = evaluate_c5(h)
        if c < bc - 1e-14:
            bc = c
            bh = h.copy()
    return bh, bc

def coefficient_search(h0, target, tlim):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    bh = h.copy()
    bc = evaluate_c5(h)
    t0 = time.time()
    for _ in range(20):
        if time.time() - t0 > tlim:
            break
        improved = False
        for p in range(n):
            f = compute_f(h)
            cm = float(np.max(f))
            if cm >= bc:
                cm = bc
            best_t = h[p]
            best_c = cm
            for trial in np.linspace(0.0, 1.0, 17):
                if abs(trial - h[p]) < 1e-6:
                    continue
                h[p] = trial
                hp = project(h, target)
                c = evaluate_c5(hp)
                if c < best_c - 1e-14:
                    best_c = c
                    best_t = hp[p]
            h = project(h, target)
            h[p] = best_t
            hp = project(h, target)
            c = evaluate_c5(hp)
            if c < bc - 1e-14:
                bc = c
                bh = hp.copy()
                h = hp.copy()
                improved = True
        if not improved:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    hard_end = t_start + budget_s * 0.92

    ns = 'initial_h_values'
    parent = None
    if ns in globals() and globals()[ns] is not None:
        try:
            parent = np.array(globals()[ns], dtype=np.float64)
        except Exception:
            parent = None
    if parent is None or len(parent) == 0:
        parent = np.full(64, 0.5, dtype=np.float64)

    n0 = len(parent)
    target0 = 0.5 * n0
    best_h = project(parent.copy(), target0)
    best_c = evaluate_c5(best_h)

    # Try reversal symmetry
    h_rev = project(parent[::-1].copy(), target0)
    c_rev = evaluate_c5(h_rev)
    if c_rev < best_c:
        best_h = h_rev
        best_c = c_rev

    # Phase 1: Gradient descent at parent resolution
    rem1 = min(100.0, max(10.0, (hard_end - time.time()) * 0.25))
    if time.time() < hard_end:
        h1, c1 = proj_desc(best_h, target0, rem1, tau0=0.004)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Try higher resolutions
    resolutions = []
    n0 = len(best_h)
    for nc in [96, 128, 160, 192]:
        if nc <= n0:
            continue
        if n0 > 1 and nc % n0 == 0:
            resolutions.append(nc)
        elif nc >= 2 * n0:
            resolutions.append(nc)
    if not resolutions:
        resolutions = [128, 192]

    for nc in resolutions:
        if time.time() > hard_end - 60:
            break
        rem2 = max(20.0, (hard_end - time.time()) * 0.35)
        hc = interp_to(best_h, nc)
        tc = 0.5 * nc
        hc, cc = proj_desc(hc, tc, rem2, tau0=0.003)
        if cc < best_c - 1e-13:
            best_c = cc
            best_h = hc

    # Phase 3: SLSQP polishing
    if time.time() < hard_end - 30:
        rem_slsqp = min(180.0, hard_end - time.time() - 30)
        best_h, best_c = _make_slsqp_polish(best_h, rem_slsqp)

    # Phase 4: Coefficient search on best
    if time.time() < hard_end - 15:
        rem4 = min(80.0, hard_end - time.time() - 15)
        n_best = len(best_h)
        best_h, best_c = coefficient_search(best_h, 0.5 * n_best, rem4)

    # Phase 5: Final gradient descent
    if time.time() < hard_end - 5:
        rem5 = max(5.0, hard_end - time.time() - 5)
        n_best = len(best_h)
        best_h, best_c = proj_desc(best_h, 0.5 * n_best, rem5, tau0=0.002)

    # Final validation and projection
    n_final = len(best_h)
    t_final = 0.5 * n_final
    best_h = project(best_h, t_final)
    best_c = evaluate_c5(best_h)

    return (best_h, best_c, n_final)
