def exact_eval(h):
    """Compute C5 = max of np.correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(z) * dx)

def smooth_obj_grad(h, tau):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    # Compute all correlation values
    # np.correlate(h, b, 'full') gives 2n-1 values
    z = np.correlate(h, b, mode='full') * dx
    # Smooth max
    zmax = np.max(z)
    # Stable log-sum-exp
    shifted = (z - zmax) / tau
    mask = shifted > -50
    p = np.zeros_like(z)
    p[mask] = np.exp(shifted[mask])
    p /= np.sum(p)
    F = zmax + tau * np.log(np.sum(p))
    # Gradient: sum_q p_q * grad_z_q
    # grad_z_q wrt h: for lag ell (0-indexed in correlate output, centered at n-1)
    # z_q = dx * sum_{valid} h_{i+ell}(1-h_i)
    # dz_q/dh_j = dx * (b_{j-ell} if valid) - dx * (h_{j+ell} if valid)
    # In correlate(h, b, 'full'), index q corresponds to lag ell = q - (n-1)
    grad = np.zeros(n)
    for q in range(1, 2*n-1):  # skip q=0 which has p~0 usually, but include all
        if p[q] < 1e-15:
            continue
        ell = q - (n - 1)
        if ell == 0:
            grad += p[q] * dx * (b - h)  # dx*(1-2h)
        elif ell > 0:
            L = n - ell
            # z_ell = dx * dot(h[ell:], b[:L])
            # d/dh_i: for i >= ell: dx*b[i-ell]; for i < L: -dx*h[i+ell]
            grad[ell:] += p[q] * dx * b[:L]
            grad[:L] -= p[q] * dx * h[ell:]
        else:
            a = -ell
            L = n - a
            # z_{-a} = dx * dot(h[:L], b[a:])
            # d/dh_i: for i < L: dx*b[i+a]; for i >= a: -dx*h[i-a]
            grad[:L] += p[q] * dx * b[a:]
            grad[a:] -= p[q] * dx * h[:L]
    return F, grad

def project_box_sum(h, T):
    """Project h onto {x: 0<=x_i<=1, sum(x)=T}"""
    n = len(h)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Final repair
    diff = T - np.sum(y)
    if abs(diff) > 1e-12:
        # Distribute to elements not at bounds
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if np.any(mask):
            y[mask] += diff / np.sum(mask)
            y = np.clip(y, 0.0, 1.0)
    return y

    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), T)
    best_h = h.copy()
    best_score = exact_eval(h)
    
    alpha = 0.02
    tau = tau0

    best_h = h.copy()
    best_score = exact_eval(h)
    tau = 0.012
    alpha = 0.008
    n_active = 0
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        F, grad = smooth_obj_grad(h, tau)
        gnorm = np.linalg.norm(grad, ord=np.inf)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        a = alpha
        accepted = False
        for _ in range(12):
            h_new = project_box_sum(h + a * direction, T)
            F_new, _ = smooth_obj_grad(h_new, tau)
            if F_new <= F + 1e-13:
                h = h_new
                accepted = True
                break
            a *= 0.5
        if accepted:
            alpha = min(a * 1.05, 0.04)
        else:
            alpha = max(a * 0.4, 1e-7)
        if (it + 1) % 8 == 0:
            s = exact_eval(h)
            if s < best_score - 1e-13:
                best_score = s
                best_h = h.copy()
                n_active += 1
        if (it + 1) % 40 == 0:
            tau = max(tau * 0.82, 1.5e-4)
            alpha = 0.006
    s = exact_eval(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()
    return best_h, best_score

def epigraph_polish(h0, T, deadline, maxiters=200, n_active_cons=30):
    n = len(h0)
    dx = 2.0 / n
    b = 1.0 - h0
    z0 = np.correlate(h0, b, mode='full') * dx
    t0 = float(np.max(z0)) + 0.005
    # Select most active lags
    order = np.argsort(z0)[::-1]
    active = order[:min(n_active_cons, len(order))]
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    h_start = h0.astype(float).copy()
    x0 = np.concatenate([h_start, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.65)]
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x[:n]) - T,
             'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}]
    for q in active:
        ell = q - (n - 1)
        def con_fun(x, q=q, ell=ell, n=n, dx=dx):
            hh = x[:n]
            bb = 1.0 - hh
            if ell == 0:
                return -(dx * np.dot(hh, bb) - x[-1])
            elif ell > 0:
                L = n - ell
                return -(dx * np.dot(hh[ell:], bb[:L]) - x[-1])
            else:
                a2 = -ell
                L = n - a2
                return -(dx * np.dot(hh[:L], bb[a2:]) - x[-1])
        def con_jac(x, q=q, ell=ell, n=n, dx=dx):
            hh = x[:n]
            bb = 1.0 - hh
            gj = np.zeros(n + 1)
            gj[-1] = 1.0
            if ell == 0:
                gj[:n] = -dx * (bb - hh)
            elif ell > 0:
                L = n - ell
                gj[ell:n] += -dx * bb[:L]
                gj[0:L] -= -dx * hh[ell:]
            else:
                a2 = -ell
                L = n - a2
                gj[0:L] += -dx * bb[a2:]
                gj[a2:] -= -dx * hh[0:L]
            return gj
        cons.append({'type': 'ineq', 'fun': con_fun, 'jac': con_jac})
    try:
        result = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiters, 'ftol': 1e-11})
        h_new = project_box_sum(result.x[:n].copy(), T)
        return h_new
    except Exception:
        return h0

def make_variants(h, T):
    n = len(h)
    v = [h.copy(), h[::-1].copy(), (1.0 - h).copy(), (1.0 - h[::-1]).copy()]
    return [project_box_sum(x, T) for x in v]

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15.0
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_score = 1.0
    
    # Get parent
    h_init = None
    try:
        if 'initial_h_values' in globals():
            h_arr = np.asarray(globals()['initial_h_values'], dtype=float)
            if h_arr.ndim == 1 and len(h_arr) >= 10:
                h_init = h_arr
    except Exception:
        pass
    
    if h_init is None:
        # Construct a simple starting point
        n = 250
        T = n / 2.0
        h_init = np.full(n, 0.5)
        h_init[np.arange(0, n, 2)] = 1.0
        h_init[np.arange(1, n, 2)] = 0.0
        h_init = project_box_sum(h_init, T)
        best_score = exact_eval(h_init)
        best_h = h_init.copy()
    else:
        n = len(h_init)
        T = n / 2.0
        h_init = project_box_sum(h_init, T)
        best_score = exact_eval(h_init)
        best_h = h_init.copy()
    
    # Phase 1: Main PGD on parent (most important)
    phase_deadline = min(deadline, t_start + 350)
    if time.time() < phase_deadline:
        h_res, s_res = pgd_optimize(best_h, T, phase_deadline, n_iter_max=12000)
        if s_res < best_score:
            best_score = s_res
            best_h = h_res.copy()
    
    # Phase 2: Epigraph polish
    if time.time() < deadline - 120 and len(best_h) <= 600:
        h_res = epigraph_polish(best_h, T, min(deadline, time.time() + 80), maxiters=200)
        s_res = exact_eval(h_res)
        if s_res < best_score:
            best_score = s_res
            best_h = h_res.copy()
    
    # Phase 3: PGD again after epigraph
    if time.time() < deadline - 80:
        h_res, s_res = pgd_optimize(best_h, T, min(deadline - 50, time.time() + 120), n_iter_max=8000)
        if s_res < best_score:
            best_score = s_res
            best_h = h_res.copy()
    
    # Phase 4: Try symmetries of original parent
    if time.time() < deadline - 100:
        variants = make_variants(h_init, T)
        for vi, hv in enumerate(variants):
            if time.time() > deadline - 40:
                break
            s_try = exact_eval(hv)
            if s_try < best_score:
                best_score = s_try
                best_h = hv.copy()
            if time.time() < deadline - 20:
                h_res, s_res = pgd_optimize(hv, T, min(deadline - 10, time.time() + 60), n_iter_max=4000)
                if s_res < best_score:
                    best_score = s_res
                    best_h = h_res.copy()
    
    # Phase 5: Random perturbations from best
    n_best = len(best_h)
    T_best = n_best / 2.0
    if time.time() < deadline - 60:
        for trial in range(8):
            if time.time() > deadline - 20:
                break
            noise = rng.normal(0, 0.03, size=n_best)
            h_pert = project_box_sum(best_h + noise, T_best)
            if time.time() < deadline - 15:
                h_res, s_res = pgd_optimize(h_pert, T_best, min(deadline - 5, time.time() + 40), n_iter_max=3000)
                if s_res < best_score:
                    best_score = s_res
                    best_h = h_res.copy()
    
    # Phase 6: Final epigraph + PGD
    if time.time() < deadline - 15 and len(best_h) <= 600:
        h_res = epigraph_polish(best_h, T_best, min(deadline, time.time() + 10), maxiters=100)
        s_res = exact_eval(h_res)
        if s_res < best_score:
            best_score = s_res
            best_h = h_res.copy()
    
    # Final validation and repair
    n_final = len(best_h)
    T_final = n_final / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = T_final - np.sum(best_h)
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-9) & (best_h < 1.0 - 1e-9)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
    best_score = exact_eval(best_h)
    return (best_h, best_score, n_final)
