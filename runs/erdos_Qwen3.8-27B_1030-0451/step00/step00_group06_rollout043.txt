def project_feasible(y, target_sum):
    """Project y onto {x: sum(x) = target_sum, 0 <= x <= 1}.
    Uses bisection on the water level."""
    n = len(y)
    # f(tau) = sum(clip(y - tau, 0, 1)) is decreasing in tau
    # f(min(y)-1) = n, f(max(y)) = 0
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(y - tau, 0.0, 1.0)

def enforce_exact_sum(h, target_sum):
    """Adjust h so that sum(h) is exactly target_sum (floating point)."""
    diff = target_sum - h.sum()
    if abs(diff) < 1e-15:
        return h
    # Distribute the difference to elements not at 0 or 1
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    if interior.sum() > 0:
        h[interior] += diff / interior.sum()
        h = np.clip(h, 0.0, 1.0)
    else:
        # All at boundaries, adjust one
        # Find an element to adjust
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
    return h

def compute_c5_official(h):
    """Compute C5 using the official formula."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def resample_h(h, new_n):
    """Resample h from its current length to new_n using linear interpolation."""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n  # cell centers
    # Actually, let's use the midpoint rule interpretation

    # For resampling, interpolate at new cell centers

    new_h = np.interp(new_x, old_x, h, left=h[0], right=h[-1])
    return new_h

The key question is: what kind of h minimizes the maximum overlap?

Intuition: We want h to be "spread out" in a way that for every shift k, the overlap between h and (1-h) shifted by k is small. This means h should be close to 1 where (1-h) shifted is close to 0, and vice versa.

This is related to the concept of "complementary sequences" or "Berlekamp sequences" in signal processing.

One approach: Start with a periodic structure. If h has period p (in the discrete sense), then the overlaps at shifts that are multiples of p will have a specific structure.

Another approach: Think of it as a binary problem first. If h is 0 or 1 (with exactly n/2 ones), then I(k) = dx * (number of i where h[i]=1 and h[i+k]=0, for the appropriate range). We want to minimize the maximum of this over k.

This is similar to minimizing the maximum "disagreement" between a binary sequence and its shifts. This is related to binary sequences with low autocorrelation.

For the continuous version (h in [0,1]), we have more flexibility.

Let me think about what the optimal structure might look like. 

If we divide [0,2] into alternating blocks of 1s and 0s, with block length L, then for shifts that are multiples of 2L, the overlap is large (both are the same pattern). For shifts of L, the overlap is 0 (complementary).

The maximum overlap would be at shift 0: I(0) = dx * sum h[i]*(1-h[i]). For binary h, this is 0. For the constraint sum(h) = n/2, the minimum of I(0) is 0 (achieved by binary h).

def project_feasible(y, target_sum):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(80):

        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:

        else:

    if abs(diff) < 1e-14:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)

def compute_gradient_fft(h, weights):
    n = len(h)
    # Build W from weights
    # weights has length 2n-1, index k corresponds to lag d = k-(n-1)
    # W[a] = sum of weights[k] for k such that |k-(n-1)| = a
    W = np.zeros(n)
    W[0] = weights[n-1]
    for a in range(1, n):
        W[a] = weights[n-1-a] + weights[n-1+a]
    
    one_minus_h = 1.0 - h
    neg_h = -h
    
    # Part1 = causal convolution of W with (1-h)
    part1 = fftconvolve(W, one_minus_h, mode='full')[:n]
    # Part2 = correlation of (-h) with W (first n elements)
    part2 = np.correlate(neg_h, W, mode='full')[:n]
    
    return part1 + part2

    h = project_feasible(h.copy(), target_sum)

        if time_fn() < 5.0:

        # Anneal temperature and learning rate
        t = it / max(1, max_iter - 1)

def project_feasible(y, target_sum):

    lo = y.min() - 1.0

    for _ in range(60):

        s = np.clip(y - mid, 0.0, 1.0).sum()
