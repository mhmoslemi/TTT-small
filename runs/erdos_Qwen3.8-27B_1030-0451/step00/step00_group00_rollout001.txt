    h[:m] = p

    p = h[:m].copy()
    return p

def smooth_objective_and_grad(p, eta):
    h = build_h(p)
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    max_R = np.max(R)
    shifted = eta * (R - max_R)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    logsumexp = max_R + np.log(sum_exp) / eta
    obj = dx * logsumexp
    
    weights = exp_shifted / sum_exp
    
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):

        w = weights[k]
        if w < 1e-14:
            continue

            grad_h += w * (1.0 - 2.0 * h)

            grad_h[d:] += w * (1.0 - h[:-d])
            grad_h[:-d] -= w * h[d:]
        else:
            dp = -d
            grad_h[:-dp] += w * (1.0 - h[dp:])
            grad_h[dp:] -= w * h[:-dp]
    
    grad_p = grad_h[:m] - grad_h[m:][::-1]
    return obj, dx * grad_p

def coordinate_refine(p, max_passes=5, n_trials=7):
    """Try small perturbations of each coordinate."""
    m = len(p)
    best_p = p.copy()
    best_h = build_h(best_p)
    best_c = exact_objective(best_h)
    
    # Get gradient to guide search
    _, grad = smooth_objective_and_grad(p, 50.0)
    
    for path in range(max_passes):
        improved = False
        order = np.argsort(np.abs(grad))[::-1]  # Try most important coords first

            base = best_p[j]
            # Try values centered around base, guided by gradient
            step_sizes = np.array([0.01, 0.005, 0.002, 0.001])

                        best_h = trial_h
