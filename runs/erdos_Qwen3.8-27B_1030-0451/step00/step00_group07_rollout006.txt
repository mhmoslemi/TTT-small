def mass_project(x, target):
    """Project x to [0,1]^N with sum = target using bisection + repair."""
    x = np.asarray(x, dtype=np.float64).copy()
    N = len(x)
    if N == 0:
        return x
    # Replace non-finite
    x[~np.isfinite(x)] = 0.5
    
    def F(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0))
    
    # Find bracket
    lo = -x.max() - 1.0
    hi = 1.0 - x.min() + 1.0
    while F(lo) > target:
        lo -= 1.0
    while F(hi) < target:
        hi += 1.0
    
    # Bisection
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if F(mid) > target:
            hi = mid
        else:
            lo = mid
    
    result = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    
    # Repair with fsum
    s = math.fsum(result)
    residual = s - target
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to reduce
            order = np.argsort(-result)
            for idx in order:
                if residual <= 0:
                    break
                reduce = min(residual, result[idx])
                result[idx] -= reduce
                residual -= reduce
        else:
            order = np.argsort(result)
            for idx in order:
                if residual >= 0:
                    break
                add = min(-residual, 1.0 - result[idx])
                result[idx] += add
                residual += add
    result = np.clip(result, 0.0, 1.0)
    return result

def solve_lp(h, corr, active_indices, dx, trust=0.1):
    """Solve the local LP to find a descent direction."""
    N = len(h)
    M = len(active_indices)
    if M == 0:
        return None
    
    # Build constraint matrix
    G = build_active_grads(h, corr, active_indices, dx)  # (M, N)
    
    # A_ub: G @ delta - m <= -corr[active_indices]
    # Row i: [G[i, :], -1] @ [delta, m] <= -corr[active_indices][i]
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = -corr[active_indices]
    
    # A_eq: sum(delta) = 0
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: max(-h[i], -trust) <= delta[i] <= min(1-h[i], trust)
    lb = np.maximum(-h, -trust)
    ub = np.minimum(1.0 - h, trust)
    bounds = list(zip(lb, ub))
    bounds.append((None, None))  # m is free
    
    # Objective: minimize m
    c_obj = np.zeros(N + 1)
    c_obj[-1] = 1.0
    
    try:

    if res.status == 0 and res.x is not None:
        return res.x[:N]  # return the delta
    elif res.status == 3:  # infeasible
        return None
    else:
        return None

        return x

            F_lo -= 1.0

        else:
            F_lo = mid
    result = np.clip(x + 0.5 * (F_lo + F_hi), 0.0, 1.0)
    s = math.fsum(result)
    residual = s - target
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(-result)
            for idx in order:
                if residual <= 1e-13:
                    break
                red = min(residual, result[idx])
                result[idx] -= red
                residual -= red
        else:
            order = np.argsort(result)
            for idx in order:
                if residual >= -1e-13:
                    break
                add = min(-residual, 1.0 - result[idx])
                result[idx] += add
                residual += add
    return np.clip(result, 0.0, 1.0)

def grad_for_shift(s, h, dx):
    N = len(h)
    grad = np.zeros(N)
    js = max(0, s)
    je = min(N, s + N)
    if js < je:
        grad[js:je] += (1.0 - h[js - s:je - s])
    js2 = max(0, -s)
    je2 = min(N, N - s)
    if js2 < je2:
        grad[js2:je2] -= h[js2 + s:je2 + s]
    return grad * dx

def build_active_grads(h, active_indices, dx):
    N = len(h)
    M = len(active_indices)
    G = np.zeros((M, N))
    for i, idx in enumerate(active_indices):
        s = idx - (N - 1)
        G[i] = grad_for_shift(s, h, dx)
    return G

def solve_lp(h, corr, active_indices, dx, trust):
    N = len(h)
    M = len(active_indices)
    if M == 0 or trust < 1e-8:
        return None
    G = build_active_grads(h, active_indices, dx)
    A_ub = np.hstack([G, -np.ones((M, 1))])
    b_ub = -corr[active_indices]
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    lb = np.maximum(-h, -trust)
    ub = np.minimum(1.0 - h, trust)
    bounds = list(zip(lb, ub))
    bounds.append((None, None))
    c_obj = np.zeros(N + 1)
    c_obj[-1] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status == 0 and res.x is not None:
        return res.x[:N]
    return None

def transfer(h, new_N):
    N_old = len(h)
    dx_new = 2.0 / new_N
    dx_old = 2.0 / N_old
    result = np.zeros(new_N)
    for i in range(new_N):
        x_l = i * dx_new
        x_r = (i + 1) * dx_new
        j_l = max(0, int(np.floor(x_l / dx_old)))
        j_r = min(N_old, int(np.ceil(x_r / dx_old)))
        area = 0.0
        for j in range(j_l, j_r):
            ov_l = max(x_l, j * dx_old)
            ov_r = min(x_r, (j + 1) * dx_old)
            if ov_r > ov_l:
                area += h[j] * (ov_r - ov_l)
        result[i] = area / dx_new
    return result

def verify_gradients(h, dx, seed=0):
    N = len(h)
    rng = np.random.default_rng(seed)
    test_shifts = list(range(-5, 6)) + [N // 3, -(N // 3)]
    test_shifts = [s for s in test_shifts if -N + 1 <= s <= N - 1]
    test_indices = rng.choice(N, size=min(10, N), replace=False)
    eps = 1e-7
    for s in test_shifts[:6]:
        idx = s + (N - 1)
        grad = grad_for_shift(s, h, dx)
        for j in test_indices:
            h_p = h.copy()
            h_m = h.copy()
            h_p[j] = min(1.0, h[j] + eps)
            h_m[j] = max(0.0, h[j] - eps)
            vp = np.correlate(h_p, 1.0 - h_p, mode='full')[idx] * dx
            vm = np.correlate(h_m, 1.0 - h_m, mode='full')[idx] * dx
            fd = (vp - vm) / (h_p[j] - h_m[j])
            if abs(fd - grad[j]) > 1e-4:
                return False
    return True

def optimize_at_resolution(h, N, deadline, rng, target_val=0.3808):
    dx = 2.0 / N
    trust = 0.12
    consecutive_fails = 0
    best_local = evaluate_exact(h)
    best_local_h = h.copy()
    momentum = np.zeros(N)
    iter_count = 0
    grad_verified = verify_gradients(h, dx, seed=iter_count)
    
    alpha_list = [1.0, 0.7, 0.5, 0.3, 0.15, 0.08]
    
    while time.time() < deadline - 3.0:
        iter_count += 1
        corr = objective_array(h)
        current_max = float(np.max(corr))
        
        if current_max < best_local - 1e-12:
            best_local = current_max
            best_local_h = h.copy()
            consecutive_fails = 0
        
        if current_max <= target_val - 0.0001:
            break
        
        # Select active constraints
        threshold = current_max - 0.004
        n_active = min(len(corr), max(15, int(0.25 * len(corr))))
        active_mask = corr >= threshold
        n_above = int(np.sum(active_mask))
        if n_above > 0 and n_above <= n_active:
            active_indices = np.where(active_mask)[0]
        else:
            active_indices = np.argsort(corr)[-n_active:][::-1]
        
        # Try LP
        delta = None
        if grad_verified and trust > 1e-5:
            delta = solve_lp(h, corr, active_indices, dx, trust)
        
        # If LP failed, try subgradient
        if delta is None:
            G = build_active_grads(h, active_indices, dx)
            grad_avg = G.mean(axis=0)
            gn = np.linalg.norm(grad_avg)
            if gn > 1e-14:
                # Use momentum
                momentum = 0.5 * momentum + 0.5 * (-grad_avg / gn)
                mn = np.linalg.norm(momentum)
                if mn > 1e-14:
                    delta = momentum / mn * trust
                else:
                    delta = None
            else:
                delta = None
        
        if delta is None:
            # Random perturbation
            delta = (rng.random(N) - 0.5) * trust * 2.0
            consecutive_fails += 1
            if consecutive_fails > 30:
                break
            continue
        
        # Line search
        improved = False
        for alpha in alpha_list:
            candidate = mass_project(h + alpha * delta, N / 2.0)
            val = evaluate_exact(candidate)
            if np.isfinite(val) and val < current_max - 1e-13:
                h = candidate
                improved = True
                break
        
        if improved:
            trust = min(0.4, trust * 1.08)
            consecutive_fails = 0
        else:
            trust *= 0.65
            consecutive_fails += 1
            if trust < 1e-6:
                # Try larger random jump
                h = mass_project(h + (rng.random(N) - 0.5) * 0.3, N / 2.0)
                trust = 0.1
                consecutive_fails = 0
            elif consecutive_fails % 15 == 0:
                # Random restart blend
                blend = mass_project(0.7 * h + 0.3 * (0.5 + 0.3 * (rng.random(N) - 0.5)), N / 2.0)
                if evaluate_exact(blend) < best_local + 0.01:
                    h = blend
                    trust = 0.15
                    consecutive_fails = 0
    
    return best_local_h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    reserve = max(5.0, 0.04 * budget_s)
    deadline = start_time + budget_s - reserve
    rng = np.random.default_rng(seed)
    
    # Get initial h
    h0 = None
    try:
        h0_raw = np.asarray(initial_h_values, dtype=np.float64).flatten()
        if len(h0_raw) > 0 and np.all(np.isfinite(h0_raw)):
            N_raw = len(h0_raw)
            h0 = mass_project(h0_raw, N_raw / 2.0)
    except (NameError, UnboundLocalError, Exception):
        pass
    
    if h0 is None:
        N0 = 120
        h0 = mass_project(np.full(N0, 0.5), N0 / 2.0)
    
    best_h = h0.copy()
    best_val = evaluate_exact(best_h)
    
    # Build resolution plan
    N_start = len(h0)
    resolutions = [N_start]
    if budget_s > 200:
        n = N_start
        while n * 2 <= min(2400, 4 * N_start) and len(resolutions) < 4:
            n *= 2
            resolutions.append(n)
    
    # Allocate time budget across resolutions
    total_units = sum(r / N_start for r in resolutions)
    available = max(10.0, deadline - start_time)
    phase_deadlines = []
    t = start_time
    for r in resolutions:
        frac = (r / N_start) / total_units
        t += available * frac
        phase_deadlines.append(min(t, deadline - 3.0))
    
    current_h = h0.copy()
    
    for res_idx, target_N in enumerate(resolutions):
        phase_dl = phase_deadlines[res_idx]
        if time.time() >= phase_dl - 2.0:
            continue
        
        # Transfer to target resolution
        if target_N != len(current_h):
            current_h = transfer(current_h, target_N)
            current_h = mass_project(current_h, target_N / 2.0)
        
        # Optimize
        opt_h, opt_val = optimize_at_resolution(
            current_h, target_N, phase_dl, rng
        )
        
        if opt_val < best_val - 1e-12:
            best_val = opt_val
            best_h = opt_h.copy()
        else:
            # Keep the better of current and previous
            cur_val = evaluate_exact(current_h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = current_h.copy()
        
        current_h = opt_h
    
    # Try a few random blends for diversity if time remains
    if time.time() < deadline - 10.0 and len(best_h) > 0:
        N_best = len(best_h)
        forTrial in range(5):
            if time.time() >= deadline - 5.0:
                break
            blend = mass_project(
                0.6 * best_h + 0.4 * mass_project(rng.random(N_best), N_best / 2.0),
                N_best / 2.0
            )
            bv = evaluate_exact(blend)
            if np.isfinite(bv) and bv < best_val - 1e-12:
                best_val = bv
                best_h = blend
    
    # Final validation
    best_h = mass_project(best_h, len(best_h) / 2.0)
    final_val = evaluate_exact(best_h)
    if np.isfinite(final_val):
        best_val = final_val
    
    # Safety: ensure we don't return worse than initial
    if h0 is not None:
        h0_val = evaluate_exact(h0)
        if np.isfinite(h0_val) and h0_val < best_val:
            best_h = h0.copy()
            best_val = h0_val
    
    # Ultimate fallback
    if not np.isfinite(best_val) or best_val > 0.45:
        N_fb = 120
        best_h = mass_project(np.full(N_fb, 0.5), N_fb / 2.0)
        best_val = evaluate_exact(best_h)
    
    return (best_h, best_val, len(best_h))
