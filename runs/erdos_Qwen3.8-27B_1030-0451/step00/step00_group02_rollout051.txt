import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def objective(h):
    return compute_c5(h)

def constraints(n):
    # sum(h) = n/2
    # 0 <= h[i] <= 1
    cons = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0, 'jac': lambda h: np.ones(n)}
    ]
    bounds = [(0, 1)] * n
    return cons, bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values if available
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 81
        h0 = np.zeros(n0)
        h0[:n0//2] = 1.0
        h0[n0//2] = 0.5  # to make sum = n0/2
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    def elapsed():
        return time.time() - start_time
    
    # Try optimization from the initial point
    cons, bounds = constraints(n0)
    
    result = minimize(objective, h0, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 500, 'ftol': 1e-10})
    
    if result.success or result.fun < best_c5:
        h_opt = np.clip(result.x, 0, 1)
        # Project onto sum constraint
        h_opt = project(h_opt, n0)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Try with different n values
    for n in [100, 150, 200, 250]:
        if elapsed() > budget_s * 0.7:
            break
        
        # Create initial guess
        h_init = create_initial(n, rng)
        c5_init = compute_c5(h_init)
        
        cons_n, bounds_n = constraints(n)
        result = minimize(objective, h_init, method='SLSQP',
                         bounds=bounds_n, constraints=cons_n,

        if result.fun < best_c5:
            h_opt = np.clip(result.x, 0, 1)

            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    return best_h, best_c5, len(best_h)

def project(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    # Simple approach: adjust
    if s > target:
        # Reduce elements that are close to 1
        diff = s - target
        # Distribute reduction
        h = h - diff / n
        h = np.clip(h, 0, 1)
    elif s < target:
        diff = target - s
        h = h + diff / n
        h = np.clip(h, 0, 1)
    return h

def create_initial(n, rng):
    """Create a feasible initial point."""
    # Start with a block structure and add randomness
    h = np.zeros(n)
    h[:n//2] = 1.0
    if n % 2 == 1:
        h[n//2] = 0.5
    # Add small perturbations
    h = h + rng.uniform(-0.1, 0.1, n)
    h = np.clip(h, 0, 1)
    h = project(h, n)

def compute_correlations(h):
    """Compute all correlation values."""
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlations(h)
    return np.max(c) * dx

def get_active_shifts(h, top_m=20):
    """Get the top m shifts by correlation value."""
    c = compute_correlations(h)
    idx = np.argsort(c)[::-1][:top_m]
    return idx, c[idx]

def slsqp_optimize(h0, n, maxiter=500):
    """Optimize using SLSQP."""
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return compute_c5(x)
    
    result = minimize(obj, h0, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-8})
    return np.clip(result.x, 0, 1)

def project_to_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    target = n / 2.0

        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break

            if not np.any(mask):

            if not np.any(mask):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project to {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact projection
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-10:
        # Distribute remainder
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def make_feasible(n, pattern='constant', rng=None):
    """Create a feasible initial point."""
    if pattern == 'constant':
        h = np.full(n, 0.5)
    elif pattern == 'ramp':
        h = np.linspace(0.9, 0.1, n)
    elif pattern == 'random':
        h = rng.uniform(0, 1, n) if rng is not None else np.random.uniform(0, 1, n)
    elif pattern == 'block':
        h = np.zeros(n)
        h[:n//2] = 1.0
        if n % 2 == 1:
            h[n//2] = 0.5
    elif pattern == 'sine':
        x = np.linspace(0, 2*np.pi, n)
        h = 0.5 + 0.3 * np.sin(x)
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

def slsqp_optimize(h0, n, maxiter=500, ftol=1e-10):
    """Optimize C5 using SLSQP."""
    h0 = project_feasible(h0.copy(), n)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(compute_c5, h0, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
    h_opt = project_feasible(np.clip(result.x, 0, 1), n)
    return h_opt, compute_c5(h_opt)

**Active Set Successive Minimization:**
1. Start with h₀.
2. Compute all g_k(h₀), find the active set A = {k : g_k(h₀) ≈ max}.
3. Solve the subproblem: minimize max_{k∈A} g_k(h) subject to constraints and g_k(h) ≤ max + ε for k ∉ A.
4. Update A and repeat.

def active_set_optimize(h0, n, max_rounds=20, top_m=5):
    """Active set optimization for C5."""
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for round_ in range(max_rounds):
        c = np.correlate(h, 1.0 - h, mode="full")
        c5_vals = c * dx
        max_val = np.max(c5_vals)
        
        # Find active set (top m values within tolerance of max)
        threshold = max_val - dx * 0.01  # small tolerance
        active = np.where(c5_vals >= threshold)[0]
        
        if len(active) > top_m:
            # Keep only top m
            top_idx = np.argsort(c5_vals[active])[::-1][:top_m]
            active = active[top_idx]
        
        # Solve subproblem: minimize max over active set
        # Use SLSQP with inequality constraints for active shifts
        def sub_obj(x):
            c_sub = np.correlate(x, 1.0 - x, mode="full")
            return np.max(c_sub[active]) * dx
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
        # Add inequality constraints for inactive shifts (optional, for correctness)
        bounds = [(0.0, 1.0)] * n
        
        try:
            result = minimize(sub_obj, h, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 100, 'ftol': 1e-10})
            h_new = project_feasible(np.clip(result.x, 0, 1), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
            else:
                break
        except:
            break
    
    return h, best_c5
