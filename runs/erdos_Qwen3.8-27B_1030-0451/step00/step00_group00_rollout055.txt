    """Compute the correlation array C."""
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradient(h, p, dx):
    """Compute gradient of sum_j p[j] * C[j]."""
    N = len(h)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-15:
            continue
        if j <= N - 1:
            lo1, hi1 = 0, j
            lo2, hi2 = N - 1 - j, N - 1
        else:
            lo1, hi1 = j - (N - 1), N - 1
            lo2, hi2 = 0, 2 * (N - 1) - j

def project_capped_simplex(v, S):
    """Project v onto {h : 0<=h<=1, sum(h)=S}."""
    N = len(v)

        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)

import numpy as np
import time

def compute_C(h, dx):
    """Compute the overlap correlation array."""
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_grad(h, p, dx):
    """Compute gradient: sum_j p[j] * dC[j]/dh."""
    N = len(h)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        w = p[j]
        if w < 1e-14:
            continue
        if j <= N - 1:
            lo1, hi1 = 0, j
            lo2, hi2 = N - 1 - j, N - 1
        else:
            lo1, hi1 = j - (N - 1), N - 1
            lo2, hi2 = 0, 2 * (N - 1) - j
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1) - j + (N - 1)
            grad[lo1:hi1 + 1] += w * dx * (1.0 - h[idx])
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1) + j - (N - 1)
            grad[lo2:hi2 + 1] -= w * dx * h[idx2]
    return grad

def project(v, S):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=S} via bisection."""
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def optimize(h, N, dx, S, budget, tau_init=40.0, lr_init=0.08, seed=42):

    h = project(h.copy(), S)
    best_c5 = compute_C(h, dx).max()
    best_h = h.copy()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iters = 0

    while time.time() - start < budget:
        C = compute_C(h, dx)
        c5 = C.max()

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Softmax weights
        shift = C - C.max()
        exp_tau = tau * shift
        exp_tau -= exp_tau.max()
        p = np.exp(exp_tau)
        ps = p.sum()
        if ps < 1e-300:
            p[C.argmax()] = 1.0
        else:
            p /= ps

        # Gradient
        grad = compute_grad(h, p, dx)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        # Backtracking line search
        alpha = lr
        h_new = h
        c_new = c5
        for _ in range(12):
            cand = project(h - alpha * grad, S)
            cc = compute_C(cand, dx).max()
            if cc < c5 - 1e-12:
                h_new = cand
                c_new = cc
                break
            alpha *= 0.5

        h = h_new
        iters += 1

        # Adapt parameters

            lr = min(lr * 1.1, 0.8)

def local_search_binary(h, N, dx, S, budget, seed=0):

                    ones = np.where(best_h > 0.5)[0]

                break

    """Create a structured initial point."""
    rng = np.random.RandomState(seed)
    # Start with alternating blocks
    h = np.zeros(N)
    # 4 blocks of 1s and 4 blocks of 0s

        end = start + block
        if end <= N:
            h[start:end] = 1.0
    # Add noise
    h += 0.1 * rng.randn(N)
    return h
