    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, n, target_sum):

def interpolate_to(h_src, n_src, n_dst):
    """Interpolate h from n_src to n_dst points preserving integral."""
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    # Use linear interpolation
    h_interp = np.interp(x_dst, x_src, np.concatenate([h_src, [h_src[0]]]))
    target_sum = n_dst / 2.0
    return project_feasible(h_interp, n_dst, target_sum)

def block_candidate(n, rng):
    """Generate a block-like candidate."""
    h = np.zeros(n)
    # Random blocks
    pos = 0
    while pos < n:
        block_len = int(rng.randint(1, n // 4 + 1))
        val = rng.random()
        end = min(pos + block_len, n)
        h[pos:end] = val
        pos = end
    return project_feasible(h, n, n / 2.0)

def cosine_candidate(n, rng):
    """Generate a cosine-like candidate."""
    x = np.linspace(0, 2 * np.pi, n)
    a = rng.uniform(0.5, 2.0)
    b = rng.uniform(0, 2*np.pi)
    h = 0.5 + a * 0.3 * np.cos(a * x + b)
    return project_feasible(h, n, n / 2.0)
