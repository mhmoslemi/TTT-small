def repair_profile(arr):
    n = len(arr)
    h = np.asarray(arr, dtype=float).reshape(-1)
    if n == 0: return h, False
    if not np.all(np.isfinite(h)):
        # replace nan with 0.5
        h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    # adjust by descending/ascending
    diff = np.sum(h) - target
    tol = 1e-10
    # try several passes
    for _ in range(3):
        diff = np.sum(h) - target
        if abs(diff) <= 1e-13: break
        if diff > 0:
            order = np.argsort(-h, kind='stable')
            rem = diff
            for idx in order:
                if rem <= 0: break
                d = min(h[idx], rem)
                if d > 0:
                    h[idx] -= d
                    rem -= d
            if rem > 1e-12:
                # forcibly take from largest
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] - rem)
        else:
            rem = -diff
            order = np.argsort(h, kind='stable')
            for idx in order:
                if rem <= 0: break
                d = min(1.0 - h[idx], rem)
                if d > 0:
                    h[idx] += d
                    rem -= d
            if rem > 1e-12:
                idx = int(np.argmin(h))
                h[idx] = min(1.0, h[idx] + rem)
    h = np.clip(h,0,1)
    # final exact adjustment at a variable with slack
    diff = np.sum(h) - target
    if abs(diff) > 0:
       # find index with room in direction
       if diff > 0:
           idx = int(np.argmax(h))
           room = h[idx]
           d = min(room, diff)
           h[idx] -= d
       else:
           idx = int(np.argmin(h))
           room = 1.0 - h[idx]
           d = min(room, -diff)
           h[idx] += d
       # try last element to fix residual
       s = np.sum(h[:-1]) if n>1 else 0.0
       val = target - s
       if 0.0 <= val <= 1.0:
           h[-1] = val
    # final clip and small fix
    h = np.clip(h,0,1)
    diff = np.sum(h) - target
    ok = np.all(np.isfinite(h)) and np.all(h >= -1e-12) and np.all(h <= 1+1e-12) and abs(diff) <= 1e-8

def active_slsqp_optimize(h, time_limit, rng):
    if time_limit <= 0: return h, np.inf
    best_h, best_c, ok = evaluate_profile(h)
    if not ok: return h, np.inf
    working_h = best_h.copy(); working_c = best_c
    deadline = time.monotonic()+time_limit
    it = 0
    eps_vals = (2e-5,5e-5,1e-4,2e-4,5e-4,1e-4)
    K_vals = (30,50,70,50,30,50)
    no_improve = 0; stall = 0
    while True:
       left = deadline - time.monotonic()
       if left < 90: break
       it += 1
       # working repair/eval
       working_h, cok = repair_profile(working_h)
       if not cok:
          working_h=best_h.copy(); working_c=best_c
       c_work, vals, okev = evaluate_profile(working_h)
       if not okev: break
       if c_work < best_c - 1e-13:

       else:
          no_improve +=1

       active = select_active(vals, c_work, K=K, eps=eps, max_count=90)

def pair_swap_polish(h, time_limit):
    h, ok = repair_profile(h)
    if not ok or time_limit <= 0 or h.size <2:
       best, bc, _ = evaluate_profile(h); return best, bc
    best = h.copy(); best_c, _ = c5_value_fast(best)
    n = best.size
    deadline = time.monotonic() + time_limit
    while time.monotonic() < deadline:
       h = best.copy()
       cur = best_c
       improved = False
       for i in range(n-1):
          if time.monotonic() >= deadline: break
          hi = h[i]
          for j in range(i+1,n):
             hj = h[j]
             if abs(hi-hj) < 1e-13: continue
             h[i] = hj; h[j] = hi
             v, _ = c5_value_fast(h)
             if np.isfinite(v) and v < cur - 1e-13:
                cur = v; best = h.copy(); improved = True; break
             else:

import numpy as np

def repair_profile(arr):
    a = np.asarray(arr, dtype=float).reshape(-1)
    n = a.size
    if n == 0:
        return a, False

    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.5)
    a = np.clip(a, 0.0, 1.0)

    target = 0.5 * float(n)

    for _ in range(4):
        diff = float(np.sum(a) - target)
        if abs(diff) <= 1e-13:
            break

        if diff > 0.0:
            order = np.argsort(-a, kind='stable')
            rem = diff
            for idx in order:
                if rem <= 0.0:
                    break
                d = float(a[idx])
                if d > rem:
                    d = rem
                if d > 0.0:
                    a[idx] = a[idx] - d
                    rem -= d
            if rem > 1e-12:
                idx = int(np.argmax(a))
                a[idx] = max(0.0, float(a[idx]) - rem)
        else:
            rem = -diff
            order = np.argsort(a, kind='stable')
            for idx in order:
                if rem <= 0.0:
                    break
                d = 1.0 - float(a[idx])
                if d > rem:
                    d = rem
                if d > 0.0:
                    a[idx] = a[idx] + d
                    rem -= d
            if rem > 1e-12:
                idx = int(np.argmin(a))
                a[idx] = min(1.0, float(a[idx]) + rem)

        a = np.clip(a, 0.0, 1.0)

    diff = float(np.sum(a) - target)
    if diff > 0.0:
        idx = int(np.argmax(a))
        a[idx] = max(0.0, float(a[idx]) - diff)
    elif diff < 0.0:
        idx = int(np.argmin(a))
        a[idx] = min(1.0, float(a[idx]) - diff)

    if n > 1:
        s = float(np.sum(a[:-1]))
        val = target - s
        if 0.0 <= val <= 1.0:
            a[-1] = val

    a = np.clip(a, 0.0, 1.0)
    ok = (
        np.all(np.isfinite(a))
        and np.all(a >= -1e-12)
        and np.all(a <= 1.0 + 1e-12)
        and abs(float(np.sum(a)) - target) <= 1e-8
    )
    return a, ok


def evaluate_profile(h):
    h, ok = repair_profile(h)
    if not ok or h.size == 0:
        return np.inf, None, False
    dx = 2.0 / float(h.size)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return np.inf, vals, False
    return float(np.max(vals)), vals, True


def c5_value_fast(h):
    dx = 2.0 / float(h.size)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return np.inf, vals
    return float(np.max(vals)), vals


def compute_values_at_indices(h, idx):
    n = int(h.size)
    dx = 2.0 / float(n)
    q = 1.0 - h
    idx = np.asarray(idx, dtype=int)
    out = np.empty(idx.size, dtype=float)

    for k in range(idx.size):
        i = int(idx[k])
        s = n - 1 - i
        if s >= 0:
            m = n - s
            out[k] = dx * float(np.dot(h[:m], q[s:s + m]))
        else:
            r = -s
            m = n - r
            out[k] = dx * float(np.dot(h[r:r + m], q[:m]))
    return out


def make_fallback(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = (
        0.5
        + 0.45 * np.sin(2.0 * np.pi * 2.0 * x)
        + 0.30 * np.cos(2.0 * np.pi * 5.0 * x + 0.7)
        + 0.20 * np.sin(2.0 * np.pi * 11.0 * x + 1.3)
    )
    if rng is not None:
        h = h + rng.normal(0.0, 0.10, size=n)
    h = np.clip(h, 0.0, 1.0)
    h, ok = repair_profile(h)
    if not ok:
        h = np.zeros(n, dtype=float)
        h[n // 2] = 1.0
        h, ok = repair_profile(h)
    return h


def add_noise(h, rng, scale):
    h0 = np.asarray(h, dtype=float).reshape(-1).copy()
    if rng is not None and scale > 0.0:
        h0 = h0 + rng.normal(0.0, scale, size=h0.size)
    h1, ok = repair_profile(h0)
    if ok:
        return h1
    return h0


def select_active(vals, current_c, K=80, eps=1e-4, max_count=120, floor=None):
    m = int(vals.size)
    if m == 0:
        return np.empty(0, dtype=int)

    idx = np.argsort(vals)[::-1]

    if K is None or K >= m:
        active = idx.copy()
    else:
        active = idx[:K]

    threshold = float(current_c) - float(eps)
    if floor is not None and float(floor) < threshold:
        threshold = float(floor)

    within = idx[vals >= threshold]
    active = np.unique(np.concatenate((active, within)))

    if max_count is not None and active.size > max_count:
        order = np.argsort(vals[active])[::-1]
        active = active[order[:max_count]]

    return active


def _slsqp_objective(x):
    return float(x[-1])


def _slsqp_objective_jac(x):
    x = np.asarray(x, dtype=float)
    g = np.zeros(x.size, dtype=float)
    g[-1] = 1.0
    return g


def _slsqp_eq(x):
    x = np.asarray(x, dtype=float)
    n = int(_STATE.get('n', x.size - 1))
    if not np.all(np.isfinite(x)):
        return 0.0
    return float(np.sum(x[:n]) - _STATE.get('S', 0.5 * float(n)))


def _slsqp_ineq(x):
    x = np.asarray(x, dtype=float)
    n = int(_STATE.get('n', x.size - 1))
    active = _STATE.get('active_indices', None)

    if active is None:
        active = np.arange(max(0, 2 * n - 1), dtype=int)
    else:
        active = np.asarray(active, dtype=int)

    m = active.size
    if m == 0:
        return np.empty(0, dtype=float)

    if not np.all(np.isfinite(x)):
        return np.full(m, 1e6, dtype=float)

    h = x[:n]
    t = float(x[n])
    vals = compute_values_at_indices(h, active)
    return vals - t


def slsqp_solve_active(h, active_idx, maxiter=30):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return h

    _STATE['n'] = int(n)
    _STATE['S'] = float(0.5 * n)
    _STATE['active_indices'] = np.asarray(active_idx, dtype=int)

    c0, _ = c5_value_fast(h)
    if not np.isfinite(c0):
        return h

    t0 = float(c0) + 1e-7
    if t0 > 1.0:
        t0 = 1.0

    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = t0

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'type': 'eq', 'fun': _slsqp_eq},
        {'type': 'ineq', 'fun': _slsqp_ineq},
    ]
    maxiter = max(5, int(maxiter))

    try:
        res = minimize(
            _slsqp_objective,
            x0,
            jac=_slsqp_objective_jac,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-12, 'maxiter': maxiter, 'disp': False},
        )
        if res.x is not None:
            hnew = np.asarray(res.x[:n], dtype=float)
            if np.all(np.isfinite(hnew)):
                return hnew
    except Exception:
        pass

    return h

def pair_swap_polish(h, time_limit):
    h, ok = repair_profile(h)
    if not ok or h.size < 2:
        hh, cc, _ = evaluate_profile(h)
        return hh, cc
    if time_limit <= 0.0:
        hh, cc, _ = evaluate_profile(h)
        return hh, cc

    best = h.copy()
    best_c, _ = c5_value_fast(best)
    n = best.size
    deadline = time.monotonic() + float(time_limit)

    while time.monotonic() < deadline:
        cur_h = best.copy()
        cur_c = best_c
        improved = False

        for i in range(n - 1):
            if time.monotonic() >= deadline:
                break
            hi = cur_h[i]
            for j in range(i + 1, n):
                hj = cur_h[j]
                if abs(hi - hj) < 1e-13:
                    continue

                cur_h[i] = hj
                cur_h[j] = hi
                v, _ = c5_value_fast(cur_h)

                if np.isfinite(v) and v < cur_c - 1e-13:
                    cur_c = v
                    best = cur_h.copy()
                    improved = True
                    break
                else:
                    cur_h[i] = hi
                    cur_h[j] = hj

            if improved or time.monotonic() >= deadline:
                break

        if not improved:
            break

    best, best_c, _ = evaluate_profile(best)
    return best, float(best_c) if np.isfinite(best_c) else np.inf


def anneal_polish(h, time_limit, rng):
    h, ok = repair_profile(h)
    if not ok or h.size < 2:
        hh, cc, _ = evaluate_profile(h)
        return hh, cc
    if time_limit <= 0.0:
        hh, cc, _ = evaluate_profile(h)
        return hh, cc

    best = h.copy()
    current = h.copy()
    best_c, _ = c5_value_fast(best)
    current_c = best_c

    deadline = time.monotonic() + float(time_limit)
    T = 3e-5
    n = current.size
    evals = 0

    while time.monotonic() < deadline:
        if rng is not None and rng.random() < 0.35:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

            d = float(rng.choice(_DELTAS))
            oi = current[i]
            oj = current[j]

            if oi > oj:
                delta = min(d, float(oi), 1.0 - float(oj))
                if delta <= 1e-12:
                    continue
                current[i] = oi - delta
                current[j] = oj + delta
            else:
                delta = min(d, float(oj), 1.0 - float(oi))
                if delta <= 1e-12:
                    continue
                current[i] = oi + delta
                current[j] = oj - delta

            v, _ = c5_value_fast(current)
            delta_c = float(v) - float(current_c)

            if (
                np.isfinite(v)
                and (
                    delta_c < -1e-13
                    or (T > 0.0 and rng.random() < math.exp(-delta_c / T))
                )
            ):
                current_c = float(v)
                if v < best_c - 1e-13:
                    best_c = float(v)
                    best = current.copy()
            else:
                current[i] = oi
                current[j] = oj
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

            oi = current[i]
            oj = current[j]
            if abs(oi - oj) < 1e-12:
                continue

            current[i] = oj
            current[j] = oi
            v, _ = c5_value_fast(current)
            delta_c = float(v) - float(current_c)

            if (
                np.isfinite(v)
                and (
                    delta_c < -1e-13
                    or (T > 0.0 and rng.random() < math.exp(-delta_c / T))
                )
            ):
                current_c = float(v)
                if v < best_c - 1e-13:
                    best_c = float(v)
                    best = current.copy()
            else:
                current[i] = oi
                current[j] = oj

        T = max(1e-8, T * 0.9995)
        evals += 1

        if evals % 2000 == 0:
            if current_c - best_c > 0.01:
                current = best.copy()
                current_c = best_c
                T = min(T, 5e-5)

            current, okc = repair_profile(current)
            if not okc:
                current = best.copy()
                current_c = best_c
            else:

                if np.isfinite(cc):

                    if current_c < best_c - 1e-13:
                        best_c = current_c
                        best = current.copy()

    best, best_c, _ = evaluate_profile(best)
    return best, float(best_c) if np.isfinite(best_c) else np.inf
