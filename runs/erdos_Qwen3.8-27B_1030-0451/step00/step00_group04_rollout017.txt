        return np.sum(np.clip(x - lam, 0.0, 1.0))
    
    # Find bracket
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    # sum_clipped is decreasing in lam
    # We want sum_clipped(lam) = target
    # If lam is very small, sum_clipped is close to n (all clipped to 1)
    # If lam is very large, sum_clipped is close to 0

        s = sum_clipped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny numerical error in the sum
    err = target - np.sum(result)
    if abs(err) > 1e-10:
        # Distribute error to elements not at bounds
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            result[mask] += err / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)

    z = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(z))


def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    err = target - result.sum()
    if abs(err) > 1e-12:
        mask = (result > 1e-9) & (result < 1.0 - 1e-9)
        if mask.sum() > 0:
            result[mask] += err / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result


def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full") * dx
    z_max = z.max()
    shifted = (z - z_max) / tau
    ev = np.exp(np.clip(shifted, -60.0, 0.0))
    se = ev.sum()
    F = z_max + tau * np.log(max(se, 1e-300))
    p = ev / se
    g = np.zeros(n)
    for q in range(len(z)):
        pq = p[q]
        if pq < 1e-11:
            continue
        w = pq * dx
        shift = q - (n - 1)
        if shift >= 0:
            L = n - shift
            if L > 0:
                g[shift:] += w * (1.0 - h[:L])
                g[:L] -= w * h[shift:]
        else:
            a = -shift
            L = n - a
            if L > 0:
                g[:L] += w * (1.0 - h[a:])
                g[a:] -= w * h[:L]
    return F, g


def pgd(h0, tau0, tau_min, max_iter, tlimit):
    n = len(h0)
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_s = evaluate_c5(h)
    F, g = smooth_max_grad(h, tau0)
    tau = tau0
    alpha = 0.03
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > tlimit:
            break
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-13 or not np.all(np.isfinite(g)):
            break
        d = -g / gnorm
        found = False
        for tr in range(25):
            a = alpha * (0.6 ** tr)
            hn = project_feasible(h + a * d, n)
            Fn, gn = smooth_max_grad(hn, tau)
            if Fn < F - 1e-12:
                hs = evaluate_c5(hn)
                if hs < best_s - 1e-10:
                    best_s = hs
                    best_h = hn.copy()
                h, F, g = hn, Fn, gn
                found = True
                break
        if not found:
            alpha *= 0.5
            if alpha < 1e-15:
                break
        if it % 35 == 34:
            tau = max(tau * 0.78, tau_min)
            F, g = smooth_max_grad(h, tau)
        if it % 50 == 0:
            s = evaluate_c5(h)
            if s < best_s - 1e-10:
                best_s = s
                best_h = h.copy()
    s = evaluate_c5(h)
    if s < best_s - 1e-10:
        best_s, best_h = s, h.copy()
    return best_h, best_s


class EpiProblem:
    def __init__(self, h0, active_idx, tbudget):
        self.n = len(h0)
        self.dx = 2.0 / self.n
        self.n2 = 2 * self.n - 1
        self.active = active_idx
        self.tbudget = tbudget
        self.t0 = time.time()
        self.h0 = h0
        self.nc = len(active_idx)

    def obj(self, x):
        return x[-1]

    def objg(self, x):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g

    def conval(self, x):
        if time.time() - self.t0 > self.tbudget:
            return np.ones(self.nc) * 100.0
        h = x[:-1]
        t = x[-1]
        z = np.correlate(h, 1.0 - h, mode="full") * self.dx
        return z[self.active] - t

    def conjac(self, x):
        if time.time() - self.t0 > self.tbudget:
            return np.ones((self.nc, self.n + 1))
        h = x[:-1]
        n, dx = self.n, self.dx
        J = np.zeros((self.nc, n + 1))
        J[:, -1] = -1.0
        for i, q in enumerate(self.active):
            shift = q - (n - 1)
            if shift >= 0:
                L = n - shift
                if L > 0:
                    J[i, shift:shift + L] = (1.0 - h[:L]) * dx
                    J[i, :L] += -h[shift:] * dx
            else:
                a = -shift
                L = n - a
                if L > 0:
                    J[i, :L] = (1.0 - h[a:]) * dx
                    J[i, a:] += -h[:L] * dx
        return J

    def eqfun(self, x):
        return np.sum(x[:-1]) - self.n / 2.0

    def eqjac(self, x):
        J = np.ones(self.n + 1)
        J[-1] = 0.0
        return J


def slsqp_polish(h0, tlimit, n_active=60):
    n = len(h0)
    score0 = evaluate_c5(h0)
    z = np.correlate(h0, 1.0 - h0, mode="full") * (2.0 / n)
    active = np.argsort(z)[::-1][:min(n_active, 2 * n - 1)]
    epi = EpiProblem(h0, active, tlimit)
    x0 = np.concatenate([h0, [score0 + 0.002]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.7)]
    maxiter = max(50, min(400, int(n / 2)))
    res = minimize(
        epi.obj, x0, jac=epi.objg, method='SLSQP', bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': epi.conval, 'jac': epi.conjac},
            {'type': 'eq', 'fun': epi.eqfun, 'jac': epi.eqjac}
        ],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    if res.x is not None:
        h_sqp = project_feasible(np.asarray(res.x[:-1], dtype=float), n)
        s = evaluate_c5(h_sqp)
        if s < score0:
            return h_sqp, s
    return h0, score0


def upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_up = np.empty(n_new)
    for i in range(n_new):
        src = i / factor
        i0 = int(src)
        i1 = min(i0 + 1, n - 1)
        frac = src - i0
        h_up[i] = (1 - frac) * h[i0] + frac * h[i1]
    return project_feasible(h_up, n_new)


def downsample(h, factor):
    n = len(h)
    n_orig = n // factor
    h_blocks = h[:n_orig * factor].reshape(factor, n_orig)
    h_avg = h_blocks.mean(axis=0)
    return project_feasible(h_avg, n_orig)


def random_perturb(h, rng, strength=0.05):
    n = len(h)
    noise = rng.standard_normal(n) * strength
    h_new = project_feasible(h + noise, n)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    total_budget = budget_s - 15.0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            h_init = np.asarray(initial_h_values, dtype=float).copy()
        except Exception:
            h_init = None
    else:
        h_init = None

    if h_init is None or len(h_init) == 0:
        n = 250
        h_init = np.full(n, 0.5)

    n = len(h_init)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_score = evaluate_c5(h_init)

    def time_left():
        return total_budget - (time.time() - t_start)

    # Phase 1: PGD on initial resolution
    if time_left() > 100:
        tl = min(time_left() * 0.35, 300)
        try:
            h1, s1 = pgd(best_h, 0.02, 1e-5, 600, tl)
            if s1 < best_score - 1e-10:
                best_score = s1
                best_h = h1
        except Exception:
            pass

    # Phase 2: SLSQP polish
    if time_left() > 40:
        tl = min(time_left() * 0.15, 120)
        try:
            h2, s2 = slsqp_polish(best_h, tl, n_active=50)
            if s2 < best_score - 1e-10:
                best_score = s2
                best_h = h2
        except Exception:
            pass

    # Phase 3: PGD with complement symmetry
    if time_left() > 60:
        tl = min(time_left() * 0.15, 120)
        h_comp = project_feasible(1.0 - best_h, n)
        try:
            h3, s3 = pgd(h_comp, 0.015, 1e-5, 300, tl)
            if s3 < best_score - 1e-10:
                best_score = s3
                best_h = h3
        except Exception:
            pass

    # Phase 4: Upsample and optimize
    if time_left() > 120 and n <= 600:
        tl = min(time_left() * 0.2, 250)
        n2 = n * 2
        if n2 <= 2400:
            h_up = upsample(best_h, 2)
            try:
                h4, s4 = pgd(h_up, 0.015, 1e-5, 400, tl)
                if s4 < best_score - 1e-10:
                    best_score = s4
                    best_h = h4
            except Exception:
                pass

    # Phase 5: SLSQP on best (possibly higher resolution)
    if time_left() > 40:
        tl = min(time_left() * 0.15, 100)
        try:
            h5, s5 = slsqp_polish(best_h, tl, n_active=50)
            if s5 < best_score - 1e-10:
                best_score = s5
                best_h = h5
        except Exception:
            pass

    # Phase 6: Random perturbations + PGD
    if time_left() > 60:
        ntries = 3
        per_try = time_left() / (ntries + 0.5) / len(best_h) * 300
        for trial in range(ntries):
            if time_left() < 20:
                break
            h_pert = random_perturb(best_h, rng, strength=0.03)
            try:
                h6, s6 = pgd(h_pert, 0.012, 1e-5, 200, min(per_try, time_left() * 0.3))
                if s6 < best_score - 1e-10:
                    best_score = s6
                    best_h = h6
            except Exception:
                pass

    # Phase 7: Final PGD polish
    if time_left() > 30:
        tl = min(time_left() * 0.5, 120)
        try:
            h7, s7 = pgd(best_h, 0.008, 1e-6, 300, tl)
            if s7 < best_score - 1e-10:
                best_score = s7
                best_h = h7
        except Exception:
            pass

    # Final SLSQP
    if time_left() > 25:
        tl = min(time_left() * 0.4, 60)
        try:
            h8, s8 = slsqp_polish(best_h, tl, n_active=40)
            if s8 < best_score - 1e-10:
                best_score = s8
                best_h = h8
        except Exception:
            pass

    # Final validation and repair
    best_h = project_feasible(best_h, len(best_h))
    final_score = evaluate_c5(best_h)
    if final_score > best_score:
        best_score = final_score

    # Ensure valid output
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_score = evaluate_c5(best_h)

    return (best_h, best_score, len(best_h))
