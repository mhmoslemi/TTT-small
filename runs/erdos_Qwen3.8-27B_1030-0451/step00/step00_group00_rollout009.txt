from scipy.optimize import minimize

def build_from_p(p):
    m = len(p)
    n = 2 * m
    h = np.empty(n)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return dx * R.max()

def get_p_from_h(h):
    n = len(h)
    m = n // 2
    return h[:m].copy()

def softmax_obj(p, tau):
    m = len(p)
    n = 2 * m
    dx = 1.0 / m
    h = build_from_p(p)
    R = np.correlate(h, 1.0 - h, mode="full")
    max_R = R.max()
    shifted = tau * (R - max_R)
    logsumexp = max_R + np.log(np.exp(shifted).sum()) / tau
    return dx * logsumexp

def softmax_grad(p, tau):
    m = len(p)
    n = 2 * m
    dx = 1.0 / m
    h = build_from_p(p)
    R = np.correlate(h, 1.0 - h, mode="full")
    max_R = R.max()
    shifted = tau * (R - max_R)
    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()
    
    pi = np.concatenate([np.arange(m), np.arange(n-1, m-1, -1)])
    si = np.concatenate([np.ones(m), -np.ones(m)])
    
    grad = np.zeros(m)
    for idx in range(2*n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        s = idx - (n - 1)
        j_min = max(0, -s)
        j_max = min(n - 1 - s, n - 1)
        if j_min > j_max:
            continue
        j_arr = np.arange(j_min, j_max + 1)
        js = j_arr + s
        np.add.at(grad, pi[js], si[js] * (1.0 - h[j_arr]) * w)
        np.add.at(grad, pi[j_arr], -h[js] * si[j_arr] * w)
    
    return dx * grad

def softmax_obj_grad(p, tau):
    m = len(p)

    dx = 1.0 / m

    logsumexp = max_R + np.log(exp_s.sum()) / tau
    obj = dx * logsumexp
    weights = exp_s / exp_s.sum()
    
    pi = np.concatenate([np.arange(m), np.arange(n-1, m-1, -1)])
    si = np.concatenate([np.ones(m), -np.ones(m)])

        np.add.at(grad, pi[j_arr], -h[js] * si[j_arr] * w)
    
    return obj, dx * grad

And dR_s/dp[i] = si(js)*(1-h[j]) for j+s where pi(j+s)==i, minus h[js]*si(j) for j where pi(j)==i.

Let me reorganize. For a fixed i:
- The indices j where pi(j)==i are: j=i and j=n-1-i (if both are in [0,n-1], which they always are for i<m since n-1-i >= n-1-(m-1) = m).

So pi(j)==i only for j=i and j=n-1-i.
Similarly, pi(js)==i only for js=i and js=n-1-i, i.e., j=i-s and j=n-1-i-s.

So dR_s/dp[i] = si(i)*(1-h[i-s])_valid + si(n-1-i)*(1-h[n-1-i-s])_valid - h[i+s]*si(i) - h[n-1-i+s]*si(n-1-i)

where (x)_valid means the contribution is 0 if the index is out of [0,n-1].

si(i) = 1 for i<m (always true)
si(n-1-i) = -1

So dR_s/dp[i] = (1-h[i-s])_valid - (1-h[n-1-i-s])_valid - h[i+s] + h[n-1-i+s]_valid

where the _valid suffix means the term is 0 if the index is out of range.

This gives a much faster way to compute the gradient! For each i and each s, it's just a few lookups. Total: O(m * 2n) = O(n^2) but with a very small constant.

import numpy as np
import time
from scipy.optimize import minimize

def build_from_p(p):
    m = len(p)
    h = np.empty(2 * m)
    h[:m] = p
    hum [m:] = 1.0 - p[::-1]
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(R.max())

def get_p_from_h(h):
    m = len(h) // 2
    return h[:m].copy()

def softmax_obj_grad(p, tau):
    m = len(p)
    n = 2 * m
    dx = 1.0 / m
    h = build_from_p(p)
    R = np.correlate(h, 1.0 - h, mode="full")
    max_R = R.max()
    shifted = tau * (R - max_R)
    exp_s = np.exp(shifted)
    obj = dx * (max_R + float(np.log(exp_s.sum())) / tau)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(m)
    for i in range(m):
        acc = 0.0
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-14:
                continue
            s = idx - (n - 1)
            a = i - s
            b = n - 1 - i - s
            c = i + s
            d = n - 1 - i + s
            if 0 <= a < n:
                acc += wi * (1.0 - h[a])
            if 0 <= b < n:
                acc -= wi * (1.0 - h[b])
            if 0 <= c < n:
                acc -= wi * h[c]
            if 0 <= d < n:
                acc += wi * h[d]
        grad[i] = dx * acc
    
    return obj, grad

def lgbfsb_optimize(p, tau, maxiter=300):
    m = len(p)
    bounds = [(0.0, 1.0)] * m
    
    def fun(p_):
        obj, _ = softmax_obj_grad(p_, tau)
        return obj
    
    def fun_grad(p_):
        _, g = softmax_obj_grad(p_, tau)
        return g
    
    res = minimize(fun_grad, p, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8, 'maxls': 50})
    return res.x

def coord_descent(p, n_passes=100, seed=42, time_limit=None, start_time=None):
    rng = np.random.RandomState(seed)
    m = len(p)
    cur_c5 = compute_c5(build_from_p(p))
    
    delta_sets = [
        [0.01, -0.01, 0.001, -0.001, 0.0001, -0.0001],
        [0.005, -0.005, 0.0005, -0.0005, 0.00005, -0.00005],
        [0.001, -0.001, 0.0001, -0.0001, 0.00001, -0.00001],
        [0.0, 1.0],  # push to binary
    ]
    
    for dset in delta_sets:
        for pass_idx in range(n_passes):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit:
                    return p, cur_c5
            improved = False
            order = rng.permutation(m)
            for i in order:
                best_val = p[i]
                best_obj = cur_c5
                for d in dset:
                    nv = p[i] + d
                    if nv < -0.1 or nv > 1.1:
                        continue
                    nv = np.clip(nv, 0.0, 1.0)
                    if nv == p[i]:
                        continue
                    p[i] = nv
                    c5 = compute_c5(build_from_p(p))
                    if c5 < best_obj:
                        best_obj = c5
                        best_val = nv
                p[i] = best_val
                if best_obj < cur_c5:
                    cur_c5 = best_obj
                    improved = True
            if not improved:
                break
    
    return p, cur_c5

from scipy.optimize import minimize

def build_from_p(p):
    m = len(p)
    h = np.empty(2 * m)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(R.max())

def get_p_from_h(h):
    m = len(h) // 2
    return h[:m].copy()

def softmax_obj_grad(p, tau):
    m = len(p)
    n = 2 * m
    dx = 1.0 / m
    h = build_from_p(p)
    R = np.correlate(h, 1.0 - h, mode="full")
    max_R = float(R.max())
    shifted = tau * (R - max_R)
    exp_s = np.exp(shifted)
    s = float(exp_s.sum())
    obj = dx * (max_R + np.log(s) / tau)
    w = exp_s / s
    
    grad = np.zeros(m)
    for i in range(m):
        acc = 0.0
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-14:
                continue
            lag = idx - (n - 1)
            a = i - lag
            if 0 <= a < n:
                acc += wi * (1.0 - h[a])
            b = (n - 1 - i) - lag
            if 0 <= b < n:
                acc -= wi * (1.0 - h[b])
            c = i + lag
            if 0 <= c < n:
                acc -= wi * h[c]
            d = (n - 1 - i) + lag
            if 0 <= d < n:
                acc += wi * h[d]
        grad[i] = dx * acc
    
    return obj, grad
