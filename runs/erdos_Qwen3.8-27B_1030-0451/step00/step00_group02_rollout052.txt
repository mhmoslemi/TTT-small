def fw_step(h, n, tau, seed=None):
    """Frank-Wolfe step."""
    rng = np.random.default_rng(seed) if seed is not None else np.random.default_rng()
    S = n / 2.0
    cur_score = exact_score(h)
    
    g = smooth_gradient(h, tau)
    order = np.argsort(g)
    u = np.zeros(n)
    num_full = int(np.floor(S))
    frac = S - num_full
    u[order[:num_full]] = 1.0
    if frac > 1e-10:
        u[order[num_full]] = frac
    u = project_feasible(u, n, S)
    
    best_h = h.copy()
    best_sc = cur_score
    for gamma in [0.9, 0.6, 0.4, 0.2, 0.1, 0.05, 0.01]:
        new_h = (1 - gamma) * h + gamma * u
        new_h = project_feasible(new_h, n, S)
        sc = exact_score(new_h)
        if sc < best_sc - 1e-14:
            best_h = new_h
            best_sc = sc
    return best_h, best_sc

def smooth_optimize(h, n, t_start, time_limit):

def tau_sequence(score):
    """Generate decreasing tau sequence."""
    tau0 = min(0.05, max(2e-4, 0.1 * score))
    taus = []
    t = tau0
    while t > 1e-5:
        taus.append(t)
        t *= 0.4
    taus.append(1e-5)
    return taus

def interpolate_to_dim(h, new_n):

            continue
    return best_h, best_sc


def fw_step(h, n, tau, seed=0):
    rng = np.random.default_rng(seed)
    S = n / 2.0
    cur_score = exact_score(h)
    g = SmoothGrad(n, tau)(h)
    order = np.argsort(g)
    u = np.zeros(n)
    num_full = int(math.floor(S))
    frac = S - num_full
    u[order[:num_full]] = 1.0
    if frac > 1e-10 and num_full < n:
        u[order[num_full]] = frac
    u = project_feasible(u, n, S)
    best_h = h.copy()
    best_sc = cur_score
    for gamma in [0.8, 0.5, 0.3, 0.15, 0.05, 0.01]:
        if time.time() - t0_global > deadline_global:
            break
        new_h = (1.0 - gamma) * h + gamma * u
        new_h = project_feasible(new_h, n, S)
        sc = exact_score(new_h)
        if sc < best_sc - 1e-14:
            best_h = new_h
            best_sc = sc
    return best_h, best_sc


def pair_descend(h, n, seed, max_iter=2000, time_limit=60):
    global t0_global, deadline_global
    rng = np.random.default_rng(seed)
    S = n / 2.0
    best = h.copy()
    best_score = exact_score(best)
    no_improve = 0
    tau = min(max(1e-5, 0.015 * best_score), 0.04)
    grad_fn = SmoothGrad(n, tau)

    for it in range(max_iter):
        if time.time() > deadline_global:
            break
        if time.time() - (t0_global) > max_iter * 1000:
            break
        if no_improve > 500:
            break

        g = grad_fn(best)
        gi = int(np.argmin(g))
        gj = int(np.argmax(g))

        if gi != gj and abs(g[gi] - g[gj]) > 1e-12:
            max_d = min(1.0 - best[gi], best[gj])
            if max_d > 1e-12:
                improved = False
                for frac in [1.0, 0.7, 0.4, 0.2, 0.08, 0.02, 0.005]:
                    d = frac * max_d
                    new_h = best.copy()
                    new_h[gi] += d
                    new_h[gj] -= d
                    new_h = project_feasible(new_h, n, S)
                    sc = exact_score(new_h)
                    if sc < best_score - 1e-14:
                        best = new_h
                        best_score = sc
                        no_improve = 0
                        improved = True
                        break
                if improved:
                    continue

        found = False
        for _ in range(8):
            if time.time() > deadline_global:
                break
            ri = int(rng.integers(0, n))
            rj = int(rng.integers(0, n))
            if ri == rj:
                continue
            for (a, b) in [(ri, rj), (rj, ri)]:
                max_d2 = min(1.0 - best[a], best[b])
                if max_d2 < 1e-12:
                    continue
                d = float(rng.random()) * max_d2
                new_h = best.copy()
                new_h[a] += d
                new_h[b] -= d
                new_h = project_feasible(new_h, n, S)
                sc = exact_score(new_h)
                if sc < best_score - 1e-14:
                    best = new_h
                    best_score = sc
                    no_improve = 0
                    found = True
                    break
            if found:
                break

        if not found:
            no_improve += 1
            if no_improve % 80 == 0:
                best, best_score = fw_step(best, n, tau, seed=seed + it)
                if no_improve > 1000:
                    break

    return best, best_score


def make_inits(n, seed, count=12):
    rng = np.random.default_rng(seed)
    S = n / 2.0
    cands = []
    z = np.full(n, 0.5)
    cands.append(z.copy())
    for _ in range(count):
        z = rng.random(n)
        cands.append(project_feasible(z, n, S))
    for _ in range(count):
        z = (rng.random(n) < 0.5).astype(float)
        cands.append(project_feasible(z, n, S))
    for period in [3, 4, 5, 7]:
        pattern = np.zeros(period)
        num_ones = min(period, int(round(S / n * period)))
        pattern[:num_ones] = 1.0
        h = np.tile(pattern, n // period + 1)[:n]
        cands.append(project_feasible(h, n, S))
    return cands


def interpolate_to_dim(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    S = new_n / 2.0
    return project_feasible(h_new, new_n, S)


def improve_candidate(h_cand, best_state, n, t_start, time_limit, seed, do_smooth=True):
    sc = exact_score(h_cand)
    if sc >= best_state['score'] + 1e-8:
        return
    h_opt, sc_opt = pair_descend(h_cand, n, seed=seed, max_iter=1200, time_limit=min(40, time_limit))
    if sc_opt < best_state['score'] - 1e-12:
        best_state['h'] = h_opt
        best_state['score'] = sc_opt
        best_state['n'] = n
    if do_smooth and time.time() - t_start < time_limit * 0.7:
        h_sm, sc_sm = smooth_optimize(h_opt, n, t_start=time.time(),
                                      time_limit=min(60, time_limit - (time.time() - t_start)),
                                      seed=seed + 1)
        if sc_sm < best_state['score'] - 1e-12:
            best_state['h'] = h_sm
            best_state['score'] = sc_sm
            best_state['n'] = n
            if time.time() - t_start < time_limit:
                h_p2, sc_p2 = pair_descend(h_sm, n, seed=seed + 2,
                                           max_iter=800, time_limit=30)
                if sc_p2 < best_state['score'] - 1e-12:
                    best_state['h'] = h_p2
                    best_state['score'] = sc_p2
                    best_state['n'] = n


t0_global = 0
deadline_global = 0


def run(seed=42, budget_s=1100, **kwargs):
    global t0_global, deadline_global
    t0_global = time.time()
    deadline_global = t0_global + max(30, budget_s - 8.0)

    # Get parent construction
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=float)
        if pv.ndim == 1 and len(pv) >= 4 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except (NameError, TypeError, AttributeError):
        parent = None

    if parent is not None:
        n_parent = len(parent)
        S = n_parent / 2.0
        parent = np.clip(parent, 0.0, 1.0)
        parent = project_feasible(parent, n_parent, S)
        parent_score = exact_score(parent)
    else:
        n_parent = 81
        parent = np.full(n_parent, 0.5)
        parent_score = exact_score(parent)

    best_state = {'h': parent.copy(), 'score': parent_score, 'n': n_parent}

    # Generate candidates
    all_cands = [parent.copy()]
    comp = project_feasible(1.0 - parent, n_parent, n_parent / 2.0)
    all_cands.append(comp)
    inits = make_inits(n_parent, seed, count=8)
    all_cands.extend(inits)

    # Score and sort
    scored = []
    for c in all_cands:
        sc = exact_score(c)
        scored.append((sc, c))
    scored.sort(key=lambda x: x[0])

    # Phase 1: Local optimize top candidates
    top_k = min(6, len(scored))
    for i in range(top_k):
        if time.time() > deadline_global - 60:
            break
        sc, h_c = scored[i]
        if sc > best_state['score'] + 0.05:
            continue
        remain = deadline_global - time.time() - 30
        if remain < 10:
            break
        improve_candidate(h_c, best_state, n_parent, t0_global, remain, seed=seed + i * 777, do_smooth=(i < 3))

    # Phase 2: Try complement transform of best
    if time.time() < deadline_global - 60:
        comp_best = project_feasible(1.0 - best_state['h'], best_state['n'], best_state['n'] / 2.0)
        improve_candidate(comp_best, best_state, best_state['n'], t0_global, 40, seed=seed + 500, do_smooth=False)

    # Phase 3: Try different dimensions
    if time.time() < deadline_global - 40:
        for factor in [1.5, 2.0]:
            if time.time() > deadline_global - 30:
                break
            new_n = int(round(n_parent * factor))
            new_n = min(new_n, 200)
            if new_n <= 20 or new_n == best_state['n']:
                continue
            h_interp = interpolate_to_dim(best_state['h'], new_n)
            improve_candidate(h_interp, best_state, new_n, t0_global, 40, seed=seed + 1000 + new_n, do_smooth=True)

        # Also try smaller dimensions
        for factor in [0.75]:
            if time.time() > deadline_global - 20:
                break
            new_n = int(round(n_parent * factor))
            new_n = max(new_n, 30)
            if new_n == best_state['n']:
                continue
            h_interp = interpolate_to_dim(best_state['h'], new_n)
            improve_candidate(h_interp, best_state, new_n, t0_global, 30, seed=seed + 2000, do_smooth=False)

    # Phase 4: Final refinement
    if time.time() < deadline_global - 10:
        remain = deadline_global - time.time() - 5
        h_final, sc_final = pair_descend(best_state['h'], best_state['n'],
                                         seed=seed + 99999, max_iter=5000, time_limit=remain)
        if sc_final < best_state['score'] - 1e-13:
            best_state['h'] = h_final
            best_state['score'] = sc_final

    # Final validation and return
    bh = best_state['h']
    bn = best_state['n']
    S = bn / 2.0
    bh = project_feasible(bh, bn, S)
    final_score = exact_score(bh)
    assert np.all(np.isfinite(bh))
    assert bh.shape == (bn,)
    assert abs(float(np.sum(bh)) - S) < 1e-8, f"Sum={np.sum(bh)}, S={S}"
    assert np.all(bh >= -1e-12) and np.all(bh <= 1.0 + 1e-12)

    return bh, final_score, bn
