    ell = q - (n - 1)
    g = np.zeros(n)
    if ell == 0:

    elif ell > 0:

        g[0:L] += (1.0 - h[ell:n])
        g[ell:n] -= h[0:L]
    else:

        L = n - a
        g[a:n] += (1.0 - h[0:L])
        g[0:L] -= h[a:n]

def project_box_sum(h_raw, T):

    h = np.clip(h_raw, 0.0, 1.0)

    lo, hi = -1.0, 1.0
    for _ in range(100):

        if s < T:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Repair residual
    err = T - np.sum(h_out)
    if abs(err) > 1e-12:

def slp_step(h, rho, K, max_t):
    n = len(h)
    z_vec = exact_z_vector(h)
    z_max = z_vec[np.argmax(z_vec)]
    
    # Select active constraints
    gap = max(1e-4, 0.01 * z_max)
    active_mask = z_vec >= (z_max - gap)
    active_idx = np.where(active_mask)[0]
    
    # Cap to K
    if len(active_idx) > K:
        # Take top K
        order = np.argsort(-z_vec[active_idx])
        active_idx = active_idx[order[:K]]
    
    # Ensure argmax is included
    argmax_q = np.argmax(z_vec)
    if argmax_q not in active_idx:
        active_idx = np.concatenate([active_idx, [argmax_q]])
    
    # Also include center (q = n-1, which is lag 0)
    center_q = n - 1
    if center_q not in active_idx:
        active_idx = np.append(active_idx, center_q)
    
    # Build LP
    # Variables: d[0..n-1], t (index n)
    # min t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: g_q · d - t <= -z_q for each active q
    A_ub = np.zeros((len(active_idx), n + 1))
    b_ub = np.zeros(len(active_idx))
    for i, q in enumerate(active_idx):
        gq = grad_z(h, q)
        A_ub[i, 0:n] = gq
        A_ub[i, n] = -1.0
        b_ub[i] = -z_vec[q]  # z_q(h) is the exact value
    
    # Equality: sum d = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = [0.0]
    
    # Bounds
    lb = np.maximum(-h, -rho)
    ub = np.minimum(1.0 - h, rho)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, max_t))
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:n]
        y = h + d
        y = np.clip(y, 0.0, 1.0)
        return y
    return None

    T = n / 2.0

    best_score = current_score

def pair_transfer_polish(h, time_limit, time_start, max_accepted=50):
    """Pairwise transfer local search."""
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    dx = 2.0 / n
    T = n / 2.0
    
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    accepted = 0
    
    for it in range(200):
        if accepted >= max_accepted:
            break
        if time.time() - time_start > time_limit - 20:
            break
        
        z_vec = exact_z_vector(h)
        z_max = z_vec[np.argmax(z_vec)]
        
        # Weighted gradient from active constraints
        gap = max(1e-5, 0.02 * z_max)
        active_mask = z_vec >= (z_max - gap)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > 50:
            order = np.argsort(-z_vec[active_idx])
            active_idx = active_idx[order[:50]]
        
        # Compute mixed gradient
        g_mix = np.zeros(n)
        for q in active_idx:
            g_mix += grad_z(h, q)
        g_mix /= len(active_idx)
        
        # Find transfer: move from high gradient to low gradient
        g_pos = np.where(g_mix > 0.01)[0]
        g_neg = np.where(g_mix < -0.01)[0]
        
        if len(g_pos) == 0 or len(g_neg) == 0:
            # Try any pair with gradient difference
            if n >= 2:
                i = int(np.argmax(g_mix))
                j = int(np.argmin(g_mix))
                if g_mix[i] - g_mix[j] < 0.01:
                    break
                g_pos = np.array([i])
                g_neg = np.array([j])
            else:
                break
        
        # Pick best pair
        best_pair = None
        best_improvement = 0
        # Sample pairs
        n_sample = min(20, len(g_pos))
        m_sample = min(20, len(g_neg))
        for ii in range(n_sample):
            i = g_pos[ii]
            for jj in range(m_sample):
                j = g_neg[jj]
                if i == j:
                    continue
                # Transfer from i to j
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta < 1e-8:
                    continue
                for frac in [0.1, 0.3, 0.5, 0.8, 1.0]:
                    delta = max_delta * frac
                    if delta < 1e-10:
                        continue
                    h_candidate = h.copy()
                    h_candidate[i] -= delta
                    h_candidate[j] += delta
                    if h_candidate[i] < -1e-12 or h_candidate[j] > 1 + 1e-12:
                        continue
                    sc = exact_score(h_candidate)
                    if sc < current_score - best_improvement:
                        best_improvement = current_score - sc
                        best_pair = (i, j, delta)
        
        if best_pair is not None:
            i, j, delta = best_pair
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, T)
            current_score = exact_score(h)
            if current_score < best_score - 1e-12:
                best_score = current_score
                best_h = h.copy()
            accepted += 1
        else:
            break
    
    return best_h, best_score
