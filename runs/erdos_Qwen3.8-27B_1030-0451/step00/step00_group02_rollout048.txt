    exp_shifted = np.exp(tau * (corr - max_c))
    return exp_shifted / np.sum(exp_shifted)

def _gradient_smoothed(h, tau):

    for d in range(2 * n - 1):

        lag = d - (n - 1)

            # grad[j] -= w*h[j+lag] for j in [0, n-1-lag]
