import numpy as np
import time
from scipy.optimize import minimize

def exact_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return np.max(corr) * dx

def project_feasible(z, target_sum):
    """Project z onto {0 <= x <= 1, sum(x) = target_sum}."""
    n = len(z)
    lo = -2.0 - np.max(np.abs(z))
    hi = 2.0 + np.max(np.abs(z))
    for _ in range(80):
        mu = 0.5 * (lo + hi)
        y = np.clip(z - mu, 0.0, 1.0)
        s = np.sum(y)
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    y = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Fix tiny sum error
    diff = target_sum - np.sum(y)
    if abs(diff) > 1e-12:
        # Distribute over interior points
        interior = (y > 1e-10) & (y < 1.0 - 1e-10)
        if np.any(interior):
            y[interior] += diff / np.sum(interior)
            y = np.clip(y, 0.0, 1.0)
    return y

def smooth_objective(h, dx, tau):
    """Log-sum-exp smooth max of correlation values."""
    n = len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    vals = corr * dx
    m = vals.max()
    if not np.isfinite(m):
        return 1e10
    return m + tau * np.log(np.sum(np.exp((vals - m) / tau)))

def smooth_gradient(h, dx, tau):
    """Gradient of smooth objective."""
    n = len(h)
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    vals = corr * dx
    m = vals.max()
    exp_vals = np.exp((vals - m) / tau)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        ws = w[k] * dx
        if abs(ws) < 1e-15:
            continue
        n_start = max(0, -s)
        n_end = min(n, n - s)
        # First: positions j = n+s
        grad[n_start + s:n_end + s] += ws * (1.0 - h[n_start:n_end])
        # Second: positions j = n
        grad[n_start:n_end] += ws * (-h[n_start + s:n_end + s])
    return grad

def fw_extreme_point(grad, target_sum):
    """Find extreme point of polytope minimizing grad · x."""
    n = len(grad)
    order = np.argsort(grad)
    u = np.zeros(n)
    S = target_sum
    full = int(np.floor(S))
    frac = S - full
    u[order[:full]] = 1.0
    if frac > 1e-10:
        u[order[full]] = frac
    return u

def local_search(h, dx, target_sum, max_iters=5000, time_limit=5.0, seed=42):
    """Combined local search: paired moves + FW."""
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    t_start = time.time()
    
    patience = 0
    max_patience = 200
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if patience > max_patience:
            break
        
        # Occasionally try FW step
        if it % 50 == 0 and cur_c5 > 0.37:
            tau = max(0.002, 0.05 * cur_c5)
            g = smooth_gradient(h, dx, tau)
            u = fw_extreme_point(g, target_sum)
            # Line search
            for gamma in [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]:
                candidate = (1 - gamma) * h + gamma * u
                candidate = project_feasible(candidate, target_sum)
                new_c5 = exact_c5(candidate)
                if new_c5 < cur_c5 - 1e-12:
                    h = candidate
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    patience = 0
                    break
        
        # Paired move
        # Choose indices to modify
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        
        # Try to increase h[a] and decrease h[b], or vice versa
        # Use gradient to decide direction
        # Simple: try both directions
        improved = False
        for direction in [1, -1]:
            if direction == 1:
                # h[a] += d, h[b] -= d
                max_d = min(1.0 - h[a], h[b])
            else:
                # h[a] -= d, h[b] += d
                max_d = min(h[a], 1.0 - h[b])
            
            if max_d < 1e-10:
                continue
            
            # Try several step sizes
            for frac in [0.5, 0.2, 0.1, 0.05, 0.3, 0.01]:
                d = frac * max_d
                if d < 1e-10:
                    continue
                candidate = h.copy()
                if direction == 1:
                    candidate[a] += d
                    candidate[b] -= d
                else:
                    candidate[a] -= d
                    candidate[b] += d
                # Check feasibility (should be guaranteed by construction)
                if candidate.min() < -1e-10 or candidate.max() > 1.0 + 1e-10:
                    continue
                new_c5 = exact_c5(candidate)
                if new_c5 < cur_c5 - 1e-12:
                    h = candidate
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    patience = 0
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            patience += 1
    
    return best_h, best_c5

def slsqp_smooth(h0, dx, target_sum, tau_init=0.05, time_limit=10.0, seed=42):
    """SLSQP optimization of smooth surrogate."""
    n = len(h0)
    h0 = project_feasible(h0.copy(), target_sum)
    t_start = time.time()
    
    def obj(x):
        if time.time() - t_start > time_limit:
            return 1e10
        return smooth_objective(x, dx, tau_init)
    
    def obj_grad(x):
        if time.time() - t_start > time_limit:
            return np.zeros(n)
        return smooth_gradient(x, dx, tau_init)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_res = project_feasible(res.x, target_sum)
            c5_res = exact_c5(h_res)
            return h_res, c5_res
    except Exception:
        pass
    return h0, exact_c5(h0)

def interpolate_h(h_old, n_new):
    """Interpolate h from old resolution to new resolution."""

    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

import numpy as np

from scipy.optimize import minimize


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(z, target_sum):
    n = len(z)
    lo = -3.0 - np.max(np.abs(z))
    hi = 3.0 + np.max(np.abs(z))
    for _ in range(100):
        mu = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mu, 0.0, 1.0))
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(y)
    if abs(diff) > 1e-14:
        interior = (y > 1e-12) & (y < 1.0 - 1e-12)
        if np.any(interior):
            y[interior] += diff / np.sum(interior)
            y = np.clip(y, 0.0, 1.0)
    return y


def smooth_objective(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    if not np.isfinite(m):
        return 1e10
    return m + tau * np.log(np.sum(np.exp((vals - m) / tau)))


def smooth_gradient(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    exp_vals = np.exp((vals - m) / tau)
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        ws = w[k] * dx
        if abs(ws) < 1e-15:
            continue
        n_start = max(0, -s)
        n_end = min(n, n - s)
        if n_end <= n_start:
            continue
        grad[n_start + s:n_end + s] += ws * (1.0 - h[n_start:n_end])
        grad[n_start:n_end] += ws * (-h[n_start + s:n_end + s])
    return grad

        candidates.append(make_thresholded(best_h, target))
        candidates.append(project_feasible(best_h + rng.randn(n) * 0.1, target))
        candidates.append(project_feasible(best_h + rng.randn(n) * 0.25, target))
        candidates.append(project_feasible(rng.rand(n), target))
        candidates.append(make_periodic(n, target, 3, 0.0))
        candidates.append(make_periodic(n, target, 5, 0.3))
        candidates.append(make_periodic(n, target, 7, 0.7))

        # Score all candidates
        scored = []
        for c in candidates:
            c5_c = exact_c5(c)
            scored.append((c5_c, c))
        scored.sort(key=lambda x: x[0])

        # Local search on top candidates
        n_ls = min(len(scored), 6)
        ls_budget = min(40.0, time_left() / (n_ls + 1))
        for i in range(n_ls):
            if time_left() < 30:
                break
            c5_c, h_c = scored[i]
            h_res, c5_res = local_search(h_c, dx, target,
                                         max_iters=4000, time_limit=ls_budget,
                                         seed=seed + i * 7)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n

    # Phase 2: SLSQP smooth optimization (multiple tau values)
    if time_left() > 100:
        for tau in [0.04, 0.015, 0.005, 0.002]:
            if time_left() < 30:
                break
            sl_time = min(25.0, time_left() / 4)
            h_sl, c5_sl = slsqp_smooth(best_h, dx, target, tau, time_limit=sl_time)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
            # Follow with local search
            if time_left() > 40:
                h_ls, c5_ls = local_search(h_sl, dx, target,
                                           max_iters=3000, time_limit=min(20.0, time_left() / 3),
                                           seed=seed + 200)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

    # Phase 3: Try higher resolution
    if time_left() > 100:
        for n_new in [162, 200]:
            if time_left() < 60:
                break
            h_interp = interpolate_h(best_h, n_new)
            target_new = n_new / 2.0
            dx_new = 2.0 / n_new
            h_interp = project_feasible(h_interp, target_new)
            c5_interp = exact_c5(h_interp)

            ls_time = min(40.0, time_left() / 3)
            h_res, c5_res = local_search(h_interp, dx_new, target_new,
                                         max_iters=3000, time_limit=ls_time,
                                         seed=seed + 500)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_new

            if time_left() > 50:
                h_sl, c5_sl = slsqp_smooth(h_res, dx_new, target_new,
                                           tau=0.01, time_limit=min(20.0, time_left() / 2))
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl.copy()
                    best_n = n_new

                if time_left() > 30:
                    h_ls2, c5_ls2 = local_search(h_sl, dx_new, target_new,
                                                 max_iters=2000, time_limit=min(15.0, time_left() / 2),
                                                 seed=seed + 600)
                    if c5_ls2 < best_c5:
                        best_c5 = c5_ls2
                        best_h = h_ls2.copy()
                        best_n = n_new

        # Try to downsample if at higher resolution
        if best_n > 81 and time_left() > 40:
            for n_down in [81, 100]:
                h_down = interpolate_h(best_h, n_down)
                target_down = n_down / 2.0
                dx_down = 2.0 / n_down
                h_down = project_feasible(h_down, target_down)
                c5_down = exact_c5(h_down)
                if c5_down < best_c5:
                    best_c5 = c5_down
                    best_h = h_down.copy()
                    best_n = n_down
                else:
                    # Try local search at downsampled resolution
                    h_ls, c5_ls = local_search(h_down, dx_down, target_down,
                                               max_iters=1500, time_limit=min(15.0, time_left() / 2),
                                               seed=seed + 700)
                    if c5_ls < best_c5:
                        best_c5 = c5_ls
                        best_h = h_ls.copy()
                        best_n = n_down

    # Phase 4: Final intensive refinement
    if time_left() > 20:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        rem = max(5.0, time_left() - 3.0)
        h_res, c5_res = local_search(best_h, dx, target,
                                     max_iters=20000, time_limit=rem,
                                     seed=seed + 999)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Final validation and return
    best_h = project_feasible(best_h, float(best_n) / 2.0)
    final_c5 = exact_c5(best_h)

    return best_h, final_c5, best_n
