import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def project_box_sum(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(c, T):
    x = T * c
    x = x - x.max()
    return x.max() / T + np.log(np.sum(np.exp(x))) / T

def smooth_max_grad(c, T):
    x = T * c
    x = x - x.max()
    w = np.exp(x) / np.sum(np.exp(x))
    return w  # gradient w.r.t. c

def run(seed=42, budget_s=1100, **kwargs):

    target = n / 2.0

            # Gradient of f w.r.t. h
            w = smooth_max_grad(c, T)  # weights for each shift

                s2 = max(0, -k)

    w = np.exp(xc_s) / np.sum(np.exp(xc_s))
    
    grad = np.zeros(n)
    omh = 1.0 - h
    k_vals = np.arange(2 * n - 1) - (n - 1)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        k = k_vals[j]
        s1 = max(0, k)
        e1 = min(n, k + n)
        if e1 > s1:
            grad[s1:e1] += wj * omh[s1 - k:e1 - k]
        s2 = max(0, -k)
        e2 = min(n, n - k)
        if e2 > s2:
            grad[s2:e2] -= wj * h[s2 + k:e2 + k]
    return grad * dx
