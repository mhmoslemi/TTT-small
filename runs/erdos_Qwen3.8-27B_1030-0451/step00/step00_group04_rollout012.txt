def feasible_project(x, n):
    """Project x onto the feasible set: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(x.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = np.sum(h) - target
    if abs(diff) < 1e-10:
        return h
    # Adjust: if sum too high, reduce from entries near 1; if too low, increase from entries near 0
    if diff > 0:
        # Need to reduce by diff
        # Best to reduce entries closest to 1
        order = np.argsort(h)[::-1]  # descending
        for idx in order:
            reduce = min(diff, h[idx])
            h[idx] -= reduce
            diff -= reduce
            if diff <= 1e-12:

    else:

        order = np.argsort(h)  # ascending
        need = -diff
        for idx in order:
            increase = min(need, 1.0 - h[idx])
            h[idx] += increase
            need -= increase
            if need <= 1e-12:

    return h

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full')
    return np.max(z) * dx

def project_feasible(x, n):
    h = np.clip(x, 0.0, 1.0).copy()
    target = n / 2.0
    diff = np.sum(h) - target
    if diff > 0:
        order = np.argsort(h)[::-1]
        for idx in order:
            r = min(diff, h[idx])
            h[idx] -= r
            diff -= r
            if diff < 1e-12:
                break
    elif diff < 0:
        order = np.argsort(h)
        need = -diff
        for idx in order:
            r = min(need, 1.0 - h[idx])
            h[idx] += r
            need -= r
            if need < 1e-12:
                break
    return h

def weighted_gradient(h, active_lags, weights):
    """Compute weighted gradient of active correlations."""
    n = len(h)
    g = np.zeros(n)
    for lag, w in zip(active_lags, weights):
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            # corr = sum_{m=0}^{n-1-lag} h[m]*(1-h[m+lag])
            # d/dh[m] = (1-h[m+lag]) for m in [0, n-1-lag]
            #           - h[m-lag] for m in [lag, n-1]
            idx1 = np.arange(0, n - lag)
            g[idx1] += w * (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            g[idx2] -= w * h[idx2 - lag]
        else:
            # lag < 0, let s = -lag > 0
            s = -lag
            # corr = sum_{m=s}^{n-1} h[m]*(1-h[m-s])
            # d/dh[m] = (1-h[m-s]) for m in [s, n-1]
            #           - h[m+s] for m in [0, n-1-s]
            idx1 = np.arange(s, n)
            g[idx1] += w * (1.0 - h[idx1 - s])
            idx2 = np.arange(0, n - s)
            g[idx2] -= w * h[idx2 + s]
    return g

def find_active_lags(z, n_active=10, threshold=None):
    """Find lags that are near the maximum."""
    max_z = np.max(z)
    if threshold is None:
        threshold = max_z - max_z * 0.05  # within 5% of max
    active = np.where(z >= threshold)[0]
    # Convert to lag indices: lag = k - (n-1)
    # But we just need the indices in z
    if len(active) > n_active:
        # Keep only the top n_active
        order = np.argsort(z[active])[::-1][:n_active]
        active = active[order]
    return active, z[active]

def line_search_pair(h, i, j, z_current, max_improvement):
    """Try pair transfer (i,j), return best (h_new, new_score) or None."""
    n = len(h)
    tau_lo = max(-h[i], h[j] - 1.0)
    tau_hi = min(1.0 - h[i], h[j])
    if tau_hi - tau_lo < 1e-10:
        return None, None
    
    best_score = exact_score(h)
    best_tau = 0.0
    
    # Sample
    n_probes = 16
    taus = np.linspace(tau_lo, tau_hi, n_probes)
    scores = np.zeros(n_probes)
    for k, tau in enumerate(taus):
        h_new = h.copy()
        h_new[i] += tau
        h_new[j] -= tau
        scores[k] = exact_score(h_new)
    
    best_idx = np.argmin(scores)
    if scores[best_idx] >= best_score - 1e-12:
        return None, None
    
    # Refine with golden section
    lo = taus[max(0, best_idx-1)]
    hi = taus[min(n_probes-1, best_idx+1)]
    for _ in range(10):
        m1 = lo + 0.382 * (hi - lo)
        m2 = lo + 0.618 * (hi - lo)
        h1 = h.copy(); h1[i] += m1; h1[j] -= m1
        h2 = h.copy(); h2[i] += m2; h2[j] -= m2
        s1 = exact_score(h1)
        s2 = exact_score(h2)
        if s1 < s2:
            hi = m2
        else:
            lo = m1
    
    tau_best = (lo + hi) / 2
    h_new = h.copy()
    h_new[i] += tau_best
    h_new[j] -= tau_best
    s_new = exact_score(h_new)
    
    if s_new < best_score - 1e-12:
        return h_new, s_new
    return None, None

    n_starts = len(scored)
    time_per_start = (budget_s * 0.80) / n_starts
    time_per_start = min(time_per_start, 80.0)

    for idx, (s, hs) in enumerate(scored):
        elapsed = time.time() - start_time_global
        remaining = (start_time_global + budget_s) - time.time() - 15.0
        if remaining < 10:
            break
        t_budget = min(time_per_start, remaining)
        dl = time.time() + t_budget
        result_h, result_s = local_search(hs, max_iters=20000, deadline=dl)
        if result_s < best_score:
            best_score = result_s
            best_h = result_h.copy()
        if s < best_score:
            best_score = s
            best_h = hs.copy()

    # Second pass: more iterations on the best start
    remaining = (start_time_global + budget_s) - time.time() - 10.0
    if remaining > 30:
        result_h, result_s = local_search(best_h, max_iters=50000, deadline=time.time() + remaining)
        if result_s < best_score:
            best_score = result_s
            best_h = result_h.copy()

    # Final cleanup
    best_h = project_feasible(best_h, n)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_score = exact_score(best_h)

    return best_h, final_score, n
