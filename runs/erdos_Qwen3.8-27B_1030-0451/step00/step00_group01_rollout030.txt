import numpy as np
import time
from scipy.optimize import minimize


def project_sum_box(x, target):
    """Project x onto {y: 0<=y_i<=1, sum(y)=target}."""
    lo = -x.max() - 0.1
    hi = 1.0 - x.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(x + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            m = y < 1.0
            if m.any():
                y[m] += r * (1.0 - y[m]) / (1.0 - y[m]).sum()
        else:
            m = y > 0.0
            if m.any():
                y[m] += r * y[m] / y[m].sum()
    return np.clip(y, 0.0, 1.0)


def eval_c5(h):
    """Compute C5 = max of full correlation of h with 1-h, scaled by dx."""
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def lse_objective(x, beta):
    """Log-sum-exp smoothed objective."""
    dx = 2.0 / len(x)
    vals = np.correlate(x, 1.0 - x, mode="full") * dx
    m = vals.max()
    e = np.exp(np.minimum(beta * (vals - m), 0.0))
    return m + np.log(e.sum()) / beta


def lse_gradient(x, beta):
    """Gradient of log-sum-exp smoothed objective."""
    n = len(x)
    dx = 2.0 / n
    vals = np.correlate(x, 1.0 - x, mode="full") * dx
    m = vals.max()
    e = np.exp(np.minimum(beta * (vals - m), 0.0))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            i = np.arange(n - lag)
            g[i] += wk * dx * (1.0 - x[i + lag])
            g[i + lag] -= wk * dx * x[i]
        else:
            ll = -lag
            i = np.arange(ll, n)
            g[i] += wk * dx * (1.0 - x[i - ll])
            i2 = np.arange(n - ll)
            g[i2] -= wk * dx * x[i2 + ll]
    return g
