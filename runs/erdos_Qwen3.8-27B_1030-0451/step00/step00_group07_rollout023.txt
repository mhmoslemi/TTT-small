import numpy as np
import time
import math
from scipy.optimize import linprog

def mass_project(x, target):
    """Project x onto {0 <= x_i <= 1, sum(x) = target}"""
    N = len(x)
    x = np.array(x, dtype=np.float64).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0, 1)
    
    # Bisect on lambda: sum(clip(x + lambda, 0, 1)) = target
    lo = -1.0 - np.max(x)  # F(lo) = 0
    hi = 1.0 - np.min(x)   # F(hi) = N
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x + lam, 0, 1)
    
    # Fix residual
    residual = math.fsum(result) - target
    if abs(residual) > 1e-12:
        if residual > 0:
            # Too much mass, remove from largest values
            order = np.argsort(-result)
            for i in order:
                adj = min(residual, result[i])
                result[i] -= adj
                residual -= adj
                if abs(residual) < 1e-14:
                    break
        else:
            # Too little mass, add to smallest values
            order = np.argsort(result)
            for i in order:
                adj = min(-residual, 1.0 - result[i])
                result[i] += adj
                residual += adj
                if abs(residual) < 1e-14:
                    break
        result = np.clip(result, 0, 1)
    
    return result

    N = len(h)
    dx = 2.0 / N

    N = len(h)
    dx = 2.0 / N

    """Compute gradient of E_s with respect to h."""
    N = len(h)
    grad = np.zeros(N)
    if s == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif s > 0:
        # E_s = sum_{n=0}^{N-s-1} h[n+s] * (1-h[n])
        # d/dh[i]: (1-h[i-s]) if 0<=i-s<N (i.e., s<=i<N) + (-h[i+s]) if i+s<N (i.e., i<N-s)
        for n in range(N - s):

            i = n + s
            grad[i] += dx * (1.0 - h[n])

            grad[n] += dx * (-h[n + s])

            # (1-h[n]) term: derivative w.r.t h[n] is -h[n-t]
            grad[n] += dx * (-h[n - t])
    return grad

import time
import math
from scipy.optimize import linprog

def mass_project(x, target):
    N = len(x)
    x = np.asarray(x, dtype=np.float64).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    
    # Bisection on lambda
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(x + lam, 0.0, 1.0)
    
    # Fix residual
    resid = math.fsum(result) - target
    if abs(resid) > 1e-13:
        if resid > 0:
            order = np.argsort(-result)
            for i in order:
                adj = min(resid, result[i])
                result[i] -= adj
                resid -= adj
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(result)
            for i in order:
                adj = min(-resid, 1.0 - result[i])
                result[i] += adj
                resid += adj
                if abs(resid) < 1e-15:
                    break
        result = np.clip(result, 0.0, 1.0)
    return result

def evaluate_exact(h):
    N = len(h)
    if N == 0:
        return float('inf')
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def objective_array(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_all_gradients(h, dx, active_shifts):
    """Compute gradients for given shifts. active_shifts is list of shift values."""
    N = len(h)
    grads = []
    for s in active_shifts:
        grad = np.zeros(N)
        if s == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        elif s > 0:
            # E_s = sum_{n=0}^{N-s-1} h[n+s] * (1-h[n])
            # d/dh[i] = dx*(1-h[i-s]) for i in [s, N-1] + dx*(-h[i+s]) for i in [0, N-s-1]
            grad[s:] += dx * (1.0 - h[:-s])
            grad[:N-s] += dx * (-h[s:])
        else:
            t = -s
            # E_{-t} = sum_{n=t}^{N-1} h[n-t] * (1-h[n])
            # d/dh[i] = dx*(1-h[i+t]) for i in [0, N-t-1] + dx*(-h[i-t]) for i in [t, N-1]
            grad[:N-t] += dx * (1.0 - h[t:])
            grad[t:] += dx * (-h[:-t])
        grads.append(grad)
    return np.array(grads)  # shape (len(active_shifts), N)
