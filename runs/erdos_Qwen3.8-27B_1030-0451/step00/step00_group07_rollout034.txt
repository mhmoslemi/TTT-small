```python
def project_to_constraint(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}."""
    n = len(h)
    # Find t such that sum(clamp(h - t, 0, 1)) = target_sum
    lo = h.min() - 1
    hi = h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
```

```python
    lo = h.min() - 1.0

        if s > target_sum:

            hi = mid

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_subgradient_smoothed(h, temperature=0.005):

    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Softmax weights over the lags
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / temperature)
    weights = exp_c / exp_c.sum()
    
    # Compute weighted gradient
    # For lag index k (0 to 2n-2), shift s = k - (n-1)
    # g_k[j] = (1-h[j-s]) * 1[0<=j-s<n] - h[j+s] * 1[0<=j+s<n]
    
    grad = np.zeros(n)
    s_vals = np.arange(-(n-1), n)

        if w < 1e-12:
            continue

        if s >= 0:

            if neg_s < n:
                grad[:n-neg_s] += w * one_minus_h[neg_s:]
```

For a given pair (i, j) and delta, the new overlap for lag k is:
new_overlap_k = old_overlap_k + delta * a_k

where a_k = [(1-h[i+s_k]) * 1[0<=i+s_k<n] - h[i-s_k] * 1[0<=i-s_k<n]] - [(1-h[j+s_k]) * 1[0<=j+s_k<n] - h[j-s_k] * 1[0<=j-s_k<n]]

So I need to find delta that minimizes max_k (old_overlap_k + delta * a_k).

This is a 1D convex optimization problem. The function is convex piecewise linear. I can solve it by:
1. Finding all breakpoints (where old_overlap_k + delta * a_k = old_overlap_l + delta * a_l for some k, l)
2. Checking the minimum at each breakpoint and at the boundaries

But there are O(n^2) breakpoints in the worst case, which is too many.

Alternative: use the fact that at the optimum, either:
- delta is at a boundary (delta_lo or delta_hi)
- The maximum is achieved by at least two lags (and they're equal)

```python
def project_to_constraint(h, target_sum):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_smoothed_subgradient(h, temperature):
    n = len(h)
    omm = 1.0 - h
    corr = np.correlate(h, omm, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / temperature)
    weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += w * omm[:n - s]
                grad[:n - s] -= w * h[s:]
        else:
            ns = -s
            if ns < n:
                grad[:n - ns] += w * omm[ns:]
                grad[ns:] -= w * h[:n - ns]
    return grad
```
