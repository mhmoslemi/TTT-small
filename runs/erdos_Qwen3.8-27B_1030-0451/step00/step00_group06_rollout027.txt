def compute_delta_C(h, C, i, j, delta, n):
    """Compute the change in C when h[i] -= delta and h[j] += delta."""
    dc = np.zeros(n)
    
    # Position i: h[i] changes by -delta
    # Shifted role: for m=0..i, C[m] has term h[i]*(1-h[i-m]), change = -delta*(1-h[i-m])
    if i >= 0 and i < n:
        # m ranges from 0 to min(i, n-1)
        m_end = min(i, n-1) + 1
        # h[i-m] for m=0..m_end-1: h[i], h[i-1], ..., h[i-(m_end-1)]
        # = h[i : i-m_end : -1] if m_end > 1, or h[i:i] (empty) ... 
        # Actually h[i::-1] gives h[i], h[i-1], ..., h[0] (length i+1)
        # We want the first m_end elements: h[i::-1][:m_end]
        arr = h[i::-1][:m_end]
        dc[:m_end] += -delta * (1 - arr)
    
    # Base role: for m=0..n-1-i, C[m] has term h[i+m]*(1-h[i]), change = +delta*h[i+m]

    # Linear terms
