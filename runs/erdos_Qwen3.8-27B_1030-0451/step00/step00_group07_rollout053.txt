def mass_project(y, target):

        else:

def resample_vector(h_src, N_src, N_dst, rng):
    """Resample from N_src to N_dst resolution"""
    x_src = np.linspace(0, 2, N_src, endpoint=False) + 1.0/N_src
    x_dst = np.linspace(0, 2, N_dst, endpoint=False) + 1.0/N_dst
    # Use cumulative sum for integration

    cum *= (2.0 / N_src)  # scale to actual integral
    # Find for each dst point, the integral over its cell

    x_dst_right = x_dst_left + 2.0 / N_dst

def pair_move(h, i, j, amount, N):
    """Transfer 'amount' from position i to position j"""

    new_h[j] = h[j] + amount
    if new_h[i] < 0 or new_h[j] > 1:
        return None
    new_h = np.clip(new_h, 0, 1)
