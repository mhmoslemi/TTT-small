    """Compute gradient of smooth max objective.
    h: array of length N
    alpha: array of length 2N-1 (weights for each lag)
    dx: step size
    """
    N = len(h)
    grad = np.zeros(N)
    one_minus_h = 1.0 - h
    
    for k in range(2*N - 1):
        a = alpha[k]
        if a < 1e-14:

        lo = max(0, -d)

        if lo <= hi:

    """Project y onto {h: 0 <= h <= 1, sum(h) = target_sum}"""
    N = len(y)

    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(y - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP constraint/objective
_G_T = 0.0
_G_N = 0
_G_TAU = 1.0

def _c_eq_fun(h):
    return float(np.sum(h) - _G_T)

def _c_eq_jac(h):
    return np.ones(_G_N)

def _c_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _c_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    active = np.where(alpha > 1e-12)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi + 1] += a * omh[lo - d:hi + 1 - d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= a * h[lo2 + d:hi2 + 1 + d]
    return grad * dx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def _smooth_optimize(h0, n, deadline):
    global _G_T, _G_N, _G_TAU
    _G_T = n / 2.0
    _G_N = n

    target = n / 2.0
    c5_init = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_init

    dx = 2.0 / n
    c_vals = np.correlate(h0, 1.0 - h0, 'full') * dx
    spread = max(1e-6, c_vals.max() - c_vals.mean())
    tau0 = max(0.015, 0.3 * spread)

    n_tau = 14
    tau_sched = tau0 * (0.55 ** np.arange(n_tau))

    constraints = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]
    bounds = [(0.0, 1.0)] * n

    h_curr = h0.copy()
    stall_count = 0

    for i, tau in enumerate(tau_sched):
        if time.time() > deadline - 3:
            break
        _G_TAU = tau
        maxiter = 40 if tau > 0.01 else 80
        try:
            res = minimize(_c_obj, h_curr, method='SLSQP', jac=_c_grad,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass

        c5e = evaluate_c5(h_curr)
        if c5e < best_c5 - 1e-10:
            best_c5 = c5e
            best_h = h_curr.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 4 and i < len(tau_sched) - 2:
                # Add small perturbation to escape local min
                rng_esc = np.random.default_rng(hash((n, i)) & 0xFFFFFFFF)
                h_curr = project_feasible(h_curr + rng_esc.standard_normal(n) * 0.005, target)
                stall_count = 0

    return best_h, best_c5

def _perturb_optimize(h0, n, deadline, rng):
    """Try a few random perturbations and optimize from each."""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    target = n / 2.0

    for amp in [0.003, 0.008, 0.015, 0.025]:
        for trial in range(3):
            if time.time() > deadline - 3:
                break
            pert = rng.standard_normal(n) * amp
            h_try = project_feasible(h0 + pert, target)
            h_opt, c5_opt = _smooth_optimize(h_try, n, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    # Get parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        n_par = 120
        parent = np.full(n_par, 0.5)

    n_par = len(parent)
    # Ensure parent is feasible
    parent = project_feasible(parent, n_par / 2.0)
    par_c5 = evaluate_c5(parent)

    best_h = parent.copy()
    best_c5 = par_c5
    best_n = n_par

    rng = np.random.default_rng(seed)

    # Build list of N values to try
    n_list = []
    for mult in [1, 2, 3, 4, 5, 6, 8]:
        n = n_par * mult
        if n <= 2400 and n not in n_list:
            n_list.append(n)
    n_list.sort()

    # Optimize at each resolution
    for n in n_list:
        if time.time() > deadline - 20:
            break
        remaining = deadline - time.time()
        if remaining < 15:
            break

        h0 = interp_h(parent, n)
        h_opt, c5_opt = _smooth_optimize(h0, n, deadline)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    # Additional perturbations on best solution if time allows
    if time.time() < deadline - 30:
        n = best_n
        h_opt, c5_opt = _perturb_optimize(best_h, n, deadline, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # If we have a good solution at smaller N, try refining at slightly larger N
    if time.time() < deadline - 40:
        n_next = best_n + n_par
        if n_next <= 2400:
            h0 = interp_h(best_h, n_next)
            h_opt, c5_opt = _smooth_optimize(h0, n_next, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_next

    # Final feasibility check and cleanup
    best_n = len(best_h)
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] -= diff / c
    np.clip(best_h, 0.0, 1.0, out=best_h)
    # Renormalize if clip changed sum
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] += diff / c
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
