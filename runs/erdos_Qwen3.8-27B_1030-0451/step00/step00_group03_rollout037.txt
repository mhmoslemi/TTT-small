def exact_c5(h):
    """Compute the exact C5 value using the official formula."""
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    c = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    return float(c)

def project_mean_box(y):

    y = np.asarray(y, dtype=np.float64).ravel().copy()

    target = n / 2.0
    
    # Replace non-finite
    y[~np.isfinite(y)] = 0.5

        return np.sum(np.clip(y + alpha, 0.0, 1.0))
    
    # Find alpha such that phi(alpha) = target
    a_lo = -1e6
    a_hi = 1e6
    
    # Tighten bounds
    a_lo = max(a_lo, target - np.sum(y) - 1.0)  # rough
    a_hi = min(a_hi, np.sum(1.0 - y) - target + 1.0)  # rough

            a_lo = a_mid
        else:
            a_hi = a_mid

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n

    return float(c)

def project_mean_box(y):

        return np.sum(np.clip(y + alpha, 0.0, 1.0))

    a_hi = 2.0

    for _ in range(20):
        if phi(a_lo) > target:
            a_lo -= 1.0
        else:
            break
    for _ in range(20):
        if phi(a_hi) < target:
            a_hi += 1.0
        else:
            break

        else:

    x_new = (0.5 + np.arange(n_new)) * 2.0 / n_new
    return np.interp(x_new, x_old, h, left=0.0, right=0.0)

def get_feasible_interval(h0, v):
    t_lo = 0.0
    t_hi = 0.0
    pos = v > 1e-14

    if np.any(neg):
        t_hi = min(t_hi, float(np.max(-h0[neg] / v[neg])))
        t_lo = max(t_lo, float(np.min((1.0 - h0[neg]) / v[neg])))
    return t_lo, t_hi

def line_search_exact(h0, v, t_lo, t_hi, n_coarse=16, n_fine=10):

        c = exact_c5(h0)

    best_c = float('inf')

            continue
        c = exact_c5(h_t)

            best_t = t

        margin = width * 0.1

    hi = min(t_hi, best_t + margin)

            best_t = t
