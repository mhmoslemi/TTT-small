def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target} via bisection on shift."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:

                h_tr = transfer_resolution(best_h, n_t)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c:
                    best_h = h_tr.copy()
                    best_c = c_tr
                # Optimize on new grid
                sub_dl = min(deadline - 10, time.time() + min(180, time_left() * 0.5))
                h_a, c_a = method_lp_trust(best_h, sub_dl, max_iters=100)
                if c_a < best_c:
                    best_h = h_a
                    best_c = c_a
                sub_dl2 = min(deadline - 10, time.time() + min(60, time_left() * 0.3))
                h_b, c_b = method_projo_grad(best_h, sub_dl2, max_iters=500)
                if c_b < best_c:
                    best_h = h_b
                    best_c = c_b
            except Exception:
                pass

    # Phase 6: Final fine polish
    if time_left() > 10:
        try:
            h6, c6 = method_projo_grad(best_h, deadline - 1, max_iters=5000)
            if c6 < best_c:
                best_h = h6
                best_c = c6
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
