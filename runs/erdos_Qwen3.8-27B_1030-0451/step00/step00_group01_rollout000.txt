def project_sum_box(x, target):
    """Project x onto {y : 0 <= y <= 1, sum(y) = target}."""
    n = len(x)
    # Bisection on lambda: find lambda such that sum(clip(x+lambda, 0, 1)) = target
    lo = -x.max() - 1
    hi = 1 - x.min() + 1
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0, 1).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + 0.5 * (lo + hi), 0, 1)
    # Fix residual
    residual = target - y.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            # Need to increase sum - add to elements not at 1
            room = 1.0 - y
            mask = room > 1e-12
            if mask.any():
                y[mask] += residual * room[mask] / room[mask].sum()
        else:
            # Need to decrease sum - subtract from elements not at 0
            space = y
            mask = space > 1e-12
            if mask.any():
                y[mask] += residual * space[mask] / space[mask].sum()
    y = np.clip(y, 0, 1)
    return y

def exact_eval(h):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx) where dx = 2/len(h)."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def lag_gradient(h, d):
    """Gradient of the lag-d correlation value w.r.t. h.
    val(d) = sum_{i valid} (1-h[i]) * h[i+d]
    where valid means 0<=i<n and 0<=i+d<n.
    Returns gradient vector (unmultiplied by dx).
    """
    n = len(h)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n, n - d)
    # First part: d(k) in (1-h[k]) * h[k+d], derivative w.r.t. h[k] is h[k+d]
    if i_lo < i_hi:
        k_range = np.arange(i_lo, i_hi)
        grad[k_range] += h[k_range + d]
    # Second part: derivative w.r.t. h[k+d] is -(1-h[k])
    j_lo = i_lo + d  # = max(0, -d) + d = max(d, 0)
    j_hi = i_hi + d  # = min(n, n-d) + d = min(n+d, n)
    if j_lo < j_hi:
        k_range2 = np.arange(j_lo, j_hi)
        grad[k_range2] -= (1.0 - h[k_range2 - d])
    return grad

def lag_value(h, d):
    """Compute the correlation value at lag d (before multiplying by dx)."""
    n = len(h)
    i_lo = max(0, -d)
    i_hi = min(n, n - d)
    if i_lo >= i_hi:
        return 0.0
    i = np.arange(i_lo, i_hi)
    return np.sum((1.0 - h[i]) * h[i + d])

def all_lag_values(h):
    """Compute all lag correlation values (before dx)."""
    n = len(h)
    vals = np.zeros(2 * n - 1)
    for d in range(-(n-1), n):
        vals[d + n - 1] = lag_value(h, d)
    return vals

def logsumexp_obj(h, beta):
    """Log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    v = all_lag_values(h) * dx  # shaped values in [0, 1]
    m = v.max()
    # Numerical stability
    u = np.exp(beta * (v - m))
    Z = u.sum()
    f = (m + np.log(Z) / beta) if Z > 0 else m
    # gradient
    w = u / Z if Z > 0 else np.zeros_like(v)
    w = np.maximum(w, 1e-30)
    w = w / w.sum()
    grad = np.zeros(n)
    for idx, d in enumerate(range(-(n-1), n)):
        if w[idx] > 1e-15:
            grad += w[idx] * lag_gradient(h, d) * dx
    return f, grad

def active_lp(h, v_vals, d_vals, radius, target, max_rows=100):
    """Solve active-constraint LP in a trust region around h.
    Minimize t s.t. for active lags: a_i . x <= t + (a_i . h - v_i)
    Plus: 0 <= x <= 1, sum(x) = target, |x - h| <= radius (box).
    Returns candidate x or None.
    """

    # Select active lags (top rows)
    order = np.argsort(v_vals)[::-1]
    active = order[:max_rows]
    
    # Build A_ub, b_ub for linprog (minimize t)

    rows = []
    rhs = []

        row = np.zeros(n + 1)
        row[:n] = a
        row[n] = -1.0
        rows.append(row)
        rhs.append(c)
    
    A_ub = np.array(rows) if rows else np.zeros((1, n+1))
    b_ub = np.array(rhs) if rows else np.array([1e10])

    bounds = [(max(0, h[i] - radius), min(1, h[i] + radius)) for i in range(n)]
    bounds.append((-0.5, 1.0))  # t bounds

def slsqp_smooth(h, target, beta, deadline, maxiter=200):
    """Run SLSQP on the log-sum-exp smoothed objective."""
    n = len(h)
    
    def obj_fn(x):
        f, g = logsumexp_obj(x, beta)
        return f, g

    return None

import numpy as np
from scipy.optimize import minimize, linprog
import time


def project_sum_box(x, target):
    """Project x onto {y : 0 <= y <= 1, sum(y) = target}."""
    n = len(x)
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            room = 1.0 - y
            mask = room > 1e-12
            if mask.any():
                y[mask] += residual * room[mask] / room[mask].sum()
        else:
            space = y
            mask = space > 1e-12
            if mask.any():
                y[mask] += residual * space[mask] / space[mask].sum()
    return np.clip(y, 0.0, 1.0)


def exact_eval(h):
    """C5 = max(correlate(h, 1-h, 'full') * dx)."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def all_lag_values_raw(h):
    """Compute raw lag correlation values (before dx). Length 2n-1."""
    n = len(h)
    vals = np.empty(2 * n - 1)
    for idx in range(2 * n - 1):
        d = idx - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        if i_lo >= i_hi:
            vals[idx] = 0.0
        else:
            i = np.arange(i_lo, i_hi)
            vals[idx] = np.sum((1.0 - h[i]) * h[i + d])
    return vals


def lag_gradient_raw(h, d):
    """Gradient of raw lag-d value w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    i_lo = max(0, -d)
    i_hi = min(n, n - d)
    if i_lo < i_hi:
        k = np.arange(i_lo, i_hi)
        grad[k] -= h[k + d]
    j_lo = max(d, 0)
    j_hi = min(n + d, n)
    if j_lo < j_hi:
        k2 = np.arange(j_lo, j_hi)
        grad[k2] += (1.0 - h[k2 - d])
    return grad


def logsumexp_obj_grad(h, beta):
    """Smooth surrogate objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    v_raw = all_lag_values_raw(h)
    v = v_raw * dx
    m = v.max()
    u = np.exp(beta * (v - m))
    Z = u.sum()
    if Z > 0:
        w = u / Z
        f = m + np.log(Z) / beta
    else:
        w = np.zeros_like(v)
        f = m
    w = np.maximum(w, 1e-30)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(len(v)):
        if w[idx] > 1e-12:
            d = idx - (n - 1)
            grad += w[idx] * lag_gradient_raw(h, d) * dx
    return f, grad


def active_lp_step(h, target, radius, max_rows=80):
    """One active-constraint LP trust-region step. Returns candidate or None."""
    n = len(h)
    v_raw = all_lag_values_raw(h)
    # Select top lags
    order = np.argsort(v_raw)[::-1]
    active = order[:min(max_rows, len(order))]

    rows = []
    rhs = []
    for idx in active:
        d = idx - (n - 1)
        a = lag_gradient_raw(h, d)
        v0 = v_raw[idx]
        c = a @ h - v0
        row = np.zeros(n + 1)
        row[:n] = a
        row[n] = -1.0
        rows.append(row)
        rhs.append(c)

    if not rows:
        return None
    A_ub = np.array(rows)
    b_ub = np.array(rhs)

    lb = np.maximum(0.0, h - radius)
    ub = np.minimum(1.0, h + radius)
    bounds = list(zip(lb, ub))
    bounds.append((-10.0, 10.0))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None


def slsqp_step(h, target, beta, maxiter=100):
    """One SLSQP call on the smooth surrogate."""
    n = len(h)

    def obj_fn(x):
        return logsumexp_obj_grad(x, beta)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            x = np.clip(res.x, 0.0, 1.0)
            return project_sum_box(x, target)
    except Exception:
        pass
    return None


def projected_grad_step(h, target, beta, n_iter=150, step=0.05):
    """Projected gradient on smooth surrogate. Returns (best_x, best_c)."""
    x = h.copy()
    best_x = x.copy()
    best_c = exact_eval(x)
    for _ in range(n_iter):
        f, g = logsumexp_obj_grad(x, beta)
        g_proj = g - g.mean()
        x_new = project_sum_box(x - step * g_proj, target)
        c_new = exact_eval(x_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_x = x_new.copy()
        x = x_new
        step *= 0.97
        if step < 1e-6:
            break
    return best_x, best_c


def transfer_resolution(h_old, n_new):
    """Interpolate to new grid size and project."""
    x_old = np.arange(len(h_old)) / len(h_old)
    x_new = np.arange(n_new) / n_new
    h_interp = np.interp(x_new, x_old, h_old)
    return project_sum_box(h_interp, n_new / 2.0)
