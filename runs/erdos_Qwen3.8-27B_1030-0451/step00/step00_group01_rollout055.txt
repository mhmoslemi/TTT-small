import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_sum_box(x, target):
    """Euclidean projection onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(x)
    if target <= 0 or target >= n:
        return np.clip(x, 0, 1)
    lo = -x.max() - 0.5
    hi = 1.0 - x.min() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(x + lam, 0.0, 1.0)
    # Tiny residual fix
    res = target - result.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        if mask.sum() > 0:
            result[mask] += res / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def corr_vals(h):
    """All lag correlation values scaled by dx"""
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def score(h):
    """C5 score = max lag value"""
    return corr_vals(h).max()

def lag_grad_unscaled(h, k):
    """Gradient of lag k value (before dx scaling) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:])
        if k > 0:
            g[k:] += -h[:m]
        else:
            g += -h
    else:
        kp = -k
        m = n - kp
        if m > 0:
            g[kp:] += (1.0 - h[:m])
        if kp > 0:
            g[:m] += -h[kp:]
    return g

def smooth_obj_and_grad(h, beta):
    """Log-sum-exp objective and its gradient"""
    n = len(h)
    dx = 2.0 / n
    v = corr_vals(h)
    m = v.max()
    shifted = beta * (v - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    Z = exps.sum()
    obj = m + np.log(Z) / beta
    w = exps / Z
    # Gradient
    g = np.zeros(n)
    for pos in range(2 * n - 1):
        wk = w[pos]
        if wk < 1e-16:
            continue
        k = pos - (n - 1)
        g += wk * lag_grad_unscaled(h, k)
    g *= dx
    return obj, g

def slsqp_optimize(h0, beta, maxiter=200):
    """One SLSQP call with log-sum-exp smoothing"""
    n = len(h0)
    target = n / 2.0
    
    def objective(x):
        o, _ = smooth_obj_and_grad(x, beta)
        return o
    
    def objective_grad(x):
        _, g = smooth_obj_and_grad(x, beta)
        return g
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: x.sum() - target,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(objective, h0, jac=objective_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.success or (res.x is not None and np.isfinite(res.fun)):
        x = project_sum_box(res.x, target)
        return x
    return None

def projected_subgradient(h0, time_limit):
    """Projected subgradient with active-set weighting"""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c = score(h0)
    start = time.time()
    
    alpha = 0.5 / n  # initial step size
    beta_temp = 20.0  # temperature for weighting
    
    for iteration in range(5000):
        if time.time() - start > time_limit:
            break
        v = corr_vals(h)
        m = v.max()
        # Weights: softmax-ish
        shifted = beta_temp * (v - m)
        shifted = np.clip(shifted, -300, 0)
        w = np.exp(shifted)
        w /= w.sum()
        
        g = np.zeros(n)
        for pos in range(2 * n - 1):
            if w[pos] < 1e-14:
                continue
            k = pos - (n - 1)
            g += w[pos] * lag_grad_unscaled(h, k)
        g *= (2.0 / n)
        
        h_new = project_sum_box(h - alpha * g, target)
        c_new = score(h_new)

import numpy as np
import time
from scipy.optimize import minimize, linprog


def project_sum_box(x, target):
    """Euclidean projection onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(x)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = -x.max() - 1.0
    hi = 1.0 - x.min() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(x + lam, 0.0, 1.0)
    res = target - result.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += res / cnt
            result = np.clip(result, 0.0, 1.0)
    return result


def corr_vals(h):
    """All lag correlation values scaled by dx"""
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_score(h):
    return corr_vals(h).max()


def lag_grad_raw(h, k):
    """Unscaled gradient of lag k value w.r.t. h (multiply by dx for scaled)"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        g[:m] += (1.0 - h[k:])
        if k > 0:
            g[k:] += -h[:m]
        else:
            g += -h
    else:
        kp = -k
        m = n - kp
        g[kp:] += (1.0 - h[:m])
        if kp > 0:
            g[:m] += -h[kp:]
    return g


def lse_obj_grad(h, beta):
    """Log-sum-exp smoothed objective and gradient"""
    n = len(h)
    dx = 2.0 / n
    v = corr_vals(h)
    m = v.max()
    shifted = beta * (v - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    Z = exps.sum()
    obj = m + np.log(Z) / beta
    w = exps / Z
    g = np.zeros(n)
    for pos in range(2 * n - 1):
        wk = w[pos]
        if wk < 1e-15:
            continue
        k = pos - (n - 1)
        g += wk * lag_grad_raw(h, k)
    g *= dx
    return obj, g


def do_slq(h0, beta, maxiter=150):
    """One SLSQP minimization call"""
    n = len(h0)
    target = n / 2.0

    def f(x):
        o, _ = lse_obj_grad(x, beta)
        return o

    def fgrad(x):
        _, g = lse_obj_grad(x, beta)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=fgrad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum_box(res.x, target)
    except Exception:
        pass
    return None


def subgrad_phase(h0, tlimit):
    """Projected subgradient with softmax weighting"""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c = c5_score(h0)
    t0 = time.time()
    alpha = 2.0 / n
    temp = 15.0

    while time.time() - t0 < tlimit:
        v = corr_vals(h)
        m = v.max()
        shifted = temp * (v - m)
        shifted = np.clip(shifted, -300, 0)
        w = np.exp(shifted)
        Z = w.sum()
        if Z < 1e-300:
            break
        w /= Z
        g = np.zeros(n)
        for pos in range(2 * n - 1):
            if w[pos] < 1e-14:
                continue
            k = pos - (n - 1)
            g += w[pos] * lag_grad_raw(h, k)
        g *= (2.0 / n)
        hn = project_sum_box(h - alpha * g, target)
        cn = c5_score(hn)
        if cn < best_c - 1e-14:
            best_h = hn.copy()
            best_c = cn
            h = hn
            alpha = min(alpha * 1.03, 5.0 / n)
        else:
            alpha *= 0.7
            h = hn
        temp *= 1.005
    return best_h, best_c

        lb = np.maximum(0.0, h - radius)
        ub = np.minimum(1.0, h + radius)
        lb_full = np.append(lb, -0.5)
        ub_full = np.append(ub, 1.0)
        c_obj = np.append(np.zeros(n), 1.0)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        if not a_ub_list:
            break
        A_ub = np.array(a_ub_list)
        b_ub = np.array(b_ub_list)
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb_full, ub_full)),
                          method='highs')
        except Exception:
            break

        if res.status == 0 and res.x is not None:
            x_new = project_sum_box(res.x[:n], target)
            c_new = c5_score(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new
                radius = min(radius * 1.15, 0.25)
            else:
                radius *= 0.55
        else:
            radius *= 0.5
        if radius < 0.003:
            radius = 0.12
    return best_h, best_c


def transfer_res(h, n_new):
    """Interpolate to new grid size"""
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    return project_sum_box(h_new, target)


def multi_slq(h0, tlimit):
    """Run SLSQP with increasing beta"""
    n = len(h0)
    t0 = time.time()
    h = h0.copy()
    best_h = h0.copy()
    best_c = c5_score(h0)
    betas = [4, 8, 16, 32, 64, 128, 300, 700, 1500]
    for beta in betas:
        if time.time() - t0 > tlimit:
            break
        h_tri = do_slq(h, beta, maxiter=120)
        if h_tri is not None:
            c_tri = c5_score(h_tri)
            if c_tri < best_c - 1e-13:
                best_h = h_tri
                best_c = c_tri
                h = h_tri
    return best_h, best_c


def binary_local_search(h0, tlimit):
    """Try to improve by making h more binary and doing pair flips"""
    n = len(h0)
    target = n / 2.0
    t0 = time.time()
    best_h = h0.copy()
    best_c = c5_score(h0)

    # Snap to binary
    h_bin = (h0 > 0.5).astype(float)
    h_bin = project_sum_box(h_bin, target)
    c_bin = c5_score(h_bin)
    if c_bin < best_c:
        best_h = h_bin.copy()
        best_c = c_bin

    # Pair swaps
    ones = np.where(h_bin >= 0.999)[0]
    zeros = np.where(h_bin <= 0.001)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best_h, best_c

    for _ in range(2000):
        if time.time() - t0 > tlimit:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c_new = c5_score(h_bin)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def remaining():
        return max(0, budget_s - 15 - (time.time() - t_start))

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, Exception):
        h0 = np.full(100, 0.5)

    n = len(h0)
    target = n / 2.0
    h0 = project_sum_box(h0, target)
    best_h = h0.copy()
    best_c = c5_score(h0)

    # Phase 1: SLSQP on initial grid
    if remaining() > 40:
        h_new, c_new = multi_slq(best_h, min(remaining() - 30, 250))
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Phase 2: Subgradient polish
    if remaining() > 40:
        h_new, c_new = subgrad_phase(best_h, min(remaining() - 20, 120))
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Phase 3: LP active set
    if remaining() > 40:
        h_new, c_new = lp_phase(best_h, min(remaining() - 20, 180))
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Phase 4: Resolution transfer + optimize on finer grid
    if remaining() > 150:
        n_cur = len(best_h)
        targets = []
        if n_cur <= 100:
            targets = [168, 252]
        elif n_cur <= 200:
            targets = [280, 360]
        else:
            targets = [int(n_cur * 1.5), int(n_cur * 2)]

        for n_new in targets:
            if remaining() < 100 or n_new > 500:
                break
            h_trans = transfer_res(best_h, n_new)
            c_trans = c5_score(h_trans)
            h_curr = h_trans.copy()
            c_curr = c_trans

            t_sub = time.time()
            sub_budget = min(remaining() - 60, 400)
            # SLSQP on new grid
            if sub_budget > 30:
                h_curr, c_curr = multi_slq(h_curr, min(sub_budget - 60, 250))
            # Subgradient
            sub_elapsed = time.time() - t_sub
            if sub_budget - sub_elapsed > 30:
                h_curr, c_curr = subgrad_phase(h_curr, min(sub_budget - sub_elapsed - 10, 100))
            # LP
            sub_elapsed = time.time() - t_sub
            if sub_budget - sub_elapsed > 30:
                h_curr, c_curr = lp_phase(h_curr, min(sub_budget - sub_elapsed - 10, 120))

            if c_curr < best_c - 1e-13:
                best_h, best_c = h_curr, c_curr

    # Phase 5: Binary local search on best grid
    if remaining() > 30:
        h_new, c_new = binary_local_search(best_h, min(remaining() - 10, 80))
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Phase 6: Final SLSQP polish
    if remaining() > 20:
        h_new, c_new = multi_slq(best_h, min(remaining() - 5, 80))
        if c_new < best_c - 1e-13:
            best_h, best_c = h_new, c_new

    # Final validation
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = c5_score(best_h)
    if final_c < best_c:
        best_c = final_c
    else:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
