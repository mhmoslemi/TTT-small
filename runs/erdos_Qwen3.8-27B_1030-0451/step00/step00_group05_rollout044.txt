import numpy as np
import time

def evaluate_c5(h):
    """Compute C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def project_feasible(h, n):
    """Project h onto [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to increase sum: add to elements that are < 1
        # Priority to elements with most room
        room = 1.0 - h
        # Distribute diff proportional to room
        total_room = np.sum(room)
        if total_room < 1e-15:
            return h  # Can't fix
        h = h + diff * room / total_room
    else:
        # Need to decrease sum
        total = np.sum(h)
        h = h * target / total
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Add/subtract from a single element
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def resample(h_old, n_new):
    """Resample h_old to n_new points."""
    n_old = len(h_old)
    x_old = np.arange(n_old)
    # Map new indices to old index space
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def compute_gradient(h, corr, active_idx, weights, dx):
    """Compute gradient of F = sum p_k * C_k with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    c0 = n - 1  # center index in full correlation
    
    for idx, p in zip(active_idx, weights):
        k = idx - c0  # lag value
        if k == 0:
            # C_0 = dx * sum h[i]*(1-h[i])
            # dC_0/dh[j] = dx * (1 - 2*h[j])
            grad += p * dx * (1.0 - 2.0 * h)
        elif k > 0:
            # C_k = dx * sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # dC_k/dh[j] = dx*(1-h[j-k]) for j in [k, n-1] - dx*h[j+k] for j in [0, n-1-k]
            for j in range(k, n):
                grad[j] += p * dx * (1.0 - h[j - k])
            for j in range(0, n - k):
                grad[j] -= p * dx * h[j + k]
        else:
            R = -k
            # C_{-R} = dx * sum_{i=R}^{n-1} h[i-R]*(1-h[i])
            # dC_{-R}/dh[j] = dx*(1-h[j+R]) for j in [0, n-1-R] - dx*h[j-R] for j in [R, n-1]
            for j in range(0, n - R):
                grad[j] += p * dx * (1.0 - h[j + R])
            for j in range(R, n):
                grad[j] -= p * dx * h[j - R]
    
    return grad

    """Compute gradient of F = sum p_k * C_k with respect to h."""
    n = len(h)
    grad = np.zeros(n)

            grad[:n-k] -= p * dx * h[k:]
        else:
            R = -k
            grad[:n-R] += p * dx * (1.0 - h[R:])
            grad[R:] -= p * dx * h[:n-R]
    
    return grad

    """Project v onto {h: 0 <= h <= 1, sum(h) = target_sum}."""
    n = len(v)
    lo, hi = -2.0, 2.0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid

            hi = mid
