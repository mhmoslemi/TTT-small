import numpy as np
import time

def exact_c5(h):
    """Compute the exact C5 value for a feasible h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def ensure_feasible(y, n):
    """Project y onto {x: 0<=x_i<=1, sum(x)=n/2} using shift-bisection."""
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(y, 0.0, 1.0)
    target = n / 2.0
    # Bisection on alpha: find alpha such that sum(clip(y + alpha, 0, 1)) = target
    alpha_lo = -1.0 - np.max(y)  # ensures sum = 0
    alpha_hi = 2.0 - np.min(y)   # ensures sum = n
    for _ in range(80):
        alpha_mid = 0.5 * (alpha_lo + alpha_hi)
        s = np.sum(np.clip(y + alpha_mid, 0.0, 1.0))
        if s < target:
            alpha_lo = alpha_mid
        else:
            alpha_hi = alpha_mid
    alpha = 0.5 * (alpha_lo + alpha_hi)
    x = np.clip(y + alpha, 0.0, 1.0)
    # Fix any residual
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            # Add to positions closest to 1
            idx = np.argsort(-x)[:max(1, int(abs(residual)) + 1)]
            for i in idx:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
        else:
            idx = np.argsort(x)[:max(1, int(abs(residual)) + 1)]
            for i in idx:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
                if abs(residual) < 1e-12:
                    break
    return x

def precompute_sensitivity(h):
    """Precompute S[i, m] = sensitivity of correlation[m] to h[i]."""
    n = len(h)
    L = 2 * n - 1
    S = np.zeros((n, L))
    one_minus_h = 1.0 - h
    for m in range(L):
        k = m - (n - 1)
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        if lo > hi:
            continue
        # For each i, compute S[i, m]
        # Part 1: i appears as h[n+k] where n = i-k, valid if lo <= i-k <= hi
        # Part 2: i appears in (1-h[n]) where n = i, valid if lo <= i <= hi
        for i in range(n):
            s = 0.0
            n1 = i - k
            if lo <= n1 <= hi:
                s += one_minus_h[n1]
            if lo <= i <= hi:
                s -= h[i + k]
            S[i, m] = s
    return S

def find_optimal_d(c, diff, d_min, d_max, num_evals=25):
    """Find d in [d_min, d_max] minimizing max_k (c[k] + d * diff[k])."""
    if d_min > d_max:
        return 0.0, np.max(c)
    if abs(d_max - d_min) < 1e-15:
        val = np.max(c + d_min * diff)
        return d_min, val
    ds = np.linspace(d_min, d_max, num_evals)
    # Compute max for each d
    # c + d*diff for each d
    max_vals = np.max(c[None, :] + ds[:, None] * diff[None, :], axis=1)
    best_idx = np.argmin(max_vals)
    return ds[best_idx], max_vals[best_idx]
