def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    # weights
    m = c.max()
    e = np.exp(sigma*(c - m))
    w = e / e.sum()   # softmax over j (length 2n-1)
    grad = np.zeros(n)
    center = n-1
    # p=0
    grad += w[center] * (1.0 - 2.0*h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp == 0.0 and wm == 0.0:
            continue
        # dV_{+p}/dh
        # A_plus[j]=1-h[j-p] for j in p..n-1
        # B_plus[j]=h[j+p] for j in 0..n-1-p
        if wp > 0:
            grad[p:]   += wp * (1.0 - h[:-p])   # j from p to n-1, h[j-p] = h[0..n-1-p]
            grad[:-p] -= wp * (h[p:])           # j from 0 to n-1-p, h[j+p]=h[p..n-1]
        if wm > 0:
            # C_minus[j]=1-h[j+p] for j in 0..n-1-p
            grad[:-p] += wm * (1.0 - h[p:])
            # D_minus[j]=h[j-p] for j in p..n-1
            grad[p:]   -= wm * (h[:-p])
    grad *= dx
    return grad

def anneal(h0, rng, time_limit, target):
    h = h0.copy()
    cur = objective(h)
    best_h = h.copy(); best_c = cur
    T = 1e-3
    T_end = 1e-7
    n = len(h)
    t0=time.monotonic()
    steps = 0
    while time.monotonic()-t0 < time_limit:
        r = rng.random()
        i = rng.integers(n); j = rng.integers(n)
        if i==j: continue
        if r < 0.5:
            # move mass
            d = rng.random()*0.2
            a = h.copy()
            # try move i->j and j->i
            if d <= min(h[i], 1.0-h[j]):
                a[i]-=d; a[j]+=d
            elif d <= min(h[j],1.0-h[i]):
                a[j]-=d; a[i]+=d
            else:
                continue
            # ensure exact sum preserved (it is: -d + d)
            na = objective(a)
        else:
            a = h.copy(); a[i],a[j]=a[j],a[i]
            na = objective(a)
        steps+=1
        if na < cur - 1e-14:
            h=a; cur=na
            if na<best_c-1e-12: best_c=na; best_h=a.copy()
        else:
            energy = na-cur
            if rng.random() < np.exp(-energy/T):
                h=a; cur=na
        # cool
        frac = min(1.0, (time.monotonic()-t0)/time_limit)
        T = T_end + (T-T_end)*(1-frac)
    return best_h, best_c

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def project_exact(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on threshold."""
    h = np.asarray(h, dtype=float)
    lo = -1.5
    hi = 1.5
    # S(t) = sum clip(h - t, 0, 1), decreasing in t
    for _ in range(70):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    # tighten final residual
    d = out.sum() - target
    if abs(d) > 0:
        # push remaining onto the entries with room, evenly, then reclip
        pad = 1e-9
        slack_up = 1.0 - out
        slack_dn = out
        if d < 0:
            add = -d
            room = slack_up.sum()
            if room > 0:
                out = out + (add / room) * slack_up
        else:
            add = d
            room = slack_dn.sum()
            if room > 0:
                out = out - (add / room) * slack_dn
        out = np.clip(out, 0.0, 1.0)
    return out


def repair(h, target):
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    h = project_exact(h, target)
    # final micro-correction to make sum essentially exact
    d = h.sum() - target
    n = len(h)
    if n and abs(d) > 1e-14:
        idx = int(np.argmax(h if d > 0 else (1.0 - h)))
        h = h.copy()
        h[idx] = float(np.clip(h[idx] - d, 0.0, 1.0))
    return h


def objective(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def lse_value(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    val = m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
    return dx * val


def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp <= 1e-18 and wm <= 1e-18:
            continue
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * (h[p:])
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * (h[:-p])
    grad *= dx
    return grad


def fd_gradient(h, sigma, eps=1e-5):
    g = np.zeros(len(h))
    for j in range(len(h)):
        hp = h.copy(); hp[j] += eps
        hm = h.copy(); hm[j] -= eps
        g[j] = (lse_value(hp, sigma) - lse_value(hm, sigma)) / (2.0 * eps)
    return g


def _grad_ok(h, sigma):
    try:
        gc = lse_gradient(h, sigma)
        gf = fd_gradient(h, sigma, eps=1e-5)
        denom = max(float(np.max(np.abs(gf))) , 1e-12)
        return float(np.max(np.abs(gc - gf))) / denom < 1e-2
    except Exception:
        return True

    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

    best_h = repair(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_c5 = objective(best_h)
    n_pts = len(best_h)
    return best_h, float(best_c5), int(n_pts)
