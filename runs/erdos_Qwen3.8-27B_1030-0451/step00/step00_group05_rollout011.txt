def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def compute_correlations(h):
    """Compute all correlation values."""
    dx = 2.0 / len(h)
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(v, n):
    """Project v onto {0 <= h[i] <= 1, sum(h) = n/2}."""
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clamp(v[i]-lambda, 0, 1)) = target
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Final adjustment to ensure exact sum
    diff = target - h.sum()
    # Distribute diff among non-saturated elements
    mask = (h > 0) & (h < 1)
    if mask.any():
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, C, active_indices, weights):
    """Compute gradient of weighted sum of active correlations."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros(n)
    c0 = n - 1
    
    for idx, w in zip(active_indices, weights):
        if w < 1e-15:
            continue
        R = idx - c0  # lag
        if R == 0:
            G += w * dx * (1 - 2 * h)
        elif R > 0:
            # C = dx * sum_{i=0}^{n-1-R} h[i+R] * (1-h[i])
            # dC/dh[j] = -dx*h[j+R] for j in [0, n-1-R]
            # dC/dh[j] = dx*(1-h[j-R]) for j in [R, n-1]
            G[:n-R] += -w * dx * h[R:]
            G[R:] += w * dx * (1 - h[:n-R])
        else:
            # R < 0, let R_abs = -R
            R_abs = -R
            # C = dx * sum_{i=0}^{n-1-R_abs} h[i] * (1-h[i+R_abs])

            # dC/dh[j] = -dx*h[j-R_abs] for j in [R_abs, n-1]
            G[:n-R_abs] += w * dx * (1 - h[R_abs:])
            G[R_abs:] += -w * dx * h[:n-R_abs]
