    N = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()

    sm = s.max()
    e = np.exp(s - sm)
    esum = e.sum()
    obj = m + tau * (sm + np.log(esum))
    alpha = e / esum
    grad = np.correlate(alpha, 1.0 - 2.0 * h, mode='valid')
    # Add penalty
    viol = h.sum() - N / 2.0
    obj += lam * viol * viol
    grad = grad + 2.0 * lam * viol
    return obj, grad

    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    # Bisection on uniform shift
    lo, hi = -1.0, 1.0
    for _ in range(80):

        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction

    if abs(resid) > 1e-10:

    if abs(resid) > 1e-15:

import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter
import time

def exact_c5(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def lse_obj_grad(h, tau, lam):
    N = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = (c - m) / tau
    sm = s.max()
    e = np.exp(s - sm)
    esum = e.sum()
    obj = m + tau * (sm + np.log(esum))
    alpha = e / esum
    grad = np.correlate(alpha, 1.0 - 2.0 * h, mode='valid')
    viol = h.sum() - N / 2.0
    obj += lam * viol * viol
    grad = grad + 2.0 * lam * viol
    return obj, grad

def obj_wrapper(x, tau, lam):
    v, _ = lse_obj_grad(x, tau, lam)
    return v

def grad_wrapper(x, tau, lam):
    _, g = lse_obj_grad(x, tau, lam)
    return g

def project_feasible(h, N):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-10:
        if resid > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += resid / len(idx)
            h = np.clip(h, 0.0, 1.0)
    resid = target - h.sum()
    if abs(resid) > 1e-15:
        if resid > 0:
            i = int(np.argmax(1.0 - h))
        else:
            i = int(np.argmin(h))
        h[i] = np.clip(h[i] + resid, 0.0, 1.0)
    return h

def scale_h(h_orig, N_new):
    N_orig = len(h_orig)
    if N_orig == N_new:
        return h_orig.copy()
    x_old = np.arange(N_orig, dtype=float) * (2.0 / N_orig)
    x_new = np.arange(N_new, dtype=float) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h_orig)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, N_new)

def optimize_at_size(h0, N, time_budget, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    lam = 2000.0
    c_init = np.correlate(h, 1.0 - h, mode='full')
    c_range = c_init.max() - c_init.min()
    tau_start = max(0.3, 0.3 * c_range)
    n_taus = max(3, min(6, int(time_budget / 15)))
    taus = []
    for i in range(n_taus):
        t = tau_start * (0.45 ** i)
        taus.append(max(t, 0.03))
    maxiter = max(40, min(150, 350 - N // 3))
    for tau in taus:
        if time_budget < 3:
            break
        t0 = time.time()
        try:
            res = minimize(obj_wrapper, h, args=(tau, lam),
                          jac=grad_wrapper, method='SLSQP',
                          bounds=[(0.0, 1.0)] * N,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, N)
            c5_val = exact_c5(h_new)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        time_budget -= (time.time() - t0)
    return best_h, best_c5

def perturb_smooth(h, N, rng, amp):
    noise = rng.standard_normal(N)
    noise = gaussian_filter(noise, sigma=max(2.0, N / 25.0))
    std = noise.std()
    if std > 1e-10:
        noise = noise / std * amp
    else:
        noise *= amp
    h_new = h + noise
    return project_feasible(h_new, N)

def peak_focused_step(h, N, rng, time_budget):
    """Try to reduce the top peaks specifically."""
    best_h = h.copy()
    best_c5 = exact_c5(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / N
    c5 = c.max() * dx
    top_k = min(5, len(c))
    peaks = np.argsort(c)[-top_k:][::-1]
    for pk in peaks:
        if time_budget < 2:
            break
        t0 = time.time()
        d = pk - (N - 1)
        lo = max(0, -d)
        lo2 = lo + d
        slots = N - abs(d)
        for _ in range(20):
            if c5 - exact_c5(best_h) < 1e-6:
                break
            h_try = h.copy()
            a = h_try[lo:lo+slots]
            b = h_try[lo2:lo2+slots]
            overlap = (a * (1.0 - b)).sum() * dx
            shift = 0.005 * rng.uniform(-1, 1)
            h_try[lo:lo+slots] = np.clip(h_try[lo:lo+slots] + shift, 0, 1)
            h_try = project_feasible(h_try, N)
            c5_try = exact_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
            time_budget -= (time.time() - t0)
            if time_budget < 1:
                break
    return best_h, best_c5

    t_start = time.time()

    def elapsed():
