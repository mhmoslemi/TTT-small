import time

def score_exact(h):
    n = len(h)
    dx = 2.0 / n
    Q = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(Q), Q

def compute_slope(h, t):
    n = len(h)
    L = 2 * n - 1
    slope = np.zeros(L)
    r1_lo = (n - 1) - t
    r1_hi = (n - 1) + (n - 1 - t)
    slope[r1_lo:r1_hi + 1] += 1.0 - h
    r2_lo = t  # (n-1) + (t-(n-1)) = t
    r2_hi = (n - 1) + t
    slope[r2_lo:r2_hi + 1] -= h[::-1]
    return slope

def pair_line_minimize(Q, A, G_pos, s_lo, s_hi):
    """Minimize max(Q + A*s + G*s^2) over s in [s_lo, s_hi].
    G is nonzero only at positions G_pos (up to 2 positions)."""
    # Golden section search
    if s_hi - s_lo < 1e-12:
        return 0.5 * (s_lo + s_hi)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    a = s_lo
    b = s_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    fc = line_objective(Q, A, G_pos, c)
    fd = line_objective(Q, A, G_pos, d)
    
    for _ in range(80):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = line_objective(Q, A, G_pos, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = line_objective(Q, A, G_pos, d)
    
    return 0.5 * (a + b)

def line_objective(Q, A, G_pos, s):
    """Compute max(Q + A*s + G*s^2) where G is nonzero only at G_pos."""
    val = Q + A * s
    for r in G_pos:
        val[r] += s * s
    return np.max(val)

def try_pair(h, Q, u, v, dx):
    """Try a pair transfer between u and v. Returns (new_h, new_Q, improvement) or None."""
    n = len(h)
    L = 2 * n - 1
    
    # Feasible interval
    s_lo = max(-h[u], h[v] - 1.0)
    s_hi = min(1.0 - h[u], h[v])
    if s_hi - s_lo < 1e-10:
        return None

    m = abs(u - v)

    new_h[u] += s_opt
    new_h[v] -= s_opt
    
    # Check bounds
    new_h = np.clip(new_h, 0.0, 1.0)
    
    # Check sum
    if abs(np.sum(new_h) - n / 2) > 1e-6:
        return None
    
    # Exact evaluation
    new_Q = np.correlate(new_h, 1.0 - new_h, mode='full')
    new_score = dx * np.max(new_Q)
    old_score = dx * np.max(Q)
    
    improvement = old_score - new_score
    if improvement > 1e-8:
        return new_h, new_Q, improvement
    
    return None
