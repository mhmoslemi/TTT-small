def project_feasible(x, target=None):
    x = np.asarray(x, dtype=float).ravel().copy()
    N = x.size
    if N == 0:
        return x
    if target is None:
        target = 0.5 * N
    else:
        target = float(target)

    x = np.clip(x, 0.0, 1.0)
    s0 = float(np.sum(x))
    if abs(s0 - target) > 1e-13:
        lo = -float(np.max(x)) - 1e-9
        hi = 1.0 - float(np.min(x)) + 1e-9
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            y = np.clip(x + mid, 0.0, 1.0)
            if float(np.sum(y)) > target:
                hi = mid
            else:
                lo = mid
        x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(np.sum(x))
    if abs(diff) > 1e-15:
        if diff > 0.0:
            order = np.argsort(1.0 - x)[::-1]
            for i in order:
                if diff <= 1e-15:
                    break
                room = 1.0 - x[i]
                add = room if room < diff else diff
                if add > 0.0:
                    x[i] += add
                    diff -= add
        else:
            order = np.argsort(x)[::-1]
            for i in order:
                if -diff <= 1e-15:
                    break
                rem = x[i]
                sub = rem if rem < -diff else -diff
                if sub > 0.0:
                    x[i] -= sub
                    diff += sub

    for _ in range(5):
        diff = target - float(np.sum(x))
        if abs(diff) <= 1e-12:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            add = min(1.0 - x[idx], diff)
            x[idx] += add
        else:
            idx = int(np.argmax(x))
            sub = min(x[idx], -diff)
            x[idx] -= sub

    x = np.clip(x, 0.0, 1.0)
    diff = target - float(np.sum(x))
    if abs(diff) > 0.0:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + diff)
    return x


def _linear_conv(a, b):
    if fftconvolve is not None and (a.size + b.size - 1) > 1024:
        return fftconvolve(a, b, mode="full")
    return np.convolve(a, b, mode="full")


def _L_values(x):
    N = x.size
    pref = np.empty(N + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(x)
    L = np.empty(2 * N - 1, dtype=float)
    L[N - 1] = pref[N]
    if N > 1:
        L[:N - 1] = pref[1:N]
        L[N:] = pref[N] - pref[1:N]
    return L


def _raw_C(x):
    N = x.size
    if N <= 700:
        return np.correlate(x, 1.0 - x, mode="full")
    L = _L_values(x)
    Q = _linear_conv(x, x[::-1])
    C = L - Q
    if not np.all(np.isfinite(C)):
        return np.correlate(x, 1.0 - x, mode="full")
    return C


def exact_c5(x):
    N = x.size
    if N < 2:
        return 1.0
    C = _raw_C(np.asarray(x, dtype=float))
    m = float(np.max(C))
    if not math.isfinite(m):
        return 1e100
    if m < 0.0:
        m = 0.0
    return m * (2.0 / N)


def exact_c5_direct(x):
    N = x.size
    if N < 2:
        return 1.0
    C = np.correlate(np.asarray(x, dtype=float), 1.0 - np.asarray(x, dtype=float), mode="full")
    m = float(np.max(C))
    if not math.isfinite(m):
        return 1e100
    if m < 0.0:
        m = 0.0
    return m * (2.0 / N)


def _C_single(x, idx):
    N = x.size
    d = idx - (N - 1)
    if d >= 0:
        L = float(np.sum(x[d:]))
    else:
        L = float(np.sum(x[:N + d]))
    m = abs(int(d))
    if m == 0:
        Q = float(np.dot(x, x))
    else:
        Q = float(np.dot(x[m:], x[:N - m]))
    return L - Q


def _grad_C_single(x, idx):
    N = x.size
    d = idx - (N - 1)
    gL = np.zeros(N, dtype=float)
    if d >= 0:
        gL[d:] = 1.0
    else:
        gL[:N + d] = 1.0

    m = abs(int(d))
    gQ = np.zeros(N, dtype=float)
    if m == 0:
        gQ[:] = 2.0 * x
    else:
        gQ[:N - m] += x[m:]
        gQ[m:] += x[:N - m]
    return gL - gQ


def _weighted_grad_L(alpha):
    N = (alpha.size + 1) // 2
    g = np.full(N, alpha[N - 1], dtype=float)
    if N > 1:
        wpos = alpha[N:]
        wneg = alpha[:N - 1][::-1]
        cumpos = np.empty(N, dtype=float)
        cumpos[0] = 0.0
        cumpos[1:] = np.cumsum(wpos)
        g += cumpos
        tmp = np.empty(N, dtype=float)
        tmp[0] = 0.0
        tmp[1:] = wneg
        cumneg = np.cumsum(tmp)
        g += cumneg[N - np.arange(N) - 1]
    return g


def _weighted_grad_Q(x, alpha):
    N = x.size
    center = alpha[N - 1]
    g = 2.0 * center * x
    if N > 1:
        w = np.zeros(N, dtype=float)
        w[1:] = alpha[:N - 1][::-1] + alpha[N:]
        c1 = _linear_conv(x, w)
        c2 = _linear_conv(x, w[::-1])
        g += c1[:N]
        g += c2[N - 1:N - 1 + N]
    return g


def _gradC_weighted(x, alpha):
    active = np.nonzero(alpha > 1e-14)[0]
    if active.size == 0:
        return np.zeros(x.size, dtype=float)
    if active.size <= 12:
        g = np.zeros(x.size, dtype=float)
        for idx in active:
            g += float(alpha[idx]) * _grad_C_single(x, int(idx))
        return g
    return _weighted_grad_L(alpha) - _weighted_grad_Q(x, alpha)


def _smooth_objective_and_grad(x):
    N = x.size
    tau = float(CURRENT_TAU)
    if tau <= 1e-10:
        tau = 1e-10
    if not np.all(np.isfinite(x)):
        return 1e100, np.zeros(N, dtype=float)

    C = _raw_C(x)
    if not np.all(np.isfinite(C)):
        return 1e100, np.zeros(N, dtype=float)

    m = float(np.max(C))
    z = (C - m) / tau
    ez = np.exp(z)
    s = float(np.sum(ez))
    dx = 2.0 / N
    if s <= 0.0 or not math.isfinite(s):
        idx = int(np.argmax(C))
        alpha = np.zeros_like(C, dtype=float)
        alpha[idx] = 1.0
        gC = _gradC_weighted(x, alpha)
        return dx * m, dx * gC

    alpha = ez / s
    val = dx * (m + tau * math.log(s))
    gC = _gradC_weighted(x, alpha)
    grad = dx * gC
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), grad


def _smooth_obj(x):
    global CACHE_X, CACHE_VAL, CACHE_GRAD
    if CACHE_X is None or x.size != CACHE_X.size or not np.array_equal(x, CACHE_X):
        v, g = _smooth_objective_and_grad(x)
        CACHE_X = x.copy()
        CACHE_VAL = float(v)
        CACHE_GRAD = g.copy()
    return CACHE_VAL


def _smooth_grad(x):
    global CACHE_X, CACHE_VAL, CACHE_GRAD
    if CACHE_X is None or x.size != CACHE_X.size or not np.array_equal(x, CACHE_X):
        v, g = _smooth_objective_and_grad(x)
        CACHE_X = x.copy()
        CACHE_VAL = float(v)
        CACHE_GRAD = g.copy()
    return CACHE_GRAD


def _sum_fun(x):
    return float(np.sum(x) - 0.5 * len(x))


def _sum_jac(x):
    return np.ones(len(x), dtype=float)


def _sum_z_fun(z):
    return float(np.sum(z[:-1]) - 0.5 * len(z[:-1]))


def _sum_z_jac(z):
    N = len(z) - 1
    g = np.zeros(N + 1, dtype=float)
    g[:N] = 1.0
    return g


def _slsqp_maxiter(N):
    if N <= 200:
        return 80
    if N <= 500:
        return 60
    if N <= 1000:
        return 45
    return 30


def _slsqp_step(x, N, tau, maxiter):
    global CURRENT_TAU, CACHE_X
    x0 = project_feasible(x, 0.5 * N)
    CURRENT_TAU = float(tau)
    CACHE_X = None

    bounds = [(0.0, 1.0)] * N
    cons = {"type": "eq", "fun": _sum_fun, "jac": _sum_jac}
    options = {"maxiter": int(maxiter), "ftol": 1e-12, "disp": False}

    try:
        res = minimize(
            _smooth_obj,
            x0,
            jac=_smooth_grad,
            method="SLSQP",
            bounds=bounds,
            constraints=[cons],
            options=options,
        )
        xr = res.x
        if xr is None or not np.all(np.isfinite(xr)):
            return x0, exact_c5(x0)
        xr = project_feasible(xr, 0.5 * N)
        c = exact_c5(xr)
        return xr, c
    except Exception:
        return x0, exact_c5(x0)


def fallback_initial(N):
    if N < 2:
        N = 2
    t = np.linspace(0.0, 2.0, int(N))
    x = 0.5 + 0.45 * np.cos(np.pi * t)
    return project_feasible(x, 0.5 * N)


def upsample(h, N):
    old = np.asarray(h, dtype=float).ravel()
    N = int(N)
    if N < 2:
        N = 2
    if old.size == 0:
        return fallback_initial(N)
    if old.size == N:
        return project_feasible(old, 0.5 * N)
    old_coords = np.arange(old.size) * (2.0 / old.size)
    new_coords = np.arange(N) * (2.0 / N)
    x = np.interp(new_coords, old_coords, old)
    return project_feasible(x, 0.5 * N)


def repeat_start(base, N):
    old = np.asarray(base, dtype=float).ravel()
    N = int(N)
    if old.size <= 0:
        return fallback_initial(N)
    if N % old.size == 0:
        return project_feasible(np.repeat(old, N // old.size), 0.5 * N)
    return upsample(old, N)


def threshold_start(h, N):
    base = upsample(np.asarray(h, dtype=float).ravel(), N)
    x = np.zeros(N, dtype=float)
    k = int(np.floor(0.5 * N))
    if k > 0:
        top = np.argsort(-base)[:k]
        x[top] = 1.0
    return project_feasible(x, 0.5 * N)


def perturb_start(base, N, amp, seed):
    rng = np.random.default_rng(int(seed))
    x = upsample(np.asarray(base, dtype=float).ravel(), N)
    noise = rng.standard_normal(N)
    if N > 8:
        noise = np.convolve(noise, np.ones(3) / 3.0, mode="same")
    return project_feasible(x + float(amp) * noise, 0.5 * N)


def smooth_optimize(x, N, deadline):
    x = project_feasible(x, 0.5 * N)
    c = exact_c5(x)
    best_x = x.copy()
    best_c = c

    if N <= 300:
        taus = [1.2, 0.45, 0.15]
    elif N <= 600:
        taus = [2.0, 0.7, 0.25]
    else:
        taus = [3.0, 1.0, 0.35]

    for tau in taus:
        if time.time() > deadline - 45.0:
            break
        xr, cr = _slsqp_step(x, N, tau, _slsqp_maxiter(N))
        if cr < best_c - 1e-12:
            best_c = cr
            best_x = xr.copy()
            x = xr
    return best_x, best_c


def _peak_obj(z):
    return float(z[-1])


def _peak_obj_grad(z):
    N = len(z) - 1
    g = np.zeros(N + 1, dtype=float)
    g[-1] = 1.0
    return g


def _peak_constraint_fun(z):
    A = ACTIVE_INDICES
    h = z[:-1]
    t = float(z[-1])
    out = np.empty(int(A.size), dtype=float)
    for i in range(int(A.size)):
        out[i] = t - _C_single(h, int(A[i]))
    return out


def _peak_constraint_jac(z):
    A = ACTIVE_INDICES
    N = len(z) - 1
    K = int(A.size)
    J = np.empty((K, N + 1), dtype=float)
    h = z[:-1]
    for i in range(K):
        J[i, :N] = -_grad_C_single(h, int(A[i]))
        J[i, N] = 1.0
    return J


def _make_bounds_z(N, t_upper):
    return [(0.0, 1.0)] * N + [(0.0, max(0.1, float(t_upper)))]


def _peak_maxiter(N):
    if N <= 200:
        return 150
    if N <= 500:
        return 100
    if N <= 1000:
        return 70
    return 50


def peak_refine(x, deadline, max_rounds=10, K=8):
    global ACTIVE_INDICES
    x = project_feasible(x)
    N = x.size
    C = _raw_C(x)
    maxC = float(np.max(C))
    best_raw = maxC
    best_x = x.copy()
    best_c = best_raw * (2.0 / N)
    t = maxC + max(0.5, 0.01 * maxC)

    for _ in range(int(max_rounds)):
        if time.time() > deadline - 20.0:
            break

        C = _raw_C(x)
        maxC = float(np.max(C))
        if maxC < best_raw - 1e-12:
            best_raw = maxC
            best_x = x.copy()
            best_c = best_raw * (2.0 / N)

        if maxC >= best_raw - 1e-12:
            x = best_x.copy()
            C = _raw_C(x)
            maxC = float(np.max(C))

        delta = max(0.15, 0.03 * maxC)
        active = np.where(C >= maxC - delta)[0]
        if active.size > K:
            active = active[np.argsort(-C[active])[: int(K)]]
        if active.size == 0:
            active = np.array([int(np.argmax(C))], dtype=int)
        ACTIVE_INDICES = active.astype(int)

        t = max(t, maxC + 0.05)
        z = np.empty(N + 1, dtype=float)
        z[:N] = x
        z[-1] = t

        t_upper = max(t + 0.1, maxC + max(2.0, 0.1 * maxC))
        bounds = _make_bounds_z(N, t_upper)
        cons_ineq = {"type": "ineq", "fun": _peak_constraint_fun, "jac": _peak_constraint_jac}
        cons_eq = {"type": "eq", "fun": _sum_z_fun, "jac": _sum_z_jac}
        options = {"maxiter": _peak_maxiter(N), "ftol": 1e-12, "disp": False}

        try:
            res = minimize(
                _peak_obj,
                z,
                jac=_peak_obj_grad,
                method="SLSQP",
                bounds=bounds,
                constraints=[cons_eq, cons_ineq],
                options=options,
            )
            nx = res.x[:N] if res.x is not None else None
            if nx is None or not np.all(np.isfinite(nx)):
                break
            nx = project_feasible(nx)
            nC = _raw_C(nx)
            nmax = float(np.max(nC))
            if not math.isfinite(nmax):
                break
            if nmax < best_raw - 1e-12:
                best_raw = nmax
                best_x = nx.copy()
                best_c = nmax * (2.0 / N)
            if nmax <= maxC + 1e-7:
                x = nx
                curmax = nmax
            else:
                curmax = maxC
            t = curmax + max(0.1, 0.005 * curmax)
        except Exception:
            break

    return best_x, best_c


def subgrad_polish(x, deadline, max_steps=25):
    x = project_feasible(x)
    N = x.size
    best_x = x.copy()
    best_c = exact_c5(x)
    step = 0.15

    for _ in range(int(max_steps)):
        if time.time() > deadline - 8.0:
            break

        C = _raw_C(x)
        maxC = float(np.max(C))
        delta = max(0.1, 0.02 * maxC)
        active = np.where(C >= maxC - delta)[0]
        if active.size > 12:
            active = active[np.argsort(-C[active])[:12]]
        if active.size == 0:
            active = np.array([int(np.argmax(C))], dtype=int)

        alpha = np.zeros(C.shape, dtype=float)
        alpha[active] = 1.0 / float(active.size)
        gC = _gradC_weighted(x, alpha)
        g = gC * (2.0 / N)
        norm = float(np.linalg.norm(g))
        if not math.isfinite(norm) or norm < 1e-12:
            break
        direction = g / norm

        trial_x = None
        trial_c = best_c
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            y = x - step * mult * direction
            y = project_feasible(y)
            cy = exact_c5(y)
            if cy < trial_c - 1e-13:
                trial_x = y
                trial_c = cy

        if trial_x is not None:
            x = trial_x
            best_c = trial_c
            best_x = x.copy()
            step = min(0.5, max(step * 0.9, 0.02))
        else:
            step *= 0.5
            if step < 1e-4:
                break

    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        safe_budget = max(60.0, float(budget_s))
    except Exception:
        safe_budget = 1100.0
    deadline = t0 + safe_budget - 45.0

    parent = None
    raw = globals().get("initial_h_values", None)
    try:
        p = np.asarray(raw, dtype=float).ravel()
        if p.size >= 2 and np.all(np.isfinite(p)):
            parent = project_feasible(p, 0.5 * p.size)
    except Exception:
        parent = None

    if parent is None:
        parent = fallback_initial(120)

    best_h = parent.copy()
    best_n = int(best_h.size)
    best_c = exact_c5(best_h)

    N0 = int(parent.size)
    max_n = int(min(1000, max(N0 * 7, 200)))

    nodes = [N0]
    for mult in (2, 3, 4, 5, 6, 7):
        N = int(round(N0 * mult))
        if N % 2 == 1:
            N += 1
        if 2 <= N <= max_n:
            nodes.append(N)

    uniq = []
    for n in nodes:
        if n not in uniq:
            uniq.append(n)
    uniq.sort()
    if N0 in uniq:
        uniq.remove(N0)
        uniq.insert(0, N0)

    if safe_budget < 500.0:
        uniq = uniq[:2]
    elif safe_budget < 800.0:
        uniq = uniq[:4]
    else:
        uniq = uniq[:6]

    seed_base = int(seed) if seed is not None else 42

    for N in uniq:
        if time.time() > deadline - 100.0:
            break

        starts = []
        if best_h.size == N:
            starts.append(best_h.copy())
        else:
            starts.append(upsample(best_h, N))

        if best_h.size != parent.size:
            starts.append(upsample(parent, N))

        if N % best_h.size == 0 and N <= 800:
            starts.append(repeat_start(best_h, N))

        if N <= 700:
            starts.append(threshold_start(best_h, N))
        if N <= 500:
            starts.append(perturb_start(best_h, N, 0.05, seed_base + N + 1))

        if N == N0:
            starts.append(perturb_start(parent, N, 0.07, seed_base + N + 2))

        unique_starts = []
        for s in starts:
            try:
                s = project_feasible(np.asarray(s, dtype=float).ravel(), 0.5 * N)
            except Exception:
                continue
            dup = False
            for u in unique_starts:
                if float(np.max(np.abs(u - s))) < 1e-7:
                    dup = True
                    break
            if not dup:
                unique_starts.append(s)
            if len(unique_starts) >= 3:
                break

        for s in unique_starts:
            if time.time() > deadline - 60.0:
                break
            x, c = smooth_optimize(s, N, deadline)
            if c < best_c:
                best_c = c
                best_h = x.copy()
                best_n = int(x.size)

        if N == N0 and time.time() < deadline - 320.0:
            x, c = peak_refine(best_h, deadline - 280.0, max_rounds=4, K=6)
            if c < best_c:
                best_c = c
                best_h = x.copy()
                best_n = int(x.size)

    if time.time() < deadline - 40.0:
        if best_n <= 300:
            K = 10
        elif best_n <= 800:
            K = 8
        else:
            K = 6
        tleft = deadline - time.time()
        maxr = int(min(18, max(2, tleft / 30.0)))
        x, c = peak_refine(best_h, deadline - 25.0, max_rounds=maxr, K=K)
        if c < best_c:
            best_c = c
            best_h = x.copy()
            best_n = int(x.size)

    if time.time() < deadline - 15.0:
        x, c = subgrad_polish(best_h, deadline - 8.0, max_steps=15)
        if c < best_c:
            best_c = c
            best_h = x.copy()
            best_n = int(x.size)

    best_h = project_feasible(best_h, 0.5 * best_h.size)
    final_c = exact_c5_direct(best_h)
    if not math.isfinite(final_c):
        final_c = exact_c5(best_h)
    return best_h, float(final_c), int(best_h.size)
