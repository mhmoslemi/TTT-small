def repair_vector(z, n):

    y = np.clip(z, 0.0, 1.0)

def pair_descent(h, n, cur_score, num_tries, rng, time_limit):
    """Try random pair exchanges to improve"""
    best_h = h.copy()
    best_score = cur_score
    improved = True
    while improved and (time_limit is None or time.time() < time_limit):
        improved = False
        for _ in range(num_tries):

                best_score = new_score

    return best_h, best_score

def sa_optimize(h, n, cur_score, rng, time_limit, T0=0.004, Tend=0.0002, rounds=200):

    best_h = h.copy()
    best_score = cur_score
    cur_h = h.copy()
    cur_s = cur_score
    
    T = T0
    decay = (Tend / T0) ** (1.0 / max(rounds, 1))
    
    for r in range(rounds):
        if time_limit is not None and time.time() >= time_limit:
            break
        T *= decay

        j = rng.randint(n)

            cur_s = new_s
            if cur_s < best_score - 1e-12:

                best_score = cur_s

    m = np.max(corr_scaled)
    val = m + tau * np.log(np.sum(np.exp((corr_scaled - m) / tau)))

    total_search_time = effective_end - time.time() - 15
    weights = [1.0, 0.7, 0.6, 0.5, 0.4, 0.3]
    total_w = sum(w for w in weights[:len(dims)])
    
    for dim_idx, n in enumerate(dims):
        time_left = effective_end - time.time()
        if time_left < 15:
            break
        
        w = weights[dim_idx] if dim_idx < len(weights) else 0.3
        alloc = total_search_time * w / total_w
        
        # Get starting vector
        if n == len(best_h):
            h = best_h.copy()
            score = best_score
        elif n == n_parent:
            h = parent.copy()
            score = exact_score(h, n)
        else:
            h = transfer_to_dim(parent, n_parent, n)
            score = exact_score(h, n)
        
        t_limit = min(time.time() + alloc, effective_end - 10)
        if t_limit - time.time() < 10:
            continue
        
        # Optimize
        h_out, score_out = optimize_dim(h, n, score, rng, t_limit)
        
        if score_out < best_score - 1e-12:
            best_h = h_out
            best_n = n
            best_score = score_out
    
    # Final verification
    final_score = exact_score(best_h, best_n)
    if final_score < best_score:
        best_score = final_score
    
    return best_h, best_score, best_n
