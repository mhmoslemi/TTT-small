import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_sum_box(x, target):
    """Project x onto {y in [0,1]^n : sum(y) = target} using bisection on lambda."""
    n = len(x)
    # If already close, just clip
    if abs(sum(x) - target) < 1e-12:
        return np.clip(x, 0.0, 1.0)
    
    def sum_projected(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0))
    
    # Find lambda such that sum_projected(lambda) = target
    lo, hi = -max(x) - 0.1, 1.0 - min(x) + 0.1
    if sum_projected(lo) > target:
        return np.clip(x + lo, 0.0, 1.0)
    if sum_projected(hi) < target:
        return np.clip(x + hi, 0.0, 1.0)
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = sum_projected(mid)
        if abs(s - target) < 1e-12:
            break
        if s > target:
            hi = mid
        else:
            lo = mid
    
    result = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix tiny residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            free = result < 0.999999
            nfree = np.sum(free)
            if nfree > 0:
                per = min(residual / nfree, 0.999999 - result[free])
                result[free] += per
        else:
            free = result > 1e-6
            nfree = np.sum(free)
            if nfree > 0:
                per = max(residual / nfree, 1e-6 - result[free])
                result[free] += per
    return result

def corr_values(h):
    """Compute all correlation values: np.correlate(h, 1-h, mode='full') * dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_evaluate(h):
    """Compute C5 = max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals))

def grad_lag(h, k, dx):
    """Gradient of v_lag = dx * sum_{i valid} h[i+k]*(1-h[i]) w.r.t h, for lag k."""
    n = len(h)
    g = np.zeros(n)
    # First term: (1-h[p-k]) for p where 0<=p-k<n and 0<=p<n
    # p-k >= 0 => p >= k (if k > 0); p-k < n => p < n+k
    # p >= 0, p < n
    p_start = max(0, k)
    p_end = min(n, n + k)
    if p_start < p_end:
        idx = np.arange(p_start, p_end)
        g[idx] += (1.0 - h[idx - k])
    # Second term: -h[p+k] for p where 0<=p<n and 0<=p+k<n
    # p+k >= 0 => p >= -k (if k < 0); p+k < n => p < n-k
    # p >= 0, p < n
    q_start = max(0, -k)
    q_end = min(n, n - k)
    if q_start < q_end:
        idx2 = np.arange(q_start, q_end)
        g[idx2] -= h[idx2 + k]
    return dx * g

def grad_all_lags(h):
    """Gradient matrix: shape (2n-1, n). Row l corresponds to lag l-(n-1)."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for l in range(2 * n - 1):

        G[l] = grad_lag(h, k, dx)

def lse_objective_and_grad(h, beta):
    """Log-sum-exp objective and its gradient. Returns (value, gradient)."""
    vals = corr_values(h)
    mx = np.max(vals)
    shifted = beta * (vals - mx)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    obj = mx + np.log(Z) / beta
    weights = exp_vals / Z
    # Gradient = weighted sum of individual lag gradients
    n = len(h)
    grad = np.zeros(n)
    dx = 2.0 / n
    for l in range(2 * n - 1):
        if weights[l] > 1e-10:
            k = l - (n - 1)
            grad += weights[l] * grad_lag(h, k, dx)
    return obj, grad

def slsqp_step(h0, beta, maxiter=200):
    """One SLSQP step minimizing LSE with constraints."""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    def obj_and_grad(h):
        v, g = lse_objective_and_grad(h, beta)
        return v, g
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(obj_and_grad, h0, jac=True, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if result.success or result.fun < obj_and_grad(h0)[0]:
        h_new = project_sum_box(result.x, target)
        return h_new
    return None

def active_lp_step(h0, radius, n_active_max, dx):
    """Active-set LP in a trust region."""
    n = len(h0)
    target = n / 2.0
    vals = corr_values(h0)
    
    # Select active lags (top values)
    top_k = min(n_active_max, 2 * n - 1)
    active_idx = np.argsort(-vals)[:top_k]
    
    # Build LP: min t s.t. a_i^T h - t <= a_i^T h0 - v_i for active i
    # Plus box and sum constraints
    # Variables: [h (n), t (1)] → total n+1
    
    # Objective: minimize t → c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints (A_ub x <= b_ub)
    A_rows = []
    b_rows = []
    
    for l in active_idx:
        k = l - (n - 1)
        gl = grad_lag(h0, k, dx)
        # a_i^T h - t <= a_i^T h0 - v_i
        # a_i^T h - t - (a_i^T h0 - v_i) <= 0
        row = np.zeros(n + 1)
        row[:n] = gl
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(gl @ h0 - vals[l])
    
    # Trust region: -radius <= h-p - h0 <= radius → h-p <= h0+radius, -h-p <= -h0+radius
    A_trust_lo = np.zeros((n, n + 1))
    A_trust_lo[:, :n] = -np.eye(n)
    A_rows.extend(A_trust_lo)
    b_rows.extend(h0 - radius)
    
    A_trust_hi = np.zeros((n, n + 1))
    A_trust_hi[:, :n] = np.eye(n)
    A_rows.extend(A_trust_hi)
    b_rows.extend(h0 + radius)
    
    # Box: -h <= 0, h <= 1
    A_box_lo = np.zeros((n, n + 1))
    A_box_lo[:, :n] = -np.eye(n)
    A_rows.extend(A_box_lo)
    b_rows.extend(np.zeros(n))
    
    A_box_hi = np.zeros((n, n + 1))
    A_box_hi[:, :n] = np.eye(n)
    A_rows.extend(A_box_hi)
    b_rows.extend(np.ones(n))
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds on variables
    bounds = [(max(0.0, h0[i] - radius), min(1.0, h0[i] + radius)) for i in range(n)]
    bounds.append((-0.5, 1.0))  # t bounds
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:n]
            h_new = project_sum_box(h_new, target)
            return h_new
    except Exception:
        pass
    return None

def transfer_resolution(h_old, n_new):
    """Interpolate h from old grid to new grid."""
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def steepest_descent(h0, beta, n_steps, step_sizes, dx):
    """Projected steepest descent on LSE."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_local = h.copy()
    best_val = exact_evaluate(h)
    
    for step in range(n_steps):
        obj, grad = lse_objective_and_grad(h, beta)
        # Project gradient onto feasible subspace (remove mean to preserve sum)
        grad_projected = grad - np.mean(grad)
        norm = np.linalg.norm(grad_projected)
        if norm < 1e-12:
            break
        for alpha in step_sizes:
            h_trial = h - alpha * grad_projected / norm
            h_trial = project_sum_box(h_trial, target)
            c_val = exact_evaluate(h_trial)
            if c_val < best_val - 1e-10:
                best_val = c_val
                best_local = h_trial.copy()
            h_trial2 = project_sum_box(h - alpha * grad_projected, target)
            if np.sum(h_trial2) == target:
                h = h_trial2
                break
    return best_local, best_val
