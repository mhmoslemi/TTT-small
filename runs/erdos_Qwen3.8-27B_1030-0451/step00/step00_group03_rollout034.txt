def project_feasible(y, n=None):

    if n is None:
        n = len(y)
    target = n / 2.0
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    # Bisection on alpha: x = clip(y + alpha, 0, 1), find alpha s.t. sum(x) = target
    lo = -max(y.max(), 0) - 1.0
    hi = 1.0 - min(y.min(), 1) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # Distribute residual to elements with room
        if residual > 0:
            room = 1.0 - x
            room_idx = np.where(room > 1e-10)[0]
            if len(room_idx) > 0:
                # Add to largest room elements
                order = room_idx[np.argsort(-room[room_idx])]

import numpy as np
from time import monotonic

_best_h = None
_best_c5 = float('inf')

def update_best(h, c5):
    global _best_h, _best_c5
    if h is not None and np.all(np.isfinite(h)) and c5 < _best_c5 - 1e-12:
        _best_h = h.copy()
        _best_c5 = float(c5)

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y, n=None):
    if n is None:
        n = len(y)
    target = n / 2.0
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0).astype(np.float64)
    lo = -np.max(y) - 0.5
    hi = 1.0 - np.min(y) + 0.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            room = 1.0 - x
            idx = np.where(room > 1e-8)[0]
            if len(idx) > 0:
                amt = min(residual, room[idx].sum())
                fracs = room[idx] / room[idx].sum()
                x[idx] += fracs * amt
        else:
            idx = np.where(x > 1e-8)[0]
            if len(idx) > 0:
                amt = min(-residual, x[idx].sum())
                fracs = x[idx] / x[idx].sum()
                x[idx] -= fracs * amt
    return x

def interpolate_to_dim(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    x_src = (0.5 + np.arange(n_src)) * 2.0 / n_src
    x_tgt = (0.5 + np.arange(n_tgt)) * 2.0 / n_tgt
    g = np.interp(x_tgt, x_src, h_src)
    return project_feasible(g, n_tgt)

def try_pair(h, i, j, rng, n_trials=8):
    hi_v = h[i]
    hj_v = h[j]
    d_lo = max(-hi_v, hj_v - 1.0)
    d_hi = min(1.0 - hi_v, hj_v)
    width = d_hi - d_lo
    if width < 1e-9:
        return None, None
    current_c5 = exact_c5(h)
    mids = [d_lo, d_hi, (d_lo + d_hi) * 0.5]
    for f in [0.2, 0.4, 0.6, 0.8]:
        mids.append(d_lo + f * width)
    for _ in range(3):
        mids.append(d_lo + rng.random() * width)
    best_c5 = current_c5
    best_h = None
    for d in mids:
        d = float(np.clip(d, d_lo, d_hi))
        g = h.copy()
        g[i] = hi_v + d
        g[j] = hj_v - d
        c5 = exact_c5(g)
        if c5 < best_c5:
            best_c5 = c5
            best_h = g
    if best_h is not None:
        return best_h, best_c5
    return None, None

def active_pairs(h, rng, max_pairs=60):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(corr)
    K = min(16, max(5, L // 12))
    top_lags = np.argsort(corr)[::-1][:K]
    pairs = set()
    for m in top_lags:
        k = int(m) - (n - 1)
        if k >= 0:
            for i in range(n - k):
                j = i + k
                if i != j:
                    pairs.add((min(i, j), max(i, j)))
        else:
            r = -k
            for i in range(r, n):
                j = i - r
                if i != j:
                    pairs.add((min(i, j), max(i, j)))
    sorted_idx = np.argsort(h)
    n_ext = max(4, n // 6)
    for t in range(n_ext):
        a = sorted_idx[t]
        b = sorted_idx[n - 1 - t]
        if a != b:
            pairs.add((min(a, b), max(a, b)))
    n_rand = max(6, n // 5)
    for _ in range(n_rand):
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a != b:
            pairs.add((min(a, b), max(a, b)))
    plist = list(pairs)
    rng.shuffle(plist)
    return plist[:max_pairs]

def pair_descent(h, c5, rng, max_time, start_t):
    n = len(h)
    cur_h = h.copy()
    cur_c5 = c5
    stall = 0
    while monotonic() - start_t < max_time:
        pairs = active_pairs(cur_h, rng, max_pairs=50)
        improved = False
        for (i, j) in pairs:
            new_h, new_c5 = try_pair(cur_h, i, j, rng, n_trials=8)
            if new_h is not None:
                cur_h = new_h
                cur_c5 = new_c5
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 15:
                break
            # Try a wider search with more random pairs
            for _ in range(5):
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a == b:
                    continue
                new_h, new_c5 = try_pair(cur_h, a, b, rng, n_trials=10)
                if new_h is not None:
                    cur_h = new_h
                    cur_c5 = new_c5
                    improved = True
                    stall = 0
                    break
        check_time = (monotonic() - start_t) % 2.0
        if check_time > 1.9 and monotonic() - start_t > max_time:
            break
    return cur_h, cur_c5

        update_best(h_r, c5_r)

    for dim in dims:
        if dim == n_parent:

        t_left = hard_stop - monotonic()
        if t_left < 20:

        h_dim = interpolate_to_dim(_best_h, dim)
        c5_dim = exact_c5(h_dim)
        alloc = min(t_left * 0.25, 180.0)
        if alloc < 10:
            alloc = min(t_left * 0.4, 100.0)

    if t_left > 5 and _best_h is not None:

    if abs(h_out.sum() - target_sum) > 1e-9:

    c5_out = exact_c5(h_out)
    update_best(h_out, c5_out)

    return _best_h, _best_c5, len(_best_h)
