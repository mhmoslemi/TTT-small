Alternative: precompute the correlation matrix and update it incrementally. When I change h[i] by δ, the correlation changes by:
Δc[m] = δ * (contribution of h[i] to c[m])

The contribution of h[i] to c[m] (for lag k = m-(n-1)):
- If k ≥ 0: h[i] appears as h[i'+k] when i'=i-k (if 0≤i-k≤n-1-k), contributing (1-h[i-k])
              h[i] appears in (1-h[i']) when i'=i (if 0≤i≤n-1-k), contributing -h[i+k]
- If k < 0 (r=-k): h[i] appears as h[i'-r] when i'=i+r (if r≤i+r≤n-1, i.e., 0≤i≤n-1-r), contributing (1-h[i+r])
              h[i] appears in (1-h[i']) when i'=i (if r≤i≤n-1), contributing -h[i-r]

So I can maintain the correlation vector and update it in O(n) per coordinate change. This would be much faster than O(n²) for full recomputation.

Let me implement this:

```python
def compute_corr(h):
    """Compute full correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')

def corr_sensitivity(h, i):
    """Compute how c[m] changes when h[i] changes by 1 unit (all other h fixed).
    Returns vector s of length 2n-1 where c_new[m] = c[m] + delta * s[m]."""
    n = len(h)
    s = np.zeros(2 * n - 1)
    
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            # c[m] = sum_{i'=0}^{n-1-k} h[i'+k] * (1-h[i'])
            # h[i] appears as h[i'+k] when i+k = i + k... wait
            # h[i] is the first factor when i = i'+k, i.e., i' = i-k, valid if 0≤i-k≤n-1-k
            if 0 <= i - k <= n - 1 - k:
                s[m] += (1.0 - h[i - k])
            # h[i] appears in (1-h[i']) when i' = i, valid if 0≤i≤n-1-k
            if 0 <= i <= n - 1 - k and i + k < n:
                s[m] -= h[i + k]
        else:
            r = -k
            # c[m] = sum_{i'=r}^{n-1} h[i'-r] * (1-h[i'])
            # h[i] is first factor when i = i'-r, i.e., i' = i+r, valid if r≤i+r≤n-1
            if r <= i + r <= n - 1:
                s[m] += (1.0 - h[i + r])
            # h[i] is in (1-h[i']) when i' = i, valid if r≤i≤n-1
            if r <= i <= n - 1 and i - r >= 0:
                s[m] -= h[i - r]
    
    return s
```

```python
def pair_move_eval(h, corr, i, j, d):
    """Evaluate the new C5 after h[i]+=d, h[j]-=d, using incremental update.
    Returns (new_c5, new_corr) or (inf, None) if infeasible."""
    n = len(h)
    hi_new = h[i] + d
    hj_new = h[j] - d
    if hi_new < -1e-12 or hi_new > 1 + 1e-12 or hj_new < -1e-12 or hj_new > 1 + 1e-12:
        return float('inf'), None
    
    # Compute sensitivity of corr to h[i]
    # For each lag m (k = m - (n-1)):
    #   if k >= 0: contribution from h[i] as first factor: (1-h[i-k]) if 0<=i-k<=n-1-k
    #              contribution from h[i] in (1-h[i']): -h[i+k] if 0<=i<=n-1-k and i+k<n
    #   if k < 0 (r=-k): contribution from h[i] as first factor: (1-h[i+r]) if r<=i+r<=n-1
    #                     contribution from h[i] in (1-h[i']): -h[i-r] if r<=i<=n-1 and i-r>=0
    
    # Vectorized computation:
    k_range = np.arange(-(n-1), n)  # k = m - (n-1) for m = 0, ..., 2n-2
    
    # For k >= 0:
    pos_k = k_range >= 0
    # h[i] as first factor at position i'+k = i, so i' = i-k, valid if 0 <= i-k <= n-1-k
    # This means k <= i and i-k <= n-1-k, i.e., i <= n-1. Always true.
    # So condition: 0 <= i-k, i.e., k <= i. And k >= 0.
    cond_first_pos = pos_k & (k_range <= i) & (i <= n-1)  # k in [0, i]
    # h[i] in (1-h[i']) at position i'=i, valid if 0 <= i <= n-1-k, i.e., k <= n-1-i
    # And i+k < n.
    cond_second_pos = pos_k & (k_range <= n-1-i) & (i+k_range < n)
    
    # For k < 0 (r = -k):
    neg_k = k_range < 0
    r = -k_range  # positive
    # h[i] as first factor at position i'-r = i, so i' = i+r, valid if r <= i+r <= n-1
    # i.e., 0 <= i <= n-1-r = n-1+k. Since k<0, n-1+k < n-1.
    # Condition: i <= n-1-r = n-1+k → k >= i-(n-1). And r >= 1 (k <= -1).
    cond_first_neg = neg_k & (i + r <= n - 1)  # i+r <= n-1, i.e., r <= n-1-i
    # h[i] in (1-h[i']) at position i'=i, valid if r <= i <= n-1, i.e., -k <= i, i.e., k >= -i
    # And i-r >= 0, i.e., i >= r = -k, i.e., k >= -i
    cond_second_neg = neg_k & (i >= r) & (i <= n-1)
    
    s_i = np.zeros(2*n-1)
    s_i[cond_first_pos] += (1.0 - h[i - k_range[cond_first_pos]])
    s_i[cond_second_pos] -= h[i + k_range[cond_second_pos]]
    s_i[cond_first_neg] += (1.0 - h[i + r[cond_first_neg]])
    s_i[cond_second_neg] -= h[i - r[cond_second_neg]]
    
    # Similarly for s_j
    s_j = np.zeros(2*n-1)
    cond_first_pos_j = pos_k & (k_range <= j)
    cond_second_pos_j = pos_k & (k_range <= n-1-j) & (j+k_range < n)
    cond_first_neg_j = neg_k & (j + r <= n - 1)
    cond_second_neg_j = neg_k & (j >= r) & (j <= n-1)
    
    s_j[cond_first_pos_j] += (1.0 - h[j - k_range[cond_first_pos_j]])
    s_j[cond_second_pos_j] -= h[j + k_range[cond_second_pos_j]]
    s_j[cond_first_neg_j] += (1.0 - h[j + r[cond_first_neg_j]])
    s_j[cond_second_neg_j] -= h[j - r[cond_second_neg_j]]
    
    # New correlation
    new_corr = corr + d * s_i - d * s_j
    new_c5 = float(np.max(new_corr) * 2.0 / n)
    
    return new_c5, new_corr
```
