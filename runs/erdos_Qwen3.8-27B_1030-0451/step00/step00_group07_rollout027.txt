- As λ → +∞: F(λ) → N

So there exists a unique λ* with F(λ*) = N/2 (since 0 < N/2 < N).

We can find λ* by bisection.

After finding λ*, the residual sum(clip(y+λ*, 0, 1)) - N/2 should be 0 (up to numerical precision). If there's a tiny residual due to floating point, we can distribute it among the free coordinates (those not at 0 or 1).

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    if N == 0:
        return float('inf')
    dx = 2.0 / N
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals))

def mass_project(x, target):
    """Project x onto {0<=h_i<=1, sum(h_i)=target}."""
    N = len(x)
    x = np.array(x, dtype=np.float64)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    
    # Bisection for lambda
    lo = -2.0
    hi = 2.0
    # Ensure F(lo) <= target <= F(hi)
    while np.sum(np.clip(x + lo, 0, 1)) > target:
        lo -= 1.0
    while np.sum(np.clip(x + hi, 0, 1)) < target:
        hi += 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fine correction
    residual = target - np.sum(result)
    if abs(residual) > 1e-13:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(free):
            n_free = np.sum(free)
            result[free] += residual / n_free
            result = np.clip(result, 0.0, 1.0)
        else:
            # All at bounds; distribute to whichever can absorb
            if residual > 0:
                order = np.argsort(1.0 - result)
            else:
                order = np.argsort(result)
            for idx in order:
                if residual > 0:
                    add = min(residual, 1.0 - result[idx])
                    result[idx] += add
                    residual -= add
                else:
                    rem = min(-residual, result[idx])
                    result[idx] -= rem
                    residual += rem

    """Compute gradient of C[k]*dx with respect to h."""
    t = k - (N - 1)
    g = np.zeros(N)
    if t >= 0:

        # d/dh[m] = (1-h[m-t]) for m in [t, N-1] + (-h[m+t]) for m in [0, N-1-t]
        for m in range(t, N):
            g[m] += (1.0 - h[m - t])
        for m in range(0, N - t):
            g[m] += (-h[m + t])
    else:
        s = -t  # s > 0
        # C[k] = sum_{j=0}^{N-1-s} h[j](1-h[j+s])
        # d/dh[m] = (1-h[m+s]) for m in [0, N-1-s] + (-h[m-s]) for m in [s, N-1]
        for m in range(0, N - s):
            g[m] += (1.0 - h[m + s])
        for m in range(s, N):
            g[m] += (-h[m - s])
    g *= dx
    return g

def upsample(h, new_N):
    """Upsample h (length N) to new_N using piecewise constant averaging."""
    N = len(h)
    if new_N == N:
        return h.copy()
    dx_old = 2.0 / N
    dx_new = 2.0 / new_N
    result = np.zeros(new_N)
    # For each new cell, compute the average of the old cells that overlap
    # New cell i covers [i*dx_new, (i+1)*dx_new]
    # Old cell j covers [j*dx_old, (j+1)*dx_old]
    for i in range(new_N):
        x_start = i * dx_new
        x_end = (i + 1) * dx_new
        total = 0.0
        # Find overlapping old cells
        j_start = int(math.floor(x_start / dx_old))
        j_end = int(math.ceil(x_end / dx_old)) - 1
        j_start = max(0, j_start)
        j_end = min(N - 1, j_end)
        for j in range(j_start, j_end + 1):
            ov_start = max(x_start, j * dx_old)
            ov_end = min(x_end, (j + 1) * dx_old)
            if ov_end > ov_start:
                total += h[j] * (ov_end - ov_start)
        result[i] = total / dx_new
    return result

def downsample(h, new_N):
    """Downsample h (length N) to new_N using averaging."""
    return upsample(h, new_N)  # Same operation works both ways

def subgradient_step(h, alpha, N):

def evaluate_exact(h):
    """Exact C5 evaluation."""
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    if N == 0:
        return float('inf')
    dx = 2.0 / N
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals))

def mass_project(x, target):
    """Euclidean projection onto {0<=h<=1, sum(h)=target}."""
    x = np.array(x, dtype=np.float64, copy=True)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    N = len(x)
    if N == 0:
        return x
    
    # Bisection for lambda
    lo, hi = -3.0, 3.0
    while np.sum(np.clip(x + lo, 0, 1)) > target:
        lo -= 1.0
    while np.sum(np.clip(x + hi, 0, 1)) < target:
        hi += 1.0
    
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(x + mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    
    # Fine correction for floating point
    residual = target - result.sum()
    if abs(residual) > 1e-14:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        nf = int(free.sum())
        if nf > 0:
            result[free] += residual / nf
            result = np.clip(result, 0.0, 1.0)
    return result
