def exact_c(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_pressure(h, n, top_k=50):
    """Compute per-coordinate pressure from top active lags."""
    corr_raw = np.correlate(h, 1-h, mode='full')  # length 2n-1
    cvals = corr_raw * (2.0/n)
    cmax = np.max(cvals)
    
    # Get indices of top lags
    idx = np.argsort(cvals)[::-1][:top_k]

    pressure = np.zeros(n)

def exact_c(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def pressure_vector(h, n):
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    cvals = corr_raw * (2.0 / n)
    cmax = float(np.max(cvals))
    tau = max(1e-4, 0.025 * cmax)
    w = np.exp((cvals - cmax) / tau)
    w = w / w.sum()
    p = np.zeros(n)
    for qi in range(2 * n - 1):
        wi = w[qi]
        if wi < 1e-10:
            continue
        ell = qi - (n - 1)
        if ell == 0:
            p += wi * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            p[ell:] += wi * (1.0 - h[:L])
            p[:L] -= wi * h[ell:]
        else:
            a = -ell
            L = n - a
            p[a:] += wi * (1.0 - h[:L])
            p[:L] -= wi * h[a:]
