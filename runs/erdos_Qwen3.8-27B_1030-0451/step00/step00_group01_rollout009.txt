        no_improve = 0
        while time.monotonic() < deadline - 5.0 and no_improve < 40:
            improved_gen = False
            for i in range(pop_size):
                if time.monotonic() > deadline - 5.0:
                    break
                a, b, c = rng.choice(pop_size, size=3, replace=False)
                trial = pop[a] + 0.6 * (pop[b] - pop[c])
                mask = rng.random(dim_fine) < 0.5
                if not mask.any():
                    mask[rng.integers(dim_fine)] = True
                trial = np.where(mask, trial, pop[i])

                h_trial = _decode(anchor, basis, trial, scale_fine, n)
                s_trial = _exact_score(h_trial)

                if s_trial <= scores[i] + 1e-14:
                    pop[i] = trial
                    scores[i] = s_trial
                    improved_gen = True
                    if s_trial < best_score - 1e-12:
                        best_score = s_trial
                        best_h = h_trial.copy()
                        anchor = best_h.copy()
            no_improve = 0 if improved_gen else no_improve + 1

    # Phase 4: Direct coordinate perturbations
    if time.monotonic() < deadline - 3.0:
        for _ in range(5):
            if time.monotonic() > deadline - 3.0:
                break
            improved = False
            for i in range(n):
                if time.monotonic() > deadline - 3.0:
                    break
                for delta in [0.01, -0.01]:
                    h_try = anchor.copy()
                    h_try[i] += delta
                    h_try = _project_feasible(h_try, n)
                    s_try = _exact_score(h_try)
                    if s_try < best_score - 1e-13:
                        best_score = s_try
                        best_h = h_try.copy()
                        anchor = best_h.copy()
                        improved = True
            if not improved:
                break

    # Final validation
    best_h = _project_feasible(best_h, n)
    final_score = _exact_score(best_h)
    if final_score > best_score + 1e-9:
        # Something went wrong, try to recover
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _project_feasible(best_h, n)
        final_score = _exact_score(best_h)

    return (best_h, final_score, n)
