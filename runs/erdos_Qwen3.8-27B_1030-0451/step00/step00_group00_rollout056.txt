import numpy as np

def score_exact(h):
    n = len(h)
    if n < 4:
        return float('inf'), None
    dx = 2.0 / n
    Q = np.correlate(h, 1.0 - h, mode='full')
    c = float(np.max(Q)) * dx
    return c, Q

def compute_AG(h, u, v):
    n = len(h)
    L = 2 * n - 1
    m = abs(u - v)
    A = np.zeros(L)
    G = np.zeros(L)
    offset = n - 1
    
    # G at positions offset+m and offset-m
    r_pos = offset + m
    r_neg = offset - m
    if 0 <= r_pos < L:
        G[r_pos] = 1.0
    if 0 <= r_neg < L:
        G[r_neg] = 1.0
    
    # A[r] = p_u(r) - p_v(r)
    # p_t(r) = I(0<=t+d<n)*(1-h[t+d]) - I(0<=t-d<n)*h[t-d]
    # where d = r - offset
    for r in range(L):
        d = r - offset
        # p_u
        pu = 0.0
        k1 = u + d
        if 0 <= k1 < n:
            pu += (1.0 - h[k1])
        k2 = u - d
        if 0 <= k2 < n:
            pu -= h[k2]
        # p_v
        pv = 0.0
        k3 = v + d
        if 0 <= k3 < n:
            pv += (1.0 - h[k3])
        k4 = v - d
        if 0 <= k4 < n:
            pv -= h[k4]
        A[r] = pu - pv
    
    return A, G

def line_minimize(Q, A, G, s_lo, s_hi, n_iters=50):
    # Minimize F(s) = max(Q + A*s + G*s^2) over s in [s_lo, s_hi]
    # F is convex
    # Golden section search
    if s_hi - s_lo < 1e-14:
        return (s_lo + s_hi) / 2.0
    
    # Check endpoints and midpoint first
    candidates = [s_lo, s_hi, (s_lo + s_hi) / 2.0]
    
    def F(s):
        vals = Q + A * s + G * s * s
        return np.max(vals)
    
    best_s = s_lo
    best_f = F(s_lo)
    for s in candidates[1:]:
        f = F(s)
        if f < best_f:
            best_f = f
            best_s = s
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = s_lo, s_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = F(c)
    fd = F(d)
    
    for _ in range(n_iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = F(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = F(d)
    
    s_mid = (a + b) / 2.0
    f_mid = F(s_mid)
    if f_mid < best_f:
        best_f = f_mid
        best_s = s_mid
    
    return best_s
