        mid = (lo + hi) * 0.5
        if float(np.sum(np.clip(x - mid, 0.0, 1.0))) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_logsumexp(corr, h, n, dx, M, threshold=1e-12):
    max_c = float(np.max(corr))
    exp_c = np.exp(M * (corr - max_c))
    s = float(np.sum(exp_c))
    if s < 1e-300:
        return np.zeros(n)
    w = exp_c / s
    grad = np.zeros(n)
    active = np.where(w > threshold)[0]
    for k in active:
        wk = w[k]
        d = int(k) - (n - 1)
        if d >= 0:
            m1 = np.arange(d, n)
            if len(m1) > 0:
                grad[m1] += wk * dx * (1.0 - h[m1 - d])
            m2 = np.arange(0, n - d)
            if len(m2) > 0:
                grad[m2] -= wk * dx * h[m2 + d]
        else:
            dd = -d
            m1 = np.arange(0, n - dd)
            if len(m1) > 0:
                grad[m1] += wk * dx * (1.0 - h[m1 + dd])
            m2 = np.arange(dd, n)
            if len(m2) > 0:
                grad[m2] -= wk * dx * h[m2 - dd]
    return grad

def optimize_from(h, n, dx, target_sum, budget_s, t0, M_start=20.0, M_end=300.0, lr_start=0.03, max_iters=200000):
    best_c5 = float(np.max(compute_corr(h, dx)))
    best_h = h.copy()
    M = M_start
    lr = lr_start
    
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 5:
            break
        corr = compute_corr(h, dx)
        obj = float(np.max(corr))
        if obj < best_c5 - 1e-12:
            best_c5 = obj
            best_h = h.copy()
        
        grad = grad_logsumexp(corr, h, n, dx, M)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        step = lr
        accepted = False
        for _ in range(8):
            h_try = project_to_box_sum(h - step * grad, target_sum)
            obj_try = float(np.max(compute_corr(h_try, dx)))
            if obj_try < obj + 1e-14:
                h = h_try
                accepted = True
                break
            step *= 0.5
        if not accepted:
            lr *= 0.85
            if lr < 1e-12:
                break
        
        progress = it / max(1.0, max_iters)
        M = M_start + (M_end - M_start) * progress
        lr = max(lr_start * 0.5, lr * 1.001)
    
    return h, best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h0 = np.array(initial_h_values, dtype=np.float64)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def eval_c5(h):
        return float(np.max(compute_corr(h, dx)))
    
    # Phase 1: Optim
    h = project_to_box_sum(h0.copy(), target_sum)
    h, c5_1, best_h_1 = optimize_from(h, n, dx, target_sum, budget_s, t0,
                                        M_start=15.0, M_end=200.0, lr_start=0.05, max_iters=80000)
    best_c5 = c5_1
    best_h = best_h_1.copy()
    
    # Phase 2: Sharpened optimization from best
    if time.time() - t0 < budget_s - 30:
        h2, c5_2, best_h_2 = optimize_from(best_h.copy(), n, dx, target_sum, budget_s, t0,
                                            M_start=200.0, M_end=500.0, lr_start=0.015, max_iters=60000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = best_h_2.copy()
    
    # Phase 3: Random restarts
    if time.time() - t0 < budget_s - 20:
        for restart in range(30):
            if time.time() - t0 > budget_s - 10:
                break
            noise_scale = 0.03 + 0.02 * restart
            h_r = best_h + np.random.randn(n) * noise_scale
            h_r = project_to_box_sum(h_r, target_sum)
            h_r, c5_r, best_h_r = optimize_from(h_r, n, dx, target_sum, budget_s, t0,
                                                 M_start=100.0, M_end=400.0, lr_start=0.01, max_iters=20000)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = best_h_r.copy()
    
    # Final verification
    final_c5 = eval_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, n
