import numpy as np
import time


def _project_feasible(z, n):
    """Project z to {h in [0,1]^n : sum(h) = n/2} via bisection on lambda."""
    target = n * 0.5
    lo = float(np.min(z)) - 2.0
    hi = float(np.max(z)) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(z - lam, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        interior = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(interior) > 1:
            per = residual / len(interior)
            h[interior] += per
            h = np.clip(h, 0.0, 1.0)
            residual2 = target - np.sum(h)
            if abs(residual2) > 1e-13:
                h[interior[0]] += residual2
                h = np.clip(h, 0.0, 1.0)
        elif len(interior) == 1:
            h[interior[0]] += residual
            h = np.clip(h, 0.0, 1.0)
    return h


def _score(h):
    """Compute exact C5 = max_k integral h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(res) * dx)


def _build_basis(n, d):
    """Build d Fourier basis modes (sin/cos pairs at increasing frequencies)."""
    x = (np.arange(n) + 0.5) / float(n)
    B = np.zeros((n, d))
    for j in range(d):
        k = j // 2 + 1
        if j % 2 == 0:
            B[:, j] = np.sin(2.0 * np.pi * k * x)
        else:
            B[:, j] = np.cos(2.0 * np.pi * k * x)
        mn = float(np.max(np.abs(B[:, j])))
        if mn > 1e-15:
            B[:, j] /= mn
    return B


def _resample(h, new_n):
    """Resample h to new length via linear interpolation, then project."""
    n = len(h)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(new_n) + 0.5) / float(new_n)
    h_new = np.interp(x_new, x_old, h)
    return _project_feasible(h_new, new_n)


def _eval_latent(anchor, basis, b, delta, n):
    """Decode latent vector to feasible h and return (h, score)."""
    z = anchor + delta * (basis @ b)
    h = _project_feasible(z, n)
    c = _score(h)
    return h, c


def _de_run(anchor, n, d, delta, pop_size, max_evals, rng, best_c, best_h):
    """Differential evolution in latent subspace. Returns (best_c, best_h)."""
    basis = _build_basis(n, d)
    pop = np.zeros((pop_size, d))
    for i in range(1, pop_size):
        pop[i] = rng.uniform(-1.0, 1.0, d)

    costs = np.full(pop_size, 999.0)
    eval_count = 0

    # Evaluate initial population
    for i in range(pop_size):
        h, c = _eval_latent(anchor, basis, pop[i], delta, n)
        costs[i] = c
        eval_count += 1
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_cost = np.min(costs)
    stagnation = 0

    for gen in range(5000):
        if eval_count >= max_evals:
            break
        improved_this_gen = False
        F = 0.5 + 0.3 * rng.random()
        CR = 0.7 + 0.2 * rng.random()

        for i in range(pop_size):
            if eval_count >= max_evals:
                break
            # DE/rand/1 mutation
            choices = rng.choice(pop_size, size=3, replace=False)
            r1, r2, r3 = int(choices[0]), int(choices[1]), int(choices[2])
            v = pop[r1] + F * (pop[r2] - pop[r3])

            # Binomial crossover
            gates = rng.random(d) < CR
            if not np.any(gates):
                gates[rng.integers(0, d)] = True
            trial = pop[i].copy()
            trial[gates] = v[gates]
            trial = np.clip(trial, -5.0, 5.0)

            h, c = _eval_latent(anchor, basis, trial, delta, n)
            eval_count += 1

            if c <= costs[i]:
                pop[i] = trial
                costs[i] = c
                improved_this_gen = True
                if c < best_c:
                    best_c = c
                    best_h = h.copy()

        if improved_this_gen:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 8:
                # Reset worst individual
                worst = int(np.argmax(costs))
                best_idx = int(np.argmin(costs))
                pop[worst] = pop[best_idx] + rng.uniform(-0.3, 0.3, d)

                if c < best_c:

def _direct_perturb(anchor, n, num_steps, step_size, rng, best_c, best_h):
    """Try direct coordinate perturbations with projection."""
    improved = 0
    for _ in range(num_steps):
        idx = rng.integers(0, n)
        direction = rng.choice([-1.0, 1.0])
        z = anchor.copy()
        z[idx] += direction * step_size

        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            improved += 1
            anchor = h.copy()
    return best_c, best_h, improved
