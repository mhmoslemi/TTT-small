            grad, scaled, p = compute_smooth_gradient(h, T, top_k)
            grad_c = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_c)
            if gnorm < 1e-16:
                break
            
            accepted = False
            for trial_factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                trial_lr = lr * trial_factor
                h_new = h - trial_lr * grad_c
                h_new = project_to_feasible(h_new, n)
                _, c5_new = compute_c5_full(h_new)
                if c5_new <= c5_cur + 1e-14:
                    h = h_new
                    c5_cur = c5_new
                    if c5_new < best_ref[0]:
                        best_ref[0] = c5_new
                        best_ref[1] = h.copy()
                    accepted = True
                    break
            
            if not accepted:
                lr *= 0.7
                T = max(T * 0.95, 0.0005)
                if lr < 1e-12 and T <= 0.0006:
                    break
    
    # Final verification
    best_h = project_to_feasible(best_ref[1].copy(), len(best_ref[1]))
    _, c5_final = compute_c5_full(best_h)
    
    return best_h, c5_final, len(best_h)
