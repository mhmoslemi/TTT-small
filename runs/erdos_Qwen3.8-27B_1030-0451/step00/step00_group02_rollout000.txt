def project_feasible(z, target):
    """Project z onto {0 <= h <= 1, sum(h) = target} using bisection on offset."""
    n = len(z)
    lo = -2.0

        starts.append(gen_random_start(n_p, rng))
    for _ in range(3):
        starts.append(gen_threshold_start(h_p, rng))

    # Evaluate and record starts
    for s in starts:
        if is_feasible(s):
            c_s = exact_score(s)
            record(s, c_s, n_p)

    # Phase 1: Local search on best candidate
    if time.time() < deadline:
        t_phase1 = min(300.0, (deadline - time.time()) * 0.4)
        best_h, best_c = local_search(best_h, t_phase1, seed=seed)
        best_n = len(best_h)
        record(best_h, best_c, best_n)

    # Phase 2: SLSQP on best
    if time.time() < deadline:
        t_left = deadline - time.time()
        if t_left > 60:
            t_phase2 = min(200.0, t_left * 0.35)
            h_sl, c_sl = slsqp_optimize(best_h, time_limit_per_run=t_phase2 / 3.0, max_runs=3)
            record(h_sl, c_sl, len(h_sl))
            # Local refine after SLSQP
            if time.time() < deadline - 30:
                t_rem = deadline - time.time() - 30
                hr, cr = local_search(h_sl, min(100.0, t_rem * 0.5), seed=seed + 1)
                record(hr, cr, len(hr))
            best_h = h_sl
            best_c = c_sl
            best_n = len(h_sl)

    # Phase 3: Try different starts
    if time.time() < deadline:
        t_left = deadline - time.time()
        if t_left > 60:
            for i, s in enumerate(starts[1:5]):
                if time.time() > deadline - 30:
                    break
                t_alloc = min(80.0, (deadline - time.time() - 30) / max(1, (5 - i)))
                if t_alloc < 10:
                    break
                hr, cr = local_search(s.copy(), t_alloc, seed=seed + 100 + i)
                record(hr, cr, len(hr))

    # Phase 4: Try different dimensions
    if time.time() < deadline:
        t_left = deadline - time.time()
        if t_left > 80 and best_n <= 200:
            for new_n in [best_n - 1, best_n + 1, int(best_n * 1.3), int(best_n * 2.0)]:
                if time.time() > deadline - 40:
                    break
                if new_n < 20 or new_n > 300:
                    continue
                h_trans = interp_to_dim(best_h, new_n)
                c_trans = exact_score(h_trans)
                record(h_trans, c_trans, new_n)
                if c_trans < best_c + 0.02:
                    t_alloc = min(60.0, max(10.0, (deadline - time.time() - 40) * 0.3))
                    hr, cr = local_search(h_trans, t_alloc, seed=seed + 500)
                    record(hr, cr, new_n)

    # Phase 5: Final refinement
    if time.time() < deadline - 10 and best_h is not None:
        t_final = max(5.0, deadline - time.time() - 10)
        hr, cr = local_search(best_h, t_final, seed=seed + 999)
        record(hr, cr, len(hr))

    # Final validation
    if best_h is None:
        best_h = np.full(81, 0.5)
        best_n = 81
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c = exact_score(best_h)

    return best_h, best_c, best_n
