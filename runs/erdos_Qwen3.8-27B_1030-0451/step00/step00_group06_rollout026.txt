    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    h = np.empty(N)
    M = N // 2
    h[:M] = x
    if N % 2 == 1:
        h[M] = 0.5
        h[M+1:] = 1.0 - x[::-1]

    """Smoothed objective using log-sum-exp."""
    x = np.clip(x, 0.0, 1.0)

    return compute_c5(h)

def obj_smooth_func(x, N, alpha):
    h = assemble_anti(x, N)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    cmax = np.max(c)
    return cmax + np.log(np.sum(np.exp(alpha * (c - cmax)))) / alpha

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=np.float64)
    except:
        h_parent = None
    
    N = len(h_parent) if h_parent is not None else 417
    M = N // 2
    
    # Extract anti-palindromic parameters from parent
    if h_parent is not None:
        x0 = np.clip(h_parent[:M].copy(), 0.0, 1.0)
    else:
        x0 = np.linspace(0.1, 0.9, M)
    
    # Evaluate
    h0 = assemble_anti(x0, N)
    c5_0 = compute_c5(h0)
    
    # Set initial best
    if h_parent is not None:
        c5_parent = compute_c5(h_parent)
        if c5_parent <= c5_0:
            best_h = h_parent.copy()
            best_c5 = c5_parent
        else:
            best_h = h0.copy()
            best_c5 = c5_0
    else:
        best_h = h0.copy()
        best_c5 = c5_0
    
    # Calibrate evaluation time
    t0 = time.time()
    for _ in range(200):
        obj_exact_clipped(x0[:50], min(101, N))  # quick test
    # Actually just time one real eval
    t0 = time.time()
    obj_exact_clipped(x0, N)
    t_eval = max(time.time() - t0, 1e-6)
    
    def elapsed():
        return time.time() - t_start
    
    def remaining():
        return budget_s - elapsed()
    
    # Phase 1: Optimize from parent with L-BFGS-B continuation
    x_current = x0.copy()
    if remaining() > 100:
        for alpha in [20, 50, 100, 300]:
            if remaining() < 50:
                break
            try:
                res = minimize(obj_smooth_func, x_current, args=(N, alpha),
                             method='L-BFGS-B', bounds=[(0.0, 1.0)] * M,
                             options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10})
                x_candidate = np.clip(res.x, 0.0, 1.0)
                c5_cand = obj_exact_clipped(x_candidate, N)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = assemble_anti(x_candidate, N).copy()
                x_current = x_candidate
            except:
                pass
    
    # Phase 1b: Powell polish
    if remaining() > 30:
        try:
            maxfev = min(20000, int(remaining() * 0.3 / t_eval))
            maxfev = max(maxfev, 5000)
            res = minimize(obj_exact_clipped, x_current, args=(N,), method='Powell',
                         options={'maxfev': maxfev, 'xtol': 1e-13, 'ftol': 1e-15})
            x_candidate = np.clip(res.x, 0.0, 1.0)
            c5_cand = obj_exact_clipped(x_candidate, N)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = assemble_anti(x_candidate, N).copy()
            x_current = x_candidate
        except:
            pass
    
    # Phase 2: Many restarts
    # Time per restart: L-BFGS-B (~50 evals) + Powell (~2000 evals) ≈ 2050 * t_eval
    t_per_restart = 2500 * t_eval
    n_restarts = int(min(200, max(20, remaining() * 0.5 / t_per_restart)))
    
    for r in range(n_restarts):
        if remaining() < 100:
            break
        
        # Generate starting point
        if r < 3:
            x_r = x_current + rng.normal(0, 0.02 * (r+1), M)
        elif r % 5 == 0:
            x_r = rng.uniform(0, 1, M)
        elif r % 5 == 1:
            t = np.linspace(0, 2*np.pi, M)
            x_r = 0.5 + rng.uniform(0.1, 0.3)*np.sin(t) + rng.uniform(0.05, 0.2)*np.cos(2*t)
            x_r = np.clip(x_r, 0, 1)
        elif r % 5 == 2:
            bp = rng.integers(M//4, 3*M//4)
            lo = rng.uniform(0, 0.3)
            hi = rng.uniform(0.7, 1)
            x_r = np.concatenate([np.full(bp, lo), np.full(M-bp, hi)])
            x_r += rng.normal(0, 0.08, M)
            x_r = np.clip(x_r, 0, 1)
        elif r % 5 == 3:
            x_r = np.linspace(rng.uniform(0,0.3), rng.uniform(0.7,1), M)
            x_r += rng.normal(0, 0.05, M)
            x_r = np.clip(x_r, 0, 1)
        else:
            # Perturbed best
            x_best_left = np.clip(best_h[:M], 0, 1) if len(best_h) == N else x0
            x_r = x_best_left + rng.normal(0, 0.1, M)
            x_r = np.clip(x_r, 0, 1)
        
        # L-BFGS-B warm start
        x_local = x_r.copy()
        try:
            res = minimize(obj_smooth_func, x_local, args=(N, 100),
                         method='L-BFGS-B', bounds=[(0.0, 1.0)] * M,
                         options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-8})
            x_local = np.clip(res.x, 0.0, 1.0)
        except:
            pass
        
        # Powell polish
        try:
            maxfev = min(5000, int(remaining() * 0.02 / t_eval))
            maxfev = max(maxfev, 1000)
            res = minimize(obj_exact_clipped, x_local, args=(N,), method='Powell',
                         options={'maxfev': maxfev, 'xtol': 1e-11, 'ftol': 1e-13})
            x_local = np.clip(res.x, 0.0, 1.0)
        except:
            pass
        
        c5_local = obj_exact_clipped(x_local, N)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = assemble_anti(x_local, N).copy()
    
    # Phase 3: Try nearby N values
    if remaining() > 60:
        for dN in [2, -2, 4, -4, 34, -34]:
            N_try = N + dN
            if N_try % 2 == 0:
                N_try += 1
            if N_try < 100 or N_try > 2000 or remaining() < 40:
                continue
            M_try = N_try // 2
            # Interpolate
            old_x = (np.arange(len(best_h)) + 0.5) * 2.0 / len(best_h)
            new_x = (np.arange(N_try) + 0.5) * 2.0 / N_try
            h_interp = np.clip(np.interp(new_x, old_x, best_h), 0, 1)
            x_try = h_interp[:M_try].copy()
            
            try:
                res = minimize(obj_smooth_func, x_try, args=(N_try, 200),
                             method='L-BFGS-B', bounds=[(0.0, 1.0)] * M_try,
                             options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-9})
                x_try_opt = np.clip(res.x, 0.0, 1.0)
                
                maxfev = min(5000, int(remaining() * 0.03 / t_eval))
                res = minimize(obj_exact_clipped, x_try_opt, args=(N_try,), method='Powell',
                             options={'maxfev': maxfev, 'xtol': 1e-12, 'ftol': 1e-14})
                x_try_opt = np.clip(res.x, 0.0, 1.0)
            except:
                x_try_opt = x_try
            
            h_try = assemble_anti(x_try_opt, N_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
