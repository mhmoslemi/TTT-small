                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40})
    return np.clip(result.x, 0.0, 1.0)

def coordinate_refine(p, n_rounds=5, delta=0.05, n_grid=11, time_limit=None, t_ref=None):
    m = len(p)
    best_val = exact_objective(build_h_from_p(p))

    for rnd in range(n_rounds):
        if time_limit is not None and t_ref is not None:
            if time.time() - t_ref > time_limit:
                break
        improved = False
        for i in range(m):
            lo = max(0.0, p[i] - delta)
            hi = min(1.0, p[i] + delta)
            if lo >= hi:
                continue
            candidates = np.linspace(lo, hi, n_grid)
            cur = p[i]
            best_local = cur
            best_local_val = best_val
            for c in candidates:
                if abs(c - cur) < 1e-12:
                    continue
                p[i] = c
                val = exact_objective(build_h_from_p(p))
                if val < best_local_val:
                    best_local_val = val
                    best_local = c
            p[i] = best_local
            if best_local_val < best_val - 1e-15:
                best_val = best_local_val
                improved = True
        if not improved:
            break
    return p, best_val

def transfer_p(p_src, n_tgt):
    h_src = build_h_from_p(p_src)
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    h_tgt = np.interp(x_tgt, x_src, h_src, period=2.0)
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    p_tgt = extract_p_from_h(h_tgt)
    return p_tgt

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 10 - (time.time() - t_start)

    # Get parent
    parent = initial_h_values.astype(np.float64).copy()
    n_parent = len(parent)

    # Project to antisymmetric manifold
    p_parent = extract_p_from_h(parent)
    h_anti = build_h_from_p(p_parent)
    val_anti = exact_objective(h_anti)
    val_parent = exact_objective(parent)

    best_p = p_parent.copy()
    best_val = min(val_anti, val_parent)
    if val_parent < val_anti:
        # Parent is better non-antisymmetric, but we can only work with antisymmetric
        # Use the antisymmetric projection as our working point
        best_val = val_anti

    # Stage 1: L-BFGS-B with increasing smoothness
    betas_seq = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    for beta in betas_seq:
        if time_left() < 90:
            break
        p_opt = optimize_l_bfgs(best_p, beta, maxiter=120)
        val_opt = exact_objective(build_h_from_p(p_opt))
        if val_opt < best_val:
            best_p = p_opt.copy()
            best_val = val_opt

    # Stage 2: Coordinate refinement
    if time_left() > 30:
        best_p, best_val = coordinate_refine(best_p, n_rounds=6, delta=0.05, n_grid=13,
                                             time_limit=time_left() - 10, t_ref=t_start)

    # Stage 3: Multi-resolution search
    trial_ns = [256, 200, 338, 400, 512, 64]
    for n_try in trial_ns:
        if time_left() < 120:
            break
        if n_try % 2 != 0:
            n_try += 1
        if n_try == len(best_p) * 2:
            continue  # skip same resolution

        p_try = transfer_p(best_p, n_try)

        # Optimize
        for beta in [10.0, 50.0]:
            if time_left() < 60:
                break
            p_try = optimize_l_bfgs(p_try, beta, maxiter=80)

        val_try = exact_objective(build_h_from_p(p_try))

        if val_try < best_val:
            best_p = p_try.copy()
            best_val = val_try
            if time_left() > 30:
                best_p, best_val = coordinate_refine(best_p, n_rounds=4, delta=0.03, n_grid=11,
                                                     time_limit=time_left() - 10, t_ref=t_start)

    # Stage 4: Random perturbations + restart
    best_p_copy = best_p.copy()
    for pert_scale in [0.02, 0.01, 0.005]:
        if time_left() < 60:
            break
        p_pert = best_p_copy + rng.normal(0, pert_scale, size=best_p_copy.shape[0])
        p_pert = np.clip(p_pert, 0.0, 1.0)
        p_pert = optimize_l_bfgs(p_pert, 50.0, maxiter=80)
        val_pert = exact_objective(build_h_from_p(p_pert))
        if val_pert < best_val:
            best_p = p_pert.copy()
            best_val = val_pert
            best_p_copy = best_p.copy()
            best_p, best_val = coordinate_refine(best_p, n_rounds=4, delta=0.03, n_grid=11,
                                                 time_limit=time_left() - 10, t_ref=t_start)

    # Stage 5: Final fine coordinate refinement
    if time_left() > 15:
        best_p, best_val = coordinate_refine(best_p, n_rounds=10, delta=0.02, n_grid=17,
                                             time_limit=time_left() - 5, t_ref=t_start)

    # Final: use the best resolution
    h_final = build_h_from_p(best_p)
    h_final = np.clip(h_final, 0.0, 1.0)
    final_val = exact_objective(h_final)
    n_final = len(h_final)

    return h_final, final_val, n_final
