def compute_c5(h, dx):

def gradient_of_max_shift(h, dx, n):
    """Compute gradient of the active (max) correlation term"""
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    max_idx = np.argmax(corr)
    d = max_idx - (n - 1)  # shift
    
    # Gradient of S(h, d) w.r.t. h
    # For shift d: S = sum_i h[i] * (1-h[i+d]) over valid i
    # dS/dh[j] = (1-h[j+d]) - h[j-d] (with boundary handling)
    grad = np.zeros(n)
    if d >= 0:
        # Valid i: 0 <= i < n and 0 <= i+d < n, so 0 <= i < n-d
        for i in range(n - d):
            grad[i] += (1 - h[i + d])
            grad[i + d] += -h[i]  # from -h[i]*h[i+d] term, derivative w.r.t. h[i+d]
    else:
        d_abs = -d
        for i in range(d_abs, n):
            grad[i] += (1 - h[i - d_abs])
            grad[i - d_abs] += -h[i]
    
    return grad * dx

import numpy as np
import time
import cvxpy as cp

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection on shift."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    """Compute C5 value from h array (assumes n = len(h))."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def grad_shift(h, d, n):
    """Gradient of S(h,d) = sum_{valid i} h[i]*(1-h[i+d]) w.r.t. h."""
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < d < n:
        g[:n-d] += 1.0 - h[d:]
        g[d:] += -h[:n-d]
    elif -n < d < 0:
        da = -d
        g[da:] += 1.0 - h[:n-da]
        g[:n-da] += -h[da:]
    return g

def projected_gradient(h, dx, n, lr, top_k=5):
    """One step of projected subgradient descent on max_k S(h,k)*dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        d = idx - (n - 1)
        grad += grad_shift(h, d, n)
    grad *= dx / top_k
    h_new = h - lr * grad
    return project_feasible(h_new, n / 2.0)

def lp_cutting_plane(h, dx, n, max_cp_iters=30):
    """LP-based refinement using cutting plane."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_S = corr.max()
    
    h_var = cp.Variable(n)
    t = cp.Variable(initialize=max_S * dx)
    base_constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    # Start with top-10 active constraints
    top_idx = np.argsort(corr)[-10:]
    cut_constraints = []
    for idx in top_idx:
        d = idx - (n - 1)
        g = grad_shift(h, d, n)
        c0 = corr[idx]
        cut_constraints.append(c0 + float(g @ (h_var - h)) <= t)
    
    for cp_iter in range(max_cp_iters):
        constraints = base_constraints + cut_constraints
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver='CLARABEL', verbose=False)
        except:
            break
        
        if prob.status != 'optimal' or h_var.value is None:
            break
        
        h_new = h_var.value
        t_val = t.value
        
        # Check for violated cuts
        violated = []
        for k in range(2 * n - 1):
            d = k - (n - 1)
            g = grad_shift(h, d, n)
            lin_val = corr[k] + float(np.dot(g, h_new - h))
            if lin_val > t_val + 1e-6:
                violated.append((lin_val - t_val, k))
        
        if not violated:
            break
        
        # Add most violated cut
        violated.sort(reverse=True)
        _, k_new = violated[0]
        d = k_new - (n - 1)
        g = grad_shift(h, d, n)
        c0 = corr[k_new]
        cut_constraints.append(c0 + float(g @ (h_var - h)) <= t)
    
    if h_var.value is not None:
        h_final = np.clip(h_var.value, 0.0, 1.0)
        h_final = project_feasible(h_final, n / 2.0)
        return h_final
    return h.copy()
