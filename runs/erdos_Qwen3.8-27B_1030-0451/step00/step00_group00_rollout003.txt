```python
def minimize_1d_convex(Q, A, G, s_lo, s_hi, max_iter=100):
    """Minimize F(s) = max_r(Q[r] + A[r]*s + G[r]*s^2) over [s_lo, s_hi]."""
    if s_hi - s_lo < 1e-15:
        return 0.5 * (s_lo + s_hi)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    a, b = s_lo, s_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    fc = np.max(Q + A * c + G * c * c)
    fd = np.max(Q + A * d + G * d * d)
    
    for _ in range(max_iter):
        b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = np.max(Q + A * c + G * c * c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = np.max(Q + A * d + G * d * d)
    
    return 0.5 * (a + b)
```

```python
    """Select a pair based on the current active lags."""
    max_q = np.max(Q)
    active = np.where(Q >= max_q - 1e-8)[0]
    
    # For each active lag, compute the subgradient vector p_t(r)
    # and find the "best" (u, v) pair
    # For simplicity, just use the first active lag
    r_star = active[0]
    d = r_star - offset
    
    # p_t = I(0 <= t-d < n)*(1-h[t-d]) - I(0 <= t+d < n)*h[t+d]
```

```python
    hi1 = min(n, n + d)  # exclusive
    if lo1 < hi1:
        p[lo1:hi1] += (1.0 - h[lo1-d:hi1-d])
    
    # Second term: I(0 <= t+d < n)*h[t+d]
    # t+d >= 0 means t >= -d; t+d < n means t < n-d
    # So t ranges from max(0, -d) to min(n-1, n-d-1)
    lo2 = max(0, -d)
    hi2 = min(n, n - d)  # exclusive
    if lo2 < hi2:
        p[lo2:hi2] -= h[lo2+d:hi2+d]
    
    return p
```

I want this to be negative. So I want p_u(r) - p_v(r) < 0 for all active r (or at least for the "dominant" ones).
