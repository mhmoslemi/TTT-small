def rank_fill(n, scores):
    target = n / 2.0
    order = np.argsort(scores)[::-1]
    h = np.zeros(n)
    m = int(math.floor(target + 1e-12))
    frac = target - m
    for k in range(min(m, n)):
        h[order[k]] = 1.0
    if frac > 1e-10 and m < n:
        h[order[m]] = frac
    return h

def sa_optimize(h, n, T, n_iters, rng, best_h, best_c5, t_start, t_budget):
    """Run SA with pair moves. Returns updated best."""
    cur_h = h.copy()
    cur_c = exact_score(cur_h)
    
    T_min = 1e-7
    T_decay = 0.997
    reheat_every = 3000
    
    for it in range(n_iters):
        if time.monotonic() - t_start > t_budget * 0.93:
            break
        
        # Pick pair
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        hi = cur_h[i]
        hj = cur_h[j]
        
        # h[i] += t, h[j] -= t
        t_lo = max(-hi, hj - 1.0)
        t_hi = min(1.0 - hi, hj)
        
        if t_hi - t_lo < 1e-12:
            continue
        
        t = float(rng.uniform(t_lo, t_hi))
        
        cur_h[i] += t
        cur_h[j] -= t
        
        new_c = exact_score(cur_h)
        delta = new_c - cur_c
        
        accept = False
        if delta < -1e-14:
            accept = True
        elif T > T_min:
            if delta < 2.0 * T * 10:  # avoid overflow
                if float(rng.random()) < math.exp(-delta / T):
                    accept = True
        
        if accept:
            cur_c = new_c
            if new_c < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = new_c
        else:
            cur_h[i] -= t
            cur_h[j] += t
        
        T = max(T_min, T * T_decay)
        if (it + 1) % reheat_every == 0:
            T = min(T * 2.5, 0.1)
    
    return cur_h, cur_c, best_h, best_c5

def make_score_vector(n, variant, seed_val):
    """Generate a score vector for rank-fill."""
    idx = np.arange(n, dtype=float)
    r = np.random.default_rng(seed_val)
    
    if variant == 0:
        phi = (1.0 + math.sqrt(5.0)) / 2.0
        return np.abs(np.sin(idx * phi + 0.1 * variant))
    elif variant == 1:
        return np.abs(np.sin(idx * math.sqrt(2.0) + 0.3))
    elif variant == 2:
        return r.random(n)
    elif variant == 3:
        # Low-discrepancy: use Kronecker sequence
        return np.abs(np.sin(2.0 * math.pi * (idx * ((math.sqrt(5.0) - 1.0) / 2.0) % 1.0)))
    elif variant == 4:

        return np.where((idx % 2 == 0), 1.0, 0.5) + r.random(n) * 0.1
    elif variant == 5:
        return np.sin(idx * 0.5) * np.cos(idx * 1.3)
    elif variant == 6:
        # Wavelength 3 pattern
        return 0.5 + 0.5 * np.sin(2.0 * math.pi * idx / 3.0)
    elif variant == 7:
        # Two-scale
        return 0.5 * np.sin(2.0 * math.pi * idx / 7.0) + 0.3 * np.sin(2.0 * math.pi * idx / 3.0 + 1.0)
    elif variant == 8:

    elif variant == 9:

        for i in range(n):
            t[i] = bin(i).count('1') % 2
        return t + r.random(n) * 0.1
    else:

def lse_smooth_objective(h, n, tau):
    """Log-sum-exp smoothing of the max correlation."""
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full") * dx
    maxc = np.max(corr)
    exps = np.exp((corr - maxc) / tau)
    return maxc + tau * math.log(np.sum(exps))

            deltas = np.linspace(-0.1, 0.1, grid_size)
            best_t = old_val
            best_local_c = cur_c

                if c < best_local_c:
                    best_local_c = c
                    best_t = new_val
            h[i] = best_t
            if best_t != old_val:
                improved = True
                cur_c = best_local_c
            else:
                h[i] = old_val
        if not improved:
            break
    return cur_c

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return float(np.max(corr) * dx)


def repair_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff < 0:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = -diff / float(np.sum(mask))
            h[mask] = np.maximum(h[mask] - step, 0.0)
        else:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = diff / float(np.sum(mask))
            h[mask] = np.minimum(h[mask] + step, 1.0)
    h = np.clip(h, 0.0, 1.0)
    err = target - np.sum(h)
    if abs(err) > 1e-12:
        if err > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + err, 1.0)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(h[idx] + err, 0.0)
    return h

def lse_objective(h, n, tau):
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full") * dx
    maxc = np.max(corr)
    s = np.sum(np.exp((corr - maxc) / tau))
    return float(maxc + tau * np.log(s))


def lse_gradient(h, n, tau):
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full") * dx
    maxc = np.max(corr)
    weights = np.exp((corr - maxc) / tau)
    total = weights.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights /= total
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if w < 1e-10:
            continue
        if lag == 0:
            contrib = 1.0 - 2.0 * h
        elif lag > 0:
            contrib = np.zeros(n)
            contrib[:n - lag] += (1.0 - h[lag:])
            contrib[lag:] -= h[:n - lag]
        else:
            L = -lag
            contrib = np.zeros(n)
            contrib[L:] += (1.0 - h[:n - L])
            contrib[:n - L] -= h[L:]
        grad += w * contrib
    return grad * dx


class LSEOpt:
    def __init__(self, n, tau):
        self.n = n
        self.tau = tau
        self.target = n / 2.0

    def fun(self, x):
        return lse_objective(x, self.n, self.tau)

    def jac(self, x):
        return lse_gradient(x, self.n, self.tau)

    def con_fun(self, x):
        return np.sum(x) - self.target

    def con_jac(self, x):
        return np.ones(self.n)


def sa_optimize(h, n, T_init, max_iters, rng, best_h, best_c5, t_start, budget):

    reheat_period = 2000

    for it in range(max_iters):

        hj = cur_h[j]
        t_lo = max(-hi, hj - 1.0)
        t_hi = min(1.0 - hi, hj)
        if t_hi - t_lo < 1e-12:
            continue

        if rng.random() < 0.3:
            t = float(rng.uniform(-0.05, 0.05))
            t = max(t_lo, min(t_hi, t))
        else:
            t = float(rng.uniform(t_lo, t_hi))

        cur_h[i] += t
        cur_h[j] -= t
        new_c = exact_score(cur_h)
        delta = new_c - cur_c

        accept = False
        if delta < -1e-14:
            accept = True
        elif T > T_min and delta < 1.0:
            if float(rng.random()) < math.exp(-delta / T):
                accept = True

        if accept:
            cur_c = new_c
            if new_c < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = new_c
        else:
            cur_h[i] -= t
            cur_h[j] += t

        T = max(T_min, T * T_decay)
        if (it + 1) % reheat_period == 0 and n_reheats < 8:
            T = min(T * 3.0, 0.08)
            n_reheats += 1

    return cur_h, cur_c, best_h, best_c5
