def project_feasible(y, target_sum):
    """Project y onto {h: 0 <= h <= 1, sum(h) = target_sum}"""
    # Bisection on lambda
    lo, hi = y.min() - target_sum - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2, 0, 1)

        idx = np.argmax(np.abs(h - 0.5))

    c = np.correlate(h, 1 - h, mode='full') * dx

    c = np.correlate(h, 1 - h, mode='full') * dx
    # For numerical stability
    c_max = c.max()
    log_sum = c_max + eps * np.log(np.sum(np.exp((c - c_max) / eps)))
    weights = np.exp((c - c_max) / eps)
    weights /= weights.sum()

import numpy as np

def project_feasible(y, n):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    target = 0.5 * n
    lo = y.min() - target - 1.0
    hi = y.max() + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(np.abs(h - 0.5))
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                adj = min(diff, 1.0 - h[i])
            else:
                adj = max(diff, -h[i])
            h[i] += adj
            diff -= adj
    return h


def c5_val(h, n):
    """Compute C5 = max_k correlate(h,1-h,full)*dx."""
    dx = 2.0 / n
    return float((np.correlate(h, 1.0 - h, mode='full') * dx).max())


def grad_max(h, n):
    """Return (c5, gradient of the argmax correlation term)."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k = int(np.argmax(c))
    m = n - 1
    d = k - m
    g = np.zeros(n)
    if d >= 0:
        g[d:] = 1.0 - h[:n - d]
        g[:n - d] += -h[d:]
    else:
        dd = -d
        g[:n - dd] = 1.0 - h[dd:]
        g[dd:] += -h[:n - dd]
    return float(c.max()), g * dx


def smooth_FG(h, n, eps):
    """Log-sum-exp smoothed objective and its gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = n - 1
    cs = c / eps
    cm = cs.max()
    ex = np.exp(cs - cm)
    sw = ex.sum()
    F = eps * (cm + np.log(sw))
    w = ex / sw
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-7:
            continue
        d = k - m
        if d >= 0:
            g[d:] += wk * (1.0 - h[:n - d])
            g[:n - d] += wk * (-h[d:])
        else:
            dd = -d
            g[:n - dd] += wk * (1.0 - h[dd:])
            g[dd:] += wk * (-h[:n - dd])
    return F, g * dx

    """Gradient of softmax-weighted sum of top-k correlation terms."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = n - 1
    idx = np.argsort(c)[-topk:]
    vals = c[idx] / eps
    vals = vals - vals.max()
    w = np.exp(vals)
    w /= w.sum()
    g = np.zeros(n)
