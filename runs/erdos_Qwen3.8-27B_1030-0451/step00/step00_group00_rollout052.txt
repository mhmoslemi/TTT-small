def min_convex_1d(f, lo, hi, tol=1e-10, max_iter=100):
    """Minimize a convex function on [lo, hi] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return 0.5 * (a + b)

    return best_h, final_score, n
