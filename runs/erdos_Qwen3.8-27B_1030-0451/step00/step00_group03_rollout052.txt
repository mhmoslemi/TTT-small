                h_start = np.full(dim, 0.5)
        elif r < 0.60:
            h_start = h_start[::-1].copy()

        # Run pair descent
        remaining = hard_stop - guard_s - time.monotonic()
        sweep_time = min(40.0, remaining * 0.25)
        if sweep_time < 1.0:
            break
        h_opt, c_opt = pair_descent(h_start, max_sweeps=200,
                                     time_limit=sweep_time,
                                     start_time=time.monotonic(),
                                     rng=rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = len(h_opt)

        # Early stop if target well exceeded
        if best_c <= 0.38075:
            break

    # Final intensive polish on global best
    if best_h is not None:
        remaining_time = hard_stop - 5.0 - time.monotonic()
        if remaining_time > 3.0:
            for _ in range(3):
                if time.monotonic() > hard_stop - 3.0:
                    break
                h_opt, c_opt = pair_descent(best_h, max_sweeps=500,
                                             time_limit=remaining_time * 0.3,
                                             start_time=time.monotonic(),
                                             rng=rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
                    best_n = len(h_opt)
        # One more pass with different starting perturbation
        remaining_time = hard_stop - 3.0 - time.monotonic()
        if remaining_time > 2.0:
            h_pert = best_h + 0.005 * rng.standard_normal(best_n)
            h_pert = ensure_feasible(h_pert)
            if h_pert is not None:
                h_opt, c_opt = pair_descent(h_pert, max_sweeps=300,
                                             time_limit=remaining_time * 0.8,
                                             start_time=time.monotonic(),
                                             rng=rng)
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
                    best_n = len(h_opt)

    if best_h is None:
        h_out = np.full(64, 0.5)
        return h_out, exact_c5(h_out), 64

    h_out = best_h.copy()
    c_out = exact_c5(h_out)
    n_out = len(h_out)
    return h_out, c_out, n_out
