    if k >= 0:
        L = n - k
        g[k:] += (1.0 - h[:L])
        g[:L] -= h[k:]
    else:
        r = -k
        L = n - r
        g[:L] += (1.0 - h[r:])
        g[r:] -= h[:L]
    return g * dx

    f = corr_values(h)
    fmax = f.max()
    z = (f - fmax) / tau
    z = np.where(z < -30, -30.0, z)

def project_feasible(y, n):
    """Euclidean projection onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.5, posinf=2.0, neginf=-2.0)
    lo = -np.max(y) - 0.5

        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) * 0.5
    h = np.clip(y + alpha, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            rem = residual
            for i in order:
                if rem < 1e-14:
                    break
                step = min(room[i], rem)
                h[i] += step
                rem -= step

            rem = -residual
            for i in order:
                if rem < 1e-14:
                    break
                step = min(h[i], rem)
                h[i] -= step
                rem -= step
    return h


def eval_c5(h):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    """All correlation values times dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_f_k(h, k, dx):
    """Gradient of f_m(h) with respect to h, for shift k."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        g[k:] += (1.0 - h[:L])
        g[:L] -= h[k:]
    else:
        r = -k
        L = n - r
        g[:L] += (1.0 - h[r:])
        g[r:] -= h[:L]
    return g * dx


def smooth_gradient(h, dx, tau, max_active=150):
    """Gradient of log-sum-exp smoothed max of f_m."""
    n = len(h)
    f = corr_values(h)
    fmax = f.max()
    z = (f - fmax) / tau
    z = np.where(z < -40, -40.0, z)
    e = np.exp(z)
    total = e.sum()
    if total < 1e-300:
        return np.zeros(n)
    s = e / total
    if len(s) > max_active:
        idx = np.argpartition(s, -max_active)[-max_active:]
        s2 = np.zeros_like(s)
        s2[idx] = s[idx]
        s = s2 / s2.sum()
    grad = np.zeros(n)
    active = s > 1e-7
    for m in np.where(active)[0]:
        k = m - (n - 1)
        grad += s[m] * grad_f_k(h, k, dx)
    return grad
