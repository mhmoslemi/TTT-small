import numpy as np
import time

def project_to_sum(h, target):
    """Project h to [0,1]^n with sum = target using shift+bisection."""
    h = np.clip(h, 0, 1).astype(np.float64)
    n = len(h)
    if abs(np.sum(h) - target) < 1e-12:
        return h
    # Bisection on alpha: find alpha such that sum(clip(h + alpha, 0, 1)) = target
    lo = -np.max(h) - 1e-15
    hi = 1.0 - np.min(h) + 1e-15
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:
        # Distribute residual
        if residual > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] = np.clip(result[idx] + per, 0, 1)
    return result

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(corr) * dx)

def evaluate_pair_swap(h, i, j, num_trials=8):
    """Try mass exchanges between positions i and j, return best new h if improved."""
    n = len(h)
    h_i = h[i]
    h_j = h[j]
    d_min = max(-h_i, h_j - 1.0)
    d_max = min(1.0 - h_i, h_j)
    if d_max - d_min < 1e-10:
        return None, inf
    
    current_c5 = exact_c5(h)
    
    # Candidate d values
    width = d_max - d_min
    mid = (d_min + d_max) / 2
    candidates = set()
    candidates.add(d_min + 1e-10)
    candidates.add(d_max - 1e-10)
    candidates.add(mid)
    for f in [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9]:
        candidates.add(d_min + f * width)
    
    best_h = None
    best_c = current_c5
    for d in candidates:
        if d <= d_min + 1e-12 or d >= d_max - 1e-12:
            continue
        h_new = h.copy()
        h_new[i] = h_i + d
        h_new[j] = h_j - d
        c = exact_c5(h_new)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h_new
    return best_h, best_c

def run_pair_descent(h, max_sweeps=100, time_limit=None, start_time=None):
    """Perform pairwise coordinate descent."""
    n = len(h)
    h = h.copy()
    current_c5 = exact_c5(h)
    
    for sweep in range(max_sweeps):
        if time_limit and start_time:
            if time.time() - start_time > time_limit:
                break
        improved = False
        # Shuffle pairs
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng_local = np.random.default_rng(sweep * 1000 + 42)
        rng_local.shuffle(pairs)
        
        for i, j in pairs:
            if time_limit and start_time:
                if time.time() - start_time > time_limit:
                    break
            h_new, c_new = evaluate_pair_swap(h, i, j)
            if h_new is not None:
                h = h_new
                current_c5 = c_new
                improved = True
        if not improved:
            break
    return h, current_c5

    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    n = len(h)

import numpy as np
import time

def project_to_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} via shift+bisection."""
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    n = len(h)
    target = float(target)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-13:
        if residual > 0:
            free = np.where(result < 1.0 - 1e-10)[0]
        else:
            free = np.where(result > 1e-10)[0]
        if len(free) > 0:
            result[free] += residual / len(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def exact_c5(h):
    """Compute C5 = max_k dx * corr[k]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(corr) * dx)

def compute_subgrad(h):
    """Compute weighted subgradient of C5 at h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    max_val = np.max(corr)
    active = corr > (max_val - 0.02 * dx)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if not active[idx]:
            continue
        k = idx - (n - 1)
        if k >= 0:
            for i in range(n - k):
                g[i] += (1.0 - h[i + k])
                g[i + k] -= h[i]
        else:
            r = -k
            for i in range(r, n):
                g[i] += (1.0 - h[i - r])
                g[i - r] -= h[i]
    return g * dx, corr, max_val * dx

def try_pair(h, i, j, cur_c5):
    """Try mass exchange between i and j. Returns (new_h, new_c5) or (None, cur_c5)."""
    h_i, h_j = h[i], h[j]
    d_lo = max(-h_i, h_j - 1.0)
    d_hi = min(1.0 - h_i, h_j)
    if d_hi - d_lo < 1e-10:
        return None, cur_c5
    w = d_hi - d_lo
    best_h, best_c = None, cur_c5
    fracs = (0.08, 0.2, 0.35, 0.5, 0.65, 0.8, 0.92)
    for f in fracs:
        d = d_lo + f * w
        if d <= d_lo + 1e-12 or d >= d_hi - 1e-12:
            continue
        hn = h.copy()
        hn[i] = h_i + d
        hn[j] = h_j - d
        c = exact_c5(hn)
        if c < best_c:
            best_c = c
            best_h = hn
    return best_h, best_c

def local_opt(h, rng, time_left_fn, max_sweeps=2000):
    """Local optimization via gradient-guided pairwise swaps."""
    n = len(h)
    h = h.copy()
    cur_c5 = exact_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    no_improve = 0

    for sweep in range(max_sweeps):
        if time_left_fn() < 5.0:
            break

        g, corr, max_c = compute_subgrad(h)
        candidates = set()

        # Gradient-guided pairs: move mass from high-g to low-g
        pos = np.where(g > 0.01)[0]
        neg = np.where(g < -0.01)[0]
        if len(pos) > 0 and len(neg) > 0:
            for _ in range(min(80, len(pos) * len(neg))):
                i = pos[rng.integers(len(pos))]
                j = neg[rng.integers(len(neg))]
                candidates.add((min(i, j), max(i, j)))

        # Active-lag guided pairs
        top_lags = np.argsort(corr)[::-1][:min(12, 2 * n - 1)]
        for idx in top_lags:
            k = idx - (n - 1)
            if 0 < k < n:
                for i in range(0, n - k, max(1, (n - k) // 8)):
                    if h[i] * (1.0 - h[i + k]) > 0.02:
                        candidates.add((i, i + k))
            elif -n < k < 0:
                r = -k
                for i in range(r, n, max(1, (n - r) // 8)):
                    if h[i] * (1.0 - h[i - r]) > 0.02:
                        candidates.add((i - r, i))

        # Random pairs
        for _ in range(min(40, n * n // 10)):
            a, b = rng.integers(0, n, 2)
            if a != b:
                candidates.add((min(a, b), max(a, b)))

        agenda = list(candidates)
        rng.shuffle(agenda)

        improved = False
        for (i, j) in agenda:
            if time_left_fn() < 5.0:
                break
            h_new, c_new = try_pair(h, i, j, cur_c5)
            if h_new is not None and c_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c_new
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()

        no_improve = no_improve + 1 if not improved else 0
        if no_improve >= 5:
            break

    return best_h, best_c5

def expand_h(h, k):
    """Expand h by integer factor k."""
    return np.repeat(h, k)

def contract_h(h, k):
    """Contract h by integer factor k via block means."""
    n = len(h)
    if n % k != 0 or n // k < 4:
        return None
    m = n // k
    return h.reshape(m, k).mean(axis=1)

def interp_transfer(h_src, n_dst):
    """Interpolate h_src to length n_dst."""
    n_src = len(h_src)
    x0 = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x1 = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x1, x0, h_src)

def generate_perturbation(h, sigma, rng):
    """Generate a perturbed and projected version of h."""
    h_pert = h + sigma * rng.standard_normal(len(h))
    return project_to_sum(h_pert, len(h) / 2.0)

def generate_block_perturbation(h, rng):
    """Perturb a random contiguous block."""
    n = len(h)
    blk_len = rng.integers(max(2, n // 16), max(3, n // 4))
    start = rng.integers(0, n - blk_len + 1)
    delta = rng.uniform(-0.3, 0.3)
    h_new = h.copy()
    h_new[start:start + blk_len] += delta
    return project_to_sum(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)
