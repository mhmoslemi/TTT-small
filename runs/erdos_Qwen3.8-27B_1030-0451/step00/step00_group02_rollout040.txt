import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    """Compute the exact C5 objective."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    return float(np.max(c))
    
def smooth_objective(h, tau):
    """Log-sum-exp smooth approximation of max(c)."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    # Shift for numerical stability
    cmax = np.max(c)
    if cmax > tau * 50:
        return cmax + tau * np.log(np.sum(np.exp((c - cmax) / tau)))
    log_sum = np.log(np.sum(np.exp(c / tau)))
    return tau * log_sum

def smooth_gradient(h, tau):
    """Gradient of log-sum-exp objective."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    
    # Weights for each lag
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / tau)
    w = exp_c / np.sum(exp_c)
    
    # Compute gradient analytically
    # c_k = dx * sum_j h[j] * (1 - h[j+s]) where s = k - (n-1)
    # dc_k/dh[i] = dx * [(1-h[i+s])*I(valid1) - h[i-s]*I(valid2)]
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        s = k - (n - 1)
        # Valid j range: max(0, -s) <= j < min(n, n-s)
        j_start = max(0, -s)
        j_end = min(n, n - s)
        
        # For each valid j, the term is h[j] * (1 - h[j+s])
        # dc_k/dh[i] gets contribution from:
        #   j = i: (1 - h[i+s]) * dx (if i+s in [0,n))
        #   j+s = i, i.e., j = i-s: -h[i-s] * dx (if i-s in [0,n) and i-s in valid j range)
        
        for j in range(j_start, j_end):
            # h[j] contributes to dc_k/dh[j] as (1-h[j+s])*dx
            grad[j] += w[k] * dx * (1.0 - h[j + s])
            # h[j+s] contributes to dc_k/dh[j+s] as -h[j]*dx
            grad[j + s] += w[k] * dx * (-h[j])
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    return float(np.max(c))

def ensure_feasible(h, n):
    """Project h to feasibility: 0<=h<=1, sum(h)=n/2."""
    S = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for the level set
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = S - np.sum(h)
    # Distribute over non-saturated entries
    non_sat = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.sum(non_sat) > 0:
        h[non_sat] += residual / np.sum(non_sat)
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction
    residual = S - np.sum(h)
    idx = np.argmin(np.abs(h - 0.5))
    if 0 <= idx < n:
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def smooth_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / tau)
    w = exp_c / np.sum(exp_c)
    obj = tau * (cmax + np.log(np.sum(exp_c)))
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_start >= j_end:
            continue
        cands = h[j_start:j_end]
        comps = h[j_start + s:j_end + s]
        wk = w[k] * dx
        grad[j_start:j_end] += wk * (1.0 - comps)
        grad[j_start + s:j_end + s] += wk * (-cands)
    
    return obj, grad

class SmoothOpt:
    def __init__(self, n, tau):
        self.n = n
        self.tau = tau
    
    def fun(self, x):
        try:
            obj, _ = smooth_objective_and_grad(x, self.tau)
            return obj
        except:
            return 1e10
    
    def grad(self, x):
        try:
            _, g = smooth_objective_and_grad(x, self.tau)
            return g
        except:
            return np.zeros(self.n)
    
    def constraint_fun(self, x):
        return np.sum(x) - self.n / 2.0
    
    def constraint_grad(self, x):
        return np.ones(self.n)

def interpolate_vector(h_src, n_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_dst = (np.arange(n_dst) + 0.5) / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

def pair_search(h, max_iter=500, time_limit=None, t0=None):
    """Pair-based local search."""
    n = len(h)
    h = h.copy()
    cur_score = exact_score(h)
    rng = np.random.default_rng(12345)
    patience = 50
    no_improve = 0
    
    for it in range(max_iter):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        
        # Compute gradient
        tau = max(5e-5, 0.01 * cur_score)
        _, grad = smooth_objective_and_grad(h, tau)
        
        # Find best pair: i with lowest grad, j with highest grad
        # But also try random pairs for exploration
        if no_improve < patience:
            # Gradient-guided
            i = np.argmin(grad)
            j = np.argmax(grad)
        else:
            # Random
            perm = rng.permutation(n)
            i, j = perm[0], perm[1]
        
        if i == j:
            continue
        
        # Direction: increase h[i], decrease h[j] (since grad[i] < grad[j])
        # Feasible d range
        d_max = min(1.0 - h[i], h[j])
        if d_max < 1e-12:
            no_improve += 1
            continue
        
        # Try several d values
        best_d = 0
        best_s = cur_score
        for frac in [0.1, 0.3, 0.5, 0.8, 0.15, 0.05, 0.7, 0.2]:
            d = frac * d_max
            new_h = h.copy()
            new_h[i] += d
            new_h[j] -= d
            if new_h[i] < -1e-12 or new_h[i] > 1 + 1e-12 or new_h[j] < -1e-12 or new_h[j] > 1 + 1e-12:
                continue
            new_s = exact_score(new_h)
            if new_s < best_s - 1e-14:
                best_s = new_s
                best_d = d
        
        if best_d > 0:
            h[i] += best_d
            h[j] -= best_d
            h = np.clip(h, 0.0, 1.0)
            cur_score = best_s
            no_improve = 0
        else:
            no_improve += 1
    
    return h, cur_score

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    return float(np.max(c))


def ensure_feasible(h, n):
    S = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = S - np.sum(h)
    non_sat = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.sum(non_sat) > 0:
        h[non_sat] += residual / np.sum(non_sat)
        h = np.clip(h, 0.0, 1.0)
    residual = S - np.sum(h)
    if abs(residual) > 1e-15:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    cmax = np.max(c)
    exp_c = np.exp((c - cmax) / tau)
    w = exp_c / np.sum(exp_c)
    obj = tau * (cmax + np.log(np.sum(exp_c)))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_start >= j_end:
            continue
        cands = h[j_start:j_end]
        comps = h[j_start + s:j_end + s]
        wk = w[k] * dx
        grad[j_start:j_end] += wk * (1.0 - comps)
        grad[j_start + s:j_end + s] += wk * (-cands)
    return obj, grad


class SmoothOpt:
    def __init__(self, n, tau):
        self.n = n
        self.tau = tau

    def fun(self, x):
        try:
            obj, _ = smooth_obj_grad(x, self.tau)
            if not np.isfinite(obj):
                return 1e10
            return obj
        except Exception:
            return 1e10

    def grad(self, x):
        try:
            _, g = smooth_obj_grad(x, self.tau)
            if not np.all(np.isfinite(g)):
                return np.zeros(self.n)
            return g
        except Exception:
            return np.zeros(self.n)

    def eq_fun(self, x):
        return np.sum(x) - self.n / 2.0

    def eq_grad(self, x):
        return np.ones(self.n)


def try_slsqp(h, n, tau, maxiter=400):
    opt = SmoothOpt(n, tau)
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': opt.eq_fun, 'jac': opt.eq_grad}
    try:
        res = minimize(opt.fun, h, jac=opt.grad, method='SLSQP',
                       bounds=bounds, constraints=[constraint],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = ensure_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return None


def try_projected_gradient(h, n, tau, max_iter=2000, time_limit=None, t0=None):
    h = h.copy()
    for it in range(max_iter):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-14:

        alpha = 1.0 / (gnorm + 1e-15)

            obj_new, _ = smooth_obj_grad(h_new, tau)
            if obj_new < obj - 1e-15:
                h = h_new
                obj = obj_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            tau = max(tau * 0.8, 1e-6)
    return h

def interpolate_vector(h_src, n_dst):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_dst = (np.arange(n_dst) + 0.5) / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst


def gen_random_balanced(n, rng):
    S = n / 2.0
    h = rng.random(n)
    h = ensure_feasible(h, n)
    return h


def gen_interval_random(n, rng):
    S = n / 2.0
    h = np.zeros(n)

        start = int(rng.integers(0, n))

        h[start:start + length] += 1.0

    return h
