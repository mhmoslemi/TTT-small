```python
def compute_gradient(h, active_lags, weights, N):
    """
    Compute the gradient of sum_d w_d * L[N-1+d] with respect to h.
    active_lags: array of d values (can be negative, zero, or positive)
    weights: corresponding weights
    """
    N = len(h)
    g = np.zeros(N)
    for d, w in zip(active_lags, weights):
        if d == 0:
            g += w * (1 - 2*h)
        elif d > 0:
            # L = sum_{n=d}^{N-1} h[n](1-h[n-d])
            # ∂L/∂h[j] = (1-h[j-d]) for j >= d, minus h[j+d] for j < N-d
            g[d:] += w * (1 - h[:N-d])  # (1-h[j-d]) for j from d to N-1
            g[:N-d] -= w * h[d:]  # -h[j+d] for j from 0 to N-1-d
        else:
            d_prime = -d
            # L = sum_{n=0}^{N-1-d'} h[n](1-h[n+d'])
            # ∂L/∂h[j] = (1-h[j+d']) for j < N-d', minus h[j-d'] for j >= d'
            g[:N-d_prime] += w * (1 - h[d_prime:])  # (1-h[j+d']) for j from 0 to N-1-d'
            g[d_prime:] -= w * h[:N-d_prime]  # -h[j-d'] for j from d' to N-1
    return g
```

```python
        if s > target_sum:
            lo = mid
        else:

    return np.correlate(h, 1.0 - h, 'full') * (2.0 / N)

    g = np.zeros(N)

            g[:N-d] -= w * h[d:]
        else:
            dp = -d
            g[:N-dp] += w * (1.0 - h[dp:])
            g[dp:] -= w * h[:N-dp]
```

```python
def _lags_fft(h):
    N = len(h)
    nfft = 2 * N
    H = np.fft.rfft(h, nfft)
    B = np.fft.rfft(1.0 - h, nfft)
    corr = np.fft.irfft(H * np.conj(B), nfft)[:2*N-1]
    # numpy correlate convention: corr[k] = sum_n h[n] * (1-h)[n-k+N-1]
    # irfft gives circular correlation, need to adjust
    # Actually for full linear correlation:
    # np.correlate(h, 1-h, 'full')[k] = sum_n h[n] * (1-h)[n - k + N - 1]
    # The FFT-based approach: ifft(fft(h) * conj(fft(1-h))) gives circular cross-correlation
    # For linear, we zero-pad to 2N
```

```python
def _project(y, S):
    """Project y onto {x: 0<=x_i<=1, sum(x)=S} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def _lags(h):
    """Compute all lag values: np.correlate(h, 1-h, 'full') * dx"""
    N = len(h)
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / float(N))

def _subgrad(h, lags, k):
    """
    Compute weighted subgradient from top-k lags.
    Returns gradient of (sum w_i * L[i]) * dx where L[i] are the top-k lag values.
    """
    N = len(h)
    if k >= len(lags):
        k = len(lags)
    idx = np.argpartition(lags, -k)[-k:]
    w = lags[idx].copy()
    w = np.maximum(w, 0)
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.ones(k) / k
    else:
        w = w / wsum
    
    dx = 2.0 / float(N)
    g = np.zeros(N)
    center = N - 1
    
    for i in range(k):
        d = int(idx[i]) - center
        wi = w[i] * dx
        if d == 0:
            g += wi * (1.0 - 2.0 * h)
        elif d > 0:
            g[d:] += wi * (1.0 - h[:N - d])
            g[:N - d] -= wi * h[d:]
        else:
            dp = -d
            g[:N - dp] += wi * (1.0 - h[dp:])
            g[dp:] -= wi * h[:N - dp]
    
    return g
```
