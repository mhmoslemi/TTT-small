def find_optimal_t(c, g, h_i0, t_lo, t_hi):
    lo = t_lo
    hi = t_hi
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        vals = c + (mid - h_i0) * g
        mx = vals.max()
        mask = vals >= mx - 1e-10
        gm = g[mask]
        if gm.min() <= 1e-10 and gm.max() >= -1e-10:
            break
        elif gm.max() < -1e-10:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)
