    diff = current - target

    # Need to adjust by -diff
    if diff > 0:
        # Need to reduce sum by diff
        # Sort by value descending, reduce from higher values

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    return vals.max(), vals

def project_to_constraint(h, target_sum):

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        
        # Line search
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step_size = lr * shrink
            h_try = h - step_size * grad
            h_try = project_to_constraint(h_try, target_sum)
            c5_try, _ = compute_c5(h_try)
            if c5_try < c5 - 1e-12:
                h = h_try
                break
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

def c5_subgradient(h, vals):
    n = len(h)
    dx = 2.0 / n
    c5 = vals.max()
    # Active set: lags where vals is within tolerance of max
    active = np.where(vals >= c5 - 1e-9)[0]
    if len(active) == 0:
        active = np.array([np.argmax(vals)])
    
    grad = np.zeros(n)
    
    for idx in active:
        lag = idx - (n - 1)
        r = abs(lag)
        
        if r == 0:
            # corr_0 = sum h[i](1-h[i]), d/dh[j] = 1-2h[j]
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            # corr_{+r} = sum_{i=0}^{n-1-r} h[i+r](1-h[i])
            # d/dh[j] = (1-h[j-r])*I(r≤j) - h[j+r]*I(j≤n-1-r)
            mask_upper = (np.arange(n) >= r)
            mask_lower = (np.arange(n) <= n-1-r)
            grad[mask_upper] += (1.0 - h[np.arange(n)[mask_upper] - r])
            grad[mask_lower] += -h[np.arange(n)[mask_lower] + r]
        else:
            # corr_{-r} = sum_{i=r}^{n-1} h[i-r](1-h[i])
            # d/dh[j] = (1-h[j+r])*I(j≤n-1-r) - h[j-r]*I(r≤j≤n-1)
            mask_lower = (np.arange(n) <= n-1-r)
            mask_upper = (np.arange(n) >= r)
            grad[mask_lower] += (1.0 - h[np.arange(n)[mask_lower] + r])
            grad[mask_upper] += -h[np.arange(n)[mask_upper] - r]
    
    grad *= dx / len(active)
    return grad
