        if c < best_c - 1e-14:

            best_h = ht

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = 0.5 * n
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v - lam, 0.0, 1.0)

def grad_active(h, active, weights):
    """Compute gradient of weighted sum of active correlations."""
    n = len(h)
    dx = 2.0 / n
    c0 = n - 1
    g = np.zeros(n)
    for idx, w in zip(active, weights):
        k = idx - c0
        c = w * dx
        if k == 0:
            g += c * (1.0 - 2.0 * h)
        elif k > 0:
            nk = n - k
            g[k:] += c * (1.0 - h[:nk])
            g[:nk] -= c * h[k:]
        else:
            R = -k
            nR = n - R
            g[:nR] += c * (1.0 - h[R:])
            g[R:] -= c * h[:nR]
    return g

def active_lags(corr, T, K=30):
    """Get active correlation indices and weights."""
    M = corr.max()
    idx = np.where(corr >= M - T)[0]
    if len(idx) > K:
        top = np.argsort(corr[idx])[::-1][:K]
        idx = idx[top]
    if len(idx) == 0:
        idx = np.array([np.argmax(corr)])
    d = M - corr[idx]
    w = np.exp(-d / T)
    w /= w.sum()
    return idx, w

def gradient_step(h, T, alphas):
    """One gradient descent step with backtracking."""
    n = len(h)
    corr = compute_corr(h)
    idx, w = active_lags(corr, T)
    g = grad_active(h, idx, w)
    g -= g.mean()
    best_h, best_c = h, corr.max()
    for a in alphas:
        v = h - a * g
        ht = project(v, n)
        c_new = compute_corr(ht).max()
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = ht
    return best_h, best_c

def resample(h, n_new):

    if n == n_new:

    x_new = np.arange(n_new) * 2.0 / n_new
    y = np.interp(x_new, x_old, h)
    return project(y, n_new)
