        dims = [n_parent, 96, 128, 160, 192]
    elif n_parent <= 128:
        dims = [n_parent, 160, 192, 224]
    else:
        dims = [n_parent, n_parent + 32]

    dim_cycle = 0
    stagnation_count = 0

    while time.monotonic() < hard_stop:
        remaining = hard_stop - time.monotonic()
        if remaining < 2.0:
            break

        dim = dims[dim_cycle % len(dims)]
        dim_cycle += 1

        time_per_round = min(remaining * 0.12, 25.0)
        time_limit = time.monotonic() + time_per_round

        if dim == best_n:
            h_seed = best_h.copy()
        else:
            h_seed = transfer_resolution(best_h, dim, rng)
            c_seed = exact_c5(h_seed)
            if c_seed > best_c5 + 0.15:
                continue

        h_res, c_res = run_pair_descent(h_seed, dim, rng, time_limit, hard_stop)
        if c_res < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_res.copy(), c_res, dim
            stagnation_count = 0
        else:
            stagnation_count += 1

        if time.monotonic() < hard_stop - 3.0:
            sigma = 0.02 + 0.05 * rng.random()
            h_mut = shifted_projection(best_h + rng.normal(0, sigma, best_n), best_n)
            t2 = min(time.monotonic() + 8.0, hard_stop - 1.0)
            h_m2, c_m2 = run_pair_descent(h_mut, best_n, rng, t2, hard_stop)
            if c_m2 < best_c5 - 1e-12:
                best_h, best_c5 = h_m2.copy(), c_m2
                stagnation_count = 0

            if rng.random() < 0.3:
                h_jit = shifted_projection(best_h + rng.normal(0, 0.15, best_n), best_n)
                t3 = min(time.monotonic() + 5.0, hard_stop - 1.0)
                h_j2, c_j2 = run_pair_descent(h_jit, best_n, rng, t3, hard_stop)
                if c_j2 < best_c5 - 1e-12:
                    best_h, best_c5 = h_j2.copy(), c_j2

    remaining = polish_stop - time.monotonic()
    if remaining > 3.0:
        h_fin, c_fin = run_pair_descent(best_h, best_n, rng, polish_stop, polish_stop)
        if c_fin < best_c5 - 1e-12:
            best_h, best_c5 = h_fin, c_fin

    remaining = polish_stop - time.monotonic()
    if remaining > 2.0:
        for _ in range(3):
            if time.monotonic() > polish_stop:
                break
            h_r, c_r = run_pair_descent(best_h, best_n, rng,
                                        min(time.monotonic() + remaining * 0.25, polish_stop),
                                        polish_stop)
            if c_r < best_c5 - 1e-12:
                best_h, best_c5 = h_r, c_r

    best_h = shifted_projection(best_h, len(best_h))
    final_c5 = exact_c5(best_h)

    return best_h, final_c5, len(best_h)
