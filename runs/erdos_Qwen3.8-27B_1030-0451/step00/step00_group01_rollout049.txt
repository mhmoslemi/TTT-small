    Z = np.sum(u)
    obj = m + np.log(Z) / beta
    w = u / Z

    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        cnt = np.sum(mask)

            np.clip(result, 0.0, 1.0, out=result)
    return result

    n = len(h)

    n_min = max(0, -lag)
    n_max = min(n, n - lag)
    grad = np.zeros(n)
    m1_lo = max(0, n_min + lag)
    m1_hi = min(n - 1, n_max - 1 + lag)
    if m1_lo <= m1_hi:

        grad[idx] += (1.0 - h[idx - lag])
    if n_min <= n_max - 1:
        idx = np.arange(n_min, n_max)
        grad[idx] -= h[idx + lag]
    return grad * dx

import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_sum_box(h, target):
    """Project h onto {0<=h_i<=1, sum(h)=target}"""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += residual / cnt
            np.clip(result, 0.0, 1.0, out=result)
    return result

def evaluate_c5(h):
    dx = 2.0 / len(h)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))

def all_lag_values(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    """Gradient of lag value (scaled by dx) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    n_min = max(0, -lag)
    n_max = min(n, n - lag)
    grad = np.zeros(n)
    m1_lo = max(0, n_min + lag)
    m1_hi = min(n - 1, n_max - 1 + lag)
    if m1_lo <= m1_hi:
        idx = np.arange(m1_lo, m1_hi + 1)
        grad[idx] += (1.0 - h[idx - lag])
    if n_min <= n_max - 1:
        idx = np.arange(n_min, n_max)
        grad[idx] -= h[idx + lag]
    return grad * dx

def weighted_lag_gradient(h, weights):
    """Compute sum_k weights[k] * grad_lag(h, lag_k)"""
    n = len(h)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if weights[idx] > 1e-12:
            lag = idx - (n - 1)
            grad += weights[idx] * grad_lag(h, lag)
    return grad

def logsumexp_obj_grad(h, beta):
    """Log-sum-exp smoothed objective and its gradient"""
    v = all_lag_values(h)
    m = np.max(v)
    shifted = np.clip(beta * (v - m), -50.0, 50.0)
    u = np.exp(shifted)
    Z = np.sum(u)
    obj = m + np.log(Z) / beta
    w = u / Z
    grad = weighted_lag_gradient(h, w)
    return obj, grad

def solve_lp(h0, radius, n_active=None):
    """LP trust-region: minimize t s.t. linearized lag constraints, box, sum"""
    n = len(h0)
    v = all_lag_values(h0)
    n_total = 2 * n - 1
    if n_active is not None and n_total > n_active:
        active = np.argsort(-v)[:n_active]
    else:
        active = np.arange(n_total)
    ka = len(active)
    A_ub = np.zeros((ka, n + 1))
    b_ub = np.zeros(ka)
    for i in range(ka):
        lag = active[i] - (n - 1)
        a = grad_lag(h0, lag)
        A_ub[i, :n] = a
        A_ub[i, n] = -1.0
        b_ub[i] = np.dot(a, h0) - v[active[i]]
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    if np.sum(lb) > n / 2.0 + 1e-7 or np.sum(ub) < n / 2.0 - 1e-7:
        return None
    bounds = list(zip(lb, ub)) + [(0.0, 1.0)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='HiGHS')
        if res.success:
            return project_sum_box(res.x[:n], n / 2.0)
    except Exception:
        pass
    return None

def slsqp_optimize(h0, beta, maxiter=80):
    """SLSQP with log-sum-exp objective"""
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    n_eval = [0]
    def fun(x):
        n_eval[0] += 1
        return logsumexp_obj_grad(x, beta)[0]
    def jac(x):
        return logsumexp_obj_grad(x, beta)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h0, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum_box(res.x, target)
    except Exception:
        pass
    return None

def transfer_resolution(h_old, n_new):
    """Interpolate h to a new grid size"""
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_sum_box(h_interp, n_new / 2.0)

def gradient_descent_polish(h0, beta, max_iter=400):
    """Projected gradient with backtracking line search"""
    n = len(h0)
    target = n / 2.0
    h_cur = h0.copy()
    c5_cur = evaluate_c5(h_cur)
    step = 0.1
    for _ in range(max_iter):
        _, grad = logsumexp_obj_grad(h_cur, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        direction = -grad / gn
        improved = False
        s = step
        for bt in range(40):
            x_try = project_sum_box(h_cur + s * direction, target)
            c5_try = evaluate_c5(x_try)
            if c5_try < c5_cur - 1e-14:
                h_cur = x_try
                c5_cur = c5_try
                improved = True
                break
            s *= 0.5
        if improved:
            step = min(0.5, s * 1.2)
        else:
            step *= 0.5
            if step < 1e-14:
                break
    return h_cur, c5_cur
