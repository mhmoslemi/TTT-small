        ni = int(interior.sum())
        if ni > 0:

            y = np.clip(y, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    phi = c * dx
    m = phi.max()
    return m + tau * np.log(np.sum(np.exp((phi - m) / tau)))

    dx = 2.0 / n

    phi = c * dx
    m = phi.max()
    exp_phi = np.exp((phi - m) / tau)
