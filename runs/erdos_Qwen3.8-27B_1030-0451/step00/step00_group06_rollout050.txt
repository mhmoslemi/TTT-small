import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    h_init = np.array(initial_h_values, dtype=float)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    def elapsed():
        return time.time() - start_time
    
    def time_left():
        return budget_s - 10.0  # leave 10s margin
    
    def compute_c5(h):
        """Compute C5 = max of correlation * dx"""
        N = len(h)
        dx = 2.0 / N
        corr = np.correlate(h, 1.0 - h, mode='full')
        return corr * dx
    
    def compute_grad(h, weights, dx):
        N = len(h)
        o = 1.0 - h
        conv_wo = np.convolve(weights, o, 'full')
        A = conv_wo[N-1 : 2*N-1]
        corr_hw = np.correlate(h, weights, 'full')
        B = corr_hw[3*N-3 : 2*N-3 : -1]
        return dx * (A - B)
    
    def project_to_feasible(h, N):
        """Project h onto {x : 0 <= x_i <= 1, sum(x) = N/2}"""
        target = N / 2.0
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
        # Fix rounding to ensure exact sum
        # Actually, with floating point, sum might not be exactly N/2
        # The problem says sum(h) == n_points / 2 exactly
        # Let's adjust: if sum is off, fix one element
        s = proj.sum()
        diff = target - s
        if abs(diff) > 1e-10:
            # Add diff to the element with most room
            idx = np.argmin(np.minimum(proj, 1.0 - proj))
            room = min(proj[idx], 1.0 - proj[idx]) if diff < 0 else (1.0 - proj[idx]) if diff > 0 else 0
            # Actually just add to the max element (for positive diff) or min
            if diff > 0:
                idx = np.argmin(1.0 - proj)  # element closest to 1
            else:
                idx = np.argmax(proj)  # element closest to 0
            proj[idx] += diff
            proj = np.clip(proj, 0.0, 1.0)
        return proj
    
    def soft_max_weights(c_vals, temperature):
        c_max = np.max(c_vals)
        exp_c = np.exp((c_vals - c_max) / temperature)
        return exp_c / np.sum(exp_c)
    
    def optimize_projection_grad(h, n_iters, lr, temp, rng):
        """Projected gradient descent with soft-max weights"""
        N = len(h)
        dx = 2.0 / N
        h = h.copy()
        
        for it in range(n_iters):
            c_vals = compute_c5(h)
            weights = soft_max_weights(c_vals, temp)

    def project_feasible(h, N):
        target = N / 2.0
        lo, hi = -2.0, 2.0
        for _ in range(70):
            mid = (lo + hi) * 0.5
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        lam = (lo + hi) * 0.5
        proj = np.clip(h - lam, 0.0, 1.0)
        diff = target - proj.sum()
        if abs(diff) > 1e-14:
            if diff > 0:
                idx = np.argmin(1.0 - proj)
            else:
                idx = np.argmax(proj)
            proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
        return proj

    def compute_c5_vals(h):
        N = len(h)
        dx = 2.0 / N
        return np.correlate(h, 1.0 - h, mode='full') * dx

    def compute_grad(h, weights, dx):
        N = len(h)
        o = 1.0 - h
        conv_wo = np.convolve(weights, o, 'full')
        A = conv_wo[N - 1:2 * N - 1]
        corr_hw = np.correlate(h, weights, 'full')
        B = corr_hw[3 * N - 3:2 * N - 3:-1]
        return dx * (A - B)

    def softmax_weights(c_vals, temperature):
        c_max = np.max(c_vals)
        e = np.exp((c_vals - c_max) / temperature)
        return e / np.sum(e)

    def optimize_pg(h, n_iters, lr, temp):
        N = len(h)
        dx = 2.0 / N
        h = h.copy()
        for it in range(n_iters):
            c_vals = compute_c5_vals(h)
            w = softmax_weights(c_vals, temp)
            grad = compute_grad(h, w, dx)
            grad -= grad.mean()
            h = h - lr * grad
            h = project_feasible(h, N)
        return h

    def line_search_polish(h, n_pairs, rng_local):
        N = len(h)
        h = h.copy()
        c5 = evaluate_erdos_solution(h)
        improved = True
        while improved and time_left() > 30:
            improved = False
            for _ in range(n_pairs):
                if time_left() < 30:
                    break
                i, j = rng_local.choice(N, size=2, replace=False)
                lo, hi = max(-h[i], h[j] - 1.0), min(1.0 - h[i], h[j])
                if hi - lo < 1e-12:
                    continue
                t_vals = np.linspace(lo, hi, 21)
                best_t = 0.0
                best_c5_local = c5
                for t in t_vals:
                    h_try = h.copy()
                    h_try[i] = h[i] + t
                    h_try[j] = h[j] - t
                    if h_try[i] < -1e-12 or h_try[i] > 1 + 1e-12:
                        continue
                    if h_try[j] < -1e-12 or h_try[j] > 1 + 1e-12:
                        continue
                    c5_try = evaluate_erdos_solution(h_try)
                    if c5_try < best_c5_local - 1e-9:
                        best_c5_local = c5_try
                        best_t = t
                if abs(best_t) > 1e-12:
                    h[i] += best_t
                    h[j] -= best_t
                    c5 = best_c5_local
                    improved = True
        return h, c5

    # Initial evaluation
    best_h = project_feasible(h_init.copy(), len(h_init))
    best_c5 = evaluate_erdos_solution(best_h)

    # Optimization over multiple resolutions
    n_base = len(h_init)
    n_targets = sorted(set([n_base, 257, 419, 511, 512, 625, 768, 1024]))
    n_targets = [n for n in n_targets if n <= 2499]

    def resample_to(h_src, n_dst):
        N_src = len(h_src)
        if N_src == n_dst:
            return h_src.copy()
        x_src = np.linspace(0.0, 2.0, N_src, endpoint=False)
        x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
        h_res = np.interp(x_dst, x_src, h_src)
        return project_feasible(h_res, n_dst)

    for n_t in n_targets:
        if time_left() < 90:
            break

        # Start from both initial and best
        h_a = resample_to(h_init, n_t)
        h_b = resample_to(best_h, n_t)
        c5_a = evaluate_erdos_solution(h_a)
        c5_b = evaluate_erdos_solution(h_b)
        h_cur = h_a if c5_a <= c5_b else h_b
        c5_cur = min(c5_a, c5_b)

        # Temperature annealing with projected gradient
        for temp in [4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03]:
            if time_left() < 60:
                break
            n_it = 400 if temp > 1.0 else 600
            h_opt = optimize_pg(h_cur, n_it, lr=1.5, temp=temp)
            c5 = evaluate_erdos_solution(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_cur = h_opt

        # Random perturbations and re-optimize
        for _ in range(8):
            if time_left() < 45:
                break
            sigma = rng.uniform(0.03, 0.15)
            h_p = h_cur + rng.normal(0, sigma, n_t)
            h_p = project_feasible(h_p, n_t)
            h_opt = optimize_pg(h_p, n_iters=300, lr=1.0, temp=0.3)
            c5 = evaluate_erdos_solution(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_cur = h_opt

    # Final polish on best solution with line search
    N_best = len(best_h)
    if time_left() > 60:
        best_h, best_c5 = line_search_polish(best_h, n_pairs=500, rng_local=rng)

    # Ensure final solution is feasible
    best_h = project_feasible(best_h, N_best)
    best_c5 = evaluate_erdos_solution(best_h)

    return (best_h, best_c5, N_best)
