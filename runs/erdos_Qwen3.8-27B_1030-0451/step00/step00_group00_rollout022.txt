def compute_c5(h, dx):

    c = np.correlate(h, 1-h, mode="full") * dx
    return c.max(), c

def project_onto_convex_set(h, target_sum):
    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target_sum}."""
    h = np.clip(h, 0, 1)
    # Water-filling to achieve exact sum
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    # If diff > 0, we need to increase some elements
    # If diff < 0, we need to decrease some elements
    if diff > 0:
        # Can only increase elements that are < 1
        mask = h < 1.0
        capacity = (1.0 - h[mask]).sum()
        if capacity <= 0:
            return h
        scale = min(1.0, diff / capacity)
        h[mask] += scale * (1.0 - h[mask])
    else:
        # Can only decrease elements that are > 0
        mask = h > 0.0
        capacity = h[mask].sum()
        if capacity <= 0:
            return h
        scale = min(1.0, -diff / capacity)
        h[mask] -= scale * h[mask]
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max(), c

def project_feasible(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        if not mask.any():
            return h
        capacity = (1.0 - h[mask]).sum()
        if capacity > 0:
            step = min(1.0, diff / capacity)
            h[mask] += step * (1.0 - h[mask])
    else:
        mask = h > 0.0
        if not mask.any():
            return h
        capacity = h[mask].sum()
        if capacity > 0:
            step = min(1.0, -diff / capacity)
            h[mask] -= step * h[mask]
    # Final exact adjustment
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Adjust one element
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0, 1)
    return h

def grad_shift(h, p):
    """Gradient of correlation at shift p (|s|=p) w.r.t. h, before multiplying by dx."""
    n = len(h)
    grad = np.zeros(n)
    if p == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        if n - p > 0:
            grad[p:] = 1.0 - h[:n-p]
            grad[:n-p] -= h[p:]
    return grad

def weighted_gradient(h, c_values, dx, top_k=20):
    """Compute gradient focusing on top-k correlation values."""
    n = len(h)
    # Get top-k indices
    k = min(top_k, len(c_values))
    top_idx = np.argsort(c_values)[-k:]
    # Weights proportional to values (or just uniform over top-k)
    w = c_values[top_idx]
    w = w / w.sum()
    
    grad = np.zeros(n)
    for idx, weight in zip(top_idx, w):
        s = idx - (n - 1)
        p = abs(s)
        grad += weight * grad_shift(h, p)
    return grad * dx

    best_c5, _ = compute_c5(h, dx)
    for it in range(max_iters):
        improved = False
        # Random coordinate pairs
        rng = np.random.default_rng(it)
        for _ in range(50):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            # Try moving mass from j to i
            for delta in [0.01, 0.05, 0.1]:
                if h[i] + delta <= 1 and h[j] - delta >= 0:
                    h_new = h.copy()
                    h_new[i] += delta
                    h_new[j] -= delta
                    c5_new, _ = compute_c5(h_new, dx)
                    if c5_new < best_c5 - 1e-10:
                        h = h_new
                        best_c5 = c5_new
                        improved = True

                if h[i] - delta >= 0 and h[j] + delta <= 1:
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    c5_new, _ = compute_c5(h_new, dx)
                    if c5_new < best_c5 - 1e-10:
                        h = h_new
                        best_c5 = c5_new
                        improved = True
                        break
        if not improved:
            break
    return h, best_c5

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max(), c

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        if not mask.any():
            return h
        capacity = (1.0 - h[mask]).sum()
        if capacity > 1e-15:
            step = min(1.0, diff / capacity)
            h[mask] += step * (1.0 - h[mask])
    else:
        mask = h > 0.0
        if not mask.any():
            return h
        capacity = h[mask].sum()
        if capacity > 1e-15:
            step = min(1.0, -diff / capacity)
            h[mask] -= step * h[mask]
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def grad_for_shift(h, p):
    """Gradient of c_p (correlation at shift p) w.r.t. h. p >= 0."""
    n = len(h)
    grad = np.zeros(n)
    if p == 0:
        grad[:] = 1.0 - 2.0 * h
    elif p < n:
        grad[p:] = 1.0 - h[:n-p]
        grad[:n-p] -= h[p:]
    return grad

def projected_gradient_step(h, c_values, dx, lr, top_k=30):
    n = len(h)
    k = min(top_k, len(c_values))
    top_idx = np.argsort(c_values)[-k:]
    w = c_values[top_idx].copy()
    w = np.maximum(w, 1e-10)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx_int, weight in zip(top_idx, w):
        s = idx_int - (n - 1)
        p = abs(int(s))
        grad += weight * grad_for_shift(h, p)
    grad *= dx
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, h.sum())
    return h_new

    order = np.argsort(h)[::-1]

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        if not mask.any():
            return h
        capacity = (1.0 - h[mask]).sum()
        if capacity > 1e-15:
            step = min(1.0, diff / capacity)
            h[mask] += step * (1.0 - h[mask])
    else:
        mask = h > 0.0
        if not mask.any():
            return h
        capacity = h[mask].sum()
        if capacity > 1e-15:
            step = min(1.0, -diff / capacity)
            h[mask] -= step * h[mask]
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:

            idx = int(np.argmin(h))
        else:

    """Compute the change in all correlation values (before *dx) when h[a]: 0->1, h[b]: 1->0."""
    # c_k for k = 0, ..., 2n-2, shift s = k - (n-1), p = |s|
    delta = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        p = abs(k - (n - 1))
        if p == 0:

            # Flip a: 0->1: change in linear: +1, change in quad: -1 (h[a]^2 goes 0->1)
            # Flip b: 1->0: change in linear: -1, change in quad: +1
            delta[k] = (1 - 1) + (-1 + 1)  # = 0? No...

            # After flip: sum h'[i] - sum h'[i]^2
            # Change: (h'[a]-h[a]) + (h'[b]-h[b]) - ((h'[a]^2-h[a]^2) + (h'[b]^2-h[b]^2))
            # = (1-0) + (0-1) - ((1-0) + (0-1))
            # = 1 - 1 - (1 - 1) = 0
            # So for p=0, the change is always 0! Makes sense: sum h is preserved.
