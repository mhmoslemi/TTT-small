def project_to_feasible(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}."""
    x = h.copy()
    x = np.clip(x, 0, 1)
    # Bisection for Lagrange multiplier
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.clip(h + mid, 0, 1)
        s = np.sum(y)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix any residual
    residual = target_sum - np.sum(y)
    if abs(residual) < 1e-10:
        return y
    # Distribute residual
    if residual > 0:
        # Add to entries that are < 1
        idx = np.where(y < 1.0)[0]
        if len(idx) > 0:
            add = residual / len(idx)
            y[idx] = np.clip(y[idx] + add, 0, 1)

        if len(idx) > 0:
            sub = -residual / len(idx)
            y[idx] = np.clip(y[idx] - sub, 0, 1)

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def is_feasible(h, tol=1e-8):
    if np.any(h < -tol) or np.any(h > 1 + tol):
        return False
    if abs(np.sum(h) - len(h) / 2.0) > 1e-6:
        return False
    if not np.all(np.isfinite(h)):
        return False
    return True

def project_feasible(h, target_sum):
    """Project onto {x: sum(x)=target_sum, 0<=x<=1} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    res = target_sum - np.sum(y)
    if abs(res) > 1e-12:
        if res > 0:
            room = 1.0 - y
            idx = np.where(room > 1e-15)[0]
        else:
            room = y.copy()
            idx = np.where(room > 1e-15)[0]
        if len(idx) > 0:
            per = res / len(idx)
            y[idx] = np.clip(y[idx] + per, 0.0, 1.0)

    """Compute subgradient of the active (max) constraint. Returns (grad, active_shift)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_idx = np.argmax(corr)
    k = active_idx - (n - 1)
    g = np.zeros(n)

        m = n - k
        g[0:m] += dx * (1.0 - h[k:k+m])
        g[k:k+m] -= dx * h[0:m]
    else:
        k_a = -k
        m = n - k_a
        g[k_a:k_a+m] += dx * (1.0 - h[0:m])
        g[0:m] -= dx * h[k_a:k_a+m]

def weighted_subgrad(h, dx, top_m=5):
    """Weighted subgradient using top-m active constraints."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top m shifts
    top_idx = np.argsort(corr)[-top_m:]
    max_val = corr[top_idx[0]]
    # Weights: exponential based on proximity to max
    weights = np.exp((corr[top_idx] - max_val) / (0.01 * max_val + 1e-10))
    weights = weights / weights.sum()
    
    g = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            g[0:m] += w * dx * (1.0 - h[k:k+m])
            g[k:k+m] -= w * dx * h[0:m]
        else:
            k_a = -k
            m = n - k_a
            g[k_a:k_a+m] += w * dx * (1.0 - h[0:m])
            g[0:m] -= w * dx * h[k_a:k_a+m]
