def compute_correlation(h):

def project_to_constraint(h, n):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = n/2}"""
    target = n / 2.0
    lo, hi = -max(h), 1 - min(h)
    for _ in range(100):
        mid = (lo + hi) / 2
        clamped = np.clip(h - mid, 0, 1)
        s = clamped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    clamped = np.clip(h - (lo+hi)/2, 0, 1)
    return clamped

def gradient_overlap(h, k, n):
    """Compute gradient of c_k = sum_n h[n+k]*(1-h[n]) w.r.t. h"""
    g = np.zeros(n)
    # For each j, ∂c_k/∂h[j] = (1-h[j-k]) - h[j+k] (with boundary conditions)
    valid_j = np.arange(n)

    mask2 = (valid_j + k >= 0) & (valid_j + k < n)
    g[mask2] -= h[np.where(mask2, valid_j + k, 0)]
    return g

def soft_max_c5(h, tau):
    c = np.correlate(h, 1-h, mode="full")
    c_scaled = c * (2.0 / len(h))  # multiply by dx
    # Soft max
    max_c = c_scaled.max()
    exps = np.exp(tau * (c_scaled - max_c))
    return max_c + np.log(exps.sum()) / tau, exps / exps.sum()

def gradient_soft_max_c5(h, tau, weights):
