def exact_c5(h):
    """Compute C5 = max_k C(k) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr

def project(h, n):
    """Project h to satisfy 0 <= h <= 1 and sum(h) = n/2."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum; add to coordinates below 1
        room = 1.0 - h
        room_sum = room.sum()
        if room_sum < 1e-12:
            return h
        h += diff * room / room_sum
    else:
        # Need to decrease sum; subtract from coordinates above 0
        room = h
        room_sum = room.sum()
        if room_sum < 1e-12:
            return h
        h += diff * room / room_sum
