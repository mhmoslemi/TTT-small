import time

def evaluate_c5(h):
    """Compute C5 = max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return c.max()

def project_feasible(y, S):
    """Project y onto {0<=x<=1, sum(x)=S}"""
    n = len(y)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        if x.sum() < S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix exact sum
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mask.any():
            idxs = np.where(mask)[0]
            x[idxs[0]] += diff
    x = np.clip(x, 0.0, 1.0)
    return x

def compute_lse_weights(c_vals, alpha):
    """Compute softmax weights from correlation values."""
    # c_vals has length 2*N-1
    scaled = alpha * c_vals
    scaled -= scaled.max()  # numerical stability
    w = np.exp(scaled)
    w /= w.sum()
    return w

def compute_gradient(h, w, dx, N):
    """Compute gradient of LSE objective.
    w: weights of length 2N-1 (normalized)
    Returns gradient of same shape as h.
    """
    grad = dx * np.ones(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (N - 1)
        if lag >= 0:
            # h[j+lag] valid for j=0..N-1-lag
            grad[:N - lag] -= wk * h[lag:]
            # h[j-lag] valid for j=lag..N-1
            grad[lag:] -= wk * h[:N - lag]

            grad[l:] -= wk * h[:N - l]
            # h[j-lag] = h[j+l], valid for j=0..N-1-l
            grad[:N - l] -= wk * h[l:]
    return grad

    """Optimize h using projected gradient with LSE smoothing."""
    best_h = h0.copy()
    best_val = evaluate_c5(h0)
    S = N / 2.0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Compute correlation values
        c = np.correlate(h0, 1 - h0, mode="full") * dx
        
        # Compute weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = compute_lse_weights(c, alpha)
        
        # Compute gradient
        grad = compute_gradient(h0, w, dx, N)
        
        # Adaptive step size
        step = step_start / (1.0 + 0.01 * it)
        
        # Try step
        new_h = h0 - step * grad
        new_h = project_feasible(new_h, S)
        
        new_val = evaluate_c5(new_h)
        
        if new_val < best_val:
            best_h = new_h.copy()
            best_val = new_val
            h0 = new_h
        else:
            # Reduce step
            step_start = step_start * 0.9
            new_h = h0 - step_start * grad
            new_h = project_feasible(new_h, S)
            new_val = evaluate_c5(new_h)
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
                h0 = new_h
    
    return best_h, best_val

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return c.max()

def project_feasible(y, S):
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        if x.sum() < S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

    if abs(diff) > 1e-10:
        mask = (x > 1e-8) & (x < 1.0 - 1e-8)
        if mask.any():
            idxs = np.where(mask)[0]
            x[idxs[0]] += diff

    scaled = alpha * c_vals
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s > 1e-30:
        w /= s
    else:
        w = np.ones(len(c_vals)) / len(c_vals)
    return w
