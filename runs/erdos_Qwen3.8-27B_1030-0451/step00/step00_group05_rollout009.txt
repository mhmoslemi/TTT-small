            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                pert = rng.standard_normal(n) * 0.01
                pert -= pert.mean()
                h = project_feasible(h + pert, n)
                no_improve = 0
                continue

        # Find top-K lags
        sorted_idx = np.argsort(C)[::-1]
        active = sorted_idx[:top_k]
        # Equal weights
        w_active = np.zeros(len(C))
        w_active[active] = 1.0 / top_k

        # Compute gradient
        dx = 2.0 / n
        c0 = n - 1
        grad = np.zeros(n)
        for k in active:
            r = k - c0
            wk = 1.0 / top_k
            if r == 0:
                grad += wk * dx * (1.0 - 2.0 * h)
            elif r > 0:
                e1 = min(r - 1, n - 1 - r)
                if e1 >= 0:
                    grad[0:e1 + 1] += wk * dx * (-h[r:r + e1 + 1])
                if r <= n - 1 - r:
                    ln = n - 2 * r
                    grad[r:n - r] += wk * dx * ((1.0 - h[0:ln]) - h[2 * r:n])
                grad[n - r:] += wk * dx * (1.0 - h[0:r])
            else:
                s = -r
                e1 = min(s - 1, n - 1 - s)
                if e1 >= 0:
                    grad[0:e1 + 1] += wk * dx * (1.0 - h[s:s + e1 + 1])
                if s <= n - 1 - s:
                    ln = n - 2 * s
                    grad[s:n - s] += wk * dx * ((1.0 - h[2 * s:n]) - h[0:ln])
                grad[n - s:] += wk * dx * (-h[0:s])

        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        direction = -grad / gnorm

        accepted = False
        for _ in range(20):
            h_try = project_feasible(h + step * direction, n)
            c5_try = eval_c5(h_try)
            if c5_try < cur_max:
                h = h_try
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step = 0.003

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def time_ok():
        return (time.time() - start_time) < budget_s - 30

    best_h = None
    best_c5 = float('inf')

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=np.float64)
        n_init = len(h_init)
        # Ensure feasibility
        h_init = project_feasible(h_init, n_init)
        c5_init = eval_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init
    else:
        # Create a reasonable starting vector
        n_init = 131
        h_init = np.full(n_init, 0.5)
        # Add some structure
        t = np.linspace(0, 4 * np.pi, n_init)
        h_init = 0.5 + 0.2 * np.sin(t)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init.copy()
        best_c5 = eval_c5(h_init)

    # Phase 1: Projected gradient with softmax (main workhorse)
    if time_ok():
        h_opt, c5_opt = optimize_projected_gradient(best_h, time_ok, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Top-K active set optimization
    if time_ok():
        remaining = budget_s - 30 - (time.time() - start_time)
        if remaining > 60:
            h_opt, c5_opt = optimize_topk(best_h, time_ok, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 3: Try resampling to nearby n values
    if time_ok():
        n_cur = len(best_h)
        for dn in [-2, -1, 1, 2, -3, 3, -4, 4]:
            if not time_ok():
                break
            n_try = n_cur + dn
            if n_try < 50 or n_try > 2500:
                continue
            h_res = resample_to(best_h, n_try)
            # Quick optimization on resampled version
            h_opt, c5_opt = optimize_projected_gradient(h_res, time_ok, rng)
            if c5_opt < best_c5:

                best_h = h_opt

    # Phase 4: More refined optimization on best found
    if time_ok():
        remaining = budget_s - 30 - (time.time() - start_time)
        if remaining > 30:
            # Do another round with smaller steps
            n = len(best_h)
            h = best_h.copy()
            alpha = 0.002
            step = 0.001
            for _ in range(2000):
                if not time_ok():
                    break
                C = compute_corrs(h)
                cur_max = C.max()
                if cur_max < best_c5:
                    best_c5 = cur_max
                    best_h = h.copy()
                grad = active_gradient(h, C, alpha=alpha)
                grad -= grad.mean()
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-16:
                    break
                direction = -grad / gnorm
                h_try = project_feasible(h + step * direction, n)
                c5_try = eval_c5(h_try)
                if c5_try < cur_max:
                    h = h_try
                    if cur_max >= best_c5:
                        best_c5 = c5_try
                        best_h = h.copy()
                else:
                    step *= 0.7
                alpha = max(0.0005, alpha * 0.999)

    # Phase 5: Final fine-tuning with very small steps
    if time_ok():
        n = len(best_h)
        h = best_h.copy()
        for _ in range(5000):
            if not time_ok():
                break
            C = compute_corrs(h)
            cur_max = C.max()
            if cur_max < best_c5:

                best_h = h.copy()

            sorted_idx = np.argsort(C)[::-1]
            active = sorted_idx[:5]
            dx = 2.0 / n
            c0 = n - 1
            grad = np.zeros(n)
            for k in active:

                wk = 0.2
                if r == 0:
                    grad += wk * dx * (1.0 - 2.0 * h)
                elif r > 0:
                    e1 = min(r - 1, n - 1 - r)
                    if e1 >= 0:
                        grad[0:e1 + 1] += wk * dx * (-h[r:r + e1 + 1])
                    if r <= n - 1 - r:
                        ln = n - 2 * r
                        grad[r:n - r] += wk * dx * ((1.0 - h[0:ln]) - h[2 * r:n])
                    grad[n - r:] += wk * dx * (1.0 - h[0:r])
                else:
                    s = -r
                    e1 = min(s - 1, n - 1 - s)
                    if e1 >= 0:
                        grad[0:e1 + 1] += wk * dx * (1.0 - h[s:s + e1 + 1])
                    if s <= n - 1 - s:
                        ln = n - 2 * s
                        grad[s:n - s] += wk * dx * ((1.0 - h[2 * s:n]) - h[0:ln])
                    grad[n - s:] += wk * dx * (-h[0:s])
            grad -= grad.mean()
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break
            direction = -grad / gnorm
            h_try = project_feasible(h + 0.0005 * direction, n)
            c5_try = eval_c5(h_try)
            if c5_try < cur_max:
                h = h_try

    # Final verification
    n_points = len(best_h)
    best_c5 = eval_c5(best_h)
    # Ensure feasibility
    best_h = project_feasible(best_h, n_points)
    best_c5 = eval_c5(best_h)

    return (best_h, best_c5, n_points)
