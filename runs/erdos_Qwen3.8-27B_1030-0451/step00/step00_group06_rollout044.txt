import numpy as np
import time

def project_to_simplex_box(v, target_sum):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(v)
    # Bisection on lambda: find lambda such that sum(clamp(v - lambda, 0, 1)) = target_sum
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h)) * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def compute_gradient(h):
    """Compute subgradient of C5 at h (averaged over active lags)."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    # Active set: lags where c is close to max
    active = np.where(c >= cmax - 1e-10)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)  # the lag
        # grad contribution: (1 - h[i-k]) * 1[0<=i-k<n] - h[i+k] * 1[0<=i+k<n]
        # For i in [0, n-1]:
        #   i-k in [0, n-1] means i in [k, k+n-1] intersect [0, n-1]
        #   i+k in [0, n-1] means i in [-k, n-1-k] intersect [0, n-1]
        if k >= 0:
            # i-k valid for i in [k, n-1]
            if k < n:
                grad[k:] += (1.0 - h[:n-k])
            # i+k valid for i in [0, n-1-k]
            if k < n:
                grad[:n-k] -= h[k:]
        else:
            kk = -k
            # i-k = i+kk valid for i in [0, n-1-kk]
            if kk < n:
                grad[:n-kk] += (1.0 - h[kk:])
            # i+k = i-kk valid for i in [kk, n-1]
            if kk < n:
                grad[kk:] -= h[:n-kk]
    grad *= dx / len(active)
    return grad, cmax * dx

        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode='full')

    return _corr(h).max() * (2.0 / n)

def _grad_topk(h, corr, top_k=10, temp=0.02):
    """Subgradient using top-k lags with exponential weights."""
    n = len(h)
    cmax = corr.max()
    # Get indices of top_k largest values
    if top_k >= len(corr):
        top = np.arange(len(corr))
    else:
        top = np.argpartition(corr, -top_k)[-top_k:]
    top = top[np.argsort(corr[top])[::-1]]
    diffs = cmax - corr[top]
    w = np.exp(-diffs / (temp * cmax + 1e-15))

            jk = -k
