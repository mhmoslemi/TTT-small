import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max()), corr

def project_feasible(v, n):
    h = np.clip(v.astype(np.float64), 0.0, 1.0)
    target = n * 0.5
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smoothed_gradient(h, k_active, beta):
    n = len(h)
    dx = 2.0 / n
    c0 = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    M = float(corr.max())
    total = len(corr)
    k = min(k_active, total)
    idx = np.argpartition(corr, -k)[-k:]
    idx = idx[np.argsort(corr[idx])[::-1]]
    w = np.exp((corr[idx] - M) * beta)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(k)
        w[0] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        lag = int(ki) - c0
        if lag == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            L = lag
            grad[:n - L] += wi * dx * (1.0 - h[L:])
            grad[L:] += -wi * dx * h[:n - L]
        else:
            L = -lag
            grad[L:] += wi * dx * (1.0 - h[:n - L])
            grad[:n - L] += -wi * dx * h[L:]
    return grad, M

def optimize_from(h_init, k_active, beta, max_iter, step0, patience_limit, rng):
    n = len(h_init)
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_val, _ = evaluate_c5(h)
    step = step0
    patience = 0
    for it in range(max_iter):
        if step < 1e-13:
            break
        grad, M = smoothed_gradient(h, k_active, beta)
        grad -= grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        direction = -grad / gn
        found = False
        s = step
        for _ in range(60):
            h_new = project_feasible(h + s * direction, n)
            v_new, _ = evaluate_c5(h_new)
            if v_new < best_val:
                h = h_new
                best_h = h.copy()
                best_val = v_new
                patience = 0
                step = min(s * 1.2, step0)
                found = True
                break
            s *= 0.5
        if not found:
            patience += 1
            step = max(step * 0.85, 1e-13)
            if patience % 25 == 0:
                ns = 0.02 + 0.06 * (patience / max(patience_limit, 1))
                noise = rng.standard_normal(n) * ns
                h_pert = project_feasible(h + noise, n)
                v_pert, _ = evaluate_c5(h_pert)
                if v_pert < best_val:
                    h = h_pert
                    best_h = h.copy()
                    best_val = v_pert
                    patience = 0
                    step = max(step0 * 0.4, 0.05)
            if patience >= patience_limit:
                break
        if it > 0 and it % 400 == 0:
            beta = min(beta * 1.25, 300.0)
    return best_h, best_val

def resample_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)
