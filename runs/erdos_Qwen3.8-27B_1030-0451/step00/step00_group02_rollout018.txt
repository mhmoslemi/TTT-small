def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")

def project_onto_feasible(h, target_sum):
    """Project h onto {0≤x≤1, sum(x)=target_sum}."""
    # Use bisection on the Lagrange multiplier

def solve_lp(h0, n):
    """Solve the LP relaxation at point h0."""
    # Build constraint matrix
    # Variables: h[0..n-1], t (total n+1 variables)
    # For each lag k (0..2n-2), build the linear constraint
    # Plus: 0 ≤ h ≤ 1, sum(h) = n/2
    
    m = 2*n - 1  # number of lag constraints
    A_ub = np.zeros((m, n+1))
    b_ub = np.zeros(m)
    
    for k in range(m):
        s = k - (n-1)
        # Valid i: max(0,-s) ≤ i ≤ min(n-1, n-1-s)
        i_lo = max(0, -s)
        i_hi = min(n-1, n-1-s)
        b_val = 0.0
        for i in range(i_lo, i_hi+1):
            # Coefficient of h[i] in first sum
            A_ub[k, i] += 1.0
            # Coefficient of h[i+s] in second sum (with coefficient -2*h0[i])
            j = i + s
            if 0 <= j < n:
                A_ub[k, j] += -2.0 * h0[i]
            b_val += h0[i] * (h0[i+s] if 0 <= i+s < n else 0.0)
        # Constraint: A_ub[k,:] @ [h; t] ≤ -b_val
        # i.e., (sum h[i] - 2*sum h0[i]*h[i+s]) - t ≤ -b_val
        A_ub[k, n] = -1.0
        b_ub[k] = -b_val
