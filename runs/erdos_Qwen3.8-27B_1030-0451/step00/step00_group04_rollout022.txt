    q is index in [0, 2n-2], corresponding to lag = q - (n-1).

import numpy as np
import time
from scipy.optimize import linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def exact_z_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_z(h, q):
    n = len(h)
    lag = q - (n - 1)
    g = np.zeros(n)
    a = abs(lag)
    if a == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        L = n - a
        g[a:n] += (1.0 - h[0:L])
        g[0:L] -= h[a:n]
    return g

def project_box_sum(h_raw, T):
    """Project h_raw to satisfy 0<=h<=1 and sum(h)=T using bisection on shift + small correction."""
    n = len(h_raw)
    # Bisection to find alpha such that sum(clip(h_raw + alpha, 0, 1)) = T
    lo = -h_raw.max() - 1.0
    hi = 1.0 - h_raw.min() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h_raw + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    h = np.clip(h_raw + alpha, 0.0, 1.0)
    # Small correction for floating point
    err = T - np.sum(h)
    if abs(err) > 1e-12:
        # Distribute error
        if err > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        n_fix = len(idx)
        if n_fix > 0:
            h[idx] += err / n_fix
            h = np.clip(h, 0.0, 1.0)
    return h

def slp_optimize(h0, max_iters=100, time_budget=60.0, start_time=None):
    """Run SLP optimization starting from h0."""
    if start_time is None:
        start_time = time.time()
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    T = n / 2.0
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    rho = max(1e-3, 0.02 * dx * 10)  # initial trust radius
    min_rho = 1e-5
    max_rho = 0.05
    stall = 0
    max_stall = 20
    
    for it in range(max_iters):
        if time.time() - start_time > time_budget:
            break
        if current_score >= best_score:
            # We're at best, try to improve
            pass
        if stall > max_stall:
            break
        
        # Compute z vector and active set
        zvec = exact_z_vector(h)
        z_max = np.max(zvec)
        # Active set: top K or those within gap
        gap = max(1e-5, 0.02 * z_max)
        active_mask = zvec >= (z_max - gap)
        active_idx = np.where(active_mask)[0]
        # Cap active set size
        if len(active_idx) > 64:
            order = np.argsort(-zvec[active_idx])
            active_idx = active_idx[order[:64]]
        
        # Build LP
        K = len(active_idx)
        if K == 0:
            break
        
        # Objective: min t
        c_obj = np.concatenate([np.zeros(n), [1.0]])
        
        # Inequality constraints: dx * g_q^T d - t <= -z_q for each active q
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        for iq, q in enumerate(active_idx):
            g = grad_z(h, q) * dx
            A_ub[iq, :n] = g
            A_ub[iq, n] = -1.0
            b_ub[iq] = -zvec[q]
        
        # Equality: sum d = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds
        lb = np.maximum(0.0, h - rho)
        ub = np.minimum(1.0, h + rho)
        bounds = list(zip(lb, ub)) + [(0.0, np.inf)]
        
        # Solve
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n]
                h_new = h + d
                h_new = np.clip(h_new, 0.0, 1.0)
                # Fix sum
                h_new = project_box_sum(h_new, T)
                score_new = exact_score(h_new)
                if score_new < current_score - 1e-12:
                    h = h_new
                    current_score = score_new
                    if score_new < best_score:
                        best_score = score_new
                        best_h = h.copy()
                    stall = 0
                    rho = min(max_rho, rho * 1.1)
                else:
                    stall += 1
                    rho = max(min_rho, rho * 0.5)
            else:
                stall += 1
                rho = max(min_rho, rho * 0.5)
        except Exception:
            stall += 1
            rho = max(min_rho, rho * 0.5)
    
    return best_h, best_score

    if start_time is None:
        start_time = time.time()
    h = h.copy()
    n = len(h)
    T = n / 2.0
    dx = 2.0 / n
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    no_improve = 0
    max_no_improve = 40
    
    fracs = np.array([0.1, 0.3, 0.5, 0.7, 0.9, 1.0])
    
    for it in range(500):

        if no_improve > max_no_improve:

        zvec = exact_z_vector(h)
        z_max = np.max(zvec)
        gap = max(1e-5, 0.02 * z_max)
        active_mask = zvec >= (z_max - gap)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > 32:

            active_idx = active_idx[order[:32]]
        
        # Average gradient over active constraints
        g_avg = np.zeros(n)
        for q in active_idx:
            g_avg += grad_z(h, q)
        g_avg /= len(active_idx)
        
        # Find transfer: decrease h at high g index, increase at low g index
        g_sorted_idx = np.argsort(g_avg)
        # Source: highest g (want to decrease), Target: lowest g (want to increase)
        improved = False
        for _ in range(20):
            i = g_sorted_idx[np.random.randint(n)]  # high g
            j = g_sorted_idx[np.random.randint(n)]  # low g
            # Actually let's pick from top and bottom
            break
        
        # Better: try several pairs
        top_idx = g_sorted_idx[-min(20, n):]  # highest gradient
        bot_idx = g_sorted_idx[:min(20, n)]   # lowest gradient
        
        for i in top_idx[:10]:
            for j in bot_idx[:10]:
                if i == j:

                max_delta = min(h[i], 1.0 - h[j])
                if max_delta < 1e-8:

                    h_alt[i] -= delta

                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:

        else:
            no_improve = 0

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def exact_z_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def grad_z(h, q):
    n = len(h)
    lag = q - (n - 1)
    g = np.zeros(n)
    a = abs(lag)
    if a == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        L = n - a
        g[a:n] += (1.0 - h[0:L])
        g[0:L] -= h[a:n]
    return g

def project_box_sum(h_raw, T):
    n = len(h_raw)
    # Bisection for alpha: sum(clip(h_raw + alpha, 0, 1)) = T
    f = h_raw.copy()
    lo = -f.max() - 1.0
    hi = 1.0 - f.min() + 1.0

            lo = mid
        else:
            hi = mid

    # Fix small residual

import numpy as np
import time
from scipy.optimize import linprog


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def exact_z_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def grad_z(h, q):
    n = len(h)
    lag = q - (n - 1)
    g = np.zeros(n)
    a = abs(lag)
    if a == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        L = n - a
        g[a:n] += (1.0 - h[0:L])
        g[0:L] -= h[a:n]
    return g


def project_box_sum(h_raw, T):
    n = len(h_raw)
    lo = -h_raw.max() - 1.0
    hi = 1.0 - h_raw.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h_raw + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    h = np.clip(h_raw + alpha, 0.0, 1.0)
    err = T - np.sum(h)
    if abs(err) > 1e-12:
        if err > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        nm = np.sum(mask)
        if nm > 0:
            h[mask] += err / nm
            h = np.clip(h, 0.0, 1.0)
    return h


def slp_optimize(h0, time_budget=120.0, start_time=None, max_iters=300):
    if start_time is None:
        start_time = time.time()
    h = h0.astype(np.float64).copy()
    n = len(h)
    T = n / 2.0
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    rho = 0.008
    min_rho = 1e-7
    max_rho = 0.03
    stall = 0
    max_stall = 40

    for it in range(max_iters):
        if time.time() - start_time > time_budget:
            break
        if stall > max_stall:
            break

        zvec = exact_z_vector(h)
        z_max = np.max(zvec)
        gap = max(1e-6, 0.015 * z_max)
        active_mask = zvec >= (z_max - gap)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > 80:
            order = np.argsort(-zvec[active_idx])
            active_idx = active_idx[order[:80]]
        K = len(active_idx)
        if K == 0:
            break

        c_obj = np.concatenate([np.zeros(n), [1.0]])
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        for iq, q in enumerate(active_idx):
            g = grad_z(h, q) * (2.0 / n)
            A_ub[iq, :n] = g
            A_ub[iq, n] = -1.0
            b_ub[iq] = -zvec[q]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)

        lb = np.maximum(0.0, h - rho)
        ub = np.minimum(1.0, h + rho)
        bounds = list(zip(lb, ub)) + [(0.0, 10.0)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                stall += 1
                rho = max(min_rho, rho * 0.5)
                continue

            t_opt = res.x[n]
            if t_opt >= current_score - 1e-12:
                stall += 1
                rho = max(min_rho, rho * 0.5)
                continue

            d = res.x[:n]
            h_new = h + d
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_box_sum(h_new, T)
            if not np.all(np.isfinite(h_new)):
                stall += 1
                rho = max(min_rho, rho * 0.5)
                continue
            score_new = exact_score(h_new)
            if score_new < current_score - 1e-13:
                h = h_new
                current_score = score_new
                if score_new < best_score:
                    best_score = score_new
                    best_h = h.copy()
                stall = 0
                rho = min(max_rho, rho * 1.08)
            else:
                stall += 1
                rho = max(min_rho, rho * 0.55)
        except Exception:
            stall += 1
            rho = max(min_rho, rho * 0.5)

    return best_h, best_score


def pairwise_polish(h0, time_budget=60.0, start_time=None, max_rounds=300):
    if start_time is None:
        start_time = time.time()
    h = h0.astype(np.float64).copy()
    n = len(h)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    no_improve = 0
    max_no_improve = 80
    fracs = [0.05, 0.15, 0.3, 0.5, 0.75, 1.0]
    n_source = 20
    n_target = 20

    for rnd in range(max_rounds):
        if time.time() - start_time > time_budget:
            break
        if no_improve > max_no_improve:
            break

        zvec = exact_z_vector(h)
        z_max = np.max(zvec)
        gap = max(1e-6, 0.015 * z_max)
        active_mask = zvec >= (z_max - gap)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > 25:
            order = np.argsort(-zvec[active_idx])
            active_idx = active_idx[order[:25]]

        g_avg = np.zeros(n)
        for q in active_idx:
            g_avg += grad_z(h, q)
        g_avg /= max(1, len(active_idx))

        g_order = np.argsort(g_avg)
        sources = g_order[-n_source:]
        targets = g_order[:n_target]

        improved = False
        for i in sources:
            if improved:
                break
            for j in targets:
                if i == j:
                    continue
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta < 1e-9:
                    continue
                for frac in fracs:
                    delta = frac * max_delta
                    if delta < 1e-11:
                        continue
                    h_alt = h.copy()
                    h_alt[i] -= delta
                    h_alt[j] += delta
                    score_alt = exact_score(h_alt)
                    if score_alt < current_score - 1e-13:
                        h = h_alt
                        current_score = score_alt
                        if score_alt < best_score:
                            best_score = score_alt
                            best_h = h.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_h, best_score


def directed_transfer_polish(h0, time_budget=60.0, start_time=None, max_rounds=200):

        zvec = exact_z_vector(h)

        improved = False
        for q in active_idx:

                break

                                    improved = True
                                    break
                        if improved:
                            break
                continue

            L = n - a
            # For lag a: corr = sum_{j=0}^{L-1} h[j+a]*(1-h[j])

                            if t == source_idx:

                                    best_h = h.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_h, best_score
