    """Project v onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    # Bisection on the shift
    lo = -v.max() - 1.0
    hi = -v.min() + 1.0
    for _ in range(100):

        candidate = np.clip(v + mid, 0.0, 1.0)
        s = candidate.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(v + 0.5*(lo+hi), 0.0, 1.0)
    # Exact fix: adjust the largest-free-capacity element
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:

        if diff > 0:

        else:

        result[idx] = np.clip(result[idx], 0.0, 1.0)

        hi2 = min(n, n + k)
        if hi2 > lo2:

def softmin_gradient(h, T):
    """Gradient of (1/T)*log(sum_k exp(-T*S_k)) where S_k = sum_i h[i]*h[i+k].
    We want to minimize this (= maximize the softmin of S_k)."""
    n = len(h)
    # Compute S_k for all k
    # S_k = sum_i h[i]*h[i+k] for valid i
    # This is the autocorrelation of h
    auto = np.correlate(h, h, mode='full')
    # auto[j] = S_{j-(n-1)}
    
    # Softmin weights
    neg_T_S = -T * auto
    neg_T_S -= neg_T_S.max()  # for numerical stability
    weights = np.exp(neg_T_S)
    weights /= weights.sum()
    
    # Gradient: d/dh[m] of (1/T)*log(sum_k exp(-T*S_k))
    # = -sum_k weights[k] * S_k_grad_at_m
    # S_k = sum_i h[i]*h[i+k]
    # dS_k/dh[m] = h[m-k] + h[m+k] (for valid indices)
    
    grad = np.zeros(n)
    for j in range(2*n-1):
        wj = weights[j]
        if wj < 1e-8:
            continue
        k = j - (n-1)
        # dS_k/dh[m] = h[m-k] (if m-k valid) + h[m+k] (if m+k valid)
        lo1 = max(0, k)
        hi1 = min(n, n+k)
        if hi1 > lo1:
            grad[lo1:hi1] += wj * h[lo1-k:hi1-k]
        lo2 = max(0, -k)
        hi2 = min(n, n-k)
        if hi2 > lo2:
            grad[lo2:hi2] += wj * h[lo2+k:hi2+k]
    
    # The objective is (1/T)*log(sum exp(-T*S)), gradient is:
    # = (1/T) * sum_k (-T * exp(-T*S_k)) / (sum_j exp(-T*S_j)) * dS_k/dh
    # = -sum_k weights[k] * dS_k/dh
    grad = -grad
    return grad

For our case, the "smoothing kernel" has weights that decay as w_k, so the stable step size depends on the spectrum of the smoothing operator.

A safe choice is to use a small step size and increase it gradually. Or use a line search.

Let me also consider: the problem is to maximize min_k S_k. The optimal solution should have all S_k (for k in the "active" set) equal. This is like a Chebyshev approximation problem.

def project_feasible(v, target_sum):

def softmin_update(h, T):
    """Compute the update direction for the softmin objective.
    Returns the direction to ADD to h (i.e., -gradient of softmin)."""
    n = len(h)
    auto = np.correlate(h, h, mode='full')
    # Softmin weights: w_k proportional to exp(-T * S_k)
    neg_T_S = -T * auto
    neg_T_S -= neg_T_S.max()
    weights = np.exp(neg_T_S)
    weights /= weights.sum()
    
    # Update direction: sum_k w_k * (h[m-k] + h[m+k]) for valid indices
    direction = np.zeros(n)
    for j in range(2*n-1):
        wj = weights[j]
        if wj < 1e-10:
            continue
        k = j - (n-1)
        # h[m-k] for valid m: m-k in [0,n-1], so m in [k, n-1+k] -> [max(0,k), min(n-1, n-1+k)]
        lo1 = max(0, k)

    return direction
