import numpy as np
import time
from scipy.optimize import minimize, linprog

def grad_lag_k(h, k, n):
    """Gradient of v_k (lag value) with respect to h."""
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        if k < n:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
    else:
        m = -k
        if m < n:
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
    return g

def project_sum_box(x, target):
    """Project x onto {0<=x_i<=1, sum(x)=target}."""
    n = len(x)
    lo = -max(x.max(), 0.0) - 0.01
    hi = 1.0 - min(x.min(), 1.0) + 0.01
    def S(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0))
    for _ in range(70):
        mid = 0.5*(lo+hi)
        if S(mid) > target:
            hi = mid
        else:
            lo = mid
    lam = 0.5*(lo+hi)
    result = np.clip(x + lam, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-14:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.any(free):
            result[free] += residual / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lag_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def ls_exp_obj_grad(h, beta):
    """Log-sum-exp objective and gradient, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    v = lag_values(h)
    m = np.max(v)
    shifted = np.clip(beta * (v - m), -500, 500)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    f = (m + np.log(Z) / beta) * dx
    w = exp_s / Z
    # Gradient: dx * sum_k w_k * grad(v_k)
    grad = np.zeros(n)
    n_full = len(v)
    # Only process lags with significant weight
    thresh = 1e-14
    for idx in range(n_full):
        if w[idx] < thresh:
            continue
        k = idx - (n - 1)
        grad += w[idx] * grad_lag_k(h, k, n)
    return f, grad * dx

def ls_exp_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    v = lag_values(h)
    m = np.max(v)
    shifted = np.clip(beta * (v - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def slsqp_round(h0, beta, maxiter=300, ftol=1e-12):
    """Single SLSQP round with given beta."""
    n = len(h0)
    target = n / 2.0
    def obj(h):
        return ls_exp_obj(h, beta)
    def obj_g(h):
        _, g = ls_exp_obj_grad(h, beta)
        return g
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=obj_g, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.x is not None:

def active_grad_descent(h0, n_iters=400, step_init=0.01, M=8, beta_w=5.0):

    n = len(h0)
    target = n / 2.0
    h_cur = h0.copy()
    best_h = h0.copy()

    step = step_init

        idx_top = np.argsort(v)[::-1][:M]

        w_exp = np.exp(beta_w * (w - w.max()))
        w_exp = w_exp / w_exp.sum()
        grad = np.zeros(n)

            grad += w_exp[i] * grad_lag_k(h_cur, k, n)

        k = idx - (n - 1)
        g = grad_lag_k(h0, k, n)
        row = np.zeros(nv)
        row[:n] = g
        row[-1] = -1.0
        rows.append(row)
        b_ub.append(g @ h0 - v0[idx])
    # h >= h0 - radius
    for i in range(n):
        row = np.zeros(nv)
        row[i] = -1.0
        rows.append(row)
        b_ub.append(-(h0[i] - radius))
    # h <= h0 + radius
    for i in range(n):
        row = np.zeros(nv)
        row[i] = 1.0
        rows.append(row)
        b_ub.append(h0[i] + radius)
    A_ub = np.array(rows)
    b_ub = np.array(b_ub)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(-0.5, 1.5)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            h_trial = project_sum_box(res.x[:n], target)
            return h_trial
    except Exception:
        pass
    return None

def lp_phase(h0, n_iters=50, radius_init=0.15, M=10):
    """Trust-region LP phase."""
    n = len(h0)
    target = n / 2.0
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c = exact_c5(h0)
    radius = radius_init
    no_improve = 0
    for it in range(n_iters):
        v = lag_values(h_cur)
        idx_top = np.argsort(v)[::-1][:M]
        h_try = trust_region_lp_solve(h_cur, idx_top, radius, n)
        if h_try is not None:
            c_try = exact_c5(h_try)
            if c_try < best_c - 1e-12:
                best_h = h_try.copy()
                best_c = c_try
                h_cur = h_try
                no_improve = 0
                radius = min(radius * 1.2, 0.3)
            else:
                no_improve += 1
                radius *= 0.6
        else:
            no_improve += 1
            radius *= 0.6
        if radius < 0.005 or no_improve > 15:
            break
    return best_h, best_c

def transfer_to_size(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) + 0.5
    x_new = np.arange(n_new) + 0.5
    x_old = x_old * 2.0 / n_old
    x_new = x_new * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    def remaining():
        return budget_s - (time.time() - t_start)
    def ok(min_s=5.0):
        return remaining() > min_s

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n0 = len(h0)
    except (NameError, Exception):
        n0 = 84
        h0 = np.full(n0, 0.5)

    h0 = project_sum_box(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c = exact_c5(h0)
    n_best = n0

    # Phase 1: SLSQP with increasing beta
    if ok(60):

                    best_c = c_try

                best_c = c_try

    # Phase 4: More SLSQP at higher beta
    if ok(60):

            h_try = slsqp_round(best_h, beta, maxiter=300, ftol=1e-13)
            if h_try is not None:
                c_try = exact_c5(h_try)
                if c_try < best_c - 1e-12:
                    best_h = h_try
                    best_c = c_try

    # Phase 5: Resolution transfer to n=168
    if ok(120):

        if c_fine < best_c - 1e-12:
            best_h = h_fine
            best_c = c_fine
            n_best = n_fine
        if ok(60):
            for beta in [5, 20, 100, 300]:
                if not ok(30):
                    break
                h_try = slsqp_round(best_h, beta, maxiter=200)
                if h_try is not None:
                    c_try = exact_c5(h_try)
                    if c_try < best_c - 1e-12:
                        best_h = h_try
                        best_c = c_try
                        n_best = len(best_h)
        if ok(40):
            h_try, c_try = lp_phase(best_h, n_iters=40, radius_init=0.1, M=12)
            if c_try < best_c - 1e-12:
                best_h = h_try
                best_c = c_try
                n_best = len(best_h)
        if ok(30):
            h_try, c_try = active_grad_descent(best_h, n_iters=400,
                                              step_init=0.005, M=10)
            if c_try < best_c - 1e-12:

    # Phase 6: Try n=252 if time allows
    if ok(120):
