import numpy as np
import time
from scipy.optimize import minimize

def exact_c5(h):
    """Compute the exact C5 value for a given h."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, T):
    """Project h onto {0 <= x <= 1, sum(x) = T} using bisection on lambda."""
    n = len(h)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = T
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    def sum_at(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    # Bisection
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = sum_at(mid)
        if s > T:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Small correction for floating point
    diff = T - np.sum(result)
    if abs(diff) > 1e-12:
        # Distribute correction among elements not at bounds
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_max_and_grad(h, tau):
    """Compute smooth max of correlation lags and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    z = c * dx
    
    # Softmax weights
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted / tau)
    p = exp_z / np.sum(exp_z)
    
    F = (np.max(z) + tau * np.log(np.sum(exp_z)))
    
    # Gradient: sum_q p_q * dC_ell_q/dh * dx
    # For lag ell = q - (n-1):
    # dC_ell/dh[j] = (1-h[j-ell])*[0<=j-ell<n] - h[j+ell]*[0<=j+ell<n]
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        if p[q] < 1e-15:
            continue
        ell = q - (n - 1)
        if ell >= 0:
            # (1-h[j-ell]) for j in [ell, n-1]
            # -h[j+ell] for j in [0, n-1-ell]
            grad[ell:] += p[q] * (1.0 - h[:n-ell]) * dx
            grad[:n-ell] += p[q] * (-h[ell:]) * dx
        else:
            a = -ell
            # (1-h[j+a]) for j in [0, n-1-a]
            # -h[j-a] for j in [a, n-1]
            grad[:n-a] += p[q] * (1.0 - h[a:]) * dx
            grad[a:] += p[q] * (-h[:n-a]) * dx
    
    return F, grad

def pgd_optimize(h0, T, budget_s, seed=42):
    """Projected gradient descent with smooth max surrogate."""
    n = len(h0)
    dx = 2.0 / n
    h = project_box_sum(h0.copy(), T)
    
    # Initial tau
    c = np.correlate(h, 1.0 - h, mode='full')
    z = c * dx
    z_range = np.max(z) - np.min(z)
    tau = np.clip(0.15 * z_range + 0.005, 0.003, 0.03)
    tau_min = 2e-5
    
    F, _ = smooth_max_and_grad(h, tau)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    alpha = 0.02
    start_time = time.time()
    deadline = start_time + budget_s
    
    i = 0
    while time.time() < deadline - 5:
        F, grad = smooth_max_and_grad(h, tau)
        
        # Check gradient norm
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-10:
            tau *= 0.7
            if tau < tau_min:
                break
            continue
        
        # Normalize direction
        direction = -grad / gnorm
        
        # Line search
        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = alpha * factor
            h_new = project_box_sum(h + step * direction, T)
            F_new, _ = smooth_max_and_grad(h_new, tau)
            if F_new < F - 1e-12:
                h = h_new
                F = F_new
                accepted = True
                break
            elif F_new < F:
                h = h_new
                F = F_new
                accepted = True
                break

            if alpha < 1e-10:
                alpha = 0.02
                tau *= 0.7

def slsqp_polish(h0, T, maxiter=200):
    """Polish using SLSQP epigraph formulation."""
    n = len(h0)
    dx = 2.0 / n
    h0 = project_box_sum(h0.copy(), T)
    c5_init = exact_c5(h0)
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    # Nonlinear constraints: corr(h)*dx - t <= 0
    def constraint_val(x):
        h = x[:n]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        return c - t
    
    def constraint_jac(x):
        h = x[:n]
        t = x[-1]
        # Jacobian is (2n-1) x (n+1)
        J = np.zeros((2*n-1, n+1))
        # Last column: -1 for all rows
        J[:, -1] = -1.0
        # First n columns: d(c_q)/d(h_j) * dx
        for q in range(2*n-1):
            ell = q - (n-1)
            if ell >= 0:
                if n - ell > 0:
                    J[q, ell:] += (1.0 - h[:n-ell]) * dx
                    J[q, :n-ell] += (-h[ell:]) * dx
            else:
                a = -ell
                if n - a > 0:
                    J[q, :n-a] += (1.0 - h[a:]) * dx
                    J[q, a:] += (-h[:n-a]) * dx
        return J
    
    # Equality constraint: sum(h) = T
    def eq_val(x):
        return x[:n].sum() - T
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    x0 = np.concatenate([h0, [c5_init + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    cons = [
        {'type': 'ineq', 'fun': lambda x: -constraint_val(x), 'jac': lambda x: -constraint_jac(x)},
        {'type': 'eq', 'fun': eq_val, 'jac': eq_jac}
    ]
    
    try:
        res = minimize(objective, x0, jac=objective_grad,
                      bounds=bounds, constraints=cons,
                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(res.x[:n], T)
        c5_new = exact_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h0, c5_init

def try_symmetries(h, T, budget_per=30):
    """Try symmetry transformations and optimize each."""
    results = []
    transforms = [
        h.copy(),
        h[::-1].copy(),
        (1.0 - h).copy(),
        (1.0 - h)[::-1].copy(),
    ]
    
    for t_h in transforms:

        h_opt, c5_opt = pgd_optimize(t_h, T, budget_per)
        results.append((h_opt, c5_opt))
    
    return results

def upsample(h, factor):
    """Upsample by repeating each value 'factor' times."""
    return np.repeat(h, factor)

def downsample(h, factor):
    """Downsample by averaging blocks of size 'factor'."""
    n = len(h)
    n_blocks = n // factor
    h_blocks = h[:n_blocks * factor].reshape(n_blocks, factor)
    return h_blocks.mean(axis=1)

import numpy as np

from scipy.optimize import minimize

def exact_c5(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx using np.correlate."""
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, T):
    """Project h onto {x : 0<=x_i<=1, sum(x)=T}."""
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > T:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - float(result.sum())
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            result[mask] += diff / cnt
            np.clip(result, 0.0, 1.0, out=result)
    return result

def smooth_max_and_grad(h, tau):
    """Compute softmax-smoothed max of corruption lags and gradient."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    z = c * dx
    zmax = float(z.max())
    exp_z = np.exp((z - zmax) / tau)
    exp_sum = float(exp_z.sum())
    p = exp_z / exp_sum
    F = zmax + tau * np.log(exp_sum)
    
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = p[q]
        if w < 1e-14:
            continue
        ell = q - (n - 1)
        if ell >= 0:
            L = n - ell
            if L > 0:
                grad[ell:] += w * (1.0 - h[:L]) * dx
                grad[:L] -= w * h[ell:] * dx
        else:
            a = -ell
            L = n - a
            if L > 0:
                grad[:L] += w * (1.0 - h[a:]) * dx
                grad[a:] -= w * h[:L] * dx
    return F, grad

def pgd_optimize(h0, T, budget_s):
    """PGD with smooth max surrogate and adaptive temperature."""
    n = len(h0)
    h = project_box_sum(h0.copy(), T)
    
    c = np.correlate(h, 1.0 - h, mode='full')
    z = c * (2.0 / n)
    tau = float(np.clip(0.12 * (z.max() - z.min()) + 0.004, 0.002, 0.025))
    tau_min = 5e-6
    
    F, _ = smooth_max_and_grad(h, tau)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    
    alpha = 0.025
    start = time.time()
    deadline = start + budget_s
    i = 0
    stall = 0
    
    while time.time() < deadline - 1.0:
        F, grad = smooth_max_and_grad(h, tau)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-13:
            tau *= 0.6
            if tau < tau_min:
                break
            continue
        
        direction = -grad / gnorm
        accepted = False
        for f in (1.0, 0.5, 0.25, 0.125, 0.0625):
            h_new = project_box_sum(h + alpha * f * direction, T)
            F_new, _ = smooth_max_and_grad(h_new, tau)
            if F_new <= F + 1e-15:
                h = h_new
                F = F_new
                accepted = True
                break
        
        if not accepted:
            alpha *= 0.5
            if alpha < 1e-13:
                alpha = 0.025
                tau *= 0.6
                if tau < tau_min:
                    break
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0
            i += 1
        
        if i % 10 == 0:
            c5_val = exact_c5(h)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()
        
        if i % 30 == 0:
            tau = max(tau * 0.8, tau_min)
    
    c5_val = exact_c5(h)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h.copy()
    return best_h, best_c5

def slsqp_polish(h0, T, tau, maxiter):
    """SLSQP polish with smooth max as objective."""
    n = len(h0)
    h0 = project_box_sum(h0.copy(), T)
    
    def obj(x):
        F, _ = smooth_max_and_grad(x, tau)
        return F
    
    def obj_grad(x):
        _, g = smooth_max_and_grad(x, tau)
        return g
    
    def eq_con(x):
        return float(x.sum()) - T
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h0.copy(), jac=obj_grad, bounds=bounds,
                      constraints=cons, method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_box_sum(res.x, T)
        return h_new, exact_c5(h_new)
    except Exception:
        return h0, exact_c5(h0)

import numpy as np
import time
from scipy.optimize import minimize

def exact_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, T):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > T:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - float(result.sum())
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            result[mask] += diff / cnt
            np.clip(result, 0.0, 1.0, out=result)
    return result

def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    z = c * dx
    zmax = float(z.max())
    exp_z = np.exp((z - zmax) / tau)
    exp_sum = float(exp_z.sum())
    p = exp_z / exp_sum
    F = zmax + tau * np.log(exp_sum)
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = p[q]
        if w < 1e-14:
            continue
        ell = q - (n - 1)
        if ell >= 0:
            L = n - ell
            if L > 0:
                grad[ell:] += w * (1.0 - h[:L]) * dx
                grad[:L] -= w * h[ell:] * dx
        else:
            a = -ell
            L = n - a
            if L > 0:
                grad[:L] += w * (1.0 - h[a:]) * dx
                grad[a:] -= w * h[:L] * dx
    return F, grad

def pgd_optimize(h0, T, budget_s):
    n = len(h0)
    h = project_box_sum(h0.copy(), T)
    c = np.correlate(h, 1.0 - h, mode='full')
    z = c * (2.0 / n)
    tau = float(np.clip(0.12 * (z.max() - z.min()) + 0.004, 0.003, 0.025))
    tau_min = 3e-6
    F, _ = smooth_max_and_grad(h, tau)
    best_h = h.copy()
    best_c5 = exact_c5(h)
    alpha = 0.025
    start = time.time()
    deadline = start + budget_s
    i = 0
    stall = 0
    while time.time() < deadline - 1.0:
        F, grad = smooth_max_and_grad(h, tau)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-13:
            tau *= 0.6
            if tau < tau_min:
                break
            continue
        direction = -grad / gnorm
        accepted = False
        for f in (1.0, 0.5, 0.25, 0.125, 0.0625):
            h_new = project_box_sum(h + alpha * f * direction, T)
            F_new, _ = smooth_max_and_grad(h_new, tau)
            if F_new <= F + 1e-15:
                h = h_new
                F = F_new
                accepted = True
                break
        if not accepted:
            alpha *= 0.5
            if alpha < 1e-13:
                alpha = 0.025
                tau *= 0.6
                if tau < tau_min:
                    break
            stall += 1
