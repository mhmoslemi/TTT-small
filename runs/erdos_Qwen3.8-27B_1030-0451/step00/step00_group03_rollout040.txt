    lo, hi = -max(np.abs(y.min()), 1.0) - 1, max(np.abs(y.max()), 1.0) + 1

        else:
            lo = mid

    h = np.clip(y + alpha, 0.0, 1.0)

    residual = target_sum - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:

            room = 1.0 - h
            if room.sum() > 0:
                add = np.minimum(room, residual / n)
                if add.sum() < residual:

                    order = np.argsort(room)[::-1]
                    remaining = residual - add.sum()
                    for idx in order:
                        if remaining <= 0:
                            break
                        amt = min(room[idx], remaining)
                        h[idx] += amt
                        room[idx] = 1.0 - h[idx]
                        remaining -= amt
                else:
                    h += add
        else:
            # Need to subtract
            space = h
            if space.sum() > 0:

                        if remaining <= 0:
                            break
                        amt = min(space[idx], remaining)
                        h[idx] -= amt
                        space[idx] = h[idx]
                        remaining -= amt

                    h -= sub
    
    return h

import time
from scipy.optimize import minimize

def project_feasible(y, target_sum):
    y = np.asarray(y, dtype=np.float64).copy()
    n = len(y)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(y, -2.0, 2.0)
    
    lo = 0.0 - y.max() - 0.1
    hi = 1.0 - y.min() + 0.1
    
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    alpha = (lo + hi) * 0.5
    h = np.clip(y + alpha, 0.0, 1.0)
    
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            room = 1.0 - h
            mask = room > 1e-15
            if mask.sum() > 0:
                h[mask] += residual / mask.sum()
                h = np.clip(h, 0.0, 1.0)
        else:
            space = h
            mask = space > 1e-15
            if mask.sum() > 0:
                h[mask] += residual / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    
    # Final exact fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        # Find indices to adjust
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(residual) < 1e-14:
                break
            if residual > 0:
                amt = min(1.0 - h[idx], residual)
            else:
                amt = max(h[idx] + residual, -residual)
                amt = min(h[idx], -residual)
            h[idx] += amt
            residual = target_sum - h.sum()
    
    return h

def evaluate_c5(h):

    dx = 2.0 / n

def lag_gradient(h, lag, dx):

    g = np.zeros(n, dtype=np.float64)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[lag:] += (1.0 - h[:L])
            g[:L] += -h[lag:]
    else:
        r = -lag
        L = n - r
        if L > 0:
            g[r:] += (1.0 - h[:L])

def lse_smoothed_gradient(h, dx, tau, n_top):
    """Compute gradient of LSE-smoothed max using top n_top lags."""
    lags = compute_lags(h)
    fmax = lags.max()
    shifted = (lags - fmax) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    # Only use top n_top for gradient computation
    n = len(h)
    idx_sorted = np.argsort(lags)[::-1][:n_top]
    w_sum = weights[idx_sorted].sum()
    if w_sum < 1e-15:
        return np.zeros(n, dtype=np.float64), fmax
    w_top = weights[idx_sorted] / w_sum
    
    grad = np.zeros(n, dtype=np.float64)
    for i, m in enumerate(idx_sorted):
        lag = m - (n - 1)
        grad += w_top[i] * lag_gradient(h, lag, dx)
    
    return grad, fmax

def pgd_optimize(h0, n_iters=500, tau_start=0.05, tau_end=0.005, eta=0.01, 
                 n_top=30, time_limit=None, t_start=None):
    """Projected gradient descent with LSE smoothing."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_h = h.copy()
    best_c = evaluate_c5(h)
    
    tau = tau_start
    tau_ratio = (tau_end / tau_start) ** (1.0 / max(n_iters, 1))
    
    for it in range(n_iters):
        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        
        grad, fmax = lse_smoothed_gradient(h, dx, tau, n_top)
        
        if np.allclose(grad, 0):
            break
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Try step
        for scale in [eta, eta*0.5, eta*0.25, eta*0.1, eta*0.05]:
            step = scale * grad / gn
            y = h - step
            h_new = project_feasible(y, target_sum)
            c_new = evaluate_c5(h_new)
            if c_new < best_c:
                h = h_new
                best_h = h.copy()
                best_c = c_new
                break
            elif it < n_iters // 2:
                # In exploration phase, allow small worsening
                smoothed = fmax + tau * np.log(np.sum(np.exp(np.clip((compute_lags(h_new) - fmax)/tau, -50, 50))))
                if c_new < best_c + tau * 0.1:
                    h = h_new
                    break
        else:
            # No step accepted; increase tau slightly
            tau *= 1.2
            if tau > 0.1:
                break
        
        tau *= tau_ratio
    
    return best_h, best_c

def slsqp_polish(h0, active_lags, maxiter=50, time_limit=None, t_start=None):
    """SLSQP optimization of min t s.t. f_m(h) <= t for active m."""
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    current_lags = compute_lags(h0)
    t0 = current_lags[active_lags].max() + 1e-4
    
    def objective(x):
        return x[n]  # t is the last variable
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # Constraints: f_m(h) <= t for each active lag
    # Nonlinear constraint: t - f_m(h) >= 0
    def cons_fn(x):
        h = x[:n]
        t = x[n]
        result = np.zeros(len(active_lags))
        for i, m in enumerate(active_lags):
            lag = m - (n - 1)
            # Compute f_m
            if lag >= 0:
                L = n - lag
                result[i] = t - dx * np.dot(h[lag:lag+L], 1.0 - h[:L])
            else:
                r = -lag
                L = n - r
                result[i] = t - dx * np.dot(h[r:r+L], 1.0 - h[:L])
        # Equality: sum(h) - target = 0
        result = np.append(result, np.sum(h) - target_sum)
        return result
    
    def cons_jac(x):
        h = x[:n]
        t = x[n]
        n_act = len(active_lags)
        J = np.zeros((n_act + 1, n + 1))
        for i, m in enumerate(active_lags):
            lag = m - (n - 1)
            J[i, n] = 1.0  # d(t - f_m)/dt = 1
            g = lag_gradient(h, lag, dx)
            J[i, :n] = -g  # d(t - f_m)/dh = -df_m/dh
        # Equality constraint
        J[n_act, :n] = 1.0  # d(sum(h) - target)/dh = 1
        return J
    
    x0 = np.append(h0, t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, t0 + 0.1)]
    
    try:
        res = minimize(objective, x0, method='SLSQP', 
                      bounds=bounds, constraints=[
                          {'type': 'ineq', 'fun': cons_fn, 'jac': cons_jac},
                      ],
                      options={'maxiter': maxiter, 'ftol': 1e-8})
        
        if res.success or res.x is not None:
            h_new = project_feasible(res.x[:n], target_sum)
            c_new = evaluate_c5(h_new)
            return h_new, c_new
    except Exception:
        pass
    
    return h0.copy(), evaluate_c5(h0)

def upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation, then project."""
    n = len(h)
    target_sum = n_new / 2.0
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    # Actually not periodic - use standard interpolation
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    h_new = project_feasible(h_new, target_sum)
    return h_new

    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()

import numpy as np
import time

def project_feasible(y, target_sum):
    """Exact projection onto {0<=h<=1, sum(h)=target_sum}."""
    y = np.asarray(y, dtype=np.float64).copy()
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    
    lo = -y.max() - 0.5
    hi = 1.0 - y.min() + 0.5
    
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    alpha = 0.5 * (lo + hi)
    h = np.clip(y + alpha, 0.0, 1.0)
    
    # Fix tiny residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-15)[0]
            if len(idx) > 0:
                h[idx] += residual / len(idx)
                h = np.clip(h, 0.0, 1.0)
        else:
            space = h.copy()
            idx = np.where(space > 1e-15)[0]
            if len(idx) > 0:

                h = np.clip(h, 0.0, 1.0)
    
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def all_lag_gradients(h, dx):
    """Compute gradients of all lag functions. Returns array of shape (2n-1, n)."""
    n = len(h)
    grads = np.zeros((2 * n - 1, n), dtype=np.float64)
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        if lag >= 0:
            L = n - lag
            if L > 0:
                grads[m, lag:] += (1.0 - h[:L])
                grads[m, :L] += -h[lag:]
        else:
            r = -lag
            L = n - r
            if L > 0:
                grads[m, r:] += (1.0 - h[:L])
                grads[m, :L] += -h[r:]
    grads *= dx
    return grads

def upsample_h(h, n_new):
    n = len(h)

    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new, n_new / 2.0)

        fmax = lags.max()

        shifted = np.clip(shifted, -30, 30)
