def smooth_objective(h, dx, tau):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    c_shifted = c - c_max  # for numerical stability
    exp_c = np.exp(tau * c_shifted)
    f = (c_max + np.log(exp_c.sum()) / tau)
    return f

def smooth_gradient(h, dx, tau):

    c_max = c.max()

    grad *= dx

Wait, I have a bug in the subgradient. Let me re-derive.

c[k] = dx * sum_i h[i] * (1 - h[i + shift_k]) where shift_k = k - (n-1)

The gradient of c[k] w.r.t. h[j]:
- h[j] appears in the sum when i = j (as the first factor): contribution to c[k] is h[j] * (1 - h[j+shift_k])
  Derivative w.r.t. h[j]: (1 - h[j+shift_k]) [if j+shift_k is in range]
- h[j] appears in the sum when i+shift_k = j, i.e., i = j-shift_k (as the second factor): contribution is h[j-shift_k] * (1 - h[j])
  Derivative w.r.t. h[j]: -h[j-shift_k] [if j-shift_k is in range]

So: ∂c[k]/∂h[j] = dx * [(1 - h[j+shift_k]) * 1{j+shift_k ∈ [0,n)} - h[j-shift_k] * 1{j-shift_k ∈ [0,n)}]

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()

def project_box_hyperplane(h, target_sum):
    n = len(h)
    if n == 0:
        return h
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    Z = exp_c.sum()
    f = c_max + np.log(Z) / tau
    weights = exp_c / Z
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(c_raw)):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (n - 1)
        m1 = (idx + shift >= 0) & (idx + shift < n)
        if m1.any():
            grad[m1] += w * (1.0 - h[idx[m1] + shift])
        m2 = (idx - shift >= 0) & (idx - shift < n)
        if m2.any():
            grad[m2] -= w * h[idx[m2] - shift]
    grad *= dx
    return f, grad

def slaqp_optimize(h0, n, target_sum, tau, max_iter=500):
    cons = [{'type': 'eq', 'fun': lambda z: z.sum() - target_sum,
             'jac': lambda z: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    def obj(z):
        return smooth_obj_and_grad(z, tau)[0]
    def jac(z):
        return smooth_obj_and_grad(z, tau)[1]
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-10})
    return project_box_hyperplane(res.x, target_sum)

def subgradient_polish(h0, n, target_sum, max_iters=200000, time_limit=None, t0=None):
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.4
    alpha = 0.05
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    no_improve = 0
    for it in range(max_iters):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = np.argmax(c)
        shift = k_star - (n - 1)
        idx = np.arange(n)
        grad = np.zeros(n)
        m1 = (idx + shift >= 0) & (idx + shift < n)
        grad[m1] += (1.0 - h[idx[m1] + shift])
        m2 = (idx - shift >= 0) & (idx - shift < n)
        grad[m2] -= h[idx[m2] - shift]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        v = mu * v - alpha * grad / gnorm
        h_new = project_box_hyperplane(h + v, target_sum)
        c5_new = evaluate_c5(h_new)
        if c5_new <= best_c5:
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                alpha *= 0.7
                v *= 0.3
                no_improve = 0
                if alpha < 0.002:
                    break
        if it % 5000 == 0:
            alpha = max(alpha, 0.02)
    return best_h, best_c5

def local_search(h0, n, target_sum, rng, n_iters=20000, time_limit=None, t0=None):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        # Randomly perturb 1-3 coordinates
        n_pert = rng.integers(1, 4)
        indices = rng.choice(n, size=n_pert, replace=False)
        h_trial = h.copy()
        for i in indices:
            h_trial[i] = rng.uniform(0, 1)
        h_trial = project_box_hyperplane(h_trial, target_sum)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
    return best_h, best_c5

def fista_smoothed(h0, n, target_sum, tau, max_iters=50000, time_limit=None, t0=None):
    dx = 2.0 / n
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    L = 10.0  # Lipschitz constant estimate
    step = 1.0 / L
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        # Gradient step
        _, grad = smooth_obj_and_grad(y, tau)
        z = y - step * grad
        # Project
        h_new = project_box_hyperplane(z, target_sum)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        h = h_new
        t_k = t_new
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Adapt step size
        if it % 500 == 0 and L > 0.1:
            L *= 0.9
            step = 1.0 / L
    return best_h, best_c5

    rng = np.random.default_rng(seed)
    
    def elapsed():
        return time.time() - start_time

    # Get initial point
    if 'initial_h_values' in globals():

    n = len(h0)

        h_cand = slaqp_optimize(best_h, n, target_sum, tau, max_iter=400)
        c5 = evaluate_c5(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()
    
    # FISTA polished with moderate smoothing
    if time_left() > 60:
        h_f, c5_f = fista_smoothed(best_h, n, target_sum, tau=80,
                                    max_iters=30000, time_limit=min(300, time_left()-30), t0=start_time)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Subgradient polish
    if time_left() > 30:
        h_sg, c5_sg = subgradient_polish(best_h, n, target_sum,
                                          max_iters=100000, time_limit=min(300, time_left()-20), t0=start_time)
        if c5_sg < best_c5:

    # Local search
    if time_left() > 20:
