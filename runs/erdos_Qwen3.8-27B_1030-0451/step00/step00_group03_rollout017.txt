        lo, hi = hi, lo
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) / 2.0, 0, 1)
    r = target - np.sum(x)
    if abs(r) > 1e-12:
        if r > 0:
            for i in np.argsort(-(1.0 - x)):
                if r <= 1e-14: break
                a = min(r, 1.0 - x[i])
                x[i] += a; r -= a
        else:
            for i in np.argsort(-x):
                if -r <= 1e-14: break
                a = min(-r, x[i])
                x[i] -= a; r += a
    return x

def resample_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_interp = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_mean_box(h_interp, n_new)

def pair_swap_optimize(h, max_passes=50, time_limit=None, t_start=None):
    """Optimize h using pair swap (coordinate descent on pairs)."""
    h = h.copy()
    n = len(h)
    best_c = exact_c5(h)
    rng = np.random.default_rng(42)
    
    for p in range(max_passes):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        improved = False
        # Randomize pair order for diversity
        order = rng.permutation(n * (n-1) // 2)
        for flat_idx in order:
            # Convert flat index to (i, j)
            # flat_idx = i*(n-1) - i*(i+1)//2 + j - i - 1 ... this is complex
            # Simpler: just iterate all pairs
            pass
        # Actually let me just do all pairs in random order
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng.shuffle(pairs)
        
        for i, j in pairs:
            # Transfer from j to i: h[i] += t, h[j] -= t
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-10:
                continue
            # Golden section search
            # Actually, for speed, do a small grid
            ts = np.linspace(t_lo, t_hi, 7)
            best_t = 0.0
            best_local_c = exact_c5(h)
            for t in ts:
                h[i] += t
                h[j] -= t
                c = exact_c5(h)
                if c < best_local_c:
                    best_local_c = c
                    best_t = t
                h[i] -= t
                h[j] += t
            if best_t != 0.0 and best_local_c < best_c - 1e-12:
                h[i] += best_t
                h[j] -= best_t
                best_c = best_local_c
                improved = True
        
        if not improved:
            break
    
    return h, best_c

def slsqp_optimize(h, maxiter=200, t_start=None, time_limit=None):
    """Optimize using SLSQP with epigraph formulation."""
    n = len(h)
    dx = 2.0 / n
    h0 = h.copy()
    
    # Variables: [h_0, ..., h_{n-1}, T]
    x0 = np.append(h0, exact_c5(h0) + 0.01)
    
    # Objective: minimize T
    def objective(x):
        if time_limit and t_start and time.monotonic() - t_start > time_limit:
            raise ValueError("timeout")
        return x[-1]
    
    # Bounds
    bounds = [(0, 1)] * n + [(0, 2)]
    
    # Equality constraint: sum(h) = n/2
    def eq_constraint(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    # Inequality constraints: T - cor_k * dx >= 0 for each lag
    # cor_k = np.correlate(h, 1-h, mode='full')[k]
    # We need: T >= dx * cor_k for all k
    # i.e., T - dx * cor_k >= 0
    
    def ineq_constraints(x):
        h_x = x[:n]
        T = x[-1]
        cor = np.correlate(h_x, 1.0 - h_x, mode="full")
        return T - cor * dx
    
    def ineq_grad(x):
        h_x = x[:n]
        n_lags = 2 * n - 1
        grad = np.zeros((n_lags, n + 1))
        # For each lag m (0 to 2n-2), the constraint is T - dx * cor[m] >= 0
        # d/dT = 1
        # d/dh[p] of cor[m]:
        # cor[m] = sum over valid (i, j) of h[i] * (1-h[j]) where i-j = m-(n-1)
        # Let s = m - (n-1)
        # cor[m] = sum_{i: max(0,s) <= i <= min(n-1, n-1+s)} h[i] * (1-h[i-s])
        # Wait, I need to be more careful.
        # numpy correlate: cor[m] = sum_n a[n+k] * v[n] where k = m - (len(v)-1)
        # With a=h, v=1-h: cor[m] = sum_n h[n + k] * (1-h[n]) where k = m - (n-1)
        # valid n: 0 <= n < n AND 0 <= n+k < n
        # So: max(0, -k) <= n <= min(n-1, n-1-k)
        
        # d cor[m] / d h[p] = (1-h[p-k]) * [if p-k valid] + (-h[p+k]) * [if p+k valid]
        # Wait: if p = n+k (from the h[n+k] term), then n = p-k, valid if 0<=p-k<n
        #        if p = n (from the (1-h[n]) term), then the contribution is -h[n+k] = -h[p+k], valid if 0<=p+k<n
        
        for m in range(n_lags):
            s = m - (n - 1)  # the shift
            grad[m, -1] = 1.0  # dT/dT
            for p in range(n):
                dc = 0.0
                # Contribution from h[n+s] term: if p = n+s, then n = p-s
                if 0 <= p - s < n:
                    n_idx = p - s
                    dc += (1.0 - h_x[n_idx])
                # Contribution from (1-h[n]) term: if p = n, then n+k = p+s = h term
                if 0 <= p + s < n:
                    dc += (-h_x[p + s])
                grad[m, p] = -dx * dc
        
        return grad
    
    constraints = [
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_constraints, 'jac': ineq_grad}
    ]
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-10})
    
    if result.success or result.fun < np.inf:
        h_new = project_mean_box(result.x[:n], n)
        c_new = exact_c5(h_new)
        return h_new, c_new
    return h, exact_c5(h)
