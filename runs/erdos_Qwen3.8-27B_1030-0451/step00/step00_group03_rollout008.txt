def project_to_constraint(y, target):

def _smooth_descent(h0, budget, start):

    n = len(h0)

    target = n / 2.0
    h = h0.copy()

    best_c5 = _eval(h)

    n_active = min(2 * n - 1, max(40, n))
    active = np.argsort(-f)[:n_active]

    tau = max(1e-4, 0.008 * best_c5)
    eta = 0.02
    no_improve = 0
    it = 0

    while time.time() - start < budget:
        it += 1
        f = _corr(h) * dx
        f_act = f[active]
        f_max = f_act.max()
        z = np.clip((f_act - f_max) / tau, -60, 60)
        exp_z = np.exp(z)
        w = exp_z / exp_z.sum()
        smooth = f_max + tau * np.log(exp_z.sum())

        grad = np.zeros(n)
        for i in range(len(active)):
            if w[i] > 1e-8:

            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = len(h2)

        # Phase 3: Local perturbation
        remaining = deadline - time.time()
        if remaining > 5 and len(best_h) == n_res:
            lp_budget = min(remaining * 0.15, 15.0)
            h3, c5_3 = _local_perturb(best_h, lp_budget, time.time(), rng)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = len(h3)

        # Early exit if target met
        if best_c5 <= 0.38075:
            break

    # Final polish on best
    remaining = deadline - time.time()
    if remaining > 10:
        h_f, c5_f = _slsqp_polish(best_h, min(remaining * 0.6, 40.0), time.time())
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = len(best_h)

    # Final validation
    best_h = _project(best_h, len(best_h) / 2.0)
    final_c5 = _eval(best_h)

    return (best_h, final_c5, len(best_h))
