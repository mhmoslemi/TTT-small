        return 0.0, current_c5
    
    best_delta = 0.0
    best_val = current_c5
    grid = np.linspace(d_lo, d_hi, n_pts)
    
    for d in grid:
        if abs(d) < 1e-12:
            continue
        h[i] = hi + d
        h[j] = hj - d
        val = exact_c5(h)
        if val < best_val:
            best_val = val
            best_delta = d
    
    h[i] = hi
    h[j] = hj
    
    if best_delta == 0.0:
        return 0.0, current_c5
    
    # Refine around best
    span = (d_hi - d_lo) / (n_pts - 1)
    lo = max(d_lo, best_delta - span)
    hi_r = min(d_hi, best_delta + span)
    fine_grid = np.linspace(lo, hi_r, 8)
    
    for d in fine_grid:
        if abs(d) < 1e-12:
            continue
        h[i] = hi + d
        h[j] = hj - d
        val = exact_c5(h)
        if val < best_val:
            best_val = val
            best_delta = d
    
    h[i] = hi
    h[j] = hj
    return best_delta, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 8.0  # safety margin
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64).flatten()
    else:
        h = np.full(250, 0.5, dtype=np.float64)
    
    n = len(h)
    h = project_feasible(h, n)
    
    best_c5 = exact_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t_start
    
    def time_ok():
        return elapsed() < deadline
    
    # Phase 1: Pair-coordinate descent with shuffled sweeps
    sweep = 0
    no_improve_count = 0
    
    while time_ok():
        sweep += 1
        # Get active (interior) coordinates
        interior = np.where((h > 0.003) & (h < 0.997))[0]
        active_n = len(interior)
        if active_n < 2:
            break
        
        # Shuffle
        perm = rng.permutation(active_n)
        
        sweep_best = best_c5
        sweep_improvements = 0
        pairs_tried = 0
        max_pairs = min(active_n * 12, 40000)
        
        for idx in range(len(perm)):
            if not time_ok():
                break
            if pairs_tried >= max_pairs:
                break
            
            i = interior[perm[idx]]
            # Try several j values for this i
            n_j_tried = 0
            for jidx in range(min(15, len(perm) - idx - 1)):
                j = interior[perm[idx + 1 + jidx]]
                pairs_tried += 1
                
                if abs(h[i] - h[j]) < 1e-5:
                    continue
                
                delta, new_c5 = pair_grid_search(h, i, j, sweep_best, n_pts=10)
                
                if delta != 0.0 and new_c5 < sweep_best - 1e-11:
                    h[i] += delta
                    h[j] -= delta
                    h[i] = np.clip(h[i], 0.0, 1.0)
                    h[j] = np.clip(h[j], 0.0, 1.0)
                    sweep_best = new_c5
                    sweep_improvements += 1
                    # Update best
                    if sweep_best < best_c5 - 1e-12:
                        best_c5 = sweep_best
                        best_h = h.copy()
                    break  # re-evaluate i with new state next iteration
            
            if not time_ok():
                break
        
        if sweep_improvements == 0:
            no_improve_count += 1
            # Escape: random perturbation
            if no_improve_count >= 2:
                n_pert = min(5, n)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h_pert = h.copy()
                h_pert[pert_idx] += rng.uniform(-0.03, 0.03, size=n_pert)
                h_pert = project_feasible(h_pert, n)
                c5_pert = exact_c5(h_pert)
                if c5_pert < sweep_best:
                    h = h_pert
                    sweep_best = c5_pert
                    if c5_pert < best_c5:
                        best_c5 = c5_pert
                        best_h = h.copy()
                no_improve_count = 0
        else:
            no_improve_count = 0
        
        # Early exit
        if best_c5 <= 0.38079:
            break
    
    # Phase 2: Fine-tuning with more grid points on best solution
    if time_ok():
        h = best_h.copy()
        c5_cur = exact_c5(h)
        for _ in range(3):
            if not time_ok():
                break
            interior = np.where((h > 0.003) & (h < 0.997))[0]
            active_n = len(interior)
            if active_n < 2:
                break
            perm = rng.permutation(active_n)
            for idx in range(len(perm)):
                if not time_ok():
                    break
                i = interior[perm[idx]]
                for jidx in range(min(8, len(perm) - idx - 1)):
                    j = interior[perm[idx + 1 + jidx]]
                    if abs(h[i] - h[j]) < 1e-6:
                        continue
                    delta, new_c5 = pair_grid_search(h, i, j, c5_cur, n_pts=20)
                    if delta != 0.0 and new_c5 < c5_cur - 1e-13:
                        h[i] += delta
                        h[j] -= delta
                        h[i] = np.clip(h[i], 0.0, 1.0)
                        h[j] = np.clip(h[j], 0.0, 1.0)
                        c5_cur = new_c5
                        if c5_cur < best_c5 - 1e-12:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
    
    # Phase 3: Triple-coordinate moves for final escape
    if time_ok():
        h = best_h.copy()
        interior = np.where((h > 0.05) & (h < 0.95))[0]
        for _ in range(500):
            if not time_ok():
                break
            if len(interior) < 3:
                break
            triple = rng.choice(interior, size=3, replace=False)
            h_try = h.copy()
            # Move mass: increase one, decrease others
            src = triple[rng.randint(3)]
            dsts = [t for t in triple if t != src]
            amount = rng.uniform(0.01, 0.05)
            h_try[src] -= amount
            for d in dsts:
                h_try[d] += amount / len(dsts)
            h_try = project_feasible(h_try, len(h))
            c5_try = exact_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
    
    # Final validation and return
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0.0, 1.0)
    # Ensure sum is exact
    residual = n / 2.0 - np.sum(final_h)
    if abs(residual) > 1e-12:
        for k in range(n):
            if 0.0 < final_h[k] < 1.0:
                final_h[k] += residual
                final_h[k] = np.clip(final_h[k], 0.0, 1.0)
                break
    
    final_c5 = exact_c5(final_h)
    
    return (final_h, final_c5, n)
