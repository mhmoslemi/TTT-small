import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        # Add diff to elements with most room
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in order:
            if remaining < 1e-14: break
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for i in order:
            if remaining < 1e-14: break
            sub = min(remaining, h[i])
            h[i] -= sub
            remaining -= sub
    return h

def pair_move(h, j, k, c_vals, n):
    """Find optimal δ for h[j]+=δ, h[k]-=δ"""
    # Compute B_m = A_{m,j} - A_{m,k}
    # A_{m,j} = coefficient of h[j] in c_m
    # For lag m (shift s = m - (n-1)):
    #   A_{m,j} = (1-h[j+s]) if 0≤j+s<n (j is first factor)
    #            + (-h[j-s]) if 0≤j-s<n (j is second factor)
    #            + (1-2*h[j]) if s==0 (the h[j]*(1-h[j]) term)
    
    # Actually, let me just compute the change directly
    # c_m(new) = c_m(old) + δ * (contrib_j - contrib_k)
    # where contrib_j is the "coefficient" of h[j] in c_m
    
    # For efficiency, compute A for j and k
    def get_coeff(h, pos):
        A = np.zeros(2*n - 1)
        for s in range(-(n-1), n):
            idx = n - 1 + s
            if s == 0:
                A[idx] += (1.0 - 2.0 * h[pos])
            else:
                if 0 <= pos + s < n:
                    A[idx] += (1.0 - h[pos + s])
                if 0 <= pos - s < n:
                    A[idx] += (-h[pos - s])
        return A
    
    Aj = get_coeff(h, j)
    Ak = get_coeff(h, k)
    B = Aj - Ak
    
    # Minimize max_m (c_vals[m] + B[m] * δ)
    # δ range: [max(-h[j], h[k]-1), min(1-h[j], h[k])]
    d_min = max(-h[j], h[k] - 1.0)
    d_max = min(1.0 - h[j], h[k])
    
    if d_min >= d_max:
        return 0.0
    
    # Use grid search (fast enough)
    best_d = 0.0
    best_val = np.max(c_vals)
    
    # Coarse grid
    grid = np.linspace(d_min, d_max, 50)
    for d in grid:
        val = np.max(c_vals + B * d)
        if val < best_val:
            best_val = val
            best_d = d
    
    # Fine grid around best
    if d_min < best_d < d_max:
        margin = (d_max - d_min) / 50.0
        grid2 = np.linspace(max(d_min, best_d - margin), min(d_max, best_d + margin), 50)
        for d in grid2:
            val = np.max(c_vals + B * d)
            if val < best_val:
                best_val = val
                best_d = d
    
    return best_d

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        remaining = diff
        for i in order:
            if remaining < 1e-14:
                break
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
    else:
        order = np.argsort(-h)
        remaining = -diff
        for i in order:
            if remaining < 1e-14:
                break
            sub = min(remaining, h[i])
            h[i] -= sub
            remaining -= sub
    return h

def get_coeff(h, pos, n):

    for s in range(-(n - 1), n):

        else:
            if 0 <= pos + s < n:
                A[idx] += (1.0 - h[pos + s])
            if 0 <= pos - s < n:
                A[idx] += (-h[pos - s])
    return A

        if feasible(mid):
            hi = mid
        else:
            lo = mid
    
    tau = hi
    
    # Find δ range
    lower = d_min
    upper = d_max
    for i in range(len(c_vals)):
        b = B[i]
        a = c_vals[i]
        if b > 1e-15:
            upper = min(upper, (tau - a) / b)
        elif b < -1e-15:
            lower = max(lower, (tau - a) / b)
    
    lower = max(lower, d_min)
    upper = min(upper, d_max)
    
    if lower >= upper:
        # Pick one that minimizes
        candidates = [lower, max(d_min, min(d_max, lower))]
        best_d = d_min
        best_val = np.inf
        for d in candidates:
            d = max(d_min, min(d_max, d))
            val = np.max(c_vals + B * d)
            if val < best_val:
                best_val = val
                best_d = d
        return best_d
    
    return (lower + upper) / 2.0
