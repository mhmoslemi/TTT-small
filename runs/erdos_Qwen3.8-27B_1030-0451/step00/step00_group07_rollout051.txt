def _adjust_sum(x, target):
    x = np.asarray(x, dtype=float)
    r = float(target - x.sum())
    if abs(r) < 1e-13:
        return x
    if abs(r) < 1e-7:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                i = int(idx[np.argmin(x[idx])])
                inc = min(float(1.0 - x[i]), r)
                x[i] += inc
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                i = int(idx[np.argmax(x[idx])])
                dec = min(float(x[i]), -r)
                x[i] -= dec
        x = np.clip(x, 0.0, 1.0)
        if abs(float(x.sum()) - target) < 1e-10:
            return x
    if r > 0:
        idx = np.where(x < 1.0)[0]
        if idx.size:
            order = idx[np.argsort(x[idx])]
            rem = r
            for i in order:
                inc = min(float(1.0 - x[i]), rem)
                x[i] += inc; rem -= inc
                if rem <= 1e-14: break
    else:
        rpos = -r
        idx = np.where(x > 0.0)[0]
        if idx.size:
            order = idx[np.argsort(x[idx])[::-1]]
            rem = rpos
            for i in order:
                dec = min(float(x[i]), rem)
                x[i] -= dec; rem -= dec
                if rem <= 1e-14: break
    x = np.clip(x, 0.0, 1.0)
    return x

def _adjust_sum(x, target):
    x = np.asarray(x, dtype=float)
    r = float(target - x.sum())
    if abs(r) < 1e-13:
        return x

    if abs(r) < 1e-7:
        if r > 0.0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                i = int(idx[np.argmin(x[idx])])
                inc = min(float(1.0 - x[i]), r)
                x[i] += inc
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                i = int(idx[np.argmax(x[idx])])
                dec = min(float(x[i]), -r)
                x[i] -= dec
        x = np.clip(x, 0.0, 1.0)
        if abs(float(x.sum()) - target) < 1e-10:
            return x

    if r > 0.0:
        idx = np.where(x < 1.0)[0]
        if idx.size:
            order = idx[np.argsort(x[idx])]
            rem = r
            for i in order:
                inc = min(float(1.0 - x[i]), rem)
                x[i] += inc
                rem -= inc
                if rem <= 1e-14:
                    break
    else:
        rpos = -r
        idx = np.where(x > 0.0)[0]
        if idx.size:
            order = idx[np.argsort(x[idx])[::-1]]
            rem = rpos
            for i in order:
                dec = min(float(x[i]), rem)
                x[i] -= dec
                rem -= dec
                if rem <= 1e-14:
                    break

    x = np.clip(x, 0.0, 1.0)
    return x


def make_feasible(y, target):
    y = np.asarray(y, dtype=float).ravel()
    n = y.size
    if n == 0:
        return y
    target = float(target)
    if not np.all(np.isfinite(y)):
        y = np.full(n, 0.5, dtype=float)
    if target < 0.0:
        target = 0.0
    if target > float(n):
        target = float(n)

    lo = -1.0 - float(np.max(y))
    hi = 1.0 - float(np.min(y))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid

    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    x = _adjust_sum(x, target)
    return x


def fix_sum_small(x, target):
    x = np.asarray(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    x = _adjust_sum(x, target)
    if abs(float(x.sum()) - target) > 1e-8:
        x = make_feasible(x, target)
    return x


def corr_for_d(h, d):
    n = len(h)
    if abs(d) >= n:
        return 0.0
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    if d > 0:
        return float(np.sum(h[d:] * (1.0 - h[:-d])))
    dd = -d
    return float(np.sum(h[:-dd] * (1.0 - h[dd:])))


def _calibrate_corr_direction():
    n = 9
    y = np.sin(np.arange(n, dtype=float) * 1.7) + 0.5
    h = make_feasible(y, 0.5 * n)
    cact = np.correlate(h, 1.0 - h, "full")
    best_dir = 1
    best_err = np.inf
    ctr = n - 1
    for direc in (1, -1):
        raw = np.empty(2 * n - 1, dtype=float)
        for k in range(2 * n - 1):
            d = direc * (k - ctr)
            raw[k] = corr_for_d(h, d)
        err = float(np.max(np.abs(raw - cact)))
        if err < best_err:
            best_err = err
            best_dir = direc
    return best_dir


def _get_corr_dir():
    global _CORR_DIR, _CORR_DIR_SET
    if not _CORR_DIR_SET:
        try:
            _CORR_DIR = _calibrate_corr_direction()
        except Exception:
            _CORR_DIR = 1
        _CORR_DIR_SET = True
    return _CORR_DIR


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, "full") * dx


def exact_C5(h):
    if len(h) == 0:
        return 0.0
    c = corr_scaled(h)
    if c.size == 0:
        return 0.0
    v = float(np.max(c))
    if not np.isfinite(v):
        return float("inf")
    return v


def select_lags(C, cap=None, slack=0.005):
    m = C.size
    if m == 0:
        return np.empty(0, dtype=int)
    maxc = float(np.max(C))
    if cap is None:
        cap = min(800, 2 * m)
    near = np.where(C >= maxc - max(float(slack), 1e-6))[0]
    if near.size > cap:
        order = np.argsort(C, kind="stable")
        return order[::-1][:cap].astype(int)
    want = max(16, m // 8)
    if near.size < want:
        order = np.argsort(C, kind="stable")
        top = order[::-1][:want]
        combined = np.unique(np.concatenate((near, top)))
        if combined.size > cap:
            return order[::-1][:cap].astype(int)
        return combined.astype(int)
    return near.astype(int)


def lp_sqp_step(h, C, sel):
    sel = np.asarray(sel, dtype=int)
    if sel.size == 0:
        return None, None
    n = len(h)
    dx = 2.0 / float(n)
    target = 0.5 * float(n)
    msel = sel.size
    A = np.zeros((msel, n + 1), dtype=float)
    b = np.empty(msel, dtype=float)
    dirf = _get_corr_dir()
    ctr = n - 1

    for ii, k in enumerate(sel):
        row = A[ii, :n]
        row.fill(0.0)
        d = dirf * (int(k) - ctr)
        if d == 0:
            row[:] = dx * (1.0 - 2.0 * h)
        elif d > 0:
            row[d:] = dx * (1.0 - h[:-d])
            row[:-d] += -dx * h[d:]
        else:
            dd = -d
            row[:n - dd] = dx * (1.0 - h[dd:])
            row[dd:] += -dx * h[:-dd]
        A[ii, n] = -1.0
        b[ii] = float(row.dot(h)) - float(C[k])

    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([target], dtype=float)
    bounds = [(0.0, 1.0)] * (n + 1)

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None, None

    if res is None or not getattr(res, "success", False) or res.x is None:
        return None, None
    x = np.asarray(res.x, dtype=float).ravel()
    t = float(x[n])
    x = fix_sum_small(x[:n], target)
    if not np.all(np.isfinite(x)) or not np.isfinite(t):
        return None, None
    return x, t


def _line_eval(x0, s, lam, target):
    if lam <= 0.0 or lam > 1.0:
        return None, float("inf")
    x = x0 + lam * s
    if not np.all(np.isfinite(x)):
        return None, float("inf")
    if np.any(x < -1e-8) or np.any(x > 1.0 + 1e-8):
        return None, float("inf")
    x = np.clip(x, 0.0, 1.0)
    x = fix_sum_small(x, target)
    c = exact_C5(x)
    if not np.isfinite(c):
        return None, float("inf")
    return x, c


def line_search_exact(x0, x1, c0, target):
    s = x1 - x0
    if np.linalg.norm(s) < 1e-14:
        return None, None
    best_lam = 0.0
    best_c = c0
    lams = (1.0, 0.75, 0.5, 0.333333333, 0.2, 0.1, 0.05, 0.02, 0.01, 0.003, 0.001, 0.0003)
    for lam in lams:
        x, c = _line_eval(x0, s, lam, target)
        if x is not None and c < best_c - 1e-13:
            best_c = c
            best_lam = lam

    if best_lam > 0.0:
        base = best_lam
        for fac in (0.5, 2.0, 0.7, 1.4, 0.4, 1.8):
            lam = base * fac
            if 0.0 < lam <= 1.0:
                x, c = _line_eval(x0, s, lam, target)
                if x is not None and c < best_c - 1e-13:
                    best_c = c
                    best_lam = lam

    if best_lam <= 0.0:
        return None, None
    return best_lam, best_c


def binarize_h(h, target):
    n = len(h)
    if abs(float(target) - float(round(target))) > 1e-8:
        return None
    cnt = int(round(target))
    if cnt <= 0 or cnt >= n:
        return None
    order = np.argsort(h, kind="stable")[::-1][:cnt]
    b = np.zeros(n, dtype=float)
    b[order] = 1.0
    return b


def _slsqp_eq_fun(x):
    return float(np.sum(x) - _SL_TARGET)


def _slsqp_eq_jac(x):
    return np.ones_like(x, dtype=float)


def _slsqp_obj(x, tau, dx):
    C = np.correlate(x, 1.0 - x, "full") * dx
    m = float(np.max(C))
    if tau <= 1e-15:
        return m
    a = (C - m) / tau
    s = float(np.sum(np.exp(a)))
    if s <= 1e-300:
        return m
    return m + tau * float(np.log(s))


def _slsqp_jac(x, tau, dx):
    n = len(x)
    C = np.correlate(x, 1.0 - x, "full") * dx
    m = float(np.max(C))

    if tau <= 1e-14:
        act = C >= m - 1e-12
        alpha = np.zeros_like(C, dtype=float)
        cnt = float(np.count_nonzero(act))
        if cnt <= 0.0:
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha[act] = 1.0 / cnt
    else:
        a = (C - m) / tau
        e = np.exp(a)
        s = float(np.sum(e))
        if s <= 1e-300:
            alpha = np.zeros_like(C, dtype=float)
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha = e / s

    G = np.zeros(n, dtype=float)
    dirf = _get_corr_dir()
    ctr = n - 1
    for k in range(2 * n - 1):
        ak = float(alpha[k])
        if ak <= 1e-15:
            continue
        coeff = ak * dx
        d = dirf * (int(k) - ctr)
        if d == 0:
            G += coeff * (1.0 - 2.0 * x)
        elif d > 0:
            G[d:] += coeff * (1.0 - x[:-d])
            G[:-d] += -coeff * x[d:]
        else:
            dd = -d
            G[:n - dd] += coeff * (1.0 - x[dd:])
            G[dd:] += -coeff * x[:-dd]
    return G


def slsqp_optimize(h, tau, deadline):
    global _SL_TARGET
    n = len(h)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    rem = deadline - time.time()
    if rem < 20.0:
        return None, None

    _SL_TARGET = target
    maxiter = int(max(20, min(80, 8000.0 / float(max(n, 1)), 6000.0 / max(rem, 1.0))))

    try:
        res = minimize(
            _slsqp_obj,
            h,
            args=(tau, dx),
            jac=_slsqp_jac,
            method="SLSQP",
            bounds=[(0.0, 1.0)] * n,
            constraints=[{"type": "eq", "fun": _slsqp_eq_fun, "jac": _slsqp_eq_jac}],
            options={"maxiter": maxiter, "ftol": 1e-12},
        )
    except Exception:
        return None, None

    if res is None:
        return None, None
    x = np.asarray(res.x, dtype=float).ravel()
    if x.size != n or not np.all(np.isfinite(x)):
        return None, None
    x = make_feasible(x, target)
    c = exact_C5(x)
    if not np.isfinite(c):
        return None, None
    return x, c


def resample_h(h, N):
    n0 = len(h)
    target = 0.5 * float(N)
    if n0 == N:
        return make_feasible(h, target)

    cands = []
    xold = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))
    y = np.interp(xnew, xold, h)
    cands.append(make_feasible(y, target))

    if N % n0 == 0:
        cands.append(make_feasible(np.tile(h, N // n0), target))

    if n0 % N == 0:
        cands.append(make_feasible(h.reshape(N, n0 // N).mean(axis=1), target))

    if N > n0:
        idx = np.floor(np.arange(N) * (n0 / float(N))).astype(int)
        idx = np.clip(idx, 0, n0 - 1)
        cands.append(make_feasible(h[idx], target))

    if len(cands) == 1:
        return cands[0]
    return min(cands, key=exact_C5)


def _fourier_candidate(N, rng):
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    f = int(1 + rng.integers(0, 3))
    ph = float(rng.uniform(0.0, 2.0 * np.pi))
    y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph) + 0.08 * rng.normal(0.0, 1.0, size=N)
    return make_feasible(y, 0.5 * float(N))


def augment_candidate(h_base, N, rng):
    cands = [np.asarray(h_base, dtype=float).copy()]
    for sig in (0.05, 0.02, 0.01):
        cands.append(make_feasible(h_base + rng.normal(0.0, sig, size=N), 0.5 * float(N)))
    if N % 2 == 0:
        hb = binarize_h(h_base, 0.5 * float(N))
        if hb is not None:
            cands.append(hb)
    if N <= 720:
        for _ in range(2):
            cands.append(_fourier_candidate(N, rng))
    return min(cands, key=exact_C5)


def _get_initial():
    val = globals().get("initial_h_values", None)
    if val is None:
        return None
    arr = np.asarray(val, dtype=float).ravel()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    return make_feasible(arr, 0.5 * float(arr.size))


def fallback_initial(N=240):
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    best = None
    bestc = float("inf")
    for f in (1, 2, 3, 4):
        for ph in (0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi):
            y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph)
            h = make_feasible(y, 0.5 * float(N))
            c = exact_C5(h)
            if np.isfinite(c) and c < bestc:
                bestc = c
                best = h
    rng = np.random.default_rng(123)
    for _ in range(8):
        f = int(1 + rng.integers(0, 4))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph) + 0.1 * rng.normal(0.0, 1.0, size=N)
        h = make_feasible(y, 0.5 * float(N))
        c = exact_C5(h)
        if np.isfinite(c) and c < bestc:
            bestc = c
            best = h
    return best


def optimize_candidate(h, deadline, rng):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(h)
    target = 0.5 * float(n)
    h = make_feasible(h, target).copy()
    c = exact_C5(h)
    if not np.isfinite(c):
        return h, c

    if n % 2 == 0:
        hb = binarize_h(h, target)
        if hb is not None:
            cb = exact_C5(hb)
            if np.isfinite(cb) and cb < c - 1e-12:
                h = hb
                c = cb

    max_lp = int(max(20, min(120, 12000.0 / float(max(n, 1)))))
    lp_count = 0
    stall = 0
    slsqp_used = 0
    no_lp_improve = 0
    sigmas = (0.03, 0.02, 0.01, 0.005, 0.002, 0.001)

    while time.time() < deadline:
        improved = False

        if lp_count < max_lp and time.time() < deadline - 5.0:
            C = corr_scaled(h)
            if no_lp_improve >= 3 and no_lp_improve % 4 == 0:
                sel = np.arange(C.size, dtype=int)
            else:
                sel = select_lags(C)

            x_lp, _ = lp_sqp_step(h, C, sel)
            lp_count += 1

            if x_lp is not None:
                lam, c_new = line_search_exact(h, x_lp, c, target)
                if lam is not None and c_new < c - 1e-13:
                    old = c
                    h = h + lam * (x_lp - h)
                    h = np.clip(h, 0.0, 1.0)
                    h = fix_sum_small(h, target)
                    c = c_new
                    improved = True
                    if c < old - 1e-9:
                        stall = 0
                        no_lp_improve = 0
                    else:
                        stall += 1
                        no_lp_improve += 1
                    continue

        if not improved:
            stall += 1
            no_lp_improve += 1

        if (
            slsqp_used < 4
            and lp_count >= 4
            and stall >= 2
            and time.time() < deadline - 25.0
        ):
            taus = (0.03, 0.01, 0.003, 0.001, 0.0003)
            tau = taus[min(slsqp_used, len(taus) - 1)]
            x_new, c_new = slsqp_optimize(h, tau, deadline)
            slsqp_used += 1
            if x_new is not None and c_new < c - 1e-13:
                h = x_new
                c = c_new
                stall = 0
                no_lp_improve = 0
                continue
            stall += 2

        if stall >= 5:
            grad_improved = False
            dx = 2.0 / float(n)
            for tau in (0.02, 0.005, 0.001):
                if time.time() > deadline - 8.0:
                    break
                try:
                    g = _slsqp_jac(h, tau, dx)
                except Exception:
                    g = None
                if g is None or not np.all(np.isfinite(g)):
                    continue
                gp = g - float(np.mean(g))
                for alpha in (0.1, 0.05, 0.02, 0.01, 0.005, 0.002):
                    if time.time() > deadline - 5.0:
                        break
                    h2 = make_feasible(h - alpha * gp, target)
                    c2 = exact_C5(h2)
                    if np.isfinite(c2) and c2 < c - 1e-12:
                        h = h2
                        c = c2
                        stall = 0
                        no_lp_improve = 0
                        grad_improved = True
                        break
                if grad_improved:
                    break

            if grad_improved:
                continue

            pert_improved = False
            for sig in sigmas:
                if time.time() > deadline - 5.0:
                    break
                h2 = make_feasible(h + rng.normal(0.0, sig, size=n), target)
                c2 = exact_C5(h2)
                if np.isfinite(c2) and c2 < c - 1e-12:
                    h = h2
                    c = c2
                    stall = 0
                    no_lp_improve = 0
                    pert_improved = True
                    break

            if pert_improved:
                continue

            if slsqp_used >= 4 and (lp_count >= max_lp or time.time() > deadline - 20.0):
                break
            stall = max(0, stall - 4)

    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0.0:
        total = 1100.0
    hard = t0 + max(5.0, total) - 5.0

    rng = np.random.default_rng(int(seed) % (1 << 32))

    hinit = _get_initial()
    if hinit is None:
        hinit = fallback_initial(240)

    n0 = int(hinit.size)
    best_h = make_feasible(hinit, 0.5 * float(n0))
    best_c = exact_C5(best_h)
    best_n = n0

    Ns = []
    for f in (1, 2, 3, 4, 6):
        N = int(round(n0 * f))
        if 10 <= N <= 1500 and N not in Ns:
            Ns.append(N)

    if n0 < 300:
        n15 = int(round(1.5 * n0))
        if 10 <= n15 <= 1500 and n15 not in Ns:
            Ns.append(n15)

    Ns = sorted(set(Ns))
    max_targets = 4 if total < 600.0 else 6
    if len(Ns) > max_targets:
        prioritized = []
        for f in (1, 2, 3, 4, 6):
            N = int(round(n0 * f))
            if N in Ns and N not in prioritized:
                prioritized.append(N)
        for N in Ns:
            if N not in prioritized and len(prioritized) < max_targets:
                prioritized.append(N)
        Ns = prioritized

    for idx, Nt in enumerate(Ns):
        if time.time() > hard - 30.0:
            break
        if best_c <= 0.3808 - 1e-8:
            break

        remaining = hard - time.time()
        if remaining < 20.0:
            break

        if idx == 0:
            stage = min(remaining, max(15.0, 0.35 * total))
        else:
            stage = min(remaining * 0.45, max(12.0, 0.20 * total))

        stage_deadline = min(hard - 10.0, time.time() + max(5.0, stage))

        if Nt == best_n:
            h_cand = best_h.copy()
        else:
            h_cand = resample_h(best_h, Nt)

        h_cand = augment_candidate(h_cand, Nt, rng)
        h_opt, c_opt = optimize_candidate(h_cand, stage_deadline, rng)

        if np.isfinite(c_opt) and c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            best_n = int(len(h_opt))

    if time.time() < hard - 5.0:
        h_opt, c_opt = optimize_candidate(best_h, hard - 3.0, rng)
        if np.isfinite(c_opt) and c_opt < best_c:
            best_h = h_opt
            best_c = c_opt
            best_n = int(len(h_opt))

    best_n = int(best_h.size)
    best_h = make_feasible(best_h, 0.5 * float(best_n))
    best_c = exact_C5(best_h)

    if not np.isfinite(best_c):
        best_h = make_feasible(np.full(best_n, 0.5, dtype=float), 0.5 * float(best_n))
        best_c = exact_C5(best_h)

    return best_h, float(best_c), best_n
