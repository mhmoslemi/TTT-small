def is_feasible(h, n):

def project_feasible(h, n):

            h[mask] += add * space / total_space

    grad = np.zeros(n)

        valid1 = np.arange(n - lag)

        valid2 = np.arange(lag, n)
        grad[valid2] -= dx * h[valid2 - lag]

def lp_direction(h, n, c5_cur, top_k=5):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    values = corr * dx
    top_idx = np.argsort(values)[::-1][:top_k]

    rows = []
    for idx in top_idx:
        lag = idx - (n - 1)
        g = gradient_of_lag(h, lag, n)
        # Normalize
        gnorm = np.abs(g).sum()
        if gnorm > 1e-14:
            rows.append(g / gnorm)
    
    if len(rows) == 0:
        return None
    
    A = np.array(rows)  # shape (k, n)
    # We want to find d such that A @ d <= -delta for all rows, sum(d) = 0, ||d||_1 <= 1
    # Variables: d (n), s (n), delta (1) => total 2n+1
    # Constraints:
    #   A @ d <= -delta  =>  A @ d + delta <= 0  (k constraints)
    #   d_i + s_i <= 0? No, we need |d_i| <= s_i, sum(s_i) <= 1
    #   Actually: d_i - s_i <= 0, -d_i - s_i <= 0, sum(s_i) <= 1, s_i >= 0
    #   sum(d) = 0 (equality)
    #   delta >= 0
    # Maximize delta (minimize -delta)
    
    nvar = 2 * n + 1
    # c: minimize -delta
    c = np.zeros(nvar)
    c[-1] = -1.0  # maximize delta
    
    # Inequality constraints A_ub @ x <= b_ub
    A_ub_list = []
    b_ub_list = []
    
    # A_i @ d + delta <= 0 for each active row
    for i in range(len(A)):
        row = np.zeros(nvar)
        row[:n] = A[i]
        row[-1] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(1e-12)  # small positive for numerical
    
    # d_i - s_i <= 0
    for i in range(n):
        row = np.zeros(nvar)
        row[i] = 1.0
        row[n + i] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(0.0)

    for i in range(n):
        row = np.zeros(nvar)
        row[i] = -1.0
        row[n + i] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(0.0)
    
    # sum(s_i) <= 1
    row = np.zeros(nvar)
    row[n:2*n] = 1.0
    A_ub_list.append(row)
    b_ub_list.append(1.0)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.success and res.x is not None:
            d = res.x[:n]
            delta = res.x[-1]
            if delta > 1e-8:

                if dnorm > 1e-14:
                    return d / dnorm
            else:
                # Use the direction even if delta is small
                dnorm = np.abs(d).sum()
                if dnorm > 1e-14:
                    return d / dnorm
        return None
    except:
        return None

            return np.ones(n)

def gradient_lag(h, lag, n):
    """Exact gradient of F_lag(h) = dx * sum h[i]*(1-h[i+lag]) over valid indices."""
    dx = 2.0 / n

        idx = np.arange(n - lag)

        idx2 = np.arange(lag, n)
        grad[idx2] -= dx * h[idx2 - lag]
