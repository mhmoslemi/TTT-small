Now, for the "smooth max" or active-lag gradient:
The objective is C(h) = max_q corr[q] * dx

The subgradient of the max is any convex combination of the gradients of the active lags. For our purposes, we can use a weighted average where the weights are based on how close each lag is to the maximum.

def active_lag_gradient(h, frac=0.1):
    """
    Compute weighted gradient of active lags.
    Returns gradient vector and the max correlation value.
    """
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    max_val = float(np.max(corr))
    min_val = float(np.min(corr))
    
    # Active lags: those within frac of the max
    thresh = max_val - frac * (max_val - min_val + 1e-15)
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    
    # Weights: softmax-like
    active_corr = corr[active]
    temp = max(1e-6, 0.1 * (max_val - float(active_corr.min()) + 1e-15))
    weights = np.exp((active_corr - max_val) / (temp / dx))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, q in zip(weights, active):
        lag = q - (n - 1)
        if lag == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            L = n - lag
            if L > 0:
                grad[lag:] += w * dx * (1.0 - h[:L])
                grad[:L] -= w * dx * h[lag:]
        else:
            a = -lag
            L = n - a
            if L > 0:
                grad[:L] += w * dx * (1.0 - h[a:])
                grad[a:] -= w * dx * h[:L]
    
    return grad, max_val * dx, corr

def evaluate(h):

    n = len(h)

    return float(np.max(corr) * dx)

def feasible_fix(h, target_sum):
