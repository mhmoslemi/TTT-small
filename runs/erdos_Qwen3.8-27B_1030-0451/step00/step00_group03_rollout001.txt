import numpy as np

def project_onto_feasible(y, target_sum):
    """Project y onto {0<=h<=1, sum(h)=target_sum}"""
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    
    # Handle non-finite
    y = np.nan_to_num(y, nan=0.5, posinf=0.5, neginf=0.5)
    
    # Bisection on alpha: find alpha such that sum(clip(y+alpha, 0, 1)) = target_sum
    lo = -1.0 - y.max()
    hi = 1.0 - y.min()
    
    def S(alpha):
        return np.sum(np.clip(y + alpha, 0.0, 1.0)) - target_sum
    
    # Check endpoints
    if S(0) <= 0:
        # Already sum <= target, need to increase
        lo = 0.0
    if S(1.0) >= 0:
        hi = 1.0
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if S(mid) > 0:
            hi = mid
        else:
            lo = mid
    
    result = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            rooms = 1.0 - result
            order = np.argsort(-rooms)
            for i in order:
                step = min(rooms[i], residual)
                result[i] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            residual = -residual
            order = np.argsort(result)
            for i in order:
                step = min(result[i], residual)
                result[i] -= step
                residual -= step
                if abs(residual) < 1e-14:
                    break
    
    return result

def evaluate_c5(h):
    """Compute C5 = max_k integrate h(x)(1-h(x+k)) dx"""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    """Compute all correlation values f_m"""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smoothed_lse(values, tau):
    """Smoothed max using log-sum-exp"""
    fmax = np.max(values)
    z = (values - fmax) / tau
    # Clip to prevent overflow
    z = np.clip(z, -50, 50)
    log_sum = np.log(np.sum(np.exp(z)))
    return fmax + tau * log_sum

def gradient_smoothed(h, tau, top_k=None):
    """
    Compute gradient of smoothed objective.
    Uses top-k active lags for efficiency.
    """
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    f = compute_correlations(h)  # length 2n-1
    
    if top_k is not None and top_k < len(f):
        idx = np.argsort(f)[-top_k:]
        f_active = f[idx]
    else:
        idx = np.arange(len(f))
        f_active = f
    
    f_max = np.max(f_active)
    z = (f_active - f_max) / tau
    z = np.clip(z, -50, 50)
    w = np.exp(z)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(len(f_active)) / len(f_active)
    else:
        w = w / w_sum
    
    # Gradient: sum_m w_m * df_m/dh
    grad = np.zeros(n, dtype=np.float64)
    
    for i, m in enumerate(idx):
        if w[i] < 1e-6:
            continue
        k = m - (n - 1)  # lag
        # f_m = dx * sum_j h[j+k]*(1-h[j]) for valid j
        # df_m/dh[j] = -dx * h[j+k] (for valid j)
        # df_m/dh[j+k] = dx * (1-h[j]) (for valid j)
        
        if k >= 0:
            L = n - k
            # j ranges from 0 to L-1
            grad[:L] += w[i] * dx * (-h[k:k+L])
            grad[k:k+L] += w[i] * dx * (1.0 - h[:L])
        else:
            r = -k
            L = n - r
            # j ranges from r to n-1
            grad[r:r+L] += w[i] * dx * (-h[:L])
            grad[:L] += w[i] * dx * (1.0 - h[r:r+L])
    
    return grad

def projected_gradient_descent(h0, max_iter=2000, tau_init=0.005, tau_min=5e-4, 
                                 top_k=None, eta_init=0.01, time_limit=None, t_start=None):
    """Run projected gradient descent with smoothed objective"""
    h = np.array(h0, dtype=np.float64)
    n = len(h)
    target_sum = n / 2.0
    tau = tau_init
    eta = eta_init
    best_h = h.copy()

    no_improve = 0
    
    if top_k is None:
        top_k = min(n, max(30, n // 2))
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        f = compute_correlations(h)
        fmax = np.max(f)
        
        # Update best
        if fmax < best_c5 - 1e-12:
            best_c5 = fmax
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Try increasing tau or breaking
            if no_improve > 500:
                break
            tau = max(tau * 1.5, tau_min * 10)
            no_improve = 0
            continue
        
        # Compute gradient
        grad = gradient_smoothed(h, tau, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Line search
        best_trial = None
        best_trial_c5 = best_c5
        for trial_factor in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            step = eta * trial_factor / gnorm
            y = h - step * grad

            if c5_trial < best_trial_c5 - 1e-12:

                break

        else:

            eta *= 0.5
            if eta < 1e-6:
                break

        if fmax < best_c5:

    return best_h, best_c5

def slsqp_refine(h0, active_lags, time_budget=30.0):

    h0 = np.asarray(h0, dtype=np.float64)
    n = len(h0)
    target_sum = n / 2.0
    dx = 2.0 / n
    c5_0 = evaluate_c5(h0)
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t
    # Constraints: sum(h) = target_sum, f_m(h) <= t for m in active_lags, 0 <= h <= 1
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    constraints = []
    jacobian = []
    
    # Equality: sum(h) = target_sum
    def eq_con(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_grad(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    constraints.append({'type': 'eq', 'fun': eq_con, 'jac': eq_grad})
    
    # Inequality: t - f_m(h) >= 0 for active lags
    for m in active_lags:
        k = m - (n - 1)
        
        if k >= 0:
            L = n - k
            j1 = np.arange(L)
            j2 = j1 + k
        else:
            r = -k
            L = n - r
            j1 = np.arange(r, n)
            j2 = np.arange(L)
        
        def ineq_con(x, j1=j1, j2=j2, dx=dx):
            h = x[:n]
            t = x[-1]
            f_m = dx * np.sum(h[j2] * (1.0 - h[j1]))
            return t - f_m
        
        def ineq_grad(x, j1=j1, j2=j2, dx=dx):
            h = x[:n]
            g = np.zeros(n + 1)
            g[-1] = 1.0
            g[j1] += -dx * h[j2]
            g[j2] += dx * (1.0 - h[j1])
            return g
        
        constraints.append({'type': 'ineq', 'fun': ineq_con, 'jac': ineq_grad})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h0, [c5_0 + 0.001]])
    
    t_start = time.time()

        result = minimize(objective, x0, method='SLSQP',
                         jac=objective_grad,
                         bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8})
    except Exception:
        return h0, c5_0

        return h0, c5_0

        h_new = project_onto_feasible(h_new, target_sum)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_0:
            return h_new, c5_new
    
    return h0, c5_0

def get_top_lags(h, k):
    """Get indices of top-k correlation values"""
    f = compute_correlations(h)
    return np.argsort(f)[-k:]

def resample_h(h, new_n):
    """Resample h to new resolution using linear interpolation"""
    n = len(h)
    dx_old = 2.0 / n
    # Sample points are at centers: (i + 0.5) * dx_old
    x_old = (np.arange(n) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    
    # Interpolate
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    h_new = project_onto_feasible(h_new, new_n / 2.0)
    return h_new

def project_onto_feasible(y, target_sum):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -y.max() - 0.1
    hi = 1.0 - y.min() + 0.1
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            rooms = 1.0 - result
            order = np.argsort(-rooms)
            for i in order:
                step = min(rooms[i], residual)
                result[i] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            residual = -residual
            order = np.argsort(result)
            for i in order:
                step = min(result[i], residual)
                result[i] -= step
                residual -= step
                if abs(residual) < 1e-14:
                    break
    return result

def evaluate_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def gradient_smoothed(h, tau, f_vals, top_k=None):
    n = len(h)
    dx = 2.0 / n
    f = f_vals
    if top_k is not None and top_k < len(f):
        idx = np.argsort(f)[-top_k:]
        f_a = f[idx]
    else:
        idx = np.arange(len(f))
        f_a = f
    f_max = np.max(f_a)
    z = np.clip((f_a - f_max) / tau, -50.0, 50.0)
    w = np.exp(z)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(len(f_a)) / len(f_a)
    else:
        w = w / w_sum
    grad = np.zeros(n, dtype=np.float64)
    for i, m in enumerate(idx):
        if w[i] < 1e-7:
            continue
        k = m - (n - 1)
        wi = w[i] * dx
        if k >= 0:
            L = n - k
            grad[:L] += wi * (-h[k:k+L])
            grad[k:k+L] += wi * (1.0 - h[:L])
        else:
            r = -k
            L = n - r
            grad[r:r+L] += wi * (-h[:L])
            grad[:L] += wi * (1.0 - h[r:r+L])
    return grad

def pgd(h0, max_iter, tau_init, tau_min, top_k, time_limit, t_start):
    h = np.array(h0, dtype=np.float64)
    n = len(h)
    target_sum = n / 2.0
    tau = tau_init
    eta = 0.02
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        f = compute_correlations(h)
        fmax = np.max(f)
        if fmax < best_c5 - 1e-12:
            best_c5 = fmax
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        if no_improve > 100 and no_improve % 50 == 0:
            tau = min(tau * 1.4, 0.02)
        grad = gradient_smoothed(h, tau, f, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        improved = False
        for factor in [1.0, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:

        return x[-1]

        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

        return np.sum(x[:n]) - target_sum

def resample_h(h, new_n):

                if time.time() > hard_end - 2:
                    break
                n_a, n_b = cs[i][0], cs[j][0]
                n_common = max(n_a, n_b)
                h_a = resample_h(cs[i][2], n_common) if n_a != n_common else cs[i][2].copy()
                h_b = resample_h(cs[j][2], n_common) if n_b != n_common else cs[j][2].copy()
                for alpha in [0.3, 0.5, 0.7]:
                    h_avg = project_onto_feasible(alpha * h_a + (1 - alpha) * h_b, n_common / 2.0)
                    try_accept(h_avg, evaluate_c5(h_avg))
    
    # Final validation
    best_h = project_onto_feasible(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, best_n)
