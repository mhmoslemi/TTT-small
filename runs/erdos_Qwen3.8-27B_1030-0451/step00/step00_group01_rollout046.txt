    """Solve LP to find a minimizing direction using active constraints."""
    target = n / 2.0
    dx = 2.0 / n
    v = np.correlate(h0, 1.0 - h0, mode='full') * dx
    max_v = np.max(v)
    
    # Select active lags (those within some fraction of max)
    threshold = 0.95 * max_v
    active_indices = np.where(v >= threshold)[0]
    if len(active_indices) > max_active:
        # Keep the top max_active
        order = np.argsort(v[active_indices])[::-1][:max_active]
        active_indices = active_indices[order]
    
    # Build LP
    n_lags = len(active_indices)
    # A_ub: n_lags x (n+1)
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)

        a = lag_gradient(h0, L)  # this is the gradient of v_L
        A_ub[idx, :n] = a
        A_ub[idx, n] = -1.0
        b_ub[idx] = a @ h0 - v[k]

    bounds = [(0, 1)] * n + [(0, None)]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:

            c5_trial = evaluate_c5(h_trial)
            return h_trial, c5_trial
    except Exception:
        pass
    
    return None, None

import numpy as np
from scipy.optimize import minimize, linprog

def project_sum_box(x, target):
    """Project x onto {0<=h<=1, sum(h)=target}."""
    n = len(x)
    x = np.asarray(x, dtype=float)
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    y = np.clip(x + lam, 0.0, 1.0)
    residual = target - np.sum(y)
    if abs(residual) > 1e-12:
        dist = np.minimum(y, 1.0 - y)
        order = np.argsort(-dist)
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - y[i])
                y[i] += add
                residual -= add
            else:
                sub = min(-residual, y[i])
                y[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return y

def evaluate_c5(h):
    n = len(h)

    res = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(res))

def smooth_obj_grad(h, beta):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(v)
    shifted = np.clip(beta * (v - m), -500, 500)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    f = m + np.log(Z) / beta
    w = exp_vals / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = k - (n - 1)
        if L >= 0:
            ie = n - L
            grad[:ie] += wk * dx * (1.0 - h[L:L+ie])
            grad[L:L+ie] -= wk * dx * h[:ie]
        else:
            M = -L
            grad[M:] += wk * dx * (1.0 - h[:-M])
            grad[:-M] -= wk * dx * h[M:]
    return f, grad

def lag_grad_at(h, L):
    """Gradient of v_L at point h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if L >= 0:
        ie = n - L
        grad[:ie] += dx * (1.0 - h[L:L+ie])
        grad[L:L+ie] -= dx * h[:ie]
    else:
        M = -L
        grad[M:] += dx * (1.0 - h[:-M])
        grad[:-M] -= dx * h[M:]
    return grad

            bounds7 = [(0, 1)] * len(best_h)
            res7 = minimize(obj7, best_h, method='SLSQP', jac=jac7,
                           bounds=bounds7, constraints=cons7,
                           options={'maxiter': 100, 'ftol': 1e-14})
            if res7.x is not None:
                h7 = project_sum_box(res7.x, target)
                c5_7 = evaluate_c5(h7)
                if c5_7 < best_c5 - 1e-12:
                    best_h = h7
                    best_c5 = c5_7
        except Exception:
            pass

    # Final validation
    best_h = project_sum_box(best_h, target)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum
    residual = target - np.sum(best_h)
    if abs(residual) > 1e-10:
        best_h = project_sum_box(best_h, target)
    best_c5 = evaluate_c5(best_h)
    best_h = best_h.astype(np.float64)

    return best_h, best_c5, len(best_h)
