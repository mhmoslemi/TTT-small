import numpy as np
import time

def project_box_sum(y, target_sum):
    """Project y onto {x : sum(x) = target_sum, 0 <= x <= 1} via bisection."""
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def compute_corr(h):
    """Full cross-correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def c5_from_corr(corr, dx):
    """C5 value from correlation array."""
    return np.max(corr) * dx

def compute_grad_fft(h, w, n, fft_size):
    """Compute gradient using FFT. w has length 2n-1."""
    W = w  # length 2n-1
    u = 1.0 - h  # length n
    Wh_f = np.fft.rfft(W, fft_size)
    u_f = np.fft.rfft(u, fft_size)
    A_full = np.fft.irfft(Wh_f * u_f, fft_size)
    A = A_full[n - 1: 2 * n - 1]
    hr_f = np.fft.rfft(h[::-1], fft_size)
    B_full = np.fft.irfft(Wh_f * hr_f, fft_size)
    B = B_full[n - 1: 2 * n - 1][::-1]
    return A - B

def make_initial(n, seed):
    """Create a diverse initial function."""
    rng = np.random.RandomState(seed)
    r = rng.rand()
    if r < 0.25:
        h = np.full(n, 0.5)
    elif r < 0.5:
        h = np.zeros(n)
        h[::2] = 1.0
    elif r < 0.75:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    else:
        n_b = rng.randint(n // 5, n // 2)
        starts = np.sort(rng.choice(n, n_b, replace=False))
        for s in starts:
            e = min(s + rng.randint(1, n // 4), n)
            h[s:e] = 1.0
    h = np.clip(h, 0, 1)
    h = project_box_sum(h, n / 2.0)
    h = h + rng.normal(0, 0.01, n)
    return project_box_sum(h, n / 2.0)

def optimize_local(h_init, n, max_time, start_time, seed):

    h = project_box_sum(h, target)
    fft_size = 1

    corr = compute_corr(h)
    c5 = c5_from_corr(corr, dx)
    best_local = c5

    beta = 5.0
    lr = 0.008
    no_improve = 0
