def compute_c(d, h, N):
    """Compute C(d) = sum_i h[i]*(1-h[i+d]) for valid i."""
    lo = max(0, -d)
    hi = min(N-1, N-1-d)
    if lo > hi:
        return 0.0
    i_vals = np.arange(lo, hi+1)
    return np.sum(h[i_vals] * (1 - h[i_vals + d]))

    """Compute C(d) for all d from -(N-1) to N-1."""
    d_vals = np.arange(-(N-1), N)
    c_vals = np.array([compute_c(d, h, N) for d in d_vals])
    return c_vals  # indexed by d + (N-1)

def smooth_max(c_vals, tau):
    """Log-sum-exp smooth max."""
    m = np.max(c_vals)
    return m + tau * np.log(np.sum(np.exp((c_vals - m) / tau)))

    """Gradient of smooth max with respect to c_vals."""
    m = np.max(c_vals)
    weights = np.exp((c_vals - m) / tau)
    weights /= np.sum(weights)
    return weights

        d = k - (N - 1)
        lo = max(0, -d)
        hi = min(N - 1, N - 1 - d)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += a * (1.0 - h[idx + d])
        j_lo = lo + d
        j_hi = hi + d
        if j_lo <= j_hi:
            j_vals = np.arange(j_lo, j_hi + 1)
            grad[j_vals] -= a * h[j_vals - d]
    grad *= (2.0 / N)
    return obj, grad
