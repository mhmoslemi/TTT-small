def project_feasible(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Iterative water-filling
    h = np.clip(h, 0, 1)
    for _ in range(100):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to decrease sum
            mask = h < 1  # Can decrease any element, even those at 1? No, can't go below 0
            mask = h > 0  # Can only decrease elements > 0
            if not mask.any():
                break
            # Decrease elements proportionally or uniformly
            # Simple: decrease all elements > 0 uniformly
            step = min(diff / mask.sum(), h[mask].min())
            h[mask] -= step
        else:
            # Need to increase sum
            mask = h < 1
            if not mask.any():
                break
            step = min(-diff / mask.sum(), 1 - h[mask].max())
            h[mask] += step
    return h
