def project_to_constraint(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0, 1)
    # Binary search for the right adjustment
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break

        else:
            mask = h > 0

    corr = np.correlate(h, 1 - h, mode="full")

import time

def project_to_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return h

def compute_c5(h, dx):
    """Compute C5 = max_k overlap_k * dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx), corr * dx

def grad_lag(h, d):
    """Gradient of overlap at displacement d."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        return g
    if d > 0:
        g[:n-d] += 1.0
        g[:n-d] -= h[d:]
        g[d:] -= h[:n-d]
    else:
        m = -d
        g[m:] += 1.0
        g[m:] -= h[:n-m]
        g[:n-m] -= h[m:]
    return g

    t0 = time.time()
    best_h = h.copy()
    best_c5, _ = compute_c5(h, dx)
    lr = 0.02
    stalled = 0

    """Guided SA step: bias toward swaps that help the worst lag."""
    worst_d = int(np.argmax(overlaps)) - (n - 1)
    # Find positions that contribute to the worst lag
    if worst_d > 0:
        # overlap = sum h[i](1-h[i+worst_d]) for i in [0, n-1-worst_d]

def project_to_constraint(h, target_sum):
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx), corr * dx

def grad_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        return g
    if d > 0:
        g[:n-d] += 1.0
        g[:n-d] -= h[d:]
        g[d:] -= h[:n-d]
    else:
        m = -d
        g[m:] += 1.0
        g[m:] -= h[:n-m]
        g[:n-m] -= h[m:]
    return g
