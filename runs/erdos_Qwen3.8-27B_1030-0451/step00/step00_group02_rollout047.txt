Now let me think about the projection more carefully.

The constraint set is: {h ∈ R^n : 0 ≤ h_i ≤ 1, sum(h) = n/2}

The projection onto this set can be done in two steps:
1. Project onto {0 ≤ h_i ≤ 1}: just clip
2. Project onto {sum(h) = n/2} ∩ {0 ≤ h_i ≤ 1}: this is the intersection of a hyperplane with a box.

The projection onto the intersection is:
h* = argmin ||h - h*||² s.t. 0 ≤ h*_i ≤ 1, sum(h*) = n/2

By KKT, h*_i = clip(h_i + λ, 0, 1) for some λ that makes sum(h*) = n/2.

We find λ by bisection since sum(clip(h_i + λ, 0, 1)) is monotonically increasing in λ.

```python
def project_onto_constraint(h, n):
    """Project h onto {0 <= h_i <= 1, sum(h) = n/2}"""
    target = n / 2.0
    
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0, 1).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    
    return np.clip(h + (lo + hi) / 2, 0, 1)
```

Now for the subgradient computation. Let me be very precise.

Given h of length n, corr = np.correlate(h, 1-h, "full") has length 2n-1.

For index k (0-indexed) in corr:
corr[k] = sum_{i} h[i] * (1-h[i-k+n-1])

where the sum is over i such that 0 ≤ i ≤ n-1 AND 0 ≤ i-k+n-1 ≤ n-1.

The valid range of i: max(0, k-n+1) ≤ i ≤ min(n-1, k).

Let me define:
- lo = max(0, k - n + 1)
- hi = min(n-1, k)

Then corr[k] = sum_{i=lo}^{hi} h[i] * (1 - h[i - k + n - 1])

Let's define j = i - k + n - 1, so i = j + k - n + 1.
When i = lo, j = lo - k + n - 1
When i = hi, j = hi - k + n - 1

So corr[k] = sum_{j=j_lo}^{j_hi} h[j + k - n + 1] * (1 - h[j])

where j_lo = lo - k + n - 1 and j_hi = hi - k + n - 1.

Now, ∂corr[k]/∂h[m]:

corr[k] = sum_{i=lo}^{hi} h[i] - sum_{i=lo}^{hi} h[i] * h[i-k+n-1]

Let's call the first term A and the second term B.

∂A/∂h[m] = 1 if lo ≤ m ≤ hi, else 0.

For B:
B = sum_{i=lo}^{hi} h[i] * h[i-k+n-1]

Let l = k - n + 1 (so i-k+n-1 = i+l).

B = sum_{i=lo}^{hi} h[i] * h[i+l]

∂B/∂h[m] = h[m+l] * I[lo ≤ m ≤ hi] + h[m-l] * I[lo ≤ m-l ≤ hi]

Wait, the second term: if we're differentiating with respect to h[m] in the factor h[i+l], we need i+l = m, so i = m-l. This contributes h[m-l] (from h[i]) if m-l is in [lo, hi].

So:
∂B/∂h[m] = h[m+l] * I[lo ≤ m ≤ hi] + h[m-l] * I[lo ≤ m-l ≤ hi]

But we also need the indices m+l and m-l to be in [0, n-1] for h[m±l] to be defined.

So:
∂corr[k]/∂h[m] = I[lo ≤ m ≤ hi] - h[m+l] * I[lo ≤ m ≤ hi] * I[0 ≤ m+l ≤ n-1] - h[m-l] * I[lo ≤ m-l ≤ hi] * I[0 ≤ m-l ≤ n-1]

This is O(1) per (k, m) pair, so for top_m lags and n elements, it's O(top_m * n).

```python
def compute_subgradient(h, top_m=10):
    n = len(h)
    corr = np.correlate(h, 1-h, "full")
    
    # Get top_m indices
    top_idx = np.argsort(corr)[-top_m:]
    
    grad = np.zeros(n)
    
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        l = k - n + 1  # lag
        
        # ∂A/∂h[m] = 1 for m in [lo, hi]
        grad[lo:hi+1] += 1.0
        
        # -h[m+l] for m in [lo, hi] and 0 ≤ m+l ≤ n-1
        if l >= 0:
            # m+l >= 0 always (since m >= lo >= 0)
            # m+l <= n-1 means m <= n-1-l
            m_end = min(hi, n - 1 - l)
            if lo <= m_end:
                grad[lo:m_end+1] -= h[lo+l:m_end+1+l]
        else:
            # m+l >= 0 means m >= -l
            m_start = max(lo, -l)
            m_end = hi
            if m_start <= m_end:
                grad[m_start:m_end+1] -= h[m_start+l:m_end+1+l]
        
        # -h[m-l] for m-l in [lo, hi] and 0 ≤ m-l ≤ n-1
        # m in [lo+l, hi+l] ∩ [0, n-1]
        m_start2 = max(0, lo + l)
        m_end2 = min(n - 1, hi + l)
        if m_start2 <= m_end2:
            grad[m_start2:m_end2+1] -= h[m_start2-l:m_end2+1-l]
    
    return grad / top_m
```

def project_onto_constraint(h, n):
    """Project h onto {0 <= h_i <= 1, sum(h) = n/2}"""
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgradient(h, top_m=20):
    """Compute subgradient of max_k corr[k] using top_m active lags"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, "full")
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    
    for k in top_idx:
        s_val = k - n + 1
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        
        # d/dh[m] of sum_{i=lo}^{hi} h[i] = 1 for m in [lo, hi]
        grad[lo:hi+1] += 1.0
        
        # d/dh[m] of sum_{i=lo}^{hi} h[i]*h[i-s_val]
        # Part 1: m is the first index (i=m), contributes h[m-s_val]
        m1_s = max(lo, s_val)
        m1_e = min(hi, n - 1 + s_val)
        if m1_s <= m1_e:
            grad[m1_s:m1_e+1] -= h[m1_s - s_val : m1_e + 1 - s_val]
        
        # Part 2: m is the second index (i-s_val=m, i=m+s_val), contributes h[m+s_val]
        m2_s = max(0, lo - s_val)
        m2_e = min(n - 1, hi - s_val)
        if m2_s <= m2_e:
            grad[m2_s:m2_e+1] -= h[m2_s + s_val : m2_e + 1 + s_val]
    
    return grad / float(top_m), corr

import numpy as np
import time

def project_onto_constraint(h, n):
    """Project h onto {0 <= h_i <= 1, sum(h) = n/2}"""
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgradient(h, top_m=25):
    """Compute subgradient of max_k corr[k] using top_m active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, "full")
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        s_val = k - n + 1
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 1.0
        m1_s = max(lo, s_val)
        m1_e = min(hi, n - 1 + s_val)
        if m1_s <= m1_e:
            grad[m1_s:m1_e+1] -= h[m1_s - s_val : m1_e + 1 - s_val]
        m2_s = max(0, lo - s_val)
        m2_e = min(n - 1, hi - s_val)
        if m2_s <= m2_e:
            grad[m2_s:m2_e+1] -= h[m2_s + s_val : m2_e + 1 + s_val]
    return grad / float(top_m), corr

def get_objective(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, "full")
    return corr.max() * dx, corr

def optimize_h(h0, n, budget_end, rng, max_top_m=30):
    """Run momentum gradient descent on a single starting point."""
    h = project_onto_constraint(h0.copy(), n)
    velocity = np.zeros(n)
    beta = 0.85
    alpha = 1.5 / n
    best_local = get_objective(h, n)[0]
    no_improve = 0
    best_h_local = h.copy()
    best_val_local = best_local
    
    while time.time() < budget_end:
        grad, corr = compute_subgradient(h, top_m=max_top_m)
        c5 = corr.max() * (2.0 / n)
        
        if c5 < best_val_local:
            best_val_local = c5
            best_h_local = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        velocity = beta * velocity - alpha * grad
        h_new = project_onto_constraint(h + velocity, n)
        
        corr_new = np.correlate(h_new, 1.0 - h_new, "full")
        c5_new = corr_new.max() * (2.0 / n)
        
        if c5_new <= c5 * 1.005:
            h = h_new
            alpha = min(alpha * 1.03, 4.0 / n)
        else:
            velocity *= 0.3
            alpha *= 0.6
        
        if no_improve > 400:
            scale = 0.4 / np.sqrt(n)
            h = project_onto_constraint(h + rng.normal(0, scale, n), n)
            velocity = np.zeros(n)
            alpha = 1.5 / n
            no_improve = 0
    
    return h, best_h_local, best_val_local
