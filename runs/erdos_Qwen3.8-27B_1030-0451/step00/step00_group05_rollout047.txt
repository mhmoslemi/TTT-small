import numpy as np
from scipy.optimize import minimize_scalar

def project_feasible(v, n):
    """Project v onto {h: 0 <= h[i] <= 1, sum(h) = n/2}"""
    # Binary search for lambda such that sum(clip(v - lambda, 0, 1)) = n/2
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    """Compute C5 = max of correlate(h, 1-h, full) * dx"""
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    c = np.correlate(h, u, mode="full") * dx
    return c.max(), c

def gradient_of_corr_lag(h, k, n, dx):
    """
    Compute gradient of C[k] (the k-th element of full correlation * dx) w.r.t. h.
    k is the index in the full correlation output (0 to 2n-2).
    Center is n-1.
    """
    grad = np.zeros(n)
    center = n - 1
    r = k - center  # lag offset
    
    if r >= 0:
        # c[index] = sum_{i=0}^{n-1-r} h[i+r] * (1-h[i])
        # d/dh[j]:
        #   if j in [r, n-1]: (1-h[j-r])  [h[j] appears as h[i+r] where i = j-r]
        #   if j in [0, n-1-r]: -h[j+r]   [h[j] appears in (1-h[i]) where i = j]
        valid_i = np.arange(0, n - r)  # i from 0 to n-1-r

        # (1-h[i]) terms: for i = j, j ranges from 0 to n-1-r
        # Combined:
