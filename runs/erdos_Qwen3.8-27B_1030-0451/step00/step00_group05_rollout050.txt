    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(g - lam, 0.0, 1.0)
    # Correct tiny floating point errors
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        nm = mask.sum()
        if nm > 0:
            h[mask] += diff / nm
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return (compute_correlation(h) * dx).max()

def grad_weighted(h, w):
    """Compute dx * sum_k w[k] * dCorr_k/dh."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    c = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-14:
            continue
        L = k - c
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            m = n - L
            g[:m] += wk * (1.0 - h[L:])
            g[L:] += wk * (-h[:m])
        else:
            M = -L
            m = n - M
            g[M:] += wk * (1.0 - h[:m])
            g[:m] += wk * (-h[M:])
    return g * dx

def gradient_step(h, n, T, step):
    """One step of softmax-weighted gradient descent with backtracking."""
    dx = 2.0 / n
    C = compute_correlation(h) * dx
    Cmax = C.max()
    
    # Softmax weights
    z = C / T
    z -= z.max()
    p = np.exp(z)
    p /= p.sum()
    
    # Gradient
    g = grad_weighted(h, p)
    g_tan = g - g.mean()
    d = -g_tan
    
    # Backtracking line search
    ss = step
    for _ in range(30):
        h_new = project_feasible(h + ss * d, n)
        c5_new = c5_of(h_new)
        if c5_new < Cmax - 1e-14:
            return h_new, c5_new
        ss *= 0.5
        if ss < 1e-12:
            break
    return h, Cmax

def resample(h_old, n_new):
    """Resample from old grid to new grid."""
    n_old = len(h_old)
    old_x = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    new_x = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(new_x, old_x, h_old)
