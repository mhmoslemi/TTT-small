def compute_overlaps(h, dx):
    """Compute all overlap values I(d) = dx * sum h[i]*(1-h[i+d]) for d=0..N-1."""
    N = len(h)
    overlaps = np.zeros(N)
    for d in range(N):
        overlaps[d] = dx * np.dot(h[:N-d], 1 - h[d:])
    return overlaps

def compute_gradient(h, dx, active_d, weights):
    """Compute gradient of weighted sum of overlaps."""
    N = len(h)
    grad = np.zeros(N)
    for idx, d in enumerate(active_d):
        w = weights[idx]
        if N - d > 0:
            grad[:N-d] += w * dx * (1 - h[d:N])
        if d > 0:
            grad[d:] += -w * dx * h[:N-d]
    return grad

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def grad_c(h, active_idx, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(active_idx)):
        k = active_idx[i]
        w = weights[i]
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        if lo1 <= hi1:
            p = np.arange(lo1, hi1 + 1)
            v = p - k + n - 1
            grad[p] += w * (1.0 - h[v])
        lo2 = max(0, n - 1 - k)
        hi2 = min(n - 1, 2 * n - 2 - k)
        if lo2 <= hi2:
            p2 = np.arange(lo2, hi2 + 1)
            v2 = p2 + k - n + 1
            grad[p2] -= w * h[v2]
    return grad

def project(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - np.sum(x)
    if abs(d) > 1e-13:
        m = (x > 1e-9) & (x < 1.0 - 1e-9)
        if np.any(m):
            x[m] += d / np.count_nonzero(m)
            np.clip(x, 0.0, 1.0, out=x)
    return x

def interp_n(h, nn):
    n = len(h)
    ox = (np.arange(n) + 0.5) * (2.0 / n)
    nx = (np.arange(nn) + 0.5) * (2.0 / nn)
    return np.clip(np.interp(nx, ox, h), 0.0, 1.0)
