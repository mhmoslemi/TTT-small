def project_sum_box(x, target):
    n = len(x)
    lo, hi = -max(x.min(), 0.0) - 1.0, 1.0 - min(x.max(), 1.0) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - result.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        if mask.any():
            caps = (1.0 - result[mask]) if residual > 0 else result[mask]
            total = caps.sum()
            if total > 0:
                result[mask] += residual * caps / total
    return np.clip(result, 0.0, 1.0)

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lag(h, lag, n):
    """Gradient of v_lag = dx * sum_i h[i]*(1-h[i+lag]) w.r.t. h"""
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        # First factor: j in [0, n-1-lag]
        if n - 1 - lag >= 0:
            j = np.arange(0, n - lag)
            g[j] += dx * (1.0 - h[j + lag])
        # Second factor: j in [lag, n-1]
        if lag <= n - 1:
            j = np.arange(lag, n)
            g[j] -= dx * h[j - lag]
    else:
        m = -lag
        # First factor: j in [m, n-1]
        if m <= n - 1:

            g[j] += dx * (1.0 - h[j - m])

        if n - 1 - m >= 0:

            g[j] -= dx * h[j + m]

    """Compute log-sum-exp of correlation values and its gradient."""
    n = len(h)
    dx = 2.0 / n
    v = all_corr_values(h)
    M = v.max()
    u = np.exp(beta * (v - M))
    Z = u.sum()
    f = M + np.log(Z) / beta
    w = u / Z
    
    # Gradient: sum_lag w_lag * grad_v_lag
    g = np.zeros(n)
    # For efficiency, we can compute this as a weighted sum
    # But for n=84, even looping is fine
    for k in range(len(v)):
        if w[k] < 1e-15:
            continue
        lag = k - (n - 1)
        g += w[k] * grad_for_lag(h, lag, n)
    return f, g

    """Get gradients for the most active lags."""
    v = all_corr_values(h)
    # Sort by value, take top ones
    idx = np.argsort(v)[::-1]
    active = []
    for k in idx[:max_active]:
        lag = k - (n - 1)
        g = grad_for_lag(h, lag, n)
        active.append((lag, g, v[k]))
    return active, v

def solve_lp(h0, v0, grads_info, radius, n, target_sum):
    """Solve trust-region LP to reduce max overlap.
    Returns candidate h or None.
    """
    # Variables: [h_0, ..., h_{n-1}, t]
    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # Trust region + box: h0_i - r <= h_i <= h0_i + r, and 0 <= h_i <= 1
    # => -1*(h0_i - r) <= -h_i <= -(h0_i + r) ... no, let's use bounds
    
    # Actually, let's use bounds for the box+trust region
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    
    # Active constraint linearizations:
    # grad_lag @ h - t <= grad_lag @ h0 - v0_lag
    # => [grad_lag, -1] @ [h, t] <= grad_lag @ h0 - v0_lag
    A_rows = []
    b_vals = []
    for lag, g, v_lag in grads_info:
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(g @ h0 - v_lag)
    
    if A_rows:
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
    else:
        A_ub = None
        b_ub = None
    
    # Equality: sum(h) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds
    bounds = list(zip(lb, ub))
    bounds.append((0.0, 1.0))  # t is free-ish, but bounded
    # Actually t should be bounded below by 0 and above by 1
    bounds[-1] = (0.0, 1.0)
    
    try:

                      bounds=bounds, method='highs',

    if not res.success:
        return None
    
    h_cand = res.x[:n]
    h_cand = project_sum_box(h_cand, target_sum)

        f, g = logsumexp_obj_and_grad(x, beta)
        return f

        f, g = logsumexp_obj_and_grad(x, beta)
        return g

import numpy as np
import time
from scipy.optimize import linprog, minimize


def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(x)
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - result.sum()
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = result < 1.0 - 1e-9
        else:
            mask = result > 1e-9
        if mask.any():
            caps = (1.0 - result[mask]) if resid > 0 else result[mask]
            tot = caps.sum()
            if tot > 1e-15:
                result[mask] += resid * caps / tot
    return np.clip(result, 0.0, 1.0)


def evaluate_exact(h):
    """Compute C5 = max(correlate(h, 1-h, full) * dx)."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def all_corr_values(h):
    """Return all (2n-1) correlation values scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_for_lag(h, lag, n):
    """Analytic gradient of v_lag = dx * sum_i h[i]*(1-h[i+lag]) w.r.t. h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        lo1 = 0
        hi1 = n - lag
        if hi1 > lo1:
            j = np.arange(lo1, hi1)
            g[j] += dx * (1.0 - h[j + lag])
        lo2 = lag
        hi2 = n
        if hi2 > lo2:
            j = np.arange(lo2, hi2)
            g[j] -= dx * h[j - lag]
    else:
        m = -lag
        lo1 = m
        hi1 = n
        if hi1 > lo1:
            j = np.arange(lo1, hi1)
            g[j] += dx * (1.0 - h[j - m])
        lo2 = 0
        hi2 = n - m
        if hi2 > lo2:
            j = np.arange(lo2, hi2)
            g[j] -= dx * h[j + m]
    return g


def logsumexp_obj_and_grad(h, beta):
    """Compute smoothed log-sum-exp objective and gradient."""
    n = len(h)
    v = all_corr_values(h)
    M = v.max()
    shifted = beta * (v - M)
    shifted = np.clip(shifted, -700, 700)
    u = np.exp(shifted)
    Z = u.sum()
    f = M + np.log(Z) / beta
    w = u / Z
    g = np.zeros(n)
    active_idx = np.where(w > 1e-12)[0]
    for k in active_idx:
        lag = k - (n - 1)
        g += w[k] * grad_for_lag(h, lag, n)
    return f, g


def solve_lp_active(h0, radius, n, target_sum, max_active=60, timeout=20):
    """Trust-region LP to reduce max overlap using active constraints."""
    v0 = all_corr_values(h0)
    idx = np.argsort(v0)[::-1]
    thresh = max(0.005, 0.4 * v0.max())
    active = [(k - (n - 1), v0[k]) for k in idx if v0[k] > thresh]
    if len(active) < 3:
        active = [(k - (n - 1), v0[k]) for k in idx[:max_active]]
    active = active[:max_active]

    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0

    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    if not np.all(lb < ub):
        return None

    A_rows = []
    b_vals = []
    for lag, v_lag in active:
        g = grad_for_lag(h0, lag, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(g @ h0 - v_lag)

    A_ub = np.array(A_rows) if A_rows else None
    b_ub = np.array(b_vals) if b_vals else None

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])

    bounds = [(lb[i], ub[i]) for i in range(n)]
    bounds.append((max(0.0, v0.max() - 2 * radius), min(1.0, v0.max() + 2 * radius)))

    try:
        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None

    if res is None or not res.success:
        return None
    h_cand = res.x[:n]
    if not np.all(np.isfinite(h_cand)):
        return None
    h_cand = project_sum_box(h_cand, target_sum)
    return h_cand


def slsqp_smooth(h0, beta, n, target_sum, maxiter=200):
    """SLSQP on log-sum-exp smoothed objective."""
    def obj(x):
        f, _ = logsumexp_obj_and_grad(x, beta)
        return f

    def grad(x):
        _, g = logsumexp_obj_and_grad(x, beta)
        return g

    def sum_eq(x):
        return x.sum() - target_sum

    def sum_eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
    except Exception:
        return None
    if res is not None and res.x is not None:
        x = np.array(res.x)
        if np.all(np.isfinite(x)):
            return project_sum_box(x, target_sum)
    return None


def projected_subgradient(h0, n, target_sum, step_size=0.02, n_iter=200):
    """Projected subgradient method on the active lag."""
    h = h0.copy()
    best_local = evaluate_exact(h)
    best_h_local = h.copy()
    for it in range(n_iter):
        v = all_corr_values(h)
        k_max = np.argmax(v)
        lag = k_max - (n - 1)
        g = grad_for_lag(h, lag, n)
        h_new = h - step_size * g
        h_new = project_sum_box(h_new, target_sum)
        c_new = evaluate_exact(h_new)
        if c_new < best_local:
            best_local = c_new
            best_h_local = h_new.copy()
        h = h_new
        step_size *= 0.999
    return best_h_local, best_local


def transfer_resolution(h_old, n_new):
    """Interpolate from old grid to new grid."""
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_sum_box(h_new, n_new / 2.0)

        # Also try sinusoidal on this grid

            h_sin = make_sinusoidal(n_new, omega)
            c_sin = evaluate_exact(h_sin)
            if c_sin < c_new:
                h_new = h_sin
                c_new = c_sin

        # Polish on new grid
        h_polished, c_polished = polish(h_new, n_new, n_new / 2.0, time_alloc)

        if c_polished < best_c - 1e-12:
            best_h = h_polished.copy()
            best_c = c_polished
            best_n = n_new

    # Final ALIP + SLSQP polish if time remains
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30 and best_c > 0.375:
        n_cur = len(best_h)
        time_alloc = min(remaining * 0.5, 150)
        if time_alloc > 20:
            best_h, best_c = polish(best_h, n_cur, n_cur / 2.0, time_alloc)

    # One more transfer attempt if we have time
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 60:
        current_n = len(best_h)
        n_next = min(current_n + 42, 300)
        if n_next > current_n:
            time_alloc = min(remaining * 0.4, 120)
            if time_alloc > 30:
                h_new = transfer_resolution(best_h, n_next)
                h_polished, c_polished = polish(h_new, n_next, n_next / 2.0, time_alloc)
                if c_polished < best_c - 1e-12:
                    best_h = h_polished.copy()
                    best_c = c_polished

    # Final validation
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    best_c = evaluate_exact(best_h)

    return best_h, float(best_c), len(best_h)
