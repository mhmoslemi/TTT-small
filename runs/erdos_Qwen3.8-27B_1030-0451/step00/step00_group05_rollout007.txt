def exact_score(h):
    """Compute exact C5 value for feasible h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()

def check_feasible(h, tol=1e-9):
    """Check if h is feasible."""
    n = len(h)
    if n == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    target = n / 2.0
    if abs(h.sum() - target) > tol * max(1, n):
        return False
    if h.min() < -1e-10:
        return False
    if h.max() > 1.0 + 1e-10:
        return False
    return True

def project_feasible(h):
    """Project h to the feasible set 0<=h<=1, sum(h)=n/2."""
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=np.float64).copy()
    # Iterative water-filling
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # Adjust: add/subtract diff/n to all, then re-clip
        # Better: only adjust free variables
        free = (h > 0) & (h < 1)
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute to interior points
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, top_k=20, tau=0.01):
    """
    Compute gradient of the smooth approximation to max correlation.
    Uses softmax weighting over top-k correlation values.
    Returns gradient array same shape as h.
    """
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode="full") * dx
    M = 2 * n - 1
    center = n - 1
    
    # Find top values
    sorted_idx = np.argsort(full)[::-1]
    top_idx = sorted_idx[:min(top_k, M)]
    
    # Smooth weights
    max_val = full[top_idx[0]]
    weights = np.exp((full[top_idx] - max_val) / tau)
    weights /= weights.sum()
    
    # Build weight array over full indices
    w_full = np.zeros(M)
    w_full[top_idx] = weights
    
    # Accumulate gradient
    g = np.zeros(n)
    
    # Zero lag (center index)
    w0 = w_full[center]
    if w0 > 0:
        g += w0 * (1.0 - 2.0 * h) * dx
    
    # Positive and negative lags
    for r in range(1, n):
        wp = w_full[center + r]  # weight for positive lag r
        wn = w_full[center - r]  # weight for negative lag r
        L = n - r
        
        if wp > 0:
            # c[r] = dx * sum_{i=0}^{L-1} h[i+r]*(1-h[i])
            # d/dr: h[k] as h[i+r] when k-r in [0,L-1], i.e., k in [r, n-1]
            #         h[k] in (1-h[i]) when k in [0, L-1]
            # ∂c[r]/∂h[k] = dx*(1-h[k-r]) for k in [r, n-1]
            #              - dx*h[k+r] for k in [0, L-1]
            g[r:] += wp * dx * (1.0 - h[:L])       # k from r to n-1, h[k-r] = h[0:L]
            g[:L] += -wp * dx * h[r:]               # k from 0 to L-1, h[k+r] = h[r:n]
        
        if wn > 0:
            # c[-r] = dx * sum_{i=0}^{L-1} h[i]*(1-h[i+r])
            # ∂c[-r]/∂h[k] = dx*(1-h[k+r]) for k in [0, L-1]
            #              - dx*h[k-r] for k in [r, n-1]
            g[:L] += wn * dx * (1.0 - h[r:])        # k from 0 to L-1, h[k+r] = h[r:n]
            g[r:] += -wn * dx * h[:L]               # k from r to n-1, h[k-r] = h[0:L]
    
    return g

def smooth_score(h, top_k=20, tau=0.01):
    """Compute smooth approximation to max correlation for SLSQP."""
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode="full") * dx
    sorted_idx = np.argsort(full)[::-1]
    top_idx = sorted_idx[:min(top_k, len(full))]
    max_val = full[top_idx[0]]
    weights = np.exp((full[top_idx] - max_val) / tau)
    weights /= weights.sum()
    return np.dot(weights, full[top_idx])

def try_pair_exchange(h, i, j, delta):
    """Try a pair exchange, return new h if feasible, else None."""
    ni = h[i] + delta
    nj = h[j] - delta
    if ni < -1e-12 or ni > 1.0 + 1e-12:

    if nj < -1e-12 or nj > 1.0 + 1e-12:

    new_h = h.copy()
    new_h[i] = max(0.0, min(1.0, ni))
    new_h[j] = max(0.0, min(1.0, nj))
    return new_h

def symmetrize(h):

            h_new[i] = p
            h_new[j] = 1.0 - p

import numpy as np
from scipy.optimize import minimize
import time
import math


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())


def check_feasible(h, tol=1e-8):
    n = len(h)
    if n == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    target = n / 2.0
    if abs(h.sum() - target) > tol * n:
        return False
    if h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return False
    return True


def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=np.float64).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:
            break
        h[free] += diff / nfree
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_gradient(h, top_k=20, tau=0.005):
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode="full") * dx
    M = 2 * n - 1
    center = n - 1
    sorted_idx = np.argsort(full)[::-1]
    k = min(top_k, M)
    top_idx = sorted_idx[:k]
    max_val = full[top_idx[0]]
    weights = np.exp((full[top_idx] - max_val) / max(tau, 1e-10))
    weights /= weights.sum()
    w_full = np.zeros(M)
    w_full[top_idx] = weights
    g = np.zeros(n)
    w0 = w_full[center]
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h) * dx
    for r in range(1, n):
        wp = w_full[center + r]
        wn = w_full[center - r]
        L = n - r
        if wp > 1e-15:
            g[r:] += wp * dx * (1.0 - h[:L])
            g[:L] += -wp * dx * h[r:]
        if wn > 1e-15:
            g[:L] += wn * dx * (1.0 - h[r:])
            g[r:] += -wn * dx * h[:L]
    return g


def smooth_score(h, top_k=20, tau=0.005):
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode="full") * dx
    M = len(full)
    sorted_idx = np.argsort(full)[::-1]
    k = min(top_k, M)
    top_idx = sorted_idx[:k]
    max_val = full[top_idx[0]]
    weights = np.exp((full[top_idx] - max_val) / max(tau, 1e-10))
    weights /= weights.sum()
    return float(np.dot(weights, full[top_idx]))


def try_pair_exchange(h, i, j, delta):
    ni = h[i] + delta
    nj = h[j] - delta
    if ni < -1e-12 or ni > 1.0 + 1e-12:
        return None
    if nj < -1e-12 or nj > 1.0 + 1e-12:
        return None
    new_h = h.copy()
    new_h[i] = max(0.0, min(1.0, ni))
    new_h[j] = max(0.0, min(1.0, nj))
    return new_h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    idx = np.searchsorted(x_old, x_new, side='right') - 1
    idx = np.clip(idx, 0, n_old - 1)
    h_new = h_old[idx].copy()
    return project_feasible(h_new)


def symmetrize(h):
    n = len(h)
    h_new = np.zeros(n)
    for i in range(n):
        j = n - 1 - i
        if i < j:
            p = (h[i] + (1.0 - h[j])) / 2.0
            p = max(0.0, min(1.0, p))
            h_new[i] = p
            h_new[j] = 1.0 - p
        elif i == j:
            h_new[i] = 0.5
    return project_feasible(h_new)


def slsqp_polish(h0, n_best, deadline, reserve):
    n = len(h0)
    tau = max(0.005, 0.01 * 0.38)
    top_k = 20
    h_flat = h0.copy()

    def obj(x):
        return smooth_score(x, top_k=top_k, tau=tau)

    def obj_grad(x):
        return compute_gradient(x, top_k=top_k, tau=tau)

    constraints = [
        {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    t_rem = deadline - reserve - time.monotonic() - 5
    if t_rem < 10:
        return h0, exact_score(h0)
    try:
        res = minimize(obj, h_flat, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_opt = project_feasible(res.x)
            if check_feasible(h_opt):
                return h_opt, exact_score(h_opt)
    except Exception:
        pass
    return h0, exact_score(h0)
