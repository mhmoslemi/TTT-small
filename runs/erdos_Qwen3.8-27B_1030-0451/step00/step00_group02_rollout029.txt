    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    shifted = (corr - M) / tau
    # Handle potential overflow
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w = w / w.sum()

            if n - lag > 0:

import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def ensure_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-13
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def paired_optimize(h, i, j, n, n_grid=200):
    """Optimize the pair (i,j) by grid search over the transfer amount."""
    a, b = h[i], h[j]
    # t ranges from -b to 1-a
    t_min, t_max = -b, 1.0 - a
    if t_max - t_min < 1e-12:
        return h, None
    
    # Precompute correlation at t=0
    h0 = h.copy()
    corr0 = correlate(h0, 1.0 - h0, mode='full').astype(np.float64)
    
    dx = 2.0 / n
    center = n - 1
    
    # For each lag k, compute the coefficient of t
    # c_k(t) = c_k(0) + t * coeff_k (for k != 0)
    # c_0(t) = c_0(0) + 2t(h[j]-h[i]) - 2t^2 (for k = 0)
    
    # Actually, let me just compute directly for efficiency
    # For a given t: h_new[i] = a+t, h_new[j] = b-t
    # Instead of recomputing full correlation, use the linearity
    
    # coeff_k for k != 0:
    # d/dt [h[i]*(1-h[i+k])] = (1-h[i+k]) if i+k valid
    # d/dt [h[j-k]*(1-h[j])] ... wait
    # Let me just use the direct formula
    
    # For lag r (where lag = r - center):
    # The change in corr[r] when h[i]→h[i]+t, h[j]→h[j]-t:
    # = t * [(1-h[i+lag]) * 1_{0<=i+lag<n} - h[i-lag] * 1_{0<=i-lag<n}]
    #   - t * [(1-h[j+lag]) * 1_{0<=j+lag<n} - h[j-lag] * 1_{0<=j-lag<n}]
    
    lags = np.arange(2*n-1) - center
    coeff = np.zeros(2*n-1)
    for r in range(2*n-1):
        lag = lags[r]
        ci = 0.0
        if 0 <= i + lag < n:
            ci += (1.0 - h[i+lag])
        if 0 <= i - lag < n:
            ci -= h[i-lag]
        cj = 0.0
        if 0 <= j + lag < n:
            cj += (1.0 - h[j+lag])
        if 0 <= j - lag < n:
            cj -= h[j-lag]
        coeff[r] = ci - cj
    
    # For r = center (lag=0):
    # c_0(t) = c_0(0) + 2t(h[j]-h[i]) - 2t^2
    # But coeff[center] should capture the linear part
    # Let's verify: for lag=0, ci = (1-h[i]) - h[i] = 1-2h[i], cj = 1-2h[j]
    # coeff[center] = (1-2h[i]) - (1-2h[j]) = 2(h[j]-h[i]) ✓
    # And the quadratic term is -2t^2 for the h[i] part and -2t^2 for h[j] part? 
    # Wait no. Let me recompute.
    # c_0 = sum_m h[m](1-h[m])
    # Change: (a+t)(1-a-t) + (b-t)(1-b+t) - a(1-a) - b(1-b)
    # = [a(1-a) + t(1-2a) - t^2] + [b(1-b) - t(1-2b) - t^2] - a(1-a) - b(1-b)
    # = t(1-2a) - t^2 - t(1-2b) - t^2
    # = t(2b-2a) - 2t^2
    # So the quadratic term is -2t^2 (only for lag 0)
    
    # For the objective: C5(t) = dx * max_r [corr0[r] + t*coeff[r] - 2t^2*1_{r=center}]
    # We want to minimize this over t in [t_min, t_max]
    
    # Grid search
    ts = np.linspace(t_min, t_max, n_grid)
    best_t = 0.0
    best_val = float('inf')
    
    # Vectorized: for each t, compute max corr
    # corr(t) = corr0 + t*coeff - 2t^2 * e_center
    # where e_center has 1 at position center, 0 elsewhere
    e_center = np.zeros(2*n-1)
    e_center[center] = 1.0
    
    # For efficiency, compute in batches
    batch_size = 5000
    for start in range(0, len(ts), batch_size):
        t_batch = ts[start:start+batch_size]
        # (batch, 2n-1) array
        corr_batch = corr0[np.newaxis, :] + t_batch[:, np.newaxis] * coeff[np.newaxis, :] \
                     - 2*t_batch[:, np.newaxis]**2 * e_center[np.newaxis, :]
        vals = dx * np.max(corr_batch, axis=1)
        for idx in range(len(t_batch)):
            if vals[idx] < best_val:
                best_val = vals[idx]
                best_t = t_batch[idx]
    
    if best_val < float('inf'):
        h_new = h.copy()
        h_new[i] = a + best_t
        h_new[j] = b - best_t
        h_new = ensure_feasible(h_new, n)
        return h_new, best_val
    return h, None
