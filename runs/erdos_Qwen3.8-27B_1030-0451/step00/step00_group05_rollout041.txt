    dx = 2.0 / n
    u = 1.0 - h
    corr = np.correlate(h, u, mode="full") * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    """Project h onto the feasible set: 0 <= h[i] <= 1, sum(h) = target_sum."""
    h = np.array(h, dtype=float)
    n = len(h)
    
    # Iterative projection (Dykstra's algorithm or simple clipping + rescaling)
    for _ in range(50):
        # Clip to [0, 1]
        h_clipped = np.clip(h, 0.0, 1.0)
        # Check sum
        s = h_clipped.sum()
        if abs(s - target_sum) < 1e-12:
            return h_clipped
        # Simple rescaling won't work well with box constraints
        # Use a different approach: find alpha such that sum(clip(h + alpha, 0, 1)) = target_sum
        # This is a 1D root finding problem
        lo = -h.max() - 1
        hi = 1 - h.min() + 1
        for _ in range(60):
            mid = (lo + hi) / 2
            h_test = np.clip(h + mid, 0.0, 1.0)
            s_test = h_test.sum()
            if s_test < target_sum:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    corr = np.correlate(h, u, mode="full") * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    n = len(h)
    
    for _ in range(30):
        # Find alpha such that sum(clip(h + alpha, 0, 1)) = target_sum
        lo = -h.max() - 0.1
        hi = 1.0 - h.min() + 0.1
        for _ in range(50):
            mid = (lo + hi) / 2.0
            h_test = np.clip(h + mid, 0.0, 1.0)
            s_test = h_test.sum()
            if s_test < target_sum:
                lo = mid
            else:
                hi = mid
        
        h_new = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
        if np.max(np.abs(h_new - h)) < 1e-14:
            h = h_new
            break
        h = h_new
    
    return h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feasible(h_old, n_new / 2.0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def compute_smoothed_gradient(h, dx, temperature=0.003, active_width=0.008, max_active=20):
    """Compute gradient of the smoothed max objective."""
    n = len(h)
    u = 1.0 - h
    corr = np.correlate(h, u, mode="full") * dx
    M = np.max(corr)
    
    # Find active indices: those within active_width of max
    active_mask = corr >= M - active_width
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) > max_active:
        # Take top max_active
        active_vals = corr[active_idx]
        top_k = np.argsort(active_vals)[-max_active:]
        active_idx = active_idx[top_k]
    
    if len(active_idx) == 0:
        return np.zeros(n), 0.0
    
    # Compute softmax weights
    diffs = np.maximum(0.0, M - corr[active_idx])
    weights = np.exp(-diffs / temperature)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(active_idx)) / len(active_idx)
    else:
        weights = weights / w_sum
    
    grad = np.zeros(n)
    c0 = n - 1
    
    for ai, idx in enumerate(active_idx):
        w = weights[ai]
        R = idx - c0
        
        if R == 0:
            grad[:] += w * dx * (1.0 - 2.0 * h)
        elif R > 0:
            nterms = n - R
            # For j in [0, nterms-1]: contribution from (1-h[j]) term is -h[j+R]
            grad[:nterms] += w * dx * (-h[R:])
            # For j in [R, n-1]: contribution from h[j] term (where j = i+R, i=j-R) is (1-h[j-R])
            grad[R:] += w * dx * (1.0 - h[:nterms])
        else:
            S = -R
            nterms = n - S
            # For j in [0, nterms-1]: contribution from h[j] term is (1-h[j+S])
            grad[:nterms] += w * dx * (1.0 - h[S:])
            # For j in [S, n-1]: contribution from (1-h[j]) term (where j = i+S, i=j-S) is -h[j-S]
            grad[S:] += w * dx * (-h[:nterms])
    
    return grad, M

For τ < 0, let τ = -S: sum_{i=S}^{n-1} h[i] * (1-h[i-S]) = sum_{j=0}^{n-1-S} h[j+S] * (1-h[j])

So:
- Positive lag R = τ > 0: G[c0+R] = dx * sum_{i=0}^{n-1-R} h[i] * (1-h[i+R])
- Negative lag (τ = -S, S > 0): G[c0-S] = dx * sum_{i=0}^{n-1-S} h[i+S]
