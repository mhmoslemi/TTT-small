import numpy as np
import time
from scipy.optimize import linprog

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(z) * dx)

def exact_z_vector(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full") * dx
    return z

def project_box_sum(x, target_sum):
    """Project x onto {y: 0<=y[i]<=1, sum(y)=target_sum} using bisection."""
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    
    def f(alpha):
        return np.sum(np.clip(x + alpha, 0.0, 1.0))
    
    # Find bracket
    lo, hi = -x.max() - 1, 1 - x.min() + 1
    flo, fhi = f(lo), f(hi)
    
    if flo > target_sum:
        return np.clip(x + lo, 0.0, 1.0)
    if fhi < target_sum:
        return np.clip(x + hi, 0.0, 1.0)
    
    for _ in range(100):
        mid = (lo + hi) / 2
        fm = f(mid)
        if fm < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    
    result = np.clip(x + (lo + hi) / 2, 0.0, 1.0)
    
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute to elements not at bounds
        if residual > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        if np.sum(mask) > 0:
            n_target = int(np.sum(mask))
            result[mask] += residual / n_target
            result = np.clip(result, 0.0, 1.0)
    
    return result

def grad_z(h, q):
    """Gradient of the q-th correlation value (scaled by dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    ell = q - (n - 1)
    g = np.zeros(n)
    
    if ell == 0:
        g = (1.0 - 2.0 * h) * dx
    elif ell > 0:
        L = n - ell
        # For i in [0, L-1]: g[i] += dx * (1 - h[i+ell])
        g[0:L] += dx * (1.0 - h[ell:n])
        # For i in [ell, n-1]: g[i] -= dx * h[i-ell]
        g[ell:n] -= dx * h[0:L]
    else:
        a = -ell
        L = n - a
        # For i in [a, n-1]: g[i] += dx * (1 - h[i-a])
        g[a:n] += dx * (1.0 - h[0:L])
        # For i in [0, L-1]: g[i] -= dx * h[i+a]
        g[0:L] -= dx * h[a:n]
    
    return g

def slp_local_search(h0, time_limit, rng, target_c=0.3808):
    """Sequential Linear Programming local search."""
    n = len(h0)
    h = h0.copy()
    current_c = exact_score(h)
    best_h = h.copy()
    best_c = current_c
    
    rho = max(0.01, 0.03 * n / 250)  # trust region radius
    min_rho = 1e-5
    stall = 0
    max_stall = 30
    max_iter = 500
    
    t_start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Compute current z vector and active set
        zvec = exact_z_vector(h)
        z_max = np.max(zvec)
        current_c = z_max
        
        # Select active constraints
        gap = max(1e-4, 0.01 * current_c)
        active = np.where(zvec >= z_max - gap)[0]
        
        # Cap active set
        K_max = min(len(active), 40 if n <= 300 else 20)
        if len(active) > K_max:
            order = np.argsort(-zvec[active])[:K_max]
            active = active[order]
        
        # Build LP
        m = len(active)
        # Variables: h[0..n-1], t  => n+1 variables, t at index n
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        # A_ub: active constraints
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        
        for idx, q in enumerate(active):
            g = grad_z(h, q)
            zq = zvec[q]  # this is already scaled by dx
            # Linearized: g · h_new - t <= g · h - zq
            A_ub[idx, :n] = g
            A_ub[idx, n] = -1.0
            b_ub[idx] = np.dot(g, h) - zq
        
        # Bounds
        bounds = [(max(0.0, h[i] - rho), min(1.0, h[i] + rho)) for i in range(n)]
        bounds.append((0.0, max(0.99, current_c + 0.1)))
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [n / 2.0]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success or res.x is None:
                rho *= 0.5
                if rho < min_rho:
                    break
                continue
        except Exception:
            rho *= 0.5
            if rho < min_rho:
                break
            continue
        
        h_new = np.asarray(res.x[:n], dtype=np.float64)
        
        # Check feasibility
        if not np.all(h_new >= -1e-9) or not np.all(h_new <= 1 + 1e-9):
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_box_sum(h_new, n / 2.0)
        
        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_cand = (1 - alpha) * h + alpha * h_new
            h_cand = np.clip(h_cand, 0.0, 1.0)
            h_cand = project_box_sum(h_cand, n / 2.0)
            c_cand = exact_score(h_cand)
            if c_cand < current_c - 1e-12:
                h = h_cand
                current_c = c_cand
                improved = True
                break
        
        if improved:
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
            stall = 0
            rho = min(rho * 1.1, 0.05)
        else:
            stall += 1
            rho *= 0.5
            if rho < min_rho or stall > max_stall:
                break
    
    return best_h, best_c

        # This is good if g[i] < g[j] (reduces g·h)

def upscale(h, factor):

import numpy as np
import time
from scipy.optimize import linprog


def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(z) * dx)


def exact_z_vector(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_box_sum(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x

    def f(alpha):
        return np.sum(np.clip(x + alpha, 0.0, 1.0))

    lo, hi = -x.max() - 2.0, 2.0 - x.min()
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if f(mid) < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break

    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        if np.any(mask):
            result[mask] += residual / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result


def grad_z(h, q):
    n = len(h)
    dx = 2.0 / n
    ell = q - (n - 1)
    g = np.zeros(n)
    if ell == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif ell > 0:
        L = n - ell
        if L > 0:
            g[0:L] += dx * (1.0 - h[ell:n])
            g[ell:n] -= dx * h[0:L]
    else:
        a = -ell
        L = n - a
        if L > 0:
            g[a:n] += dx * (1.0 - h[0:L])
            g[0:L] -= dx * h[a:n]
    return g


def slp_local_search(h0, time_limit, rng, target_c=0.3808):
    n = len(h0)
    h = h0.copy()
    current_c = exact_score(h)
    best_h = h.copy()
    best_c = current_c

    rho = 0.02
    min_rho = 5e-6
    stall = 0
    max_stall = 40
    max_iter = 2000
    t_start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t_start > time_limit:
            break

        zvec = exact_z_vector(h)
        z_max = np.max(zvec)
        current_c = z_max

        gap = max(2e-4, 0.008 * current_c)
        active_mask = zvec >= z_max - gap
        active = np.where(active_mask)[0]

        K_max = min(len(active), 50 if n <= 300 else 25)
        if len(active) > K_max:
            order = np.argsort(-zvec[active])[:K_max]
            active = active[order]

        m = len(active)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)

        for idx, q in enumerate(active):
            g = grad_z(h, q)
            zq = zvec[q]
            A_ub[idx, :n] = g
            A_ub[idx, n] = -1.0
            b_ub[idx] = np.dot(g, h) - zq

        bounds = []
        for i in range(n):
            lo_b = max(0.0, h[i] - rho)
            hi_b = min(1.0, h[i] + rho)
            if lo_b > hi_b:
                lo_b, hi_b = h[i], h[i]
            bounds.append((lo_b, hi_b))
        bounds.append((0.0, current_c + 0.5))

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success or res.x is None:
                rho *= 0.5
                if rho < min_rho:
                    break
                continue
        except Exception:
            rho *= 0.5
            if rho < min_rho:
                break
            continue

        h_new = np.asarray(res.x[:n], dtype=np.float64)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_box_sum(h_new, n / 2.0)

        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.03]:
            h_cand = (1.0 - alpha) * h + alpha * h_new
            h_cand = np.clip(h_cand, 0.0, 1.0)
            h_cand = project_box_sum(h_cand, n / 2.0)
            c_cand = exact_score(h_cand)
            if c_cand < current_c - 1e-13:
                h = h_cand
                current_c = c_cand
                improved = True
                break

        if improved:
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
            stall = 0
            if rho < 0.05:
                rho *= 1.15
        else:
            stall += 1
            rho *= 0.55
            if rho < min_rho or stall > max_stall:
                break

    return best_h, best_c


def pairwise_transfer(h0, time_limit, rng):
    n = len(h0)
    h = h0.copy()
    current_c = exact_score(h)
    best_h = h.copy()
    best_c = current_c
    t_start = time.monotonic()
    max_rounds = 500

    for rnd in range(max_rounds):
        if time.monotonic() - t_start > time_limit:
            break

        zvec = exact_z_vector(h)
        z_max = np.max(zvec)
        gap = max(2e-4, 0.01 * current_c)
        active = np.where(zvec >= z_max - gap)[0]
        if len(active) == 0:
            active = np.array([np.argmax(zvec)])

        weights = np.exp(-(zvec[active] - z_max) * 30.0)
        weights /= np.sum(weights)

        g_mix = np.zeros(n)
        for idx, q in enumerate(active):
            g_mix += weights[idx] * grad_z(h, q)

        order = np.argsort(g_mix)
        n_try = min(30, n)
        sources = order[-n_try:]
        targets = order[:n_try]

        found = False
        for s in sources:
            if found:
                break
            for t_idx in targets:
                if s == t_idx:
                    continue
                if h[s] < 1e-12 or h[t_idx] > 1.0 - 1e-12:
                    continue
                max_d = min(h[s], 1.0 - h[t_idx])
                if max_d < 1e-12:
                    continue
                best_d = None
                best_cc = current_c
                for frac in [0.05, 0.1, 0.2, 0.35, 0.55, 0.8, 1.0]:
                    d = frac * max_d
                    h[s] -= d
                    h[t_idx] += d
                    cc = exact_score(h)
                    if cc < best_cc:
                        best_cc = cc
                        best_d = d
                    else:
                        h[s] += d
                        h[t_idx] -= d
                if best_d is not None:
                    h[s] -= best_d
                    h[t_idx] += best_d
                    current_c = best_cc
                    found = True
                    break

        if found:
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            found_rand = False
            for _ in range(10):
                s = int(rng.integers(n))
                t_idx = int(rng.integers(n))
                if s == t_idx or h[s] < 1e-12 or h[t_idx] > 1.0 - 1e-12:
                    continue
                max_d = min(h[s], 1.0 - h[t_idx])
                if max_d < 1e-12:
                    continue
                d = 0.1 * max_d * rng.uniform(0.5, 2.0)
                d = min(d, max_d)
                h[s] -= d
                h[t_idx] += d
                cc = exact_score(h)
                if cc < current_c - 1e-13:
                    current_c = cc
                    found_rand = True
                    if current_c < best_c:
                        best_c = current_c
                        best_h = h.copy()
                    break
                h[s] += d
                h[t_idx] -= d
            if not found_rand and not found:
                break

    return best_h, best_c


def upscale_interp(h, factor):
    n = len(h)
    n_new = n * factor
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, n_new / 2.0)
