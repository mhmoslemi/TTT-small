import numpy as np
import time

def project_onto_simplex(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h_i) = target_sum}"""
    n = len(h)
    lo, hi = np.min(h) - 1, np.max(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, 'full') * dx"""
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

import numpy as np

def project_simplex(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def c5_value(h):
    """Compute C5 for a given h vector."""
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def smooth_objective_grad(h, temp):
    """Compute log-sum-exp smooth max and its gradient."""
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, 'full') * dx
    f_max = np.max(f)
    shifted = temp * (f - f_max)
    exp_f = np.exp(shifted)
    denom = np.sum(exp_f)
    p = exp_f / denom
    obj = (f_max + np.log(denom)) / temp
    
    # Gradient computation using convolutions
    # grad[j] = sum_k p[k]*(1-h[j-k])*I(0<=j-k<n) - sum_k p[k]*h[j+k]*I(0<=j+k<n)
    # First term = conv(1-h, p)[:n]
    A = np.convolve(1.0 - h, p, 'full')[:n]
    # Second term: B[j] = sum_{k=0}^{n-1-j} p[k]*h[j+k]
    # B[j] = conv(p, h[::-1])[n-1-j]
    B_full = np.convolve(p, h[::-1], 'full')
    B = B_full[n-1::-1]
    
    return A - B, obj

def c5_value(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def smooth_grad_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, 'full') * dx
    f_max = f.max()

    obj = (f_max + np.log(w_sum)) / temp

    B_full = np.convolve(p, h[::-1], 'full')
    B = B_full[n - 1::-1]
    return A - B, obj
