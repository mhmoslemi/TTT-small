def transfer_resolution(h_src, N_dst):
    """Transfer a solution from N_src to N_dst resolution via interpolation."""
    N_src = len(h_src)
    x_src = np.linspace(0, 2, N_src + 1)  # cell boundaries
    x_dst = np.linspace(0, 2, N_dst + 1)
    
    # Piecewise constant: h_src[i] is the value on [2*i/N_src, 2*(i+1)/N_src)
    # For dst cell j: [2*j/N_dst, 2*(j+1)/N_dst)
    target_dst = N_dst / 2.0
    
    h_dst = np.zeros(N_dst)
    for j in range(N_dst):
        left = 2.0 * j / N_dst
        right = 2.0 * (j + 1) / N_dst
        # Which source cells overlap?
        total_area = 0.0
        cell_len = 2.0 / N_src
        i_start = int(left // cell_len)
        i_end = int(right // cell_len) + 1
        i_start = max(0, i_start)
        i_end = min(N_src, i_end)
        for i in range(i_start, i_end):
            s_left = max(left, i * cell_len)
            s_right = min(right, (i + 1) * cell_len)
            if s_right > s_left:
                total_area += h_src[i] * (s_right - s_left)
        h_dst[j] = total_area / (right - left)
    
    # Actually this should give integral preservation automatically
    # But let's project to be safe
    h_dst = project_mass(h_dst, target_dst)

def exact_metric(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_mass(h, target):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = math.fsum(z) - target
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(z)[::-1]
            rem = residual
            for i in idx:
                dec = min(z[i], rem)
                z[i] -= dec
                rem -= dec
                if rem <= 1e-14:
                    break
        else:
            rem = -residual
            idx = np.argsort(z)
            for i in idx:
                inc = min(1.0 - z[i], rem)
                z[i] += inc
                rem -= inc
                if rem <= 1e-14:
                    break
    return z


def binarize(h):

    target = N // 2
    sorted_idx = np.argsort(-h)
    bits = np.zeros(N)
    bits[sorted_idx[:target]] = 1.0
    return bits

def transfer_resolution(h_src, N_dst):
    N_src = len(h_src)
    cell_src = 2.0 / N_src
    h_dst = np.zeros(N_dst)
    for j in range(N_dst):
        left = 2.0 * j / N_dst
        right = 2.0 * (j + 1) / N_dst
        total = 0.0
        i_start = max(0, int(math.floor(left / cell_src)))
        i_end = min(N_src, int(math.ceil(right / cell_src)))
        for i in range(i_start, i_end):
            s_l = max(left, i * cell_src)
            s_r = min(right, (i + 1) * cell_src)
            if s_r > s_l:
                total += h_src[i] * (s_r - s_l)
        h_dst[j] = total / (right - left)
    return project_mass(h_dst, N_dst / 2.0)

            best_val = v_t

    # Phase 4: Try higher resolutions
    for N_try in [160, 200, 240]:
        remaining = deadline - time.time()
        if remaining < 40:
            break
        if N_try >= 2500:
            break
        h_try = transfer_resolution(best_h, N_try)
        v_try = exact_metric(h_try)

        t_bin = min(80.0, remaining * 0.2)
        if t_bin > 5:
            bits = binarize(h_try)
            b_b, b_v = binary_search(bits, t_bin, rng)
            if b_v < v_try:
                h_try = b_b
                v_try = b_v

        remaining = deadline - time.time()
        t_cont = min(80.0, remaining * 0.2)
        if t_cont > 5:
            h_p, v_p = continuous_search(h_try, t_cont, rng)
            if v_p < v_try:
                h_try = h_p
                v_try = v_p

        if v_try < best_val - 1e-12:
            best_h = h_try
            best_val = v_try

    # Phase 5: Final binary polish on best
    remaining = deadline - time.time()
    if remaining > 20:
        t_final = min(150.0, remaining * 0.5)
        bits = binarize(best_h)
        b_best, b_val = binary_search(bits, t_final, rng)
        if b_val < best_val - 1e-12:
            best_h = b_best
            best_val = b_val

    # Phase 6: Final continuous polish
    remaining = deadline - time.time()
    if remaining > 20:
        t_chemp = min(100.0, remaining * 0.5)
        h_f, v_f = continuous_search(best_h, t_chemp, rng)
        if v_f < best_val - 1e-12:
            best_h = h_f
            best_val = v_f

    # Final validation
    best_h = np.asarray(best_h, dtype=np.float64)
    N_out = len(best_h)
    target = N_out / 2.0
    best_h = project_mass(best_h, target)
    final_val = exact_metric(best_h)

    return best_h, final_val, N_out
