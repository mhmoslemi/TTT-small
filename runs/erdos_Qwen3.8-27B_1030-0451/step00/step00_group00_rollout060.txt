import numpy as np

def project_onto_feasible(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target} using bisection."""
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def compute_grad(h, weights, n, dx):
    """Compute gradient of weighted sum of correlation values."""
    W_arr = weights[::-1]
    # A[j] = sum of W_arr[j : j+n]
    W_cumsum = np.zeros(2 * n)  # W_arr has length 2n-1, cumsum has length 2n

    A = W_cumsum[n:] - W_cumsum[:n]  # A[j] for j=0..n-1

        if s > target:
            lo = mid

    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def grad_full(h, weights, dx):
    n = len(h)
    W_arr = weights[::-1]
    W_cs = np.zeros(len(W_arr) + 1)

    A = W_cs[n:] - W_cs[:n]

    C = np.correlate(h, W_arr, mode='full')[n-1:2*n-1]

                grad[Lp:] += (1.0 - h[:n-Lp]) * dx
            if n - 1 - Lp >= 0:

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    n = len(h0)

    best_c5 = c5_of(h, dx)
    best_h = h.copy()
    
    rng = np.random.RandomState(seed)
    eta = 0.02
    mom = np.zeros(n)
    beta = 0.9
    stagnation = 0
    
    while time.time() - t0 < budget_s - 3:
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cur = float(np.max(corr))
        
        if cur < best_c5 - 1e-12:
            best_c5 = cur
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Reset if stuck
        if stagnation > 300:
            h = h + rng.normal(0, 0.15, n)
            h = project_feasible(h, target)
            eta = 0.02
            mom = np.zeros(n)
            stagnation = 0
            continue

        # Use top-30 with exponential weights
        top_m = min(30, 2*n-1)
        top_idx = np.argsort(corr)[-top_m:]
        top_vals = corr[top_idx]
        # softmax weights
        w = np.exp((top_vals - top_vals.max()) / 0.01)
        w /= w.sum()

        weights = np.zeros(2*n-1)
        for i, idx in enumerate(top_idx):
            weights[idx] = w[i]

        grad = grad_via_conv(h, weights, dx)

        mom = beta * mom + (1 - beta) * grad

        h_new = project_feasible(h_new, target)
        new_c5 = c5_of(h_new, dx)

            h = h_new
            eta = min(eta * 1.01, 0.1)
        else:
            eta *= 0.5
            mom = beta * mom  # keep momentum but don't take the step
            if eta < 1e-6:
                stagnation += 100
