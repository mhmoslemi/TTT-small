def mass_project(x, target):
    """Project x to [0,1]^N with sum = target using bisection on shift."""
    x = np.asarray(x, dtype=np.float64).copy()
    N = len(x)
    # Replace non-finite
    x[~np.isfinite(x)] = 0.5
    
    def F(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0))
    
    # Find bracket
    lo = -2.0
    hi = 2.0
    while F(lo) > target:
        lo -= 1.0
    while F(hi) < target:
        hi += 1.0
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if F(mid) < target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fine adjustment
    residual = math.fsum(result) - target
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to reduce
            order = np.argsort(-result)
            i = 0
            while residual > 1e-12 and i < N:
                reduce = min(result[order[i]], residual)
                result[order[i]] -= reduce
                residual -= reduce
                i += 1
        else:
            # Need to increase
            order = np.argsort(result)
            i = 0
            while residual < -1e-12 and i < N:
                add = min(1.0 - result[order[i]], -residual)
                result[order[i]] += add
                residual += add
                i += 1
    
    return result

    N = len(h)
    corr = objective_array(h)
    max_val = np.max(corr)
    
    # Find active lags (within tolerance of max)
    tol = 1e-4 + 1e-6 * max_val
    active = np.where(corr >= max_val - tol)[0]
    
    # Compute gradient as weighted average
    grad = np.zeros(N)

        s = k - (N - 1)  # lag
        # L(s,h) = dx * sum_{i} h[i] * (1-h[i+s]) for valid i
        # dL/dh[j]:
        for i in range(N):
            j = i

            if s == 0:

                    grad[j] += dx * (1.0 - h[right])
                    # j = right, appears as right factor
                    grad[right] += -dx * h[i]
    
    # Normalize by number of active lags
    grad /= len(active)
    return grad, max_val

def transfer_to_resolution(h_src, n_dst):
    """Transfer a piecewise constant function from n_src to n_dst resolution."""
    N_src = len(h_src)
    dx_src = 2.0 / N_src
    dx_dst = 2.0 / n_dst
    
    # For each destination cell, compute the average value from source
    h_dst = np.zeros(n_dst)
    for i in range(n_dst):
        # Destination cell [i*dx_dst, (i+1)*dx_dst]
        x_start = i * dx_dst
        x_end = (i + 1) * dx_dst
        # Find overlapping source cells
        src_start = int(math.floor(x_start / dx_src))
        src_end = int(math.ceil(x_end / dx_src))
        src_start = max(0, src_start)
        src_end = min(N_src, src_end)
        
        total_area = 0.0
        for j in range(src_start, src_end):
            s_start = j * dx_src
            s_end = (j + 1) * dx_src
            overlap_start = max(x_start, s_start)
            overlap_end = min(x_end, s_end)
            if overlap_end > overlap_start:
                total_area += h_src[j] * (overlap_end - overlap_start)
        
        h_dst[i] = total_area / dx_dst
    
    return h_dst

def binary_local_search(h, max_iters, seed, deadline):

    rng = np.random.default_rng(seed)

        zeros_idx = np.where(h == 0.0)[0]

            h[i] = 0.0
            h[j] = 1.0

                current_val = val
                improved = True
                if val < best_val:
                    best_h = h.copy()
                    best_val = val
                break

                h[i] = 1.0
                h[j] = 0.0
        
        if not improved:
            no_improve += 1
            if no_improve > 30:

                fo = rng.choice(ones_idx, size=min(n, len(ones_idx)), replace=False)
                fz = rng.choice(zeros_idx, size=min(n, len(zeros_idx)), replace=False)
                h[fo] = 0.0
                h[fz] = 1.0
                current_val = evaluate_exact(h)
                no_improve = 0
                if current_val < best_val:
                    best_h = h.copy()
                    best_val = current_val
    
    return best_h, best_val
