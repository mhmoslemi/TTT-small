import numpy as np
import time

def eval_c5_from_corr(corr, n):
    return np.max(corr) * (2.0 / n)

def get_delta_vec(h, p, n):
    """Compute v_p[m] = h[p+k] + h[p-k] for all m, where k = m-(n-1)."""
    k = np.arange(-(n-1), n)  # length 2n-1
    idx_plus = p + k
    idx_minus = p - k
    v = np.zeros(2*n-1, dtype=np.int8)
    mask_p = (idx_plus >= 0) & (idx_plus < n)
    mask_m = (idx_minus >= 0) & (idx_minus < n)
    v[mask_p] += h[idx_plus[mask_p]]
    v[mask_m] += h[idx_minus[mask_m]]
    return v

def optimize_binary_local(h, n, rng, time_budget=30):
    """Local search + simulated annealing on binary sequence."""
    h = h.copy()
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    start = time.time()
    
    temperature = 2.0  # initial temperature (in units of corr, which are integers)
    cooling = 0.999
    
    n_stuck = 0
    
    while time.time() - start < time_budget:
        # Find hot shifts
        max_val = np.max(corr)
        hot_shifts = np.where(corr >= max_val - 1)[0]  # shifts within 1 of max
        
        # Pick a random hot shift
        m_star = hot_shifts[rng.randint(len(hot_shifts))]
        k_star = m_star - (n - 1)
        
        # Find candidate i (position of 1) with low "potential"
        # potential_i = h[i+k*] + h[i-k*]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Sample some candidates
        n_sample = min(100, len(ones))
        i_cands = ones[rng.choice(len(ones), size=n_sample, replace=False)]
        j_cands = zeros[rng.choice(len(zeros), size=n_sample, replace=False)]
        
        # Compute potentials
        i_pot = np.array([int(h[i+k_star]) if 0<=i+k_star<n else 0 + int(h[i-k_star]) if 0<=i-k_star<n else 0 for i in i_cands])
        j_pot = np.array([int(h[j+k_star]) if 0<=j+k_star<n else 0 + int(h[j-k_star]) if 0<=j-k_star<n else 0 for j in j_cands])
        
        # Want i with low potential, j with high potential
        i_best = i_cands[np.argmin(i_pot)]
        j_best = j_cands[np.argmax(j_pot)]
        
        # Evaluate the swap
        delta = get_delta_vec(h, i_best, n) - get_delta_vec(h, j_best, n)
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        if new_max <= current_max:
            # Accept
            h[i_best] = 0
            h[j_best] = 1
            corr = new_corr
            current_max = new_max
            n_stuck = 0
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:

            if rng.rand() < np.exp(-(new_max - current_max) / temperature):

                h[j_best] = 1
                corr = new_corr
                current_max = new_max
                n_stuck = 0
            else:
                n_stuck += 1
        
        temperature *= cooling

import numpy as np
import time

def random_binary(n, rng):
    """Create random binary sequence with n//2 ones."""
    h = np.zeros(n, dtype=np.int8)
    positions = rng.choice(n, size=n//2, replace=False)
    h[positions] = 1
    return h

def init_from_fractional(frac_vals, n, rng):
    """Resample fractional values to length n and round to binary."""
    # Resample using interpolation
    old_n = len(frac_vals)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, n, endpoint=False)
    resampled = np.interp(new_x, old_x, frac_vals)
    # Round to binary with correct sum
    target = n // 2
    # Use a threshold approach
    sorted_idx = np.argsort(resampled)[::-1]
    h = np.zeros(n, dtype=np.int8)
    h[sorted_idx[:target]] = 1
    return h

def compute_corr(h):
    """Compute full correlation of h and 1-h."""
    hf = h.astype(np.float64)
    return np.correlate(hf, 1.0 - hf, mode='full')

def eval_binary_c5(h):
    n = len(h)
    corr = compute_corr(h)
    return float(np.max(corr)) * (2.0 / n)

def delta_for_swap(h, i, j, n):
    """Compute the change in corr when swapping h[i]=1->0 and h[j]=0->1."""
    k = np.arange(-(n-1), n)
    # v_i[m] = h[i+k] + h[i-k] (0 for OOB)
    # v_j[m] = h[j+k] + h[j-k] (0 for OOB)
    # delta = v_i - v_j
    idx_ip = i + k
    idx_im = i - k
    idx_jp = j + k
    idx_jm = j - k
    
    delta = np.zeros(2*n-1, dtype=np.int8)

    delta[mask] += h[idx_ip[mask]]
    mask = (idx_im >= 0) & (idx_im < n)
    delta[mask] += h[idx_im[mask]]
    mask = (idx_jp >= 0) & (idx_jp < n)
    delta[mask] -= h[idx_jp[mask]]
    mask = (idx_jm >= 0) & (idx_jm < n)
    delta[mask] -= h[idx_jm[mask]]
    
    return delta
