    diff = c - result.sum()
    if abs(diff) > 1e-13:
        mask = (result > 1e-14) & (result < 1.0 - 1e-14)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def make_harmonic_init(n, num_harmonics=6, seed=0):
    rng = np.random.RandomState(seed)
    x = np.arange(n) * 2.0 / n
    h = np.full(n, 0.5)
    amplitudes = 1.0 / np.sqrt(np.arange(1, num_harmonics + 1, dtype=float))
    phases = rng.uniform(0, 2 * np.pi, num_harmonics)
    for m in range(num_harmonics):
        h += 0.3 * amplitudes[m] * np.cos(np.pi * (m + 1) * x + phases[m])
    return project_polytope(h, n / 2.0)

def subgradient_optimize(h, n, max_iters, time_limit_fn):
    alpha = 1.0 / np.sqrt(n)
    best_local = compute_c5(h)
    best_h_local = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    for it in range(max_iters):
        if time_limit_fn():
            break
        kmax = int(np.argmax(corr))
        shift = kmax - (n - 1)
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n - shift] += 1.0 - h[shift:]
            grad[shift:] -= h[:n - shift]
        else:
            grad[-shift:] += 1.0 - h[:n + shift]
            grad[:n + shift] -= h[-shift:]
        h = project_polytope(h - alpha * grad, n / 2.0)
        alpha *= 0.99995
        if (it + 1) % 2000 == 0:
            c = compute_c5(h)
            if c < best_local:
                best_local = c
                best_h_local = h.copy()
            corr = np.correlate(h, 1.0 - h, mode="full")
    c_final = compute_c5(h)
    if c_final < best_local:
        best_local = c_final
        best_h_local = h.copy()
    return best_h_local, best_local

def binary_swap_search(h, n, max_iters, time_limit_fn):
    hb = (h > 0.5).astype(np.float64)
    cnt = int(hb.sum())
    if cnt != n // 2:
        ones_i = np.where(hb == 1.0)[0]
        zeros_i = np.where(hb == 0.0)[0]
        np.random.shuffle(ones_i)
        np.random.shuffle(zeros_i)
        if cnt > n // 2:
            for i in ones_i[:cnt - n // 2]:
                hb[i] = 0.0
        else:
            for i in zeros_i[:n // 2 - cnt]:
                hb[i] = 1.0
    cb = compute_c5(hb)
    best_b = cb
    best_hb = hb.copy()
    no_improve = 0
    for it in range(max_iters):
        if time_limit_fn():
            break
        ones_i = np.where(hb == 1.0)[0]
        zeros_i = np.where(hb == 0.0)[0]
        a = int(ones_i[np.random.randint(len(ones_i))])
        b = int(zeros_i[np.random.randint(len(zeros_i))])
        hb[a] = 0.0
        hb[b] = 1.0
        c_new = compute_c5(hb)
        if c_new < cb - 1e-12:
            cb = c_new
            no_improve = 0
            if cb < best_b:
                best_b = cb
                best_hb = hb.copy()
        else:
            hb[a] = 1.0
            hb[b] = 0.0
            no_improve += 1
            if no_improve > 3000:
                break
    return best_hb, best_b

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0

    def exceed(frac):
        return (time.time() - t0) > budget_s * frac

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        c0 = compute_c5(init)
        best_c5, best_h = c0, init.copy()
    except Exception:
        pass

    for n in [128, 256, 512]:
        if exceed(0.5):
            break
        if best_h is not None and len(best_h) == n:
            h = best_h.copy()
        elif best_h is not None:
            h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(best_h)), best_h)
        else:
            h = make_harmonic_init(n, 6, seed)
        h = project_polytope(h, n / 2.0)
        c = compute_c5(h)
        if c < best_c5:
            best_c5, best_h = c, h.copy()

        h_opt, c_opt = subgradient_optimize(h, n, 60000, lambda: exceed(0.52))
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()

        if not exceed(0.6):
            try:
                res = minimize(compute_c5, h_opt, method='SLSQP',
                             bounds=[(0.0, 1.0)] * n,
                             constraints=[{'type': 'eq',
                                           'fun': lambda x: np.sum(x) - n / 2.0,
                                           'jac': lambda x: np.ones(n)}],
                             options={'maxiter': 400, 'ftol': 1e-12})
                hp = project_polytope(res.x, n / 2.0)
                cp = compute_c5(hp)
                if cp < best_c5:
                    best_c5, best_h = cp, hp.copy()
            except Exception:
                pass
        h_opt2, c_opt2 = subgradient_optimize(hp if not exceed(0.6) else h_opt, n, 30000, lambda: exceed(0.62))
        if c_opt2 < best_c5:
            best_c5, best_h = c_opt2, h_opt2.copy()

    if not exceed(0.82):
        for n in [128, 256]:
            if exceed(0.82):
                break
            if best_h is not None and len(best_h) == n:
                hs = best_h.copy()
            elif best_h is not None:
                hs = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(best_h)), best_h)
            else:
                hs = np.zeros(n)
                hs[:n // 2] = 1.0
            hb, cb = binary_swap_search(hs, n, 100000, lambda: exceed(0.82))
            if cb < best_c5:
                best_c5, best_h = cb, hb.copy()

    if best_h is not None and not exceed(0.95):
        n = len(best_h)
        try:
            res = minimize(compute_c5, best_h, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq',
                                       'fun': lambda x: np.sum(x) - n / 2.0,
                                       'jac': lambda x: np.ones(n)}],
                         options={'maxiter': 200, 'ftol': 1e-13})
            hp = project_polytope(res.x, n / 2.0)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5, best_h = cp, hp.copy()
        except Exception:
            pass

    if best_h is None:
        n = 128
        x = np.arange(n) * 2.0 / n
        best_h = project_polytope(0.5 + 0.4 * np.cos(np.pi * x), n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
