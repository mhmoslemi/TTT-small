    n_conj = 2*n - 1
    n_var = n + 1  # d[0..n-1] + tau

    c[-1] = 1.0  # minimize tau

    A_ub[:, :n] = dx * G  # G is (2n-1, n)
    A_ub[:, -1] = -1.0
    b_ub = -C_vals  # (2n-1,)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h0[i], -rho)
        hi = min(1.0 - h0[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, None))  # tau unbounded
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:

        return d

def smooth_gradient_descent(h, n, dx, T, lr, max_iter, t0, budget):
    """Minimize smooth max approximation."""
    for it in range(max_iter):
        if time.time() - t0 > budget:
            break
        C = np.correlate(h, 1-h, mode="full") * dx
        # Softmax weights
        C_max = np.max(C)
        weights = np.exp(T * (C - C_max))
        weights /= np.sum(weights)
        
        # Gradient
        G = compute_gradients(h, n)  # (2n-1, n)
        grad = dx * (G.T @ weights)  # (n,)
        
        # Projected gradient step
        h_new = h - lr * grad
        # Project to [0,1] and sum = n/2
        h_new = project(h_new, n)
        
        # Check improvement
        c5_new = np.max(np.correlate(h_new, 1-h_new, mode="full") * dx)
        c5_old = np.max(C)
        if c5_new < c5_old:
            h = h_new
        else:
            lr *= 0.5
    
    return h

            G[j, lag:] += (1.0 - h0[:n - lag])
            G[j, :n - lag] -= h0[lag:]

                    best_c5 = c5_pert
                    best_h = h_pert.copy()
                    h = h_pert.copy()
                    stuck_count = 0
                    rho = 0.05
                    break
            else:
                break
        if rho < 1e-7:
            break
    return best_h, best_c5

def smooth_gradient_step(h, n, dx, T, lr):
    C = compute_corr_vals(h)
    C_shifted = C - np.max(C)
    w = np.exp(T * C_shifted)
    w /= np.sum(w)
    G = compute_gradients(h, n)
    grad = dx * (G.T @ w)
    h_new = h - lr * grad
    h_new = project_onto_simplex(h_new, n)
    return h_new

def smooth_optimize(h, n, dx, time_limit, rng):
    best_h = h.copy()
    best_c5 = evaluate(h)
    T = 30.0
    lr = 0.01
    stuck = 0
    while time.time() < time_limit:
        h_new = smooth_gradient_step(h, n, dx, T, lr)
        c5_new = evaluate(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stuck = 0
        else:
            lr *= 0.7
            stuck += 1
        if stuck > 50:
            # Random restart from best
            h = best_h.copy() + rng.normal(0, 0.01, n)
            h = project_onto_simplex(h, n)
            lr = 0.01
            stuck = 0
        if lr < 1e-8:
            h = best_h.copy()
            lr = 0.01
    return best_h, best_c5

def random_binary(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h

def pairwise_swap_optimize(h, n, dx, time_limit, rng):

            new_vj = vj + (vj + vi - new_vi) * 0 if False else vj + (vi - new_vi)
            # Simpler: try specific redistributions

        # Simple swap-based move
        delta = rng.uniform(-0.2, 0.2)
        h_new = h.copy()

def multi_resolution(h, n, dx, time_limit, rng):
    """Downsample, optimize, upsample."""
    # Try a coarser resolution
    n_coarse = max(n // 2, 10)
    # Average pairs
    if n_coarse * 2 >= n:
        h_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            h_coarse[i] = np.mean(h[i*2:(i*2+2)])
    else:
        h_coarse = np.mean(h.reshape(-1, n_coarse), axis=1)
    
    # Ensure sum constraint
    target_coarse = n_coarse / 2.0
    h_coarse = project_onto_simplex(h_coarse, n_coarse)
    
    # Optimize at coarse resolution
    dx_coarse = 2.0 / n_coarse
    h_opt_coarse, c5_coarse = lp_optimize(h_coarse, n_coarse, dx_coarse,
                                           time_limit * 0.4, rng)
    
    # Upsample
    h_up = np.zeros(n)
    for i in range(n):
        h_up[i] = h_opt_coarse[i // 2]
    h_up = project_onto_simplex(h_up, n)
    
    # Refine at fine resolution
    h_fine, c5_fine = lp_optimize(h_up, n, dx, time_limit * 0.5, rng)
    return h_fine, c5_fine

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except NameError:
        h0 = random_binary(130, rng)
    
    n0 = len(h0)
    dx0 = 2.0 / n0
    
    # Ensure sum constraint
    h0 = project_onto_simplex(h0, n0)
    
    best_h = h0.copy()
    best_c5 = evaluate(best_h)
    
    # Phase 1: LP trust region on the given construction (50% budget)
    t1 = t0 + budget_s * 0.45
    h1, c5_1 = lp_optimize(h0.copy(), n0, dx0, t1, rng)
    if c5_1 < best_c5:
        best_h = h1.copy()
        best_c5 = c5_1
    
    # Phase 2: Smooth gradient on the best (10% budget)
    t2 = t1 + budget_s * 0.10
    h2, c5_2 = smooth_optimize(best_h.copy(), len(best_h), 2.0/len(best_h), t2, rng)
    if c5_2 < best_c5:
        best_h = h2.copy()
        best_c5 = c5_2
    
    # Phase 3: Multi-resolution (15% budget)
    t3 = t2 + budget_s * 0.15
    h3, c5_3 = multi_resolution(h0.copy(), n0, dx0, t3 - time.time(), rng)
    if c5_3 < best_c5:
        best_h = h3.copy()
        best_c5 = c5_3
    
    # Phase 4: Try binary sequences with even n (10% budget)
    t4 = t3 + budget_s * 0.10
    for n_try in [128, 130, 126]:
        if time.time() > t4:
            break
        h_bin = random_binary(n_try, rng)
        # Optimize
        dx_try = 2.0 / n_try
        h_opt, c5_opt = lp_optimize(h_bin.copy(), n_try, dx_try,
                                     min(t4, time.time() + budget_s * 0.03), rng)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    # Phase 5: Final polish on the best (remaining time)
    t5 = t0 + budget_s - 10
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    h5, c5_5 = lp_optimize(best_h.copy(), n_best, dx_best, t5, rng)
    if c5_5 < best_c5:
        best_h = h5.copy()
        best_c5 = c5_5
    
    # Final: one more smooth pass if time allows
    if time.time() < t0 + budget_s - 3:
        h_final, c5_final = smooth_optimize(best_h.copy(), n_best, dx_best,
                                            t0 + budget_s - 2, rng)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return best_h, best_c5, len(best_h)
