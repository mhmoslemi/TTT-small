And max_changed = max of c[j] + α*da[j] + α²*c_quadratic[j] for positions where perturbation is nonzero

The affected positions are those j where:
- 0 ≤ p - (j-(n-1)) < n (i.e., p-k is valid, where k=j-(n-1))
- OR 0 ≤ q - (j-(n-1)) < n
- OR 0 ≤ p + (j-(n-1)) < n
- OR 0 ≤ q + (j-(n-1)) < n

This gives at most 4 contiguous ranges of j values. For n=84, the total affected set is at most about 4n ≈ 336 positions, but the output only has 2n-1 = 167 positions. So potentially all positions are affected.

Hmm, for p near the center and q near the center, the affected range might cover most of the 167 positions. In that case, I might as well just compute the full perturbation.

OK, let me just go with the straightforward approach: compute the full perturbation vector for each pair, do a grid search, and refine. For n=84, this is fast enough.

Let me write the final code now. I'll make it robust and efficient.

def evaluate_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fix_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            return h
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = np.sum(free)
        if nfree == 0:
            # All at boundaries - do a hard adjustment
            if diff > 0:
                idx = np.argmax(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmin(h)
                h[idx] = max(0.0, h[idx] + diff)
            return h
        per = diff / nfree
        h[free] += per
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_perturbation(h, p, q, n):
    """Compute linear (da) and quadratic (cq) perturbation for pair move p->q."""
    m = 2 * n - 1
    da = np.zeros(m)
    cq = np.zeros(m)
    
    for j in range(m):
        k = j - (n - 1)  # lag
        val_da = 0.0
        # corr(d,v)_k: -(1-h[p-k]) + (1-h[q-k])
        if 0 <= p - k < n:
            val_da -= (1.0 - h[p - k])
        if 0 <= q - k < n:
            val_da += (1.0 - h[q - k])
        # -corr(a,d)_k: +h[p+k] - h[q+k]
        if 0 <= p + k < n:
            val_da += h[p + k]
        if 0 <= q + k < n:
            val_da -= h[q + k]
        da[j] = val_da
        
        # -corr(d,d)_k
        val_cq = 0.0
        if k == 0:
            val_cq = -2.0
        elif k == (p - q):
            val_cq = 1.0
        elif k == (q - p):
            val_cq = 1.0
        cq[j] = val_cq
    
    return da, cq

def line_search(h, p, q, dx, n, c0):
    """
    Find best alpha for moving mass from p to q.
    c0 = current correlation vector (length 2n-1)
    Returns (best_alpha, best_c5)
    """
    a_hi = min(h[p], 1.0 - h[q])
    if a_hi <= 1e-12:
        return 0.0, np.inf
    
    da, cq = compute_perturbation(h, p, q, n)
    
    # Grid search
    n_grid = 25
    alphas = np.linspace(0.0, a_hi, n_grid)
    best_a = 0.0
    best_val = np.max(c0)
    
    for a in alphas:
        c_a = c0 + a * da + a * a * cq
        val = np.max(c_a) * dx
        if val < best_val * dx - 1e-14:
            best_val = val / dx  # store unscaled
            best_a = a
    
    if best_a == 0.0 and best_val >= np.max(c0):
        return 0.0, np.max(c0) * dx
    
    # Refine with golden section
    # Determine bracket
    idx_best = int(best_a / a_hi * (n_grid - 1))
    lo = alphas[max(0, idx_best - 1)]
    hi = alphas[min(n_grid - 1, idx_best + 1)]
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(50):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1 = np.max(c0 + m1 * da + m1 * m1 * cq)
        f2 = np.max(c0 + m2 * da + m2 * m2 * cq)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-14:
            break
    
    best_a = (lo + hi) / 2.0
    best_val = np.max(c0 + best_a * da + best_a * best_a * cq) * dx
    
    return best_a, best_val

def compute_gradient(h, dx, n):
    """Compute supergradient for pair selection."""
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)
    
    # Active positions (close to max)
    threshold = max_c - 0.02 * abs(max_c) - 1e-10
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    n_act = len(active)
    g = np.zeros(n)
    
    for idx in active:
        k = idx - (n - 1)
        # ∂V_k/∂h_i = dx * [(1-h[i-k]) if 0≤i-k<n - h[i+k] if 0≤i+k<n] / n_act
        # Vectorized:
        # For term 1: i-k valid means i in [k, k+n-1], intersect [0,n-1]
        # For term 2: i+k valid means i in [-k, n-1-k], intersect [0,n-1]
        
        i1_lo = max(0, k)
        i1_hi = min(n, n + k)
        if i1_lo < i1_hi:
            g[i1_lo:i1_hi] += dx * (1.0 - h[i1_lo - k:i1_hi - k]) / n_act
        
        i2_lo = max(0, -k)
        i2_hi = min(n, n - k)
        if i2_lo < i2_hi:
            g[i2_lo:i2_hi] -= dx * h[i2_lo + k:i2_hi + k] / n_act
    
    return g
