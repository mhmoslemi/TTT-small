def project_feasible(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}"""
    n = len(h)
    
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    # Ensure f(lo) > 0 and f(hi) < 0
    # f(-2) = sum(clip(h+2, 0, 1)) - target = n - target > 0 (since target = n/2)
    # f(2) = sum(clip(h-2, 0, 1)) - target = 0 - target < 0

        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h, n):
    """Compute the subgradient of the max correlation.
    Returns the gradient direction for the maximizing lag."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    s = k_star - (n - 1)  # the shift
    
    # Gradient of c_s w.r.t. h[j]:
    # (1 - h[j-s]) * 1{0 <= j-s < n} - h[j+s] * 1{0 <= j+s < n}
    grad = np.zeros(n)
    if s >= 0:
        # j-s >= 0 means j >= s
        grad[s:] = 1.0 - h[:n-s]  # (1-h[j-s]) for j >= s
        # j+s < n means j < n-s
        grad[:n-s] -= h[s:]  # -h[j+s] for j < n-s
    else:
        # s < 0, let d = -s > 0
        d = -s
        # j-s = j+d >= 0 always (since j >= 0), valid if j+d < n
        grad[:n-d] = 1.0 - h[d:]  # (1-h[j+d]) for j < n-d
        # j+s = j-d >= 0 means j >= d
        grad[d:] -= h[:n-d]  # -h[j-d] for j >= d
    
    return grad, k_star, corr

def smoothed_gradient(h, n, eps):
    """Compute the smoothed gradient using softmax weighting."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = np.exp(corr / eps)
    weights /= weights.sum()
    
    # Sum of weighted gradients
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        else:
            d = -s
            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n-d]
    
    return grad, corr
