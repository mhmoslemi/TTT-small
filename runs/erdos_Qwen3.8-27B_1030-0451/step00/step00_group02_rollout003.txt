import numpy as np

def exact_score(h, n):
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(dx * np.max(corr))

def is_feasible(h, n, tol=1e-9):
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    target = n / 2.0
    if abs(np.sum(h) - target) > tol:
        return False
    return True

def repair_vector(z, n):
    """Project z to feasible set: 0<=h<=1, sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(z, 0.0, 1.0)
    # Bisection on reduce amount
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(vals)
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def make_rank_fill(n, scores, target=None):
    """Create balanced vector from scores via rank fill"""
    if target is None:
        target = n / 2.0
    m = int(math.floor(target + 1e-12))
    frac = target - m
    h = np.zeros(n)
    order = np.argsort(scores, kind='stable')[::-1]
    h[order[:m]] = 1.0
    if frac > 1e-12 and m < n:
        h[order[m]] = frac
    return h

def make_score_vector(n, kind, phase=0.0, seed_val=0):
    """Generate deterministic score vectors for rank fill"""
    idx = np.arange(n, dtype=float)
    if kind == 0:
        golden = (math.sqrt(5) - 1) / 2
        return np.sin(2 * math.pi * (idx * golden + phase))
    elif kind == 1:
        sqrt2 = math.sqrt(2)
        return np.cos(2 * math.pi * (idx * sqrt2 / n + phase))
    elif kind == 2:
        rng = random.Random(seed_val)
        return np.array([rng.random() for _ in range(n)])
    elif kind == 3:
        phi = (1 + math.sqrt(5)) / 2
        return np.sin(idx * phi + phase) * np.cos(idx * 0.7 + phase * 2)
    elif kind == 4:
        # Alternating pattern
        return np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    elif kind == 5:
        # Block pattern
        block = max(1, n // 4)
        s = np.zeros(n)
        for b in range(4):
            start = b * block
            s[start:min(start+block, n)] = b + 0.5
        return s
    else:
        return np.random.default_rng(seed_val + kind).random(n)

def transfer_dims(h_old, n_old, n_new):

def lse_objective_smooth(h, n, tau):
    """Log-sum-exp smoothed objective"""
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full') * dx
    max_c = np.max(corr)
    shifted = (corr - max_c) / tau
    shift = np.max(shifted)
    lse = max_c + tau * (shift + np.log(np.sum(np.exp(shifted - shift))))
    return lse

def lse_gradient(h, n, tau):
    """Gradient of LSE objective"""
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full') * dx
    max_c = np.max(corr)
    shifted = (corr - max_c) / tau
    shift = np.max(shifted)
    weights = np.exp(shifted - shift)
    weights /= np.sum(weights)
    # Gradient of corr[i] w.r.t. h
    # corr[i] = sum_j h[j] * q[j+i-(n-1)] for valid j
    # Actually np.correlate(h, q, mode='full') gives:
    # result[k] = sum_j h[j+k-(n-1)] * q[j] for valid j
    # Let's derive: for k in [0, 2n-2], lag = k - (n-1)
    # result[k] = sum_{j: 0<=j<n, 0<=j+lag<n} h[j+lag] * q[j]
    #           = sum_j h[j+lag] * (1 - h[j])
    # d/dh[i] result[k] = (1-h[i]) * [i in range for h[j+lag]] 
    #                     + (-1) * h[i] * [i in range for q[j] = 1-h[j] where j+lag=i]
    # For lag = k-(n-1):
    # First term: h[j+lag] where j+lag = i => j = i-lag, valid if 0<=i-lag<n
    # Second term: q[j] = 1-h[j] where j = i and h[j+lag] = h[i+lag], valid if 0<=i+lag<n
    # So d/dh[i] = (1-h[i]) * I(0<=i-lag<n) - h[i] * I(0<=i+lag<n)
    grad = np.zeros(n)
    for k in range(2*n-1):
        lag = k - (n - 1)
        if weights[k] < 1e-15:
            continue
        # First term: j = i - lag, valid 0 <= i-lag < n
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            grad[lo1:hi1] += weights[k] * (1.0 - h[lo1:hi1])
        # Second term: i is the q index, h[i+lag] where i+lag valid
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] -= weights[k] * h[lo2:hi2]
    return dx * grad

def sa_search(h, n, t_start, time_limit, seed=42):
    """Simulated annealing with pair moves"""
    rng = random.Random(seed)
    cur = h.copy()
    cur_score = exact_score(cur, n)
    best = cur.copy()
    best_score = cur_score
    
    # Temperature schedule
    T = max(0.005, cur_score - 0.380)
    T_min = 0.0005
    cooling = 0.995
    
    while time.monotonic() - t_start < time_limit:
        if T < T_min:
            break
        
        # Random pair
        i = rng.randint(0, n-1)
        j = rng.randint(0, n-1)
        if i == j:
            continue
        
        # Random direction and magnitude
        direction = 1 if rng.random() < 0.5 else -1
        t = direction * rng.uniform(0.01, 0.5)
        
        trial = cur.copy()
        trial[i] += t
        trial[j] -= t
        
        if trial[i] < 0 or trial[i] > 1 or trial[j] < 0 or trial[j] > 1:
            T *= cooling
            continue
        
        new_score = exact_score(trial, n)
        delta = new_score - cur_score
        
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur = trial
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best = cur.copy()
                best_score = cur_score
        
        T *= cooling
    
    return best, best_score

        return lse_objective_smooth(x, n, tau)

        return lse_gradient(x, n, tau)
    
    def constraint(x):
        return np.sum(x) - n/2.0

        return np.ones(n)

import numpy as np
import time
import math
import random
from scipy.optimize import minimize

def exact_score(h, n):
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(dx * np.max(corr))

def is_feasible(h, n, tol=1e-8):
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    target = n / 2.0
    if abs(float(np.sum(h)) - target) > tol:
        return False
    return True

def repair_vector(z, n):
    target = n / 2.0
    h = np.clip(z.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        vals = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(vals)) > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(result < 1.0 - 1e-13)[0]
        else:
            idx = np.where(result > 1e-13)[0]
        if len(idx) > 0:
            chunk = residual / len(idx)
            result[idx] += chunk
            result = np.clip(result, 0.0, 1.0)
    return result

def make_rank_fill(n, scores):
    target = n / 2.0
    m = int(math.floor(target + 1e-12))
    frac = target - m
    h = np.zeros(n)
    order = np.argsort(scores, kind='stable')[::-1]
    if m > 0:
        h[order[:m]] = 1.0
    if frac > 1e-12 and m < n:
        h[order[m]] = frac
    return h

def make_score_vec(n, kind, phase=0.0, sv=0):
    idx = np.arange(n, dtype=float)
    if kind == 0:
        g = (math.sqrt(5) - 1) / 2.0
        return np.sin(2.0 * math.pi * (idx * g + phase))

    elif kind == 2:
        rng = random.Random(sv)
        return np.array([rng.random() for _ in range(n)])
    elif kind == 3:
        phi = (1.0 + math.sqrt(5.0)) / 2.0
        return np.sin(idx * phi + phase) + 0.5 * np.cos(idx * 0.7 + phase * 2.0)
    elif kind == 4:
        return np.where(np.arange(n) % 2 == 0, 1.0, 0.0) + 0.1 * np.sin(idx * 0.3 + phase)
    elif kind == 5:
        block = max(1, n // 3)
        s = np.zeros(n)
        for b in range(3):
            st = b * block
            s[st:min(st + block, n)] = 1.0 - b * 0.3
        return s + 0.05 * np.sin(idx + phase)
    else:
        rng = np.random.default_rng(sv + kind * 7919)
        return rng.random(n)

def transfer_dims(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return repair_vector(h_new, n_new)

def lse_smooth(x, n, tau):
    dx = 2.0 / n
    q = 1.0 - x
    corr = np.correlate(x, q, mode='full') * dx
    mc = np.max(corr)
    sh = (corr - mc) / tau
    ms = np.max(sh)
    return mc + tau * (ms + math.log(float(np.sum(np.exp(sh - ms)))))

def lse_grad(x, n, tau):
    dx = 2.0 / n
    q = 1.0 - x
    corr = np.correlate(x, q, mode='full') * dx
    mc = np.max(corr)
    sh = (corr - mc) / tau
    ms = np.max(sh)
    w = np.exp(sh - ms)
    w /= float(np.sum(w))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            grad[lo1:hi1] += w[k] * (1.0 - x[lo1:hi1])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            grad[lo2:hi2] -= w[k] * x[lo2:hi2]
    return dx * grad

def pair_search(h, n, t_start, time_limit, seed=42):

def sa_local(h, n, t_start, time_limit, seed=42):

    best = cur.copy()

    T_min = 0.0003
    cool = 0.997

    while time.monotonic() - t_start < time_limit:
        if T < T_min:
            break
        i = rng.randint(0, n - 1)
        j = rng.randint(0, n - 1)
        if i == j:
            T *= cool
            continue
        t = rng.uniform(0.005, 0.5) * (1 if rng.random() < 0.5 else -1)
        trial = cur.copy()
        trial[i] += t
        trial[j] -= t
        if trial[i] < 0 or trial[i] > 1 or trial[j] < 0 or trial[j] > 1:
            T *= cool
            continue

        delta = ns - cur_score

            cur = trial
            cur_score = ns
            if cur_score < best_score - 1e-12:
                best = cur.copy()
                best_score = cur_score

    return best, best_score

def lse_polish(h, n, t_start, time_limit):
    def obj(x):
        return lse_smooth(x, n, 0.015)

    def jac(x):
        return lse_grad(x, n, 0.015)

    def g_con(x):
        return float(np.sum(x)) - n / 2.0

    def g_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': g_con, 'jac': g_jac}]
    try:
        res = minimize(obj, h.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 60, 'ftol': 1e-8, 'eps': 1e-6})
        if time.monotonic() - t_start > time_limit:
            return None
        x = np.clip(res.x, 0.0, 1.0)
        x = repair_vector(x, n)
        if is_feasible(x, n):
            return x
    except Exception:
        pass
    return None
