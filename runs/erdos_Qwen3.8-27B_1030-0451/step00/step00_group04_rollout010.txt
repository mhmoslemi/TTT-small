import numpy as np
import time
from scipy.optimize import linprog

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(h, T):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    # Bisection for shift
    lo, hi = -max(float(np.max(h)) + 1.0, 1.0), max(1.0 - float(np.min(h)) + 1.0, 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s < T:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    # Fine correction
    err = T - float(y.sum())
    for _ in range(50):
        if abs(err) < 1e-14:
            break
        mask = (y > 1e-15) & (y < 1.0 - 1e-15)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        d = err / cnt
        y[mask] = np.clip(y[mask] + d, 0.0, 1.0)
        err = T - float(y.sum())
    return y

def raw_grad_z(h, ell):
    """Unscaled gradient of z at lag ell (ell can be negative)."""
    n = len(h)
    a = abs(ell)
    if a >= n:
        return np.zeros(n)
    L = n - a
    g = np.zeros(n)
    g[a:n] += (1.0 - h[0:L])
    g[0:L] -= h[a:n]
    return g

def active_set(z_vec, z_max, gap_frac, k_max):
    """Select active lag indices."""
    order = np.argsort(-z_vec)
    gap = max(1e-5, gap_frac * z_max)
    active = order[z_vec[order] >= z_max - gap]
    if len(active) > k_max:
        active = active[:k_max]
    if order[0] not in active:
        active = np.append(active, order[0])
    return active

def slp_optimize(h_init, T, time_limit, start_time, seed, max_iters=80, k_max=30, gap_frac=0.02):
    h = project_box_sum(h_init.copy(), T)
    n = len(h)
    curr_sc = exact_score(h)
    best_h = h.copy()
    best_sc = curr_sc
    rho = max(0.015, 4.0 / n)
    stall = 0
    rng = np.random.default_rng(seed)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 3:
            break
        
        z_vec = np.correlate(h, 1.0 - h, mode='full')
        z_max = float(z_vec.max())
        
        # Active set
        act = active_set(z_vec, z_max, gap_frac, k_max)
        K = len(act)
        
        # Convert lag indices to ell values
        # z_vec index i corresponds to ell = i - (n-1)
        lags = act - (n - 1)
        
        # Build LP
        G = np.zeros((K, n))
        b = np.zeros(K)
        for i in range(K):
            g = raw_grad_z(h, lags[i])
            G[i] = g
            b[i] = float(g @ h) - float(z_vec[act[i]])
        
        # LP: min t s.t. G@y - t <= b, sum(y)=T, bounds
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.hstack([G, -np.ones((K, 1))])
        b_ub = b
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 0:n] = 1.0
        b_eq = np.array([T])
        lb = np.maximum(0.0, h - rho)
        ub = np.minimum(1.0, h + rho)
        bounds = [(lb[i], ub[i]) for i in range(n)]
        bounds.append((0.0, z_max + 1.0))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            rho *= 0.6
            stall += 1
            if stall > 20 or rho < 1e-5:
                break
            continue
        
        if not res.success:
            rho *= 0.6
            stall += 1
            if stall > 20 or rho < 1e-5:
                break
            continue
        
        y_new = res.x[0:n]
        
        # Line search
        best_alpha = 0.0
        best_local = curr_sc
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            cand = (1.0 - alpha) * h + alpha * y_new
            cand = project_box_sum(cand, T)
            sc = exact_score(cand)
            if sc < best_local - 1e-14:
                best_local = sc
                best_alpha = alpha
        
        if best_alpha > 0:

            new_sc = exact_score(new_h)
            if new_sc < best_sc - 1e-14:

                best_h = new_h.copy()
            h = new_h
            curr_sc = new_sc
            stall = 0
            # Slightly increase rho on success
            rho = min(rho * 1.3, 0.05)
        else:
            rho *= 0.6
            stall += 1
        
        if stall > 25 or rho < 1e-5:

    return best_h, best_sc

    n = len(h)

    best_h = h.copy()

            break
        
        z_vec = np.correlate(h, 1.0 - h, mode='full')
        z_max = float(z_vec.max())

        act = np.where(z_vec >= z_max - gap)[0]
