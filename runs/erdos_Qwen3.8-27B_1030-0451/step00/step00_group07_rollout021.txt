    """Smooth approximation of max(x)."""
    x_shifted = x - np.max(x)
    return np.max(x) + (1.0/beta) * np.log(np.mean(np.exp(beta * x_shifted)))

    """Compute np.correlate(h, 1-h, mode='full')."""
    return np.correlate(h, 1-h, mode="full")

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _local_search_binary(h, n, dx, time_limit=60.0, rng=None):
    start = time.time()
    if rng is None:
        rng = np.random
    h = h.astype(float).copy()
    current_c5 = _compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    temp = 0.004

    while time.time() - start < time_limit:
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = _compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        temp *= 0.9998

    return best_h, best_c5

def _smooth_max_grad(h, n, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    c_dx = c * dx
    c_max = np.max(c_dx)
    shifted = c_dx - c_max
    exp_vals = np.exp(beta * shifted)
    total = np.sum(exp_vals)
    if total == 0:
        return c_max, np.zeros(n)
    weights = exp_vals / total
    fval = c_max + (1.0 / beta) * np.log(total / len(total) * len(c_dx))
    # Actually: smooth_max = (1/beta)*log(sum(exp(beta*x)))
    # Using shifted: c_max + (1/beta)*log(mean(exp(beta*(x-c_max))))
    # = c_max + (1/beta)*log(sum(exp(beta*shifted))/len)
    fval = c_max + (1.0 / beta) * np.log(total / len(c_dx))

    grad = np.zeros(n)
    for ells in range(-(n - 1), n):
        k = ells + (n - 1)
        w = weights[k] * dx
        if ells >= 0:
            jr1 = np.arange(ells, n)
            if len(jr1) > 0:
                grad[jr1] += w * (1.0 - h[jr1 - ells])
            jr2 = np.arange(0, n - ells)
            if len(jr2) > 0 and ells > 0:
                grad[jr2] -= w * h[jr2 + ells]
        else:
            m = -ells
            jr1 = np.arange(0, n - m)
            if len(jr1) > 0:
                grad[jr1] += w * (1.0 - h[jr1 + m])
            jr2 = np.arange(m, n)
            if len(jr2) > 0:
                grad[jr2] -= w * h[jr2 - m]
    return fval, grad

def _continuous_optimize(h0, n, dx, beta=80.0, maxiter=150, time_limit=90.0):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def objfun(x):
        if time.time() - start > time_limit:
            raise OverflowError("time")
        return _smooth_max_grad(x, n, dx, beta)

    try:
        result = minimize(objfun, h0, jac=True, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-10})
        if result.x is not None:
            h_new = np.clip(result.x, 0.0, 1.0)
            return h_new, _compute_c5(h_new)
    except Exception:
        pass
    return h0, _compute_c5(h0)

def _adjust_sum(h, target_sum):
    h = h.copy()
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def _make_binary_from_continuous(h, n_target):
    n_ones = n_target // 2
    sorted_idx = np.argsort(h)[::-1]
    binary = np.zeros(n_target)
    binary[sorted_idx[:n_ones]] = 1.0
    return binary

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.ones(120)

    n_init = len(h0)
    dx_init = 2.0 / n_init
    best_h = h0.copy()
    best_c5 = _compute_c5(h0)

    # Phase 1: Binarize initial and local search
    binary_init = _make_binary_from_continuous(h0, n_init)
    c5_bin = _compute_c5(binary_init)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = binary_init.copy()

    if remaining() > 150:
        tl = min(180, remaining() * 0.25)
        h_b, c5_b = _local_search_binary(binary_init, n_init, dx_init, tl, rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 2: Continuous optimization from best
    if remaining() > 90:
        tl = min(100, remaining() * 0.15)
        h_c, c5_c = _continuous_optimize(best_h, n_init, dx_init,
                                          beta=100, maxiter=200, time_limit=tl)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

    # Phase 3: Multi-start with random binary sequences
    n_randoms = 5
    for r in range(n_randoms):
        if remaining() < 120:
            break
        h_rand = np.zeros(n_init)
        ones_positions = rng.choice(n_init, size=n_init // 2, replace=False)
        h_rand[ones_positions] = 1.0
        tl = min(90, remaining() * 0.2)
        h_r, c5_r = _local_search_binary(h_rand, n_init, dx_init, tl, rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 4: Try different n values
    n_targets = [200, 300, 500, 800, 1200, 2000]
    best_n = len(best_h)

    for n_t in n_targets:
        if remaining() < 150:
            break
        if n_t <= best_n:
            continue

        dx_t = 2.0 / n_t
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_t, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        h_interp = _adjust_sum(h_interp, n_t / 2.0)

        # Binary local search from interpolated
        binary_t = _make_binary_from_continuous(h_interp, n_t)
        tl = min(90, remaining() * 0.3)
        h_bt, c5_bt = _local_search_binary(binary_t, n_t, dx_t, tl, rng)
        if c5_bt < best_c5:
            best_c5 = c5_bt
            best_h = h_bt
            best_n = n_t

        # Continuous optimization from interpolated
        if remaining() > 120 and n_t <= 800:
            tl2 = min(60, remaining() * 0.2)
            h_ct, c5_ct = _continuous_optimize(h_interp, n_t, dx_t,
                                               beta=80, maxiter=100, time_limit=tl2)
            if c5_ct < best_c5:
                best_c5 = c5_ct
                best_h = h_ct
                best_n = n_t

    # Phase 5: Final polish
    if remaining() > 60:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        if np.allclose(best_h, np.round(best_h)):
            best_h = np.round(best_h)
            tl = min(90, remaining() * 0.6)
            h_f, c5_f = _local_search_binary(best_h, n_f, dx_f, tl, rng)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
        elif remaining() > 40:
            tl = min(40, remaining() * 0.4)
            h_f, c5_f = _continuous_optimize(best_h, n_f, dx_f,
                                              beta=120, maxiter=100, time_limit=tl)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f

    # Final constraint check
    n_final = len(best_h)
    target_sum = n_final / 2.0
    best_h = _adjust_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n_final
