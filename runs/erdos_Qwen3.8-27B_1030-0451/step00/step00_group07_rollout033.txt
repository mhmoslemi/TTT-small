import numpy as np

def exact_metric(h):
    """C5 = max(np.correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_vector(y, target):
    """Project onto {0<=x<=1, sum=target}"""
    y = np.asarray(y, dtype=np.float64).copy()
    n = len(y)
    if n == 0:
        return y
    lo = float(np.min(y)) - target - 2.0
    hi = 1.0 - float(np.max(y)) + target + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    res = math.fsum(z) - target
    if res > 1e-10:
        for i in np.argsort(-z):
            d = min(z[i], res)
            z[i] -= d
            res -= d
            if res < 1e-13:
                break
    elif res < -1e-10:
        for i in np.argsort(z):
            d = min(1.0 - z[i], -res)
            z[i] += d
            res += d
            if res > -1e-13:
                break
    return z

def resample_h(h, new_n):
    """Resample PW-constant h to new_n points."""
    n_old = len(h)
    if n_old == new_n:
        return h.copy()
    dx_old = 2.0 / n_old
    dx_new = 2.0 / new_n
    cum = np.zeros(n_old + 1)
    np.cumsum(h * dx_old, out=cum[1:])
    new_h = np.zeros(new_n)
    for j in range(new_n):
        t0 = j * dx_new
        t1 = (j + 1) * dx_new
        i0 = min(int(t0 / dx_old), n_old - 1)
        i1 = min(int(t1 / dx_old), n_old - 1)
        f0 = cum[i0] + (t0 - i0 * dx_old) * h[i0]
        f1 = cum[i1] + (t1 - i1 * dx_old) * h[i1]
        new_h[j] = (f1 - f0) / dx_new
    return repair_vector(new_h, new_n / 2.0)
