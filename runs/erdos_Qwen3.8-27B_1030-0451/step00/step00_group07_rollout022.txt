```python
    hi = 2.0

        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fix rounding
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        for i in idx:
            if diff > 0 and h[i] < 1:
                add = min(diff, 1-h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
```

```python
import numpy as np
import time

def exact_c5(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1-h, mode="full").max() * dx

def smooth_max_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1-h, mode="full")
    m = c.max()
    shifted = (c - m) / tau
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    w = e / e.sum()
    val = (m + tau * np.log(e.sum())) * dx
    
    grad = np.zeros(N)
    for k in range(2*N-1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (N-1)
        lo1 = max(0, lag)
        hi1 = min(N-1, N-1+lag)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * (1.0 - h[lo1-lag:hi1-lag+1])
        lo2 = max(0, -lag)
        hi2 = min(N-1, N-1-lag)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += wk * (-h[lo2+lag:hi2+lag+1])
    
    grad *= dx
    return val, grad

def project(y, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        if np.clip(y-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y-(lo+hi)/2, 0, 1)
    d = target - h.sum()
    if abs(d) > 1e-12:
        free = np.where((h>1e-10)&(h<1-1e-10))[0]
        if len(free)>0:
            h[free] += d/len(free)
            h = np.clip(h, 0, 1)
    return h

def interpolate(h, n_new):
    n = len(h)
    x = np.arange(n) * n_new / n
    return np.interp(x, np.arange(n), h)

def pgd(h0, n, rng, check_fn, verbose=False):
    target = n/2.0
    h = project(h0, target)
    best = h.copy()
    best_v = exact_c5(h)
    
    n_iter = 3000
    # tau schedule
    c_scale = np.correlate(h, 1-h, mode="full").max()
    tau0 = 0.1 * c_scale
    tau_end = 0.005 * c_scale
    tau0 = max(tau0, 0.01)
    tau_end = max(tau_end, 0.001)
    
    # Learning rate schedule
    # Start with moderate lr, decrease
    lr0 = 0.5 / c_scale  # since gradient is of order c_scale * dx
    
    for it in range(n_iter):
        if check_fn():
            break
        t = it / n_iter
        tau = tau0 * (tau_end/tau0) ** t
        lr = lr0 * (1 - 0.5*t)
        
        val, grad = smooth_max_grad(h, tau)
        h_new = project(h - lr * grad, target)
        v = exact_c5(h_new)
        if v < best_v:
            best_v = v
            best = h_new.copy()
        h = h_new
    
    return best, best_v
```

```python
import numpy as np

def exact_c5(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.correlate(h, 1-h, mode="full").max() * dx)

def smooth_max_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1-h, mode="full")
    m = c.max()
    shifted = (c - m) / tau
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    w = e / e.sum()
    val = (m + tau * np.log(e.sum())) * dx
    
    grad = np.zeros(N)
    for k in range(2*N-1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (N-1)
        lo1 = max(0, lag)
        hi1 = min(N-1, N-1+lag)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * (1.0 - h[lo1-lag:hi1-lag+1])
        lo2 = max(0, -lag)
        hi2 = min(N-1, N-1-lag)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += wk * (-h[lo2+lag:hi2+lag+1])
    grad *= dx
    return val, grad

def project_feasible(y, target):
    N = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        if np.clip(y-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y-(lo+hi)/2, 0, 1)
    d = target - h.sum()
    if abs(d) > 1e-12:
        free = np.where((h>1e-10)&(h<1-1e-10))[0]
        if len(free)>0:
            h[free] += d/len(free)
            h = np.clip(h, 0, 1)
    return h
```

```python
import numpy as np
import time

def exact_c5(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.correlate(h, 1 - h, mode="full").max() * dx)

def smooth_max_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1 - h, mode="full")
    m = c.max()
    shifted = (c - m) / tau
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    w = e / e.sum()
    val = (m + tau * np.log(e.sum())) * dx
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (N - 1)
        lo1 = max(0, lag)
        hi1 = min(N - 1, N - 1 + lag)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wk * (1.0 - h[lo1 - lag:hi1 - lag + 1])
        lo2 = max(0, -lag)
        hi2 = min(N - 1, N - 1 - lag)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += wk * (-h[lo2 + lag:hi2 + lag + 1])
    grad *= dx
    return val, grad

def project_feasible(y, target):
    N = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(y - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2, 0, 1)
    d = target - h.sum()
    if abs(d) > 1e-12:
        free = np.where((h > 1e-10) & (h < 1 - 1e-10))[0]
        if len(free) > 0:
            h[free] += d / len(free)
            h = np.clip(h, 0, 1)
    return h

def interp_upsample(h, n_new):
    n = len(h)
    x = np.arange(n_new) * n / n_new
    return np.interp(x, np.arange(n), h)

def pgd_optimize(h0, n, check_time):
    target = n / 2.0
    h = project_feasible(h0, target)
    best = h.copy()
    best_v = exact_c5(h)
    c = np.correlate(h, 1 - h, mode="full")
    c_max = max(c.max(), 1e-6)
    tau_start = max(0.1 * c_max, 1.0)
    tau_end = max(0.004 * c_max, 0.1)
    n_iter = 20000
    lr_init = n * 0.3
    for it in range(n_iter):
        if check_time():
            break
        t = it / n_iter
        tau = tau_start * (tau_end / tau_start) ** t
        lr = lr_init * (1.0 - 0.6 * t)
        val, grad = smooth_max_grad(h, tau)
        h = project_feasible(h - lr * grad, target)
        if it % 30 == 0:
            v = exact_c5(h)
            if v < best_v:
                best_v = v
                best = h.copy()
    v = exact_c5(h)
    if v < best_v:
        best_v = v
        best = h.copy()
    return best, best_v

def binary_local_search(h, n, max_iters, rng, check_time):
    target = n // 2
    h = h.copy()
    sorted_idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    best_v = exact_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iters):
        if check_time():
            break
        a = int(ones[rng.integers(len(ones))])
        b = int(zeros[rng.integers(len(zeros))])
        if a == b:
            continue
        h[a] = 0
        h[b] = 1
        v = exact_c5(h)
        if v < best_v - 1e-15:
            best_v = v
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[a] = 1
            h[b] = 0
    return h, best_v

def directed_binary_search(h, n, max_rounds, rng, check_time):
    """Focus on the worst lag for binary swaps"""
    target = n // 2
    h = h.copy()
    sorted_idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[sorted_idx[:target]] = 1.0
    best_v = exact_c5(h)
    for rnd in range(max_rounds):
        if check_time():
            break
        c = np.correlate(h, 1 - h, mode="full")
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Find candidates that reduce count at k_star
        best_swap = None
        best_v_try = best_v
        n_candidates = min(200, len(ones) * len(zeros))
        for _ in range(n_candidates):
            a = int(ones[rng.integers(len(ones))])
            b = int(zeros[rng.integers(len(zeros))])
            if a == b:
                continue
            # Check if this swap reduces count at lag
            reduce = 0
            if 0 <= a + lag < n and h[a + lag] == 0 and h[a] == 1:
                reduce += 1
            if 0 <= a - lag < n and h[a - lag] == 1 and h[a] == 1:
                reduce -= 1
            if 0 <= b + lag < n and h[b + lag] == 0 and h[b] == 0:
                reduce -= 1
            if 0 <= b - lag < n and h[b - lag] == 1 and h[b] == 0:
                reduce += 1
            if reduce <= 0:
                continue
            h[a] = 0
            h[b] = 1
            v = exact_c5(h)
            if v < best_v_try:
                best_v_try = v
                best_swap = (a, b)
            h[a] = 1
            h[b] = 0
        if best_swap is not None:
            a, b = best_swap
            h[a] = 0
            h[b] = 1
            best_v = best_v_try
    return h, best_v

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    def time_up():
        return (time.time() - start) > budget_s - 90

    # Get initial
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).flatten()
    else:
        h_init = np.full(120, 0.5)

    n0 = len(h_init)
    best_h = project_feasible(h_init, n0 / 2.0)
    best_v = exact_c5(best_h)
    best_n = n0
    print(f"Initial: N={best_n}, C5={best_v:.6f}")

    # PGD at original N
    if not time_up():
        h_opt, v_opt = pgd_optimize(best_h, n0, time_up)
        if v_opt < best_v:
            best_v, best_h, best_n = v_opt, h_opt, n0
            print(f"PGD N={n0}: C5={best_v:.6f}")

    # Try larger N values
    ns_to_try = []
    for mult in [2, 3, 4, 5, 6]:
        n = n0 * mult
        if n <= 2400:
            ns_to_try.append(n)
    ns_to_try = sorted(set(ns_to_try))

    for n in ns_to_try:
        if time_up():
            break
        h_up = interp_upsample(h_init, n)
        h_up = project_feasible(h_up, n / 2.0)
        v_up = exact_c5(h_up)
        h_opt, v_opt = pgd_optimize(h_up, n, time_up)
        if time_up():
            break
        if v_opt < best_v:
            best_v, best_h, best_n = v_opt, h_opt, n
            print(f"PGD N={n}: C5={best_v:.6f}")

    # Try PGD on the current best (continued optimization)
    if not time_up():
        h_opt, v_opt = pgd_optimize(best_h, best_n, time_up)
        if v_opt < best_v:
            best_v, best_h = v_opt, h_opt
            print(f"Additional PGD N={best_n}: C5={best_v:.6f}")

    # Binary local search
    if not time_up():
        h_bin, v_bin = directed_binary_search(best_h, best_n, 50, rng, time_up)
        if v_bin < best_v:
            best_v, best_h, best_n = v_bin, h_bin, best_n
            print(f"Directed binary: C5={best_v:.6f}")

    # Random binary search
    if not time_up():
        h_bin2, v_bin2 = binary_local_search(best_h, best_n, 30000, rng, time_up)
        if v_bin2 < best_v:
            best_v, best_h, best_n = v_bin2, h_bin2, best_n
            print(f"Random binary: C5={best_v:.6f}")

    # More PGD with random restarts
    if not time_up():
        for restart in range(5):
            if time_up():
                break
            noise = rng.normal(0, 0.02, best_n)
            h_restart = project_feasible(best_h + noise, best_n / 2.0)
            h_opt, v_opt = pgd_optimize(h_restart, best_n, time_up)
            if v_opt < best_v:
                best_v, best_h = v_opt, h_opt
                print(f"Restart {restart}: C5={best_v:.6f}")

    # Final directed binary polish
    if not time_up():
        h_bin3, v_bin3 = directed_binary_search(best_h, best_n, 100, rng, time_up)
        if v_bin3 < best_v:
            best_v, best_h, best_n = v_bin3, h_bin3, best_n
            print(f"Final binary polish: C5={best_v:.6f}")

    print(f"Final: N={best_n}, C5={best_v:.6f}")
    return best_h, best_v, best_n
```
