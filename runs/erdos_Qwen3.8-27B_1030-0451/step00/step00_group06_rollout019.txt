    step = step_init

        lags = np.correlate(h, 1-h, mode='full') * dx

        shifted = alpha * (lags - np.max(lags))  # for numerical stability

def evaluate(h):

    n = len(h)

def compute_lags(h):

import numpy as np

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project(h, target_sum):
    n = len(h)
    if n == 0:
        return h.copy()
    lo = -float(np.max(h)) - 0.5
    hi = 1.0 - float(np.min(h)) + 0.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - float(np.sum(result))
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = result < 1.0 - 1e-14
            idx = np.where(mask)[0]
            if len(idx) > 0:
                result[idx[-1]] = min(1.0, result[idx[-1]] + diff)
        else:
            mask = result > 1e-14
            idx = np.where(mask)[0]
            if len(idx) > 0:
                result[idx[0]] = max(0.0, result[idx[0]] + diff)
    return result
