import numpy as np

def project_sum_box(x, target):
    """Project x onto {0<=y<=1, sum(y)=target}"""
    n = len(x)
    # Bisection on lambda: sum(clip(x + lambda, 0, 1)) = target
    lo, hi = -x.max() - 1, 1 - x.min() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(x + mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo+hi)/2, 0, 1)
    # Fix tiny residual
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add residual: add to elements that are < 1
            slack = 1 - y
            idx = np.where(slack > 1e-10)[0]
            if len(idx) > 0:
                y[idx] += residual * slack[idx] / slack[idx].sum()
        else:
            # Need to subtract: subtract from elements that are > 0
            avail = y
            idx = np.where(avail > 1e-10)[0]
            if len(idx) > 0:

    y = np.clip(y, 0, 1)

    y = np.clip(y, 0, 1)
    return y

    """Compute all lag overlap values (the raw correlation * dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    """Compute C5 = max of overlap values."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals))

def grad_single_lag(h, lag, n):
    """Gradient of v(lag) = sum_i h[i]*(1-h[i+lag]) for valid i, scaled by dx."""
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        # v(lag) = sum_{i=0}^{m-1} h[i] * (1-h[i+lag])
        # d/dh[j]:
        # If j < m: (1-h[j+lag])
        # If j >= lag and j-lag < m: -h[j-lag]  (i.e., j in [lag, lag+m-1] = [lag, n-1])
        if m > 0:
            g[:m] += (1 - h[lag:lag+m])  # first factor
            g[lag:] -= h[:m]  # second factor: -h[j-lag] for j in [lag, n-1]
    else:
        L = -lag
        m = n - L
        # v(lag) = sum_{j=0}^{m-1} h[j+L] * (1-h[j])
        # d/dh[j]:
        # If j >= L and j-L < m: (1-h[j-L])  (j in [L, L+m-1] = [L, n-1])
        # If j < m: -h[j+L]  (j in [0, m-1])
        if m > 0:
            g[L:] += (1 - h[:m])  # first factor: (1-h[j-L]) for j in [L, n-1]
            g[:m] -= h[L:L+m]  # second factor: -h[j+L] for j in [0, m-1]
    return g * dx

def logsumexp_obj(h, beta):
    """Log-sum-exp smoothed objective and its gradient."""
    v = compute_v(h)
    m = v.max()
    if m <= 0:
        return 0.0, np.zeros(len(h))
    u = np.exp(beta * (v - m))
    Z = u.sum()
    f = m - np.log(Z) / beta
    w = u / Z
    # Gradient: sum_l w_l * grad_v_l
    n = len(h)
    g = np.zeros(n)
    # For each lag with significant weight, add weighted gradient
    # lag for output index p is: lag = (n-1) - p
    for p in range(len(v)):
        if w[p] < 1e-12:
            continue
        lag = (n - 1) - p
        g += w[p] * grad_single_lag(h, lag, n)
    return f, g

def logsumexp_grad(h, beta):
    """Log-sum-exp smoothed objective and its gradient (combined)."""
    return logsumexp_obj(h, beta)

def slp_solve(h0, c0, n, radius, active_lags, max_iter=100):
    """Sequential Linear Programming on active constraints."""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h0.copy()
    best_c = c0
    r = radius
    
    for it in range(max_iter):
        # Build LP
        # Variables: x (length n) + t (scalar) = n+1 variables
        # Objective: minimize t → c = [0,...,0, 1]
        # Constraints:
        # For each active lag ℓ: a_ℓ · x ≤ t + (a_ℓ · h0 - v_ℓ(h0))
        #   i.e., a_ℓ · x - t ≤ a_ℓ · h0 - v_ℓ(h0)
        # 0 ≤ x ≤ 1
        # sum(x) = target
        # |x - h0| ≤ r (trust region)
        
        n_active = len(active_lags)
        # A_ub has rows for: active constraints + trust region bounds
        # Active constraint row: [a_ℓ, -1] · [x, t] ≤ a_ℓ·h0 - v_ℓ(h0)
        # Trust region: x - h0 ≤ r → [e_i, 0] · [x,t] ≤ h0_i + r_i
        #              -(x - h0) ≤ r → [-e_i, 0] · [x,t] ≤ -h0_i + r_i
        
        A_rows = []
        b_rows = []
        v0 = compute_v(h0)
        
        for lag in active_lags:
            a = grad_single_lag(h0, lag, n) / dx  # unscaled gradient (for the raw v)
            # Actually, let me just use the scaled gradient since v is scaled
            a = grad_single_lag(h0, lag, n)  # this already includes dx
            # a · x ≤ t + (a · h0 - v0[lag])
            # a · x - t ≤ a · h0 - v0[lag]
            row = np.concatenate([a, [-1.0]])
            rhs = a @ h0 - v0[(n-1)-lag]  # v0 index
            A_rows.append(row)
            b_rows.append(rhs)
        
        # Trust region bounds
        for i in range(n):
            A_rows.append(np.concatenate([np.eye(1, n, i).ravel(), [0.0]]))
            b_rows.append(h0[i] + r)
            A_rows.append(np.concatenate([-np.eye(1, n, i).ravel(), [0.0]]))
            b_rows.append(-h0[i] + r)
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        # Equality: sum(x) = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        # Bounds: 0 ≤ x ≤ 1, t unbounded (or bounded)
        bounds = [(0, 1)] * n + [(-0.5, 0.9)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='HiGHS')
        
        if not res.success:
            r *= 0.5

            best_c = c_new
            r = min(r * 1.1, 0.5)  # expand slightly
        else:
            r *= 0.6
            if r < 0.001:
                break
    
    return best_h, best_c

def compute_v_full(h):

    dx = 2.0 / n

def grad_v_lag(h, lag, n, dx):
    """Gradient of v(lag) w.r.t. h, including dx scaling."""
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return g
        # v(lag) = dx * sum_{i=0}^{m-1} h[i]*(1-h[i+lag])
        # d/dh[j] = dx * [(1-h[j+lag]) if j<m] - dx * [h[j-lag] if j>=lag and j<n]
        g[:m] += (1.0 - h[lag:lag+m])
        g[lag:n] -= h[:m]
    else:
        L = -lag
        m = n - L
        if m <= 0:
            return g
        # v(lag) = dx * sum_{j=0}^{m-1} h[j+L]*(1-h[j])
        # d/dh[j] = dx * [(1-h[j-L]) if j>=L] - dx * [h[j+L] if j<m]
        g[L:n] += (1.0 - h[:m])
        g[:m] -= h[L:L+m]
    return g * dx

    hi = float(1.0 - x.min()) + 0.01

        else:

    if abs(res) > 1e-12:
        if res > 0:
            mask = (y < 1.0 - 1e-12)
            cnt = mask.sum()
            if cnt > 0:

            if cnt > 0:

def lse_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    v = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(v.max())
    if m <= 1e-15:
        return 0.0
    shifted = np.minimum(beta * (v - m), 0.0)
    u = np.exp(shifted)
    Z = float(u.sum())
    if Z < 1e-300:
        return m
    return m - np.log(Z) / beta

def lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    v = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(v.max())
    if m <= 1e-15:
        return np.zeros(n)
    shifted = np.minimum(beta * (v - m), 0.0)
    u = np.exp(shifted)
    Z = float(u.sum())
    if Z < 1e-300:
        return np.zeros(n)
    w = u / Z
    g = np.zeros(n)
    thresh = 1e-10
    for p in range(len(v)):
        if w[p] < thresh:
            continue
        lag = (n - 1) - p
        g += w[p] * grad_v_lag(x, lag, n, dx)
    return g

    return np.ones(len(x))

    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return g

        g[lag:n] -= h[:m]

        m = n - L
        if m <= 0:

        g[L:n] += (1.0 - h[:m])

    return g * dx
