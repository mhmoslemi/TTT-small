import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c * dx))

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mu = (lo + hi) / 2.0
        y = np.clip(z - mu, 0.0, 1.0)
        if y.sum() > S:
            lo = mu
        else:
            hi = mu
    y = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.any():
            y[mask] += diff / mask.sum()
    return np.clip(y, 0.0, 1.0)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    phi = c * dx
    mp = np.max(phi)
    ep = np.exp((phi - mp) / tau)
    Z = ep.sum()
    w = ep / Z
    obj = tau * (np.log(Z) + mp / tau)
    grad = np.zeros(n)
    for m in range(len(phi)):
        if w[m] < 1e-14:
            continue
        s = m - (n - 1)
        k_lo = max(0, -s)
        k_hi = min(n - 1, n - 1 - s)
        lo1 = max(0, k_lo + s)
        hi1 = min(n - 1, k_hi + s)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += w[m] * dx * (1.0 - h[idx - s])
        lo2 = max(0, k_lo)
        hi2 = min(n - 1, k_hi)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            sh = idx + s
            valid = (sh >= 0) & (sh < n)
            if valid.any():
                grad[idx[valid]] += w[m] * dx * (-h[sh[valid]])
    return obj, grad

def fw_extreme(g, n):
    S = n / 2.0
    u = np.zeros(n)
    order = np.argsort(g)
    Si = int(S)
    Sf = S - Si
    for i in range(Si):
        u[order[i]] = 1.0
    if Sf > 1e-10:
        u[order[Si]] = Sf
    return u

    def jac(self, x):
        return smooth_grad(x, self.tau)[1]
    def c_fun(self, x):
        return x.sum() - self.S
    def c_jac(self, x):

            best_h = h_c
            n0 = n_c
        elif abs(s_re - best_s) < 1e-10 and n_c < n0:
            best_s = s_re
            best_h = h_c
            n0 = n_c
    
    # Final polish on best
    if time.time() < deadline - 20:
        try:
            best_h, best_s = pair_search(best_h, n0, max_iter=3000,
                                          time_limit=min(80, deadline - time.time() - 10),
                                          t_start=t_start, rng=rng)
        except Exception:
            pass
        if time.time() < deadline - 15:
            for tau in [0.003, 0.0005]:
                if time.time() > deadline - 15:
                    break
                try:
                    best_h, best_s = slsqp_smooth(best_h, n0, tau, max_iter=20,
                                                   time_limit=20, t_start=t_start)
                except Exception:
                    pass
            if time.time() < deadline - 10:
                try:
                    best_h, best_s = pair_search(best_h, n0, max_iter=2000,
                                                  time_limit=min(40, deadline - time.time() - 5),
                                                  t_start=t_start, rng=rng)
                except Exception:
                    pass
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_s = exact_score(best_h)
    
    # Verify
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(best_h.sum() - n_final / 2.0) < 1e-6
    
    return best_h, best_s, n_final
