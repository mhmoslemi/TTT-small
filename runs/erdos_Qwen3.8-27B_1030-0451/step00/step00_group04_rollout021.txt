def corr_and_grad(h, p):
    """
    Compute correlations z and gradient of smooth-max.
    h: array of length n
    p: softmax weights of length 2n-1 (indexed by k from -(n-1) to n-1, stored as p[k+n-1])
    Returns: z (correlations * dx), grad (gradient of smooth-max)
    """
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    
    # Compute correlations
    z = np.zeros(2*n - 1)
    grad = np.zeros(n)
    
    for k in range(-(n-1), n):
        # k is the shift
        idx = k + (n - 1)  # index into z and p
        if k >= 0:
            # C_k = sum_{i=0}^{n-1-k} h(i+k)(1-h(i))
            L = n - k
            z[idx] = np.dot(h[k:], b[:L])
            # grad contribution: for j in [k, n-1]: (1-h(j-k))
            # for j in [0, L-1]: -h(j+k)
            if p[idx] != 0:
                grad[k:] += p[idx] * b[:L]
                grad[:L] -= p[idx] * h[k:]
        else:
            a = -k
            # C_k = sum_{i=a}^{n-1} h(i+k)(1-h(i)) = sum_{i=0}^{n-1-a} h(i)(1-h(i+a))
            L = n - a
            z[idx] = np.dot(h[:L], b[a:])

            # for j in [a, n-1]: -h(j-a) [= -h(j+k)]
            if p[idx] != 0:
                grad[:L] += p[idx] * b[a:]
                grad[a:] -= p[idx] * h[:L]

import numpy as np
import time
from scipy.optimize import minimize

_g_n = 0
_g_T = 0.0
_g_dx = 0.0

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _score(h):
    return float(np.max(_corr(h)))

def _project(h, T):
    n = len(h)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    d = T - float(y.sum())
    if abs(d) > 1e-14:
        m = (y > 1e-10) & (y < 1.0 - 1e-10)
        cnt = int(m.sum())
        if cnt > 0:
            y[m] += d / cnt
            y = np.clip(y, 0.0, 1.0)
    return y

def _smooth_obj(h, tau):
    n = len(h)
    z = _corr(h)
    zm = float(z.max())
    sh = np.clip((z - zm) / tau, -60.0, 0.0)
    return zm + tau * float(np.log(np.exp(sh).sum()))

def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = _corr(h)
    zm = float(z.max())
    sh = np.clip((z - zm) / tau, -60.0, 0.0)
    e = np.exp(sh)
    p = e / e.sum()
    
    grad = np.zeros(n)
    b = 1.0 - h
    for i in range(2 * n - 1):
        wi = float(p[i])
        if wi < 1e-12:
            continue
        k = i - (n - 1)
        if k >= 0:
            L = n - k
            grad[k:] += wi * b[:L]
            grad[:L] -= wi * h[k:]
        else:
            a = -k
            L = n - a
            grad[:L] += wi * b[a:]
            grad[a:] -= wi * h[:L]
    grad *= dx
    return grad

def _pgd(h0, deadline, tau_init=None, tau_min=0.001):
    n = len(h0)
    T = n / 2.0
    h = _project(h0.copy(), T)
    best_h = h.copy()
    best_sc = _score(h)
    
    if tau_init is None:
        z0 = _corr(h)
        tau_init = max(0.01, 0.3 * (float(z0.max()) - float(z0.min())))
    tau = tau_init
    alpha = 0.1
    stall = 0
    it = 0
    
    while time.time() < deadline:
        F = _smooth_obj(h, tau)
        grad = _smooth_grad(h, tau)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-12:
            break
        d = -grad / gn
        
        acc = False
        h_new = h
        for _ in range(20):
            h_t = _project(h + alpha * d, T)
            F_t = _smooth_obj(h_t, tau)
            if F_t < F * (1.0 - 1e-12) - 1e-16:
                h_new = h_t
                acc = True
                break
            alpha *= 0.5
        
        if not acc:
            tau = max(0.85 * tau, tau_min)
            alpha = 0.03
            stall += 1
            if stall > 40:
                break
            continue
        
        h = h_new
        alpha = min(1.3 * alpha, 0.3)
        stall = 0
        it += 1
        
        if it % 5 == 0:
            sc = _score(h)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_h = h.copy()
        
        if it % 20 == 0:

    sc = _score(h)
    if sc < best_sc:
        best_sc = sc
        best_h = h.copy()
    return best_h, best_sc

    g = np.zeros(_g_n + 1)
    g[-1] = 1.0
    return g

    h = x[:n]

    m = 2 * n - 1

    return J

def _slsqp(h0, deadline, maxiter=300):
    global _g_n, _g_T, _g_dx
    n = len(h0)
    T = n / 2.0
    _g_n = n
    _g_T = T
    _g_dx = 2.0 / n
    
    h = _project(h0.copy(), T)
    c0 = _score(h)
    t0 = c0 + 0.005
    
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': _epi_nlin_fun, 'jac': _epi_nlin_jac},
        {'type': 'eq', 'fun': _epi_eq_fun, 'jac': _epi_eq_jac},
    ]
    
    try:
        res = minimize(_epi_fun, x0, jac=_epi_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return h0, c0
    
    if res.x is not None:
        h_res = _project(res.x[:n], T)
        sc = _score(h_res)
        if sc < c0:
            return h_res, sc
    return h0, c0
