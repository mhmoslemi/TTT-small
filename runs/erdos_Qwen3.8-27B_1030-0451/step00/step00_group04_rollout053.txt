def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / n
    val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def exact_z_scaled(h):

def project_box_sum(x, T):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    # Water filling to achieve sum = T
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    err = T - np.sum(y)
    if abs(err) > 1e-12:

            mask = y < 1.0
            idx = np.where(mask)[0]

    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n, dtype=np.float64)
    ell = q - (n - 1)

            return g

        g[0:L] += dx * (1.0 - h[ell:n])

def exact_z_scaled(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_box_sum(x, T):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    err = T - np.sum(y)
    if abs(err) > 1e-13:
        if err > 0:
            mask = y < 1.0 - 1e-14
            idx = np.where(mask)[0]
            if len(idx) > 0:
                y[idx] = np.minimum(1.0, y[idx] + err / len(idx))
                err2 = T - np.sum(y)
                mask2 = y < 1.0 - 1e-14
                idx2 = np.where(mask2)[0]
                if len(idx2) > 0:
                    y[idx2] = np.minimum(1.0, y[idx2] + err2 / len(idx2))
        else:
            mask = y > 1e-14
            idx = np.where(mask)[0]
            if len(idx) > 0:
                y[idx] = np.maximum(0.0, y[idx] + err / len(idx))
                err2 = T - np.sum(y)
                mask2 = y > 1e-14
                idx2 = np.where(mask2)[0]
                if len(idx2) > 0:
                    y[idx2] = np.maximum(0.0, y[idx2] + err2 / len(idx2))
    return y

def grad_of_lag_sq(h, q):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n, dtype=np.float64)
    ell = q - (n - 1)
    if ell >= 0:
        L = n - ell
        if L <= 0:
            return g
        g[0:L] += dx * (1.0 - h[ell:n])
        g[ell:n] -= dx * h[0:L]
    else:
        a = -ell
        L = n - a
        if L <= 0:
            return g
        g[a:n] += dx * (1.0 - h[0:L])
        g[0:L] -= dx * h[a:n]
    return g

def slp_step_safe(h, K, rho, t_ub):
    n = len(h)
    T = n / 2.0
    z_scaled = exact_z_scaled(h)
    s_max = float(np.max(z_scaled))
    if s_max < 1e-15:
        return None, None
    order = np.argsort(-z_scaled)
    gap = max(1e-5, 0.02 * s_max)
    active = list(order[z_scaled[order] >= s_max - gap])
    if len(active) > K:
        active = active[:K]
    if order[0] not in active:
        active.append(int(order[0]))
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub_rows = []
    b_ub_vals = []
    for q in active:
        g = grad_of_lag_sq(h, int(q))
        row = np.zeros(n + 1)
        row[0:n] = g
        row[n] = -1.0
        rhs = float(np.dot(g, h)) - float(z_scaled[int(q)])
        A_ub_rows.append(row)
        b_ub_vals.append(rhs)
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([T])
    bounds = []
    for j in range(n):
        lo_j = max(0.0, h[j] - rho)
        hi_j = min(1.0, h[j] + rho)
        if lo_j >= hi_j:
            lo_j = max(0.0, h[j] - 1e-6)
            hi_j = min(1.0, h[j] + 1e-6)
        bounds.append((lo_j, hi_j))
    bounds.append((0.0, t_ub))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.status != 0 or res.x is None:
        return None, None
    y = np.asarray(res.x[0:n], dtype=np.float64)
    y = project_box_sum(y, T)
    s_new = exact_score(y)
    return y, s_new

def pairwise_polish(h, T, max_iters=200, time_limit=60, t_start=None):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    s_cur = exact_score(h)
    best_h = h.copy()
    best_s = s_cur
    if t_start is None:
        t_start = time.time()
    fracs = np.array([0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0])
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        z_scaled = exact_z_scaled(h)
        s_max = float(np.max(z_scaled))
        if s_max < 1e-15:
            break
        order = np.argsort(-z_scaled)
        gap = max(2e-5, 0.03 * s_max)
        active = order[z_scaled[order] >= s_max - gap]
        if len(active) == 0:
            active = order[:5]
        if len(active) > 25:
            active = active[:25]
        ws = z_scaled[active].copy()
        ws = np.maximum(ws - 0.95 * s_max, 0.005)
        ws = np.exp(ws / (0.02 * s_max + 1e-12))
        ws = ws / np.sum(ws)
        g_mix = np.zeros(n)
        for w, q in zip(ws, active):
            g_mix += w * grad_of_lag_sq(h, int(q))
        thresh = 0.05 * s_max
        sources = np.where((g_mix > thresh) & (h > 0.02))[0]
        targets = np.where((g_mix < -thresh) & (h < 0.98))[0]
        if len(sources) == 0 or len(targets) == 0:
            break
        sources = sources[np.argsort(-g_mix[sources])][:15]
        targets = targets[np.argsort(-g_mix[targets])][:15]
        improved = False
        for i in sources:
            if improved:
                break
            for j in targets:
                if i == j:
                    continue
                upper = min(h[i], 1.0 - h[j])
                if upper < 1e-7:
                    continue
                best_delta = 0.0
                best_s_try = s_cur
                for f in fracs:
                    delta = f * upper
                    if delta < 1e-9:
                        continue
                    h_try = h.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    s_try = exact_score(h_try)
                    if s_try < best_s_try - 1e-13:
                        best_s_try = s_try
                        best_delta = delta
                if best_delta > 0:
                    h[i] -= best_delta
                    h[j] += best_delta
                    s_cur = exact_score(h)
                    if s_cur < best_s - 1e-13:
                        best_s = s_cur
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_s

def generate_random_smooth(n, r, T):
    x = np.zeros(n)
    n_m = int(r.integers(2, 8))
    for _ in range(n_m):
        freq = int(r.integers(1, max(2, n // 4)))
        amp = r.uniform(0.1, 0.5)
        phase = r.uniform(0.0, 2.0 * np.pi)
        x += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    x += r.uniform(0.1, 0.4)
    return project_box_sum(x, T)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    r = np.random.default_rng(int(seed))
    
    def tleft():
        return budget_s - (time.time() - t0)
    
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
        n = len(parent)
    except Exception:
        n = 250
        parent = np.full(n, 0.5)
    
    if n % 2 != 0 and n > 2:
        n = n - 1
        parent = parent[:n]
    if n < 4:
        n = 250
        parent = np.full(n, 0.5)
    
    T = n / 2.0
    parent = project_box_sum(parent, T)
    
    best_h = parent.copy()
    best_s = exact_score(best_h)
    
    # Try transforms
    for tv in [parent, parent[::-1].copy(), 1.0 - parent, (1.0 - parent)[::-1].copy()]:
        tv = project_box_sum(tv, T)
        s = exact_score(tv)
        if s < best_s - 1e-12:
            best_h = tv.copy()
            best_s = s
    
    # Upscale parent
    for scale in [2, 4]:
        nc = n * scale
        if nc > 2500 or tleft() < 60:
            break
        hc = np.repeat(parent, scale)
        Tc = nc / 2.0
        hc = project_box_sum(hc, Tc)
        sc = exact_score(hc)
        if sc < best_s - 1e-12:
            best_h = hc.copy()
            best_s = sc
            n = nc
            T = Tc
    
    # SLP phase
    rho = max(0.01, 0.4 / n)
    max_slp = 400 if n <= 500 else 200
    slp_it = 0
    stall = 0
    
    while tleft() > 40 and slp_it < max_slp:
        K = min(100, max(20, int(0.1 * (2 * n - 1))))
        res_h, res_s = slp_step_safe(best_h, K, rho, best_s + 0.1)
        if res_h is not None and res_s < best_s - 1e-12:
            best_h = res_h
            best_s = res_s
            stall = 0
            rho = max(0.003, rho * 0.9)
        else:
            stall += 1
            rho = max(0.001, rho * 0.65)
        slp_it += 1
        if stall >= 4 and tleft() > 80:
            ph, ps = pairwise_polish(best_h, T, max_iters=40,
                                      time_limit=min(25, max(5, tleft() - 50)),
                                      t_start=time.time())
            if ps < best_s - 1e-12:
                best_h = ph
                best_s = ps
                stall = 0
                rho = max(0.008, rho * 1.5)
            stall = 0
        if stall > 25:
            noise = r.normal(0, 0.015, n)
            hr = project_box_sum(best_h + noise, T)
            sr = exact_score(hr)
            if sr < best_s:
                best_h = hr
                best_s = sr
            rho = max(0.02, rho * 3.0)
            stall = 0
    
    # Extended pairwise polish
    if tleft() > 25:
        ph, ps = pairwise_polish(best_h, T, max_iters=500,
                                  time_limit=min(80, max(5, tleft() - 15)),
                                  t_start=time.time())
        if ps < best_s - 1e-12:
            best_h = ph
            best_s = ps
    
    # Random smooth seeds with quick SLP
    if tleft() > 40:
        for attempt in range(4):
            if tleft() < 20:
                break
            hr = generate_random_smooth(n, r, T)
            sr = exact_score(hr)
            if sr < best_s - 1e-12:
                best_h = hr
                best_s = sr
            # Quick SLP
            if tleft() > 30:
                ht = hr.copy()
                st = exact_score(ht)
                rho_t = 0.05
                for _ in range(40):
                    if tleft() < 20:
                        break
                    rh2, rs2 = slp_step_safe(ht, 40, rho_t, st + 0.05)
                    if rh2 is not None and rs2 < st - 1e-12:
                        ht = rh2
                        st = rs2
                        rho_t = max(0.003, rho_t * 0.85)
                    else:
                        rho_t = max(0.001, rho_t * 0.7)
                if st < best_s - 1e-12:
                    best_h = ht
                    best_s = st
    
    # Final pairwise
    if tleft() > 15:
        ph, ps = pairwise_polish(best_h, T, max_iters=200,
                                  time_limit=min(30, max(5, tleft() - 10)),
                                  t_start=time.time())
        if ps < best_s - 1e-12:
            best_h = ph
            best_s = ps
    
    best_h = project_box_sum(best_h, T)
    final_s = exact_score(best_h)
    return (best_h, final_s, len(best_h))
