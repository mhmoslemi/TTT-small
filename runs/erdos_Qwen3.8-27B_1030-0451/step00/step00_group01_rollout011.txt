A better approach: identify the active lags (those near the max) and make moves that specifically try to reduce those lags.

For lag k, the term is: T_k = sum_i h[i]*(1-h[i+k]) (for valid i).
To reduce T_k, I want to either decrease h[i] where h[i+k] is high, or increase h[i+k] where h[i] is high.

import numpy as np

def project_feasible(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2} via bisection on shift."""
    target = n / 2.0
    lo = z.min() - 1.5
    hi = z.max() + 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(z - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z - lam, 0.0, 1.0)
    r = target - h.sum()
    if r > 1e-10:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) > 0:
            order = idx[np.argsort(1.0 - h[idx])]
            for j in order:
                add = min(1.0 - h[j], r)
                h[j] += add
                r -= add
                if r < 1e-12:
                    break
    elif r < -1e-10:
        idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            order = idx[np.argsort(h[idx])]
            for j in order:
                rem = min(h[j], -r)
                h[j] -= rem
                r += rem
                if -r < 1e-12:
                    break
    return h

def compute_score(h):
    """Exact C5 score: max(correlate(h, 1-h, 'full')) * dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

    """Return the full correlation array (before dx scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')
