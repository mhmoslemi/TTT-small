```python
def project_feasible(v, target_sum):
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum}"""
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    # Distribute diff among coordinates with room
    if diff > 0:
        # Need to increase
        room = 1.0 - v
        # Add to coordinates with most room
        order = np.argsort(-room)
        idx = order[0]
        add = min(diff, room[idx])
        v[idx] += add
        diff -= add
        # If still more needed, distribute
        if diff > 1e-12:

                add = min(diff, room[order[i]])
                v[order[i]] += add
                diff -= add
                if diff < 1e-12:
                    break

        room = v
        order = np.argsort(room)
        idx = order[0]

        diff += sub
        if diff < -1e-12:
```

```python
    active_mask = c >= thresh
    active_indices = np.where(active_mask)[0]

        order = np.argsort(c[active_indices])[::-1][:top_k]

    tau = max(1e-3, 0.03 * (max_c - np.min(c_active)))
    weights = np.exp((c_active - max_c) / tau)
    weights /= weights.sum()
    
    return ells, weights
```

```python
def compute_gradient(h, active_q, weights):
    """Compute gradient of weighted sum of active correlations.
    active_q: array of q indices (0 to 2n-2)
    weights: corresponding weights
    """
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for q, w in zip(active_q, weights):
        ell = q - (n - 1)
        if ell == 0:
            g += w * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            a = ell
            if a < n:
                # c[q] = sum_{m=0}^{n-1-a} h[m+a]*(1-h[m])
                term1 = np.zeros(n)
                term1[a:] = 1.0 - h[:n-a]  # (1-h[t-a]) for t >= a
                term2 = np.zeros(n)
                term2[:n-a] = -h[a:]  # -h[t+a] for t < n-a
                g += w * dx * (term1 + term2)
        else:
            a = -ell
            if a < n:
                # c[q] = sum_{m=0}^{n-1-a} h[m]*(1-h[m+a])
                term1 = np.zeros(n)
                term1[:n-a] = 1.0 - h[a:]  # (1-h[t+a]) for t < n-a
                term2 = np.zeros(n)
                term2[a:] = -h[:n-a]  # -h[t-a] for t >= a
                g += w * dx * (term1 + term2)
    return g
```

```python
def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def compute_priority_gradient(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(c))
    thresh = c_max - 0.06 * c_max
    active = c >= thresh
    active_q = np.where(active)[0]
    if len(active_q) > 100:
        order = np.argsort(c[active_q])[::-1][:100]
        active_q = active_q[order]
    c_act = c[active_q]
    tau = max(1e-3, 0.05 * (c_max - float(c_act.min())))
    w = np.exp((c_act - c_max) / tau)
    w /= w.sum()
    g = np.zeros(n)
    for q, wi in zip(active_q, w):
        ell = int(q) - (n - 1)
        if ell == 0:
            g += wi * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            a = ell
            if a < n:
                g[a:] += wi * dx * (1.0 - h[:n - a])
                g[:n - a] += wi * dx * (-h[a:])
        else:
            a = -ell
            if a < n:
                g[:n - a] += wi * dx * (1.0 - h[a:])
                g[a:] += wi * dx * (-h[:n - a])
    return g

        return h, exact_score(h)
    best_score = exact_score(h)
    best_xi = h[i]
    xs = np.linspace(lo, hi, n_grid)

        h[i] = xi
        h[j] = s - xi

        if sc < best_score:
            best_score = sc
            best_xi = xi
```
