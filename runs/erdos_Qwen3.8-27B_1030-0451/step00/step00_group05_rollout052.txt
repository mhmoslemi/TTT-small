def pairwise_exchange(h, i, j, delta):
    """Perform exact pairwise exchange h[i]+=delta, h[j]-=delta."""

    # hj - delta >= 0 => delta <= hj
    # Min delta: hi + delta >= 0 => delta >= -hi

    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h, delta

    M = len(c)
    center = n - 1
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:top_k]
    M_val = c[active[-1]]
    spread = max(c[active[0]] - M_val, 1e-10)
    tau = spread * 0.5 + 1e-10
    weights = np.zeros(M)
    for idx in active:
        weights[idx] = math.exp((c[idx] - M_val) / tau)
    W = np.sum(weights)
    if W < 1e-30:
        weights[active[0]] = 1.0
    else:
        weights /= W
    g = np.zeros(n)
    # Zero lag
    if weights[center] > 1e-15:
        g += weights[center] * (1.0 - 2.0 * h)
    # Non-zero lags
    for r in range(1, n):
        L = n - r
        wp = weights[center + r]
        if wp > 1e-15:
            g[:L] += -wp * h[r:r+L]
            g[r:r+L] += wp * (1.0 - h[:L])
        wn = weights[center - r]
        if wn > 1e-15:
            g[:L] += wn * (1.0 - h[r:r+L])
            g[r:r+L] += -wn * h[:L]
    return g, weights

def verify_gradient(h, rng, top_k=15):
    g, _ = compute_smooth_gradient(h, top_k)
    n = len(h)
    # Check with finite difference
    eps = 1e-5
    i_test = int(rng.integers(0, n))
    h_plus = h.copy()
    h_minus = h.copy()
    j_comp = int(rng.integers(0, n))
    j_comp = (j_comp + 1) % n
    h_plus[i_test] += eps
    h_plus[j_comp] -= eps
    h_minus[i_test] -= eps
    h_minus[j_comp] += eps
    if h_plus[i_test] > 1 or h_minus[i_test] < 0 or h_plus[j_comp] < 0 or h_minus[j_comp] > 1:
        return True
    def smooth_obj(hh):
        _, w = compute_smooth_gradient(hh, top_k)
        n2 = len(hh)
        dx2 = 2.0 / n2
        c2 = np.correlate(hh, 1.0 - hh, mode="full") * dx2
        center2 = n2 - 1
        return float(np.sum(w * c2))
    fd = (smooth_obj(h_plus) - smooth_obj(h_minus)) / (2 * eps)
    gd = g[i_test] - g[j_comp]
    return abs(fd - gd) < 0.5 * (abs(fd) + abs(gd) + 1.0)

def optimize_local(h_init, time_limit, rng, current_best, top_k=15):
    n = len(h_init)
    h = np.array(h_init, dtype=np.float64).copy()
    h = project_to_feasible(h)
    local_score = exact_score(h)
    best_h = h.copy()
    best_score = local_score
    start_time = time.monotonic()
    sigma = 0.03
    T = 0.003
    sigma_min = 0.002
    T_min = 1e-5
    decay = 0.9997
    no_improve = 0
    use_grad = verify_gradient(h, rng, top_k)
    total_evals = 0

    while time.monotonic() - start_time < time_limit:
        if total_evals > 500000:
            break

        # Gradient step
        if use_grad and sigma > sigma_min * 2:
            g, _ = compute_smooth_gradient(h, top_k)
            direction = -(g - np.mean(g))
            dn = np.max(np.abs(direction))
            if dn > 1e-12:
                direction = direction / dn * sigma
                cand = h + direction
                cand = project_to_feasible(cand)
                if is_feasible(cand, 1e-6):
                    cs = exact_score(cand)
                    total_evals += 1
                    ds = cs - local_score
                    if cs < local_score or (T > T_min and float(rng.random()) < math.exp(-ds / T)):
                        h = cand
                        local_score = cs
                        if cs < best_score:
                            best_score = cs
                            best_h = h.copy()
                            no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1

        # Pairwise exchanges
        n_pairs = 8
        for _ in range(n_pairs):
            idxs = rng.integers(0, n, size=2)
            i, j = int(idxs[0]), int(idxs[1])
            if i == j:
                continue
            delta = float(rng.uniform(-sigma, sigma))
            hi, hj = h[i], h[j]
            lo = max(-hi, hj - 1.0)
            up = min(1.0 - hi, hj)
            d = max(lo, min(up, delta))
            if abs(d) < 1e-14:
                continue
            h[i] = hi + d
            h[j] = hj - d
            ns = exact_score(h)
            total_evals += 1
            ds = ns - local_score
            if ns < local_score or (T > T_min and float(rng.random()) < math.exp(-ds / T)):
                local_score = ns
                if ns < best_score:
                    best_score = ns
                    best_h = h.copy()
                    no_improve = 0
            else:
                no_improve += 1
                # Revert
                h[i] = hi
                h[j] = hj

        # Triple exchange occasionally
        if float(rng.random()) < 0.3:
            idxs = rng.integers(0, n, size=3)
            i, j, k = int(idxs[0]), int(idxs[1]), int(idxs[2])
            if len(set([i, j, k])) == 3:
                hi, hj, hk = h[i], h[j], h[k]
                d = float(rng.uniform(-sigma, sigma))
                lo1 = max(-hi, 0.0)
                up1 = min(1.0 - hi, 0.0)
                # Move d from i to j and k
                new_i = hi - d
                new_j = hj + d * 0.5
                new_k = hk + d * 0.5
                if 0 <= new_i <= 1 and 0 <= new_j <= 1 and 0 <= new_k <= 1:
                    h[i], h[j], h[k] = new_i, new_j, new_k
                    ns = exact_score(h)
                    total_evals += 1
                    ds = ns - local_score
                    if ns < local_score or (T > T_min and float(rng.random()) < math.exp(-ds / T)):
                        local_score = ns
                        if ns < best_score:
                            best_score = ns
                            best_h = h.copy()
                            no_improve = 0
                    else:
                        h[i], h[j], h[k] = hi, hj, hk
                        no_improve += 1

        sigma = max(sigma_min, sigma * decay)
        T = max(T_min, T * decay)

        if no_improve > 80:
            # Restart with perturbation
            for _ in range(5):
                i2, j2 = int(rng.integers(0, n)), int(rng.integers(0, n))
                if i2 != j2:
                    d2 = float(rng.uniform(-0.03, 0.03))
                    hi2, hj2 = h[i2], h[j2]
                    lo2 = max(-hi2, hj2 - 1.0)
                    up2 = min(1.0 - hi2, hj2)
                    d2 = max(lo2, min(up2, d2))
                    if abs(d2) > 1e-14:
                        h[i2] = hi2 + d2
                        h[j2] = hj2 - d2
            local_score = exact_score(h)
            total_evals += 1
            no_improve = 0
            sigma = max(sigma, 0.02)
            T = max(T, 0.002)

    return best_h, best_score

def make_binary_n(n):
    h = np.zeros(n, dtype=np.float64)
    m = n // 2
    h[:m] = 1.0
    if n % 2 == 1:
        h[m] = 0.5
    return h

def resample_interp(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_feasible(h_new)

def symmetrize_complement(h):
    n = len(h)
    s = h.copy()
    for i in range(n // 2):
        j = n - 1 - i
        avg = 0.5 * (s[i] + 1.0 - s[j])
        s[i] = avg
        s[j] = 1.0 - avg
    if n % 2 == 1:
        s[n // 2] = 0.5
    return project_to_feasible(s)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()

    def elapsed():
        return time.monotonic() - t_start

    def remaining():
        return budget_s - elapsed()

    # Get parent
    parent = None
    try:
        if 'initial_h_values' in globals():
            p = np.array(globals()['initial_h_values'], dtype=np.float64).copy()
            if p.size > 10 and np.all(np.isfinite(p)):
                parent = np.clip(p, 0.0, 1.0)
                parent = project_to_feasible(parent)
    except Exception:
        parent = None

    if parent is None:
        parent = make_binary_n(131)

    n_parent = len(parent)

    # Build diverse starting candidates
    starts = []
    starts.append(("parent", parent.copy()))
    starts.append(("rev", parent[::-1].copy()))
    starts.append(("comp", project_to_feasible(1.0 - parent)))
    starts.append(("sym", symmetrize_complement(parent)))
    starts.append(("rev_comp", project_to_feasible(1.0 - parent[::-1])))

    # Resampled to nearby lengths
    for nn in [n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4, n_parent + 8, n_parent - 8]:
        if nn >= 64 and nn <= 300:
            h_r = resample_interp(parent, nn)
            starts.append(("resample_%d" % nn, h_r))
            starts.append(("resample_sym_%d" % nn, symmetrize_complement(h_r)))

    # Binary patterns
    for nn in [n_parent, n_parent + 2]:
        h_b = make_binary_n(nn)
        starts.append(("binary_%d" % nn, h_b))

    # Evaluate and sort starts
    scored_starts = []
    for name, h in starts:
        s = exact_score(h)
        scored_starts.append((s, name, h))
    scored_starts.sort(key=lambda x: x[0])

    # Global tracking
    global_best_h = scored_starts[0][2].copy()
    global_best_score = scored_starts[0][0]

    # Reserve final time
    reserve = max(5.0, 0.04 * budget_s)

    # Phase 1: Optimize top starts
    n_top = min(6, len(scored_starts))
    time_per_start = (budget_s - reserve - 30) / max(1, n_top)

    for i in range(n_top):
        if remaining() < reserve + 20:
            break
        s0, name, h0 = scored_starts[i]
        t_alloc = min(time_per_start * (2 if i < 3 else 1), remaining() - reserve)
        if t_alloc < 5:
            break
        h_opt, s_opt = optimize_local(h0, t_alloc, rng, global_best_score, top_k=15)
        if s_opt < global_best_score:
            global_best_score = s_opt
            global_best_h = h_opt.copy()

    # Phase 2: Focus on global best
    if remaining() > reserve + 20:
        t_alloc = remaining() - reserve
        h_opt, s_opt = optimize_local(global_best_h, t_alloc, rng, global_best_score, top_k=12)
        if s_opt < global_best_score:
            global_best_score = s_opt
            global_best_h = h_opt.copy()

    # Phase 3: Try symmetrization of best
    if remaining() > reserve + 10:
        h_sym = symmetrize_complement(global_best_h)
        s_sym = exact_score(h_sym)
        if s_sym < global_best_score:
            global_best_score = s_sym
            global_best_h = h_sym.copy()
        if remaining() > reserve + 5:
            h_opt, s_opt = optimize_local(h_sym, remaining() - reserve, rng, global_best_score, top_k=10)
            if s_opt < global_best_score:
                global_best_score = s_opt
                global_best_h = h_opt.copy()

    # Final: ensure feasibility and compute exact score
    final_h = np.array(global_best_h, dtype=np.float64)
    final_h = project_to_feasible(final_h)
    final_score = exact_score(final_h)
    n_points = len(final_h)

    return final_h, final_score, n_points
