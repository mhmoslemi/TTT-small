def project_sum_box(x, target):
    """Project x onto {y : 0 <= y <= 1, sum(y) = target}."""
    n = len(x)
    # Bisection on lambda: find lambda such that sum(clip(x + lambda, 0, 1)) = target
    lo, hi = -x.max() - 1, 1 - x.min() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + (lo + hi) / 2, 0, 1)
    # Fix small numerical errors
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        # Proportional redistribution
        if diff > 0:
            mask = result < 1
            if mask.any():
                room = (1 - result)[mask]
                result[mask] += diff * room / room.sum()
        else:
            mask = result > 0
            if mask.any():
                room = result[mask]
                result[mask] += diff * room / room.sum()
