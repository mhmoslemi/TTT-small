```python
def project_box_hyperplane(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    
    def project_given_offset(offset):
        """Given offset, compute the projected values and their sum."""
        proj = np.clip(h + offset, 0, 1)
        return proj
    
    # Bisection on offset
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = project_given_offset(mid)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = project_given_offset((lo + hi) / 2)
    return proj
```

This is a standard technique. The projection onto {x : 0 ≤ x ≤ 1, sum(x) = S} is found by solving for the offset lambda such that sum(clip(x + lambda, 0, 1)) = S. Since the LHS is continuous and non-decreasing in lambda, bisection works.

```python
        if time_limit is not None and time.time() - start > time_limit:
            break
        
        frac = t / max(max_iters - 1, 1)
        T = T0 + (T_end - T0) * frac
        alpha = step0 / (1.0 + 0.01 * t)
        
        grad, _ = grad_smoothed(h, dx, n, T)
        
        momentum = beta * momentum - alpha * grad
        h_new = project_bh(h + momentum, n / 2.0)
        
        c5_new = eval_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            iter_since_improve = 0
        else:
            iter_since_improve += 1
        
        if iter_since_improve > 500:
            # Random restart
            h = project_bh(best_h + rng.normal(0, 0.01, n), n / 2.0)
            momentum = np.zeros(n)
            iter_since_improve = 0
        
        h = h_new
    
    return best_h, best_c5

def transfer(h, n_old, n_new):
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h)
    return project_bh(h_new, n_new / 2.0)

def ensure_exact(h, n):
    h = h.copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] += diff
    return np.clip(h, 0.0, 1.0)
```
