    p = np.clip(p, 0.0, 1.0)

        h[m+1:] = 1.0 - p[::-1]
    else:

        p = 0.5 * (h[:m] + 1.0 - h[m+1:][::-1])
    else:
        p = 0.5 * (h[:m] + 1.0 - h[m:][::-1])

    cands.append(p_res)
    cands.append(np.full(m, 0.5))
    cands.append(np.clip(rng.uniform(0.1, 0.9, m), 0, 1))
    x = np.linspace(0, 1, m)
    for freq in [1, 2, 3, 5]:
        for amp in [0.2, 0.4]:
            cands.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x), 0, 1))
    for i in range(3):
        p_step = np.full(m, 0.5)
        thr = rng.uniform(0.3, 0.7)
        p_step[x < thr] = rng.uniform(0.6, 1.0)
        p_step[x >= thr] = rng.uniform(0.0, 0.4)
        cands.append(np.clip(p_step, 0, 1))
    return cands

def _optimize_candidate(p0, n, state, deadline, seed_val):
    m = len(p0)
    p0 = np.clip(p0, 0.0, 1.0)
    p_cur = p0.copy()

    tau_seq = [0.05, 0.025, 0.0125, 0.007, 0.004, 0.002, 0.001]
    bounds = [(0.0, 1.0)] * m
    max_gsteps = max(20, min(80, 4000 // max(1, 2 * m + 1)))

    for tau in tau_seq:
        if state['done'] or _time.monotonic() > deadline - 30:
            break
        maxfun = max_gsteps * (2 * m + 1)
        maxfun = min(maxfun, 30000)
        try:
            res = minimize(
                fun=_lse_obj,
                x0=p_cur,
                args=(tau, n, state),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxfun': maxfun, 'ftol': 1e-14, 'maxiter': max_gsteps + 10,
                         'disp': False, 'eps': 1e-6}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_new = np.clip(res.x, 0.0, 1.0)
                h_new = _build_h(p_new, n)
                v_new = _exact_c5(h_new)
                if v_new < state['best_val'] - 1e-12:
                    state['best_val'] = v_new
                    state['best_h'] = h_new.copy()
                p_cur = p_new
        except Exception:
            pass

    # SLSQP polish
    if not state['done'] and _time.monotonic() < deadline - 20:
        h_cur = _build_h(p_cur, n)
        dx = 2.0 / n
        f_all = np.correlate(h_cur, 1.0 - h_cur, mode="full") * dx
        K = min(40, len(f_all), 2 * m + 10)
        idx = np.argsort(f_all)[::-1][:K]

        _slq_ctx['n'] = n
        _slq_ctx['idx'] = idx

        v_cur = float(np.max(f_all))
        z0 = np.concatenate([p_cur, [v_cur + 1e-5]])
        bounds_z = [(0.0, 1.0)] * m + [(0.0, 1.0)]

        try:
            res = minimize(
                fun=_slq_obj,
                x0=z0,
                method='SLSQP',
                bounds=bounds_z,
                constraints=[{'type': 'ineq', 'fun': _slq_con}],
                options={'maxiter': 80, 'ftol': 1e-12, 'disp': False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                p_res = np.clip(res.x[:-1], 0.0, 1.0)
                h_res = _build_h(p_res, n)
                v_res = _exact_c5(h_res)
                if v_res < state['best_val'] - 1e-12:
                    state['best_val'] = v_res
                    state['best_h'] = h_res.copy()
                    p_cur = p_res
        except Exception:
            pass

    return p_cur

def run(seed=42, budget_s=1100, **kwargs):
    t_start = _time.monotonic()
    deadline = t_start + budget_s * 0.94

    state = {
        'best_val': 1.0,
        'best_h': None,
        'deadline': deadline,
        'done': False
    }

    rng = np.random.default_rng(seed)
    parent_h = None

    try:
        if 'initial_h_values' in globals():
            iv = initial_h_values
            if iv is not None:
                parent_h = np.array(iv, dtype=np.float64).copy()
                if parent_h.size > 0 and np.all(np.isfinite(parent_h)):
                    parent_h = np.clip(parent_h, 0.0, 1.0)
                    pv = _exact_c5(parent_h)
                    state['best_val'] = pv
                    state['best_h'] = parent_h.copy()
    except Exception:
        pass

    if state['best_h'] is None:
        h0 = np.full(128, 0.5)
        state['best_val'] = _exact_c5(h0)
        state['best_h'] = h0.copy()

    parent_n = len(state['best_h'])
    parent_p = _p_from_h(state['best_h'])

    dims = sorted(set([64, 80, 96, 128, 160, 192, 256, 320, 384, 512]))
    if parent_n not in dims:
        dims.append(parent_n)
    dims = [d for d in dims if d <= 512]

    # Generate and evaluate all candidates
    all_cands = []
    for n in dims:
        p_list = _gen_candidates(n, rng, parent_p)
        for p_cand in p_list:
            h_cand = _build_h(p_cand, n)
            v_cand = _exact_c5(h_cand)
            all_cands.append((v_cand, n, p_cand.copy()))

    all_cands.sort(key=lambda x: x[0])

    # Phase 1: Optimize top candidates
    top_k = min(len(all_cands), 25)
    n_per_dim = {}

    for ci in range(top_k):
        if state['done'] or _time.monotonic() > deadline - 80:
            break
        v0, n0, p0 = all_cands[ci]
        n_per_dim[n0] = n_per_dim.get(n0, 0) + 1
        _optimize_candidate(p0, n0, state, deadline, seed + ci)

    # Phase 2: Refine best solution with smaller perturbations and SLSQP
    if not state['done'] and _time.monotonic() < deadline - 20 and state['best_h'] is not None:
        best_n = len(state['best_h'])
        best_p = _p_from_h(state['best_h'])

        for _ in range(8):
            if _time.monotonic() > deadline - 20:
                break
            h_cur = _build_h(best_p, best_n)
            dx = 2.0 / best_n
            f_all = np.correlate(h_cur, 1.0 - h_cur, mode="full") * dx
            v_cur = float(np.max(f_all))

            # SLSQP with more constraints
            K = min(80, len(f_all), 3 * best_n)
            idx = np.argsort(f_all)[::-1][:K]
            _slq_ctx['n'] = best_n
            _slq_ctx['idx'] = idx
            m = len(best_p)
            z0 = np.concatenate([best_p, [v_cur + 1e-5]])
            bounds_z = [(0.0, 1.0)] * m + [(0.0, 1.0)]
            try:
                res = minimize(
                    fun=_slq_obj, x0=z0, method='SLSQP',
                    bounds=bounds_z,
                    constraints=[{'type': 'ineq', 'fun': _slq_con}],
                    options={'maxiter': 120, 'ftol': 1e-13, 'disp': False}
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    p_res = np.clip(res.x[:-1], 0.0, 1.0)
                    h_res = _build_h(p_res, best_n)
                    v_res = _exact_c5(h_res)
                    if v_res < state['best_val'] - 1e-12:
                        state['best_val'] = v_res
                        state['best_h'] = h_res.copy()
                        best_p = p_res
                    else:
                        break
            except Exception:
                break

        # Try dimension increase
        if _time.monotonic() < deadline - 30:
            for n_big in [best_n * 2, best_n + 64, best_n + 128]:
                if n_big > 512 or _time.monotonic() > deadline - 30:
                    break
                m_big = n_big // 2
                p_big = np.interp(
                    (np.arange(m_big) + 0.5) / m_big,
                    (np.arange(len(best_p)) + 0.5) / len(best_p),
                    best_p
                )
                p_big = np.clip(p_big, 0, 1)
                h_big = _build_h(p_big, n_big)
                v_big = _exact_c5(h_big)
                if v_big < state['best_val'] + 0.01:
                    _optimize_candidate(p_big, n_big, state, deadline, seed + 999)

    # Final validation
    final_h = np.clip(state['best_h'], 0.0, 1.0)
    final_n = len(final_h)
    if abs(float(np.sum(final_h)) - final_n / 2.0) > 1e-8:
        final_h = _project_full(final_h)
    final_c5 = _exact_c5(final_h)

    return (final_h, final_c5, final_n)
