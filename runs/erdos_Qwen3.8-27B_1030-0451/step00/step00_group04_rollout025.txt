def exact_score(h):

    n = len(h)

def compute_gradient(h, active_lags):
    """Compute subgradient of C at h using active lags.
    active_lags: array of indices into the full correlation output (0 to 2n-2)
    Returns: gradient vector of length n
    """

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(x, n):
    """Project x to [0,1]^n with sum = n/2"""
    T = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if abs(s - T) < 1e-12:
        return x
    # Need to add (T - s) total, distribute to coordinates with room
    diff = T - s
    if diff > 0:
        # Need to increase sum; use coordinates not at 1
        room = 1.0 - x
        total_room = np.sum(room)
        if total_room < 1e-12:
            return x  # can't fix
        x = x + diff * room / total_room
    else:
        # Need to decrease sum; use coordinates not at 0
        space = x
        total_space = np.sum(space)
        if total_space < 1e-12:
            return x
        x = x + diff * space / total_space
    x = np.clip(x, 0.0, 1.0)
    # Final adjustment for rounding
    residual = T - np.sum(x)
    # Distribute residual to coordinates with room
    if residual > 0:
        idx = np.argsort(1.0 - x)[-int(np.ceil(abs(residual))):]
        for i in idx:
            add = min(residual, 1.0 - x[i])
            x[i] += add
            residual -= add
    elif residual < 0:
        idx = np.argsort(x)[-int(np.ceil(abs(residual))):]
        for i in idx:
            sub = min(-residual, x[i])
            x[i] -= sub
            residual += sub
    return x

def compute_subgradient(h, top_k=20, temperature=0.01):
    """Compute a subgradient of C(h) using top-k active lags with softmax weighting."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # Softmax weights over lags
    weights = np.exp((corr - max_corr) / temperature)
    total_w = np.sum(weights)
    if total_w < 1e-15:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= total_w
    
    grad = np.zeros(n)
    # Only use lags with significant weight
    active = np.where(weights > 1e-10)[0]
    for k in active:
        ell = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        if ell == 0:
            # R(0) = sum h[i]*(1-h[i])
            # dR/dh[j] = (1-2*h[j])
            grad += w * (1.0 - 2.0 * h)
        elif ell > 0:
            # dR/dh[j] = (1-h[j+ell]) for j < n-ell, minus h[j-ell] for j >= ell
            g = np.zeros(n)
            g[:n-ell] += (1.0 - h[ell:])
            g[ell:] += -h[:n-ell]
            grad += w * g
        else:
            ep = -ell
            g = np.zeros(n)
            g[ep:] += (1.0 - h[:n-ep])
            g[:n-ep] += -h[ep:]
            grad += w * g
    
    return grad, corr, max_corr

def local_search(h, max_time, rng):
    """Perform local search on h, return improved h and score."""
    n = len(h)
    h = h.copy()
    current_c = exact_score(h)
    best_local = h.copy()
    best_local_c = current_c
    start = time.time()
    
    no_improve_count = 0
    max_no_improve = 100
    
    while time.time() - start < max_time and no_improve_count < max_no_improve:
        # Compute subgradient
        grad, corr, max_corr = compute_subgradient(h)
        
        # Rank coordinates: high gradient = should decrease, low gradient = should increase
        grad_rank = np.argsort(grad)  # low to high
        # Top candidates for decrease: high grad (end of sorted)
        # Top candidates for increase: low grad (beginning of sorted)
        
        # Generate candidate pairs
        n_pairs = 30
        candidates = []
        
        # Strategy 1: gradient-based pairs
        n_per_side = n_pairs // 2
        high_idx = grad_rank[-n_per_side:]  # high gradient (decrease)
        low_idx = grad_rank[:n_per_side]    # low gradient (increase)
        
        for i in range(min(len(high_idx), n_per_side)):
            for j in range(min(len(low_idx), n_per_side)):
                a = high_idx[i]
                b = low_idx[j]
                if a != b:
                    candidates.append((a, b, 1.0))  # decrease a, increase b
        
        # Strategy 2: random pairs
        n_random = n_pairs // 2
        for _ in range(n_random):
            a = rng.integers(0, n)
            b = rng.integers(0, n)
            if a != b:
                if grad[a] > grad[b]:
                    candidates.append((a, b, 1.0))
                else:
                    candidates.append((b, a, 1.0))
        
        if not candidates:
            no_improve_count += 1
            continue
        
        # Try each candidate pair
        best_improvement = 0.0
        best_move = None
        
        for a, b, direction in candidates:
            # Transfer: h[a] -= tau, h[b] += tau (direction=1 means decrease a)

    if abs(diff) < 1e-14:
        return x

        room = 1.0 - x
        total = room.sum()
        if total > 1e-14:
            x += diff * room / total
    else:
        space = x.copy()
        total = space.sum()
        if total > 1e-14:
            x += diff * space / total

                residual += sub

            ep = -ell
            grad[ep:] += (1.0 - h[:n - ep])
            grad[:n - ep] += -h[ep:]
    grad /= n_active
    return grad

def try_pair(h, a, b, n_eval=10):
    """Try transferring from a to b. Return (best_tau, best_score) or (0, inf)."""
    tau_hi = min(h[a], 1.0 - h[b])
    if tau_hi < 1e-12:
        return 0.0, float('inf')
    best_tau = 0.0
    best_c = float('inf')
    for tau in np.linspace(0, tau_hi, n_eval)[1:]:
        h[a] -= tau
        h[b] += tau
        c = exact_score(h)
        h[a] += tau
        h[b] -= tau
        if c < best_c:
            best_c = c
            best_tau = tau
    return best_tau, best_c

def do_local_search(h, current_c, deadline, rng, max_no_improve=300):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = current_c
    no_improve = 0
    dx = 2.0 / n

    while time.time() < deadline:
        corr = compute_corr(h)
        grad = compute_subgradient(h, corr)

        # Build candidate pairs
        n_g_pairs = 25
        n_r_pairs = 15
        candidates = []

        sorted_g = np.argsort(grad)
        high = sorted_g[-n_g_pairs:]
        low = sorted_g[:n_g_pairs]

        for a in high:
            for b in low[:n_g_pairs]:
                if a != b:
                    candidates.append((int(a), int(b)))
                    candidates.append((int(b), int(a)))
                    if len(candidates) >= n_g_pairs * 2:
                        break
            if len(candidates) >= n_g_pairs * 2:
                break

        for _ in range(n_r_pairs):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                if grad[a] >= grad[b]:
                    candidates.append((a, b))
                else:
                    candidates.append((b, a))

        # Evaluate candidates
        best_improve = 0.0
        best_move = None

        for a, b in candidates:
            tau_hi = min(h[a], 1.0 - h[b])
            if tau_hi < 1e-10:
                continue
            # Quick gradient check
            if grad[a] - grad[b] < -0.01:
                continue
            n_eval = 8
            tau_hi2 = min(h[a], 1.0 - h[b])
            best_tau = 0.0
            best_c_eval = current_c
            for tau in np.linspace(0, tau_hi2, n_eval)[1:]:
                h[a] -= tau
                h[b] += tau
                c = exact_score(h)
                h[a] += tau
                h[b] -= tau
                if c < best_c_eval:
                    best_c_eval = c
                    best_tau = tau
            if best_tau > 0 and best_c_eval < current_c:
                imp = current_c - best_c_eval
                if imp > best_improve:
                    best_improve = imp
                    best_move = (a, b, best_tau)

        if best_move is not None and best_improve > 1e-14:
            a, b, tau = best_move
            h[a] -= tau
            h[b] += tau
            h = project_feasible(h, n)
            current_c = exact_score(h)
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                noise_scale = 0.005 + 0.01 * (no_improve // 200)
                noise = rng.normal(0, noise_scale, n)
                h_try = project_feasible(h + noise, n)
                c_try = exact_score(h_try)
                if c_try < current_c:
                    h = h_try
                    current_c = c_try
                no_improve += 10
            if no_improve > max_no_improve:
                break

    return best_h, best_c

def generate_fourier_start(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    n_modes = rng.integers(2, 6)
    for _ in range(n_modes):
        k = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        h += amp * np.sin(2 * np.pi * k * x / 2.0 + phase)
    return project_feasible(h, n)

def generate_block_start(n, rng):
    n_blocks = rng.integers(3, 10)
    block_size = n // n_blocks
    h = np.zeros(n)
    for i in range(n_blocks):
        val = rng.uniform(0.1, 0.9)
        start = i * block_size
        end = min(start + block_size, n)
        h[start:end] = val
    return project_feasible(h, n)

def upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_up = np.zeros(n_new)
    for i in range(n):
        start = i * factor
        end = min(start + factor, n_new)
        h_up[start:end] = h[i]
    return h_up

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get initial values
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    except (NameError, TypeError, ValueError):
        h0 = np.full(250, 0.5)

    n = len(h0)
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = exact_score(h)

    # Phase 1: Intensive local search from initial (60% of budget)
    t1 = t0 + budget_s * 0.55
    best_h, best_c = do_local_search(h, best_c, t1, rng, max_no_improve=500)

    # Phase 2: Alternative starts (20% of budget)
    t2 = t0 + budget_s * 0.75
    alt_starts = []
    # Reversed
    alt_starts.append(project_feasible(h[::-1].copy(), n))
    # Small noise perturbations
    for _ in range(3):
        noise = rng.normal(0, 0.02, n)
        alt_starts.append(project_feasible(h + noise, n))
    # Different constructions
    alt_starts.append(generate_fourier_start(n, rng))
    alt_starts.append(generate_block_start(n, rng))
    # Downsampled-then-upsampled for smoothing
    if n >= 50:
        small_n = n // 5
        h_small = np.zeros(small_n)
        for i in range(small_n):
            h_small[i] = h[i * 5: (i + 1) * 5].mean()
        h_small = project_feasible(h_small, small_n)
        alt_starts.append(upsample(h_small, 5))

    for s in alt_starts:
        if time.time() > t2:
            break
        sub_deadline = min(time.time() + 30, t2)
        h_new, c_new = do_local_search(s, exact_score(s), sub_deadline, rng, max_no_improve=200)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new

    # Phase 3: Try n*2 resolution if time permits (15% of budget)
    t3 = t0 + budget_s * 0.90
    if time.time() < t3 and budget_s > 300:
        n2 = n * 2
        h2 = upsample(best_h, 2)
        h2 = project_feasible(h2, n2)
        c2 = exact_score(h2)
        sub_deadline = min(t3, time.time() + budget_s * 0.12)
        if time.time() < sub_deadline:
            h2_new, c2_new = do_local_search(h2, c2, sub_deadline, rng, max_no_improve=300)
            if c2_new < best_c:
                best_h = h2_new
                best_c = c2_new

    # Phase 4: Final polish on best (remaining time)
    t_final = t0 + budget_s * 0.97
    if time.time() < t_final:
        best_h, best_c = do_local_search(best_h, best_c, t_final, rng, max_no_improve=200)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = exact_score(best_h)

    return (best_h, best_c, len(best_h))
