    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full') * dx
    m = len(corr)
    center = n - 1

    sorted_idx = np.argsort(corr)[::-1]

    top_vals = corr[top_idx]

        tau = max(0.02 * spread, 1e-6) if spread > 1e-8 else 1e-4

    exps = np.exp((top_vals - top_vals[0]) / tau)
    weights = exps / np.sum(exps)
    
    # Gradient accumulation
    grad = np.zeros(n)

        wl = w * dx

            L = n - R

                continue

            #   - h[k+R] for 0≤k≤L-1  [h[k] appears as h[i] with i=k]

            grad[:L] += wl * (1.0 - h[R:])
            grad[R:] += wl * (-h[:L])
    
    return grad, corr

        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        c5_new = exact_c5(h_new)
        if c5_new < c5 - 1e-14:
            h = h_new
            c5 = c5_new
            no_improve = 0
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            lr = min(lr * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve > 30:
                momentum = np.zeros(n)
                no_improve = 0
                tau *= 1.4
                tau = min(tau, 0.5)
                lr = max(0.001, lr * 0.7)
            elif no_improve > 10:
                lr = max(0.001, lr * 0.9)
    return best_h, best_c5

def optimize_exchanges(h_start, budget_s=200, seed=42):
    t0 = time.monotonic()
    n = len(h_start)
    h = h_start.copy()
    c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    rng = np.random.default_rng(seed)
    step0 = 0.04
    no_improve = 0
    while time.monotonic() - t0 < budget_s - 2:
        try:
            grad, corr = compute_smoothed_grad(h, top_k=15, tau=0.01)
        except:
            grad = np.zeros(n)
        sorted_idx = np.argsort(grad)
        ns = max(3, n // 8)
        candidates = []
        for hi in sorted_idx[-ns:]:
            for lo in sorted_idx[:ns]:
                if hi != lo and h[hi] > 0.02 and h[lo] < 0.98:
                    candidates.append((int(hi), int(lo), -1))
        for _ in range(4):
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                d = -1 if grad[i] > grad[j] else 1
                candidates.append((i, j, d))
        improved = False
        for i, j, direction in candidates:
            if time.monotonic() - t0 > budget_s - 2:
                break
            new_h, new_c5 = pair_exchange_ls(h, i, j, direction, step0=step0)
            if new_h is not None and new_c5 < c5 - 1e-12:
                h = new_h
                c5 = new_c5
                improved = True
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                break
        if not improved:
            no_improve += 1
            step0 = min(0.08, step0 * 1.15)
            if no_improve > 80:
                break
    return best_h, best_c5

def optimize_anneal(h_start, budget_s=150, seed=7):
    t0 = time.monotonic()
    n = len(h_start)
    h = h_start.copy()
    c5 = exact_c5(h)
    best_h = h.copy()
    best_c5 = c5
    rng = np.random.default_rng(seed)
    T = 0.004
    T_min = 5e-6
    T_decay = 0.999
    while time.monotonic() - t0 < budget_s - 2:
        sigma = 0.02 + 0.04 * min(T * 250, 1.0)
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        d = direction * sigma
        if direction > 0:
            d = min(d, 1.0 - h[i], h[j])
        else:
            d = -min(-d, h[i], 1.0 - h[j])
        if abs(d) < 1e-14:
            continue
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        c5_new = exact_c5(h_new)
        if c5_new <= c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            delta = c5_new - c5
            if rng.random() < math.exp(-delta / max(T, T_min)):
                h = h_new
                c5 = c5_new
        T = max(T_min, T * T_decay)
    return best_h, best_c5

def multi_phase_optimize(h_start, budget_s=300, seed=42):
    t0 = time.monotonic()
    h = h_start.copy()
    c5 = exact_c5(h)
    # Phase 1: PGD
    b1 = min(budget_s * 0.35, budget_s - 30)
    if b1 > 20:
        h1, c51 = optimize_pgd(h, budget_s=b1, top_k=20, tau_init=0.01, lr_init=0.005)
        if c51 < c5:
            h, c5 = h1, c51
    # Phase 2: Exchanges
    rem = budget_s - (time.monotonic() - t0)
    b2 = min(rem * 0.35, rem - 10)
    if b2 > 15:
        h2, c52 = optimize_exchanges(h, budget_s=b2, seed=seed)
        if c52 < c5:
            h, c5 = h2, c52
    # Phase 3: PGD with different params
    rem = budget_s - (time.monotonic() - t0)
    b3 = min(rem * 0.3, rem - 5)
    if b3 > 15:
        h3, c53 = optimize_pgd(h, budget_s=b3, top_k=30, tau_init=0.03, lr_init=0.003)
        if c53 < c5:
            h, c5 = h3, c53
    # Phase 4: Anneal
    rem = budget_s - (time.monotonic() - t0)
    b4 = max(0, rem - 3)
    if b4 > 10:
        h4, c54 = optimize_anneal(h, budget_s=b4, seed=seed + 1)
        if c54 < c5:
            h, c5 = h4, c54
    return h, c5

def try_structural_variants(h):
    n = len(h)
    results = {}
    results['orig'] = h.copy()
    results['mirror'] = h[::-1].copy()
    results['comp'] = (1.0 - h).copy()
    results['mirror_comp'] = (1.0 - h)[::-1].copy()
    cr = (1.0 - h)[::-1]
    results['sym'] = project_feasible((h + cr) / 2.0, n)
    return results

def resample_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def smooth_init(n, freq=1.0, amplitude=0.3):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + amplitude * np.sin(2 * math.pi * freq * x)
    return project_feasible(h, n)

def block_init(n, n_blocks=4):
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(0, n - n_blocks + 1, 2 * block_size):
        h[i:i + block_size] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.monotonic()
    DEADLINE = t_start + budget_s - 8

    def remaining():
        return max(0, DEADLINE - time.monotonic())

    # Get parent
    h_parent = None
    n_parent = 131
    try:
        iv = globals().get('initial_h_values', None)
        if iv is not None:
            h_parent = np.array(iv, dtype=np.float64).copy()
            n_parent = len(h_parent)
            h_parent = make_feasible(h_parent, n_parent)
    except:
        pass

    if h_parent is None:
        h_parent = np.zeros(n_parent)
        h_parent[:n_parent // 2] = 1.0
        if n_parent % 2 == 1:
            h_parent[n_parent // 2] = 0.5
        h_parent = make_feasible(h_parent, n_parent)

    # Initial best
    best_h = h_parent.copy()
    best_c5 = exact_c5(best_h)

    # Try structural variants of parent
    variants = try_structural_variants(best_h)
    for name, h_v in variants.items():
        c5_v = exact_c5(h_v)
        if c5_v < best_c5 - 1e-12:
            best_h = h_v.copy()
            best_c5 = c5_v

    # Build list of (n, initial_h) to optimize
    seeds = []
    seeds.append((n_parent, best_h.copy()))

    # Add resampled versions at nearby n
    for delta in [-2, -1, 1, 2]:
        n_try = n_parent + delta
        if 80 <= n_try <= 300:
            seeds.append((n_try, resample_to_n(best_h, n_try)))

    # Add some structured inits at various n
    for n_alt in [150, 200, 250]:
        if n_alt != n_parent and remaining() > 100:
            seeds.append((n_alt, resample_to_n(best_h, n_alt)))
            seeds.append((n_alt, smooth_init(n_alt, freq=2.0, amplitude=0.25)))

    # Optimize each seed
    n_seeds = len(seeds)
    base_budget = max(30, min(200, (budget_s * 0.7) / max(n_seeds, 1)))

    for idx, (n_s, h_s) in enumerate(seeds):
        if remaining() < 30:
            break
        # Prioritize parent and smaller n
        priority = 1.0
        if n_s == n_parent:
            priority = 1.5
        elif n_s < n_parent:
            priority = 1.2
        b = min(base_budget * priority, remaining() - 10)
        if b < 20:
            b = 20
        if remaining() - b < 15:
            continue
        try:
            h_opt, c5_opt = multi_phase_optimize(h_s, budget_s=b, seed=seed + idx)
            if c5_opt < best_c5 - 1e-12:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        except:
            pass

    # Final polish on best
    if remaining() > 30:
        b = min(remaining() - 5, 200)
        try:
            h_opt, c5_opt = multi_phase_optimize(best_h, budget_s=b, seed=seed + 100)
            if c5_opt < best_c5 - 1e-12:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        except:
            pass

    # Try structural variants of final best
    variants2 = try_structural_variants(best_h)
    for name, h_v in variants2.items():
        c5_v = exact_c5(h_v)
        if c5_v < best_c5 - 1e-12:
            best_h = h_v.copy()
            best_c5 = c5_v

    # Quick exchange polish
    if remaining() > 20:
        try:
            h_opt, c5_opt = optimize_exchanges(best_h, budget_s=min(remaining() - 3, 80), seed=seed + 200)
            if c5_opt < best_c5 - 1e-12:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        except:
            pass

    # Final feasibility check and evaluation
    best_h = np.asarray(best_h, dtype=np.float64)
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    c5_final = exact_c5(best_h)

    return (best_h, c5_final, n_final)
