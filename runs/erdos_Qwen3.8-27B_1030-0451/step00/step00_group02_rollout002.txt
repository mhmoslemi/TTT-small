            if s_d < best_score - 1e-13 or (abs(s_d - best_score) < 1e-7 and dn < best_n):
                best_h = h_d
                best_score = s_d
                best_n = dn
                try_update(best_h, best_score, best_n)

    # Phase 6: Random restarts if time remains
    if time.time() < deadline - 80:
        for ri in range(5):
            if time.time() > deadline - 30:
                break
            h_r = make_random_balanced(best_n, rng)
            # Quick SLSQP
            h_r = run_slsqp(h_r, 0.05, best_n, 100, deadline)
            if h_r is None:
                continue
            s_r = exact_score(h_r)
            if s_r < best_score:
                h_r = run_slsqp(h_r, 0.01, best_n, 100, deadline)
                if h_r is not None:
                    s_r = exact_score(h_r)
                h_r, s_r = pair_search(h_r, 500, 0.01, rng, deadline)
                if s_r < best_score - 1e-13:
                    best_h = h_r
                    best_score = s_r
                    try_update(best_h, best_score, best_n)

    # Phase 7: Final polish
    if time.time() < deadline - 10:
        tau_final = 0.002
        h_new = run_slsqp(best_h.copy(), tau_final, best_n, 200, deadline)
        if h_new is not None:
            s_new = exact_score(h_new)
            if s_new < best_score - 1e-13:
                best_h = h_new
                best_score = s_new
                try_update(best_h, best_score, best_n)

    # Final verification and correction
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-12:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
