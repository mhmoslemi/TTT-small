import time

def compute_max_corr(h):
    """Compute C5 value and the full correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def optimize_binary_local(h, max_iters=500000, patience=5000):
    """Local search: try random swaps to reduce max correlation."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    _, corr = compute_max_corr(h)
    best_val = np.max(corr)
    no_improve = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if no_improve > patience:
            break
        
        # Pick random 1 and 0
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Swap
        h[i] = 0
        h[j] = 1
        
        _, new_corr = compute_max_corr(h)
        new_val = np.max(new_corr)
        
        if new_val < best_val:
            best_val = new_val
            no_improve = 0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Undo swap
            h[i] = 1
            h[j] = 0
            no_improve += 1
    
    return h, best_val * dx

import numpy as np

def compute_c5(h):

def binarize_even(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b


def _random_binary(n):
    h = np.zeros(n)
    h[np.random.choice(n, n // 2, replace=False)] = 1.0
    return h


def _thue_morse_init(n):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=np.float64)
    n_ones = int(np.sum(tm))
    target = n // 2
    if n_ones == target:
        return tm
    elif n_ones > target:
        flips = n_ones - target
        one_idxs = np.flatnonzero(tm)
        to_flip = one_idxs[np.random.choice(flips, size=flips, replace=False)]
        tm[to_flip] = 0.0
    else:
        flips = target - n_ones
        zero_idxs = np.flatnonzero(1 - tm)
        to_flip = zero_idxs[np.random.choice(flips, size=flips, replace=False)]
        tm[to_flip] = 1.0
    return tm


def _resample_to_binary(init, n):
    n_init = len(init)
    x_old = np.linspace(0, 2, n_init, endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    h_float = np.interp(x_new, x_old, init)
    return binarize_even(h_float)


def local_search_binary(h, max_time=10, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.astype(np.float64).copy()
    dx = 2.0 / n

    _, corr = compute_c5(h)
    best_val = float(np.max(corr))

    start = time.time()
    stagnation = 0
    max_stagnation = max(2000, n * 15)

    while time.time() - start < max_time:
        one_idx = np.flatnonzero(h)
        zero_idx = np.flatnonzero(1 - h)
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break

        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]

        h[i] = 0.0

def focused_local_search(h, max_time=10, seed=0):

def simulated_annealing(h, max_time=10, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.astype(np.float64).copy()
    dx = 2.0 / n

    _, corr = compute_c5(h)
    current_val = float(np.max(corr))
    best_val = current_val
    best_h = h.copy()

    T = 2.0
    cool_rate = 0.99995
    start = time.time()

    while time.time() - start < max_time:
        one_idx = np.flatnonzero(h)
        zero_idx = np.flatnonzero(1 - h)
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break

        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]

        h[i] = 0.0
        h[j] = 1.0
        _, new_corr = compute_c5(h)
        new_val = float(np.max(new_corr))

        delta = new_val - current_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 0.01)):
            current_val = new_val
            corr = new_corr
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

        T *= cool_rate

    return best_h, best_val * dx
