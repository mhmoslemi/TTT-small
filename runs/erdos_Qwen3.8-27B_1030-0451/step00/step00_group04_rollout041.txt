    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': lambda x: _slsqp_eq_fun(x, T),
         'jac': lambda x: _slsqp_eq_jac(x)},
        {'type': 'ineq', 'fun': lambda x: _slsqp_ineq_fun(x, T, n, dx, idx),
         'jac': lambda x: _slsqp_ineq_jac(x, T, n, dx, idx)}
    ]
    try:
        res = minimize(_slsqp_obj, x0, jac=_slsqp_obj_g, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 300, 'ftol': 1e-10})
        if res.success and time.time() < deadline:
            hn = project_box_sum(res.x[:n].copy(), T)
            s = evaluate_c5(hn)
            if s < c0:
                return hn, s
    except Exception:
        pass
    return h0, c0

def make_variants(h, n, T, rng):
    vs = [h.copy()]
    vs.append(h[::-1].copy())
    vs.append((1.0 - h).copy())
    vs.append((1.0 - h)[::-1].copy())
    s = int(rng.integers(1, n))
    vs.append(np.roll(h, s).copy())
    vs.append(project_box_sum(h + rng.normal(0, 0.03, n), T))
    vs.append(project_box_sum(h + rng.normal(0, 0.08, n), T))
    # Block-wise perturbation
    bw = max(3, n // 20)
    nblocks = n // bw
    block_shifts = rng.normal(0, 0.05, nblocks)
    perturbed = h.copy()
    for i in range(nblocks):
        perturbed[i*bw:(i+1)*bw] += block_shifts[i]
    vs.append(project_box_sum(perturbed, T))
    return vs

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    try:
        h_parent = np.asarray(initial_h_values, dtype=np.float64).flatten().copy()
    except (NameError, AttributeError, ValueError):
        h_parent = None

    if h_parent is None or len(h_parent) < 10:
        n = 250
        h_parent = 0.5 * np.ones(n)

    n = len(h_parent)
    T = n / 2.0
    h_parent = project_box_sum(h_parent, T)

    best_h = h_parent.copy()
    best_s = evaluate_c5(h_parent)

    def tl():
        return budget_s - (time.time() - t_start)

    # Phase 1: Main PGD on parent
    if tl() > 60:
        d1 = time.time() + min(250, tl() - 60)
        best_h, best_s = pgd_h(h_parent, T, 0.015, 0.0005, 500, 0.4, d1)

    # Phase 2: SLSQP polish
    if tl() > 40:
        h2, s2 = slsqp_polish(best_h, T, time.time() + min(80, tl() - 30))
        if s2 < best_s:
            best_s, best_h = s2, h2

    # Phase 3: Try variants
    if tl() > 60:
        variants = make_variants(best_h, n, T, rng)
        for v in variants[1:]:
            if tl() < 25:
                break
            d3 = time.time() + min(30, tl() - 15)
            hv, sv = pgd_h(v, T, 0.01, 0.0003, 250, 0.3, d3)
            if sv < best_s:
                best_s, best_h = sv, hv
            if tl() > 30:
                hs, sc = slsqp_polish(hv, T, time.time() + min(20, tl() - 10))
                if sc < best_s:
                    best_s, best_h = sc, hs

    # Phase 4: Upsample to 2n and optimize
    if tl() > 100 and n <= 600:
        n2 = n * 2
        T2 = n2 / 2.0
        h_up = np.repeat(best_h, 2)
        h_up = project_box_sum(h_up, T2)
        d4 = time.time() + min(70, tl() - 30)
        hu, su = pgd_h(h_up, T2, 0.01, 0.0005, 250, 0.3, d4)
        # Downsample back
        h_ds = hu.reshape(-1, 2).mean(axis=1)
        h_ds = project_box_sum(h_ds, T)
        s_ds = evaluate_c5(h_ds)
        if s_ds < best_s:
            best_s, best_h = s_ds, h_ds
        elif su < best_s and tl() > 20:
            # Keep upside if better
            best_s, best_h = su, hu

    # Phase 5: Try upsample from any good starting point with SLSQP
    if tl() > 60 and len(best_h) <= 600:
        n_up2 = len(best_h) * 2 if len(best_h) <= 500 else len(best_h)
        if n_up2 > len(best_h):
            T_up2 = n_up2 / 2.0
            h_u2 = np.repeat(best_h, n_up2 // len(best_h))
            if len(h_u2) > n_up2:
                h_u2 = h_u2[:n_up2]
            h_u2 = project_box_sum(h_u2, T_up2)
            d5 = time.time() + min(40, tl() - 20)
            h_u2o, s_u2o = pgd_h(h_u2, T_up2, 0.008, 0.0003, 150, 0.25, d5)
            h_d5 = h_u2o.reshape(-1, 2).mean(axis=1) if h_u2o.size % 2 == 0 else h_u2o
            h_d5 = project_box_sum(h_d5, len(h_d5) / 2.0)
            s_d5 = evaluate_c5(h_d5)
            if s_d5 < best_s:
                best_s, best_h = s_d5, h_d5

    # Phase 6: Final SLSQP polish
    if tl() > 25 and len(best_h) <= 1000:
        nf = len(best_h)
        hf, sf = slsqp_polish(best_h, nf / 2.0, time.time() + min(30, tl() - 5),
                             n_active=min(80, 2 * nf - 1))
        if sf < best_s:
            best_s, best_h = sf, hf

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    nf = len(best_h)
    best_h = project_box_sum(best_h, nf / 2.0)
    final_s = evaluate_c5(best_h)

    return (best_h, final_s, nf)
