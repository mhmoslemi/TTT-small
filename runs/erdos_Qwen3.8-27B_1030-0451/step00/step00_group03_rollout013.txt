_ACTIVE_LAGS = None

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def build_h(p):
    p = np.clip(p, 0.0, 1.0)
    return np.concatenate([p, 1.0 - p[::-1]])

def get_corr_vals(p):
    h = build_h(p)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max(vals, tau):
    vmax = np.max(vals)
    if tau < 1e-12:
        return float(vmax)
    s = (vals - vmax) / tau
    return float(vmax + tau * np.log(np.sum(np.exp(s))))

def _lse_obj(pv, tau):
    return smooth_max(get_corr_vals(pv), tau)

def resample_p(p, new_m):
    om = len(p)
    if om == new_m:
        return p.copy()
    ot = (np.arange(om) + 0.5) / om
    nt = (np.arange(new_m) + 0.5) / new_m
    return np.clip(np.interp(nt, ot, p), 0.0, 1.0)

def structured_p(m, seed, ncomp=6):
    rng = np.random.default_rng(seed)
    x = np.arange(m, dtype=np.float64) / m
    p = np.full(m, 0.5)
    for f in range(1, ncomp + 1):
        amp = rng.uniform(0.02, 0.15) / np.sqrt(f)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        p += amp * np.sin(2.0 * np.pi * f * x + ph)
    return np.clip(p, 0.0, 1.0)

def _slsqp_obj(x):
    return x[-1]

def _slsqp_con(x):
    pv = np.clip(x[:-1], 0.0, 1.0)
    fv = get_corr_vals(pv)
    t = x[-1]
    return float(np.min(t - fv[_ACTIVE_LAGS]))

def slsqp_optimize(p_init, top_k, maxiter):
    global _ACTIVE_LAGS
    m = len(p_init)
    fvals = get_corr_vals(p_init)
    _ACTIVE_LAGS = np.argsort(fvals)[-top_k:]
    c0 = float(np.max(fvals))
    x0 = np.concatenate([np.clip(p_init, 0.0, 1.0), [c0 + 1e-6]])
    bounds = [(0.0, 1.0)] * m + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': _slsqp_con}]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-8, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x[:-1], 0.0, 1.0)
    except Exception:
        pass
    return None

def project_full(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
