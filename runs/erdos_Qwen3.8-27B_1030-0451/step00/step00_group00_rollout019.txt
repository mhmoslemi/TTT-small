import time

def score_and_Q(h):

    Q = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n

    n = len(h)

    P = np.zeros(2 * n - 1)

    P[offset - t : offset - t + n] -= h
    return P
