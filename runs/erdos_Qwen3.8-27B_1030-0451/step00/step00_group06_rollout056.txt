def compute_c5(h):

def project(h, target_sum):
    lo = -1.0
    hi = 1.0

        mid = (lo + hi) / 2.0
        c = np.clip(h - mid, 0.0, 1.0)
        s = c.sum()
        if s > target_sum:
            lo = mid
        else:

def smooth_max_grad(h, alpha):
    N = len(h)
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    total = exp_vals.sum()

import numpy as np
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(overlaps) * dx)


def project(h, target_sum):
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        c = np.clip(h - mid, 0.0, 1.0)
        s = c.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def grad_smooth_max(h, alpha):
    N = len(h)
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    m = overlaps.max()
    exp_vals = np.exp(alpha * (overlaps - m))
    total = exp_vals.sum()
    weights = exp_vals / total

    threshold = weights.max() * 1e-12
    sig = np.where(weights > threshold)[0]

    grad = np.zeros(N)
    for k in sig:
        s = N - 1 - k
        w = weights[k]
        # First term: w*(1-h[j+s]) for 0<=j+s<N
        j_lo = max(0, -s)
        j_hi = min(N, N - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo + s:j_hi + s])
        # Second term: -w*h[j-s] for 0<=j-s<N
        j_lo2 = max(0, s)
        j_hi2 = min(N, N + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 - s:j_hi2 - s]
    return grad, float(m + np.log(total) / alpha)


def optimize_local(h_init, target_sum, alpha_start, alpha_end, lr_start,
                   n_iter, time_limit, start_time, binary_pressure=0.0):
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0

    for it in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        t = it / max(n_iter, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (1.0 + 0.3 * t) ** (-1.5)

        grad, _ = grad_smooth_max(h, alpha)

        if binary_pressure > 0.0:
            grad -= binary_pressure * (2.0 * h - 1.0)

        h_new = h - lr * grad
        h_new = project(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1

        h = h_new

    return best_h, best_c5


def resample(h, N_new):
    N = len(h)
    if N == N_new:
        return h.copy()
    x_old = (np.arange(N) + 0.5) / N * 2.0
    x_new = (np.arange(N_new) + 0.5) / N_new * 2.0
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])


def random_start(N, rng):
    h = rng.rand(N)
    h = project(h, N / 2.0)
    return h


def oscillatory_start(N, freq, rng):
    x = np.linspace(0, 2, N, endpoint=False) + 1.0 / N

    h = np.clip(h, 0.0, 1.0)
    h = project(h, N / 2.0)
    return h
