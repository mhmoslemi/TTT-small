    T = x[n]

    obj = T

    grad = np.zeros(len(x))

def c5_constraints_fun(x, n, active_lags):
    """
    Returns constraint values (should be >= 0)
    Constraint: T - dx * corr_k >= 0
    """
    h = x[:n]
    T = x[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    ResultSet = np.zeros(len(active_lags))
    for i, k in enumerate(active_lags):
        ResultSet[i] = T - dx * corr[k]
    return ResultSet

def c5_constraints_grad(x, n, active_lags):

    Gradient of constraints

        G = np.zeros((n_act, n + 1))
        for i, m in enumerate(active_lags):
            k = m - (n - 1)
            for j in range(n):
                val = 0.0
                jk = j - k
                if 0 <= jk <= n - 1:
                    val += (1.0 - h[jk])
                jk2 = j + k
                if 0 <= jk2 <= n - 1:
                    val -= h[jk2]
                G[i, j] = -dx * val
            G[i, n] = 1.0
        return G
    
    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    try:
        result = minimize(obj, x0, jac=obj_jac, bounds=bounds,
                          constraints=constraints, method='SLSQP',
                          options={'maxiter': maxiter, 'ftol': 1e-14})
    except Exception:
        return None, None
    
    if result is not None and result.x is not None:
        h_new = project_feasible(result.x[:n])
        c_new = full_score(h_new)
        return h_new, c_new
    
    return None, None
