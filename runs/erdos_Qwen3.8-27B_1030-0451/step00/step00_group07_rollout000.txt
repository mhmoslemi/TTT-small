            if abs(diff) < 1e-14:
                break
        result = np.clip(result, 0.0, 1.0)
    return result

def shift_gradient_vec(h, s, N):
    g = np.zeros(N)
    if s >= 0:
        if N - s > 0:
            g[:N-s] += (1.0 - h[s:N])
            g[s:] -= h[:N-s]
    else:
        t = -s
        if N - t > 0:
            g[t:] += (1.0 - h[:N-t])
            g[:N-t] -= h[t:]
    return g

def transfer_to_resolution(h_base, n_base, n_new):
    """Transfer h from n_base to n_new resolution via cell averaging."""
    dx_base = 2.0 / n_base
    dx_new = 2.0 / n_new
    # For each new cell i, compute the average value
    # New cell i covers [i*dx_new, (i+1)*dx_new]
    # In base grid, this spans base cells from floor(i*dx_new/dx_base) to ceil((i+1)*dx_new/dx_base)-1
    # Compute overlap
    h_new = np.zeros(n_new)
    for i in range(n_new):
        x_start = i * dx_new
        x_end = (i + 1) * dx_new
        # Find overlapping base cells
        i_base_start = int(np.floor(x_start / dx_base + 1e-12))
        i_base_end = int(np.ceil(x_end / dx_base - 1e-12)) - 1
        i_base_start = max(0, min(i_base_start, n_base - 1))
        i_base_end = max(0, min(i_base_end, n_base - 1))
        total_area = 0.0
        for j in range(i_base_start, i_base_end + 1):
            # Base cell j covers [j*dx_base, (j+1)*dx_base]
            overlap_start = max(x_start, j * dx_base)
            overlap_end = min(x_end, (j + 1) * dx_base)
            if overlap_end > overlap_start:
                total_area += h_base[j] * (overlap_end - overlap_start)
        if dx_new > 0:
            h_new[i] = total_area / dx_new
    return h_new

def optimize_smooth(h, N, deadline, tau, lr, n_iter, target):
    """Projected gradient descent with log-sum-exp smoothing."""
    best_h = h.copy()
    best_val = evaluate_exact(h)
    momentum = np.zeros(N)
    beta = 0.9
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Compute objective
        E = evaluate_array(h)
        max_E = np.max(E)
        
        # Log-sum-exp weights
        shifted_E = tau * (E - max_E)
        shifted_E = np.clip(shifted_E, -100, 100)  # numerical stability
        w = np.exp(shifted_E)
        w_sum = np.sum(w)
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.zeros(2*N-1)
            w[np.argmax(E)] = 1.0
        
        # Compute gradient
        shifts = np.arange(2*N-1) - (N-1)
        grad = np.zeros(N)
        for i in range(len(E)):
            if w[i] > 1e-10:
                grad += w[i] * shift_gradient_vec(h, shifts[i], N)
        grad *= (2.0 / N)
        
        # Momentum update
        momentum = beta * momentum - lr * grad
        h_new = project_to_feasible(h + momentum, target)
        
        val = evaluate_exact(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val

def random_perturb(h, target, sigma, rng):
    """Add random perturbation and project."""
    noise = rng.normal(0, sigma, len(h))
    return project_to_feasible(h + noise, target)

def mix_two(h1, h2, alpha, target):
    """Mix two feasible vectors."""
    if len(h1) != len(h2):
        return project_to_feasible(h1, target)
    return project_to_feasible(alpha * h1 + (1-alpha) * h2, target)

import numpy as np
import time

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def evaluate_array(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_to_feasible(x, target):
    x = np.asarray(x, dtype=np.float64).copy()
    N = len(x)
    x[~np.isfinite(x)] = 0.5
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    # Fine adjustment
    r = target - np.sum(result)
    if abs(r) > 1e-12:
        interior = (result > 1e-10) & (result < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            result[interior] += r / n_int
            result = np.clip(result, 0.0, 1.0)
    # Final exact fix
    r = target - np.sum(result)
    if abs(r) > 1e-8:
        order = np.argsort(1.0 - result) if r > 0 else np.argsort(result)
        for idx in order:
            if r > 0:
                a = min(r, 1.0 - result[idx])
                result[idx] += a
                r -= a
            else:
                a = min(-r, result[idx])
                result[idx] -= a
                r += a
            if abs(r) < 1e-15:
                break
        result = np.clip(result, 0.0, 1.0)
    return result

def all_shift_gradients(h, N):
    """Compute gradients for all shifts. Returns array of shape (2N-1, N)."""
    dx = 2.0 / N
    grads = np.zeros((2*N-1, N))
    for k in range(2*N-1):
        s = k - (N-1)
        if s >= 0:
            if N - s > 0:
                grads[k, :N-s] += (1.0 - h[s:N])
                grads[k, s:] -= h[:N-s]
        else:
            t = -s
            if N - t > 0:
                grads[k, t:] += (1.0 - h[:N-t])
                grads[k, :N-t] -= h[t:]
    return grads * dx

def smooth_gradient(h, tau, N):
    """Compute gradient of log-sum-exp approximation."""
    E = evaluate_array(h)
    max_E = np.max(E)
    shifted = tau * (E - max_E)
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        # Fall back to uniform over ties
        w = np.zeros(2*N-1)
        w[E >= max_E - 1e-15] = 1.0
        w /= np.sum(w)
    else:
        w = w / sw
    
    # Weighted gradient
    shifts = np.arange(2*N-1) - (N-1)
    grad = np.zeros(N)
    dx = 2.0 / N
    for i in range(2*N-1):
        if w[i] > 1e-15:
            s = shifts[i]
            if s >= 0:
                if N - s > 0:
                    grad[:N-s] += w[i] * (1.0 - h[s:N])
                    grad[s:] -= w[i] * h[:N-s]
            else:
                t = -s
                if N - t > 0:
                    grad[t:] += w[i] * (1.0 - h[:N-t])
                    grad[:N-t] -= w[i] * h[t:]
    return grad * dx, E

    """Gradient using only top K shifts."""
    E = evaluate_array(h)
    idx = np.argsort(E)[::-1][:K]
    max_E = E[idx[0]]
    # Equal weight among top K
    shifts = np.arange(2*N-1) - (N-1)
    grad = np.zeros(N)
    dx = 2.0 / N
    for i in idx:

        if
