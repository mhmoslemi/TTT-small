    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    lo = -x.max() - 1.0
    hi = 1.0 - x.min()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))

            idx = np.argmax(h)
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    dx = 2.0 / len(h)
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(vals))

def get_corr_values(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_obj(x, beta):
    v = get_corr_values(x)
    m = np.max(v)
    sh = np.clip(beta * (v - m), -500, 500)
    e = np.exp(sh)
    return m - np.log(np.sum(e)) / beta

def smooth_grad(x, beta):
    dx = 2.0 / len(x)
    n = len(x)
    v = get_corr_values(x)
    m = np.max(v)
    sh = np.clip(beta * (v - m), -500, 500)
    e = np.exp(sh)
    w = e / np.sum(e)
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-14:
            continue
        k = mi - (n - 1)
        s1 = max(0, -k)
        e1 = min(n, n - k)
        if e1 > s1:
            g[s1:e1] += wi * dx * (1.0 - x[s1 + k:e1 + k])
        s2 = max(0, k)
        e2 = min(n, n + k)
        if e2 > s2:
            g[s2:e2] -= wi * dx * x[s2 - k:e2 - k]
    return g

def obj_func(x, beta):
    return smooth_obj(x, beta)

def grad_func(x, beta):
    return smooth_grad(x, beta)

def eq_func(x, tsum):
    return x.sum() - tsum

def eq_jac_func(x, tsum):
    return np.ones(len(x))

def do_slsqp(h, beta, tsum, maxiter=200):
    n = len(h)
    try:
        res = minimize(obj_func, h, args=(beta,),
                       jac=grad_func, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_func,
                                     'jac': eq_jac_func, 'args': (tsum,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.x is not None:
            return project_sum_box(np.array(res.x, dtype=np.float64), tsum)
    except Exception:
        pass
    return h.copy()

def transfer(h, n_new, ts):
    n = len(h)
    xo = (np.arange(n) + 0.5) * (2.0 / n)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return project_sum_box(np.interp(xn, xo, h), ts)

def binarize(h, tsum):
    n = len(h)
    h_b = (h > 0.5).astype(np.float64)
    diff = int(round(tsum - h_b.sum()))
    if diff > 0:
        z = np.where(h_b == 0)[0]
        order = np.argsort(-h[z])
        for idx in order[:diff]:
            h_b[z[idx]] = 1.0
    elif diff < 0:
        o = np.where(h_b == 1)[0]
        order = np.argsort(h[o])
        for idx in order[:(-diff)]:
            h_b[o[idx]] = 0.0
    return h_b

def swap_search(h, tsum, deadline, max_sweeps=80):
    n = len(h)
    h = h.copy()
    c = evaluate_c5(h)
    ones = set(np.where(h >= 0.5)[0].tolist())
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = set(range(n)) - ones
        for i in sorted(ones):
            if time.time() > deadline:
                break
            for j in sorted(zeros):
                h[i] = 0.0
                h[j] = 1.0
                c2 = evaluate_c5(h)
                if c2 < c - 1e-15:
                    c = c2
                    ones.discard(i)
                    ones.add(j)
                    zeros.discard(j)
                    zeros.add(i)
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    h_out = np.zeros(n)
    h_out[list(ones)] = 1.0
    return h_out, c

def anneal_search(h, tsum, deadline, rng, max_iters=4000):
    n = len(h)
    h = h.copy()
    c = evaluate_c5(h)
    ones = set(np.where(h >= 0.5)[0].tolist())
    T = 0.08
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i_list = sorted(ones)
        if not i_list:
            break
        i = i_list[rng.integers(len(i_list))]
        z_list = [x for x in range(n) if x not in ones]
        if not z_list:
            break
        j = z_list[rng.integers(len(z_list))]
        h[i] = 0.0
        h[j] = 1.0
        c_new = evaluate_c5(h)
        d = c_new - c
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-5)):
            c = c_new
            ones.discard(i)
            ones.add(j)
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * 0.9985, 1e-5)
    h_out = np.zeros(n)
    h_out[list(ones)] = 1.0
    return h_out, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)

    try:
        h_best = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, UnboundLocalError, TypeError):
        h_best = np.full(84, 0.5)

    n = len(h_best)
    tsum = n / 2.0
    h_best = project_sum_box(h_best, tsum)
    c_best = evaluate_c5(h_best)

    def tl():
        return deadline - time.time()

    if tl() < 5:
        return h_best, c_best, n

    # Phase 1: SLSQP on original grid with increasing smoothness
    for beta in [4, 8, 16, 32, 64, 128, 256, 512]:
        if tl() < 30:
            break
        h_try = do_slsqp(h_best, beta, tsum, maxiter=200)
        c_try = evaluate_c5(h_try)
        if c_try < c_best - 1e-13:
            h_best = h_try
            c_best = c_try

    # Phase 2: Binary local search (greedy swaps)
    if tl() > 60:
        h_b = binarize(h_best, tsum)
        if h_b.sum() == tsum:
            c_b = evaluate_c5(h_b)
            if c_b <= c_best + 0.02:
                h_b, c_b = swap_search(h_b, tsum, deadline, max_sweeps=100)
                if c_b < c_best:
                    h_best = h_b
                    c_best = c_b

    # Phase 3: Resolution transfer to finer grids
    for n2 in [126, 168, 252]:
        if tl() < 90:
            break
        ts2 = n2 / 2.0
        h2 = transfer(h_best, n2, ts2)
        for beta in [8, 32, 128, 512]:
            if tl() < 30:
                break
            h2 = do_slsqp(h2, beta, ts2, maxiter=150)
            c2 = evaluate_c5(h2)
            if c2 < c_best - 1e-13:
                h_best = h2.copy()
                c_best = c2
        if tl() > 60:
            h2b = binarize(h2, ts2)
            if h2b.sum() == ts2:
                h2b, c2b = swap_search(h2b, ts2, deadline, max_sweeps=50)
                if c2b < c_best:
                    h_best = h2b
                    c_best = c2b

    # Phase 4: Random restarts with SLSQP
    for r in range(20):
        if tl() < 45:
            break
        scale = rng.uniform(0.05, 0.2)
        h_r = project_sum_box(h_best + scale * rng.standard_normal(n), tsum)
        if r % 4 == 3:
            h_r = project_sum_box(rng.uniform(0, 1, n), tsum)
        for beta in [32, 128, 512]:
            if tl() < 30:
                break
            h_r = do_slsqp(h_r, beta, tsum, maxiter=150)
            c_r = evaluate_c5(h_r)
            if c_r < c_best - 1e-13:
                h_best = h_r
                c_best = c_r

    # Phase 5: Simulated annealing on best binary
    if tl() > 30:
        hb = binarize(h_best, tsum)
        if hb.sum() == tsum:
            hb, cb = anneal_search(hb, tsum, deadline, rng, max_iters=5000)
            if cb < c_best:
                h_best = hb
                c_best = cb
            # Also try swap search after annealing
            if tl() > 20:
                hb2, cb2 = swap_search(hb, tsum, deadline, max_sweeps=50)
                if cb2 < c_best:
                    h_best = hb2
                    c_best = cb2

    # Phase 6: Final polish - SLSQP one more time from best
    if tl() > 30:
        for beta in [256, 1024]:
            if tl() < 20:
                break
            h_try = do_slsqp(h_best, beta, len(h_best) / 2.0, maxiter=100)
            c_try = evaluate_c5(h_try)
            if c_try < c_best - 1e-13:
                h_best = h_try
                c_best = c_try

    # Final validation
    n_final = len(h_best)
    h_best = project_sum_box(h_best, n_final / 2.0)
    c_best = evaluate_c5(h_best)

    return h_best, c_best, n_final
