```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, alpha=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_c = np.max(corr)
    exps = np.exp(alpha * (corr - max_c))
    w = exps / exps.sum()
    u = 1.0 - 2.0 * h
    grad = np.correlate(w, u, mode="valid")
    return grad * dx

def subgrad_descent(h, n_steps=5000, alpha_start=50.0, alpha_end=500.0,
                    lr_start=0.1, lr_end=0.001, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for i in range(n_steps):
        t = i / max(n_steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        
        grad = smooth_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val

def local_search_binary(h_bin, n_iters=50000, seed=42):
    """Local search on binary solution."""
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    ones = np.where(h == 1)[0]
    
    for i in range(n_iters):
        # Pick random 1 and 0 to swap
        if len(ones) == 0:
            break
        i_idx = ones[rng.randint(len(ones))]
        zeros = np.where(h == 0)[0]
        if len(zeros) == 0:
            break
        j_idx = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h[i_idx], h[j_idx] = 0, 1
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
            ones = np.where(h == 1)[0]
        else:
            h[i_idx], h[j_idx] = 1, 0
    
    return best_h, best_val
```
