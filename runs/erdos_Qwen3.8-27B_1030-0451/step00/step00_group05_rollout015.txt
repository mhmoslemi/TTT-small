    n = len(h)
    return np.correlate(h, 1 - h, mode="full")

def compute_gradient_row(h, p):
    """Compute ∂c[p]/∂h[j] for a single lag p."""
    n = len(h)
    s = p - (n - 1)
    grad = np.zeros(n)
    # h[j] appears as h[i+s] when i = j-s
    # Contribution: (1 - h[j-s]) if 0 <= j-s <= n-1
    # h[j] appears as h[i] when i = j
    # Contribution: -h[j+s] if 0 <= j+s <= n-1
    if s >= 0:
        # j-s >= 0 means j >= s; j-s <= n-1 means j <= n-1+s (always true for j < n when s < n)
        # So for j >= s: add (1 - h[j-s])
        grad[s:] += (1 - h[:n-s])
        # j+s <= n-1 means j <= n-1-s
        # So for j <= n-1-s: subtract h[j+s]
        grad[:n-s] -= h[s:]
    else:
        ss = -s
        # j-s = j+ss >= 0 always; j+ss <= n-1 means j <= n-1-ss
        # So for j <= n-1-ss: add (1 - h[j+ss])
        grad[:n-ss] += (1 - h[ss:])
        # j+s = j-ss >= 0 means j >= ss; j-ss <= n-1 always
        # So for j >= ss: subtract h[j-ss]
        grad[ss:] -= h[:n-ss]
    return grad

def compute_full_gradient(h):
    """Compute full gradient matrix G[p, j] = ∂c[p]/∂h[j]."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        G[p] = compute_gradient_row(h, p)
    return G

            G[p, :n - s] -= h[s:]

            ss = -s
            G[p, :n - ss] += (1.0 - h[ss:])
            G[p, ss:] -= h[:n - ss]
    return G * dx

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:

    n = len(h)
    old_x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    h_interp = np.interp(new_x, old_x, h, period=2.0)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return project_feasible(h_interp, new_n)

            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)
