def repair_profile(arr):
    """Repair profile to satisfy 0 <= h[i] <= 1 and sum(h) = n/2."""
    arr = np.asarray(arr, dtype=np.float64).ravel().copy()
    n = len(arr)
    target = n / 2.0
    arr = np.clip(arr, 0.0, 1.0)
    s = np.sum(arr)
    diff = target - s
    if abs(diff) < 1e-10:
        return arr
    # Iteratively adjust
    for _ in range(100):
        s = np.sum(arr)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase: add to positions with room
            slack = 1.0 - arr
            order = np.argsort(-slack)
            add = diff / n
            for i in order:
                amt = min(add, slack[i], diff)
                arr[i] += amt
                diff -= amt
                if abs(diff) < 1e-12:
                    break
        else:
            # Need to decrease: remove from positions with value
            order = np.argsort(arr)  # start from smallest to avoid hitting 0
            # Actually, remove proportionally
            remove = -diff / n

                arr[i] -= amt
                diff += amt

    s = np.sum(arr)

def compute_penalty_gradient(h, target, margin):
    """Compute gradient of penalty sum max(0, v_k - target)^2.
    Returns gradient array and active count."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    
    # Weights: w_k = 2 * max(0, v_k - target) for active ones, 0 otherwise
    violations = np.maximum(0, vals - (target - margin))
    weights = 2.0 * violations * dx  # dx factor for converting to h-space
    
    # Only use active lags for efficiency
    active = np.where(weights > 1e-15)[0]
    if len(active) == 0:
        # Fallback: use top lags
        top_k = min(10, len(vals))
        active = np.argsort(-vals)[:top_k]
        weights = np.zeros_like(vals)
        for k in active:
            weights[k] = 2.0 * max(0, vals[k] - target) * dx
        if np.max(weights) < 1e-15:
            return None, 0
    
    g = np.zeros(n)
    
    for k_idx in active:
        w = weights[k_idx]
        if w < 1e-15:

        L = k_idx - (n - 1)  # lag
        if L == 0:
            g += w * (1.0 - 2.0 * h)

                g[i] += w * (1.0 - h[i + R])
            for i in range(R, n):
                g[i] -= w * h[i - R]
    
    return g, len(active)

    best_val = curr_val

        if g_norm < 1e-15:

        g = g / g_norm
        
        # Line search

import numpy as np
import time

def repair_profile(arr):
    arr = np.asarray(arr, dtype=np.float64).ravel().copy()
    n = len(arr)
    if n == 0:
        return arr
    target = n / 2.0
    np.clip(arr, 0.0, 1.0, out=arr)
    for _ in range(50):
        s = np.sum(arr)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            slack = 1.0 - arr
            total_slack = np.sum(slack)
            if total_slack < 1e-15:
                break
            arr += (diff / total_slack) * slack
        else:
            total_val = np.sum(arr)
            if total_val < 1e-15:
                break
            arr += (diff / total_val) * arr
    np.clip(arr, 0.0, 1.0, out=arr)
    # Final exact correction
    diff = target - np.sum(arr)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = arr < 1.0 - 1e-15
        else:
            mask = arr > 1e-15
        if np.any(mask):
            arr[mask] += diff / np.sum(mask)
            np.clip(arr, 0.0, 1.0, out=arr)
    return arr

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    return float(np.max(vals)), vals

def compute_gradient(h, vals, target, margin):
    n = len(h)
    dx = 2.0 / n
    # Weights for penalty sum max(0, v_k - (target-margin))^2
    thresh = target - margin
    viol = np.maximum(0.0, vals - thresh)
    w = 2.0 * viol * dx
    
    if np.max(w) < 1e-15:
        # Use top lags
        top = min(8, len(vals))
        idx = np.argsort(-vals)[:top]
        w = np.zeros_like(vals)
        for i in idx:
            if vals[i] > thresh:
                w[i] = 2.0 * (vals[i] - thresh) * dx
        if np.max(w) < 1e-15:
            return None
    
    g = np.zeros(n)
    # Lag 0 (output index n-1)
    w0 = w[n-1]
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    
    # Other lags
    for R in range(1, n):
        wp = w[n-1+R]  # lag +R
        wn = w[n-1-R]  # lag -R
        if wp < 1e-15 and wn < 1e-15:
            continue
        g[0:n-R] += wn * (1.0 - h[R:n]) - wp * h[R:n]
        g[R:n] += wp * (1.0 - h[0:n-R]) - wn * h[0:n-R]
    
    return g

def projected_descent(h0, max_iters, target, margin, step_scale, deadline, rng):
    h = h0.copy()
    curr_val, curr_vals = evaluate_c5(h)
    best_h = h.copy()
    best_val = curr_val
    step = step_scale
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = compute_gradient(h, curr_vals, target, margin)
        if g is None:
            break
        
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-14:
            break
        g = g / g_norm
        
        accepted = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            alpha = step * frac
            h_try = h - alpha * g
            h_try = repair_profile(h_try)
            try_val, _ = evaluate_c5(h_try)
            if try_val < curr_val - 1e-12:
                h = h_try
                curr_val = try_val
                _, curr_vals = evaluate_c5(h)
                if curr_val < best_val - 1e-12:
                    best_val = curr_val
                    best_h = h.copy()
                step = min(step * 1.1, 0.3)
                accepted = True
                break
        
        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break
    
    return best_h, best_val
