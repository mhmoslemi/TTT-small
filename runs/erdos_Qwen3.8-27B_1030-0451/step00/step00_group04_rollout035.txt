import numpy as np
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(30):
        residual = target - float(np.sum(h))
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        per = residual / float(np.sum(mask))
        h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        new_val = h[idx] + residual
        if 0.0 <= new_val <= 1.0:
            h[idx] = new_val
    return h
