```python
def _resample_step(h, n_old, n_new):
    """Resample step function from n_old to n_new points."""
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    g = np.zeros(n_new, dtype=np.float64)
    for j in range(n_new):
        x_start = j * dx_new
        x_end = (j + 1) * dx_new
        # Find which old cells overlap
        i_start = int(x_start / dx_old)
        i_end = int(np.ceil(x_end / dx_old))
        i_start = max(0, i_start)
        i_end = min(n_old, i_end)
        total_weight = 0.0
        val = 0.0
        for i in range(i_start, i_end):
            cell_start = i * dx_old
            cell_end = (i + 1) * dx_old
            overlap = min(cell_end, x_end) - max(cell_start, x_start)
            if overlap > 0:
                val += h[i] * overlap
                total_weight += overlap
        if total_weight > 0:
            g[j] = val / total_weight
    # Fix the mean
    g = _project(g, n_new)
    return g
```

def _optimize_local(h, n, rng, time_limit):
    """Local optimization using smoothed gradient descent."""
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = _evaluate(h, n)
    
    T = 5.0  # temperature
    base_step = 0.01
    no_improve = 0
    max_no_improve = 200
    
    while time.time() - start < time_limit:
        # Compute current objective
        full = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
        mx = np.max(full)
        
        # Compute softmax weights
        exponents = (full - mx) / T
        weights = np.exp(exponents)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum
        
        # Compute gradient
        grad = _compute_gradient(h, n, weights)
        
        # Project to tangent space (sum = 0)
        d = -(grad - np.mean(grad))
        
        # Check step size
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-14:
            break
        d = d / d_norm
        
        # Try a step
        step = base_step
        accepted = False
        for _ in range(5):
            h_new = _project(h + step * d, n)
            c5_new = _evaluate(h_new, n)
            if c5_new < mx - 1e-12:
                h = h_new
                accepted = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    no_improve = 0
                break
            step *= 0.5
        
        if not accepted:
            # Try random perturbation
            if no_improve > max_no_improve:
                h = _project(h + 0.01 * rng.standard_normal(n), n)
                no_improve = 0
                T = min(T * 1.5, 50.0)
                base_step = 0.01
            else:
                # Accept a slightly worse move for exploration
                h = _project(h + step * d, n)
        
        no_improve += 1
        
        # Adapt temperature
        # If many lags are active, use higher T; if few, lower T
        active = np.sum(weights > 0.01)
        if active < 3:
            T = max(T * 0.9, 1.0)
        elif active > n:
            T = min(T * 1.1, 30.0)
        
        # Slowly decrease step
        base_step = max(base_step * 0.999, 0.003)
    
    return best_h, best_c5

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    
    # Get initial values from pre-imported construction
    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    h0 = _project(h0, n0)
    c5_0 = _evaluate(h0, n0)
    
    best_h = h0.copy()
    best_c5 = c5_0
    best_n = n0
    
    # Phase 1: Primary optimization at n0 (45% of budget)
    t1 = budget_s * 0.45
    h_opt, c5_opt = _optimize_local(h0, n0, rng, t1)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt
        best_n = n0
    
    # Phase 2: Multi-directional optimization (20% of budget)
    elapsed = time.time() - t_start
    remaining = budget_s * 0.65 - elapsed
    if remaining > 10:
        h_opt2, c5_opt2 = _optimize_multidirectional(best_h, best_n, rng, remaining)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2
            best_n = h_opt2
    
    # Phase 3: Try nearby n values (20% of budget)
    elapsed = time.time() - t_start
    remaining = budget_s * 0.85 - elapsed
    if remaining > 15:
        n_values = []
        for d in [-3, -2, -1, 1, 2, 3, -5, 5, -4, 4]:
            nn = best_n + d
            if 10 <= nn <= 2500 and nn != best_n:
                n_values.append(nn)
        n_values = sorted(set(n_values))
        time_per_n = max(remaining / max(len(n_values), 1), 3.0)
        for nn in n_values:
            if time.time() - t_start > budget_s * 0.85:
                break
            h_try = _resample_step(best_h, best_n, nn)
            h_try = _project(h_try, nn)
            t_lim = min(time_per_n, 45.0)
            left = budget_s * 0.85 - (time.time() - t_start)
            t_lim = min(t_lim, left)
            if t_lim < 2:
                continue
            h_opt3, c5_opt3 = _optimize_local(h_try, nn, rng, t_lim)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3
                best_n = nn
    
    # Phase 4: Final polish from best (remaining time)
    elapsed = time.time() - t_start
    remaining = budget_s * 0.97 - elapsed
    if remaining > 3:
        h_opt4, c5_opt4 = _optimize_local(best_h, best_n, rng, remaining)
        if c5_opt4 < best_c5:
            best_c5 = c5_opt4
            best_h = h_opt4
    
    # Final validation
    final_c5 = _evaluate(best_h, best_n)
    return best_h, final_c5, best_n
