So v_k as a function of h[j] (for fixed other variables):
v_k(h[j]) = [other terms] + dx * [(1-h[j-k]) * h[j] - h[j+k] * h[j]]
= [other terms] + dx * [(1-h[j-k]) - h[j+k]] * h[j]

This is linear in h[j] for each k! So the objective max_k v_k, as a function of h[j], is the max of linear functions, which is convex in h[j]. We can minimize it by finding the point where the maximum is minimized.

For a single variable j with the constraint that sum stays constant (which means we'd need to adjust another variable, making it trickier), this becomes a 1D problem.

        target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x + lam, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = result < 1.0 - 1e-14
        else:
            mask = result > 1e-14
        if np.any(mask):
            if residual > 0:
                slack = 1.0 - result[mask]
                total = np.sum(slack)
                if total > 1e-15:
                    result = result.copy()
                    result[mask] += residual * slack / total
            else:
                total = np.sum(result[mask])
                if total > 1e-15:
                    result = result.copy()
                    result[mask] += residual * result[mask] / total
    return result

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    v = compute_corr(h)
    m = np.max(v)
    shifted = np.clip(v - m, -40.0, 40.0)
    exps = np.exp(beta * shifted)
    Z = np.sum(exps)
    obj = m + np.log(Z) / beta
    w = exps / Z
    grad = np.zeros(n)
    for k_idx in range(len(v)):
        wi = w[k_idx]
        if wi < 1e-14:
            continue
        k = k_idx - (n - 1)
        ns = max(0, -k)
        ne = min(n, n - k)
        if ns >= ne:
            continue
        idx2 = np.arange(ns, ne)
        idx1 = idx2 + k
        grad[idx1] += wi * dx * (1.0 - h[idx2])
        grad[idx2] += wi * dx * (-h[idx1])
    return obj, grad

def slsqp_run(h0, target, betas, maxiter_per=150, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    eq_fun = lambda x: np.sum(x) - target
    eq_jac = lambda x: np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    h_cur = h0.copy()
    for beta in betas:
        def _obj(x):
            val, _ = lse_obj_grad(x, beta)
            return val
        def _grad(x):
            _, g = lse_obj_grad(x, beta)
            return g
        try:
            res = minimize(_obj, h_cur, jac=_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per, 'ftol': ftol})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_cur = project_sum_box(res.x, target)
        except Exception:
            pass
    return h_cur

def lp_trust_region(h0, v0, radius, n_active=40):
    n = len(h0)
    dx = 2.0 / n
    order = np.argsort(v0)[::-1]
    active = order[:min(n_active, len(v0))]
    A_rows, b_rows = [], []
    for k_idx in active:
        k = k_idx - (n - 1)
        ns = max(0, -k)
        ne = min(n, n - k)
        if ns >= ne:
            continue
        idx2 = np.arange(ns, ne)
        idx1 = idx2 + k
        grad_k = np.zeros(n)
        grad_k[idx1] += dx * (1.0 - h0[idx2])
        grad_k[idx2] += dx * (-h0[idx1])
        rhs = np.dot(grad_k, h0) - v0[k_idx]
        row = np.zeros(n + 1)
        row[:n] = grad_k
        row[n] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    if not A_rows:
        return None
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    lb = np.maximum(0.0, h0 - radius)
    ub = np.minimum(1.0, h0 + radius)
    bnds = list(zip(lb, ub))
    bnds.append((-1.0, 2.0))
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def interp_to_grid(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.RandomState(seed)

    def time_left():
        return deadline - time.time()

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = project_sum_box(np.array(initial_h_values, dtype=float).copy())
    else:
        h_best = np.full(84, 0.5)

    n_best = len(h_best)
    c_best = evaluate_c5(h_best)

    # Phase 1: SLSQP on original grid
    if time_left() > 80:
        h_mod = slsqp_run(h_best.copy(), n_best / 2.0,
                          [5, 10, 20, 40, 80, 150, 300, 500], maxiter_per=120)
        h_mod = project_sum_box(h_mod, n_best / 2.0)
        c_mod = evaluate_c5(h_mod)
        if c_mod < c_best - 1e-12:
            h_best, c_best = h_mod, c_mod

    # Phase 2: LP trust region
    if time_left() > 80:
        v0 = compute_corr(h_best)
        radius = 0.10
        no_improve = 0
        for _ in range(8):
            if time_left() < 25:
                break
            if no_improve >= 4:
                break
            h_lp = lp_trust_region(h_best, v0, radius, n_active=50)
            if h_lp is not None:
                h_lp = project_sum_box(h_lp, n_best / 2.0)
                c_lp = evaluate_c5(h_lp)
                if c_lp < c_best - 1e-12:
                    h_best, c_best = h_lp, c_lp
                    v0 = compute_corr(h_best)
                    no_improve = 0
                    radius = min(0.15, radius * 1.2)
                else:
                    no_improve += 1
                    radius *= 0.6
            else:
                no_improve += 1
                radius *= 0.6
            if radius < 0.005:
                break

    # Phase 3: SLSQP high-beta polish
    if time_left() > 60:
        h_mod = slsqp_run(h_best.copy(), n_best / 2.0,
                          [200, 400, 800, 1200], maxiter_per=100)
        h_mod = project_sum_box(h_mod, n_best / 2.0)
        c_mod = evaluate_c5(h_mod)
        if c_mod < c_best - 1e-12:
            h_best, c_best = h_mod, c_mod

    # Phase 4: Random perturbations + local opt
    if time_left() > 120:
        best_perturb_c = c_best
        for trial in range(6):
            if time_left() < 40:
                break
            noise_scale = 0.03 * (1 - trial * 0.1)
            h_pert = h_best + rng.normal(0, noise_scale, n_best)
            h_pert = project_sum_box(h_pert, n_best / 2.0)
            h_pert = slsqp_run(h_pert, n_best / 2.0,
                               [30, 80, 200, 400], maxiter_per=60)
            h_pert = project_sum_box(h_pert, n_best / 2.0)
            c_pert = evaluate_c5(h_pert)
            if c_pert < c_best - 1e-12:
                h_best, c_best = h_pert, c_pert
                break

    # Phase 5: Transfer to finer grid
    for mult in [2, 3]:
        n_new = n_best * mult
        if n_new > 504:
            continue
        if time_left() < 150:
            break
        h_interp = interp_to_grid(h_best, n_new)
        h_interp = project_sum_box(h_interp, n_new / 2.0)
        c_interp = evaluate_c5(h_interp)

        h_try = h_interp.copy()
        c_try = c_interp

        # Optimize on finer grid
        if time_left() > 100:
            h_try = slsqp_run(h_interp, n_new / 2.0,
                              [5, 10, 20, 50, 100, 200, 400], maxiter_per=80)
            h_try = project_sum_box(h_try, n_new / 2.0)
            c_try = evaluate_c5(h_try)

        # LP on finer grid
        if time_left() > 50:
            v_try = compute_corr(h_try)
            radius = 0.06
            for _ in range(4):
                if time_left() < 20:
                    break
                h_lp2 = lp_trust_region(h_try, v_try, radius, n_active=60)
                if h_lp2 is not None:
                    h_lp2 = project_sum_box(h_lp2, n_new / 2.0)
                    c_lp2 = evaluate_c5(h_lp2)
                    if c_lp2 < c_try - 1e-12:
                        h_try, c_try = h_lp2, c_lp2
                        v_try = compute_corr(h_try)
                    else:
                        radius *= 0.6
                else:
                    radius *= 0.6
                if radius < 0.003:
                    break

        # High-beta polish on finer grid
        if time_left() > 30:
            h_try2 = slsqp_run(h_try, n_new / 2.0,
                               [300, 600, 1000], maxiter_per=50)
            h_try2 = project_sum_box(h_try2, n_new / 2.0)
            c_try2 = evaluate_c5(h_try2)
            if c_try2 < c_try:
                h_try, c_try = h_try2, c_try2

        if c_try < c_best - 1e-12:
            h_best, c_best, n_best = h_try, c_try, n_new

    # Phase 6: Final SLSQP polish
    if time_left() > 20:
        h_mod = slsqp_run(h_best.copy(), n_best / 2.0,
                          [500, 800, 1200, 2000], maxiter_per=60)
        h_mod = project_sum_box(h_mod, n_best / 2.0)
        c_mod = evaluate_c5(h_mod)
        if c_mod < c_best - 1e-12:
            h_best, c_best = h_mod, c_mod

    # Phase 7: LP final polish
    if time_left() > 15:
        v_final = compute_corr(h_best)
        radius = 0.03
        for _ in range(3):
            if time_left() < 8:
                break
            h_lp = lp_trust_region(h_best, v_final, radius, n_active=50)
            if h_lp is not None:
                h_lp = project_sum_box(h_lp, n_best / 2.0)
                c_lp = evaluate_c5(h_lp)
                if c_lp < c_best - 1e-12:
                    h_best, c_best = h_lp, c_lp
                    v_final = compute_corr(h_best)
                else:
                    radius *= 0.5
            else:
                break

    # Final validation
    h_best = project_sum_box(h_best, len(h_best) / 2.0)
    assert np.all(h_best >= -1e-12) and np.all(h_best <= 1.0 + 1e-12)
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_sum_box(h_best, len(h_best) / 2.0)
    c_final = evaluate_c5(h_best)

    return h_best, c_final, len(h_best)
