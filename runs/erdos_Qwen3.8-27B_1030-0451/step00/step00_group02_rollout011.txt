def smooth_obj_grad(h, tau):
    n = len(h)
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    exp_vals = np.exp((overlaps - m) / tau)
    sw = np.sum(exp_vals)
    value = m + tau * np.log(sw)
    weights = exp_vals / sw
    grad = np.zeros(n)

        w = weights[k_idx]
        if w > 1e-9:

            grad += w * overlap_grad(h, lag)
    return value, grad


class SmoothFunc:
    def __init__(self, tau):
        self.tau = tau
        self.val = 0.0
        self.grad = None

    def f(self, h):
        v, g = smooth_obj_grad(h, self.tau)
        self.val = v
        self.grad = g
        return v

    def g(self, h):
        if self.grad is None:
            _, self.grad = smooth_obj_grad(h, self.tau)
        return self.grad

    def fg(self, h):
        v, g = smooth_obj_grad(h, self.tau)
        self.val = v
        self.grad = g
        return (v, g)

    def constrain_val(self, h):
        n = len(h)
        return np.sum(h) - n / 2.0

    def constrain_jac(self, h):
        n = len(h)
        return np.ones((1, n))


def slsqp_solve(h, tau, maxiter=200):
    n = len(h)
    sf = SmoothFunc(tau)
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': sf.constrain_val, 'jac': sf.constrain_jac}
    try:
        res = minimize(sf.fg, h, method='SLSQP', jac=True, bounds=bounds,
                       constraints=constraint, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.fun < smooth_obj_grad(h, tau)[0]:
            h_new = project_feasible(res.x)
            return h_new
    except Exception:
        pass
    return h


def lp_direction(h, n_active=15, theta=0.01):
    n = len(h)
    overlaps = compute_overlaps(h)
    idx = np.argsort(overlaps)[::-1][:n_active]
    # Build LP: max delta s.t. grad_k · d <= -delta, ||d||_1 <= 1, sum(d)=0
    # Variables: d[0..n-1], s[0..n-1], delta
    # d_i - s_i <= 0, -d_i - s_i <= 0 for L1 norm
    # s_i >= 0 implied by s_i >= |d_i| >= 0... actually we need s_i >= 0
    # sum(s_i) <= 1
    # For each active k: grad_k · d + delta <= 0
    # sum(d) = 0
    # delta <= theta (to keep steps bounded)
    
    m_constraints = n_active + 2 * n + 1  # active + 2n (L1) + 1 (sum s)
    n_vars = 2 * n + 1
    
    c = np.zeros(n_vars)
    c[2 * n] = -1.0  # maximize delta = minimize -delta
    
    A_ub = np.zeros((m_constraints, n_vars))
    b_ub = np.zeros(m_constraints)
    
    row = 0
    for ii, k_idx in enumerate(idx):
        lag = k_idx - (n - 1)
        g = overlap_grad(h, lag)
        A_ub[row, :n] = g
        A_ub[row, 2 * n] = 1.0
        b_ub[row] = 0.0
        row += 1
    
    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = 0.0
        row += 1
    for i in range(n):
        A_ub[row, i] = -1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = 0.0
        row += 1
    # sum(s) <= 1
    A_ub[row, n:2 * n] = 1.0
    b_ub[row] = 1.0
    row += 1
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds_lp = [(None, None)] * n + [(0.0, 1.0)] * n + [(0.0, theta)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success and res.fun < -1e-12:
            d = res.x[:n]
            if np.sum(np.abs(d)) > 1e-10:
                return d
    except Exception:
        pass
    return None


def gradient_direction(h, n_active=10, temperature=0.02):
    n = len(h)
    overlaps = compute_overlaps(h)
    idx = np.argsort(overlaps)[::-1][:n_active]
    vals = overlaps[idx]
    w = np.exp((vals - vals[0]) / temperature)
    w /= np.sum(w)
    grad = np.zeros(n)
    for ii, k_idx in enumerate(idx):
        lag = k_idx - (n - 1)
        grad += w[ii] * overlap_grad(h, k_idx - (n - 1))
    grad -= np.mean(grad)
    return -grad  # negative gradient direction

def directed_pair_search(h, c5, rng, n_trials=300, deadline=None):

            break
        k_idx = top_k[rng.randint(len(top_k))]
        lag = k_idx - (n - 1)
        g = overlap_grad(h, lag)
        # We want to decrease overlap at lag: move in direction -g
        # Find i where -g[i] is large (decreasing h[i] helps if g[i]>0)
        # and j where -g[j] is large (increasing h[j] helps if g[j]<0)
        neg_g = -g
        # Source: where h is high AND neg_g is positive (decreasing h reduces F_k)
        # Target: where h is low AND neg_g is positive (increasing h reduces F_k)
        # But need sum=0: decrease source, increase target
        # Actually we need: direction d with d[i]>0 (increase h[i]), d[j]<0 (decrease h[j])
        # For decrease of F_k: we want g·d < 0, i.e., sum g_i * d_i < 0
        # If d[i] > 0 (increase) and d[j] < 0 (decrease with |d[j]|=d[i]):
        #   g·d = g[i]*amt - g[j]*amt = amt*(g[i]-g[j]) < 0 iff g[i] < g[j]
        # So: increase where g is small, decrease where g is large
        
        # Find good (increase_idx, decrease_idx) pairs
        sorted_by_g = np.argsort(g)
        inc_candidates = sorted_by_g[:n // 2]  # low g: good to increase

                try_update(h, c5)
    
    if time.time() > deadline:
        return best_h, best_c5, len(best_h)
    
    # Phase 2: Intensive local search
    t_phase2_start = time.time()
    t_phase2_deadline = deadline - 60
    
    stall_count = 0
    max_stall = 80
    
    while time.time() < t_phase2_deadline and stall_count < max_stall:
        c5_before = c5
        method_choice = rng.random()
        
        if method_choice < 0.3:
            # Gradient step
            dirn = gradient_direction(h, n_active=rng.randint(5, 15),
                                       temperature=rng.uniform(0.005, 0.05))
            h_try, c5_try = line_search_step(h, dirn, c5)
            if h_try is not None:
                h = h_try
                c5 = c5_try
                try_update(h, c5)
        elif method_choice < 0.5:
            # LP direction
            d = lp_direction(h, n_active=rng.randint(8, 20),
                           theta=rng.uniform(0.005, 0.05))
            if d is not None:
                h_try, c5_try = line_search_step(h, d, c5)
                if h_try is not None:
                    h = h_try
                    c5 = c5_try
                    try_update(h, c5)
        elif method_choice < 0.7:
            # Directed pair search
            h, c5 = directed_pair_search(h, c5, rng, n_trials=100,
                                         deadline=t_phase2_deadline)
            try_update(h, c5)
        elif method_choice < 0.85:
            # Random pair search
            h, c5 = pair_exchange_search(h, c5, rng, n_trials=100,
                                         deadline=t_phase2_deadline)
            try_update(h, c5)
        else:
            # Perturbation
            h, c5 = perturbation_search(h, c5, rng, n_trials=50,
                                        deadline=t_phase2_deadline)
            try_update(h, c5)
        
        if c5 < c5_before - 1e-13:
            stall_count = 0
        else:
            stall_count += 1
        
        # Periodic short SLSQP polish
        if stall_count > 20 and stall_count % 30 == 0 and time.time() < t_phase2_deadline - 30:
            h_new = slsqp_solve(h, 0.001, maxiter=100)
            h_new = project_feasible(h_new)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                try_update(h, c5)
            stall_count = max(0, stall_count - 10)
    
    if time.time() > deadline - 30:
        return best_h, best_c5, len(best_h)
    
    # Phase 3: Try dimension transfer
    for n_try in [n - 1, n + 1, n + 2, n - 2, n * 2, n + 16, n + 32]:
        if time.time() > deadline - 20:
            break
        if n_try < 10 or n_try > 2500:
            continue
        h_trans = interpolate_to(h, n_try)
        c5_trans = compute_c5(h_trans)
        # Quick local improvement
        if time.time() < deadline - 10:
            dirn = gradient_direction(h_trans, n_active=8, temperature=0.01)
            h_try, c5_try = line_search_step(h_trans, dirn, c5_trans)
            if h_try is not None:
                h_trans = h_try
                c5_trans = c5_try
            h_trans, c5_trans = directed_pair_search(h_trans, c5_trans, rng,
                                                      n_trials=200, deadline=deadline - 5)
        if c5_trans < best_c5 - 1e-12:
            best_h = h_trans.copy()
            best_c5 = c5_trans
            n = n_try
            h = h_trans
            c5 = c5_trans
    
    # Phase 4: Final polish
    if time.time() < deadline - 5:
        tau_final = 0.0003
        h_new = slsqp_solve(h, tau_final, maxiter=300)
        h_new = project_feasible(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Final verification
    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)
    
    # Sanity check
    if not np.all(np.isfinite(best_h)) or np.min(best_h) < -1e-10 or np.max(best_h) > 1 + 1e-10:
        # Fallback: uniform
        best_h = np.full(80, 0.5)
        final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, len(best_h)
