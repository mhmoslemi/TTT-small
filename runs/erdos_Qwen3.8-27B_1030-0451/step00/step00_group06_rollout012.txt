def project_feasible(v, target_sum):
    """Euclidean projection onto {h: 0<=h_i<=1, sum(h)=target_sum}."""
    n = len(v)
    lo = v.min() - target_sum / n - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    return float(corr.max()), corr
