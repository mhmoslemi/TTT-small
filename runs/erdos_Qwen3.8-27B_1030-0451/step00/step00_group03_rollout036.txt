import numpy as np
import time

def exact_c5(h):
    """Compute the exact C5 value for a feasible step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def shifted_projection(y):
    """Project y onto {x : 0 <= x_i <= 1, sum(x) = n/2} using shift + clip."""
    n = len(y)
    target = n / 2.0
    # Bisection on alpha: find alpha such that sum(clip(y + alpha, 0, 1)) = target
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    
    alpha_lo = -np.max(y) - 1.0  # ensures all clipped to 0
    alpha_hi = 1.0 - np.min(y) + 1.0  # ensures all clipped to 1
    
    # Check if target is achievable
    sum_lo = np.sum(np.clip(y + alpha_lo, 0, 1))
    sum_hi = np.sum(np.clip(y + alpha_hi, 0, 1))
    
    if sum_lo > target + 1e-12:
        return np.clip(y + alpha_lo, 0, 1)
    if sum_hi < target - 1e-12:
        return np.clip(y + alpha_hi, 0, 1)
    
    for _ in range(100):
        alpha_mid = 0.5 * (alpha_lo + alpha_hi)
        x = np.clip(y + alpha_mid, 0, 1)
        s = np.sum(x)
        if s < target:
            alpha_lo = alpha_mid
        else:
            alpha_hi = alpha_mid
        if abs(s - target) < 1e-14:
            break
    
    x = np.clip(y + 0.5 * (alpha_lo + alpha_hi), 0, 1)
    # Final correction for floating point
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        # Distribute residual to coordinates with room
        if residual > 0:
            mask = x < 1.0
            if np.any(mask):
                n_room = np.sum(mask)
                add = residual / n_room
                x[mask] = np.minimum(1.0, x[mask] + add)
        else:
            mask = x > 0.0
            if np.any(mask):
                n_room = np.sum(mask)
                sub = -residual / n_room
                x[mask] = np.maximum(0.0, x[mask] - sub)
    return x

def ensure_feasible(h):
    """Ensure h is in [0,1] with sum = n/2. Returns feasible array or None."""
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or len(h) == 0 or not np.all(np.isfinite(h)):
        return None
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    return shifted_projection(h)

def get_active_pairs(h, max_pairs=80):
    """Find promising (i,j) pairs for mass exchange based on correlation structure."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    L = 2 * n - 1
    
    K = min(24, max(6, L // 16), n - 1)
    top_idx = np.argsort(corr)[::-1][:K]
    
    pairs = set()
    for m in top_idx:
        k = m - (n - 1)
        if k >= 0:
            # lag k: overlap h[p+k] * (1-h[p]) for p=0..n-k-1
            for p in range(max(0, n - k - 1)):
                a = p + k
                b = p
                if a != b and a < n and b < n:
                    pairs.add((min(a,b), max(a,b)))
        else:
            r = -k
            for p in range(r, n):
                a = p
                b = p - r
                if a != b and a < n and b < n:
                    pairs.add((min(a,b), max(a,b)))
    
    # Add high-low value pairs
    if len(pairs) < max_pairs:
        order = np.argsort(h)
        low_half = order[:n//4]
        high_half = order[-n//4:]
        for a in np.random.default_rng(0).choice(len(low_half), size=min(20, len(low_half)), replace=False):
            for b in np.random.default_rng(1).choice(len(high_half), size=min(4, len(high_half)), replace=False):
                i, j = int(low_half[a]), int(high_half[b])
                if i != j:
                    pairs.add((min(i,j), max(i,j)))
    
    # Fill with random pairs if needed
    if len(pairs) < 16:
        for _ in range(200):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
    
    result = list(pairs)[:max_pairs]
    return result

def shifted_projection(y):
    """Project y to {x: 0<=x<=1, sum(x)=n/2} via shift+clip bisection."""
    n = len(y)
    target = 0.5 * n
    y = np.nan_to_num(np.asarray(y, dtype=np.float64), nan=0.5, posinf=1.0, neginf=0.0)
    a_lo = -float(np.max(y)) - 0.5

    for _ in range(80):
        a_mid = 0.5 * (a_lo + a_hi)
        x = np.clip(y + a_mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s < target:
            a_lo = a_mid
        else:
            a_hi = a_mid
    x = np.clip(y + 0.5*(a_lo+a_hi), 0.0, 1.0)
    # Fix residual
