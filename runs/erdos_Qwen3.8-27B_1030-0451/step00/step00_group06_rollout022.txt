    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:

    h = np.clip(z - theta, 0.0, 1.0)

            idx = np.argmin(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def fix_sum(h, target):
    """Fix sum of h to equal target, maintaining [0,1] bounds."""
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase some elements
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 0:
                break
            room = 1.0 - h[i]
            adj = min(diff, room)
            h[i] += adj
            diff -= adj
    else:
        # Need to decrease some elements
        order = np.argsort(h)
        for i in order:
            if diff >= 0:
                break
            room = h[i]
            adj = min(-diff, room)
            h[i] -= adj
            diff += adj
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def assemble_anti(x, N):
    h = np.empty(N, dtype=np.float64)
    M = N // 2
    h[:M] = x
    h[N - M:] = 1.0 - x[::-1]
    if N % 2 == 1:
        h[M] = 0.5
    return h


def project_anti(h):
    N = len(h)
    M = N // 2
    right = h[N - M:][::-1]
    x = (h[:M] - right + 1.0) * 0.5
    return np.clip(x, 0.0, 1.0)


def obj_anti(x, N):
    xc = np.clip(x, 0.0, 1.0)
    h = assemble_anti(xc, N)
    return compute_c5(h)


def bisection_theta(z, target, iters=50):
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def make_full_h(z, N):
    target = N * 0.5
    theta = bisection_theta(z, target)
    h = np.clip(z - theta, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmin(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h


def obj_full_smoothed(z, N, t=100.0):
    h = make_full_h(z, N)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / N)
    mc = float(corr.max())
    return mc + float(np.log(np.exp(corr - mc).sum()) / t)


def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 0:
                break
            room = 1.0 - h[i]
            adj = min(diff, room)
            h[i] += adj
            diff -= adj
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= 0:
                break
            room = h[i]
            adj = min(-diff, room)
            h[i] -= adj
            diff += adj
    return h


def evaluate_and_update(h_cand, h_best, c5_best):
    h_cand = np.clip(h_cand, 0.0, 1.0)
    c5_cand = compute_c5(h_cand)
    if c5_cand < c5_best:
        return h_cand, c5_cand
    return h_best, c5_best
