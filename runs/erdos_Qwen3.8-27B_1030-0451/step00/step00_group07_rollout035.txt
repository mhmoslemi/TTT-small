```python
def compute_smooth_obj_and_grad(h, dx, eps):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    
    c_max = np.max(c)
    log_sum = np.log(np.sum(np.exp((c - c_max) / eps)))
    obj = c_max + eps * log_sum
    
    p = np.exp((c - c_max) / eps) / np.sum(np.exp((c - c_max) / eps))
    
    shifts = np.arange(2*n - 1) - (n - 1)
    one_minus_h = 1.0 - h
    
    # Build gradient using broadcasting
    # For each j in 0..n-1 and m in 0..2n-2:
    # contribution to grad[j] from m: p[m] * (1-h[j-shifts[m]]) if 0<=j-shifts[m]<n
    #                                - p[m] * h[j+shifts[m]] if 0<=j+shifts[m]<n
    
    # This is O(n * 2n) which is fine for n <= 1000
    j = np.arange(n)  # (n,)
    s = shifts  # (2n-1,)
    
    # First term: (1-h[j-s]) where 0<=j-s<n
    # j[i] - s[m] for all i,m
    js = j[:, None] - s[None, :]  # (n, 2n-1)
    mask1 = (js >= 0) & (js < n)  # (n, 2n-1)
    # (1-h[js]) where valid
    grad1 = np.zeros((n, 2*n-1))
    grad1[mask1] = one_minus_h[js[mask1]]
    term1 = grad1 @ p  # (n,)
    
    # Second term: h[j+s] where 0<=j+s<n
    js2 = j[:, None] + s[None, :]  # (n, 2n-1)
    mask2 = (js2 >= 0) & (js2 < n)
    grad2 = np.zeros((n, 2*n-1))
    grad2[mask2] = h[js2[mask2]]
    term2 = grad2 @ p
    
    grad = dx * (term1 - term2)
    
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj_and_grad(h, dx, eps):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    e = np.exp((c - c_max) / eps)
    se = np.sum(e)
    obj = c_max + eps * np.log(se)
    p = e / se
    shifts = np.arange(2 * n - 1) - (n - 1)
    omh = 1.0 - h
    j = np.arange(n)
    s = shifts
    js1 = j[:, None] - s[None, :]
    m1 = (js1 >= 0) & (js1 < n)
    v1 = np.where(m1, omh[np.clip(js1, 0, n - 1)], 0.0)
    t1 = v1 @ p
    js2 = j[:, None] + s[None, :]
    m2 = (js2 >= 0) & (js2 < n)
    v2 = np.where(m2, h[np.clip(js2, 0, n - 1)], 0.0)
    t2 = v2 @ p
    grad = dx * (t1 - t2)
    return obj, grad

def project_h(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def resample_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    return np.interp(x_new, x_old, h)

def optimize_pgd(h0, dx, n, time_limit, rng, lr_init, momentum=0.9):
    h = project_h(h0.copy(), n / 2.0)
    v = np.zeros(n)  # momentum velocity
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_obj = smooth_obj_and_grad(h, dx, 0.01)[0]
    
    lr = lr_init
    step = 0
    no_improve = 0
    
    while time.time() < time_limit:
        # Adaptive temperature
        frac = min(step / 5000.0, 1.0)
        eps = 0.02 * (1.0 - frac) ** 2 + 0.001
        
        obj, grad = smooth_obj_and_grad(h, dx, eps)
        
        # Momentum step
        v = momentum * v - lr * grad
        h_new = project_h(h + v, n / 2.0)
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        obj_new = smooth_obj_and_grad(h_new, dx, eps)[0]
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Adapt learning rate
        if no_improve > 100:
            lr *= 0.8
            v *= 0.5
            no_improve = 0
        
        if no_improve > 500:
            # Restart with perturbation
            h = best_h + rng.randn(n) * 0.05
            h = project_h(h, n / 2.0)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0
        
        step += 1
    
    return best_h, best_c5
```

```python
def smooth_obj_and_grad(h, dx, eps):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    e = np.exp((c - c_max) / eps)
    se = np.sum(e)
    obj = c_max + eps * np.log(se)
    p = e / se
    shifts = np.arange(2 * n - 1) - (n - 1)
    omh = 1.0 - h
    j = np.arange(n)
    s = shifts
    js1 = j[:, None] - s[None, :]
    m1 = (js1 >= 0) & (js1 < n)
    v1 = np.where(m1, omh[np.clip(js1, 0, n - 1)], 0.0)
    t1 = v1 @ p
    js2 = j[:, None] + s[None, :]
    m2 = (js2 >= 0) & (js2 < n)
    v2 = np.where(m2, h[np.clip(js2, 0, n - 1)], 0.0)
    t2 = v2 @ p
    grad = dx * (t1 - t2)
```
