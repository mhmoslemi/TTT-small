def project(y, n, target_sum):
    """Project y onto {0<=h<=1, sum(h)=target_sum}"""
    y = np.asarray(y, dtype=np.float64)
    # Bisection on alpha
    lo, hi = -y.max() - 1, -y.min() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(y + mid, 0, 1).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo+hi)/2, 0, 1)
    # Fix any tiny residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-10:
        # Distribute residual
        if residual > 0:
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                d = min(room[i], residual)
                h[i] += d
                residual -= d
                if abs(residual) < 1e-14:

        else:

                residual += d

    n = len(h)
    grad = np.zeros(n)
    if k >= 0:

                    g[r:][:saved_n] -= dx_local * hh[:saved_n - r]
            return g

        constraints.append({'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac})

    try:
        result = minimize(
            obj_fun, x0, jac=obj_jac, method='SLSQP',
            bounds=bounds_list, constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        if result.fun < 1e6:
            h_sol = np.clip(result.x[:n], 0.0, 1.0)
            h_sol = project_to_feasible(h_sol, target_sum)
            c5_sol = eval_c5(h_sol)
            return h_sol, c5_sol
    except Exception:
        pass
    return None, None

def generate_random_binary(n, rng):
    h = np.zeros(n)
    num_ones = int(n // 2)
    positions = rng.choice(n, size=num_ones, replace=False)
    h[positions] = 1.0
    if n % 2 == 1:
        remaining = n - 2 * num_ones
        extra_pos = rng.choice(n, size=remaining, replace=False)
        h[extra_pos] = 0.5
    return h

def generate_waveform(n, freq1, freq2, phase1, phase2, amplitude, rng):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + amplitude * np.sin(2.0 * np.pi * freq1 * x + phase1) + \
        0.3 * amplitude * np.sin(2.0 * np.pi * freq2 * x + phase2)
    noise = rng.normal(0, 0.02, n)
    h += noise
    return project_to_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        h_init = np.full(64, 0.5, dtype=np.float64)

    n0 = len(h_init)
    target0 = n0 / 2.0
    h_init = project_to_feasible(h_init, target0)
    best_c5 = eval_c5(h_init)
    best_h = h_init.copy()
    best_n = n0

    # Phase 1: Optimize at initial resolution
    if remaining() > 30:
        h1, c1 = pgd_optimize(h_init, target0, max_iter=800, tau_init=0.01,
                              time_limit=min(80.0, remaining() * 0.25))
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
            best_n = n0

    if remaining() > 20:
        h1b, c1b = avg_grad_optimize(best_h, best_n / 2.0, max_iter=500,
                                      n_top=10, time_limit=min(40.0, remaining() * 0.15))
        if c1b < best_c5:
            best_c5 = c1b
            best_h = h1b.copy()
            best_n = len(h1b)

    # Phase 2: Upsample and optimize at higher resolutions
    resolutions = []
    if best_n <= 64:
        resolutions = [96, 128, 160, 192, 256]
    elif best_n <= 128:
        resolutions = [160, 192, 256]
    else:
        resolutions = [256, 320, 384]
    resolutions = [r for r in resolutions if r > best_n and r <= 512]

    for n_res in resolutions:
        if remaining() < 40:
            break
        target_res = n_res / 2.0
        alloc = min(remaining() * 0.45, 180.0)

        # Linear interpolation upsample
        h_res = upsample_linear(best_h, n_res)
        h_res = project_to_feasible(h_res, target_res)
        h_res, c5_res = pgd_optimize(h_res, target_res, max_iter=1500,
                                      tau_init=0.005, time_limit=alloc)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            best_n = n_res

        if remaining() < 40:
            break

        # Repeat upsample (if factor is integer)
        factor = n_res // best_n
        if factor >= 2:
            h_rep = upsample_repeat(best_h, factor)
            h_rep = project_to_feasible(h_rep, target_res)
            alloc2 = min(remaining() * 0.3, 120.0)
            h_rep, c5_rep = pgd_optimize(h_rep, target_res, max_iter=1000,
                                          tau_init=0.005, time_limit=alloc2)
            if c5_rep < best_c5:
                best_c5 = c5_rep
                best_h = h_rep.copy()
                best_n = n_res

        # Avg grad on best
        if remaining() > 30:
            h_ag, c5_ag = avg_grad_optimize(best_h, best_n / 2.0, max_iter=400,
                                             n_top=8, time_limit=min(30.0, remaining() * 0.15))
            if c5_ag < best_c5:
                best_c5 = c5_ag
                best_h = h_ag.copy()

    # Phase 3: SLSQP polishing
    if remaining() > 60:
        n_sl = min(best_n, 192)
        if n_sl != best_n:
            h_sl = upsample_linear(best_h, n_sl)
            h_sl = project_to_feasible(h_sl, n_sl / 2.0)
        else:
            h_sl = best_h.copy()
        h_sl, c5_sl = slsqp_polish(h_sl, n_sl / 2.0, n_active_lags=30,
                                    max_iter=200, time_limit=min(120.0, remaining() * 0.4))
        if h_sl is not None and c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            best_n = n_sl

    # Phase 4: Random restarts with different initial structures
    if remaining() > 60:
        n_try = best_n
        target_try = n_try / 2.0
        n_restarts = min(8, int(remaining() / 20))
        for trial in range(n_restarts):
            if remaining() < 20:
                break
            if trial == 0:
                h_r = project_to_feasible(best_h + 0.03 * rng.normal(0, 1, n_try), target_try)
            elif trial == 1:
                h_r = project_to_feasible(best_h + 0.08 * rng.normal(0, 1, n_try), target_try)
            elif trial == 2:
                h_r = generate_random_binary(n_try, rng)
            elif trial == 3:
                h_r = generate_waveform(n_try, 2.0, 5.0, rng.uniform(0, 3), rng.uniform(0, 3), 0.3, rng)
            elif trial == 4:
                h_r = generate_waveform(n_try, 3.0, 7.0, rng.uniform(0, 3), rng.uniform(0, 3), 0.25, rng)
            elif trial == 5:
                # Blend binary and continuous
                h_bin = generate_random_binary(n_try, rng)
                h_r = project_to_feasible(0.5 * best_h + 0.5 * h_bin, target_try)
            elif trial == 6:
                h_r = project_to_feasible(rng.uniform(0.2, 0.8, n_try), target_try)
            else:
                h_r = project_to_feasible(best_h[::-1], target_try)

            h_r, c5_r = pgd_optimize(h_r, target_try, max_iter=600, tau_init=0.01,
                                      time_limit=min(30.0, remaining() * 0.3))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n_try

    # Phase 5: Final fine polishing
    if remaining() > 30:
        target_fin = best_n / 2.0
        h_f, c5_f = pgd_optimize(best_h, target_fin, max_iter=3000, tau_init=0.001,
                                  time_limit=min(remaining() * 0.7, 200.0))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if remaining() > 20:
        h_f2, c5_f2 = avg_grad_optimize(best_h, best_n / 2.0, max_iter=500,
                                         n_top=5, time_limit=min(remaining() * 0.5, 60.0))
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final validation
    best_h = project_to_feasible(best_h, best_n / 2.0)
    final_c5 = eval_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, best_n
