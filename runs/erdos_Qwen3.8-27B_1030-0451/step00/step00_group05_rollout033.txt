        ones = rng.choice(n, size=n // 2, replace=False)
        h[ones] = 1.0
        h[rng.randint(n)] = 0.5
    return h


def greedy_construct(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    n_ones = int(target)
    remaining = n_ones
    for pos in range(n):
        vals = [0.0, 1.0]
        if pos == n - 1 and remaining > 0 and n % 2 == 1:
            vals = [0.5]
        elif pos == n - 1 and remaining == 1:
            vals = [1.0]
        elif pos == n - 1 and remaining == 0:
            vals = [0.0]
        elif val_filter(vals, remaining):
            vals = [1.0]
        elif pos == n - 1:
            vals = [0.0, 1.0]

        best_val = 0.0
        best_score = 1e9
        for val in vals:
            if val == 1.0 and remaining <= 0:
                continue
            h[pos] = val
            score = 0.0
            for k in range(1, pos + 1):
                if h[pos - k] == 1.0 and val == 0.0:
                    score += 1.0
                elif h[pos - k] == 0.0 and val == 1.0:
                    score += 1.0
            if score < best_score:
                best_score = score
                best_val = val
        h[pos] = best_val
        if best_val == 1.0:
            remaining -= 1
        elif best_val == 0.5:
            remaining -= 0.5

    if remaining > 0:
        zeros = np.where(h == 0.0)[0]
        np.random.shuffle(zeros)
        for i in range(min(int(remaining), len(zeros))):
            h[zeros[i]] = 1.0
    return h


def val_filter(vals, remaining):
    if remaining == 0:
        return 1.0 not in vals
    return False


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = 1e9
    best_n = 0

    def check_and_update(h, n):
        nonlocal best_h, best_c5, best_n
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        return c5

    # Phase 1: Optimize initial_h_values
    iv = locals().get('initial_h_values', None)
    if iv is None:
        iv = globals().get('initial_h_values', None)
    if iv is not None:
        h0 = np.array(iv, dtype=float)
        n0 = len(h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = fix_sum(h0, n0 / 2.0)
        c5_0 = evaluate_c5(h0)
        print("Initial C5:", c5_0)
        check_and_update(h0, n0)

        elapsed = time.time() - start_time
        t1 = min(250, budget_s - elapsed - 60)
        if t1 > 30:
            h_opt, c5_opt = lp_optimize(h0, 5000, t1, rng)
            check_and_update(h_opt, n0)

        h_round = round_to_binary(h_opt)
        h_round = fix_sum(h_round, n0 / 2.0)
        check_and_update(h_round, n0)

        if time.time() - start_time < budget_s - 40:
            t2 = min(80, budget_s - (time.time() - start_time) - 30)
            if t2 > 20:
                h_opt2, c5_opt2 = lp_optimize(h_round, 3000, t2, rng)
                check_and_update(h_opt2, n0)

    # Phase 2: Try different n values with random starts
    n_candidates = [126, 128, 130, 132, 134, 136, 138, 140, 144, 148, 152, 156, 160]
    for n_val in n_candidates:
        if time.time() - start_time > budget_s - 60:
            break
        time_left = budget_s - (time.time() - start_time) - 50
        t_per = min(40, max(10, time_left / 3.0))

        for trial in range(2):
            if time.time() - start_time > budget_s - 50:
                break
            h_init = makeInitial(n_val, rng)
            h_opt, c5_opt = lp_optimize(h_init, 2000, t_per, rng)
            c5_check = check_and_update(h_opt, n_val)
            if c5_check < 0.3808:
                break
        if best_c5 < 0.3808:
            break

    # Phase 3: Greedy + optimize
    if time.time() - start_time < budget_s - 40:
        for n_val in [128, 144, 160]:
            if time.time() - start_time > budget_s - 30:
                break
            h_g = greedy_construct(n_val, rng)
            h_g = fix_sum(h_g, n_val / 2.0)
            t_g = min(30, budget_s - (time.time() - start_time) - 20)
            if t_g > 10:
                h_opt, c5_opt = lp_optimize(h_g, 1500, t_g, rng)
                check_and_update(h_opt, n_val)

    # Phase 4: Final polish
    if best_h is not None and time.time() - start_time < budget_s - 15:
        t_final = budget_s - (time.time() - start_time) - 10
        if t_final > 20:
            h_final, c5_final = lp_optimize(best_h, 5000, t_final, rng)
            check_and_update(h_final, best_n)

    if best_h is None:
        best_h = makeInitial(128, rng)
        best_n = 128
        best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
