    _opt_state['N'] = N
    _opt_state['tau'] = tau
    bounds = [(0.0, 1.0)] * N
    constraints = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    try:
        res = minimize(
            _smooth_obj, h, jac=_smooth_grad,
            method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(res.x, N)
        return h_new
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(120, 0.5)

    N0 = len(h0)
    h0 = project_feasible(h0, N0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    best_N = N0

    n_list = []
    for n in [N0 * 2, N0 * 3, N0 * 4, 500, 600, 800, 1000, 1200, 1500]:
        if n > N0 and n <= 2400 and n not in n_list:
            n_list.append(n)
    n_list = sorted(set(n_list))

    tau_schedule = [0.04, 0.02, 0.01, 0.005, 0.003, 0.0015, 0.0008]

    for target_N in n_list:
        if time.time() > deadline:
            break

        h_curr = upsample(best_h, best_N, target_N)
        h_curr = project_feasible(h_curr, target_N)
        curr_c5 = evaluate_c5(h_curr)
        N = target_N

        c_vals = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c_scale = max(float(np.max(c_vals)), 1.0)

        maxiter_per_tau = 400
        if N > 1000:
            maxiter_per_tau = 250
        if N > 1500:
            maxiter_per_tau = 150

        for tau_frac in tau_schedule:
            if time.time() > deadline:
                break
            tau = tau_frac * c_scale
            h_new = run_slsqp(h_curr, N, tau, maxiter=maxiter_per_tau)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < curr_c5 - 1e-12:
                h_curr = h_new
                curr_c5 = new_c5

        # Random restarts
        n_restarts = 4
        if time.time() > deadline - 60:
            n_restarts = 1
        for _ in range(n_restarts):
            if time.time() > deadline:
                break
            h_restart = h_curr + rng.normal(0, 0.015, N)
            h_restart = project_feasible(h_restart, N)
            for tau_frac in [0.02, 0.008, 0.002]:
                if time.time() > deadline:
                    break
                tau = tau_frac * c_scale
                h_new = run_slsqp(h_restart, N, tau, maxiter=150)
                new_c5 = evaluate_c5(h_new)
                if new_c5 < curr_c5 - 1e-12:
                    h_curr = h_new
                    curr_c5 = new_c5

        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_curr.copy()
            best_N = N

    # Final sanity check
    best_h = project_feasible(best_h, best_N)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_N
