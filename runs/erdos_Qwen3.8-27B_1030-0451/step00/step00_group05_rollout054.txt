    # Zero lag contribution
    grad += w_zero * dx * (1 - 2*h)
    
    # Positive lag contributions
    for R, w in zip(active_lags_pos, weights_pos):
        # H[j] contributes to positions j-j+R >= 0, i.e., j >= R
        # B_pos = dx * sum_{i=0}^{n-1-R} h[i+R] * (1-h[i])
        # d/dh[j]:
        #   if j is in {R, ..., n-1}: h[j] appears as h[i+R] with i=j-R, contributes (1-h[j-R])
        #   if j is in {0, ..., n-1-R}: h[j] appears as h[i] with i=j, contributes -h[j+R]
        if R > 0 and R < n:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    corr = np.correlate(h, u, mode='full') * dx
    return np.max(corr), corr

def project_feasible(x, n):
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(x)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-15
            if not np.any(mask):
                x[:] = 1.0
                x *= target / n  # fallback
                break
            x[mask] += diff / np.sum(mask)
        else:
            mask = x > 1e-15
            if not np.any(mask):
                x[:] = 0.0
                x += -diff / n
                break
            x[mask] += diff / np.sum(mask)
        np.clip(x, 0.0, 1.0, out=x)
    # Final correction
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute to elements with room
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_surrogate_gradient(h, n, dx, corr, T, delta):
    M = np.max(corr)
    active = corr >= M - delta
    if not np.any(active):
        active = np.zeros(n*2-1, dtype=bool)
        active[np.argmax(corr)] = True
    
    active_idx = np.where(active)[0]
    # Compute softmax weights
    logits = (corr[active_idx] - M) / T
    weights = np.exp(logits)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    c0 = n - 1
    
    # Zero lag
    if c0 in active_idx:
        idx_in_active = np.where(active_idx == c0)[0][0]
        w0 = weights[idx_in_active]
        grad += w0 * dx * (1 - 2*h)
    
    # For each active non-zero lag
    for i, k in enumerate(active_idx):
        if k == c0:
            continue
        w = weights[i]
        R = abs(k - c0)
        if R == 0 or R >= n:
            continue
        
        if k > c0:
            # Positive lag R: B = dx * sum_{i=0}^{n-1-R} h[i+R]*(1-h[i])
            grad[R:] += w * dx * (1 - h[:n-R])
            grad[:n-R] += w * dx * (-h[R:])
        else:
            # Negative lag R: B = dx * sum_{i=0}^{n-1-R} h[i]*(1-h[i+R])
            grad[:n-R] += w * dx * (1 - h[R:])
            grad[R:] += w * dx * (-h[:n-R])
    
    return grad, weights, active_idx

def local_optimize(h, n, rng, max_iters, T_init, T_min, delta):
    dx = 2.0 / n
    T = T_init
    alpha = 0.05
    best_h = h.copy()
    best_c5, _ = evaluate_c5(h)
    
    for it in range(max_iters):
        # Compute current objective
        c5, corr = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient
        grad, w_active, active_idx = compute_surrogate_gradient(h, n, dx, corr, T, delta)
        
        # Subtract mean for zero-sum constraint
        grad -= np.mean(grad)
        
        grad_norm = np.max(np.abs(grad))

        # Normalized descent direction

        if c5_new < c5:
            h = h_new
            alpha = min(alpha * 1.1, 0.5)  # slight increase
        else:
            alpha *= 0.7
        
        # Anneal temperature
        T = max(T_min, T * 0.999)
    
    return best_h, best_c5
