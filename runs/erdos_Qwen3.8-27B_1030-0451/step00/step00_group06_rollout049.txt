import numpy as np
import time

def overlap_values(h, dx):
    """Return all overlap values (the full correlation times dx)."""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def max_overlap(h, dx):
    """Return the C5 value."""
    return np.max(overlap_values(h, dx))

def projected_gradient_step(h, dx, step_size, top_m=50):
    """One projected gradient step to reduce the top overlaps."""
    N = len(h)
    ov = overlap_values(h, dx)
    
    # Find top-m lags
    N_ov = len(ov)
    if top_m >= N_ov:
        top_indices = np.arange(N_ov)
    else:
        # Get indices of top m values
        top_indices = np.argpartition(ov, -top_m)[-top_m:]
    
    # Compute gradient
    grad = np.zeros(N)
    for k_idx in top_indices:
        k = k_idx - (N - 1)  # actual lag
        # For j in 0..N-1:
        #   contribution from h[j+k]*(1-h[j]): 
        #     d/dh[j] = -h[j+k] if 0<=j+k<N
        #     d/dh[j+k] = (1-h[j]) if 0<=j<N
        # Let me use the formula I derived:
        # grad_j += (1-h[j-k]) * I(0<=j-k<N) + (-h[j+k]) * I(0<=j+k<N)
        
        # Vectorized:
        # First term: (1-h[j-k]) for j where 0<=j-k<N, i.e., max(0,k) <= j <= min(N-1, N-1+k)
        lo1 = max(0, k)
        hi1 = min(N, N + k)  # exclusive
        if hi1 > lo1:
            # j in [lo1, hi1), j-k in [lo1-k, hi1-k) = [max(0,k)-k, min(N, N+k)-k)
            # If k>=0: j in [k, N), j-k in [0, N-k)
            # If k<0: j in [0, N+k), j-k in [-k, N)
            if k >= 0:
                grad[k:] += (1.0 - h[:N-k])
            else:
                grad[:N+k] += (1.0 - h[-k:])
        
        # Second term: -h[j+k] for j where 0<=j+k<N, i.e., max(0,-k) <= j <= min(N-1, N-1-k)
        lo2 = max(0, -k)

            if k >= 0:
                grad[:N-k] -= h[k:]
            else:
                grad[-k:] -= h[:N+k]

    grad -= np.mean(grad)
    
    # Take step
    h_new = h - step_size * grad

    h_new = np.clip(h_new, 0.0, 1.0)

def _project_onto_box_sum(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    N = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    
    # Bisection for lambda
    # We want sum(clip(h + lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    if s > target_sum:
        # Need to decrease, so lambda < 0
        hi = 0.0
        lo = -max(h) - 0.1
    else:
        lo = 0.0
        hi = 1.0 - min(h) + 0.1
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        trial = np.clip(h + mid, 0.0, 1.0)
        s_trial = np.sum(trial)
        if s_trial > target_sum:
            hi = mid
        else:
            lo = mid
    
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction for exactness
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute among non-saturated entries
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    corr = np.correlate(h, 1.0 - h, mode='full')

        else:

            # n+k >= 0 means n >= -k = kp

            # h[n+k] = h[n-kp], which for n=kp..N-1 gives h[0..N-1-kp]
            # (1-h[n]) for n=kp..N-1
            # d/dh[j]:
