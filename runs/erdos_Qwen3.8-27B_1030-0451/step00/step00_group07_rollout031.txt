    if abs(res) > 1e-10:

                add = min(1.0 - h[i], res)
                h[i] += add

                rem = min(h[i], -res)

    b = np.zeros(N)
    b[order[:t]] = 1.0

 def upsample(h, factor):
    """Upsample h by a factor using linear interpolation."""
    N = len(h)
    N_new = N * factor
    x_old = np.linspace(0, 2, N, endpoint=False) + 1.0/N  # cell centers
    x_new = np.linspace(0, 2, N_new, endpoint=False) + 1.0/N_new
    h_new = np.interp(x_new, x_old, h)
    return h_new

def downsample(h, factor):
    """Downsample h by averaging blocks."""
    N = len(h)
    block = N // factor
    N_new = factor
    h_new = np.zeros(N_new)
    for i in range(N_new):
        start = i * block
        end = start + block
        h_new[i] = np.mean(h[start:end])
    return h_new
