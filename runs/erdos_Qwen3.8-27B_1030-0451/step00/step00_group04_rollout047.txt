    n = len(x)

    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):

        s = y.sum()
        if s > target:
            lo = mid

            L = n - a

        for i, q in enumerate(order):

            elif ell > 0:

                J[i, ell:] += -(1.0 - h[:L])
                J[i, :L] += h[ell:]
            else:
                a = -ell
                L = n - a
                J[i, :L] += -(1.0 - h[a:])
                J[i, a:] += h[:L]
            J[i, n] = 1.0

    x0[n] = t0

    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

def project_box_sum(x, target):
    x = np.asarray(x, dtype=np.float64)
    lo = x.min() - target - 1.0
    hi = x.max() - target + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:

        else:
