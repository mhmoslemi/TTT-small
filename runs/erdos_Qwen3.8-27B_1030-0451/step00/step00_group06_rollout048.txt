```python
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    return (c * dx).max()

def smooth_max(g, alpha):
    # Numerically stable softmax
    m = g.max()
    return m + np.log(np.sum(np.exp(g - m))) / alpha

def project_to_constraint(h, target_sum):
    # Project onto {0<=h<=1, sum(h) = target_sum}
    # Use bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float((c * dx).max())

def project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)
```
