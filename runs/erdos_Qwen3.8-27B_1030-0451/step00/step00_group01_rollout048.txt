def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def full_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def ls_exp_obj(h, beta):
    v = full_corr(h)
    m = v.max()
    e = np.exp(beta * (v - m))
    Z = e.sum()
    return m + np.log(Z) / beta

def ls_exp_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    v = full_corr(h)
    m = v.max()
    e = np.exp(beta * (v - m))
    Z = e.sum()
    w = e / Z
    f = m + np.log(Z) / beta
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            mm = n - shift
            grad[shift:] += wi * dx * (1.0 - h[:mm])
            grad[:mm] += wi * (-dx) * h[shift:]
        else:
            s = -shift
            mm = n - s
            grad[:mm] += wi * dx * (1.0 - h[s:])
            grad[s:] += wi * (-dx) * h[:mm]
    return f, grad

def con_fun(h):
    return np.sum(h) - len(h) / 2.0

def con_jac(h):
    return np.ones(len(h))

        pass
    return None

    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    b_rows = []
    for idx in order:
        shift = idx - (n - 1)
        g = np.zeros(n)
        if shift >= 0:

            g[shift:] += dx * (1.0 - h0[:mm])
            g[:mm] += dx * (-h0[shift:])
        else:

            g[:mm] += dx * (1.0 - h0[s:])
            g[s:] += dx * (-h0[:mm])

        row = np.append(g, -1.0)
        rhs = g @ h0 - v[idx]
        A_rows.append(row)
        b_rows.append(rhs)
    
    A = np.array(A_rows)
    b = np.array(b_rows)

def _proj_sum(x, lam):
    return np.sum(np.clip(x + lam, 0.0, 1.0))

def project_sum_box(x, target=None):
    n = len(x)
    if target is None:
        target = n / 2.0
    x = np.asarray(x, dtype=np.float64).copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if _proj_sum(x, mid) < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            slack = 1.0 - result
            ts = slack.sum()
            if ts > 1e-14:
                result += (diff / ts) * slack
        else:
            ta = result.sum()
            if ta > 1e-14:
                result += (diff / ta) * result
        result = np.clip(result, 0.0, 1.0)
    return result

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def full_corr(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def ls_exp_fun(h, beta):
    v = full_corr(h)
    m = v.max()
    e = np.exp(beta * (v - m))
    Z = e.sum()
    return m + np.log(Z) / beta

def ls_exp_jac(h, beta):

    dx = 2.0 / n
    v = full_corr(h)
    m = v.max()

    Z = e.sum()
    w = e / Z
    f = m + np.log(Z) / beta
    grad = np.zeros(n)

        wi = w[idx]
        if wi < 1e-14:
            continue

        if shift >= 0:

        else:
            s = -shift
            mm = n - s
            grad[:mm] += wi * dx * (1.0 - h[s:])
            grad[s:] += wi * (-dx) * h[:mm]
    return grad

def con_fun(h):

def con_jac(h):
    return np.ones(len(h))

        c5_new = eval_c5(h_new)
        if np.isfinite(c5_new):

    except Exception:
        pass

def grad_shift(h, shift):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if shift >= 0:
        mm = n - shift
        g[shift:] += dx * (1.0 - h[:mm])
        g[:mm] += dx * (-h[shift:])
    else:
        s = -shift
        mm = n - s
        g[:mm] += dx * (1.0 - h[s:])
        g[s:] += dx * (-h[:mm])
    return g

def try_lp(h0, radius, top_k):
