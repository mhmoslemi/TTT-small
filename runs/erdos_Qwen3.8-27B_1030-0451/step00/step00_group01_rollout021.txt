    stall_limit = 300
    it = 0
    
    while time.time() - start < max_time:
        if it % restart_interval == it:
            pass
        if it > 0 and it % restart_interval == 0:
            h = best_h.copy()
            alpha *= 0.5
            stall = 0
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_vals = corr * dx
        max_idx = np.argmax(c5_vals)
        c5 = c5_vals[max_idx]
        lag = max_idx - (n - 1)
        
        grad = _grad_at_lag(h, lag, n, dx)
        
        # Add small subgradient from a few other active lags for diversity
        top_k = np.argsort(c5_vals)[-3:][::-1]
        for idx in top_k[1:]:
            lg = idx - (n - 1)
            grad += 0.3 * _grad_at_lag(h, lg, n, dx)
        
        alpha = max(0.001, 1.0 / (1.0 + it * 0.02))
        h_new = h - alpha * grad
        h_new = _project(h_new, n)
        
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            h = h_new.copy()
            stall += 1
        
        it += 1
        
        if stall > stall_limit:
            break
    
    # Polishing: very small steps
    polish_it = 0
    polish_alpha = 0.005
    while time.time() - start < max_time and polish_it < 500:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_vals = corr * dx
        max_idx = np.argmax(c5_vals)
        lag = max_idx - (n - 1)
        grad = _grad_at_lag(h, lag, n, dx)
        h_new = h - polish_alpha * grad
        h_new = _project(h_new, n)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        polish_it += 1
    
    return best_h, best_c5

def _make_sinusoidal(n, freq, phase, amp, seed):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0, 1)
    return _project(h, n)

def _make_block(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    n_blocks = rng.integers(3, 8)
    boundaries = np.sort(rng.choice(range(1, n), size=n_blocks - 1, replace=False))
    state = 1
    prev = 0
    for b in list(boundaries) + [n]:
        h[prev:b] = state
        state = 1 - state
        prev = b
    return _project(h, n)

def _make_thue_morse(n):
    # Thue-Morse sequence: t[i] = parity of popcount(i)
    t = np.zeros(n, dtype=int)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    h = 1.0 - t  # Map to {0,1}
    # Adjust to have exactly n/2 ones
    n_ones = np.sum(h)
    if n_ones != n // 2:
        if n_ones > n // 2:
            idxs = np.where(h == 1)[0]
            to_zero = rng_local.choice(idxs, size=n_ones - n//2, replace=False)
            h[to_zero] = 0
        else:
            idxs = np.where(h == 0)[0]
            to_one = rng_local.choice(idxs, size=n//2 - n_ones, replace=False)
            h[to_one] = 1
    return h.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_local
    rng_local = np.random.default_rng(seed)
    start_time = time.time()
    
    def elapsed():
        return time.time() - start_time
    
    def time_left():
        return budget_s - elapsed()
    
    best_h = None
    best_c5 = float('inf')
    
    # Get initial values if available
    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        # Validate
        if np.all(h_init >= 0) and np.all(h_init <= 1) and abs(np.sum(h_init) - len(h_init)/2.0) < 0.01:
            pass
        else:
            h_init = _project(h_init, len(h_init))
    
    candidates = []
    
    # Candidate 1: initial
    if h_init is not None:
        candidates.append(h_init.copy())
    
    # Candidate 2: Sinusoidal at n=84
    if h_init is None:
        candidates.append(_project(0.5 * np.ones(84), 84))
    else:
        n0 = len(h_init)
    
    # Generate various starting points at different resolutions
    resolutions = [84, 100, 120, 140, 168, 200, 252, 300, 400]
    
    for n in resolutions:
        if n == (len(h_init) if h_init is not None else 0):
            continue
        # Sinusoidal
        candidates.append(_make_sinusoidal(n, rng_local.integers(1, 5), 
                                            rng_local.uniform(0, 2*np.pi), 0.5, seed+n))
        # Block
        candidates.append(_make_block(n, seed+n))
        # Thue-Morse
        candidates.append(_make_thue_morse(n))
        # Random blocks with varying sizes
        h = np.zeros(n)
        pos = 0
        while pos < n:
            width = max(1, int(rng_local.integers(n//20, n//4)))
            h[pos:pos+width] = 1.0 if (pos // max(1, n//8)) % 2 == 0 else 0.0
            pos += width
        candidates.append(_project(h, n))
    
    # Budget allocation: give more time to initial and smaller n
    n_candidates = len(candidates)
    # Sort by priority: initial first, then smaller n
    # Assign time proportional to importance
    base_time = 20  # seconds per candidate minimum
    
    candidate_times = []
    for h in candidates:
        n = len(h)
        # Give more time to smaller n (faster iterations)
        t = max(base_time, min(120, 200.0 / (n / 84.0)))
        candidate_times.append(t)
    
    total_time = sum(candidate_times)
    avail = budget_s * 0.85
    if total_time > avail:
        scale = avail / total_time
        candidate_times = [t * scale for t in candidate_times]
    
    for i, h in enumerate(candidates):
        if time_left() < 10:
            break
        n = len(h)
        h = _project(h, n)
        c5_initial = _compute_c5(h)
        
        t_alloc = candidate_times[min(i, len(candidate_times)-1)]
        t_alloc = min(t_alloc, time_left() - 5)
        if t_alloc < 5:
            continue
        
        result_h, result_c5 = _optimize(h, n, t_alloc, seed=seed+i*1000)
        
        if result_c5 < best_c5:
            best_c5 = result_c5
            best_h = result_h.copy()
    
    # Final polishing on best solution
    if best_h is not None and time_left() > 20:
        n = len(best_h)
        polish_time = min(120, time_left() * 0.3)
        result_h, result_c5 = _optimize(best_h, n, polish_time, seed=seed+99999)
        if result_c5 < best_c5:
            best_c5 = result_c5
            best_h = result_h
    
    if best_h is None:
        h = np.zeros(84)
        h[:42] = 1.0
        best_h = _project(h, 84)
        best_c5 = _compute_c5(best_h)
    
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, len(best_h))
    best_c5 = _compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
