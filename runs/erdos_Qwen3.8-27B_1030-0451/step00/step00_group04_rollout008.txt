            if c_val < -1e-10 or c_val > 1.0 + 1e-10:

            c_val = max(0.0, min(1.0, c_val))
            h[i] = a_val
            h[j] = b_val
            h[k] = c_val
            val = evaluate_c5(h)
            if val < best_score - 1e-14:
                best_score = val
                best_vals = (a_val, b_val, c_val)
    h[i], h[j], h[k] = best_vals
    return best_score

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0

    global initial_h_values
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        h0 = np.full(250, 0.5, dtype=np.float64)

    n = len(h0)
    T = n / 2.0

    h = project_feasible(h0, T)
    current_score = evaluate_c5(h)
    best_h = h.copy()
    best_score = current_score

    rng = np.random.default_rng(seed)

    all_i = np.arange(n)
    stall_rounds = 0
    max_stall = 5
    sweep = 0

    while time.time() < deadline:
        sweep += 1
        improved_this_sweep = False

        # Subgradient step every few sweeps
        if sweep % 4 == 1:
            grad = subgradient_of_active(h, n, top_k=40, tau_frac=0.05)
            step_size = 0.02 / np.max(np.abs(grad) + 1e-10)
            h_candidate = h - step_size * grad
            h_candidate = project_feasible(h_candidate, T)
            cand_score = evaluate_c5(h_candidate)
            if cand_score < current_score - 1e-14:
                h = h_candidate
                current_score = cand_score
                if cand_score < best_score - 1e-14:
                    best_h = h.copy()
                    best_score = cand_score
                improved_this_sweep = True

        # Pair coordinate descent
        # Prioritize pairs based on subgradient
        grad = subgradient_of_active(h, n, top_k=30, tau_frac=0.06)
        press = np.abs(grad)
        press_rank = np.argsort(press)

        n_pairs = min(300, n * 2)
        pairs = []
        # High-pressure with low-pressure
        n_cross = min(40, n // 4)
        for k in range(n_cross):
            i_idx = press_rank[-1 - k]
            j_idx = press_rank[k]
            if i_idx != j_idx:
                pairs.append((min(i_idx, j_idx), max(i_idx, j_idx)))
        # Random pairs
        for _ in range(n_pairs - len(pairs)):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            if a != b:
                pairs.append((min(a, b), max(a, b)))
        rng.shuffle(pairs)

        for (i, j) in pairs:
            if time.time() > deadline:
                break
            old_s = evaluate_c5(h)
            new_s = pair_line_search(h, i, j, old_s, n_coarse=5, n_fine=8)
            if new_s < old_s - 1e-14:
                improved_this_sweep = True
                if new_s < best_score - 1e-14:
                    best_h = h.copy()
                    best_score = new_s
                    if best_score <= 0.3808:
                        break

        # Occasional triple optimization for escape
        if sweep % 5 == 0:
            n_triples = 30
            for _ in range(n_triples):
                if time.time() > deadline:
                    break
                idx = rng.choice(n, size=3, replace=False)
                old_s = evaluate_c5(h)
                new_s = triple_optimize(h, tuple(idx), old_s, grid_size=6)
                if new_s < old_s - 1e-14:
                    improved_this_sweep = True
                    if new_s < best_score - 1e-14:
                        best_h = h.copy()
                        best_score = new_s

        if not improved_this_sweep:
            stall_rounds += 1
            if stall_rounds >= max_stall:
                escaped = False
                # Multi-scale perturbation
                for scale in [0.3, 0.15, 0.05, 0.02]:
                    h_pert = h.copy()
                    for _ in range(int(n * 0.1)):
                        a = int(rng.integers(n))
                        b = int(rng.integers(n))
                        if a == b:
                            continue
                        delta = rng.uniform(-scale, scale)
                        h_pert[a] += delta
                        h_pert[b] -= delta
                    h_pert = project_feasible(h_pert, T)
                    p_score = evaluate_c5(h_pert)
                    if p_score < current_score - 1e-14:
                        h = h_pert
                        current_score = p_score
                        if p_score < best_score:
                            best_h = h.copy()
                            best_score = p_score
                        escaped = True
                        break
                if escaped:
                    stall_rounds = 0
                else:
                    break  # truly stuck
            continue
        else:
            stall_rounds = 0

        if best_score <= 0.3808:
            break

    # Final validation
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0.0, 1.0)
    residual = T - np.sum(h_final)
    if abs(residual) > 1e-10:
        interior = np.where((h_final > 1e-6) & (h_final < 1.0 - 1e-6))[0]
        if len(interior) > 0:
            h_final[interior[0]] += residual
            h_final = np.clip(h_final, 0.0, 1.0)
        residual2 = T - np.sum(h_final)
        if abs(residual2) > 1e-12:
            interior2 = np.where((h_final > 1e-8) & (h_final < 1.0 - 1e-8))[0]
            if len(interior2) > 0:
                h_final[interior2[0]] += residual2
                h_final = np.clip(h_final, 0.0, 1.0)

    final_score = evaluate_c5(h_final)

    if final_score > best_score + 1e-6:
        h_final = np.clip(h, 0.0, 1.0)
        residual = T - np.sum(h_final)
        if abs(residual) > 1e-10:
            interior = np.where((h_final > 1e-6) & (h_final < 1.0 - 1e-6))[0]
            if len(interior) > 0:
                h_final[interior[0]] += residual
                h_final = np.clip(h_final, 0.0, 1.0)
        final_score = evaluate_c5(h_final)

    return (h_final, final_score, len(h_final))
