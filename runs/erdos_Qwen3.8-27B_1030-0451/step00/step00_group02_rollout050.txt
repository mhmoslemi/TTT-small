            h[i] = ai
            h[j] = bj
        T *= cool
    return h, c5

def make_initial(n, rng):
    """Create a random binary (or near-binary) initial vector."""
    if n % 2 == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    else:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h[n // 2] = 0.5
    perm = rng.permutation(n)
    return h[perm].copy()
