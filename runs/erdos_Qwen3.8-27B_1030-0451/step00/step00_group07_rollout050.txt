```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute the exact C5 value for a feasible h."""
    N = len(h)
    dx = 2.0 / N
    corrs = np.correlate(h, 1-h, mode='full')
    return np.max(corrs) * dx

def smooth_obj_and_grad(h, tau):
    """Compute smooth objective and gradient."""
    N = len(h)
    dx = 2.0 / N
    corrs = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corrs)
    shifted = (corrs - max_c) / tau
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    alpha = exp_shifted / total
    obj = dx * (max_c + tau * np.log(total))
    
    grad = np.zeros(N)
    for r in range(2*N-1):
        if alpha[r] < 1e-10:
            continue
        lag = r - (N-1)
        w = dx * alpha[r]
        if lag >= 0:
            grad[:N-lag] += w * (1 - h[lag:N])
            grad[lag:] += w * (-h[:N-lag])
        else:
            d = -lag
            grad[d:] += w * (1 - h[:N-d])
            grad[:N-d] += w * (-h[d:])
    
    return obj, grad

def project_feasible(g, target_sum):
    """Project g onto {0 <= h <= 1, sum(h) = target_sum}."""
    N = len(g)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(g + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(g + (lo+hi)/2, 0.0, 1.0)
    # Correct tiny numerical error
    err = target_sum - np.sum(result)
    if abs(err) < 1e-6:
        # Distribute error among non-boundary elements
        mask = (result > 1e-8) & (result < 1 - 1e-8)
        if np.sum(mask) > 0:
            result[mask] += err / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result
```

```python
def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def smooth_obj_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corrs = np.correlate(h, 1-h, mode='full')
    mc = corrs.max()
    shifted = (corrs - mc) / tau
    es = np.exp(shifted)
    total = es.sum()
    alpha = es / total
    obj = dx * (mc + tau * np.log(total))
    grad = np.zeros(N)
    for r in range(2*N-1):
        if alpha[r] < 1e-10:
            continue
        lag = r - (N-1)
        w = dx * alpha[r]
        if lag >= 0:
            grad[:N-lag] += w * (1.0 - h[lag:N])
            grad[lag:] -= w * h[:N-lag]
        else:
            d = -lag
            grad[d:] += w * (1.0 - h[:N-d])
            grad[:N-d] -= w * h[d:]
    return obj, grad

def project_feasible(g, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(g + mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(g + (lo+hi)*0.5, 0, 1)
    # Fix sum
    err = target - h.sum()
    if abs(err) > 1e-12:
        mask = (h > 1e-9) & (h < 1-1e-9)
        n_mask = mask.sum()
        if n_mask > 0:
            h[mask] += err / n_mask
            h = np.clip(h, 0, 1)
    return h

    if new_n == old_n:
        return h.copy()
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n / 2.0)
```

```python
    """Find optimal h[j] given all other coordinates fixed."""
    N = len(h)
    dx = 2.0 / N
    t = h[j]
    
    # Compute A_r and B_r for each r
    # corrs_r = A_r * t + B_r
    # A_r = coefficient of h[j] in corrs_r
    # B_r = corrs_r - A_r * t (i.e., corrs_r with h[j] set to 0)
    
    h[j] = 0  # temporarily set to 0

    h[j] = 1  # temporarily set to 1
    corrs_1 = np.correlate(h, 1-h, mode='full')  # corrs with h[j]=1
    h[j] = t  # restore
    
    A = corrs_1 - corrs_0  # coefficient
    B = corrs_0  # intercept

    # The minimum is at an intersection or at a boundary
    # Intersection of lines r1 and r2: A_r1*t + B_r1 = A_r2*t + B_r2

        val = np.max(A * t_check + B)
        if val < best_val:

            best_t = t_check
    
    # Check intersections
    # For efficiency, we can be smart: only check intersections that are in [0,1]
    # and where both lines are on the upper envelope
    
    # Simple approach: check all pairs (might be slow for large N)
    # Better: just evaluate at a grid + the intersection points near the minimum
```

```python
import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def smooth_obj_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corrs = np.correlate(h, 1.0-h, mode='full')
    mc = corrs.max()
    shifted = (corrs - mc) / tau
    es = np.exp(shifted)
    total = es.sum()
    alpha = es / total
    obj = float(dx * (mc + tau * np.log(total)))
    grad = np.zeros(N)
    for r in range(2*N-1):
        a = alpha[r]
        if a < 1e-10:
            continue
        lag = r - (N-1)
        w = dx * a
        if lag >= 0:
            grad[:N-lag] += w * (1.0 - h[lag:N])
            grad[lag:] -= w * h[:N-lag]
        else:
            d = -lag
            grad[d:] += w * (1.0 - h[:N-d])
            grad[:N-d] -= w * h[d:]
    return obj, grad

def project_feasible(g, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(g + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(g + (lo+hi)*0.5, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0-1e-9)
        if mask.sum() > 0:
            h[mask] += err / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n * 0.5)

def coord_descent(h, target, max_passes=50, tol=1e-10):
    N = len(h)
    for _pass in range(max_passes):
        max_change = 0.0
        for j in range(N):
            t_save = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0-h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0-h, mode='full')
            A = c1 - c0
            B = c0
            h[j] = t_save
            # Find optimal t in [0,1]
            ts = np.linspace(0.0, 1.0, 300)
            vals = np.max(A[None,:] * ts[:,None] + B[None,:], axis=1)
            idx = np.argmin(vals)
            if 0 < idx < len(ts)-1:
                ts2 = np.linspace(ts[idx-1], ts[idx+1], 100)
                vals2 = np.max(A[None,:] * ts2[:,None] + B[None,:], axis=1)
                t_new = ts2[np.argmin(vals2)]
            else:
                t_new = ts[idx]
            change = abs(t_new - t_save)
            if change > max_change:
                max_change = change
            h[j] = t_new
        if max_change < tol:
            break
    return h

def gradient_descent(h, target, max_iter=500, base_step=0.5):
    N = len(h)
    for tau in [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]:
        step = base_step
        obj_prev = smooth_obj_grad(h, tau)[0]
        for it in range(max_iter // len([5.0,2.0,1.0,0.5,0.2,0.1,0.05,0.02])):
            obj, grad = smooth_obj_grad(h, tau)
            alpha = step
            improved = False
            for _ in range(30):
                h_new = project_feasible(h - alpha * grad, target)
                obj_new = smooth_obj_grad(h_new, tau)[0]
                if obj_new < obj - 1e-15:
                    h = h_new
                    alpha *= 1.1
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                break
            if abs(obj_new - obj_prev) < 1e-12:
                obj_prev = obj_new
    return h
```
