import numpy as np
from scipy.interpolate import interp1d
import time

def compute_corr(h, dx):
    """Compute the full correlation array multiplied by dx."""
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h, dx):
    """Compute C5 = max of correlation array."""
    return float(np.max(compute_corr(h, dx)))

def project(h, n):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = n/2}."""
    target = n / 2.0
    # Bisection for the offset
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    return result

def grad_for_lag(h, k, dx):
    """Gradient of g_k(h) at the given lag k."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k  # number of terms
        if m <= 0:
            return grad
        grad[:m] += dx * (1.0 - h[k:k+m])
        grad[k:k+m] += -dx * h[:m]
    else:
        ak = -k
        m = n - ak
        if m <= 0:
            return grad
        grad[ak:ak+m] += dx * (1.0 - h[:m])
        grad[:m] += -dx * h[ak:ak+m]
    return grad

def upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation."""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='linear', bounds_error=False)
    return np.clip(f(x_new), 0.0, 1.0)

def optimize_pg(h_init, n, dx, max_iters=2000, time_budget=100.0, use_momentum=True):
    """Projected gradient with momentum."""
    n = len(h_init)
    h = h_init.copy()
    h = project(h, n)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    v = np.zeros(n)
    start_time = time.time()
    lr = 0.01
    momentum = 0.9
    
    for it in range(max_iters):
        if time.time() - start_time > time_budget:
            break
        
        corr = compute_corr(h, dx)
        max_val = np.max(corr)
        
        # Find active lags (within a small tolerance)
        if it < 100:
            tol = 1e-8
        else:
            tol = 0.001
        
        active = np.where(corr >= max_val - tol)[0]
        
        # Average gradient over active lags
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            grad += grad_for_lag(h, k, dx)
        grad /= len(active)
        
        if use_momentum:
            v = momentum * v - lr * grad
        else:
            v = -lr * grad
        
        h_new = project(h + v, n)
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:

            best_h = h_new.copy()
            h = h_new

                best_h = h_new.copy()
                h = h_new
