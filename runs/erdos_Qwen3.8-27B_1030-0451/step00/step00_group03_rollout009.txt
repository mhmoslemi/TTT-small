Let me think about the mirror-complement parameterization more carefully:
- For even n, m = n//2
- h[0:m] = p, h[m:n] = 1 - p[::-1]
- Sum = sum(p) + sum(1-p[::-1]) = sum(p) + m - sum(p) = m = n/2 ✓
- Each pair h[i] + h[n-1-i] = p[i] + 1 - p[i] = 1 ✓

The objective: C5 = max_k correlate(h, 1-h, mode="full") * dx where dx = 2/n

2. `build_h(p, n)`: construct h from half-vector p

4. `project_full(y)`: project onto {0≤x≤1, sum(x)=n/2} for fallback

Wait, the log-sum-exp formula: log(sum(exp(x))) = max(x) + log(sum(exp(x - max(x))))
So tau * log(sum(exp(corr/tau))) = tau * (max(corr/tau) + log(sum(exp(corr/tau - max(corr/tau)))))
= tau * (max(corr)/tau + log(sum(exp((corr - max(corr))/tau))))
= max(corr) + tau * log(sum(exp((corr - max(corr))/tau)))

This is a smooth upper bound on max(corr) that approaches max(corr) as tau → 0.

For the SLSQP epigraph:
minimize t
subject to: t - corr[i] >= 0 for selected i
            0 <= p[j] <= 1

import numpy as np
from scipy.optimize import minimize
import time

_state = {'best_c': float('inf'), 'best_h': None, 'deadline': 0}

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    val = float(np.max(corr))
    if not np.isfinite(val):
        return float('inf')
    return val

def build_h(p, n):
    p = np.clip(np.asarray(p, dtype=np.float64), 0.0, 1.0)
    if n % 2 == 0:
        m = n // 2
    else:
        m = n // 2
    if len(p) != m:
        p = _interp_p(p, m)
    h = np.empty(n, dtype=np.float64)
    h[:m] = p
    if n % 2 == 1:
        h[m] = 0.5
        h[m+1:] = 1.0 - p[::-1]
    else:
        h[m:] = 1.0 - p[::-1]
    return h

def _interp_p(p, m_new):
    p = np.asarray(p, dtype=np.float64)
    m_old = len(p)
    if m_old == m_new:
        return p.copy()
    if m_old == 1:
        return np.full(m_new, p[0])
    x_old = (np.arange(m_old) + 0.5) / m_old
    x_new = (np.arange(m_new) + 0.5) / m_new
    return np.interp(x_new, x_old, p)

def p_from_h(h, n):
    h = np.asarray(h, dtype=np.float64)
    if n % 2 == 0:
        m = n // 2
        p = 0.5 * (h[:m] + 1.0 - h[m:n][::-1])
    else:
        m = n // 2
        p = 0.5 * (h[:m] + 1.0 - h[m+1:n][::-1])
    return np.clip(p, 0.0, 1.0)

def project_full(y):
    """Project y onto {0<=x<=1, sum(x) = n/2} using bisection on shift."""
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    target = n / 2.0
    
    def _sum_for_alpha(alpha):
        return np.sum(np.clip(y + alpha, 0.0, 1.0))
    
    # Find alpha such that sum(clip(y+alpha, 0, 1)) = target
    lo = -1.0 - np.max(y)
    hi = 1.0 - np.min(y)
    
    # Ensure bracket
    while _sum_for_alpha(lo) > target:
        lo -= 1.0
    while _sum_for_alpha(hi) < target:
        hi += 1.0
    
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if _sum_for_alpha(mid) > target:
            hi = mid
        else:
            lo = mid
    
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def soft_objective(p, tau, n, state):
    """Smoothed C5 objective for L-BFGS-B."""
    if time.monotonic() > state['deadline']:
        return state['best_c']
    p_clipped = np.clip(p, 0.0, 1.0)
    h = build_h(p_clipped, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_exact = float(np.max(corr))
    if np.isfinite(c_exact) and c_exact < state['best_c']:
        state['best_c'] = c_exact
        state['best_h'] = h.copy()
        state['best_p'] = p_clipped.copy()
    if tau < 1e-10:
        return c_exact
    # Numerically stable log-sum-exp
    scaled = corr / tau
    max_s = np.max(scaled)
    val = float(tau * (max_s + np.log(np.sum(np.exp(scaled - max_s)))))
    if not np.isfinite(val):
        return c_exact + 1.0
    return val

def generate_initial_candidates(n, seed, state):
    """Generate diverse initial p-vectors for dimension n."""
    m = n // 2 if n % 2 == 0 else n // 2
    rng = np.random.default_rng(seed)
    candidates = []
    
    # 1. Constant
    candidates.append(np.full(m, 0.5))
    
    # 2. Linear ramp
    candidates.append(np.linspace(0.0, 1.0, m))
    candidates.append(np.linspace(1.0, 0.0, m))
    
    # 3. Sinusoidal modes
    t = (np.arange(m) + 0.5) / m
    for k in [1, 2, 3, 4, 5, 7, 11]:
        for amp in [0.2, 0.4, 0.45]:
            candidates.append(np.clip(0.5 + amp * np.sin(2*np.pi*k*t), 0, 1))
            candidates.append(np.clip(0.5 + amp * np.cos(2*np.pi*k*t), 0, 1))
    
    # 4. Mixed harmonics
    for trial in range(8):
        coeffs = rng.uniform(-0.3, 0.3, size=5)
        vals = 0.5 + sum(c * np.sin(2*np.pi*(i+1)*t + rng.uniform(0, 2*np.pi)) 
                       for i, c in enumerate(coeffs))
        candidates.append(np.clip(vals, 0, 1))
    
    # 5. Random step functions
    for trial in range(6):
        n_steps = rng.integers(2, min(m//2+1, 12))
        boundaries = np.sort(rng.uniform(0, 1, n_steps))
        vals = rng.uniform(0, 1, n_steps + 1)
        p = np.zeros(m)
        idx = np.digitize(t, boundaries)
        p[:] = vals[idx]
        candidates.append(np.clip(p, 0, 1))
    
    return candidates

Now let me think about the SLSQP epigraph. The idea is:
- We have the exact correlation values corr[0:2n-1]
- We identify the top-k lags
- We solve: min t s.t. t >= corr_i for i in top_k, 0 <= p <= 1

The constraint t - corr_i(p) >= 0 means the function is t - corr_i(p).

For SLSQP with numerical gradients, the constraint function just needs to return the value.

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    val = float(np.max(corr))
    return val if np.isfinite(val) else float('inf')


def build_h(p, n):
    p = np.clip(np.asarray(p, dtype=np.float64), 0.0, 1.0)
    m = n // 2
    if len(p) != m:
        p = interp_p(p, m)
    h = np.empty(n, dtype=np.float64)
    h[:m] = p
    if n % 2 == 1:
        h[m] = 0.5
        h[m + 1:] = 1.0 - p[::-1]
    else:
        h[m:] = 1.0 - p[::-1]
    return h


def p_from_h(h, n):
    h = np.asarray(h, dtype=np.float64)
    m = n // 2
    if n % 2 == 0:
        p = 0.5 * (h[:m] + 1.0 - h[m:n][::-1])
    else:
        p = 0.5 * (h[:m] + 1.0 - h[m + 1:n][::-1])
    return np.clip(p, 0.0, 1.0)


def interp_p(p, m_new):
    p = np.asarray(p, dtype=np.float64)
    m_old = len(p)
    if m_old == m_new:
        return p.copy()
    if m_old == 1:
        return np.full(m_new, float(p[0]))
    x_old = (np.arange(m_old) + 0.5) / m_old
    x_new = (np.arange(m_new) + 0.5) / m_new
    return np.interp(x_new, x_old, p)
