        return np.sum(np.clip(y - lam, 0.0, 1.0))

        c5_new = _compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            step = min(step * 1.08, 0.3)
            beta = min(beta * 1.03, 100.0)
        else:
            no_improve += 1
            step *= 0.85
            beta = max(beta * 0.98, 3.0)
    return best_h, best_c5

def _transfer_resolution(h_src, n_src, n_tgt):
    if n_tgt == n_src:
        return h_src.copy()
    x_old = np.arange(n_src)
    x_new = np.arange(n_tgt)
    f = interp1d(x_old, h_src, kind='linear', fill_value=(h_src[0], h_src[-1]))
    h_new = f(x_new)
    return np.clip(h_new, 0.0, 1.0)

def _make_structural_start(n, seed):
    rng = np.random.RandomState(seed)
    target_sum = n / 2.0
    # Create periodic-like pattern with random phase
    period = rng.randint(n // 6, n // 3 + 1)
    duty = 0.4 + rng.rand() * 0.2
    x = np.arange(n)
    h = ((x % period) < period * duty).astype(float)
    h += rng.randn(n) * 0.1
    h = np.clip(h, 0.0, 1.0)
    h = _project_feasible(h, target_sum)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    n_parent = len(initial_h_values)
    h_parent = np.array(initial_h_values, dtype=float).copy()
    dx_p = 2.0 / n_parent
    h_parent = _project_feasible(h_parent, n_parent / 2.0)

    best_h = h_parent.copy()
    best_c5 = _compute_c5(best_h, dx_p)
    best_n = n_parent

    # Phase 1: Optimize at parent resolution
    t_phase1 = t_start + budget_s * 0.30
    if t_phase1 > time.time():
        h_opt, c5_opt = _optimize_local(h_parent, n_parent, t_phase1, seed)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_parent

    # Phase 2: Try nearby and other resolutions
    candidate_ns = []
    for delta in [1, 2, 3, 5, 8]:
        candidate_ns.append(n_parent + delta)
        candidate_ns.append(n_parent - delta)
    for n_raw in [300, 350, 400, 500, 600, 800, 1000]:
        if abs(n_raw - n_parent) > 5:
            candidate_ns.append(n_raw)
    candidate_ns = [n for n in candidate_ns if 100 <= n <= 2400]
    candidate_ns = sorted(set(candidate_ns), key=lambda x: abs(x - n_parent))

    n_evaluated = {n_parent}
    t_phase2_end = t_start + budget_s * 0.85
    seed_cur = seed + 200

    for n_try in candidate_ns:
        if n_try in n_evaluated:
            continue
        if time.time() >= t_phase2_end:
            break
        time_left = t_phase2_end - time.time()
        if time_left < 3.0:
            break

        # Transfer best solution to this resolution
        h_trans = _transfer_resolution(best_h, best_n, n_try)
        h_trans = _project_feasible(h_trans, n_try / 2.0)

        # Also try a structural start for diversity
        h_struct = _make_structural_start(n_try, seed_cur)

        # Optimize from transfer
        t_sub = time.time() + time_left * 0.45
        h_try1, c5_try1 = _optimize_local(h_trans, n_try, t_sub, seed_cur)

        # Optimize from structural if time remains
        t_sub2 = time.time() + (t_phase2_end - time.time()) * 0.5
        h_try2, c5_try2 = _optimize_local(h_struct, n_try, max(t_sub2, time.time() + 0.1), seed_cur + 1)

        n_evaluated.add(n_try)
        if c5_try1 < best_c5:
            best_c5 = c5_try1
            best_h = h_try1.copy()
            best_n = n_try
        if c5_try2 < best_c5:
            best_c5 = c5_try2
            best_h = h_try2.copy()
            best_n = n_try
        seed_cur += 10

    # Phase 3: Final polish at best resolution
    t_phase3_end = t_start + budget_s * 0.95
    if time.time() < t_phase3_end - 5.0:
        t_left = t_phase3_end - time.time()
        # Multiple short polish attempts with different seeds
        for polish_seed in [seed + 1000, seed + 1001, seed + 1002]:
            if time.time() >= t_phase3_end - 1.0:
                break
            t_sub = time.time() + max(1.0, t_left * 0.33)
            perturb_scale = 0.005
            rng_p = np.random.RandomState(polish_seed)
            h_pert = _project_feasible(best_h + rng_p.randn(best_n) * perturb_scale, best_n / 2.0)
            h_pol, c5_pol = _optimize_local(h_pert, best_n, t_sub, polish_seed)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = best_n

    # Final feasibility check
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = _project_feasible(best_h, n_final / 2.0)
    final_c5 = _compute_c5(best_h, dx_final)

    return best_h, final_c5, n_final
