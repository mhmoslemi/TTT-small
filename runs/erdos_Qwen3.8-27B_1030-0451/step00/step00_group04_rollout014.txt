import numpy as np
from scipy.optimize import minimize
import time

def corr_values(h):
    """Compute all correlation values z_q * dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c5(h):
    """Compute exact C5 value."""
    z = corr_values(h)
    return float(np.max(z))

def project_box_sum(x, target_sum):
    """Project x onto {h in [0,1]^N : sum(h) = target_sum}."""
    N = len(x)
    # Bisection for lambda
    x = np.asarray(x, dtype=np.float64)
    
    def f(lam):
        return np.sum(np.clip(x - lam, 0.0, 1.0)) - target_sum
    
    # Find bracket
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    # f(lo) should be positive (sum too high), f(hi) should be negative
    # Make sure
    fl = f(lo)
    fh = f(hi)
    if fl < 0:
        lo -= 1.0
        while f(lo) < 0:
            lo -= 1.0
    if fh > 0:
        hi += 1.0
        while f(hi) > 0:
            hi += 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        fm = f(mid)
        if abs(fm) < 1e-15:
            break
        if fm > 0:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    y = np.clip(x - lam, 0.0, 1.0)
    # Fix tiny numerical errors
    residual = target_sum - np.sum(y)
    if abs(residual) > 1e-12:
        # Distribute residual to elements not at bounds
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if np.any(mask):
            y[mask] += residual / np.sum(mask)
            y = np.clip(y, 0.0, 1.0)
    return y

def smooth_max_value_and_weights(z, tau):
    """Compute smooth max value and softmax weights."""
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    # Clip to avoid overflow/underflow
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    value = z_max + tau * np.log(np.sum(exp_vals))
    return value, weights

def compute_gradient(h, weights, dx):
    """Compute gradient of weighted sum of correlation values.
    
    weights: array of shape (2*N-1,) for each lag
    Returns gradient of sum_q weights[q] * z_q
    """
    N = len(h)
    grad = np.zeros(N, dtype=np.float64)
    
    # For lag ell = q - (N-1), where q is the index in the full correlation
    # z_q = dx * sum_n h[n+ell] * (1-h[n])
    
    for q in range(2 * N - 1):
        w = weights[q]
        if w < 1e-12:
            continue
        ell = q - (N - 1)
        
        if ell == 0:
            # d/dh[j] = (1-2*h[j]) * w * dx
            grad += (1.0 - 2.0 * h) * w * dx
        elif ell > 0:
            L = N - ell
            # First term: (1-h[j-ell]) for j in [ell, N-1]
            grad[ell:] += (1.0 - h[:L]) * w * dx
            # Second term: -h[j+ell] for j in [0, L-1]
            grad[:L] -= h[ell:] * w * dx
        else:
            a = -ell
            L = N - a
            # First term: (1-h[j+a]) for j in [0, L-1]
            grad[:L] += (1.0 - h[a:]) * w * dx
            # Second term: -h[j-a] for j in [a, N-1]
            grad[a:] -= h[:L] * w * dx
    
    return grad

def pgd_optimize(h0, n_iter=2000, tau_init=0.02, tau_final=0.001, 
                 alpha_init=0.02, deadline=None):
    """Projected gradient descent with smooth-max surrogate."""
    N = len(h0)
    dx = 2.0 / N
    target_sum = N / 2.0
    
    h = project_box_sum(h0.astype(np.float64).copy(), target_sum)
    best_h = h.copy()
    best_score = exact_c5(h)
    
    tau = tau_init
    alpha = alpha_init
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        z = corr_values(h)
        F, weights = smooth_max_value_and_weights(z, tau)
        grad = compute_gradient(h, weights, dx)
        
        # Normalize gradient
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-15:
            break
        direction = -grad / gnorm
        
        # Line search
        found = False
        for trial in range(8):
            h_new = project_box_sum(h + alpha * direction, target_sum)
            z_new = corr_values(h_new)
            F_new, _ = smooth_max_value_and_weights(z_new, tau)
            if F_new < F:
                h = h_new
                found = True
                break
            alpha *= 0.5
        
        if not found:
            alpha *= 0.1
            # Try one more with smaller alpha
            if alpha > 1e-10:
                h_new = project_box_sum(h + alpha * direction, target_sum)
                z_new = corr_values(h_new)
                F_new, _ = smooth_max_value_and_weights(z_new, tau)
                if F_new < F:
                    h = h_new
                    found = True
        
        if not found:
            # No improvement, shrink alpha and maybe break
            alpha *= 0.5
            if alpha < 1e-10:
                break
            continue
        
        # Check exact score
        score = exact_c5(h)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
        
        # Decay tau
        if it % 50 == 0 and tau > tau_final:
            tau = max(tau * 0.85, tau_final)
        
        # Occasionally increase alpha if things are going well
        if found and it % 10 == 0:
            alpha = min(alpha * 1.05, 0.1)
    
    return best_h, best_score

def slsqp_epigraph(h0, deadline=None, maxiter=200):
    """SLSQP minimization of epigraph formulation."""
    N = len(h0)
    dx = 2.0 / N
    target_sum = N / 2.0
    m = 2 * N - 1  # number of lags
    
    h0 = np.asarray(h0, dtype=np.float64)
    c0 = exact_c5(h0)
    t0 = c0 + 0.01  # start with some margin
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(N + 1)
        g[-1] = 1.0
        return g
    
    def constraints_func(x):
        h = x[:N]
        t = x[-1]
        z = np.correlate(h, 1.0 - h, mode="full") * dx
        return z - t
    
    def constraints_jac(x):
        h = x[:N]
        # Build Jacobian: (m, N+1)
        J = np.zeros((m, N + 1))
        J[:, -1] = -1.0
        # Compute gradient of each z_q wrt h
        for q in range(m):
            ell = q - (N - 1)
            if ell == 0:
                J[q, :N] = (1.0 - 2.0 * h) * dx
            elif ell > 0:
                L = N - ell
                J[q, ell:N] += (1.0 - h[:L]) * dx
                J[q, :L] -= h[ell:] * dx
            else:
                a = -ell
                L = N - a
                J[q, :L] += (1.0 - h[a:]) * dx
                J[q, a:N] -= h[:L] * dx
        return J
    
    def sum_constraint(x):
        return np.sum(x[:N]) - target_sum
    
    def sum_constraint_jac(x):
        J = np.zeros(N + 1)
        J[:N] = 1.0
        return J
    
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * N + [(max(0.0, c0 - 0.05), 1.0)]
    
    cons = [
        {'type': 'ineq', 'fun': constraints_func, 'jac': constraints_jac},
        {'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}
    ]
    
    try:
        result = minimize(
            obj, x0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        h_new = project_box_sum(result.x[:N], target_sum)
        score = exact_c5(h_new)
        return h_new, score
    except Exception:
        return h0, c0

def make_perturbed(h, rng, sigma=0.05):
    """Create a perturbed version of h."""
    h_pert = h + sigma * rng.standard_normal(len(h))
    return project_box_sum(h_pert, len(h) / 2.0)

def upsample_h(h, factor):
    """Upsample h by integer factor using nearest neighbor."""
    h_up = np.repeat(h, factor)
    N_up = len(h_up)
    return project_box_sum(h_up, N_up / 2.0)

def downsample_h(h, factor):
    """Downsample h by taking block means (factor must divide len(h))."""
    N = len(h)
    if N % factor != 0:
        # Trim
        N_trim = N - (N % factor)
        h = h[:N_trim]
    h_down = h.reshape(-1, factor).mean(axis=1)
    target = len(h_down) / 2.0
    return project_box_sum(h_down, target)
