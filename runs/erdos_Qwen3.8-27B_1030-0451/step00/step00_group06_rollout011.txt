import numpy as np

def project(h, target_sum):
    """Project onto {x: 0<=x[i]<=1, sum(x)=target_sum} via bisection on multiplier."""
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def c5_of(h):
    """C5 = max_k dx * correlate(h, 1-h, full)[k]"""
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max_grad(h, dx, I, t):
    """Gradient of (1/t)*log(sum exp(t*I_k)) w.r.t. h. I is the full correlation * dx."""
    N = len(h)
    I_max = I.max()
    exponents = np.clip(t * (I - I_max), -500.0, 500.0)
    w = np.exp(exponents)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(N)
    w /= ws
    g = np.zeros(N)
    Nm1 = N - 1
    for idx in range(2 * N - 1):
        wd = w[idx]
        if wd < 1e-12:
            continue
        d = idx - Nm1
        # Part 1: +wd*dx*(1-h[j-d]) for j in [max(0,d), min(Nm1, Nm1+d)]
        a = max(0, d)
        b = min(Nm1, Nm1 + d)
        if a <= b:

        c = max(0, -d)
        e = min(Nm1, Nm1 - d)

    """Coordinate-flip local search for near-binary improvement."""

            bh, bc = ho, co

    # Phase 3: Local search polish on best
    rem = budget_s - (time.time() - t0)
    if rem > 40.0:
        dl = t0 + rem - 15.0
        ho, co = local_search(bh.copy(), dl, seed + 77777)
        if co < bc:
            bh, bc = ho, co

    # Final verification
    final_c5 = c5_of(bh)
    if final_c5 <= bc + 1e-10:
        bc = final_c5
    else:
        # Something went wrong; return the initial
        bh, bc, bN = h0, c0, N0

    return bh, bc, bN
