        print(f"Phase 2 (factor={f}): score={best_score:.6f}, n={len(best_h)}, time={time.time()-t0:.1f}s")
    
    # Phase 3: Further refinement on the best grid
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 30:
        t_alloc = min(remaining * 0.4, 300.0)
        h_ref, s_ref = optimize_grid(best_h.copy(), max_time=t_alloc, max_iters=30000)
        if s_ref < best_score:
            best_h, best_score = h_ref, s_ref
        print(f"Phase 3 (refine): score={best_score:.6f}, n={len(best_h)}, time={time.time()-t0:.1f}s")
    
    # Phase 4: Random restarts / perturbations
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 30:
        rng = np.random.RandomState(seed)
        n = len(best_h)
        t_total = min(remaining * 0.4, 200.0)
        n_restarts = max(1, int(t_total / 30))
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            # Perturb: randomly swap a few pairs
            h_pert = best_h.copy()
            n_swaps = max(3, n // 10)
            for _ in range(n_swaps):
                i, j = rng.choice(n, 2, replace=False)
                # Swap the "excess" part
                hi = min(h_pert[i], 1.0 - h_pert[j])
                h_pert[i] -= hi
                h_pert[j] += hi
            # Fix sum
            diff = n / 2.0 - np.sum(h_pert)
            if abs(diff) > 1e-10:
                h_pert += diff / n
                h_pert = np.clip(h_pert, 0, 1)
            
            t_rem = budget_s - 30 - (time.time() - t0)
            t_alloc = max(5, min(t_total / n_restarts, t_rem))
            h_r, s_r = optimize_grid(h_pert.copy(), max_time=t_alloc, max_iters=10000)
            if s_r < best_score:
                best_h, best_score = h_r, s_r
        print(f"Phase 4 (restarts): score={best_score:.6f}, n={len(best_h)}, time={time.time()-t0:.1f}s")
    
    # Final: ensure feasibility
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        # Distribute among positions that can accept
        if diff > 0:
            mask = best_h < 1.0 - 1e-10
        else:
            mask = best_h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            best_h[mask] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    
    final_score = eval_score(best_h)
    if final_score < best_score:
        best_score = final_score
    
    elapsed = time.time() - t0
    print(f"Final: score={best_score:.6f}, n={len(best_h)}, time={elapsed:.1f}s")
    
    return best_h, best_score, len(best_h)
