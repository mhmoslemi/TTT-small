```python
    n = len(h)
    dx = 2.0 / n
    C = get_correlations(h)
    
    # Select top-k indices
    top_idx = np.argsort(C)[-top_k:]
    C_active = C[top_idx]
    Cmax = C_active.max()
    exps = np.exp((C_active - Cmax) / tau)
    p_active = exps / exps.sum()
    
    # Build full p vector
    p = np.zeros(len(C))
    p[top_idx] = p_active
    
    F = tau * (Cmax + np.log(exps.sum()))
    
    grad = np.zeros(n)
    c0 = n - 1
    
    if p[c0] > 1e-15:
        grad += p[c0] * dx * (1.0 - 2.0 * h)
    
    for R in range(1, n):
        ip = c0 + R
        im = c0 - R
        if p[ip] > 1e-15:
            grad[R:] += p[ip] * dx * (1.0 - h[:n-R])
            grad[:n-R] -= p[ip] * dx * h[R:]
        if p[im] > 1e-15:
            grad[:n-R] += p[im] * dx * (1.0 - h[R:])
            grad[R:] -= p[im] * dx * h[:n-R]
    
    return F, grad, C, p
```

    xs = np.linspace(lo, hi, n_eval)
    for x in xs:
        h[i] = x
        h[j] = s - x
        v = evaluate(h)
        if v < best_val:
            best_val = v
            best_x = x
    
    # Golden section refinement around best_x
    a = max(lo, best_x - (hi - lo) / n_eval)
    b = min(hi, best_x + (hi - lo) / n_eval)
    if b - a < 1e-12:
        h[i] = best_x
        h[j] = s - best_x
        return h
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] = c; h[j] = s - c; fc = evaluate(h)
    h[i] = d; h[j] = s - d; fd = evaluate(h)
    
    for _ in range(40):
        if b - a < 1e-12:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            h[i] = c; h[j] = s - c; fc = evaluate(h)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            h[i] = d; h[j] = s - d; fd = evaluate(h)
    
    h[i] = (a + b) * 0.5
    h[j] = s - h[i]
    return h
