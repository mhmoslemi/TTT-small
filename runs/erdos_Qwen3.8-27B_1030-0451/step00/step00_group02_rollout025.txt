```python
    best_h = h.copy()
    best_c5 = c5_value(h)
    lr = lr_init
    
    for it in range(n_iters):
        overlaps = np.correlate(h, 1.0 - h, mode="full")
        max_ov = np.max(overlaps)
        
        # Active set: overlaps within 0.01 of max
        active = overlaps > max_ov - 0.01
        w = active.astype(float) / np.sum(active)
        
        # Gradient of active overlaps (average)
        grad = np.zeros(N)
        for k in range(2 * N - 1):
            if not active[k]:
                continue
            s = k - (N - 1)
            wk = w[k]
            j1_start = max(0, -s)
            j1_end = min(N, N - s)
            if j1_end > j1_start:

                grad[idx1] += wk * (1.0 - h[idx1 + s])

            j2_end = min(N, N + s)
            if j2_end > j2_start:
                idx2 = np.arange(j2_start, j2_end)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(v, target_sum):
    """Project v onto the polytope {h: 0<=h_i<=1, sum h_i = target_sum}."""
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def c5_value(h):
    """Compute C5 = max_k correlate(h, 1-h, full)_k * dx."""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(overlaps) * dx)

def compute_overlap_grad(h, weights):
    """Compute weighted sum of d(overlap_k)/d(h_j) for all k."""
    N = len(h)
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = weights[k]
        if abs(wk) < 1e-16:
            continue
        s = k - (N - 1)
        # Term 1: (1-h[j+s]) for valid j+s
        j1_start = max(0, -s)
        j1_end = min(N, N - s)
        if j1_end > j1_start:
            idx1 = np.arange(j1_start, j1_end)
            grad[idx1] += wk * (1.0 - h[idx1 + s])
        # Term 2: -h[j-s] for valid j-s
        j2_start = max(0, s)
        j2_end = min(N, N + s)
        if j2_end > j2_start:
            idx2 = np.arange(j2_start, j2_end)
            grad[idx2] -= wk * h[idx2 - s]
    return grad

def smooth_obj_fn(h, alpha, mu, target_sum):
    """Smoothed max of overlaps + penalty for sum constraint."""
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    max_ov = np.max(overlaps)
    exp_shifted = np.exp(alpha * (overlaps - max_ov))
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = max_ov + np.log(sum_exp) / alpha
    return log_sum_exp + mu * (np.sum(h) - target_sum) ** 2

def smooth_grad_fn(h, alpha, mu, target_sum):
    """Gradient of smoothed max + penalty."""
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    max_ov = np.max(overlaps)
    exp_shifted = np.exp(alpha * (overlaps - max_ov))
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    grad = compute_overlap_grad(h, w)
    grad += 2.0 * mu * (np.sum(h) - target_sum)
    return grad

def optimize_lbfgsb(h, target_sum, time_limit, start_time):
    """Run L-BFGS-B with annealing schedule."""
    N = len(h)
    best_h = h.copy()
    best_c5 = c5_value(h)
    bounds = [(0.0, 1.0)] * N
    
    schedule = [
        (10.0, 10.0),
        (50.0, 10.0),
        (100.0, 50.0),
        (500.0, 50.0),
        (1000.0, 100.0),
        (5000.0, 100.0),
        (10000.0, 100.0),
    ]
    
    for alpha, mu in schedule:
        if time.time() - start_time > time_limit * 0.9:
            break
        try:
            result = minimize(
                smooth_obj_fn, h,
                method='L-BFGS-B',
                jac=smooth_grad_fn,
                args=(alpha, mu, target_sum),
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': 5000}
            )
            h = project_feasible(result.x, target_sum)
        except Exception:
            break
        c5 = c5_value(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def polish_subgrad(h, target_sum, time_limit, start_time, n_iters=5000):
    """Polish with projected subgradient on the exact max."""
    N = len(h)
    best_h = h.copy()
    best_c5 = c5_value(h)
    lr = 0.02
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if it % 500 == 0 and it > 0:
            lr *= 0.7
        
        overlaps = np.correlate(h, 1.0 - h, mode="full")
        max_ov = np.max(overlaps)
        
        # Soft active set: use softmax weights
        alpha_active = 500.0
        weights = np.exp(alpha_active * (overlaps - max_ov))
        weights /= np.sum(weights)
        
        grad = compute_overlap_grad(h, weights)
        h_new = project_feasible(h - lr * grad, target_sum)
        
        c5_new = c5_value(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            lr *= 0.98
        h = h_new
    
    return best_h, best_c5

def interpolate_h(h, N_old, N_new):
    """Interpolate step function to higher resolution."""
    x_old = (np.arange(N_old) + 0.5) * 2.0 / N_old
    x_new = (np.arange(N_new) + 0.5) * 2.0 / N_new
    return np.interp(x_new, x_old, h)

        h_pol, c5 = polish_subgrad(best_h, target_sum, budget_s * 0.55, start_time)
        if c5 < best_c5:
            best_c5 = c5

                    h_pol, c5_p = polish_subgrad(best_h, target_sum, budget_s * 0.9, start_time, n_iters=2000)
                    if c5_p < best_c5:
                        best_c5 = c5_p
                        best_h = h_pol.copy()
    
    # Phase 4: Try higher resolution
    if time.time() - start_time < budget_s * 0.8:
        for N_new in [N_init * 2, N_init * 3]:
            if N_new > 500:
                break
            if time.time() - start_time > budget_s * 0.9:
                break
            h_high = interpolate_h(best_h, len(best_h), N_new)
            ts_new = N_new / 2.0
            h_high = project_feasible(h_high, ts_new)
            h_opt, c5 = optimize_lbfgsb(h_high, ts_new, budget_s, start_time)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                if time.time() - start_time < budget_s * 0.95:

                    if c5_p < best_c5:
                        best_c5 = c5_p
                        best_h = h_pol.copy()
    
    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = c5_value(best_h)
    
    return best_h, best_c5, len(best_h)
```
