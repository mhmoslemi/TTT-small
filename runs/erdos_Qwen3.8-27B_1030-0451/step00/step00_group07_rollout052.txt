def exact_metric(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def mass_project(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} using bisection on shift."""
    h = np.asarray(h, dtype=np.float64).copy()
    # Bisection to find lambda such that sum(clip(h+lambda, 0, 1)) = target
    lo = -max(np.max(h) - 0, target) - 1.0
    hi = max(1.0 - np.min(h), target) + 1.0
    
    # Ensure lo gives sum <= target and hi gives sum >= target
    def f(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if f(mid) < target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = math.fsum(result) - target
    if abs(residual) > 1e-10:
        # Distribute residual
        if residual > 0:
            # Need to decrease
            idx = np.where(result > 1e-9)[0]
        else:
            # Need to increase
            idx = np.where(result < 1.0 - 1e-9)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            adjusted = result[idx].copy()
            adjusted -= per if residual > 0 else 0
            adjusted += per if residual < 0 else 0
            # Clip and re-distribute
            adjusted = np.clip(adjusted, 0.0, 1.0)
            result[idx] = adjusted
    return result

def make_pair_direction(h, i, j):

def find_lambda_range(h, direction):
    """Find valid lambda range for h + lambda * direction."""
    pos = np.where(direction > 0)[0]
    neg = np.where(direction < 0)[0]
    lam_max = float('inf')
    lam_min = float('-inf')
    for idx in pos:
        if direction[idx] > 0:
            cap = (1.0 - h[idx]) / direction[idx]
            lam_max = min(lam_max, cap)
    for idx in neg:
        if direction[idx] < 0:
            cap = h[idx] / (-direction[idx])
            lam_min = max(lam_min, -cap)
    return lam_min, lam_max

def binary_local_search(h, n_eval_budget, rng, time_limit):
    """Local search on binary (0/1) vectors."""
    n = len(h)
    target = n / 2.0
    if target != int(target):
        return h, exact_metric(h), 0
    target_int = int(target)
    
    # Start from threshold of h
    bits = (h >= 0.5).astype(float)
    # Ensure exactly target_int ones
    # Sort by h values, take top target_int
    order = np.argsort(h)[::-1]
    bits = np.zeros(n)
    bits[order[:target_int]] = 1.0
    
    current_obj = exact_metric(bits)
    evals = 0
    start_time = time.time()
    
    while time.time() < time_limit and evals < n_eval_budget:
        # Try flipping: pick a 1 and a 0, swap them
        ones = np.where(bits == 1.0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        j = zeros[rng.integers(0, len(zeros))]
        
        new_bits = bits.copy()
        new_bits[i] = 0.0
        new_bits[j] = 1.0

            bits = new_bits

def block_move_search(h, n_eval_budget, rng, time_limit):
    """Search using block moves (move mass between two blocks)."""
    n = len(h)
    target = n / 2.0
    current_obj = exact_metric(h)
    evals = 0
    
    while time.time() < time_limit and evals < n_eval_budget:
        # Choose two random blocks
        block_size = rng.integers(2, max(3, n // 10))
        start1 = rng.integers(0, n - block_size)
        start2 = rng.integers(0, n - block_size)
        
        # Avoid overlapping blocks
        if start1 < start2 + block_size and start2 < start1 + block_size:
            continue
        
        b1 = slice(start1, start1 + block_size)
        b2 = slice(start2, start2 + block_size)
        
        # Move mass from b1 to b2 (or vice versa)
        for direction in [1, -1]:
            for frac in [0.05, 0.1, 0.25, 0.5]:
                # Compute how much mass we can move
                if direction == 1:
                    # Move from b1 to b2
                    max_move = min(np.min(h[b1]) if len(h[b1]) > 0 else 1.0,
                                  min(1.0 - h[b2].min() if len(h[b2]) > 0 else 1.0, 1.0))
                else:
                    max_move = min(np.min(h[b2]) if len(h[b2]) > 0 else 1.0,
                                  min(1.0 - h[b1].min() if len(h[b1]) > 0 else 1.0, 1.0))
                
                move = frac * max_move
                if move < 1e-8:
                    continue

                    new_h[b1] -= move
                    new_h[b2] += move
                else:
                    new_h[b2] -= move
                    new_h[b1] += move
                
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = mass_project(new_h, target)
                new_obj = exact_metric(new_h)
                evals += 1
                if new_obj < current_obj - 1e-14:
                    h = new_h
                    current_obj = new_obj
                    break
    
    return h, current_obj, evals

def downsample(h, new_n):
    """Downsample h from its current length to new_n, preserving integral."""
    old_n = len(h)
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    target_new = new_n / 2.0
    
    # Piecewise linear interpolation
    x_old = np.linspace(0, 2, old_n + 1, endpoint=True)
    x_old = x_old[:-1] + dx_old / 2  # cell centers
    x_new = np.linspace(0, 2, new_n + 1, endpoint=True)
    x_new = x_new[:-1] + dx_new / 2  # cell centers
    
    new_h = np.interp(x_new, x_old, h)
    new_h = mass_project(new_h, target_new)
    return new_h

def upsample(h, new_n):
    """Upsample h to new_n points."""
    old_n = len(h)

import numpy as np
import time
import math

def exact_metric(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def mass_project(h, target):
    h = np.asarray(h, dtype=np.float64).copy()
    lo = -max(float(np.max(h)), target) - 1.5
    hi = max(1.0 - float(np.min(h)), target) + 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    residual = math.fsum(result) - target
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(result > 1e-12)[0]
        else:
            idx = np.where(result < 1.0 - 1e-12)[0]
        if len(idx) > 0:
            per = residual / len(idx)
            result[idx] -= per
            result[idx] = np.clip(result[idx], 0.0, 1.0)
    return result

def default_start(n):

def downsample(h, new_n):

    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(x_new, x_old, h)

def binary_descent(h, max_evals, rng, t_end):
