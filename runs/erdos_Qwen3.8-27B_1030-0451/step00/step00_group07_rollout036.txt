        h[i] = 0.0
        h[j] = 1.0

            h[j] = 0.0

def exact_metric(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def mass_project(y, target):
    y = np.asarray(y, dtype=np.float64).copy()
    n = len(y)
    if n == 0:
        return y
    lo = -1.0
    hi = 1.0
    # Ensure bracket
    while np.sum(np.clip(y + lo, 0, 1)) > target:
        lo -= 1.0
    while np.sum(np.clip(y + hi, 0, 1)) < target:
        hi += 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual with small adjustments
    for _ in range(5):
        resid = np.sum(z) - target
        if abs(resid) < 1e-10:
            break
        if resid > 0:
            idx = np.where(z > 1e-8)[0]
            if len(idx) == 0:
                break
            step = min(resid / len(idx), np.min(z[idx]) - 1e-12)
            if step > 0:
                z[idx] -= step
        else:
            idx = np.where(z < 1.0 - 1e-8)[0]
            if len(idx) == 0:
                break
            step = min(-resid / len(idx), np.min(1.0 - z[idx]) - 1e-12)
            if step > 0:
                z[idx] += step
    return np.clip(z, 0.0, 1.0)

    x_from = (np.arange(n_from) + 0.5) / n_from * 2.0
    x_to = (np.arange(n_to) + 0.5) / n_to * 2.0
    h_to = np.interp(x_to, x_from, h_from)

f(h) = max_k c_k(h) * dx

where c_k(h) = sum_n h[n] * (1-h[n-k]) for valid (n,k) pairs.

This is the maximum of bilinear functions, so it's not convex. However, at any point, if we know which k is active (gives the max), we can try to reduce c_k while not increasing others too much.

A subgradient descent approach:
- Compute all c_k
- Find the active k* (argmax)
- Compute gradient of c_{k*} with respect to h
- Take a small step in the negative gradient direction
- Project back to feasible set

The gradient of c_k with respect to h[i]:
c_k = sum_n h[n] * (1-h[n-k])

∂c_k/∂h[i] appears in two places:
1. When n = i: h[i] appears as the first factor → ∂/∂h[i] of h[i]*(1-h[i-k]) = (1-h[i-k])
2. When n-k = i, i.e., n = i+k: h[i] appears in (1-h[n-k]) = (1-h[i]) → ∂/∂h[i] of h[i+k]*(1-h[i]) = -h[i+k]

So ∂c_k/∂h[i] = (1-h[i-k]) - h[i+k] (where terms are 0 if indices out of range)

For the full objective max_k c_k, a subgradient is the gradient of the active c_{k*}:
∂f/∂h[i] = (1-h[i-k*]) - h[i+k*] (times dx)

But this only accounts for one active constraint. In practice, multiple k's might be near the max.

A better approach: use the "top" few active k's and average their gradients.

    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def mass_project(y, target):
    y = np.asarray(y, dtype=np.float64).copy()
    n = len(y)
    if n == 0:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def exact_metric(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def mass_project(y, target):
    y = np.asarray(y, dtype=np.float64).copy()
    n = len(y)
    if n == 0:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        r = float(np.sum(z)) - target
        if abs(r) < 1e-13:
            break
        if r > 0:
            m = np.where(z > 1e-12)[0]
            if len(m) == 0:
                break
            d = min(r / len(m), float(np.min(z[m])) - 1e-14)
            if d > 0:
                z[m] -= d
        else:
            m = np.where(z < 1.0 - 1e-12)[0]

                break
            d = min(-r / len(m), float(np.min(1.0 - z[m])) - 1e-14)
            if d > 0:
                z[m] += d
    return np.clip(z, 0.0, 1.0)
