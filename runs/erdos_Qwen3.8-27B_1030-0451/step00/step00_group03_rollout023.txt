def exact_c5(h):
    """Compute exact C5 value for a feasible h vector."""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def build_h_from_p(p, n):
    """Build full h from half-vector p using mirror-complement symmetry."""
    n = int(n)
    h = np.zeros(n, dtype=np.float64)
    m = n // 2
    p = np.clip(p, 0.0, 1.0)
    h[:m] = p
    if n % 2 == 1:
        h[m] = 0.5
    h[m + (n%2):] = 1.0 - p[::-1]
    return h

def p_from_h(h, n):

    n = int(n)

import numpy as np
import scipy.optimize

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def build_h(p, n):
    m = n // 2
    p = np.clip(p, 0.0, 1.0)
    h = np.empty(n, dtype=np.float64)
    h[:m] = p
    off = n % 2
    if off:
        h[m] = 0.5
    h[m + off:] = 1.0 - p[::-1]
    return h

def extract_p(h):
    n = len(h)
    m = n // 2
    p = 0.5 * (h[:m] + 1.0 - h[n-1: n-1-m: -1])
    return np.clip(p, 0.0, 1.0)

def interp_p(p_old, m_new):
    m_old = len(p_old)
    if m_old == m_new:
        return p_old.astype(np.float64).copy()
    if m_old == 0:
        return np.full(m_new, 0.5, dtype=np.float64)
    x_old = (np.arange(m_old) + 0.5) / m_old
    x_new = (np.arange(m_new) + 0.5) / m_new
    return np.clip(np.interp(x_new, x_old, p_old), 0.0, 1.0)

def lse_max(f, tau):
    fmax = float(np.max(f))
    if tau < 1e-14:
        return fmax
    s = (f - fmax) / tau
    s = np.clip(s, -500.0, 500.0)
    return fmax + tau * float(np.log(np.sum(np.exp(s))))

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lse_obj(p, n, tau, state):
    h = build_h(p, n)
    f = compute_f(h)
    c = float(np.max(f))
    if c < state['best_c']:
        state['best_c'] = c
        state['best_h'] = h.copy()
        state['best_n'] = n
    return lse_max(f, tau)

def project_feasible(y):
    n = len(y)
    target = n / 2.0
    lo = -float(np.max(y))
    hi = 1.0 - float(np.min(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def gen_seeds(n, seed):

    rng = np.random.default_rng(seed)
    seeds = [np.full(m, 0.5, dtype=np.float64)]

        for amp in [0.25, 0.4, 0.5]:

def run_lse(p_init, n, state, deadline):
    m = n // 2

    h = build_h(p, n)

    for iteration in range(30):

        f = compute_f(h)

            state['best_c'] = c

            state['best_n'] = n

        # Get gradient direction for this lag via finite difference
        # correlate(h, 1-h)[worst] = sum_i h[i+k]*(1-h[i]) where k = worst - (n-1)
        k = worst - (n - 1)
        # For lag k: sum over valid i of h[i+k] * (1-h[i])
        # Derivative w.r.t. p[j] (for j < m): d/dp_j of sum h[i+k](1-h[i])
        # h[i] depends on p if i < m, and h[i] = 1-p[n-1-i] if i >= m
        # d/dp_j h[i] = 1 if i==j, = -1 if i==n-1-j, else 0
        # d/dp_j (1-h[i]) = -1 if i==j, = 1 if i==n-1-j, else 0
        grad = np.zeros(m)
        # Determine valid range of i
        if k >= 0:
            i_lo = 0
            i_hi = n - 1 - k

            i_lo = -k
            i_hi = n - 1

            # term = h[i+k] * (1-h[i])
            # d/dp_j: from h[i+k] part: (1-h[i]) * dh[i+k]/dp_j

            hk = i + k
            if 0 <= hk < n and 0 <= i < n:

                    grad[j] -= h[hk]
                elif n - 1 - j == i:
                    grad[j] += h[hk]
        # Step in direction that reduces the worst lag
        step = 0.01
        p_try = np.clip(p_cur - step * grad, 0.0, 1.0)
        h_try = build_h(p_try, n)
        c_try = exact_c5(h_try)

            c_try = exact_c5(h_try)

    return p_cur
