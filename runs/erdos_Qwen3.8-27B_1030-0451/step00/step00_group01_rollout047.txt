def project_sum_box(x, target):
    x = np.asarray(x, dtype=np.float64)
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    result = np.clip(x + lam, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        count = int(np.sum(mask))
        if count > 0:
            result[mask] += residual / count
            result = np.clip(result, 0.0, 1.0)

def exact_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def lse_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    shifted = np.clip(beta * (c - m), -500, 500)
    es = np.exp(shifted)
    Z = es.sum()
    obj = m + np.log(Z) / beta
    w = es / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]

        grad[lo + lag:hi + lag] += dx * wk * (1.0 - h[lo:hi])
        grad[lo:hi] -= dx * wk * h[lo + lag:hi + lag]
    return obj, grad
