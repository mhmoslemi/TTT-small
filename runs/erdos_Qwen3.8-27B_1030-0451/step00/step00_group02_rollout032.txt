def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def smooth_objective(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    fmax = np.max(corr)
    T_eff = min(T, 700.0 / max(abs(fmax), 1e-10))
    return fmax + np.log(np.sum(np.exp(T_eff * (corr - fmax)))) / T_eff

def smooth_gradient(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    fmax = np.max(corr)
    T_eff = min(T, 700.0 / max(abs(fmax), 1e-10))
    exp_v = np.exp(T_eff * (corr - fmax))
    w = exp_v / np.sum(exp_v)

    A = np.correlate(w, 1.0 - h, mode="full")

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_gradient(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    fmax = float(np.max(corr))
    T_eff = min(T, 700.0 / max(abs(fmax), 1e-10))
    exp_v = np.exp(T_eff * (corr - fmax))
    w = exp_v / np.sum(exp_v)
    h_flip = h[::-1]
    A = np.correlate(w, 1.0 - h, mode="full")
    A_m = A[0:n]
    B = np.correlate(w, h_flip, mode="full")
    B_m = B[0:n][::-1]
    return dx * (A_m - B_m)


def exact_subgradient(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    a = k - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, -a)
    hi1 = min(n - 1, n - 1 - a)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += (1.0 - h[lo1 + a:hi1 + 1 + a])
    lo2 = max(0, a)
    hi2 = min(n - 1, n - 1 + a)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 - a:hi2 + 1 - a]
    return g * dx


def project_feasible(h, n):

            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
