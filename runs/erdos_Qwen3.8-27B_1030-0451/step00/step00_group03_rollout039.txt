```python
def full_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(y):
    """Project y onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    n = len(y)
    target = n / 2.0
    # Find alpha such that sum(clip(y + alpha, 0, 1)) = target
    lo = -np.max(y)
    hi = 1.0 - np.min(y)
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) / 2, 0, 1)
    # Fix residual
    residual = target - np.sum(x)
    if residual > 0:
        # Need to add residual mass
        idx = np.argsort(1 - x)[::-1]  # most room first
        for i in idx:
            add = min(residual, 1 - x[i])
            x[i] += add
            residual -= add
            if residual <= 0:
                break
    elif residual < 0:
        idx = np.argsort(x)[::-1]  # most mass first
        for i in idx:
            rem = min(-residual, x[i])
            x[i] -= rem
            residual += rem
            if residual >= 0:
                break
    return x

def coordinate_descent(h, max_sweeps=100, tol=1e-8):
    """Perform coordinate descent to minimize C5."""
    n = len(h)
    h = h.copy()
    best_c = full_score(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Find best value for h[i]
            def obj(t):
                h[i] = t
                c = full_score(h)
                return c
            # Use grid search + refinement
            # Coarse grid
            ts = np.linspace(0, 1, 51)
            vals = np.array([obj(t) for t in ts])
            best_idx = np.argmin(vals)
            # Refine around best
            if best_idx > 0 and best_idx < len(ts)-1:
                lo = ts[best_idx-1]
                hi = ts[best_idx+1]
                # Golden section
                for _ in range(20):
                    m1 = hi - 0.618*(hi-lo)
                    m2 = lo + 0.618*(hi-lo)
                    if obj(m1) < obj(m2):
                        hi = m2
                    else:
                        lo = m1
                best_t = (lo+hi)/2
            else:
                best_t = ts[best_idx]
            
            h[i] = best_t
            new_c = full_score(h)
            if new_c < best_c - 1e-12:
                best_c = new_c
                improved = True
        
        if not improved or best_c < tol:
            break
    
    return h, best_c
```

```python
def update_corr(corr, h, i, new_val):
    """Update correlation array when h[i] changes to new_val. Returns new corr (modified in place)."""
    n = len(h)
    old = h[i]
    delta = new_val - old
    if abs(delta) < 1e-15:
        return corr
    
    for s in range(-(n-1), n):
        # Shift s corresponds to index n-1+s in corr array
        idx = n - 1 + s
        d = 0.0
        # Term where j+s = i, i.e., j = i-s
        if 0 <= i - s < n:
            if s == 0:
                # Special case: only term is h[i]*(1-h[i])
                d = new_val*(1-new_val) - old*(1-old)
            else:
                d = delta * (1 - h[i-s])
        # Term where j = i
        if s != 0 and 0 <= i + s < n:
            d += -delta * h[i+s]
        corr[idx] += d
    
    h[i] = new_val
    return corr
```
