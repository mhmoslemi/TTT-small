import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute the exact C5 value for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def project_feasible(y, target_sum):
    """Project y onto {h: 0 <= h <= 1, sum(h) = target_sum} using bisection."""
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(y)
    # Bisection for alpha
    lo = -y.min() - 1.0
    hi = 1.0 - y.max() + 1.0
    # Ensure bracket
    for _ in range(10):
        alpha = (lo + hi) / 2.0
        s = np.sum(np.clip(y + alpha, 0.0, 1.0))
        if s > target_sum:
            hi = alpha
        else:
            lo = alpha
        if hi - lo < 1e-15:
            break
    alpha = (lo + hi) / 2.0
    h = np.clip(y + alpha, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            # Add to positions with most room
            rooms = 1.0 - h
            order = np.argsort(-rooms)
            for idx in order:
                step = min(rooms[idx], residual)
                h[idx] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            residual = -residual
            order = np.argsort(h)
            for idx in order:
                step = min(h[idx], residual)
                h[idx] -= step
                residual -= step
                if abs(residual) < 1e-14:
                    break
    return h

def compute_gradient_f_m(h, k):
    """Compute gradient of f_m (for lag k) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        # f = dx * sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])
        j_max = n - 1 - k
        # Role 1: p = j+k, valid p in [k, n-1]
        #   contribution: dx * (1 - h[p-k])
        j_range = np.arange(0, n - k)
        grad[j_range + k] += (1.0 - h[j_range]) * dx
        # Role 2: p = j, valid p in [0, n-1-k]
        #   contribution: dx * (-h[p+k])
        grad[j_range] += (-h[j_range + k]) * dx
    else:
        # k < 0, let r = -k
        r = -k
        # f = dx * sum_{j=r}^{n-1} h[j] * (1-h[j-r])
        # Role 1: p = j, valid p in [r, n-1]
        #   contribution: dx * (1 - h[p-r])
        j_range = np.arange(r, n)
        grad[j_range] += (1.0 - h[j_range - r]) * dx
        # Role 2: p = j-r, valid p in [0, n-1-r]
        #   contribution: dx * (-h[p+r])
        grad[j_range - r] += (-h[j_range]) * dx
    return grad

def compute_weighted_gradient(h, tau, active_indices=None):
    """Compute weighted gradient using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    fmax = np.max(corr)
    
    if active_indices is not None:
        selected = corr[active_indices]
        smax = np.max(selected)
        z = (selected - smax) / tau
        z -= np.max(z)
        w = np.exp(z) / np.sum(np.exp(z))
    else:
        z = (corr - fmax) / tau
        z -= np.max(z)
        w = np.exp(z) / np.sum(np.exp(z))
    
    if active_indices is not None:
        grad = np.zeros(n)
        for m_idx in range(len(active_indices)):
            m = active_indices[m_idx]
            k = m - (n - 1)
            grad += w[m_idx] * compute_gradient_f_m(h, k)
    else:
        grad = np.zeros(n)
        for m in range(len(corr)):
            if w[m] > 1e-6:
                k = m - (n - 1)
                grad += w[m] * compute_gradient_f_m(h, k)
    return grad, fmax

def projected_gradient_descent(h0, max_iter=500, tau0=0.003, eta0=0.01, 
                                active_frac=0.3, time_limit=100):
    """Projected gradient descent with log-sum-exp smoothing."""
    n = len(h0)
    target_sum = 0.5 * n
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    start_time = time.time()
    
    tau = tau0
    eta = eta0
    stall_count = 0
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        n_lags = 2 * n - 1
        n_active = max(10, int(n_lags * active_frac))
        
        # Get active lags (top n_active by value)
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
        top_idx = np.argpartition(corr, -n_active)[-n_active:]
        
        grad, fmax = compute_weighted_gradient(h, tau, active_indices=top_idx)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Line search
        accepted = False
        for trial in range(15):
            y = h - eta * grad / gnorm
            h_new = project_feasible(y, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < fmax + 1e-10:
                h = h_new
                accepted = True
                break
            eta *= 0.5
        
        if not accepted:
            stall_count += 1
        else:
            stall_count = 0
            
        c5_current = compute_c5(h)
        if c5_current < best_c5 - 1e-12:
            best_c5 = c5_current
            best_h = h.copy()
        
        if stall_count > 30:
            tau *= 1.5
            eta = eta0
            stall_count = 0
            if tau > 0.01:
                tau = 0.003
                break
        
        if it % 20 == 0 and it > 0:
            tau = max(0.001, tau * 0.9)
    
    return best_h, best_c5

def coordinate_descent(h0, max_iter=200, time_limit=30):
    """Coordinate descent: try small pairwise transfers."""
    n = len(h0)
    target_sum = 0.5 * n
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    start_time = time.time()
    rng = np.random.RandomState(42)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Try random pair transfer
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        delta = rng.choice([-0.05, -0.02, -0.01, 0.01, 0.02, 0.05])
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        if 0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0:
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
    
    return best_h, best_c5

def upsample(h, factor):
    """Upsample h by integer factor using nearest neighbor."""
    return np.repeat(h, factor).astype(float)

def interpolate_to(h, new_n):
    """Interpolate h to new_n points, then project to feasible set."""
    n = len(h)
    target_sum = 0.5 * new_n
    oldx = np.linspace(0, 2, n, endpoint=False)
    newx = np.linspace(0, 2, new_n, endpoint=False)
    h_interp = np.interp(newx, oldx, h)
    return project_feasible(h_interp, target_sum)

    alpha = 0.5 * (lo + hi)
    h = np.clip(y + alpha, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            rooms = 1.0 - h
            order = np.argsort(-rooms)
            for idx in order:
                step = min(rooms[idx], residual)
                if step > 0:
                    h[idx] += step
                    residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            r = -residual
            order = np.argsort(h)
            for idx in order:
                step = min(h[idx], r)
                if step > 0:
                    h[idx] -= step
                    r -= step
                if abs(r) < 1e-14:
                    break
    return h

def grad_f_m(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k]) * dx
        grad[:n-k] += (-h[k:]) * dx
    else:
        r = -k
        grad[r:] += (1.0 - h[:n-r]) * dx
        grad[:n-r] += (-h[r:]) * dx
    return grad

def weighted_grad(h, tau, top_frac=0.4):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
    n_lags = len(corr)
    n_act = max(8, int(n_lags * top_frac))
    top_idx = np.argpartition(corr, -n_act)[-n_act:]
    fmax = np.max(corr)
    z = (corr[top_idx] - fmax) / tau
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()
    grad = np.zeros(n)
    for i, m in enumerate(top_idx):
        if w[i] < 1e-8:
            continue
        k = m - (n - 1)
        grad += w[i] * grad_f_m(h, k)
    return grad, fmax, top_idx

def pgd(h0, max_iter=500, tau0=0.002, eta0=0.005, time_limit=60):
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t0 = time.time()
    tau = tau0
    eta = eta0
    stall = 0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad, fmax, _ = weighted_grad(h, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        accepted = False
        for _ in range(12):
            y = h - eta * grad / gnorm
            h_new = project_feasible(y, target)
            c5_new = compute_c5(h_new)
            if c5_new <= fmax + 1e-10:
                h = h_new
                accepted = True
                break
            eta *= 0.5
        if accepted:
            stall = 0
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            stall += 1
        if stall > 25:
            tau *= 1.4
            eta = eta0
            stall = 0
            if tau > 0.008:
                break
        if it > 0 and it % 30 == 0:
            tau = max(0.0005, tau * 0.85)
    
    return best_h, best_c5

def slsqp(h0, n_act=25, time_limit=40):
    n = len(h0)
    target = 0.5 * n
    dx = 2.0 / n
    t0 = time.time()
    
    corr = np.correlate(h0, 1.0 - h0, mode="full") * dx
    idx = np.argsort(corr)[::-1][:n_act]
    idx = np.sort(idx)
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq(x):
        h = x[:n]
        t = x[-1]
        out = np.empty(len(idx))
        for i, m in enumerate(idx):
            k = m - (n - 1)
            if k >= 0:
                val = dx * np.dot(h[k:k+n-k], 1.0 - h[:n-k])
            else:
                r = -k
                val = dx * np.dot(h[r:r+n-r], 1.0 - h[:n-r])
            out[i] = t - val
        return out
    
    def ineq_jac(x):
        h = x[:n]
        J = np.zeros((len(idx), n + 1))
        for i, m in enumerate(idx):
            k = m - (n - 1)
            J[i, :n] = -grad_f_m(h, k)
            J[i, n] = 1.0
        return J
    
    def eq(x):
        return np.sum(x[:n]) - target
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    c5_0 = compute_c5(h0)
    x0 = np.append(h0.copy(), c5_0 + 0.003)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 150, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x[:n], target)
            c5_new = compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h0, c5_0

def coord_search(h0, n_trials=2000, time_limit=40):
    n = len(h0)
    target = 0.5 * n
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t0 = time.time()
    rng = np.random.RandomState(123)
    
    for _ in range(n_trials):
        if time.time() - t0 > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-0.02, -0.01, -0.005, 0.005, 0.01, 0.02])
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        if 0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0:
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
    
    return best_h, best_c5

def interp_to(h, new_n):
    n = len(h)
    oldx = np.linspace(0, 2, n, endpoint=False)
    newx = np.linspace(0, 2, new_n, endpoint=False)
    h_i = np.interp(newx, oldx, h)
    return project_feasible(h_i, 0.5 * new_n)
