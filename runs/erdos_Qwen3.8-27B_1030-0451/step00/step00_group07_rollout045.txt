def _smooth_objective(h, N, tau):
    h = np.clip(h, 0, 1)
    C = np.correlate(h, 1-h, 'full') * (2.0/N)
    m = np.max(C)

def _evaluate(h):
    """Evaluate C5 for a given h array."""
    N = len(h)
    dx = 2.0 / N
    C = np.correlate(h, 1-h, 'full') * dx
    return np.max(C)

def _project_feasible(h, N):
    """Project h to be feasible: 0<=h<=1, sum(h)=N/2."""
    h = np.clip(h, 0, 1)
    target = N / 2.0
    # Use bisection on a shift parameter
    # F(t) = sum(clip(h + t, 0, 1)) is monotone in t
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fine-tune: adjust remaining mass
    deficit = target - np.sum(h_new)
    if abs(deficit) > 1e-12:
        # Distribute deficit among non-boundary elements
        mask = (h_new > 1e-10) & (h_new < 1 - 1e-10)
        if np.any(mask):
            h_new[mask] += deficit / np.sum(mask)
            h_new = np.clip(h_new, 0, 1)
            # Final correction
            residual = target - np.sum(h_new)
            if abs(residual) > 1e-15:
                # Add to a specific element
                idx = np.argmax(mask) if np.any(mask) else 0
                h_new[idx] += residual
                h_new = np.clip(h_new, 0, 1)
    return h_new
