import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def exact_z_vec(h):
    """Return the full correlation vector (unscaled)."""
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, T):
    """Project h to [0,1]^n with sum = T using bisection on water level."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Find alpha such that sum(clip(h + alpha, 0, 1)) = T
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < T:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = T - result.sum()
    # Distribute residual
    pos_mask = result < 1.0
    neg_mask = result > 0.0
    if diff > 0:
        mask = pos_mask
        avail = (1.0 - result[mask]).sum()
        if avail > 0:
            result[mask] += diff * result[mask].sum() / max(result[mask].sum(), 1e-15) if False else diff / mask.sum()
            # Actually just distribute evenly
            result[mask] = np.clip(result[mask] + diff / mask.sum(), 0, 1)
    elif diff < 0:
        mask = neg_mask
        result[mask] = np.clip(result[mask] + diff / mask.sum(), 0, 1)
    return result

def grad_z(h, q_offset):
    """
    Gradient of z_q with respect to h, where q_offset is the index in the full correlate output.
    q_offset = 0..2n-2, corresponding to lag k = q_offset - (n-1).
    Returns gradient of the raw (unscaled) correlation sum.
    """
    n = len(h)
    k = q_offset - (n - 1)  # actual lag
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        # z = sum_{i=0}^{L-1} h[i+k] * (1-h[i])
        # d/dh[j]: 
        #   if j >= k and j-k < L (i.e., j-k in [0,L-1]): contributes (1-h[j-k])
        #   if j < L (i.e., j in [0,L-1]): contributes -h[j+k]
        g[k:] += (1.0 - h[:L])  # for j in [k, n-1], j-k in [0, L-1]
        g[:L] -= h[k:]  # for j in [0, L-1], j+k in [k, n-1]
    else:
        m = -k
        L = n - m
        # z = sum_{i=m}^{n-1} h[i-m] * (1-h[i])
        # d/dh[j]:
        #   if j < L (j in [0, L-1]): contributes (1-h[j+m])  [j appears as i-m, so i=j+m]
        #   if j >= m (j in [m, n-1]): contributes -h[j-m]
        g[:L] += (1.0 - h[m:])  # for j in [0, L-1], j+m in [m, n-1], 1-h[j+m] = 1-h[m:][j]
        g[m:] -= h[:L]  # for j in [m, n-1], j-m in [0, L-1], -h[j-m] = -h[:L][j-m]
    return g

def slp_step(h, active_qs, rho, n):
    """
    One SLP iteration: solve LP to find next point.
    Returns (y_candidate, t_opt) or (None, None) on failure.
    """
    z_vec = exact_z_vec(h)
    
    # Build LP
    # Variables: y[0..n-1], t (index n)
    # Minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: for each active q
    # grad_q(h) · y - t <= grad_q(h) · h - z_q(h)
    A_rows = []
    b_vals = []
    for q in active_qs:
        g_q = grad_z(h, q)
        z_q = z_vec[q]
        A_row = np.zeros(n + 1)
        A_row[:n] = g_q
        A_row[n] = -1.0
        b_val = np.dot(g_q, h) - z_q
        A_rows.append(A_row)
        b_vals.append(b_val)
    
    if A_rows:
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
    else:
        A_ub = np.empty((0, n + 1))
        b_ub = np.empty(0)
    
    # Equality: sum(y) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - rho)
        hi = min(1.0, h[i] + rho)
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0, unbounded above
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        y = res.x[:n]
        t = res.x[n]
        return y, t
    return None, None

def slp_local_search(h, time_limit, max_iter=200, rho_init=0.02, target=0.3808):
    """
    SLP-based local search. Returns best h found.
    """
    n = len(h)
    T = n / 2.0
    h = project_box_sum(h.copy(), T)
    
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    
    rho = rho_init
    stall = 0
    
    t_start = time.time()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        if current_score <= target:
            break
        
        # Select active constraints
        z_vec = exact_z_vec(h)
        z_max = np.max(z_vec)
        gap = max(1e-4, 0.01 * z_max, 0.1 * (z_max / (2.0/n) - target) * (2.0/n))
        
        active_mask = z_vec >= (z_max - gap)
        active_qs = np.where(active_mask)[0]
        
        # Cap active count
        K_max = 80
        if len(active_qs) > K_max:
            # Take top K_max
            order = np.argsort(-z_vec[active_qs])
            active_qs = active_qs[order[:K_max]]
        
        # Ensure argmax is included
        argmax_q = np.argmax(z_vec)
        if argmax_q not in active_qs:
            active_qs = np.append(active_qs, argmax_q)
        
        # SLP step
        y, t_opt = slp_step(h, active_qs, rho, n)
        
        if y is not None:
            # Line search
            accepted = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                candidate = (1 - alpha) * h + alpha * y
                candidate = project_box_sum(candidate, T)
                cand_score = exact_score(candidate)
                if cand_score < current_score - 1e-12:
                    h = candidate
                    current_score = cand_score
                    if current_score < best_score:
                        best_h = h.copy()
                        best_score = current_score
                    accepted = True
                    break
            
            if accepted:
                stall = 0
                rho = max(0.005, rho * 0.9)
            else:
                stall += 1
                rho *= 0.5
                if rho < 0.001:
                    rho = rho_init * 0.3  # reset
                    stall = 0
        else:
            stall += 1
            rho *= 0.5
            if rho < 0.001:
                break
        
        if stall > 30:
            break
    
    return best_h, best_score

def pair_transfer_polish(h, time_limit, max_rounds=50):
    """
    Pairwise transfer local search. Returns best h found.
    """
    n = len(h)
    T = n / 2.0
    h = project_box_sum(h.copy(), T)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    
    t_start = time.time()
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        if no_improve > 10:
            break
        
        # Compute weighted gradient over active lags
        z_vec = exact_z_vec(h)
        z_max = np.max(z_vec)
        gap = max(1e-4, 0.01 * z_max)
        active_mask = z_vec >= (z_max - gap)
        active_qs = np.where(active_mask)[0]
        
        if len(active_qs) > 50:
            order = np.argsort(-z_vec[active_qs])
            active_qs = active_qs[order[:50]]
        
        # Compute subgradient (average of active gradients)
        g = np.zeros(n)
        for q in active_qs:
            g += grad_z(h, q)
        g /= len(active_qs)
        
        # Find promising transfers: reduce h[i], increase h[j] where g[i] > g[j]
        # Sort by gradient
        sorted_idx = np.argsort(-g)
        top_candidates = sorted_idx[:min(80, n)]
        bottom_candidates = sorted_idx[-min(80, n):]
        
        improved = False
        for i in top_candidates:
            if h[i] < 0.01:
                continue
            for j in bottom_candidates:
                if i == j or h[j] > 0.99:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 0.005:
                    continue
                
                best_delta = 0
                best_cand_score = current_score
                for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                    delta = delta_max * frac
                    if delta < 1e-6:
                        break
                    h[i] -= delta
                    h[j] += delta
                    s = exact_score(h)
                    if s < best_cand_score - 1e-12:
                        best_cand_score = s
                        best_delta = delta
                    # Restore for next try
                    h[i] += delta
                    h[j] -= delta
                
                if best_delta > 0:
                    h[i] -= best_delta
                    h[j] += best_delta
                    current_score = best_cand_score
                    if current_score < best_score:
                        best_h = h.copy()
                        best_score = current_score
                    improved = True
                    break
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_score

import numpy as np
from scipy.optimize import linprog
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def exact_z_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_box_sum(h, T):
    n = len(h)
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < T:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            result[mask] = np.clip(result[mask] + diff / cnt, 0.0, 1.0)
    return result


def grad_z(h, q_offset):
    n = len(h)
    k = q_offset - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        g[k:] += (1.0 - h[:L])
        g[:L] -= h[k:]
    else:
        m = -k
        L = n - m
        g[:L] += (1.0 - h[m:])
        g[m:] -= h[:L]
    return g


def slp_step(h, active_qs, rho):
    n = len(h)
    z_vec = exact_z_vec(h)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_rows = []
    b_vals = []
    for q in active_qs:
        g_q = grad_z(h, q)
        z_q = z_vec[q]
        A_row = np.zeros(n + 1)
        A_row[:n] = g_q
        A_row[n] = -1.0
        b_val = np.dot(g_q, h) - z_q
        A_rows.append(A_row)
        b_vals.append(b_val)
    if A_rows:
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
    else:
        A_ub = np.empty((0, n + 1))
        b_ub = np.empty(0)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo_b = max(0.0, h[i] - rho)
        hi_b = min(1.0, h[i] + rho)
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, np.inf))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    if res.success:
        return res.x[:n], res.x[n]
    return None, None


def slp_local_search(h, time_limit, max_iter, rho_init, target):
    n = len(h)
    T = n / 2.0
    h = project_box_sum(h.copy(), T)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    rho = rho_init
    stall = 0
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if current_score <= target:
            break

        z_max = np.max(z_vec)

        g = np.zeros(n)

        g /= max(1, len(active_qs))
        sorted_idx = np.argsort(-g)

        bot = sorted_idx[-min(25, n // 4):]
        improved = False

            if h[i] < 0.01:

                    continue
                delta_max = min(h[i], 1.0 - h[j])

                best_delta = 0.0

                    h[j] += delta
                    s = exact_score(h)
                    h[i] += delta
                    h[j] -= delta
                    if s < best_s - 1e-12:
                        best_s = s
                        best_delta = delta

                    h[i] -= best_delta
                    h[j] += best_delta

                        best_score = current_score
                    improved = True
                    break

                break

def subgradient_descent(h, time_limit, max_iter, target):
    n = len(h)
    T = n / 2.0
    h = project_box_sum(h.copy(), T)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    eta = 0.002 / (2.0 / n)
    t0 = time.time()
    stall = 0
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if current_score <= target:
            break
        z_vec = exact_z_vec(h)
        z_max = np.max(z_vec)
        gap = max(1e-5, 0.005 * z_max)
        active_qs = np.where(z_vec >= (z_max - gap))[0]
        if len(active_qs) == 0:
            active_qs = np.array([int(np.argmax(z_vec))])
        g = np.zeros(n)
        for q in active_qs:
            g += grad_z(h, q)
        g /= len(active_qs)
        g_scaled = g * (2.0 / n)
        candidate = project_box_sum(h - eta * g_scaled, T)
        s = exact_score(candidate)
        if s < current_score - 1e-13:
            h = candidate
            current_score = s
            if current_score < best_score:
                best_h = h.copy()
                best_score = current_score
            stall = 0
