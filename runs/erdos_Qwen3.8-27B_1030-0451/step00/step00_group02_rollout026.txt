def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def all_f_values(h):

def gradient_fk(h, k, dx, n):
    """Gradient of F_k with respect to h.
    k is the lag (0 to 2n-2 in correlate indexing, but actual lag is k-(n-1))"""
    # In correlate full mode, index r corresponds to lag = r - (n-1)
    lag = k - (n - 1)
    grad = np.zeros(n)
    for p in range(n):
        g = 0.0
        # h[p] appears as first factor when p + lag is valid
        if 0 <= p + lag < n:
            g += (1.0 - h[p + lag])
        # h[p] appears as second factor (via 1-h) when p - lag is valid
        if 0 <= p - lag < n:
            g -= h[p - lag]
        grad[p] = g * dx
    return grad

def project_box_sum(h, n):
    """Project h to [0,1]^n with sum = n/2. Iterative."""
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Add mass to positions not at 1

import numpy as np
import time
from scipy.optimize import minimize, linprog


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_f_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def gradient_fk(h, k, dx, n):
    lag = k - (n - 1)
    grad = np.zeros(n)
    for p in range(n):
        g = 0.0
        if 0 <= p + lag < n:
            g += (1.0 - h[p + lag])
        if 0 <= p - lag < n:
            g -= h[p - lag]
        grad[p] = g * dx
    return grad


def gradient_all(h, n):
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        grads[k] = gradient_fk(h, k, dx, n)
    return grads


def project_box_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def line_search(h, direction, n, c5_cur, max_beta=1.0):
    beta_max = max_beta
    for i in range(n):
        if direction[i] > 1e-14:
            bm = (1.0 - h[i]) / direction[i]
            if bm < beta_max:
                beta_max = bm
        elif direction[i] < -1e-14:
            bm = h[i] / (-direction[i])
            if bm < beta_max:
                beta_max = bm
    beta_max = max(0.0, min(beta_max, max_beta))
    if beta_max < 1e-14:
        return h, c5_cur, False
    best_beta = 0.0
    best_c5 = c5_cur
    fracs = [1.0, 0.85, 0.7, 0.55, 0.4, 0.3, 0.2, 0.12, 0.07, 0.035, 0.01]
    for frac in fracs:
        beta = beta_max * frac
        if beta < 1e-14:
            break
        new_h = h + beta * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        if abs(np.sum(new_h) - n / 2.0) > 1e-10:
            new_h = project_box_sum(new_h, n)
        nc5 = eval_c5(new_h)
        if nc5 < best_c5 - 1e-13:
            best_c5 = nc5
            best_beta = beta
    if best_beta > 0:
        new_h = h + best_beta * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        if abs(np.sum(new_h) - n / 2.0) > 1e-10:
            new_h = project_box_sum(new_h, n)
        return new_h, best_c5, True
    return h, c5_cur, False


def lp_find_direction(h, n, f_vals, c5_max, rng, max_active=20):
    dx = 2.0 / n
    m_total = len(f_vals)
    threshold = c5_max - max(0.002 * c5_max, 0.002)
    active = np.where(f_vals >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(f_vals))])
    if len(active) > max_active:
        idx = np.argsort(f_vals[active])[::-1][:max_active]
        active = active[idx]
    m = len(active)
    A = np.zeros((m, n))
    for i, k in enumerate(active):
        A[i] = gradient_fk(h, k, dx, n)
    row_norms = np.sum(np.abs(A), axis=1, keepdims=True)
    row_norms[row_norms < 1e-14] = 1.0
    A = A / row_norms
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    A_ub[:m, :n] = A
    A_ub[:m, n] = 1.0
    A_ub[m:m + n, :n] = np.eye(n)
    b_ub[m:m + n] = 1.0
    A_ub[m + n:m + 2 * n, :n] = -np.eye(n)
    b_ub[m + n:m + 2 * n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(0.0, None)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None
    if result.status != 0:
        return None
    d = result.x[:n].copy()
    eps = result.x[n]
    if eps < 1e-8 or np.sum(np.abs(d)) < 1e-10:
        return None
    d = d - np.sum(d) / n
    return d


def random_pair_direction(h, n, rng):
    i = int(rng.randint(n))
    j = int(rng.randint(n))
    attempts = 0
    while j == i and attempts < 10:
        j = int(rng.randint(n))
        attempts += 1
    if j == i:
        return None
    max_move = min(h[j], 1.0 - h[i])
    if max_move < 1e-12:
        return None
    frac = rng.uniform(0.2, 1.0)
    delta = max_move * frac
    direction = np.zeros(n)
    direction[i] = delta
    direction[j] = -delta
    return direction


def gradient_move_direction(h, n, f_vals, c5_max, rng):
    dx = 2.0 / n
    m = len(f_vals)
    threshold = c5_max - max(0.003 * c5_max, 0.003)
    active = np.where(f_vals >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(f_vals))])
    weights = f_vals[active] - threshold + 0.001
    weights = weights / np.sum(weights)
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        grad += w * gradient_fk(h, k, dx, n)
    grad = grad - np.mean(grad)
    l1 = np.sum(np.abs(grad))
    if l1 < 1e-14:
        return None
    direction = -grad / l1
    return direction


class LSEObj:
    def __init__(self, n, tau):
        self.n = n
        self.dx = 2.0 / n
        self.tau = tau

    def value(self, h):
        f = all_f_values(h)
        f_max = np.max(f)
        f_shifted = f - f_max
        return self.tau * np.log(np.sum(np.exp(f_shifted / self.tau))) + f_max

    def grad(self, h):
        f = all_f_values(h)
        f_max = np.max(f)
        f_shifted = f - f_max
        w = np.exp(f_shifted / self.tau)
        w = w / np.sum(w)
        n = self.n
        grad = np.zeros(n)
        for p in range(n):
            g = 0.0
            for k in range(2 * n - 1):
                if w[k] < 1e-10:
                    continue
                lag = k - (n - 1)
                if 0 <= p + lag < n:
                    g += w[k] * (1.0 - h[p + lag])
                if 0 <= p - lag < n:
                    g -= w[k] * h[p - lag]
            grad[p] = g * self.dx
        return grad


class SumCon:
    def __init__(self, n):
        self.n = n
        self.target = n / 2.0

    def fun(self, h):
        return np.sum(h) - self.target

    def jac(self, h):
        return np.ones(self.n)


def lse_polish(h, n, c5_cur, time_limit=120):
    t_start = time.time()
    current = h.copy()
    current_c5 = c5_cur
    n = len(current)
    sc = SumCon(n)
    bounds = [(0.0, 1.0)] * n
    tauri = [0.04, 0.015, 0.006, 0.002, 0.0008]
    max_iter_per_tau = 60
    for tau in tauri:
        if time.time() - t_start > time_limit:
            break
        obj_fn = LSEObj(n, tau)
        constraints = [{'type': 'eq', 'fun': sc.fun, 'jac': sc.jac}]
        try:
            res = minimize(obj_fn.value, current, jac=obj_fn.grad,
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': max_iter_per_tau, 'ftol': 1e-12})
            if res.success and np.all(np.isfinite(res.x)):
                new_h = np.clip(res.x, 0.0, 1.0)
                new_h = project_box_sum(new_h, n)
                nc5 = eval_c5(new_h)
                if nc5 < current_c5 - 1e-13:
                    current = new_h
                    current_c5 = nc5
        except Exception:
            pass
    return current, current_c5


def transfer_dim(h_src, n_src, n_dst):
    src_grid = np.arange(n_src) * (2.0 / n_src)
    dst_grid = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_interp = np.interp(dst_grid, src_grid, h_src, left=h_src[0], right=h_src[-1])
    h_dst = project_box_sum(h_interp, n_dst)
    return h_dst


def structured_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    numpert = max(4, n // 5)
    idx = rng.choice(n, size=numpert, replace=False)
    for i in idx:
        h[i] = rng.uniform(0.0, 1.0)
    return project_box_sum(h, n)
