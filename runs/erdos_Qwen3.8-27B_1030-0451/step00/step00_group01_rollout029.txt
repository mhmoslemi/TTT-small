import time
import math

def project_feasible(z, target_sum):
    n = len(z)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(z.min()) - 2.0

                tot = float(av.sum())

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def exact_score_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def compute_g(h, k, n, ii):
    g = np.zeros(n)
    pk = (ii + k >= 0) & (ii + k < n)
    if pk.any():
        g[pk] += h[ii[pk] + k]
    mk = (ii - k >= 0) & (ii - k < n)
    if mk.any():
        g[mk] += h[ii[mk] - k]
    return g

def directed_local_search(h, n, target_sum, deadline, rng, max_rounds=50000):
    best_h = h.copy()
    best_c = exact_score(h)
    ii = np.arange(n)
    no_improve = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        corr, _ = exact_score_and_corr(best_h)
        max_c = float(corr.max())
        threshold = max_c * 0.97
        active_idx = np.where(corr >= threshold)[0]
        lags = (active_idx - (n - 1)).astype(int)
        
        min_g = np.full(n, 2.0)
        max_g = np.zeros(n)
        for k in lags:
            k = int(k)
            g = compute_g(best_h, k, n, ii)
            min_g = np.minimum(min_g, g)
            max_g = np.maximum(max_g, g)
        
        # Sort by min_g (desc) and max_g (asc)
        order_i = np.argsort(-min_g)
        order_j = np.argsort(max_g)
        
        found = False
        n_transfers = 80
        for attempt in range(n_transfers):
            idx_i = order_i[rng.integers(min(30, len(order_i)))]
            idx_j = order_j[rng.integers(min(30, len(order_j)))]
            if idx_i == idx_j:
                continue
            if max_g[idx_j] >= min_g[idx_i]:
                continue
            delta = rng.uniform(0.01, 0.08)
            vi = best_h[idx_i] + delta
            vj = best_h[idx_j] - delta
            if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                continue
            new_h = best_h.copy()
            new_h[idx_i] = vi
            new_h[idx_j] = vj
            c_new = exact_score(new_h)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = new_h
                found = True
                # Recompute g for next iteration
                break

            # Try swap moves
            for attempt in range(20):

                j = rng.integers(n)
                if i == j or abs(best_h[i] - best_h[j]) < 1e-6:
                    continue
                new_h = best_h.copy()
                new_h[i], new_h[j] = new_h[j], new_h[i]
                c_new = exact_score(new_h)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = new_h
                    found = True
                    break
            if not found:

                for attempt in range(15):
                    i = rng.integers(n)
                    j = rng.integers(n)

                        continue

                    vi = best_h[i] + delta

                        break

                    new_h = best_h.copy()
                    for _ in range(rng.integers(2, 6)):

def build_fourier_basis(n, max_order):
    x = (np.arange(n) + 0.5) / n
    cols = []
    for m in range(1, max_order + 1):
        cols.append(np.cos(2.0 * np.pi * m * x))
        cols.append(np.sin(2.0 * np.pi * m * x))
    B = np.column_stack(cols)
    for j in range(B.shape[1]):
        nrm = np.linalg.norm(B[:, j])
        if nrm > 1e-12:
            B[:, j] /= nrm
    return B

def latent_de(h0, n, target_sum, deadline, rng, d=20, P=30):
    B = build_fourier_basis(n, d // 2 + 1)
    if B.shape[1] < d:
        d = B.shape[1]
    B = B[:, :d]
    
    F = 0.7
    CR = 0.8
    pop = np.zeros((P, d))
    for i in range(1, P):
        sc = rng.choice([0.05, 0.1, 0.15, 0.25, 0.4])
        pop[i] = rng.standard_normal(d) * sc

    pop_c[0] = exact_score(h0)
    best_c = pop_c[0]
    best_h = h0.copy()
    best_idx = 0
    
    for i in range(1, P):
        h = project_feasible(h0 + B @ pop[i], target_sum)
        c = exact_score(h)
        pop_c[i] = c
        if c < best_c:
            best_c = c
            best_h = h.copy()
            best_idx = i

                break

            v = np.clip(v, -3.0, 3.0)
            trial = pop[i].copy()
            jr = rng.integers(d)
            mask = np.concatenate([rng.random(d) < CR, [True]])[:d]
            mask[jr] = True

                pop[i] = trial
                pop_c[i] = c
            if c < gen_best:
                gen_best = c

                if pop_c[i] <= gen_best + 1e-14:
                    best_c = pop_c[i]
                    best_h = project_feasible(h0 + B @ pop[i], target_sum)
                    best_idx = i
                    break
            stale = 0

            stale += 1

                if pop_c[i] < best_c:
                    best_c = pop_c[i]
                    best_h = h.copy()
            stale = 0
            F = rng.uniform(0.5, 1.0)

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c = exact_score(best_h)
    
    return best_h, best_c, len(best_h)
