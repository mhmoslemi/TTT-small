import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def precompute_R2(n):
    R2 = np.zeros((n, 2*n-1), dtype=np.float64)
    for p in range(n):
        k_min = max(0, 1 - p)
        k_max = min(2*n - 2, p + n - 1)
        if k_min <= k_max:
            R2[p, k_min:k_max+1] = 1.0
    return R2

def compute_S_row(h, R2, p):
    n = len(h)
    S_p = np.zeros(2*n-1, dtype=np.float64)
    # Q1[p][k] = h[p-n+1+k] for valid k
    k1_min = max(0, n - 1 - p)
    k1_max = min(2*n - 2, 2*n - 2 - p)
    if k1_min <= k1_max:
        h_idx = np.arange(k1_min, k1_max+1) - (n - 1 - p)
        S_p[k1_min:k1_max+1] += h[h_idx]
    # Q2[p][k] = h[p+n-1-k] for valid k
    k2_min = max(0, 1 - p)
    k2_max = min(2*n - 2, p + n - 1)
    if k2_min <= k2_max:
        h_idx = p + n - 1 - np.arange(k2_min, k2_max+1)
        S_p[k2_min:k2_max+1] += h[h_idx]
    S_p -= R2[p]
    return S_p

def compute_S_full(h, R2):
    n = len(h)
    S = np.zeros((n, 2*n-1), dtype=np.float64)
    for p in range(n):
        S[p] = compute_S_row(h, R2, p)
    return S
