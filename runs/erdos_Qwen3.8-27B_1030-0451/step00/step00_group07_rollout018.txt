```python
def _lp_step(h, active_grads, active_vals, rho):
    """Solve LP to find a descent direction.
    
    active_grads: array of shape (K, N) - gradients of active constraints
    active_vals: array of shape (K,) - current values of active constraints
    rho: trust region radius
    
    Returns: (delta, t) or (None, None) if infeasible
    """
    n = len(h)
    k = len(active_grads)
    
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: grad_k · delta - t <= 0
    # => [grad_k, -1] · [delta, t] <= 0
    A_ub = np.zeros((k, n + 1))
    A_ub[:, :n] = active_grads
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(k)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on delta: -min(h_i, rho) <= delta_i <= min(1-h_i, rho)
    # t is unbounded (or bounded by reasonable values)
    bounds = []
    for i in range(n):
        lo = -min(h[i], rho)
        hi = min(1.0 - h[i], rho)
        bounds.append((lo, hi))
    bounds.append((None, None))  # t unbounded
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None
```

```python
def _smoothed_objective_and_grad(h, t=0.01):
    """Compute smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = _corr_array(h)
    
    # Softmax weights
    corr_scaled = corr / t
    corr_scaled -= np.max(corr_scaled)  # Numerical stability
    weights = np.exp(corr_scaled)
    weights /= np.sum(weights)
    
    # Objective
    obj = t * (np.max(corr_scaled) + np.log(np.sum(weights)))
    # Actually: logsumexp(corr/t) = max(corr)/t + log(sum(exp((corr-max(corr))/t)))
    # So f = t * [max(corr)/t + log(sum(exp((corr-max(corr))/t)))]
    #     = max(corr) + t * log(sum(exp((corr-max(corr))/t)))
    obj = np.max(corr) + t * np.log(np.sum(np.exp((corr - np.max(corr)) / t)))
    
    # Gradient: weighted average of individual gradients
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        grad += weights[k] * _grad_for_shift(s, h, dx)
    
    return obj, grad
```

```python
import numpy as np
import time
import math
from scipy.optimize import linprog, minimize

def _evaluate(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _corr_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(x, target):
    x = np.array(x, dtype=np.float64)
    n = len(x)
    if n == 0:
        return x
    x = np.clip(x, -5.0, 5.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    res = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = math.fsum(res.tolist()) - target
    if diff > 1e-12:
        for i in np.argsort(-res):
            if diff <= 1e-13: break
            rm = min(res[i], diff); res[i] -= rm; diff -= rm
    elif diff < -1e-12:
        for i in np.argsort(res):
            if -diff <= 1e-13: break
            add = min(1.0 - res[i], -diff); res[i] += add; diff += add
    return res

def _grad_for_shift(s, h, dx):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] = 1.0 - h[s:]
        g[s:] = -h[:n-s]
    else:
        t = -s
        g[t:] = 1.0 - h[:n-t]
        g[:n-t] = -h[t:]
    return g * dx

def _smooth_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = _corr_array(h)
    cmax = np.max(corr)
    scaled = (corr - cmax) / t
    exp_s = np.exp(scaled)
    w = exp_s / np.sum(exp_s)
    obj = cmax + t * np.log(np.sum(exp_s))
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            s = k - (n-1)
            grad += w[k] * _grad_for_shift(s, h, dx)
    return obj, grad

def _transfer(h_src, n_dst):
    n_src = len(h_src)
    if n_dst == n_src:
        return h_src.copy()
    # Use interpolation for efficiency
    x_src = np.linspace(0, 2.0, n_src, endpoint=False) + 1.0/n_src
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False) + 1.0/n_dst
    # For downsampling, use cell average; for upsampling, use interpolation
    if n_dst > n_src:
        # Upsample: value at center of each new cell
        result = np.interp(x_dst, x_src, h_src, period=2.0)
    else:
        # Downsample: average over each new cell
        result = np.zeros(n_dst)
        for i in range(n_dst):
            a = i * 2.0 / n_dst
            b = a + 2.0 / n_dst
            # Find which source cells overlap
            j0 = int(a * n_src / 2.0)
            j1 = min(n_src, int(b * n_src / 2.0) + 1)
            total = 0.0
            w = 0.0
            for j in range(max(0, j0), j1):
                la = max(a, j * 2.0 / n_src)
                lb = min(b, (j+1) * 2.0 / n_src)
                if lb > la:
                    total += h_src[j] * (lb - la)
                    w += (lb - la)
            result[i] = total / w if w > 0 else 0.5
    return result
```

```python
def _smooth_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = _corr_array(h)
    cmax = np.max(corr)
    scaled = (corr - cmax) / t
    exp_s = np.exp(scaled)
    s_exp = np.sum(exp_s)
    w = exp_s / s_exp
    obj = cmax + t * math.log(s_exp)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-8:
            s = k - (n - 1)
            grad += w[k] * _grad_for_shift(s, h, dx)
    return obj, grad

def _transfer(h_src, n_dst):
    n_src = len(h_src)
    if n_dst == n_src:
        return h_src.copy()
    if n_dst > n_src:
        x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
        x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
        result = np.interp(x_dst, x_src, h_src, left=h_src[0], right=h_src[-1])
    else:
        result = np.zeros(n_dst)
        for i in range(n_dst):
            a = i * 2.0 / n_dst
            b = a + 2.0 / n_dst
            j0 = max(0, int(a * n_src / 2.0))
            j1 = min(n_src, int(b * n_src / 2.0) + 1)
            total = 0.0
            for j in range(j0, j1):
                la = max(a, j * 2.0 / n_src)
                lb = min(b, (j + 1) * 2.0 / n_src)
                if lb > la:
                    total += h_src[j] * (lb - la)
            result[i] = total / (2.0 / n_dst)
    return result

def _pg_optimize(h, n, deadline, rng, t_start=0.05, t_end=0.002, n_iter=500):
    target = n / 2.0
    h = _project(h, target)
    best_h = h.copy()
    best_val = _evaluate(h)
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        obj, grad = _smooth_obj_grad(h, t)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad_norm = grad / gnorm
        momentum = beta * momentum - lr * grad_norm
        trial = _project(h + momentum, target)
        # Backtracking line search
        alpha = 1.0
        for _bt in range(20):
            trial2 = _project(h + alpha * momentum, target)
            obj2, _ = _smooth_obj_grad(trial2, t)
            if obj2 < obj + 1e-4:
                h = trial2
                break
            alpha *= 0.5
        else:
            h = trial2
        exact_val = _evaluate(h)
        if exact_val < best_val - 1e-12:
            best_val = exact_val
            best_h = h.copy()
        # Slowly reduce learning rate
        if it % 50 == 49:
            lr *= 0.7
    return best_h, best_val

def _lp_optimize(h, n, deadline, rng):
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h, target)
    best_h = h.copy()
    best_val = _evaluate(h)
    rho = 0.08
    stall = 0
    
    for it in range(300):
        if time.time() > deadline:
            break
        corr = _corr_array(h)
        cmax = np.max(corr)
        n_active = max(5, min(2 * n - 1, int(0.2 * (2 * n - 1))))
        top_idx = np.argsort(corr)[::-1][:n_active]
        
        # Build LP
        A_ub = np.zeros((n_active, n + 1))
        for ai, k in enumerate(top_idx):
            s = k - (n - 1)
            A_ub[ai, :n] = _grad_for_shift(s, h, dx)
            A_ub[ai, n] = -1.0
        b_ub = np.zeros(n_active)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bounds = []
        for i in range(n):
            lo = -min(h[i], rho)
            hi = min(1.0 - h[i], rho)
            bounds.append((lo, hi))
        bounds.append((None, None))
        
        res = linprog(np.zeros(n + 1), A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        
        if not res.success:
            delta = rng.normal(0, 0.03, n)
            delta -= delta.mean()
        else:
            delta = res.x[:n]
            t_val = res.x[n]
            if t_val >= -1e-10:
                delta = rng.normal(0, 0.03, n)
                delta -= delta.mean()
                stall += 1
            else:
                stall = 0
        
        if stall > 8:
            delta = rng.normal(0, 0.08, n)
            delta -= delta.mean()
            stall = 0
            rho = max(0.02, rho * 0.5)
        
        best_alpha = 0.0
        best_trial = best_val
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07]:
            trial = _project(h + alpha * delta, target)
            tv = _evaluate(trial)
            if tv < best_trial - 1e-13:
                best_trial = tv
                best_alpha = alpha
                best_trial_h = trial
        
        if best_alpha > 0:
            h = best_trial_h
            if best_trial < best_val - 1e-12:
                best_val = best_trial
                best_h = h.copy()
                rho = min(0.3, rho * 1.15)
        
        if time.time() > deadline - 2.0:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    try:
        h_init = np.asarray(initial_h_values, dtype=np.float64).flatten()
    except (NameError, TypeError):
        h_init = np.full(120, 0.5)
    
    n0 = len(h_init)
    if n0 < 2:
        h_init = np.full(120, 0.5)
        n0 = 120
    
    target0 = n0 / 2.0
    h = _project(h_init, target0)
    best_h = h.copy()
    best_val = _evaluate(h)
    
    # Sanity: try a few random starts at n0
    for _ in range(3):
        rh = rng.uniform(0.1, 0.9, n0)
        rh = _project(rh, target0)
        rv = _evaluate(rh)
        if rv < best_val:
            best_val = rv
            best_h = rh.copy()
    
    # Deadline with reserve
    deadline = t0 + budget_s - 10.0
    
    # Phase 1: Optimize at native resolution
    h1, v1 = _pg_optimize(best_h, n0, min(deadline, t0 + budget_s * 0.25), rng)
    if v1 < best_val:
        best_val = v1
        best_h = h1.copy()
    
    h2, v2 = _lp_optimize(best_h, n0, min(deadline, t0 + budget_s * 0.4), rng)
    if v2 < best_val:
        best_val = v2
        best_h = h2.copy()
    
    # Phase 2: Multi-resolution
    if time.time() < deadline - 30:
        resolutions = []
        for mult in [2, 4]:
            nr = n0 * mult
            if nr <= 600:
                resolutions.append(nr)
        if n0 * 2 > 400:
            resolutions = [n0 * 2] if n0 * 2 <= 600 else []
        
        current_h = best_h
        for nr in resolutions:
            if time.time() > deadline - 15:
                break
            h_trans = _transfer(current_h, nr)
            h_trans = _project(h_trans, nr / 2.0)
            
            # Optimize at new resolution
            time_left = deadline - time.time()
            if time_left > 20:
                h3, v3 = _pg_optimize(h_trans, nr, t0 + time.time() - time.time() + time_left * 0.6, rng,
                                      t_start=0.03, t_end=0.002, n_iter=300)
                if v3 < best_val:
                    best_val = v3
                    best_h = h3.copy()
                    current_h = h3
                time_left = deadline - time.time()
            if time_left > 15:
                h4, v4 = _lp_optimize(current_h, nr, t0 + time.time() - time.time() + time_left * 0.5, rng)
                if v4 < best_val:
                    best_val = v4
                    best_h = h4.copy()
                    current_h = h4
    
    # Phase 3: Random restarts at best resolution
    if time.time() < deadline - 10:
        nb = len(best_h)
        for _ in range(20):
            if time.time() > deadline - 5:
                break
            # Perturb best solution
            scale = rng.uniform(0.05, 0.3)
            pert = rng.normal(0, scale, nb)
            cand = _project(best_h + pert, nb / 2.0)
            cv = _evaluate(cand)
            if cv < best_val:
                # Quick polish
                t_sub = min(deadline - 5, time.time() + min(30, (deadline - time.time()) * 0.3))
                h5, v5 = _pg_optimize(cand, nb, t_sub, rng, t_start=0.02, t_end=0.002, n_iter=100)
                if v5 < best_val:
                    best_val = v5
                    best_h = h5.copy()
    
    # Phase 4: Final LP polish
    if time.time() < deadline - 3:
        nb = len(best_h)
        h6, v6 = _lp_optimize(best_h, nb, deadline, rng)
        if v6 < best_val:
            best_val = v6
            best_h = h6.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, len(best_h) / 2.0)
    final_val = _evaluate(best_h)
    
    # If somehow invalid, fallback
    if not np.isfinite(final_val) or len(best_h) < 2:
        best_h = np.full(120, 0.5)
        best_h = _project(best_h, 60.0)
        final_val = _evaluate(best_h)
    
    return best_h, final_val, len(best_h)
```
