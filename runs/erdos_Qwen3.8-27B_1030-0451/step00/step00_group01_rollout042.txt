    # s.t. grads[i].dot(h) - t <= grads[i].dot(h) - vals[i]  (for each active lag)

    #              h0[j] - h[j] <= radius  ->  -h[j] <= h0[j] - radius - ...

import numpy as np
import time
from scipy.optimize import minimize, linprog


def project_sum_box(x, target):
    """Project x onto {0 <= y <= 1, sum(y) = target}."""
    n = len(x)
    lo = -x.max() - 1.0
    hi = 1.0 - x.min() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-12:
        if res > 0:
            mask = y < 1.0
        else:
            mask = y > 0.0
        if mask.any():
            slack = (1.0 - y)[mask] if res > 0 else y[mask]
            if slack.sum() > 1e-15:
                y[mask] += res * slack / slack.sum()
    return np.clip(y, 0.0, 1.0)


def compute_c5(h):
    """Return (c5_value, all_lag_values)."""
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return vals.max(), vals


def grad_lag(h, k):
    """Gradient of raw c_k (before dx) w.r.t. h for lag k (int, can be negative)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += 1.0 - h[:n - k]
            g[:n - k] -= h[k:]
    else:
        d = -k
        if d < n:
            g[:n - d] += 1.0 - h[d:]
            g[d:] -= h[:n - d]
    return g


def logsumexp_val_grad(h, beta):
    """Log-sum-exp smooth approx to max c_k and its gradient."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full') * dx
    m = raw.max()
    u = np.exp(beta * (raw - m))
    Z = u.sum()
    f = m + np.log(Z) / beta
    w = u / Z
    g = np.zeros(n)
    for idx in range(len(raw)):
        k = idx - (n - 1)
        g += w[idx] * grad_lag(h, k)
    g *= dx
    return f, g


def slsqp_eq_fun(x, target):
    return x.sum() - target


def slsqp_eq_jac(x, target):
    return np.ones(len(x))


def slsqp_obj(x, beta):
    f, g = logsumexp_val_grad(x, beta)
    return f, g


def solve_lp_step(h, n_active, radius):
    """One LP linearization step. Returns new h or None."""
    n = len(h)
    dx = 2.0 / n
    _, vals = compute_c5(h)
    n_act = min(n_active, 2 * n - 1)
    idx = np.argsort(-vals)[:n_act]
    grads = np.zeros((n_act, n))
    for i, lag_idx in enumerate(idx):
        k = int(lag_idx - (n - 1))
        grads[i] = grad_lag(h, k) * dx
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    for i in range(n_act):
        A_ub[i, :n] = grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = grads[i] @ h - vals[i]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for j in range(n):
        lo_j = max(0.0, h[j] - radius)
        hi_j = min(1.0, h[j] + radius)
        bounds.append((lo_j, hi_j))
    bounds.append((-0.5, 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.success and res.x is not None:
        h_new = res.x[:n]
        return project_sum_box(h_new, n / 2.0)
    return None


def run_lp_phase(h, max_iter, init_radius, deadline):
    """Iterate LP steps until no improvement or time runs out."""
    n = len(h)
    c_curr, _ = compute_c5(h)
    radius = init_radius
    stalled = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new = solve_lp_step(h, min(40, 2 * n - 1), radius)
        if h_new is None:
            radius *= 0.5
            stalled += 1
            if stalled > 5:
                break
            continue
        c_new, _ = compute_c5(h_new)
        if c_new < c_curr - 1e-10:
            h = h_new
            c_curr = c_new
            stalled = 0
        else:
            radius *= 0.5
            stalled += 1
            if stalled > 5:
                break
        if radius < 1e-4:
            break
    return h, c_curr


def run_slsqp_phase(h, beta_list, deadline, maxiter_per_beta=100):
    """Run SLSQP with increasing beta values."""
    n = len(h)
    target = n / 2.0
    constraints = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac,
                    'args': (target,)}]
    bounds = [(0.0, 1.0)] * n
    c_curr, _ = compute_c5(h)
    for beta in beta_list:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_obj, h.copy(), jac=True, args=(beta,),
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter_per_beta, 'ftol': 1e-12})
        except Exception:
            continue
        if res.x is not None:
            h_cand = project_sum_box(res.x, target)
            c_cand, _ = compute_c5(h_cand)
            if np.isfinite(c_cand) and c_cand < c_curr - 1e-10:
                h = h_cand
                c_curr = c_cand
    return h, c_curr


def run_grad_descent(h, beta, step_init, n_iters, deadline):
    """Projected gradient descent on log-sum-exp."""
    n = len(h)
    target = n / 2.0
    c_curr, _ = compute_c5(h)
    step = step_init
    for it in range(n_iters):
        if time.time() > deadline:
            break
        f, g = logsumexp_val_grad(h, beta)
        h_new = h - step * g
        h_new = project_sum_box(h_new, target)
        c_new, _ = compute_c5(h_new)
        if c_new < c_curr - 1e-10:
            h = h_new
            c_curr = c_new
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return h, c_curr


def transfer_resolution(h_old, n_new):
    """Interpolate h_old to a new grid of size n_new."""
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_sum_box(h_new, n_new / 2.0)
