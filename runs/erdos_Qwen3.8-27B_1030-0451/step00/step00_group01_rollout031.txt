def line_search_pair(h, n, p, q, current_max):
    """Find best α for moving mass from p to q.
    h_new[α] has h[p]-α at position p, h[q]+α at position q.
    α ∈ [0, min(h[p], 1-h[q])]"""
    eps = 1e-12
    alpha_max = min(h[p], 1.0 - h[q]) - eps
    if alpha_max <= 0:
        return 0, current_max
    
    d = np.zeros(n)
    d[p] = -1.0
    d[q] = 1.0
    
    # Compute polynomial coefficients
    one_minus_h = 1.0 - h
    a_vec = np.correlate(h, one_minus_h, mode='full')  # O(n)
    # For b_vec: corr(d, 1-h) - corr(h, d)
    # d has only 2 nonzero entries, so these are cheap
    # corr(d, one_minus_h): at position k (lag = k-(n-1)):
    #   = sum_n d[n] * one_minus_h[n+lag] = -one_minus_h[p] * I[p+lag valid] + one_minus_h[q] * I[q+lag valid]
    # Actually, let me just compute them directly since they're O(n)
    b_vec1 = np.correlate(d, one_minus_h, mode='full')  # O(n) since d is sparse
    b_vec2 = np.correlate(h, d, mode='full')  # O(n)
    b_vec = b_vec1 - b_vec2
    
    # c_vec = -corr(d, d)
    c_vec = -np.correlate(d, d, mode='full')  # very sparse
    
    # Now minimize max_l (a_l + b_l*alpha + c_l*alpha^2) over [0, alpha_max]
    # Strategy: scan grid + vertices + refine
    n_scan = 33
    alphas = np.linspace(0, alpha_max, n_scan)
    
    # Evaluate at grid points
    scores = np.zeros(n_scan)
    for i, alpha in enumerate(alphas):
        val = a_vec + b_vec * alpha + c_vec * alpha * alpha
        scores[i] = np.max(val)
    
    # Add vertices of active quadratics
    # For each l with c_l != 0: vertex at -b_l / (2*c_l)
    # Only add if in [0, alpha_max]
    mask = c_vec != 0
    if np.any(mask):
        vertices = -b_vec[mask] / (2.0 * c_vec[mask])
        valid_v = (vertices >= 0) & (vertices <= alpha_max) & np.isfinite(vertices)
        if np.any(valid_v):
            v_alpha = vertices[valid_v]
            # Limit number of vertices to add (take those that give lowest value)
            v_scores = np.array([np.max(a_vec + b_vec*va + c_vec*va*va) for va in v_alpha])
            # Merge with grid
            all_alphas = np.concatenate([alphas, v_alpha])
            all_scores = np.concatenate([scores, v_scores])
    else:
        all_alphas = alphas
        all_scores = scores
    
    # Find best
    best_idx = np.argmin(all_scores)
    best_alpha = all_alphas[best_idx]
    best_score = all_scores[best_idx]
    
    # Refine around best
    if best_alpha > 0 and best_alpha < alpha_max:
        delta = max(1e-6, alpha_max / n_scan * 0.5)
        lo = max(0, best_alpha - delta)
        hi = min(alpha_max, best_alpha + delta)
        def obj(alpha):
            val = a_vec + b_vec * alpha + c_vec * alpha * alpha
            return np.max(val)
        try:
            refined = fminbound(obj, lo, hi, xtol=1e-10, maxiter=50)
            refined_score = obj(refined)
            if refined_score < best_score:
                best_alpha = refined
                best_score = refined_score
        except:
            pass
    
    return best_alpha, best_score * (2.0/n)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def resample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    # For step function resampling: each old point covers [i/n_old, (i+1)/n_old]
    # For new point j, it's in the old bin floor(j*n_old/n_new)
    new_idx = np.arange(n_new)
    old_idx = (new_idx * n_old / n_new).astype(int)
    old_idx = np.clip(old_idx, 0, n_old - 1)
    h_new = h_old[old_idx].astype(float)
    return h_new

def make_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        room_sum = room.sum()
        if room_sum > 0:
            h += diff * room / room_sum
    else:
        avail = h.copy()
        avail_sum = avail.sum()
        if avail_sum > 0:
            h += diff * avail / avail_sum  # diff < 0, so this subtracts
    h = np.clip(h, 0.0, 1.0)
    # One more adjustment pass
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

        top_idx = np.arange(M)

    for k in top_idx:

        # S_k = {n : max(0, -lag) <= n <= min(n-1, n-1-lag)}
        n_lo = max(0, -lag)
        n_hi = min(n - 1, n - 1 - lag)
        if n_lo > n_hi:

        # For each m in 0..n-1:
        # contribution = (1-h[m+lag]) * I[m in S_k] - h[m-lag] * I[m-lag in S_k]
        # m in S_k means n_lo <= m <= n_hi
        # m-lag in S_k means n_lo <= m-lag <= n_hi, i.e., n_lo+lag <= m <= n_hi+lag
        
        mask1 = (m_arr >= n_lo) & (m_arr <= n_hi)  # for term 1
        m2 = m_arr - lag
        mask2 = (m2 >= 0) & (m2 < n) & (m2 >= n_lo) & (m2 <= n_hi)  # for term 2
        
        # Term 1: (1-h[m+lag]) for m in S_k
        m1 = m_arr + lag

        valid2 = (m2 >= 0) & (m2 < n) & mask2
        g[valid2] += -h[m2[valid2]]

def compute_subgradient(h, n, K=20):
    dx = 2.0 / n
    a_vec = np.correlate(h, 1.0 - h, mode='full')
    M = len(a_vec)
    
    if K < M:
        top_idx = np.argpartition(-a_vec, min(K, M-1))[:K]
    else:
        top_idx = np.arange(M)
    
    g = np.zeros(n)
    m = np.arange(n)
    
    for k in top_idx:
        lag = k - (n - 1)
        n_lo = max(0, -lag)
        n_hi = min(n - 1, n - 1 - lag)
        if n_lo > n_hi:
            continue
        
        # Term 1: for m in [n_lo, n_hi], add (1-h[m+lag]) if m+lag valid
        m1_lo = n_lo
        m1_hi = n_hi
        valid_m1 = np.arange(m1_lo, m1_hi + 1)
        idx1 = valid_m1 + lag
        valid = (idx1 >= 0) & (idx1 < n)
        if np.any(valid):
            g[valid_m1[valid]] += (1.0 - h[idx1[valid]])
        
        # Term 2: for m such that m-lag in [n_lo, n_hi], subtract h[m-lag]
        m2_lo = n_lo + lag
        m2_hi = n_hi + lag
        m2_lo = max(0, m2_lo)
        m2_hi = min(n - 1, m2_hi)
        if m2_lo <= m2_hi:
            valid_m2 = np.arange(m2_lo, m2_hi + 1)
            idx2 = valid_m2 - lag
            valid = (idx2 >= 0) & (idx2 < n)
            if np.any(valid):
                g[valid_m2[valid]] -= h[idx2[valid]]
    
    g *= dx
    return g, a_vec

def line_search_pair(h, n, p, q, a_vec, current_score):
    """Line search for pair move p→q. Returns (best_alpha, best_score)."""
    alpha_max = min(h[p], 1.0 - h[q])
    if alpha_max <= 1e-12:
        return 0.0, current_score
    
    lag_diff = q - p  # this is the "lag" for the d vector
    
    # Compute b_vec[k] for k = 0, ..., 2n-2
    # b_vec[k] = corr(d, 1-h)[k] - corr(h, d)[k]
    # where d[p]=-1, d[q]=1
    # corr(d, 1-h)[k] = (1-h)[q+lag] - (1-h)[p+lag] where lag = k-(n-1)
    #   = h[p+lag] - h[q+lag] (with 0 for out of bounds)
    # corr(h, d)[k] = h[q-lag] - h[p-lag] (with 0 for out of bounds)
    # So b_vec[k] = h[p+lag] - h[q+lag] - h[q-lag] + h[p-lag]
    
    k_arr = np.arange(2 * n - 1)
    lag_arr = k_arr - (n - 1)
    
    b_vec = np.zeros(2 * n - 1)
    # h[p+lag]
    idx_pp = p + lag_arr
    valid_pp = (idx_pp >= 0) & (idx_pp < n)
    b_vec[valid_pp] += h[idx_pp[valid_pp]]
    # -h[q+lag]
    idx_qq = q + lag_arr
    valid_qq = (idx_qq >= 0) & (idx_qq < n)
    b_vec[valid_qq] -= h[idx_qq[valid_qq]]
    # -h[q-lag]
    idx_ql = q - lag_arr
    valid_ql = (idx_ql >= 0) & (idx_ql < n)
    b_vec[valid_ql] -= h[idx_ql[valid_ql]]
    # +h[p-lag]
    idx_pl = p - lag_arr
    valid_pl = (idx_pl >= 0) & (idx_pl < n)
    b_vec[valid_pl] += h[idx_pl[valid_pl]]
    
    # c_vec: -2 at lag=0 (k=n-1), +1 at lag=q-p (k=n-1+q-p), +1 at lag=p-q (k=n-1+p-q)
    c_vec = np.zeros(2 * n - 1)
    k0 = n - 1
    if 0 <= k0 < 2*n-1:
        c_vec[k0] = -2.0
    k1 = n - 1 + (q - p)
    if 0 <= k1 < 2*n-1:
        c_vec[k1] += 1.0
    k2 = n - 1 + (p - q)
    if 0 <= k2 < 2*n-1:
        c_vec[k2] += 1.0
    
    # Now minimize max_l (a_l + b_l*α + c_l*α²) over [0, alpha_max]
    return _minimize_line(a_vec, b_vec, c_vec, alpha_max, n)

def _minimize_line(a_vec, b_vec, c_vec, alpha_max, n):
    """Minimize max(a + b*α + c*α²) over [0, alpha_max]."""
    dx = 2.0 / n
    
    # Coarse scan
    n_scan = 25
    alphas = np.linspace(0, alpha_max, n_scan)
    
    best_score = np.inf
    best_alpha = 0.0
    
    for alpha in alphas:
        val = a_vec + b_vec * alpha + c_vec * alpha * alpha
        s = np.max(val) * dx
        if s < best_score:
            best_score = s
            best_alpha = alpha
    
    # Add vertices of quadratics (where c != 0)
    # Vertex at -b/(2c) for each position with c != 0
    nonzero_c = c_vec != 0
    if np.any(nonzero_c):
        c_nz = c_vec[nonzero_c]
        b_nz = b_vec[nonzero_c]
        a_nz = a_vec[nonzero_c]
        vertices = -b_nz / (2.0 * c_nz)
        valid_v = (vertices >= 0) & (vertices <= alpha_max) & np.isfinite(vertices)
        if np.any(valid_v):
            v_alphas = vertices[valid_v]
            # Limit to top 20 by potential improvement
            # Score at vertex: a + b*v + c*v² = a - b²/(4c)
            v_scores = (a_nz[valid_v] + b_nz[valid_v] * v_alphas + c_nz[valid_v] * v_alphas * v_alphas)
            # Actually I need max over all positions, not just the vertex position
            for va in v_alphas[:50]:  # limit
                val = a_vec + b_vec * va + c_vec * va * va
                s = np.max(val) * dx
                if s < best_score:
                    best_score = s
                    best_alpha = va
    
    # Refine with fminbound
    if 0 < best_alpha < alpha_max:
        delta = alpha_max / n_scan * 0.5
        lo = max(0, best_alpha - delta)
        hi = min(alpha_max, best_alpha + delta)
        
        def obj(alpha):
            val = a_vec + b_vec * alpha + c_vec * alpha * alpha
            return np.max(val)
        
        try:
            refined = fminbound(obj, lo, hi, xtol=1e-12, maxiter=100)
            rs = obj(refined) * dx
            if rs < best_score:
                best_score = rs
                best_alpha = refined
        except:
            pass
    
    return best_alpha, best_score

def line_search_pair(h, n, p, q, a_vec, current_score):
    alpha_max = min(h[p], 1.0 - h[q])
    if alpha_max <= 1e-12:
        return 0.0, current_score
    dx = 2.0 / n
    
    # Compute b_vec
    k_arr = np.arange(2 * n - 1)
    lag_arr = k_arr - (n - 1)
    
    b_vec = np.zeros(2 * n - 1)
    idx = p + lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] += h[idx[m]]
    idx = q + lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] -= h[idx[m]]
    idx = q - lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] -= h[idx[m]]
    idx = p - lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] += h[idx[m]]
    
    # Compute c_vec (only 3 nonzero entries)
    c_vec = np.zeros(2 * n - 1)
    c_vec[n-1] = -2.0
    k1 = n - 1 + (q - p)
    if 0 <= k1 < 2*n-1:
        c_vec[k1] += 1.0
    k2 = n - 1 + (p - q)
    if 0 <= k2 < 2*n-1:
        c_vec[k2] += 1.0
    
    # Vectorized scan
    n_scan = 33
    alphas = np.linspace(0, alpha_max, n_scan)
    val = a_vec[None, :] + np.outer(alphas, b_vec) + np.outer(alphas**2, c_vec)
    scores = val.max(1) * dx
    
    best_i = scores.argmin()
    best_alpha = alphas[best_i]
    best_score = scores[best_i]
    
    # Refine
    if 0 < best_alpha < alpha_max:
        delta = alpha_max / n_scan
        lo = max(0, best_alpha - delta)
        hi = min(alpha_max, best_alpha + delta)
        
        # Finer scan in this interval
        fine_alphas = np.linspace(lo, hi, 33)
        val = a_vec[None, :] + np.outer(fine_alphas, b_vec) + np.outer(fine_alphas**2, c_vec)
        fine_scores = val.max(1) * dx
        fi = fine_scores.argmin()
        if fine_scores[fi] < best_score:
            best_alpha = fine_alphas[fi]
            best_score = fine_scores[fi]
    
    return best_alpha, best_score

def _compute_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _resample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    new_idx = np.arange(n_new)
    old_idx = (new_idx * n_old / n_new).astype(np.int64)
    old_idx = np.clip(old_idx, 0, n_old - 1)
    return h_old[old_idx].astype(np.float64).copy()


def _make_feasible(h, n):
    h = h.copy().astype(np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        rs = room.sum()
        if rs > 1e-15:
            h += diff * room / rs
    else:
        avail = h.copy()
        as_ = avail.sum()
        if as_ > 1e-15:
            h += diff * avail / as_
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h


def _compute_subgradient(h, n, K=25):
    dx = 2.0 / n
    a_vec = np.correlate(h, 1.0 - h, mode='full')
    M = len(a_vec)
    if K < M:
        top_idx = np.argpartition(-a_vec, min(K, M - 1))[:K]
    else:
        top_idx = np.arange(M)
    
    g = np.zeros(n)
    for k in top_idx:
        lag = k - (n - 1)
        n_lo = max(0, -lag)
        n_hi = min(n - 1, n - 1 - lag)
        if n_lo > n_hi:
            continue
        # Term 1: m in [n_lo, n_hi], add (1 - h[m+lag])
        m1 = np.arange(n_lo, n_hi + 1)
        idx1 = m1 + lag
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            g[m1[valid1]] += 1.0 - h[idx1[valid1]]
        # Term 2: m-lag in [n_lo, n_hi], subtract h[m-lag]
        m2_lo = max(0, n_lo + lag)
        m2_hi = min(n - 1, n_hi + lag)
        if m2_lo <= m2_hi:
            m2 = np.arange(m2_lo, m2_hi + 1)
            idx2 = m2 - lag
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid2):
                g[m2[valid2]] -= h[idx2[valid2]]
    
    g *= dx
    return g, a_vec


def _line_search_pair(h, n, p, q, a_vec, current_score):
    dx = 2.0 / n
    alpha_max = min(h[p], 1.0 - h[q])
    if alpha_max <= 1e-12:
        return 0.0, current_score
    
    lag_arr = np.arange(2 * n - 1) - (n - 1)
    
    # b_vec[k] = h[p+lag] - h[q+lag] - h[q-lag] + h[p-lag]
    b_vec = np.zeros(2 * n - 1)
    idx = p + lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] += h[idx[m]]
    idx = q + lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] -= h[idx[m]]
    idx = q - lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] -= h[idx[m]]
    idx = p - lag_arr
    m = (idx >= 0) & (idx < n)
    b_vec[m] += h[idx[m]]
    
    # c_vec: -2 at k=n-1, +1 at k=n-1+(q-p), +1 at k=n-1+(p-q)
    c_vec = np.zeros(2 * n - 1)
    c_vec[n - 1] = -2.0
    k1 = n - 1 + (q - p)
    if 0 <= k1 < 2 * n - 1:
        c_vec[k1] += 1.0
    k2 = n - 1 + (p - q)
    if 0 <= k2 < 2 * n - 1:
        c_vec[k2] += 1.0
    
    # Vectorized scan
    n_scan = 33
    alphas = np.linspace(0.0, alpha_max, n_scan)
    val = a_vec[None, :] + np.outer(alphas, b_vec) + np.outer(alphas * alphas, c_vec)
    scores = val.max(axis=1) * dx
    
    best_i = scores.argmin()
    best_alpha = alphas[best_i]
    best_score = scores[best_i]
    
    # Second-level refinement
    if 0.0 < best_alpha < alpha_max:
        delta = alpha_max / n_scan
        lo = max(0.0, best_alpha - delta)
        hi = min(alpha_max, best_alpha + delta)
        fine_alphas = np.linspace(lo, hi, 33)
        val2 = a_vec[None, :] + np.outer(fine_alphas, b_vec) + np.outer(fine_alphas * fine_alphas, c_vec)
        fine_scores = val2.max(axis=1) * dx
        fi = fine_scores.argmin()
        if fine_scores[fi] < best_score:
            best_alpha = fine_alphas[fi]
            best_score = fine_scores[fi]
    
    # Third-level: fminbound around best
    if 0.0 < best_alpha < alpha_max and best_score < current_score:
        delta2 = (alpha_max / n_scan) * 0.3
        lo2 = max(0.0, best_alpha - delta2)
        hi2 = min(alpha_max, best_alpha + delta2)
        
        def _obj(a):
            v = a_vec + b_vec * a + c_vec * a * a
            return float(np.max(v))
        
        try:
            ra = fminbound(_obj, lo2, hi2, xtol=1e-13, maxiter=80)
            rs = _obj(ra) * dx
            if rs < best_score - 1e-15:
                best_alpha = ra
                best_score = rs
        except:
            pass
    
    return float(best_alpha), float(best_score)


def _optimize(h, n, score, time_budget, seed):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    no_improve = 0
    
    precompute_lag_arr = np.arange(2 * n - 1) - (n - 1)
    
    while time.time() - start_time < time_budget:
        t_left = time_budget - (time.time() - start_time)
        if t_left < 0.05:
            break
        
        g, a_vec = _compute_subgradient(h, n, K=25)
        g_mean = g.mean()
        d_sd = -g + g_mean
        
        eps = 1e-10
        p_mask = h > eps
        q_mask = h < 1.0 - eps
        p_idx = np.where(p_mask)[0]
        q_idx = np.where(q_mask)[0]
        
        if len(p_idx) == 0 or len(q_idx) == 0:
            break
        
        # Candidate pairs from gradient
        candidates = []
        top_k = min(8, len(p_idx), len(q_idx))
        p_order = np.argsort(d_sd[p_idx])[:top_k]
        q_order = np.argsort(-d_sd[q_idx])[:top_k]
        
        for pi in p_order:
            for qi in q_order:
                p, q = int(p_idx[pi]), int(q_idx[qi])
                if p != q:
                    candidates.append((p, q))
        
        # Add some random candidates
        n_rand = 4
        for _ in range(n_rand):
            p = int(p_idx[rng.randint(len(p_idx))])
            q = int(q_idx[rng.randint(len(q_idx))])
            if p != q:
                candidates.append((p, q))
        
        if not candidates:
            break
        
        # De-duplicate
        seen = set()
        unique_candidates = []
        for (p, q) in candidates:
            if (p, q) not in seen:
                seen.add((p, q))
                unique_candidates.append((p, q))
        
        # Evaluate all candidates
        best_alpha = 0.0
        best_new_score = score
        best_pq = None
        
        for (p, q) in unique_candidates:
            alpha, new_score = _line_search_pair(h, n, p, q, a_vec, score)
            if alpha > 1e-12 and new_score < best_new_score:
                best_new_score = new_score
                best_alpha = alpha
                best_pq = (p, q)
        
        if best_pq is not None and best_new_score < score - 1e-13:
            h[best_pq[0]] -= best_alpha
            h[best_pq[1]] += best_alpha
            score = best_new_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break
    
    h = _make_feasible(h, n)
    score = _compute_score(h, n)
    return h, score
