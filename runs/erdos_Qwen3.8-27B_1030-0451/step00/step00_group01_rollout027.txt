                sc_ = st_
                upd(hc, sc_)
            T = max(0.0003, T * 0.9997)

    # ===================== Phase 4: Systematic coordinate perturbation =====================
    if tl() > 20:
        step = 0.05
        for coord in range(n):
            if tl() < 10:
                break
            for delta in [step, -step, 2*step, -2*step]:
                ht = bh.copy()
                ht[coord] = np.clip(ht[coord] + delta, 0, 1)
                ht = _proj(ht, S)
                s = _sc(ht)
                if s < bs - 1e-12:
                    bh = ht.copy()
                    bs = s

    # ===================== Final validation =====================
    bh = _proj(bh, len(bh) / 2.0)
    bh = np.clip(bh, 0.0, 1.0)
    # Ensure sum is exact
    r = len(bh) / 2.0 - bh.sum()
    if abs(r) > 1e-12:
        if r > 0:
            m = bh < 1.0
        else:
            m = bh > 0.0
        idx = np.where(m)[0]
        if len(idx) > 0:
            bh[idx] = np.clip(bh[idx] + r / len(idx), 0.0, 1.0)
    bs = _sc(bh)
    return (bh, bs, len(bh))
