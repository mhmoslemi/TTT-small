    delta = np.zeros(2*n-1)

    # delta_R(k) = -(1-h[p+k])*I[0<=p+k<n] + h[p-k]*I[0<=p-k<n]

    # Let's just loop... or vectorize

    # Term 1: -(1-h[p+k]) for valid k where p+k in [0,n-1], i.e., k in [-p, n-1-p]
