import numpy as np
from scipy.optimize import minimize

def project_sum_box(x, target):
    n = len(x)
    # x should already be close to [0,1]
    f = lambda lam: np.sum(np.clip(x + lam, 0.0, 1.0))
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if f(mid) < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(x + lam, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        mask = (result > 0.01) & (result < 0.99)
        if np.sum(mask) > 0:
            adjust = residual / np.sum(mask)
            result[mask] = np.clip(result[mask] + adjust, 0.0, 1.0)
    return result

def exact_c5(h):
    """Compute the exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def overlap_values(h):
    """Compute all overlap values O(s) for s = -(n-1), ..., n-1."""
    c = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    dx = 2.0 / n
    return c * dx  # length 2n-1, index m corresponds to shift s = m - (n-1)

def smooth_obj_and_grad(h, beta):
    """Log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode="full") * dx  # length 2n-1
    m = np.max(v)
    e = np.exp(beta * (v - m))
    Z = np.sum(e)
    w = e / Z  # weights
    f = m + np.log(1.0 / (beta * Z))  # wait: (1/beta)*log(Z*exp(beta*m)) = m + (1/beta)*log(Z)
    # Actually: (1/beta)*log(sum exp(beta*v)) = (1/beta)*log(Z * exp(beta*m)) = m + (1/beta)*log(Z)
    # But Z = sum exp(beta*(v-m)), so this is correct.
    
    # Gradient: grad[j] = dx * sum_s w_s * [(1-h[j+s_signed]) - h[j-s_signed]]
    # where s_signed = index - (n-1)
    # Let's compute directly:
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        ws = w[idx]
        if ws < 1e-15:
            continue
        s = idx - (n - 1)  # the shift
        # Contribution: dx * ws * [(1-h[j+s]) for valid j+s - h[j-s] for valid j-s]
        # First term: for j such that 0 <= j < n and 0 <= j+s < n
        #   j ranges from max(0, -s) to min(n-1, n-1-s)
        # Second term: for j such that 0 <= j < n and 0 <= j-s < n
        #   j ranges from max(0, s) to min(n-1, n-1+s)
        
        j1_start = max(0, -s)
        j1_end = min(n, n - s)  # exclusive
        if j1_end > j1_start:
            grad[j1_start:j1_end] += ws * (1.0 - h[j1_start + s:j1_end + s])
        
        j2_start = max(0, s)
        j2_end = min(n, n + s)  # exclusive, for j-s >= 0 means j >= s, and j-s < n means j < n+s
        if j2_end > j2_start:
            grad[j2_start:j2_end] -= ws * h[j2_start - s:j2_end - s]
    
    grad *= dx
    return f, grad

import numpy as np
from scipy.optimize import minimize
import time

def project_sum_box(x, target):
    """Project x onto {x in [0,1]^n : sum(x) = target}"""
    n = len(x)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(x + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(x + lam, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        mask = (result > 1e-6) & (result < 1.0 - 1e-6)
        cnt = int(np.sum(mask))
        if cnt > 0:
            result[mask] += residual / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def exact_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def overlap_vec(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lse_smooth(v, beta):
    """Compute log-sum-exp(v*beta)/beta and weights."""
    m = v.max()
    ev = np.exp(beta * (v - m))
    Z = ev.sum()
    val = m + np.log(Z) / beta
    w = ev / Z
    return val, w

def lse_gradient(w_vec, h, beta):
    """Gradient of the log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w_vec[idx]
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        # First part: (1-h[j+s]) for j where 0<=j<n and 0<=j+s<n
        a = max(0, -s)
        b = min(n, n - s)
        if b > a:
            g[a:b] += wi * (1.0 - h[a + s:b + s])
        # Second part: -h[j-s] for j where 0<=j<n and 0<=j-s<n
        c = max(0, s)
        d = min(n, n + s)
        if d > c:
