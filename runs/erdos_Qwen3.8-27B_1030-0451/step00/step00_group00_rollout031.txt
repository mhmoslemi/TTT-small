def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def c5_and_corr(h):

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = beta * corr * dx
    mf = np.max(f)
    ef = np.exp(f - mf)
    w = ef / np.sum(ef)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        coeff = wk * dx
        if s == 0:
            grad += coeff * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += coeff * (1.0 - h[s:])
            grad[s:] += coeff * (-h[:n-s])
        else:
            sp = -s
            grad[sp:] += coeff * (1.0 - h[:n-sp])
            grad[:n-sp] += coeff * (-h[sp:])
    return grad

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    f = beta * corr * dx
    return np.max(f) + np.log(np.sum(np.exp(f - np.max(f)))) / beta

            break

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = beta * corr * dx

    ef = np.exp(f - mf)
    Z = ef.sum()
