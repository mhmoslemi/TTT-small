    if diff > 0:

    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:

    h[mask] += diff / count
    h = np.clip(h, 0.0, 1.0)

        else:

def _compute_gradient(h, corr, dx, top_k=30, tau=None):

    n = len(h)
    center = n - 1
    if tau is None:
        max_c = np.max(corr)
        tau = max(0.005 * max_c, 1e-6)

    threshold = max_val - 5 * tau
    active_mask = corr >= threshold
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
