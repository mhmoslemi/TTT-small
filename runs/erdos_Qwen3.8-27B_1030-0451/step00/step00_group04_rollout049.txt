def compute_z(h):
    """Compute all correlation values z_k = np.correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_and_grad(h, tau):
    """Compute smooth max F_tau and its gradient"""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Stable log-sum-exp
    z_max = np.max(z)
    exp_z = np.exp((z - z_max) / tau)
    log_sum = z_max + tau * np.log(np.sum(exp_z))
    F = log_sum
    
    # Softmax weights
    p = exp_z / np.sum(exp_z)
    
    # Compute gradient
    grad = np.zeros(n)
    m = len(z)  # 2n-1
    for k in range(m):
        if p[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # Valid i: 0 <= i <= n-1-lag
            L = n - lag
            # Contribution from h[i+lag] term: for j = i+lag, i = j-lag
            # j ranges from lag to n-1
            grad[lag:] += p[k] * dx * (1.0 - h[:L])
            # Contribution from (1-h[i]) term: for j = i

        else:

            # Contribution from (1-h[i]) term: for j = i
            # j ranges from a to n-1
            grad[a:] += p[k] * dx * (-h[:-a])
    
    return F, grad, z

    #   if 0 < h[i] - lambda < 1: result is h[i] - lambda
    #   if h[i] - lambda < 0: result is 0

import numpy as np
import time

def compute_z(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def evaluate(h):
    return float(np.max(compute_z(h)))

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z_raw = np.correlate(h, 1.0 - h, mode='full')
    z = z_raw * dx
    z_max = z.max()
    exp_z = np.exp((z - z_max) / tau)
    s = exp_z.sum()
    p = exp_z / s
    F = z_max + tau * np.log(s)
    grad = np.zeros(n)
    m = 2 * n - 1
    thr = 1e-9
    for k in range(m):
        if p[k] < thr:
            continue
        w = p[k] * dx
        lag = k - (n - 1)
        if lag >= 0:
            L = n - lag
            grad[lag:] += w * (1.0 - h[:L])
            grad[:L] += w * (-h[lag:])
        else:
            a = -lag
            L = n - a
            grad[:L] += w * (1.0 - h[a:])
            grad[a:] += w * (-h[:L])
    return F, grad, z

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - hp.sum()
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            hp = hp.copy()
            hp[mask] += diff / cnt
            np.clip(hp, 0.0, 1.0, out=hp)
    return hp

def slsqp_polish(h_init, target_sum, max_iter=150, active_k=50):
    n = len(h_init)
    if n > 600:
        return h_init.copy(), evaluate(h_init)
    from scipy.optimize import minimize
    z = compute_z(h_init)
    z_max = z.max()
    t0 = z_max + 0.005
    idx = np.argsort(z)[-active_k:]
    dx = 2.0 / n

    def epi_obj(x):
        return x[-1]

    def epi_obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def epi_con(x):
        h_x = x[:-1]
        t = x[-1]
        zv = compute_z(h_x)
        return t - zv[idx]

    def epi_con_jac(x):
        h_x = x[:-1]
        m = len(idx)
        jac = np.zeros((m, n + 1))
        for i, k in enumerate(idx):
            lag = k - (n - 1)
            if lag >= 0:
                L = n - lag
                jac[i, lag:] += dx * (1.0 - h_x[:L])
                jac[i, :L] += dx * (-h_x[lag:])
            else:
                a = -lag
                L = n - a
                jac[i, :L] += dx * (1.0 - h_x[a:])
                jac[i, a:] += dx * (-h_x[:L])
            jac[i, -1] = 1.0
        return jac

    def eq_con(x):
        return np.sum(x[:-1]) - target_sum

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:-1] = 1.0
        return j

    x0 = np.concatenate([h_init, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': epi_con, 'jac': epi_con_jac},
        {'type': 'eq', 'fun': eq_con, 'jac': eq_jac}
    ]
    try:
        res = minimize(epi_obj, x0, method='SLSQP', jac=epi_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h_out = np.clip(res.x[:-1], 0.0, 1.0)
        h_out = project_box_sum(h_out, target_sum)
        sc = evaluate(h_out)
        return h_out, sc
    except Exception:
        return h_init.copy(), evaluate(h_init)
