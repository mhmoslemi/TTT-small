import numpy as np
import time
import sys

def compute_c5(h):
    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_c5_all(h):
    """Compute all correlate values (times dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def project_box_sum(h, target_sum):
    """Project h onto {x in [0,1]^n : sum(x) = target_sum} using bisection."""
    n = len(h)
    if n == 0:
        return h
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_c5(h, n_active=5):
    """Compute subgradient of C5, averaging over top n_active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    
    # Find top lags
    sorted_idx = np.argsort(corr)[::-1]
    active_idx = sorted_idx[:n_active]
    max_val = corr[active_idx[0]]
    
    # Include all lags within a small tolerance of the max
    tol = 1e-6
    active_mask = corr >= (max_val - tol)
    active_idx = np.where(active_mask)[0]
    # Limit to top n_active if there are too many
    if len(active_idx) > n_active:
        top_vals = np.sort(corr[active_idx])[::-1][:n_active]
        threshold = top_vals[-1]
        active_idx = active_idx[corr[active_idx] >= threshold]
    
    grad = np.zeros(n)
    for j in active_idx:
        k = j - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[k:] += (1.0 - h[:nk])
                grad[:nk] += -h[k:]
        else:
            k2 = -k
            nk = n - k2
            if nk > 0:
                grad[:nk] += (1.0 - h[k2:])
                grad[k2:] += -h[:nk]
    
    grad /= max(len(active_idx), 1)
    return grad

def optimize_subgradient(h, n_iter=5000, step_init=0.01, decay=0.9995, seed=42):
    """Optimize using projected subgradient method."""
    n = len(h)
    target_sum = n / 2.0
    rng = np.random.RandomState(seed)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    step = step_init
    
    for it in range(n_iter):
        if it % 100 == 0:
            # Add small random perturbation to escape local minima
            pert = rng.randn(n) * 0.01
            h = np.clip(h + pert, 0, 1)
            h = project_box_sum(h, target_sum)
        
        grad = subgradient_c5(h)
        h = h - step * grad
        h = project_box_sum(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        step *= decay
    
    return best_h, best_c5

def optimize_balancing(h, n_sweeps=100, seed=42):

                    if diff > 0.01:

                        h[i + k] -= delta
                        h[i] += delta

                    if diff > 0.01:
                        delta = min(diff * 0.1, 0.01)

        c5 = compute_c5(h)
        if c5 < best_c5:

    """Balance mass for a single lag to reduce its correlation."""
    if k == 0:
        return h
    h = h.copy()
    if k >= 0:
        nk = n - k
        # Move from h[i+k] to h[i] where h[i+k] > h[i]
        diff = h[k:] - h[:nk]

        positive = diff > 0
