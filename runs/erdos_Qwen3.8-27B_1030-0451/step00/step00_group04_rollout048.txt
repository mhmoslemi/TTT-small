import numpy as np
import time
from scipy.optimize import minimize

def c5_eval(h):
    """Exact C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_values(h):
    """Return all correlation values (unscaled)."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_obj(h, tau):
    """Compute smooth max of correlation values * dx."""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    logsum = np.log(np.sum(np.exp(shifted)))
    return z_max + tau * logsum, z, z_max

def smooth_max_grad(h, tau, z=None):
    """Compute gradient of smooth max objective."""
    n = len(h)
    dx = 2.0 / n
    if z is None:
        z = np.correlate(h, 1.0 - h, mode='full') * dx
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)  # softmax weights
    
    # Compute gradient: sum_q p_q * dz_q/dh
    grad = np.zeros(n)
    
    for q in range(2 * n - 1):
        if p[q] < 1e-12:
            continue
        ell = q - (n - 1)
        if ell == 0:
            grad += p[q] * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            # For j in [ell, n-1]: contribution (1 - h[j-ell])
            grad[ell:] += p[q] * dx * (1.0 - h[:L])
            # For j in [0, n-1-ell]: contribution -h[j+ell]
            grad[:L] -= p[q] * dx * h[ell:]

            # For j in [a, n-1]: contribution -h[j-a]
            grad[a:] -= p[q] * dx * h[:L]
    
    return grad

def pgd_optimize(h0, tau_start, tau_end, max_iter, budget_func):
    """PGD with smooth max and adaptive step size."""
    n = len(h0)
    target_sum = n / 2.0
    h = project_box_sum(h0, target_sum).copy()
    best_h = h.copy()
    best_score = c5_eval(h)
    
    tau = tau_start
    alpha = 0.1  # initial step size
    alpha_min = 1e-10
    
    F, z, z_max = smooth_max_obj(h, tau)
    
    for it in range(max_iter):
        if budget_func() < 5.0:
            break
        
        grad = smooth_max_grad(h, tau, z)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Try step with backtracking
        improved = False
        for trial in range(20):
            alpha_try = alpha * (0.5 ** trial)
            h_new = project_box_sum(h - alpha_try * grad, target_sum)
            F_new, z_new, _ = smooth_max_obj(h_new, tau)
            if F_new < F - 1e-15:
                h = h_new
                F = F_new
                z = z_new
                improved = True
                break
            alpha = max(alpha * 0.5, alpha_min)
        
        if not improved:
            alpha *= 0.5
            if alpha < alpha_min:
                break
        
        # Check hard objective periodically
        if it % 10 == 0:
            score = c5_eval(h)
            if score < best_score - 1e-12:
                best_score = score
                best_h = h.copy()
        
        # Anneal tau
        if it % 20 == 19 and tau > tau_end:
            tau = max(tau * 0.8, tau_end)
    
    # Final check
    score = c5_eval(h)
    if score < best_score:
        best_score = score
        best_h = h.copy()
    
    return best_h, best_score

import numpy as np
import time
from scipy.optimize import minimize

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def c5_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    z_max = z.max()
    exp_vals = np.exp(np.clip((z - z_max) / tau, -50.0, 0.0))
    p = exp_vals / exp_vals.sum()
    F = z_max + tau * np.log(exp_vals.sum())
    
    grad = np.zeros(n)
    w0 = p[n - 1]
    if w0 > 1e-12:
        grad += w0 * dx * (1.0 - 2.0 * h)
    
    for ell in range(1, n):
        L = n - ell
        wp = p[n - 1 + ell]
        wn = p[n - 1 - ell]
        if wp > 1e-12:
            grad[:L] += wp * dx * (1.0 - h[ell:])
            grad[ell:] -= wp * dx * h[:L]
        if wn > 1e-12:
            grad[ell:] += wn * dx * (1.0 - h[:L])
            grad[:L] -= wn * dx * h[ell:]
    
    return F, grad, z

def pgd_refine(h0, tau_start, tau_end, max_iter, time_limit):
    n = len(h0)
    target = n / 2.0
    h = project_box_sum(h0, target).astype(float)
    best_h = h.copy()
    best_score = c5_eval(h)
    t0 = time.time()
    
    tau = tau_start
    alpha = 0.05
    F, grad, z = smooth_max_and_grad(h, tau)
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if alpha < 1e-14:
            break

        a = alpha

            F_new, _, _ = smooth_max_and_grad(h_new, tau)

                F = F_new

                break

        else:

            if alpha < 1e-14:

                best_h = h.copy()
            _, grad, z = smooth_max_and_grad(h, tau)

import numpy as np
import time
from scipy.optimize import minimize

def c5_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def c5_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    z_max = z.max()
    exp_vals = np.exp(np.clip((z - z_max) / tau, -50.0, 0.0))
    p = exp_vals / exp_vals.sum()
    F = z_max + tau * np.log(exp_vals.sum())
    grad = np.zeros(n)
    w0 = p[n - 1]
    if w0 > 1e-12:
        grad += w0 * dx * (1.0 - 2.0 * h)
    for ell in range(1, n):
        L = n - ell
        wp = p[n - 1 + ell]
        wn = p[n - 1 - ell]
        if wp > 1e-12:
            grad[:L] += wp * dx * (1.0 - h[ell:])
            grad[ell:] -= wp * dx * h[:L]
        if wn > 1e-12:
            grad[ell:] += wn * dx * (1.0 - h[:L])
            grad[:L] -= wn * dx * h[ell:]
    return F, grad, z

def slsqp_refine(h0, time_limit=60):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    t_start = time.time()
    h = project_box_sum(h0.astype(float), target)
    c_init = c5_eval(h)
    z_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    K = min(50, 2 * n - 1)
    top_idx = np.argsort(z_vals)[::-1][:K]
    
    def obj_fn(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        t = x[-1]
        z = np.correlate(h_x, 1.0 - h_x, mode='full') * dx
        return z[top_idx] - t
    
    def ineq_jac(x):
        h_x = x[:n]
        J = np.zeros((K, n + 1))
        for i, q in enumerate(top_idx):
            ell = int(q) - (n - 1)
            if ell == 0:
                J[i, :n] = dx * (1.0 - 2.0 * h_x)
            elif ell > 0:
                L = n - ell
                J[i, :L] += dx * (1.0 - h_x[ell:])
                J[i, ell:] -= dx * h_x[:L]
            else:
                a = -ell
                L = n - a
                J[i, a:] += dx * (1.0 - h_x[:L])
                J[i, :L] -= dx * h_x[a:]
            J[i, n] = -1.0
        return J
    
    def eq_fun(x):
        return x[:n].sum() - target
    
    def eq_jac(x):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    
    x0 = np.concatenate([h, [c_init + 0.002]])

        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    try:
        result = minimize(obj_fn, x0, method='SLSQP', jac=obj_grad,

    except Exception:
        pass
    return h, c_init
