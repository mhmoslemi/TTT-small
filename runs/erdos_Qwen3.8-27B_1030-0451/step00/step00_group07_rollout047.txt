def smooth_max_grad(h, tau, N):
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    c_shifted = np.clip((c - m) / tau, -50, 0)
    exps = np.exp(c_shifted)
    s = np.sum(exps)
    alpha = exps / s  # shape (2N-1,)
    
    grad = np.zeros(N)
    
    for j in range(2*N - 1):
        a = alpha[j]
        if a < 1e-14:
            continue
        k = j - (N - 1)  # lag
        lo = max(0, -k)
        hi = min(N-1, N-1-k)
        if lo > hi:
            continue
        
        # First term: m in [lo+k, hi+k]
        m_lo1 = lo + k
        m_hi1 = hi + k
        if m_lo1 <= m_hi1:
            # grad[m_lo1:m_hi1+1] += dx * a * (1 - h[m_lo1-k : m_hi1-k+1])
            idx = np.arange(m_lo1, m_hi1+1)
            grad[idx] += dx * a * (1.0 - h[idx - k])
        
        # Second term: m in [lo, hi]
        if lo <= hi:
            idx = np.arange(lo, hi+1)
            grad[idx] -= dx * a * h[idx + k]
    
    return grad

        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - np.sum(h)

    N_old = len(h_old)

    old_cent = (np.arange(N_old) + 0.5) / N_old
    new_cent = (np.arange(N_new) + 0.5) / N_new
    h_new = np.interp(new_cent, old_cent, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, N_new)


def smooth_obj_and_grad(h, tau, N):
    dx = 2.0 / N
    c_full = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c_full)
    shifted = np.clip((c_full - m) / tau, -50.0, 0.0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    alpha = exps / s

    grad = np.zeros(N)

        a = alpha[j]
        if a < 1e-14:
            continue
        k = j - (N - 1)
        if k == 0:
            grad += dx * a * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += dx * a * (1.0 - h[:N - k])

        else:
            ka = -k
            grad[:N - ka] += dx * a * (1.0 - h[ka:])
            grad[ka:] -= dx * a * h[:N - ka]
    return val, grad

        v, _ = smooth_obj_and_grad(x, tau, N)
        return v

        _, g = smooth_obj_and_grad(x, tau, N)
        return g

    def cons_fun(x):
        return np.sum(x) - N / 2.0

    def cons_jac(x):
        return np.ones(N)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * N
    res = minimize(func, h0, jac=gradient, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
