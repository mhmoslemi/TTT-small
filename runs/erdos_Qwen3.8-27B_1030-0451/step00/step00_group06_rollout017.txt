def compute_C(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    C = compute_C(h)
    return float(np.max(C) * dx)

def project(h, target_sum):
    """Project onto 0<=h<=1, sum(h)=target_sum"""
    h = np.clip(h, 0.0, 1.0)
    # Bisection to find shift that makes sum correct
    # h' = clip(h + t, 0, 1), find t such that sum(h') = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0, 1).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    return h

def compute_gradient(h, active_indices, weights):
    """Compute gradient of weighted sum of C[m] for active lags"""
    n = len(h)
    grad = np.zeros(n)
    for m, w in zip(active_indices, weights):
        if w < 1e-14:
            continue
        d = m - n + 1
        if d >= 0:

                grad[:n-d] -= w * h[d:]
        else:
            nd = -d
            if nd < n:
                grad[:n-nd] += w * (1.0 - h[nd:])
                grad[nd:] -= w * h[:nd]
    return grad

def optimize_local(h0, n_top=20, budget_s=100.0, n_restart=5):
    """Local optimization using projected gradient with smoothed max"""
    n = len(h0)
    target_sum = n / 2.0
    h = project(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    t0 = time.time()
    step_size = 0.01
    temperature = 0.02

        C = compute_C(h)
        c5_current = np.max(C) * (2.0/n)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
