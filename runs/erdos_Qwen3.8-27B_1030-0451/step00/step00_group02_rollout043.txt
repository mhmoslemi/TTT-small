    """Project y onto {h: 0<=h_i<=1, sum(h_i)=c}"""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - 1  # ensures all clip to 1
    hi = y.max() + 1  # ensures all clip to 0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > c:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    return h

- Terms where j = i+s (i = j-s): contributes t*(1-h[j-s]) if 0≤j-s<n
  Old contribution: h[j]*(1-h[j-s])
  Change: (t-h[j])*(1-h[j-s])
- Terms where j = i (so s = s, and i+s = j+s): contributes h[j+s]*(1-t) if 0≤j+s<n
  Old contribution: h[j+s]*(1-h[j])
  Change: h[j+s]*((1-t)-(1-h[j])) = h[j+s]*(h[j]-t)
