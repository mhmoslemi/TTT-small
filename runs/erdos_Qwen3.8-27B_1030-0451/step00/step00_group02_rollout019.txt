            best_t = t

        if direction == 1:

        cur_h[b] -= direction * t_use
        cur_h = np.clip(cur_h, 0.0, 1.0)

def _smooth_logsumexp(h, n, t_start, time_budget):
    """Optimize smooth log-sum-exp approximation using scipy."""
    from scipy.optimize import minimize
    dx = 2.0 / n
    S = 0.5 * n
    taus = [0.05, 0.02, 0.008, 0.003]
    x0 = h.copy()
    for tau in taus:
        if abs(time.monotonic() - t_start) > t_start + time_budget:
            break
        def obj(x):
            c = dx * np.correlate(x, 1.0 - x, mode="full")
            c = c - np.max(c)
            return tau * np.log(np.sum(np.exp(c / tau))) + tau / tau * 0.0
        # Actually use stable form
        def obj(x):
            c = dx * np.correlate(x, 1.0 - x, mode="full")
            m = np.max(c)
            return m + tau * np.log(np.sum(np.exp((c - m) / tau)))
        def obj_grad(x):
            c = dx * np.correlate(x, 1.0 - x, mode="full")
            m = np.max(c)
            w = np.exp((c - m) / tau)
            w = w / np.sum(w)
            # Gradient of c[s] w.r.t. x
            g = np.zeros(n)
            for s in range(len(c)):
                shift = s - (n - 1)
                ks = np.arange(n)
                m1 = (ks + shift >= 0) & (ks + shift < n)
                g[m1] += w[s] * dx * (1.0 - x[ks[m1] + shift])
                m2 = (ks - shift >= 0) & (ks - shift < n)
                g[m2] -= w[s] * dx * x[ks[m2] - shift]
            return g
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - S,
                        'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success and np.all(np.isfinite(res.x)):
            x_new = _repair(res.x, n)
            new_score = _score(x_new)
            if new_score < _score(x0) - 1e-10:
                x0 = x_new
        else:
            break
    return x0

def _rank_fill(n, scores):
    """Create candidate from scores: top n/2 get value 1, rest get 0 (with fractional)."""
    S = 0.5 * n
    m = int(math.floor(S + 1e-10))
    frac = S - m
    order = np.argsort(scores, kind='stable')[::-1]
    h = np.zeros(n)
    h[order[:m]] = 1.0
    if frac > 1e-10 and m < n:
        h[order[m]] = frac
    return h

def _make_scores(n, kind, phase, rng):
    """Generate score vectors for rank-fill diversity."""
    idx = np.arange(n, dtype=float)
    if kind == 0:
        phi = (1.0 + math.sqrt(5.0)) / 2.0
        return np.abs(np.sin(idx * 0.381966 + phase) * phi)
    elif kind == 1:
        return np.abs(np.sin(idx * 0.618034 + phase)) + 0.1 * np.sin(idx * 1.7 + phase * 2)
    elif kind == 2:
        return np.cos(idx * 0.318310 * math.pi / 2.0 + phase) + 0.3 * np.sin(idx * 0.1 + phase)
    elif kind == 3:
        return rng.gauss(0, 1) + 0.5 * np.sin(idx * 0.2 + phase)
    elif kind == 4:

        stride = max(1, n // 7)
        return np.sin(idx * stride * 0.1 + phase) + 0.2 * np.cos(idx * 0.3)
    elif kind == 5:

    else:
        return np.sin(idx * (kind + 1) * 0.13 + phase) + rng.gauss(0, 0.5)

def _transfer(h, old_n, new_n):
    """Linearly interpolate h from old_n to new_n grid."""
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(new_x, old_x, h)
    return _repair(new_h, new_n)

def _verify(h, n):

            best_h = h1.copy()

    # Phase 2: SA from best
    remaining = budget * 0.93 - (time.monotonic() - t_start)
    if remaining > 5:
        t_alloc = remaining * 0.30
        h2, s2 = _sa_search(best_h.copy(), n_base, t_alloc, time.monotonic())
        if s2 < best_score - 1e-13:
            best_score = s2
            best_h = h2.copy()

    # Phase 3: More local search from SA result
    remaining = budget * 0.93 - (time.monotonic() - t_start)
    if remaining > 5:
        t_alloc = remaining * 0.25
        h3, s3 = _local_search(best_h.copy(), n_base, t_alloc, time.monotonic())
        if s3 < best_score - 1e-13:
            best_score = s3
            best_h = h3.copy()

    # Phase 4: Try rank-fill diverse candidates
    remaining = budget * 0.93 - (time.monotonic() - t_start)
    if remaining > 3:
        for kind in range(6):
            for phase in [0.0, 0.3, 0.7, 1.1]:
                if abs(time.monotonic() - t_start) > safe_deadline:
                    break
                scores = _make_scores(n_base, kind, phase, rng)
                cand = _rank_fill(n_base, scores)
                sc = _score(cand)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_h = cand.copy()
                # Quick local search from this candidate (very short)
                if sc < best_score + 0.05:
                    t_sm = min(3.0, remaining * 0.03)
                    h_r, s_r = _local_search(cand, n_base, t_sm, time.monotonic())
                    if s_r < best_score - 1e-13:
                        best_score = s_r
                        best_h = h_r.copy()
            if abs(time.monotonic() - t_start) > safe_deadline:
                break

    # Phase 5: Try different dimensions
    remaining = budget * 0.93 - (time.monotonic() - t_start)
    if remaining > 10:
        for factor in [0.75, 1.25, 1.5, 0.625]:
            if abs(time.monotonic() - t_start) > safe_deadline:
                break
            new_n = max(32, int(round(n_base * factor)))
            if new_n > 2500 or new_n == n_base:
                continue
            # Make sure new_n is even for exact sum
            if new_n % 2 != 0:
                new_n += 1
            cand = _transfer(best_h, n_base, new_n)
            sc = _score(cand)
            if abs(time.monotonic() - t_start) > safe_deadline:
                break
            t_sm = min(20.0, remaining * 0.15)
            h_t, s_t = _local_search(cand, new_n, t_sm, time.monotonic())
            if s_t < best_score - 1e-13:
                best_score = s_t
                best_h = h_t.copy()

    # Phase 6: Final polish - one more local search
    remaining = budget * 0.93 - (time.monotonic() - t_start)
    if remaining > 3:
        h_f, s_f = _local_search(best_h.copy(), len(best_h), remaining, time.monotonic())
        if s_f < best_score - 1e-13:
            best_score = s_f
            best_h = h_f.copy()

    # Final verification
    final_n = len(best_h)
    final_h = _repair(best_h, final_n)
    final_score = _score(final_h)
    if final_score > best_score + 1e-10:
        # Repair made it worse, use original clipped
        final_h = np.clip(best_h, 0.0, 1.0)
        final_score = _score(final_h)

    return (final_h, float(final_score), int(final_n))
