One more thought: the gradient computation can be vectorized more efficiently. Instead of looping over active lags, I can compute the full gradient for each lag in a vectorized way.

For lag L >= 0:
g[i] += (1 - h[i+L]) for i in 0..n-1-L
g[i+L] -= h[i] for i in 0..n-1-L

This is: g[:n-L] += (1 - h[L:]) and g[L:] -= h[:n-L]

For lag L < 0 (let Lp = -L):
g[i] += (1 - h[i-Lp]) for i in Lp..n-1
g[i-Lp] -= h[i] for i in Lp..n-1

This is: g[Lp:] += (1 - h[:n-Lp]) and g[:n-Lp] -= h[Lp:]

So for each lag, it's just two vector operations! Very fast.

import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, T):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > T:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    r = T - h.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += r / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, n, active_frac=0.97):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    if max_val <= 0:
        return np.zeros(n), corr
    thresh = max_val * active_frac
    active = np.where(corr >= thresh)[0]
    g = np.zeros(n)
    for m in active:
        L = m - (n - 1)
        if L >= 0:
            g[:n - L] += 1.0 - h[L:]
            g[L:] -= h[:n - L]
        else:
            Lp = -L
            g[Lp:] += 1.0 - h[:n - Lp]
            g[:n - Lp] -= h[Lp:]
    return g, corr

def try_move(h, i, j, best_score):
    max_tau = min(h[i], 1.0 - h[j])
    if max_tau < 1e-10:
        return None, best_score
    best_tau = 0.0
    best_s = best_score
    fracs = (0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 1.0)
    for f in fracs:
        t = max_tau * f
        h2 = h.copy()
        h2[i] -= t
        h2[j] += t
        s = exact_score(h2)
        if s < best_s:
            best_s = s
            best_tau = t
    if best_tau > 0:
        lo_t = max_tau * max(0.0, (best_tau / max_tau) - 0.1)
        hi_t = max_tau * min(1.0, (best_tau / max_tau) + 0.1)
        for _ in range(4):
            mid_t = 0.5 * (lo_t + hi_t)
            h2 = h.copy()
            h2[i] -= mid_t
            h2[j] += mid_t
            s_mid = exact_score(h2)
            h_left = h.copy()
            h_left[i] -= lo_t
            h_left[j] += lo_t
            s_left = exact_score(h_left)
            if s_mid < s_left:
                lo_t = mid_t
            else:
                hi_t = mid_t
        t_final = 0.5 * (lo_t + hi_t)
        h2 = h.copy()
        h2[i] -= t_final
        h2[j] += t_final
        s_final = exact_score(h2)
        if s_final < best_s:
            best_s = s_final
            best_tau = t_final
        if best_tau > 0:
            h2 = h.copy()
            h2[i] -= best_tau
            h2[j] += best_tau
            return h2, best_s
    return None, best_score

        h += rng.normal(0, 0.05, n)
