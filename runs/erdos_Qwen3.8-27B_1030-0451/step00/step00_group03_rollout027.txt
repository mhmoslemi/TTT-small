import time

def project_onto_feasible(y, target_sum):
    """Project y onto {h: 0<=h_i<=1, sum(h)=target_sum}"""
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(y)
    # Bisection on alpha: find alpha s.t. sum(clip(y+alpha, 0, 1)) = target_sum
    a_lo = -y.max() - 1.0  # ensures sum = 0
    a_hi = 2.0 - y.min()   # ensures sum = n
    
    for _ in range(80):
        a_mid = 0.5 * (a_lo + a_hi)
        s = np.clip(y + a_mid, 0.0, 1.0).sum()
        if s < target_sum:
            a_lo = a_mid
        else:
            a_hi = a_mid
    
    h = np.clip(y + 0.5*(a_lo + a_hi), 0.0, 1.0)
    
    # Fix floating point residual
    res = target_sum - h.sum()
    if abs(res) > 1e-13:
        if res > 0:
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                s = min(room[i], res)
                if s > 1e-15:
                    h[i] += s
                    res -= s
                if abs(res) < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                s = min(h[i], -res)
                if s > 1e-15:
                    h[i] -= s
                    res += s
                if abs(res) < 1e-14:
                    break
    return h

def evaluate_c5(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx approx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    """Get all correlation values times dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_gradient(h, m, n):
    """Compute gradient of f_m with respect to h"""
    dx = 2.0 / n
    k = m - (n - 1)
    j_lo = max(0, -k)
    j_hi = min(n - 1, n - 1 - k)
    g = np.zeros(n)
    
    # First term: p in [j_lo+k, j_hi+k], contribution dx*(1-h[p-k])
    p1s = max(0, j_lo + k)
    p1e = min(n - 1, j_hi + k)
    if p1s <= p1e:
        g[p1s:p1e+1] = dx * (1.0 - h[p1s-k:p1e-k+1])
    
    # Second term: p in [j_lo, j_hi], contribution -dx*h[p+k]
    if j_lo <= j_hi:
        g[j_lo:j_hi+1] += -dx * h[j_lo+k:j_hi+k+1]
    
    return g

def slsqp_refine(h0, n_active, max_iter=200, t_tol=1e-6):
    """SLSQP refinement with active lag constraints"""
    n = len(h0)
    target_sum = 0.5 * n
    dx = 2.0 / n
    
    f_vals = get_corr(h0)
    c5_0 = f_vals.max()
    
    # Select active lags
    active_idx = np.argsort(f_vals)[-n_active:]
    active_idx = sorted(active_idx)
    
    # Initial t
    t_init = c5_0 + 0.001
    
    # Variables: [h_0, ..., h_{n-1}, t]
    x0 = np.concatenate([h0, [t_init]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def obj(x):
        return x[-1]
    
    def obj_jac(x):
        j = np.zeros(n + 1)
        j[-1] = 1.0
        return j
    
    # Equality constraint: sum(h) - target_sum = 0
    def eq_fun(x):
        return x[:n].sum() - target_sum
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    # Inequality constraints: t - f_m(h) >= 0 for each active m
    def ineq_funs(x):
        h = x[:n]
        t = x[-1]
        f = get_corr(h)
        return t - f[active_idx]
    
    def ineq_jacs(x):
        h = x[:n]
        # For each active m: derivative of (t - f_m) w.r.t. [h, t]
        # = [-grad_f_m, 1]
        jac_list = []
        for m in active_idx:
            g_fm = lag_gradient(h, m, n)
            row = np.zeros(n + 1)
            row[:n] = -g_fm
            row[-1] = 1.0
            jac_list.append(row)
        return np.array(jac_list)
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_funs, 'jac': ineq_jacs}
    ]
    
    res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-8})
    
    if res.success or res.fun < t_init:
        h_new = project_onto_feasible(res.x[:n], target_sum)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_0:
            return h_new, c5_new
    
    return h0, c5_0

def lag_gradient(h, m, n):
    dx = 2.0 / n
    k = m - (n - 1)
    j_lo = max(0, -k)
    j_hi = min(n - 1, n - 1 - k)
    g = np.zeros(n)
    p1s = max(0, j_lo + k)
    p1e = min(n - 1, j_hi + k)
    if p1s <= p1e:
        g[p1s:p1e + 1] = dx * (1.0 - h[p1s - k:p1e - k + 1])
    if j_lo <= j_hi:
        g[j_lo:j_hi + 1] += -dx * h[j_lo + k:j_hi + k + 1]
    return g

    f_a = f_vals[active_idx]
    fmax_a = f_a.max()
    z = (f_a - fmax_a) / tau
    z = z - z.max()
    exp_z = np.exp(z)
    w = exp_z / exp_z.sum()
    grad = np.zeros(n)
    for i, m in enumerate(active_idx):
        wi = w[i]
        if wi < 1e-10:
            continue
        grad += wi * lag_gradient(h, m, len(h), )
    return grad

        gnorm = np.linalg.norm(grad)

        best_trial = h

        for _ in range(25):

            h_trial = project_onto_feasible(y, target_sum)
            c5_trial = evaluate_c5(h_trial)

        c5_new = evaluate_c5(h_new)
        if c5_new < c5_0 - 1e-10:
            return h_new, c5_new
    except Exception:
        pass
    return h0, c5_0


def upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation, then project."""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_onto_feasible(h_new, 0.5 * n_new)


def make_random_candidate(n, rng):
    """Generate a random feasible candidate."""
    h = rng.uniform(0.2, 0.8, size=n)
    return project_onto_feasible(h, 0.5 * n)


def make_binary_candidate(n, rng):
    """Generate a binary-ish feasible candidate."""
    h = np.zeros(n)
    k = int(round(0.5 * n))
    idx = rng.choice(n, size=k, replace=False)
    h[idx] = 1.0
    return project_onto_feasible(h, 0.5 * n)

    except (NameError, UnboundLocalError):
        h_init = np.full(64, 0.5)

    n_init = len(h_init)
    h_init = project_onto_feasible(h_init, 0.5 * n_init)
    c5_init = evaluate_c5(h_init)

    best_h = h_init.copy()
    best_c5 = c5_init
    best_n = n_init

    # Phase 1: Projected GD at multiple resolutions
    resolutions = sorted(set([n_init, 96, 128, 160, 192, 256]))
    resolutions = [r for r in resolutions if r <= 256]

    for n_res in resolutions:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 15:
            break
        gd_time = min(50.0, remaining * 0.35)

        # Start from upsampled parent
        h_start = upsample(h_init, n_res)
        tau0 = 0.004
        h_new, c5_new = projected_gd(h_start, tau0, 3000, gd_time, deadline)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_res

        # Also try from upsampled best if different resolution
        if n_res != best_n and time.time() < deadline:
            h_start2 = upsample(best_h, n_res)
            gd_time2 = min(30.0, (deadline - time.time()) * 0.3)
            if gd_time2 > 5:
                h_new2, c5_new2 = projected_gd(h_start2, 0.003, 2000, gd_time2, deadline)
                if c5_new2 < best_c5 - 1e-10:
                    best_c5 = c5_new2
                    best_h = h_new2.copy()
                    best_n = n_res

            break

                best_h = h_new.copy()

                best_h = h_new.copy()
