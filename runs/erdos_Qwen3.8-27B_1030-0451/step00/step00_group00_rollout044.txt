import numpy as np

def project_onto_constraint(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def gradient_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    obj = c_max + np.log(w_sum) / alpha
    A = np.convolve(1.0 - h, w, mode='full')[n - 1:2 * n - 1]
    B = np.convolve(h, w[::-1], mode='full')[n - 1:2 * n - 1]
    grad = dx * (A - B)
    return obj, grad
