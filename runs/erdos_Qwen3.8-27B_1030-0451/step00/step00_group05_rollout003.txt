def repair_profile(arr):
    n = len(arr)
    h = np.array(arr, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-14:
            h += diff * room / total_room
        else:
            h += diff / n
    else:
        total = np.sum(h)
        if total > 1e-14:
            h += diff * h / total
        else:
            h += diff / n
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        idx = np.argmax(1.0 - h) if residual > 0 else np.argmax(h)
        h[idx] += residual
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

    n = len(h)

    vals = corr * dx

        idx = np.argpartition(vals, -top_k)[-top_k:]
        mask = np.zeros(len(vals), dtype=bool)
        mask[idx] = True
        vals_m = np.where(mask, vals, -np.inf)
    else:
        vals_m = vals
    vals_shift = vals_m - np.max(vals_m)
    weights = np.exp(alpha * vals_shift)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros_like(vals)
        weights[np.argmax(vals)] = 1.0
    else:
        weights /= w_sum
    grad = np.zeros(n)

        final_c5 = best_c5
        best_h = repair_profile(best_h)
        final_c5, ok = evaluate_profile(best_h)
        if not ok:
            best_h = construct_seed(131, 0, rng)
            final_c5, _ = evaluate_profile(best_h)

    return best_h, final_c5, len(best_h)
