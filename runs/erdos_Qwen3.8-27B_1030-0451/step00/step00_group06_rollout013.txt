def project_to_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    lo, hi = 0.0, 1.0
    target = target_sum
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0, 1)
