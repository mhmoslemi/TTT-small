        valid1 = (m_range - offset) >= 0
        valid1 &= (m_range - offset) < n
        g[valid1] += (1.0 - h[m_range[valid1] - offset])
        # Term 2: -h[m + offset] for m where 0 <= m + offset < n
        valid2 = (m_range + offset) >= 0
        valid2 &= (m_range + offset) < n
        g[valid2] -= h[m_range[valid2] + offset]
        grads[k] = g * dx
    
    return f, grads

def solve_lp(h0, rho, f_vals, grads):
    n = len(h0)
    num_f = len(f_vals)
    
    # Variables: [d_0, ..., d_{n-1}, tau], total n+1
    # Objective: min tau
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints:
    # For each k: tau - grads[k] @ d >= f_vals[k]
    # => grads[k] @ d - tau <= -f_vals[k]
    
    A_parts = []
    b_parts = []
    
    # Epigraph constraints
    A_ep = np.zeros((num_f, n + 1))
    A_ep[:, :n] = grads  # each row is grads[k]
    A_ep[:, -1] = -1.0
    A_parts.append(A_ep)
    b_parts.append(-f_vals)
    
    # Box constraints on d:
    # d[i] <= min(rho, 1 - h0[i])
    # -d[i] <= min(rho, h0[i])  i.e. d[i] >= -min(rho, h0[i])
    
    d_upper = np.minimum(rho, 1.0 - h0)
    d_lower = np.minimum(rho, h0)
    
    A_box = np.zeros((2*n, n + 1))
    A_box[:n, np.arange(n)] = 1.0  # d[i] <= d_upper[i]
    A_box[n:, np.arange(n)] = -1.0  # -d[i] <= d_lower[i]
    A_parts.append(A_box)
    b_parts.append(np.concatenate([d_upper, d_lower]))
    
    A_ub = np.vstack(A_parts)
    b_ub = np.concatenate(b_parts)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def repair_feasibility(h, n):
    """Ensure 0 <= h <= 1 and sum(h) = n/2"""
    h = np.clip(h, 0.0, 1.0)

    n = len(h)

def compute_obj_grads(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    nf = 2 * n - 1
    grads = np.zeros((nf, n))
    idx = np.arange(n)
    for k in range(nf):
        off = k - (n - 1)

        m1 = (v1 >= 0) & (v1 < n)
        g = np.zeros(n)
        g[m1] += 1.0 - h[v1[m1]]
        v2 = idx + off
        m2 = (v2 >= 0) & (v2 < n)
        g[m2] -= h[v2[m2]]
