    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    return np.interp(x_new, x_old, h)

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')

    m = np.max(c)
    shifted = (c - m) / tau
    shifted = np.clip(shifted, -500, 500)

    sum_exp = np.sum(exp_shifted)
    S = m + tau * np.log(sum_exp)
    alpha = exp_shifted / sum_exp

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    return np.interp(x_new, x_old, h)

def project_feasible(h, N):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, tau):

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = (c - m) / tau
    shifted = np.clip(shifted, -600.0, 600.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    S = m + tau * np.log(sum_exp)
    alpha = exp_s / sum_exp
    grad = np.zeros(N)
    center = N - 1
    for k in range(len(c)):
        w = alpha[k]
        if w < 1e-14:
            continue
        lag = k - center
        if lag >= 0:
            n_valid = N - lag
            grad[:n_valid] += w * (1.0 - h[lag:lag + n_valid])
            grad[lag:lag + n_valid] -= w * h[:n_valid]
        else:
            l = -lag
            n_valid = N - l
            grad[l:l + n_valid] += w * (1.0 - h[:n_valid])
            grad[:n_valid] -= w * h[l:l + n_valid]
    grad *= dx
    return S, grad

    def obj(h):
        S, _ = smooth_max_and_grad(h, tau)
        return S
    return obj

def build_grad(tau):
    def grad(h):
        _, g = smooth_max_and_grad(h, tau)
        return g
    return grad

    target = N / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(N)}]
    return cons

def slsqp_optimize(h0, N, tau, maxiter=200, ftol=1e-12):
    target = N / 2.0
    bounds = [(0.0, 1.0)] * N
    cons = make_constraints(N)
    def obj(x):
        S, _ = smooth_max_and_grad(x, tau)
        return S
    def jac(x):
        _, g = smooth_max_and_grad(x, tau)
        return g
    res = minimize(fun=obj, x0=h0, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res
