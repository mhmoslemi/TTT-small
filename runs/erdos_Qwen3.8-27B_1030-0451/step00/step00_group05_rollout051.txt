import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
except Exception:
    _scipy_minimize = None

_GLOBAL_N = 0
_GLOBAL_T = 0.05
_GLOBAL_STOP_AT = 0.0
_PAIR_FRACS = (0.0, 1.0, 0.5, 0.25, 0.75, 0.2, 0.8)


def _stop_cb(xk):
    return time.time() > _GLOBAL_STOP_AT


def _sigmoid(x):
    x = np.asarray(x, dtype=float)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def _round_feasible(x, target):
    n = x.size
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > float(n):
        target = float(n)
    flo = int(np.floor(target + 1e-10))
    fr = target - float(flo)
    if fr > 1.0 - 1e-10:
        flo += 1
        fr = 0.0
    if flo < 0:
        flo = 0
    if flo > n:
        flo = n
    order = np.argsort(-x)
    out = np.zeros(n, dtype=float)
    if flo > 0:
        out[order[:flo]] = 1.0
    if fr > 1e-10 and flo < n:
        rem = order[flo:n]
        if rem.size > 0:
            idx = rem[int(np.argmin(np.abs(x[rem] - fr)))]
            out[idx] = fr
    return out


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float).reshape(-1)
    n = v.size
    if n == 0:
        return v
    target = float(target)
    if target < 0.0:
        target = 0.0
    elif target > n:
        target = float(n)

    if not np.all(np.isfinite(v)):
        return np.full(n, target / float(n), dtype=float)

    t = float(np.mean(v) - target / float(n))
    x = np.clip(v - t, 0.0, 1.0)
    s = float(x.sum())

    if abs(s - target) > 1e-7:
        lo = float(np.min(v)) - 1.0
        hi = float(np.max(v)) + 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            x = np.clip(v - mid, 0.0, 1.0)
            s = float(x.sum())
            if s > target:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(8):
        delta = target - float(x.sum())
        if abs(delta) <= 1e-14:
            break
        if delta > 0.0:
            mask = x < 1.0 - 1e-15
            if not np.any(mask):
                break
            cap = 1.0 - x[mask]
            cap_sum = float(cap.sum())
            if cap_sum <= 1e-15:
                break
            x[mask] += cap * (delta / cap_sum)
            x = np.clip(x, 0.0, 1.0)
        else:
            mask = x > 1e-15
            if not np.any(mask):
                break
            cap = x[mask]
            cap_sum = float(cap.sum())
            if cap_sum <= 1e-15:
                break
            x[mask] -= cap * ((-delta) / cap_sum)
            x = np.clip(x, 0.0, 1.0)

    if abs(target - float(x.sum())) > 1e-12:
        x = _round_feasible(x, target)
    return x


def resample_h(h_in, n_out):
    h_in = np.asarray(h_in, dtype=float).reshape(-1)
    n_in = h_in.size
    n_out = int(n_out)
    if n_out <= 0:
        return np.zeros(0, dtype=float)
    if n_in == 0:
        h = np.full(n_out, 0.5, dtype=float)
        return project_box_sum(h, 0.5 * n_out)
    if n_in == n_out:
        return h_in.copy()
    if not np.all(np.isfinite(h_in)):
        h_in = np.full(n_in, 0.5, dtype=float)
    x_in = (np.arange(n_in) + 0.5) * (2.0 / float(n_in))
    x_out = (np.arange(n_out) + 0.5) * (2.0 / float(n_out))
    h = np.interp(x_out, x_in, h_in, left=h_in[0], right=h_in[-1])
    return project_box_sum(h, 0.5 * n_out)


def repair_array(h, n=None):
    h = np.asarray(h, dtype=float).reshape(-1)
    if n is None:
        n = h.size
    else:
        n = int(n)
        if h.size == 0:
            h = np.full(n, 0.5, dtype=float)
        elif h.size != n:
            h = resample_h(h, n)
    return project_box_sum(h, 0.5 * n)


def c5_only(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float("inf")
    if not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def exact_smooth_value(h, T):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float("inf"), float("inf")
    dx = 2.0 / float(n)
    C = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(C.max())
    if T <= 1e-10:
        return m, m
    shifted = np.clip((C - m) / T, -745.0, 0.0)
    s = float(np.exp(shifted).sum())
    if s <= 0.0 or not np.isfinite(s):
        return m, m
    return m, float(m + T * np.log(s))


def gradient_from_p(h, p):
    n = h.size
    dx = 2.0 / float(n)
    G = np.zeros(n, dtype=float)
    if n == 0:
        return G
    c0 = n - 1
    pdx = p * dx
    G += pdx[c0] * (1.0 - 2.0 * h)
    for R in range(1, n):
        wp = pdx[c0 + R]
        if wp > 0.0:
            G[:n - R] -= wp * h[R:]
            G[R:] += wp * (1.0 - h[:n - R])
        wn = pdx[c0 - R]
        if wn > 0.0:
            G[:n - R] += wn * (1.0 - h[R:])
            G[R:] -= wn * h[:n - R]
    return G


def smooth_value_grad(h, T):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float("inf"), np.zeros(0, dtype=float)
    if not np.all(np.isfinite(h)):
        return float("inf"), np.zeros(n, dtype=float)

    dx = 2.0 / float(n)
    C = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(C.max())

    if T <= 1e-10:
        active = np.where(C >= m - 1e-12)[0]
        p = np.zeros(C.size, dtype=float)
        if active.size > 0:
            p[active] = 1.0 / float(active.size)
        else:
            p[0] = 1.0
        F = m
    else:
        shifted = np.clip((C - m) / T, -745.0, 0.0)
        e = np.exp(shifted)
        s = float(e.sum())
        if s <= 0.0 or not np.isfinite(s):
            p = np.zeros(C.size, dtype=float)
            p[int(np.argmax(C))] = 1.0
            F = m
        else:
            p = e / s
            F = float(m + T * np.log(s))

    if not np.all(np.isfinite(p)):
        p = np.zeros(C.size, dtype=float)
        p[int(np.argmax(C))] = 1.0
    g = gradient_from_p(h, p)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=float)
    return float(F), g


def z_from_h(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    eps = 1e-12
    hc = np.clip(h, eps, 1.0 - eps)
    return np.log(hc / (1.0 - hc))


def h_from_z(z, n):
    target = 0.5 * float(n)
    z = np.asarray(z, dtype=float).reshape(-1)
    if z.size != n:
        z = np.resize(z, n)
    if not np.all(np.isfinite(z)):
        return project_box_sum(np.full(n, 0.5, dtype=float), target)

    zm = float(np.max(np.abs(z))) if z.size else 0.0
    lo = -zm - 40.0
    hi = zm + 40.0

    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(_sigmoid(z + mid).sum())
        if s > target:
            hi = mid
        else:
            lo = mid

    tau = 0.5 * (lo + hi)
    h = _sigmoid(z + tau)
    return project_box_sum(h, target)


def _sm_fun(z):
    n = _GLOBAL_N
    z = np.asarray(z, dtype=float).reshape(-1)
    if z.size != n:
        return 1e100
    h = h_from_z(z, n)
    val, _ = smooth_value_grad(h, _GLOBAL_T)
    if not np.isfinite(val):
        return 1e100
    return float(val)


def _sm_jac(z):
    n = _GLOBAL_N
    z = np.asarray(z, dtype=float).reshape(-1)
    if z.size != n:
        return np.zeros(n, dtype=float)
    h = h_from_z(z, n)
    _, g = smooth_value_grad(h, _GLOBAL_T)
    if not np.all(np.isfinite(g)):
        return np.zeros(n, dtype=float)
    S = h * (1.0 - h)
    ss = float(S.sum())
    if ss <= 1e-14:
        return np.zeros(n, dtype=float)
    return S * (g - float(np.dot(g, S) / ss))


def lbfgs_optimize(h_in, n, time_limit):
    global _GLOBAL_N, _GLOBAL_T, _GLOBAL_STOP_AT
    h = repair_array(h_in, n)
    best = h.copy()
    best_m = c5_only(best)
    t0 = time.time()

    if time_limit < 5.0 or _scipy_minimize is None or n > 450:
        return best, best_m

    if time_limit > 180.0:
        T_list = (0.05, 0.02, 0.01, 0.005, 0.002, 0.001)
    elif time_limit > 80.0:
        T_list = (0.05, 0.02, 0.01, 0.005)
    elif time_limit > 35.0:
        T_list = (0.02, 0.01, 0.005)
    else:
        T_list = (0.01, 0.005)

    z = z_from_h(best)
    bounds = [(-80.0, 80.0)] * n

    for idx, T in enumerate(T_list):
        if time.time() - t0 >= time_limit:
            break
        remaining = time_limit - (time.time() - t0)
        if remaining <= 2.0:
            break
        stage_time = min(remaining / float(max(1, len(T_list) - idx)), 50.0)
        if stage_time <= 1.0:
            break
        _GLOBAL_N = n
        _GLOBAL_T = T
        _GLOBAL_STOP_AT = time.time() + stage_time
        maxiter = int(max(15, min(120, stage_time * 3.0)))
        try:
            res = _scipy_minimize(
                _sm_fun,
                z,
                jac=_sm_jac,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxiter,
                    "maxfun": maxiter * 6,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "callback": _stop_cb,
                },
            )
            if res is not None and hasattr(res, "x"):
                x_new = np.asarray(res.x, dtype=float).reshape(-1)
                if x_new.size == n and np.all(np.isfinite(x_new)):

                    m_new = c5_only(h_new)
                    if m_new < best_m:
                        best = h_new
                        best_m = m_new
                    if m_new < best_m + 5e-5:
                        z = z_from_h(h_new)
                    else:
                        z = z_from_h(best)
        except Exception:
            pass

    best = repair_array(best, n)
    best_m = c5_only(best)
    return best, best_m


def directional_polish(h_in, target, rng, time_limit):
    h = repair_array(h_in, h_in.size)
    n = h.size
    m = c5_only(h)
    best = h.copy()
    best_m = m
    t0 = time.time()

        no_imp = 0

                break
            F0, g = smooth_value_grad(h, T)
            D = -g
            D -= float(np.mean(D))
            nd = float(np.linalg.norm(D))
            if nd < 1e-11:

            Dn = D / nd
            s = step
            accepted = False

            for _ in range(8):
                h_try = project_box_sum(h - s * Dn, target)
                mt, F_try = exact_smooth_value(h_try, T)
                if (mt < m - 1e-12) or (F_try < F0 - 1e-8 and mt < m + 2e-6):
                    h = h_try
                    m = mt
                    accepted = True
                    if mt < best_m:
                        best = h.copy()
                        best_m = mt
                    break
                s *= 0.6

            if accepted:
                no_imp = 0
                step = min(step * 1.05, 0.15)
                if m > best_m + 5e-5:
                    h = best.copy()
                    m = best_m
                    step = 0.02
            else:
                no_imp += 1
                step *= 0.8
                if no_imp % 25 == 0:
                    for _ in range(30):
                        scale = max(0.002, step * 0.8)
                        pert = rng.normal(0.0, scale, n)
                        pert -= float(np.mean(pert))
                        h_try = project_box_sum(h + pert, target)
                        mt = c5_only(h_try)
                        if mt < m - 1e-12:
                            h = h_try
                            m = mt
                            if mt < best_m:
                                best = h.copy()
                                best_m = mt
                            no_imp = 0
                            break
                if no_imp > 220:
                    break

    best = repair_array(best, n)
    best_m = c5_only(best)
    return best, best_m


def pairwise_exhaustive_polish(h_in, target, rng, time_limit):
    h = repair_array(h_in, h_in.size)
    n = h.size
    if n < 2:
        m = c5_only(h)
        return h, m
    m = c5_only(h)
    best = h.copy()
    best_m = m
    t0 = time.time()
    fracs = _PAIR_FRACS

    while time.time() - t0 < time_limit:
        improved_pass = False
        pairs_checked = 0

        for i in range(n - 1):
            for j in range(i + 1, n):
                pairs_checked += 1
                if pairs_checked % 128 == 0 and time.time() - t0 >= time_limit:
                    best = repair_array(best, n)
                    best_m = c5_only(best)
                    return best, best_m

                a = h[i]
                b = h[j]
                low = max(-a, b - 1.0)
                high = min(1.0 - a, b)
                if high - low <= 1e-13:
                    continue

                orig_i = a
                orig_j = b
                span = high - low
                accepted = False

                for frac in fracs:
                    delta = low + frac * span
                    if abs(delta) < 1e-13:
                        continue
                    h[i] = a + delta
                    h[j] = b - delta
                    if h[i] < 0.0:
                        h[i] = 0.0
                    elif h[i] > 1.0:
                        h[i] = 1.0
                    if h[j] < 0.0:
                        h[j] = 0.0
                    elif h[j] > 1.0:
                        h[j] = 1.0

                    mt = c5_only(h)
                    if mt < m - 1e-12:
                        m = mt
                        accepted = True
                        improved_pass = True
                        if mt < best_m:
                            best_m = mt
                            best = h.copy()
                        break

                if not accepted:
                    h[i] = orig_i
                    h[j] = orig_j

            if time.time() - t0 >= time_limit:
                break

        h = project_box_sum(h, target)
        mh = c5_only(h)
        if mh < best_m:
            best = h.copy()
            best_m = mh
            m = mh
        else:
            if mh > best_m + 1e-9:
                h = best.copy()
                m = best_m
            else:
                m = mh

        if not improved_pass:
            break

    best = repair_array(best, n)
    best_m = c5_only(best)
    return best, best_m


def pairwise_random_polish(h_in, target, rng, time_limit):
    h = repair_array(h_in, h_in.size)
    n = h.size
    if n < 2:
        m = c5_only(h)
        return h, m
    m = c5_only(h)
    best = h.copy()
    best_m = m
    t0 = time.time()
    fracs = _PAIR_FRACS
    no_imp = 0

    while time.time() - t0 < time_limit and no_imp < 25000:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        a = h[i]
        b = h[j]
        low = max(-a, b - 1.0)
        high = min(1.0 - a, b)
        if high - low <= 1e-13:
            no_imp += 1
            continue

        orig_i = a
        orig_j = b
        span = high - low
        accepted = False

        for frac in fracs:
            delta = low + frac * span
            if abs(delta) < 1e-13:
                continue
            h[i] = a + delta
            h[j] = b - delta
            if h[i] < 0.0:
                h[i] = 0.0
            elif h[i] > 1.0:
                h[i] = 1.0
            if h[j] < 0.0:
                h[j] = 0.0
            elif h[j] > 1.0:
                h[j] = 1.0

            mt = c5_only(h)
            if mt < m - 1e-12:
                m = mt
                accepted = True
                if mt < best_m:
                    best_m = mt
                    best = h.copy()
                break

        if accepted:
            no_imp = 0
        else:
            h[i] = orig_i
            h[j] = orig_j
            no_imp += 1

        if no_imp % 2000 == 0:
            h = project_box_sum(h, target)
            mh = c5_only(h)
            if mh < best_m:
                best = h.copy()
                best_m = mh
                m = mh
            elif mh > best_m + 1e-9:
                h = best.copy()
                m = best_m
            else:
                m = mh

    best = repair_array(best, n)
    best_m = c5_only(best)
    return best, best_m


def optimize_length(h_in, n, alloc, rng):
    t0 = time.time()
    n = int(n)
    h = repair_array(h_in, n)
    target = 0.5 * float(n)
    m = c5_only(h)
    best = h.copy()
    best_m = m

    quick_alloc = min(alloc * 0.15, 60.0)
    if quick_alloc > 5.0:
        if n <= 230:
            hq, mq = pairwise_exhaustive_polish(h, target, rng, quick_alloc)
        else:
            hq, mq = pairwise_random_polish(h, target, rng, quick_alloc)
        if mq < best_m:
            best = hq
            best_m = mq
        if mq <= best_m + 1e-6:
            h = hq
        else:
            h = best.copy()

    elapsed = time.time() - t0
    lb_alloc = min(alloc * 0.35, 220.0)
    lb_alloc = min(lb_alloc, max(0.0, alloc - elapsed - 25.0))
    if lb_alloc > 15.0 and n <= 450:
        hl, ml = lbfgs_optimize(h, n, lb_alloc)
        if ml < best_m:
            best = hl
            best_m = ml
        if ml <= best_m + 1e-5:
            h = hl
        else:
            h = best.copy()

    elapsed = time.time() - t0
    dir_alloc = min(alloc * 0.30, 180.0)
    dir_alloc = min(dir_alloc, max(0.0, alloc - elapsed - 25.0))
    if dir_alloc > 10.0:
        hd, md = directional_polish(h, target, rng, dir_alloc)
        if md < best_m:
            best = hd
            best_m = md
        if md <= best_m + 1e-5:
            h = hd
        else:
            h = best.copy()

    elapsed = time.time() - t0
    rem_alloc = max(0.0, alloc - elapsed)
    if rem_alloc > 10.0:
        if n <= 230:
            hp, mp = pairwise_exhaustive_polish(h, target, rng, rem_alloc)
        else:
            hp, mp = pairwise_random_polish(h, target, rng, rem_alloc)
        if mp < best_m:
            best = hp
            best_m = mp

    best = repair_array(best, n)
    final_m = c5_only(best)
    return best, final_m


def run(seed=42, budget_s=1100, **kwargs):
    start0 = time.time()
    budget_s = float(budget_s)
    rng = np.random.default_rng(seed)

    ini = globals().get("initial_h_values", None)
    if isinstance(ini, tuple) and len(ini) >= 1:
        ini = ini[0]

    arr = None
    if ini is not None:
        try:
            arr = np.asarray(ini, dtype=float).reshape(-1)
        except Exception:
            arr = None
        if arr is None or arr.size < 3 or not np.all(np.isfinite(arr)):
            arr = None

    if arr is None:
        n0 = 131
        h0 = np.full(n0, 0.5, dtype=float)
    else:
        h0 = arr
        n0 = int(h0.size)

    h0 = repair_array(h0, n0)
    best_h = h0.copy()
    best_n = n0
    best_m = c5_only(best_h)

    if budget_s < 30.0:
        best_h = repair_array(best_h, best_n)
        best_m = c5_only(best_h)
        return best_h, float(best_m), int(best_n)

    cands = [n0]
    deltas = (1, -1, 2, -2, 3, -3, 5, -5, 7, -7, 11, -11, 19, -19, 31, -31)
    for d in deltas:
        nn = n0 + d
        if 40 <= nn <= 2500 and nn not in cands:
            cands.append(nn)
    extras = (251, 301, 351, 401, 451, 501)
    for e in extras:
        if 40 <= e <= 2500 and e not in cands:
            cands.append(e)
    cands = cands[:10]

    for idx, nn in enumerate(cands):
        elapsed = time.time() - start0
        if elapsed > budget_s - 30.0:
            break
        rem = budget_s - elapsed
        if idx == 0:
            alloc = min(rem * 0.55, 600.0)
        elif idx < 5:
            alloc = min(rem * 0.18, 140.0)
        else:
            alloc = min(rem * 0.10, 80.0)
        if alloc < 20.0:
            break

        if best_n == int(nn):
            h_start = best_h.copy()
        else:
            h_start = resample_h(best_h, int(nn))

        try:
            h_opt, m_opt = optimize_length(h_start, int(nn), alloc, rng)
        except Exception:
            continue

        if m_opt < best_m - 1e-14:
            best_h = h_opt
            best_m = m_opt
            best_n = int(nn)

    best_h = repair_array(best_h, best_n)
    best_m = c5_only(best_h)
    return best_h, float(best_m), int(best_n)
