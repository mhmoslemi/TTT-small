import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(c)

def get_gradient_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    shift1 = k - n + 1
    s1 = max(0, -shift1)
    e1 = min(n, n - shift1)
    if e1 > s1:
        grad[s1:e1] += (1.0 - h[s1 + shift1:e1 + shift1])
    shift2 = -k + n - 1
    s2 = max(0, -shift2)
    e2 = min(n, n - shift2)
    if e2 > s2:
        grad[s2:e2] -= h[s2 + shift2:e2 + shift2]
    return grad

def project_feasible(v, S):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid

    beta_m = 0.85
    start = time.time()
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_local = h_cur.copy()
    best_local_c5 = c5_cur

    max_no_improve = 200

        if time.time() - start > budget_remaining:
            break

        p = np.exp(beta_ls * c_shift)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(c)); p[k_max] = 1.0
        else:
            p /= p_sum
