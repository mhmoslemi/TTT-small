import numpy as np
import time

# Constants
EPS = 1e-12

def evaluate_c5(h):
    """Compute exact C5."""
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h):
    """Project to [0,1]^n with sum = n/2."""
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        resid = target - h.sum()
        if abs(resid) < 1e-14:
            break
        # Only adjust interior points
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not interior.any():
            # All at bounds; try adjusting boundary points
            interior = np.ones(n, dtype=bool)
        h[interior] += resid / interior.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, r):
    """Gradient of c_full[center+r] w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    L = n - abs(r)
    if L <= 0:
        return g
    if r >= 0:
        g[r:] += 1.0 - h[:L]
        g[:L] -= h[r:r+L]
    else:
        s = -r
        g[:L] += 1.0 - h[s:s+L]
        g[s:] -= h[:L]
    return g

def compute_smooth_gradient(h, tau=0.003, top_k=20):
    """Smoothed gradient using top-k softmax."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = c.max()
    k = min(top_k, len(c))
    idx = np.argsort(c)[-k:]
    vals = c[idx]
    shifted = (vals - cmax) / tau
    w = np.exp(shifted)
    w /= w.sum()
    
    g = np.zeros(n)
    center = n - 1
    for wi, j in zip(w, idx):
        r = j - center
        g += wi * dx * grad_for_lag(h, r)
    return g

def try_exchange(h, i, j, max_step=0.05, n_trials=7):
    """Try moving mass from j to i. Returns (new_h, score) or None."""
    n = len(h)
    # h[i] += d, h[j] -= d
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    # Limit step
    hi = min(hi, max_step)
    lo = max(lo, -max_step)
    if lo >= hi:
        return None
    best_h = None
    best_s = np.inf
    for d in np.linspace(max(lo+EPS, EPS), hi-EPS, n_trials):
        if d <= 0:
            continue
        new_h = h.copy()
        new_h[i] += d
        new_h[j] -= d
        s = evaluate_c5(new_h)
        if s < best_s:
            best_s = s
            best_h = new_h
    return best_h, best_s

def local_search(h, time_limit, rng, initial_tau=0.003):
    """Local search with gradient-guided exchanges."""
    n = len(h)
    h = project_feasible(h).copy()
    score = evaluate_c5(h)
    best_h = h.copy()
    best_score = score
    
    t0 = time.monotonic()
    tau = initial_tau
    top_k = 15
    stall = 0
    max_stall = 30
    
    while time.monotonic() - t0 < time_limit:
        g = compute_smooth_gradient(h, tau=tau, top_k=top_k)
        
        # Sort by gradient
        sorted_idx = np.argsort(g)
        improved = False
        
        # Try gradient-guided exchanges
        n_pairs = min(8, n // 2)
        for p in range(n_pairs):
            i = sorted_idx[p]      # low gradient (add mass)
            j = sorted_idx[-1-p]   # high gradient (remove mass)
            if i == j:
                continue
            # Move from high grad to low grad
            res = try_exchange(h, i, j, max_step=0.03)
            if res is not None:
                new_h, new_s = res
                if new_s < score - 1e-12:
                    h = new_h
                    score = new_s
                    improved = True
                    break
        
        # Try random exchanges
        if not improved:
            for _ in range(15):
                i, j = rng.choice(n, 2, replace=False)
                res = try_exchange(h, i, j, max_step=0.02)
                if res is not None:
                    new_h, new_s = res
                    if new_s < score - 1e-12:
                        h = new_h
                        score = new_s
                        improved = True
                        break
        
        if improved:
            if score < best_score:
                best_score = score
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Perturbation
                pert_size = 0.02 + 0.01 * (stall // max_stall)
                k_pert = rng.integers(2, min(6, n))
                idx = rng.choice(n, k_pert, replace=False)
                h[idx] += rng.uniform(-pert_size, pert_size, k_pert)
                h = project_feasible(h)
                score = evaluate_c5(h)
                stall = 0
                # Widen smoothing
                tau = min(tau * 1.5, 0.05)
                top_k = min(top_k + 3, 40)
            elif stall > max_stall // 2:
                # Small perturbation
                k_pert = rng.integers(1, 3)
                idx = rng.choice(n, k_pert, replace=False)
                h[idx] += rng.uniform(-0.01, 0.01, k_pert)
                h = project_feasible(h)
                score = evaluate_c5(h)
    
    return best_h, best_score
