def assemble_antipalindromic(x, N):
    """Build antipalindromic array from reduced variables x.
    For even N: x has length N/2, h[i] = (1+x[i])/2 for i=0..N/2-1,
                h[N-1-i] = 1 - h[i]

    rng = np.random.default_rng(seed)
    t_start = time.perf_counter()

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, UnboundLocalError):
        h_init = None

    if h_init is None or len(h_init) == 0:
        h_init = construct_fallback(417)

    N = len(h_init)
    h_init = np.clip(h_init, 0.0, 1.0)
    target = N / 2.0
    if abs(h_init.sum() - target) > 1e-6:
        h_init = fix_sum(h_init, target)

    best_h = h_init.copy()
    best_score = evaluate_c5(best_h)

    # Calibrate
    t_cal = time.perf_counter()
    _ = evaluate_c5(best_h)
    t_eval = max(time.perf_counter() - t_cal, 0.0005)

    def remaining():
        return budget_s - (time.perf_counter() - t_start)

    # Stage 1: Powell on anti variables at initial N
    D = N // 2
    x0 = extract_anti(best_h)
    x0 = np.clip(x0, -0.9999999, 0.9999999)

    _G['N'] = N
    _G['deadline'] = t_start + 0.50 * budget_s

    maxfev = max(300, int(0.50 * budget_s / (t_eval * 4.0)))
    maxfev = min(maxfev, 50000)

    res = minimize(objective_anti, x0, method='Powell',
                   options={'maxfev': maxfev, 'xtol': 1e-11, 'ftol': 1e-14, 'disp': False})

    if res.x is not None:
        x_best = np.clip(res.x, -1.0, 1.0)
        h_cand = assemble_anti(x_best, N)
        s_cand = evaluate_c5(h_cand)
        if s_cand < best_score:
            best_h = h_cand
            best_score = s_cand
            x0 = x_best

    # Stage 2: Nelder-Mead polish
    if remaining() > 80:
        _G['deadline'] = t_start + 0.70 * budget_s
        maxfev2 = max(200, int(0.20 * budget_s / (t_eval * 6.0)))
        maxfev2 = min(maxfev2, 30000)
        res2 = minimize(objective_anti, x0, method='Nelder-Mead',
                        options={'maxfev': maxfev2, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True, 'disp': False})
        if res2.x is not None:
            x_best = np.clip(res2.x, -1.0, 1.0)
            h_cand = assemble_anti(x_best, N)
            s_cand = evaluate_c5(h_cand)
            if s_cand < best_score:
                best_h = h_cand
                best_score = s_cand
                x0 = x_best

    # Stage 3: Coordinate descent polish
    if remaining() > 40:
        deadline_cd = t_start + 0.78 * budget_s
        max_iter_cd = int(0.08 * budget_s / (t_eval * D * 2.0))
        max_iter_cd = max(50, min(max_iter_cd, 2000))
        x0, _ = coordinate_descent_anti(x0, N, max_iter_cd, deadline_cd)
        h_cand = assemble_anti(np.clip(x0, -1, 1), N)
        s_cand = evaluate_c5(h_cand)
        if s_cand < best_score:
            best_h = h_cand
            best_score = s_cand

    # Stage 4: Random restarts on anti variables
    if remaining() > 60:
        deadline_r = t_start + 0.88 * budget_s
        n_restarts = min(15, int(0.10 * budget_s / (t_eval * 50.0)))
        for r in range(n_restarts):
            if time.perf_counter() > deadline_r:
                break
            sigma = 0.02 + 0.08 * (r % 4)
            x_pert = x0 + sigma * rng.standard_normal(D)
            x_pert = np.clip(x_pert, -0.9999999, 0.9999999)

            _G['deadline'] = deadline_r
            _G['N'] = N
            maxfev_r = max(100, int(0.5 * (deadline_r - time.perf_counter()) / (t_eval * 4.0)))
            maxfev_r = min(maxfev_r, 10000)

            res_r = minimize(objective_anti, x_pert, method='Powell',
                             options={'maxfev': maxfev_r, 'xtol': 1e-11, 'ftol': 1e-14, 'disp': False})
            if res_r.x is not None:
                x_best = np.clip(res_r.x, -1.0, 1.0)
                h_cand = assemble_anti(x_best, N)
                s_cand = evaluate_c5(h_cand)
                if s_cand < best_score:
                    best_h = h_cand
                    best_score = s_cand
                    x0 = x_best

    # Stage 5: Try different dimensions
    if remaining() > 50:
        deadline_d = t_start + 0.95 * budget_s
        N_inits = [N]
        for f in [1.2, 1.5, 0.8, 1.8, 0.7]:
            N_try = int(round(N * f))
            N_try = max(100, min(N_try, 2400))
            if N_try not in N_inits:
                N_inits.append(N_try)

        for N_new in N_inits:
            if time.perf_counter() > deadline_d:
                break
            if N_new <= 0:
                continue

            h_interp = interpolate_h(best_h, N_new)
            h_interp = fix_sum(h_interp, N_new / 2.0)

            # Make antipalindromic
            D_new = N_new // 2
            # Enforce anti-symmetry
            x_new = extract_anti(h_interp)
            x_new = np.clip(x_new, -0.9999999, 0.9999999)
            h_anti = assemble_anti(x_new, N_new)
            s_anti = evaluate_c5(h_anti)

            _G['N'] = N_new
            _G['deadline'] = deadline_d
            cost_factor = (N_new / N) ** 2
            maxfev_d = max(100, int(0.4 * (deadline_d - time.perf_counter()) / (t_eval * cost_factor * 4.0)))
            maxfev_d = min(maxfev_d, 20000)

            res_d = minimize(objective_anti, x_new, method='Powell',
                             options={'maxfev': maxfev_d, 'xtol': 1e-11, 'ftol': 1e-14, 'disp': False})
            if res_d.x is not None:
                x_best = np.clip(res_d.x, -1.0, 1.0)
                h_cand = assemble_anti(x_best, N_new)
                s_cand = evaluate_c5(h_cand)
                if s_cand < best_score:
                    best_h = h_cand
                    best_score = s_cand

    # Final validation and return
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final / 2.0)
    final_score = evaluate_c5(best_h)

    return (best_h, final_score, n_final)
