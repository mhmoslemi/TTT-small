def evaluate_c5(h):
    """Compute C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def to_binary(h, n):
    """Convert h to binary, maintaining sum constraint."""
    # Sort by value, keep top n/2 as 1s
    sorted_idx = np.argsort(h)[::-1]
    binary = np.zeros(n)
    binary[sorted_idx[:n//2]] = 1
    return binary

def local_search_binary(h, max_iters=1000, time_limit=60):
    """Local search on binary solutions via pair swaps."""
    n = len(h)
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    start_time = time.time()
    
    for iteration in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_delta = 0
        best_pair = None
        
        # Try all pair swaps
        for i in ones:
            for j in zeros:
                h_new = h.copy()
                h_new[i] = 0
                h_new[j] = 1
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5 - 1e-10:
                    best_delta = c5_new - best_c5
                    best_c5_new = c5_new
                    best_pair = (i, j)
        
        if best_pair is None:
            break  # Local minimum
        
        h[best_pair[0]] = 0
        h[best_pair[1]] = 1
        best_c5 = best_c5_new
