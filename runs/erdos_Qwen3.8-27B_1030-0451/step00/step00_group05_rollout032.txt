def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(full))

def project_to_feasible(y, S):
    n = len(y)
    if n == 0:
        return y
    y = np.asarray(y, dtype=float)
    # Bisection for lambda: x_i = clamp(y_i - lambda, 0, 1), sum(x) = S
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # Correct residual
    diff = S - x.sum()
    if abs(diff) > 1e-14:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        nf = free.sum()
        if nf > 0:
            x[free] += diff / nf
            x = np.clip(x, 0.0, 1.0)
        else:
            # All at boundary, distribute to whichever can move
            can_inc = x < 1.0
            can_dec = x > 0.0
            if diff > 0:
                nf = can_inc.sum()
                if nf > 0:
                    x[can_inc] += diff / nf
            else:
                nf = can_dec.sum()
                if nf > 0:
                    x[can_dec] += diff / nf
    return x

def compute_gradient(h, n, dx, T):
    """Compute gradient of soft-max smoothed C5. Returns (grad, full_values)."""
    full = np.correlate(h, 1.0 - h, mode='full') * dx
    C_max = float(np.max(full))
    # For numerical stability
    exps = np.exp((full - C_max) / T)
    total = exps.sum()
    if total < 1e-30:
        p = np.full(2*n-1, 1.0/(2*n-1))
    else:
        p = exps / total
    
    grad = np.zeros(n)
    c0 = n - 1
    
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        r = k - c0
        if r == 0:
            grad += pk * dx * (1.0 - 2.0 * h)
        elif r > 0:
            grad[r:] += pk * dx * (1.0 - h[:n-r])
            grad[:n-r] -= pk * dx * h[r:]
        else:
            s = -r
            grad[:n-s] += pk * dx * (1.0 - h[s:])
            grad[s:] -= pk * dx * h[:n-s]
    
    return grad, full

def optimize_local(h, n, dx, S, T_init, lr_init, max_iters, time_limit):
    """Local optimization with soft-max gradient descent."""
    T = T_init
    lr = lr_init
    h = project_to_feasible(h, S)
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    start = time.time()
    no_progress = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if time.time() > time_limit:
            break
        
        grad, full = compute_gradient(h, n, dx, T)
        # Zero-mean the gradient for sum preservation
        grad -= grad.mean()
        
        # Check gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            no_progress += 1
            if no_progress > 50:

        direction = -grad / gnorm

            if no_progress > 100:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    full = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(full))

def project_to_feasible(y, S):
    n = len(y)
    if n == 0:
        return y
    y = np.asarray(y, dtype=float).copy()
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-13:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        nf = free.sum()
        if nf > 0:
            x[free] += diff / nf
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_gradient(h, n, dx, T, top_k=None):
    full = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k is not None and top_k < len(full):
        idx = np.argpartition(full, -top_k)[-top_k:]
        f_top = full[idx].max()
        exps_full = np.zeros(len(full))
        exps_full[idx] = np.exp((full[idx] - f_top) / T)
    else:
        f_max = float(full.max())
        exps_full = np.exp((full - f_max) / T)
    tot = exps_full.sum()
    if tot < 1e-30:
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        p = exps_full / tot
    grad = np.zeros(n)
    c0 = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        r = k - c0
        if r == 0:
            grad += pk * dx * (1.0 - 2.0 * h)
        elif r > 0:
            grad[r:] += pk * dx * (1.0 - h[:n - r])
            grad[:n - r] -= pk * dx * h[r:]
        else:
            s = -r
            grad[:n - s] += pk * dx * (1.0 - h[s:])
            grad[s:] -= pk * dx * h[:n - s]
    return grad, full

def optimize_local(h, n, dx, S, T_init, lr_init, max_iters, time_limit, top_k=None):
    T = T_init
    lr = lr_init
    h = project_to_feasible(h, S)
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    t_start = time.time()
    no_prog = 0
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        grad, full = compute_gradient(h, n, dx, T, top_k=top_k)
        grad -= grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            no_prog += 1
            if no_prog > 200:
                break
            continue
        direction = -grad / gnorm
        improved = False
        step = lr
        for _ in range(40):
            h_trial = project_to_feasible(h + step * direction, S)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5 - 1e-16:
                h = h_trial
                c5 = c5_trial
                improved = True
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                break
            step *= 0.6
        if improved:
            no_prog = 0
            if it % 500 == 0 and T > 0.005:
                T *= 0.9
        else:
            no_prog += 1
            if no_prog == 50:
                T = max(T * 0.7, 0.002)
            elif no_prog == 150:
                lr = max(lr * 0.5, 1e-4)
            elif no_prog > 400:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.ones(131) * 0.5

    n = len(h0)
    S = n / 2.0
    dx = 2.0 / n
    h = project_to_feasible(h0, S)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Main optimization
    t_phase1 = budget_s * 0.45
    h, c5 = optimize_local(h, n, dx, S, T_init=0.08, lr_init=0.15,
                           max_iters=100000, time_limit=t_phase1)
    if c5 < best_c5:
        best_h, best_c5 = h, c5

    # Phase 2: Random restarts
    elapsed = time.time() - t0
    remaining = budget_s * 0.85 - elapsed
    if remaining > 5:
        n_restarts = min(20, max(3, int(remaining / 8)))
        for ri in range(n_restarts):
            if time.time() - t0 > budget_s * 0.85:
                break
            scale = 0.03 + 0.04 * np.random.rand()
            noise = np.random.randn(n) * scale
            noise -= noise.mean()
            h_start = project_to_feasible(best_h + noise, S)
            T_r = 0.02 + 0.03 * np.random.rand()
            lr_r = 0.05 + 0.05 * np.random.rand()
            tb = min(15.0, (budget_s * 0.85 - (time.time() - t0)) / max(1, n_restarts - ri))
            h_r, c5_r = optimize_local(h_start, n, dx, S, T_init=T_r, lr_init=lr_r,
                                       max_iters=20000, time_limit=tb)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 3: Try nearby resolutions
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.88:
        best_n = len(best_h)
        for delta in [-2, -1, 1, 2, -3, 3, -4, 4, -5, 5]:
            if time.time() - t0 > budget_s * 0.90:
                break
            n_try = best_n + delta
            if n_try < 50 or n_try > 2500:
                continue
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_rs = np.interp(x_new, x_old, best_h)
            h_rs = project_to_feasible(h_rs, n_try / 2.0)
            c5_rs = evaluate_c5(h_rs)
            if c5_rs < best_c5:
                best_h, best_c5 = h_rs, c5_rs
                best_n = n_try
            tb = min(10.0, (budget_s * 0.90 - (time.time() - t0)) * 0.7)
            if tb > 2:
                h_opt, c5_opt = optimize_local(h_rs, n_try, 2.0/n_try, n_try/2.0,
                                               T_init=0.02, lr_init=0.05,
                                               max_iters=10000, time_limit=tb)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    best_n = n_try

    # Phase 4: Final polish
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.95:
        n_f = len(best_h)
        S_f = n_f / 2.0
        dx_f = 2.0 / n_f
        h_p, c5_p = optimize_local(best_h, n_f, dx_f, S_f, T_init=0.005, lr_init=0.02,
                                   max_iters=30000, time_limit=budget_s * 0.95 - elapsed)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_to_feasible(best_h, n_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_final)
