import numpy as np
from scipy.optimize import minimize
import time
import functools

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return 1e10
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c * dx))

def build_h(p, n):
    p = np.clip(np.asarray(p, dtype=np.float64), 0.0, 1.0)
    m = n // 2
    h = np.empty(n, dtype=np.float64)
    if n % 2 == 0:
        h[:m] = p
        h[m:] = 1.0 - p[::-1]
    else:
        h[:m] = p
        h[m] = 0.5
        h[m + 1:] = 1.0 - p[::-1]
    return h

def p_from_h(h, n):
    m = n // 2
    p_sym = 0.5 * (h[:m] + 1.0 - h[n - 1:m - 1:-1])
    return np.clip(p_sym, 0.0, 1.0)

def project_full(y):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

def interp_p(p_old, m_new):
    m_old = len(p_old)
    if m_old == m_new:
        return p_old.copy()
    x_old = (np.arange(m_old) + 0.5) / m_old
    x_new = (np.arange(m_new) + 0.5) / m_new
    return np.clip(np.interp(x_new, x_old, p_old), 0.0, 1.0)

def _lse_wrapper(p, tau, n, state):
    p_clip = np.clip(p, 0.0, 1.0)
    h = build_h(p_clip, n)
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode="full") * dx
    fmax = float(np.max(f))
    if fmax < state['best_val']:
        state['best_val'] = fmax
        state['best_p'] = p_clip.copy()
        state['best_n'] = n
        state['best_h'] = h.copy()
    if tau < 1e-10:
        return fmax
    shifted = (f - fmax) / tau
    return fmax + tau * float(np.log(np.sum(np.exp(shifted))))

def _constr_func(x, j, m, n):
    p_tmp = np.clip(x[:m], 0.0, 1.0)
    h_tmp = build_h(p_tmp, n)
    f_tmp = np.correlate(h_tmp, 1.0 - h_tmp, mode="full") * (2.0 / n)
    return x[-1] - f_tmp[int(j)]

def _slsqp_obj(x):
    return x[-1]

def _full_lse(y, tau, n, state):
    h = project_full(y)
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode="full") * dx
    fmax = float(np.max(f))

def _try_lse(p_init, n, state, deadline, tau_seq, maxfun):

    current_p = p_init.copy()
    bounds = [(0.0, 1.0)] * m

            if res.x is not None and np.all(np.isfinite(res.x)):

                    state['best_p'] = current_p.copy()
                    state['best_n'] = n
                    state['best_h'] = h_chk.copy()
        except Exception:

def _try_slsqp(p_init, n, state, deadline):

    bounds = [(0.0, 1.0)] * m + [(0.0, 2.0)]
    constraints = []
    for j in top_idx:
        cf = functools.partial(_constr_func, j=int(j), m=m, n=n)
        constraints.append({'type': 'ineq', 'fun': cf})
    max_it = 80 if m <= 40 else (50 if m <= 80 else 30)
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_it, 'ftol': 1e-10, 'disp': False})
