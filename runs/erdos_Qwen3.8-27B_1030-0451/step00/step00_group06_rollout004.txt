import numpy as np

def _to_binary(h, n, rng):
    """Convert h to binary array of length n with sum = n//2."""
    target = n // 2
    if len(h) != n:
        x_old = np.linspace(0, 2, len(h), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, h)
    h_bin = np.where(h >= 0.5, 1.0, 0.0).astype(np.float64)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        h_bin[ones_idx[order[:n_ones - target]]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        h_bin[zeros_idx[order[:target - n_ones]]] = 1

                    h[i], h[j] = h[j], h[i]
