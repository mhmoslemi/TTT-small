    """Compute partial derivative of Q_r wrt h[t] for all r."""
    L = 2 * N - 1
    offset = N - 1
    p = np.zeros(L)
    d = np.arange(L) - offset  # d = r - offset for r = 0..L-1

    valid1 = (idx1 >= 0) & (idx1 < N)

    idx2 = t + d  # index into h for the -h[...] term
    valid2 = (idx2 >= 0) & (idx2 < N)

    offset = N - 1
    # d_r = r - offset, for r = 0..L-1
    d = np.arange(L) - offset  # shape (L,), values from -(N-1) to N-1
    
    # P1[t, r] = (1 - h[t - d_r]) * I(0 <= t - d_r < N)
    # M1[t, r] = t - d_r = t - (r - offset) = t + offset - r
    M1 = np.arange(N)[:, None] + offset - np.arange(L)[None, :]  # shape (N, L)
    valid1 = (M1 >= 0) & (M1 < N)
    P1 = np.zeros((N, L))
    P1[valid1] = 1.0 - h[M1[valid1]]
    
    # P2[t, r] = h[t + d_r] * I(0 <= t + d_r < N)
    # M2[t, r] = t + d_r = t + (r - offset) = t - offset + r
    M2 = np.arange(N)[:, None] - offset + np.arange(L)[None, :]  # shape (N, L)
    valid2 = (M2 >= 0) & (M2 < N)
    P2 = np.zeros((N, L))
    P2[valid2] = h[M2[valid2]]
    
    return P1 - P2

def line_minimize(Q, A, G, s_lo, s_hi, max_iter=80, tol=1e-13):
    """Minimize F(s) = max(Q + A*s + G*s^2) over [s_lo, s_hi]."""
    if s_hi - s_lo < tol:
        return (s_lo + s_hi) / 2.0
    lo = s_lo
    hi = s_hi
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        mid = (lo + hi) / 2.0
        vals = Q + A * mid + G * mid * mid
        f_max = np.max(vals)
        active = vals >= f_max - 1e-10
        if not np.any(active):
            active = np.array([True])  # fallback
            sg = np.array([0.0])
        else:
            sg = A[active] + 2.0 * G[active] * mid
        sg_lo = np.min(sg)
        sg_hi = np.max(sg)
        if sg_lo > 1e-14:
            hi = mid
        elif sg_hi < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = np.ones(128) * 0.5
    
    N = len(h)
    L = 2 * N - 1
    offset = N - 1
    
    Q = compute_Q(h)
    P = compute_P(h)
    
    dx = 2.0 / N
    best_score = dx * float(np.max(Q))
    best_h = h.copy()
    
    # Precompute G vectors for all possible distances
    # G_m[r] = 1 at r = offset ± m, 0 elsewhere
    # We'll compute G on the fly since it's just two non-zero entries
    
    stall = 0
    max_stall = 200
    
    while time.time() - start_time < budget_s - 15:
        improved = False
        current_Q_max = float(np.max(Q))
        
        # Generate pairs - mix of strategies
        n_pairs = 500
        pairs = np.zeros((n_pairs, 2), dtype=int)
        
        for i in range(n_pairs):
            strat = i % 5
            if strat == 0:
                u = rng.randint(0, N - 1)
                v = u + 1
            elif strat == 1:
                u = rng.randint(0, N)
                v = rng.randint(0, N)
                if u == v:
                    v = (u + 1) % N
            elif strat == 2:
                # Pairs at "structural" distances
                d = rng.choice(np.array([N//8, N//6, N//4, N//3, N//2, 2*N//3]))
                d = min(d, N - 1)
                u = rng.randint(0, N - d)
                v = u + d
            elif strat == 3:
                # Random medium distance
                d = rng.randint(1, N)
                u = rng.randint(0, N - d)
                v = u + d
            else:
                # Pairs near the "peak" of h (where h transitions)
                # Pick from regions where h is between 0.2 and 0.8
                mid_region = np.where((h > 0.1) & (h < 0.9))[0]
                if len(mid_region) > 1:
                    u = mid_region[rng.randint(len(mid_region))]
                    v = mid_region[rng.randint(len(mid_region))]
                    if u == v:
                        v = (u + 1) % N
                else:
                    u = rng.randint(0, N)
                    v = rng.randint(0, N)
                    if u == v:
                        v = (u + 1) % N
            pairs[i] = [u, v]
        
        for idx in range(n_pairs):
            if time.time() - start_time > budget_s - 15:
                break
            
            u, v = pairs[idx]
            if u == v:
                continue
            
            m = abs(u - v)
            if m == 0:
                continue
            
            # Directional derivative
            A = P[u] - P[v]
            
            # Quadratic correction
            # G is zero except at r = offset+m and r = offset-m
            # We handle this specially for speed
            
            s_lo = max(-h[u], h[v] - 1.0)
            s_hi = min(1.0 - h[u], h[v])
            if s_hi - s_lo < 1e-12:
                continue
            
            # Quick check: is there any benefit?
            # The best possible improvement is bounded by the current max minus
            # the minimum possible value. If A is all positive (or all negative)
            # for the active lags, we can skip.
            active_lags = np.where(Q >= current_Q_max - 0.01 * current_Q_max)[0]
            if len(active_lags) > 0:
                A_active = A[active_lags]
                # If for all active lags, the directional derivative has the same sign
                # as the step direction, we won't improve
                # The step range is [s_lo, s_hi]
                # Best candidates: s_lo (if A < 0) or s_hi (if A > 0)
                # Quick heuristic: check if min of (A[s_lo] or A[s_hi]) is negative
                min_A_at_slo = np.min(A_active * s_lo)
                min_A_at_shi = np.min(A_active * s_hi)
                if min_A_at_slo >= 0 and min_A_at_shi >= 0:
                    # Likely no improvement (rough check)
                    # But quadratic terms might help, so don't skip entirely
                    pass  # Don't skip; let line_minimize decide
            
            # Line minimize with quadratic terms
            # F(s) = max(Q + A*s + G*s^2)
            # G has at most 2 non-zero entries
            # We can evaluate F(s) as:
            #   base = Q + A*s
            #   base[r1] += s^2  (if r1 valid)
            #   base[r2] += s^2  (if r2 valid)
            #   F(s) = max(base)
            
            r1 = offset + m
            r2 = offset - m
            valid_r1 = 0 <= r1 < L
            valid_r2 = 0 <= r2 < L
            
            # Custom line minimize for 2 non-zero G entries
            lo = s_lo
            hi = s_hi
            for _ in range(80):
                if hi - lo < 1e-13:
                    break
                mid = (lo + hi) / 2.0
                base = Q + A * mid
                if valid_r1:
                    base[r1] += mid * mid
                if valid_r2:
                    base[r2] += mid * mid
                f_max = np.max(base)
                active = base >= f_max - 1e-10
                if not np.any(active):
                    break
                sg = np.zeros(np.sum(active))
                for ii, ai in enumerate(np.where(active)[0]):
                    g_val = 1.0 if (ai == r1 or ai == r2) else 0.0
                    sg[ii] = A[ai] + 2.0 * g_val * mid
                sg_lo = np.min(sg)
                sg_hi = np.max(sg)
                if sg_lo > 1e-14:
                    hi = mid
                elif sg_hi < -1e-14:
                    lo = mid
                else:
                    break
            s_star = (lo + hi) / 2.0
            
            # Quick evaluation
            base = Q + A * s_star
            if valid_r1:
                base[r1] += s_star * s_star
            if valid_r2:
                base[r2] += s_star * s_star
            line_max = float(np.max(base))
            
            if line_max >= current_Q_max - 1e-12:
                continue
