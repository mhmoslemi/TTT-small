    dx = 2.0 / n
    corr = get_corr_array(h)  # length 2n-1
    m = np.max(corr)
    e = np.exp(t * (corr - m))

        wi = w[idx]
        if abs(wi) < 1e-15:

        # j-k >= 0 and j-k < n, i.e., j >= k and j < n+k
        # Also j >= 0 and j < n
        j_lo = max(0, k)
        j_hi = min(n, n + k)  # exclusive
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wi * dx * (1.0 - h[j_lo-k:j_hi-k])
        # Backward: -h[j+k] for valid j
        # j+k >= 0 and j+k < n, i.e., j >= -k and j < n-k
        # Also j >= 0 and j < n
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)  # exclusive
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wi * dx * h[j_lo2+k:j_hi2+k]
    
    return grad

    return float(np.max(get_corr_array(h)))

def smooth_max_val(corr, t):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(t * (corr - m)))) / t

def smooth_max_weights(corr, t):
    m = np.max(corr)
    e = np.exp(t * (corr - m))
    return e / np.sum(e)

def smooth_grad(h, t):
    """Gradient of smooth_max(corr(h), t) wrt h."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr_array(h)
    w = smooth_max_weights(corr, t)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-15:
            continue
        # Forward contribution: (1-h[j-k]) for valid j
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wi * dx * (1.0 - h[j_lo - k:j_hi - k])
        # Backward contribution: -h[j+k] for valid j
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wi * dx * h[j_lo2 + k:j_hi2 + k]
    return grad

def mass_project(x, target):
    """Project onto {y: 0<=y_i<=1, sum(y)=target}."""
    x = np.array(x, dtype=np.float64, copy=True)
    x[~np.isfinite(x)] = 0.5
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - result.sum()
    if abs(r) > 1e-13:
        if r > 0:
            mask = result < 1.0
            cnt = mask.sum()
            if cnt > 0:
                result[mask] = np.clip(result[mask] + r / cnt, 0.0, 1.0)
        else:
            mask = result > 0.0
            cnt = mask.sum()
            if cnt > 0:
                result[mask] = np.clip(result[mask] + r / cnt, 0.0, 1.0)
    return result

def transfer_res(h, n_new):
    """Transfer h to new resolution n_new via piecewise-constant averaging."""
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    # For each new cell, compute the area-weighted average of old cells
    result = np.zeros(n_new)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    # Use cumulative sum for efficiency
    for j in range(n_new):
        x_lo = j * dx_new
        x_hi = (j + 1) * dx_new
        # Find old cells overlapping [x_lo, x_hi]
        i_start = int(np.floor(x_lo / dx_old + 1e-10))
        i_end = int(np.ceil(x_hi / dx_old - 1e-10))
        i_start = max(0, i_start)
        i_end = min(n_old, i_end)
        area = 0.0
        for i in range(i_start, i_end):
            cell_lo = i * dx_old
            cell_hi = (i + 1) * dx_old
            overlap = max(0.0, min(x_hi, cell_hi) - max(x_lo, cell_lo))
            area += h[i] * overlap
        result[j] = area / dx_new
    return result
