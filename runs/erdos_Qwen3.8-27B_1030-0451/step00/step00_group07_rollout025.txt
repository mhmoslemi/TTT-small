def smooth_max(lags, tau):
    """Smooth maximum using log-sum-exp."""
    m = np.max(lags)
    if tau < 1e-10:
        return m, lags
    shifted = (lags - m) / tau
    # Numerical stability
    max_shifted = np.max(shifted)
    if max_shifted > 500:
        shifted = shifted - max_shifted
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    return m + tau * (max_shifted + np.log(s)), exp_shifted / s

    # alpha is the softmax weights for each lag index (length 2N-1)

import numpy as np
from scipy.ndimage import zoom
import time

def compute_c5(h):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / N))

def project_feasible(h):
    N = len(h)
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -0.5, 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        interior = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            h[interior] += residual / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, tau):
    N = len(h)
    lags = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / N
    m = float(np.max(lags))
    shifted = (lags - m) / tau
    max_s = float(np.max(shifted))
    shifted -= max_s
    exp_s = np.exp(shifted)
    total = float(np.sum(exp_s))
    alpha = exp_s / total
    obj = (m + tau * (max_s + np.log(total))) * dx
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        a = alpha[k]
        if a < 1e-12:
            continue
        d = k - (N - 1)
        w = a * dx
        js = max(0, -d)
        je = min(N, N - d)
        if js < je:
            grad[js:je] += w * (1.0 - h[js + d:je + d])
        js2 = max(0, d)
        je2 = N
        if js2 < je2:
            grad[js2:je2] += -w * h[js2 - d:je2 - d]
    return obj, grad

def proj_gradient(h, tau, deadline, lr_init=0.01):
    N = len(h)
    h = project_feasible(h.copy())
    lr = lr_init
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    stall = 0
    for it in range(500000):
        if time.time() > deadline:
            break
        _, grad = smooth_obj_grad(h, tau)
        accepted = False
        lr_try = lr
        for _ in range(20):
            h_new = project_feasible(h - lr_try * grad)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                accepted = True
                break
            lr_try *= 0.5
        if accepted:
            lr = min(lr_try * 1.3, 1.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            lr *= 0.7
            stall += 1
        if stall > 500:
            break
    return best_h, best_c5
