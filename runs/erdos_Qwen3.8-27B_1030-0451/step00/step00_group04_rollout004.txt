import numpy as np
import time

def exact_eval(h):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(z))

def project_box_sum(x, T):
    """Project x onto {h: 0<=h<=1, sum(h)=T}"""
    n = len(x)
    # Bisection on lambda: clip(x - lambda, 0, 1) has sum = T
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Small repair
    diff = T - y.sum()
    if abs(diff) > 1e-15:
        # Distribute among non-bound variables
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            # Just add to all
            y += diff / n
            y = np.clip(y, 0.0, 1.0)
    return y

def compute_z(h):
    """Compute all correlation values scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_grad(h, tau):
    """Compute gradient of smooth-max surrogate."""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights (stable)
    z_shifted = z - z.max()
    p = np.exp(z_shifted / tau)
    p /= p.sum()
    
    # Now compute gradient: sum_q p_q * dC_ell/dh
    # ell_q = q - (n-1) for q = 0, ..., 2n-2
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        if p[q] < 1e-15:
            continue
        ell = q - (n - 1)
        # For this lag, gradient contribution:
        # dC_ell/dh[i] = (1-h[i-ell]) * [0<=i-ell<n] - h[i+ell] * [0<=i+ell<n]
        # Vectorized:
        if ell == 0:
            # (1-h[i]) - h[i] = 1 - 2*h[i] for all i
            grad += p[q] * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            # (1-h[i-ell]) for i >= ell: indices ell..n-1
            # -h[i+ell] for i <= n-1-ell: indices 0..n-1-ell
            grad[ell:] += p[q] * dx * (1.0 - h[:n-ell])
            grad[:n-ell] -= p[q] * dx * h[ell:]
        else:  # ell < 0, let a = -ell > 0
            a = -ell
            # (1-h[i+a]) for i <= n-1-a: indices 0..n-1-a
            # -h[i-a] for i >= a: indices a..n-1
            grad[:n-a] += p[q] * dx * (1.0 - h[a:])
            grad[a:] -= p[q] * dx * h[:n-a]
    return grad

def pgd_optimize(h0, budget_s, start_time, tau_init, tau_final):
    """Projected gradient descent with smooth-max surrogate."""
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.copy(), T)
    best_h = h.copy()
    best_score = exact_eval(h)
    z = compute_z(h)
    spread = z.max() - z.min()
    tau = max(tau_init, 0.05 * spread + 0.005)
    alpha = 0.1
    tau_min = tau_final
    
    while time.time() - start_time < budget_s - 5:
        # Decay tau
        tau = max(tau * 0.92, tau_min)
        
        # Compute gradient
        g = smooth_grad(h, tau)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        if not np.isfinite(g_norm):
            break
        
        # Normalize direction
        direction = -g / max(g_norm, 1e-12)
        
        # Line search
        F_current = None  # We use exact eval for acceptance
        for step_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = alpha * step_mult
            h_new = project_box_sum(h + step * direction, T)
            score_new = exact_eval(h_new)
            if score_new < best_score - 1e-12:
                h = h_new
                best_h = h.copy()
                best_score = score_new
                break
            elif score_new < F_current if F_current is not None else score_new < exact_eval(h):
                h = h_new
                break
        else:
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        # Occasionally update alpha up
        if np.random.random() < 0.1:
            alpha = min(alpha * 1.5, 0.5)
    
    return best_h, best_score

from scipy.optimize import minimize

def project_box_sum(x, T):
    n = len(x)
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
        else:
            y[:] += diff / n
        y = np.clip(y, 0.0, 1.0)
    return y

def compute_z(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = compute_z(h)
    z_shift = z - z.max()
    p = np.exp(z_shift / tau)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p /= p_sum
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        if p[q] < 1e-14:
            continue
        ell = q - (n - 1)
        if ell == 0:
            grad += p[q] * dx * (1.0 - 2.0 * h)
        elif ell > 0:

            grad[ell:] += p[q] * dx * (1.0 - h[:L])
            grad[:L] -= p[q] * dx * h[ell:]

            a = -ell

            grad[:L] += p[q] * dx * (1.0 - h[a:])
            grad[a:] -= p[q] * dx * h[:L]

    h = project_box_sum(h0.copy(), T)

    best_score = exact_eval(h)
    
    z = compute_z(h)
    spread = max(z.max() - z.min(), 0.01)

    st = time.time()
    n_iter = 0
    while time.time() - start_time < time_budget:
        if time.time() - st > 60:
            break
        st = time.time()

        gnorm = np.linalg.norm(g)

            break
        direction = -g / gnorm

        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = alpha * mult
            h_new = project_box_sum(h + step * direction, T)
            s = exact_eval(h_new)

def slsqp_polish(h0, t_init, maxiter=200):
    n = len(h0)
    T = n / 2.0
    dx = 2.0 / n
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def constraint_val(x):
        h = x[:n]
        t = x[-1]
        z = np.correlate(h, 1.0 - h, mode="full") * dx
        return z - t
    
    def constraint_jac(x):
        h = x[:n]
        # Jacobian of z_q - t w.r.t. x
        # For each lag q, dC_ell/dh as before, and -1 for t
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        J[:, -1] = -1.0
        for q in range(m):
            ell = q - (n - 1)
            if ell == 0:
                J[q, :n] = dx * (1.0 - 2.0 * h)
            elif ell > 0:
                L = n - ell
                J[q, ell:n] += dx * (1.0 - h[:L])
                J[q, :L] -= dx * h[ell:n]
            else:
                a = -ell
                L = n - a
                J[q, :L] += dx * (1.0 - h[a:n])
                J[q, a:n] -= dx * h[:L]
        return J
    
    def eq_constraint(x):
        return x[:n].sum() - T
    
    def eq_constraint_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    x0 = np.concatenate([h0, [max(t_init + 1e-4, 0.05)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.6)]
    
    cons = [
        {'type': 'ineq', 'fun': lambda x: -constraint_val(x), 'jac': lambda x: -constraint_jac(x)},
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_constraint_jac}
    ]
    
    res = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-9})
    
    if res.success or res.x is not None:
        h_out = project_box_sum(res.x[:n].copy(), T)
        return h_out
    return project_box_sum(h0.copy(), T)

import numpy as np
import time
from scipy.optimize import minimize

def exact_eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def project_box_sum(x, T):
    n = len(x)
    lo = float(x.min()) - 1.5
    hi = float(x.max()) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - float(y.sum())
    if abs(diff) > 1e-13:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y[:] += diff / n
        y = np.clip(y, 0.0, 1.0)
    return y

def compute_z(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = compute_z(h)
    z_max = float(z.max())
    exponents = np.clip((z - z_max) / tau, -50.0, 50.0)
    p = np.exp(exponents)
    p_sum = float(p.sum())
    if p_sum < 1e-300:
        return np.zeros(n)
    p /= p_sum
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = p[q] * dx
        if w < 1e-14:
            continue
        ell = q - (n - 1)
        if ell == 0:
            grad += w * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[:L] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:L]
    return grad

def pgd_optimize(h0, time_limit, deadline, tau_start=None):
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.astype(np.float64).copy(), T)
    best_h = h.copy()
    best_score = exact_eval(h)
    z = compute_z(h)
    spread = max(float(z.max() - z.min()), 0.005)
    tau = max(0.03 * spread, 0.008) if tau_start is None else tau_start
    alpha = 0.1
    end_time = min(time.time() + time_limit, deadline - 2.0)

    while time.time() < end_time:
        g = smooth_grad(h, tau)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-13 or not np.isfinite(gnorm):
            break
        direction = -g / gnorm
        s_cur = exact_eval(h)

        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = alpha * mult
            h_new = project_box_sum(h + step * direction, T)
            s_new = exact_eval(h_new)
            if s_new < best_score - 1e-11:
                h = h_new
                best_h = h_new.copy()
                best_score = s_new
                improved = True
                break
            if not improved and s_new <= s_cur - 1e-11:
                h = h_new
                improved = True
                break

        if not improved:
            alpha *= 0.5
            if alpha < 1e-14:
                break
        tau = max(tau * 0.87, 3e-5)

    return best_h, best_score

def slsqp_polish(h0, t_init, deadline, maxiter=100):
    n = len(h0)
    T = n / 2.0
    dx = 2.0 / n
    m = 2 * n - 1
    z_now = compute_z(h0)
    top_k = min(40, m)
    idx = np.argsort(z_now)[-top_k:]

    def obj(x):
        return x[-1]

    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def con_val(x):
        h = x[:n]
        t = x[-1]
        z = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - z[idx]

    def con_jac(x):
        h = x[:n]
        J = np.zeros((top_k, n + 1))
        J[:, -1] = 1.0
        for i in range(top_k):
            q = idx[i]
            ell = q - (n - 1)
            if ell == 0:
                J[i, :n] = -dx * (1.0 - 2.0 * h)
            elif ell > 0:
                L = n - ell
                J[i, ell:n] = -dx * (1.0 - h[:L])
                J[i, :L] = dx * h[ell:]
            else:
                a = -ell
                L = n - a
                J[i, :L] = -dx * (1.0 - h[a:])
                J[i, a:n] = dx * h[:L]
        return J

    def eq_val(x):
        return float(x[:n].sum()) - T

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    t0 = max(t_init + 1e-3, 0.05)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'ineq', 'fun': con_val, 'jac': con_jac},
        {'type': 'eq', 'fun': eq_val, 'jac': eq_jac}
    ]
    try:
        if time.time() > deadline - 15:
            return h0
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_box_sum(res.x[:n].copy(), T)
    except Exception:
        pass
    return h0

    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.default_rng(seed)

    # Get initial values

    except Exception:
        n = 250
        T = n / 2.0

    best_h = h_init.copy()
    best_score = exact_eval(h_init)

    # Phase 1: Main PGD (long run)
    t1 = min(280, (deadline - start_time) * 0.35)

        h1, s1 = pgd_optimize(h_init, t1, deadline, tau_start=0.02)

    # Phase 2: PGD with smaller tau on best
    if time.time() < deadline - 80:
        t2 = min(120, (deadline - time.time()) * 0.25)
        h2, s2 = pgd_optimize(best_h, t2, deadline, tau_start=0.005)
        if s2 < best_score:

    # Phase 3: Perturbation restarts
    n_pert = 5
    for i in range(n_pert):
        if time.time() > deadline - 80:
            break
        sigma = 0.02 + 0.015 * i
        h_pert = project_box_sum(best_h + rng.normal(0, sigma, n), T)
        t_lim = min(45, (deadline - time.time()) - 60)
        if t_lim < 5:
            break
        h_opt, s_opt = pgd_optimize(h_pert, t_lim, deadline, tau_start=0.01)

            s_slq2 = exact_eval(h_slq2)

        r = 2
        n_up = n * r

                if s_down < best_score:

    h_comp = project_box_sum(1.0 - best_h, len(best_h) / 2.0)
    s_comp = exact_eval(h_comp)
    if s_comp < best_score:
        best_h, best_score = h_comp, s_comp
    h_rev = best_h[::-1].copy()
    h_rev = project_box_sum(h_rev, len(h_rev) / 2.0)

    if s_rev < best_score:

    best_h = np.clip(best_h.astype(np.float64), 0.0, 1.0)
    T_final = len(best_h) / 2.0
    best_h = project_box_sum(best_h, T_final)
    final_score = exact_eval(best_h)

    return (best_h, final_score, len(best_h))
