    target = 0.5 * n
    # Bisection for alpha
    lo = -y.max() - 1
    hi = 1 - y.min() + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    
    h = np.clip(y + (lo + hi) / 2, 0, 1)

    residual = target - np.sum(h)

            order = np.argsort(-rooms)
            for idx in order:

                h[idx] += step
                residual -= step

            for idx in order:

    return np.array(sorted(active))

def smoothed_pgdescent(h0, n, max_iter=2000, tau_init=None, time_limit=None, t_start=None):
    """Run LSE-smoothed projected gradient descent."""
    h = h0.copy()
    dx = 2.0 / n
    
    if tau_init is None:
        tau_init = 0.003
    
    tau = tau_init
    best_c = float('inf')
    best_h = h.copy()
    
    c_val, c_vals = evaluate_c5(h)
    best_c = c_val
    best_h = h.copy()
    
    no_improve_count = 0
    tau_increase_count = 0
    
    for it in range(max_iter):
        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        
        if it % 50 == 0:
            c_val, c_vals = evaluate_c5(h)
            if c_val < best_c:
                best_c = c_val
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
            
            # Adjust tau
            if no_improve_count > 5:
                tau = min(tau * 1.5, 0.01)
                no_improve_count = 0
                tau_increase_count += 1
            elif tau > tau_init and tau_increase_count > 0:
                tau = max(tau_init, tau * 0.7)
        
        # Pick active lags
        if len(c_vals) <= 120:
            active = np.arange(len(c_vals))
        else:
            active = pick_active_lags(c_vals, 100)
        
        # Compute gradient
        grad = compute_gradient_efficient(h, c_vals, tau, n, active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Try several step sizes
        best_step_h = None
        best_step_c = c_val
        
        gnorm_f = max(gnorm, 1e-12)
        base_step = 0.5 / gnorm_f
        
        for factor in [1.0, 0.5, 0.25, 0.1]:
            step = base_step * factor
            y = h - step * grad
            h_trial = project_feasible(y, n)
            c_trial, _ = evaluate_c5(h_trial)
            if c_trial < best_step_c:
                best_step_c = c_trial
                best_step_h = h_trial
            if c_trial < best_c - 1e-10:
                break
        
        if best_step_h is not None:
            h = best_step_h
            c_val = best_step_c
        else:
            # No improvement with any step size
            # Try a small random perturbation
            pert = np.random.randn(n) * 0.01
            y = h + pert
            h_trial = project_feasible(y, n)
            c_trial, _ = evaluate_c5(h_trial)
            if c_trial < c_val:
                h = h_trial
                c_val = c_trial
            else:
                # Reset h to best
                h = best_h.copy()
                c_val, c_vals = evaluate_c5(h)
                # Increase tau more aggressively
                tau = min(tau * 2.0, 0.02)
                no_improve_count += 2
                if no_improve_count > 20:
                    break
    
    # Final evaluation
    c_final, _ = evaluate_c5(best_h)
    return best_h, c_final

def slsqp_polish(h0, n, max_active=60, time_limit_s=30):
    """Polish with SLSQP on active lags."""
    h = h0.copy()
    dx = 2.0 / n
    target = 0.5 * n
    
    c_val, c_vals = evaluate_c5(h)
    active = pick_active_lags(c_vals, max_active)
    
    t_start = time.time()
    best_h = h.copy()
    best_c = c_val
    
    for round_num in range(3):
        if time.time() - t_start > time_limit_s:
            break
        
        active = pick_active_lags(c_vals, max_active)
        n_active = len(active)
        
        def objective(x):
            return x[-1] - best_c
        
        def objective_grad(x):
            g = np.zeros(len(x))
            g[-1] = 1.0
            return g
        
        # We'll maximize t - max(f_m for m in active)  = minimize t - max(...)
        # Actually, let's minimize t subject to f_m <= t for active m
        # Variables: h[0..n-1], t  (total n+1 variables)
        
        n_vars = n + 1
        
        def obj(x):
            return x[-1]
        
        def obj_grad(x):
            g = np.zeros(n_vars)
            g[-1] = 1.0
            return g
        
        def constraint_eq(x):
            return np.sum(x[:n]) - target
        
        def constraint_eq_grad(x):
            g = np.ones(n_vars)
            g[-1] = 0.0
            return g
        
        # Inequality constraints: t - f_m >= 0 for each active m
        # f_m = dx * correlate(h, 1-h, full)[m]
        
        def constraints(x):
            h_x = x[:n]
            results = []
            for m in active:
                k = m - (n - 1)
                if k == 0:
                    f = dx * np.sum(h_x * (1.0 - h_x))
                elif k > 0:
                    if k >= n:
                        f = 0.0
                    else:
                        f = dx * np.sum(h_x[k:] * (1.0 - h_x[:n-k]))
                else:
                    r = -k
                    if r >= n:
                        f = 0.0
                    else:
                        f = dx * np.sum(h_x[:n-r] * (1.0 - h_x[r:]))
                results.append(x[-1] - f)
            return results
        
        def constraint_grad(x):
            h_x = x[:n]
            G = np.zeros((n_active, n_vars))
            for i, m in enumerate(active):
                k = m - (n - 1)
                if k == 0:
                    G[i, :n] = dx * (1.0 - 2.0 * h_x)
                elif k > 0:
                    if k < n:
                        G[i, k:n] += dx * (1.0 - h_x[:n-k])
                        G[i, :n-k] += dx * (-h_x[k:])
                else:
                    r = -k
                    if r < n:
                        G[i, :n-r] += dx * (1.0 - h_x[r:])
                        G[i, r:n] += dx * (-h_x[:n-r])
                G[i, -1] = 1.0
            return G
        
        x0 = np.append(h, c_val + 0.005)
        bounds = [(0, 1)] * n + [(0, 2.0)]
        
        cons = [
            {'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_grad},
            {'type': 'ineq', 'fun': constraints, 'jac': constraint_grad}
        ]
        
        try:
            result = minimize(
                obj, x0, method='SLSQP',
                jac=obj_grad,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
            )
            
            if result.x is not None:
                h_new = project_feasible(result.x[:n], n)
                c_new, c_vals_new = evaluate_c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    c_vals = c_vals_new
        except Exception:
            pass
    
    return best_h, best_c

def resample_h(h, new_n):
    """Resample h to new resolution."""
    old_n = len(h)
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n)

def random_candidate(n, rng):
    """Generate a random feasible candidate."""
    method = rng.integers(3)
    if method == 0:
        # Random noise around 0.5
        h = 0.5 + 0.4 * rng.random(n)
    elif method == 1:
        # Block structure
        h = np.zeros(n)
        num_blocks = rng.integers(2, 6)
        positions = np.sort(rng.choice(n, num_blocks * 2, replace=False))
        for i in range(0, num_blocks * 2, 2):
            start = positions[i]
            end = min(positions[i+1], n)
            h[start:end] = 1.0
    else:
        # Sine wave
        freq = rng.uniform(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2 + phase)
    
    return project_feasible(h, n)

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(y, n):
    target = 0.5 * n
    y = np.nan_to_num(np.asarray(y, dtype=np.float64), nan=0.5, posinf=0.5, neginf=0.5)
    lo = -y.max() - 0.1
    hi = 1.0 - y.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    alpha = (lo + hi) / 2.0
    h = np.clip(y + alpha, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            rooms = 1.0 - h
            order = np.argsort(-rooms)
            for idx in order:
                if residual <= 1e-14:
                    break
                step = min(rooms[idx], residual)
                h[idx] += step
                residual -= step
        else:
            order = np.argsort(-h)
            for idx in order:
                if -residual <= 1e-14:
                    break
                step = min(h[idx], -residual)
                h[idx] -= step
                residual += step
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def gradient_for_lag(h, k, dx, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < k < n:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += (-h[k:])
    elif -n < k < 0:
        r = -k
        g[:n - r] += (1.0 - h[r:])
        g[r:] += (-h[:n - r])
    return dx * g

def compute_lse_gradient(h, c_vals, tau, active_idx):
    c_act = c_vals[active_idx]
    z = (c_act - c_act.max()) / tau
    z = np.clip(z, -100, 100)
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(active_idx)) / len(active_idx)
    else:
        w = w / ws
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(active_idx)):
        if w[i] < 1e-15:
            continue
        m = active_idx[i]
        k = m - (n - 1)
        grad += w[i] * gradient_for_lag(h, k, dx, n)
    return grad

def pick_active(c_vals, max_count=80):
    total = len(c_vals)
    if total <= max_count:
        return np.arange(total)
    top = np.argsort(c_vals)[::-1][:max_count]
    idx_set = set(top.tolist())
    idx_set.add(0)
    idx_set.add(total // 2)
    idx_set.add(total - 1)
    return np.array(sorted(idx_set))

def pgd_optimize(h0, n, max_iter=3000, tau0=0.002, t_limit=None, t0=None):
    h = h0.copy()
    tau = tau0
    c_val, c_vals = evaluate_c5(h)
    best_c = c_val
    best_h = h.copy()
    momentum = np.zeros(n)
    mu = 0.3
    no_improve = 0
    for it in range(max_iter):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        if it % 20 == 0:
            active = pick_active(c_vals, 80)
            if it % 100 == 0:
                c_val, c_vals = evaluate_c5(h)
                if c_val < best_c:
                    best_c = c_val
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                if no_improve > 8:
                    tau = min(tau * 1.5, 0.015)
                    no_improve = 0
                elif tau > tau0 * 1.2:
                    tau = max(tau0, tau * 0.75)
            grad = compute_lse_gradient(h, c_vals, tau, active)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            momentum = mu * momentum + (1 - mu) * grad
            mgn = np.linalg.norm(momentum)
            if mgn < 1e-14:
                break
            best_trial_c = c_val
            best_trial_h = None
            base = 0.8 / mgn
            for fac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                step = base * fac
                y = h - step * momentum
                h_t = project_feasible(y, n)
                c_t, _ = evaluate_c5(h_t)
                if c_t < best_trial_c:
                    best_trial_c = c_t
                    best_trial_h = h_t
                if c_t < best_c:
                    break
            if best_trial_h is not None:
                h = best_trial_h
                c_val, c_vals = evaluate_c5(h)
            else:
                no_improve += 1
                if no_improve > 15:
                    h = best_h.copy()
                    c_val, c_vals = evaluate_c5(h)
                    tau = min(tau * 2.0, 0.02)
                    momentum *= 0.5
                    if no_improve > 40:
                        break
                else:
                    h = best_h.copy()
                    c_val, c_vals = evaluate_c5(h)
    c_final, _ = evaluate_c5(best_h)
    return best_h, c_final

def slsqp_polish(h0, n, max_active=50, t_limit_s=25):
    h = h0.copy()
    dx = 2.0 / n
    target = 0.5 * n
    c_val, c_vals = evaluate_c5(h)
    best_h = h.copy()
    best_c = c_val
    t0 = time.time()
    for rnd in range(4):
        if time.time() - t0 > t_limit_s:
            break
        active = pick_active(c_vals, max_active)
        na = len(active)
        nvars = n + 1
        active_lags = active.astype(int).copy()
        n_len = n

        def obj(x):
            return x[-1]

        def obj_g(x):
            g = np.zeros(nvars)
            g[-1] = 1.0
            return g

        def eq_con(x):
            return np.sum(x[:n_len]) - target

        def eq_con_g(x):
            g = np.ones(nvars)
            g[-1] = 0.0
            return g

        def ineq_cons(x):
            hx = x[:n_len]
            res = np.empty(na)
            for i in range(na):
                m = int(active_lags[i])
                k = m - (n_len - 1)
                if k == 0:
                    f = dx * np.sum(hx * (1.0 - hx))
                elif 0 < k < n_len:
                    f = dx * np.sum(hx[k:] * (1.0 - hx[:n_len - k]))
                elif -n_len < k < 0:
                    r = -k
                    f = dx * np.sum(hx[:n_len - r] * (1.0 - hx[r:]))
                else:
                    f = 0.0
                res[i] = x[-1] - f
            return res

        def ineq_cons_g(x):
            hx = x[:n_len]
            G = np.zeros((na, nvars))
            for i in range(na):
                m = int(active_lags[i])
                k = m - (n_len - 1)
                if k == 0:
                    G[i, :n_len] = dx * (1.0 - 2.0 * hx)
                elif 0 < k < n_len:
                    G[i, k:n_len] += dx * (1.0 - hx[:n_len - k])
                    G[i, :n_len - k] += dx * (-hx[k:])
                elif -n_len < k < 0:
                    r = -k
                    G[i, :n_len - r] += dx * (1.0 - hx[r:])
                    G[i, r:n_len] += dx * (-hx[:n_len - r])
                G[i, -1] = 1.0
            return G

        x0 = np.append(h, c_val + 0.003)
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
        cons = [
            {'type': 'eq', 'fun': eq_con, 'jac': eq_con_g},
            {'type': 'ineq', 'fun': ineq_cons, 'jac': ineq_cons_g}
        ]
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_g,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x[:n], n)
                c_new, c_new_vals = evaluate_c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    c_vals = c_new_vals
        except Exception:
            pass
    return best_h, best_c

def resample_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, new_n)

def make_binary(h, n):
    target = 0.5 * n
    h_sorted = np.sort(h)[::-1]
    if target == int(target):
        thresh = h_sorted[int(target)]
        h_bin = np.where(h >= thresh, 1.0, 0.0)
        n_ones = int(np.sum(h_bin))
        if n_ones > int(target):
            idxs = np.where(h_bin == 1.0)[0]
            h_vals = h[idxs]
            remove = np.argmin(h_vals)
            h_bin[idxs[remove]] = 0.0
        elif n_ones < int(target):
            idxs = np.where(h_bin == 0.0)[0]
            h_vals = h[idxs]
            add = np.argmax(h_vals)
            h_bin[idxs[add]] = 1.0
        return h_bin
    else:
        k = int(target)
        thresh = h_sorted[k]
        h_bin = np.where(h >= thresh, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)
        return h_bin

def random_candidate(n, rng):
    method = rng.integers(4)
    if method == 0:
        h = 0.5 + 0.45 * (2.0 * rng.random(n) - 1.0)
    elif method == 1:
        h = np.zeros(n)
        nb = rng.integers(2, 7)
        starts = np.sort(rng.choice(n, nb, replace=False))
        lens = rng.integers(2, n // 2 + 1, nb)
        for s, l in zip(starts, lens):
            h[s:s + l] = 1.0
    elif method == 2:
        x = np.linspace(0, 2, n)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(np.pi * freq * x + phase)
    else:
        h = rng.random(n)
    return project_feasible(h, n)
