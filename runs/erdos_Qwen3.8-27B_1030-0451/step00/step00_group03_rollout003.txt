def project_feasible(y, target_sum):

    n = _slsq_state['n']
    dx = _slsq_state['dx']

    lo = -float(np.max(y)) - 1.0

    dx = 2.0 / n
    fills = compute_fills(h)
    f_active = fills[active]

                    c5_blend = c5_s4
                if c5_blend < best_c5:
                    best_c5 = c5_blend
                    best_h = h_blend.copy()

    # Final polish
    if remaining() > 10:
        h_s5, c5_s5 = slsqp_polish(best_h, max_active=min(100, 2*len(best_h)-1), maxiter=200)
        if c5_s5 < best_c5:
            best_c5 = c5_s5
            best_h = h_s5.copy()

    # Final verification
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
