def score_exact(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full')
    return float(np.max(c) * dx)

def ensure_feasible(h, n):
    """Project h to box [0,1] and sum = n/2 using bisection on mu."""
    S = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for mu such that sum(clip(h - mu, 0, 1)) = S
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mu = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mu, 0.0, 1.0))
        if s > S:
            lo = mu
        else:
            hi = mu
    mu = 0.5 * (lo + hi)
    y = np.clip(h - mu, 0.0, 1.0)
    # Fix tiny residual
    residual = S - np.sum(y)
    if abs(residual) > 1e-12:
        Interior = (y > 1e-15) & (y < 1 - 1e-15)
        if np.sum(Interior) > 0:
            y[Interior] += residual / np.sum(Interior)
            y = np.clip(y, 0.0, 1.0)
    return y

class SmoothObjective:

        self.tau = tau

        dx = 2.0 / self.n
        q = 1.0 - x
        c = np.correlate(x, q, mode='full')
        scaled = c * dx / self.tau
        scaled -= np.max(scaled)
        return float(self.tau * np.log(np.sum(np.exp(scaled))))

    def __init__(self, n, tau):
        self.n = n
        self.tau = tau

    def __call__(self, x):

            return compute_smooth_gradient(x, self.tau)
        except Exception:
            return np.zeros(self.n)

    def __call__(self, x):

    def __call__(self, x):

            best_h, best_s = slqp_polish(best_h, n, deadline, n_outer=2)
            if time.time() < deadline - 5:
                best_h, best_s = local_search(best_h, n, deadline, rng, max_iter=2000)

    # Final validation
    best_h = ensure_feasible(best_h, n)
    final_s = score_exact(best_h)
    if final_s < best_s:
        best_s = final_s

    return best_h, best_s, n
