def project_feasible(z, S):

def compute_smooth_grad(h, tau):

    q = 1.0 - h
    c = np.correlate(h, q, mode='full')
    phi = c * (2.0 / n)

        q = 1.0 - x
        c = np.correlate(x, q, mode='full')
        phi = c * (2.0 / n)
        mp = np.max(phi)
        return self.tau * (np.log(np.sum(np.exp((phi - mp)/self.tau))) + mp/self.tau)

class SmoothGradFunc:
    def __init__(self, tau):
        self.tau = tau
    def __call__(self, x):
        n = len(x)
        q = 1.0 - x
        c = np.correlate(x, q, mode='full')
        phi = c * (2.0 / n)
        mp = np.max(phi)
        ep = np.exp((phi - mp)/self.tau)
        w = ep / np.sum(ep)
        grad = np.zeros(n)
        dx = 2.0 / n
        for k in range(2*n-1):
            lag = k - (n-1)
            wk = w[k] * dx
            if wk < 1e-16:
                continue
            j1 = max(0, lag)
            j2 = min(n, n+lag)
            if j1 < j2:
                grad[j1:j2] += wk * (1.0 - x[j1-lag:j2-lag])
            j3 = max(0, -lag)
            j4 = min(n, n-lag)
            if j3 < j4:
                grad[j3:j4] += wk * (-x[j3+lag:j4+lag])
        return grad

class SumEq:
    def __init__(self, S):
        self.S = S
    def fun(self, x):
        return np.sum(x) - self.S
    def jac(self, x):
        return np.ones(len(x))

def slsqp_optimize(h0, tau, maxiter=200):
    n = len(h0)
    S = n / 2.0
    bounds = list(zip(np.zeros(n), np.ones(n)))
    sf = SmoothFunc(tau)
    sg = SmoothGradFunc(tau)
    se = SumEq(S)
    try:
        res = minimize(sf, h0.copy(), jac=sg, bounds=bounds,
                      method='SLSQP',
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, S)
    except:
        pass
    return None

def pair_descent(h, max_iters=5000, time_limit=30.0, seed=0):
    t_start = time.time()
    n = len(h)
    S = n / 2.0
    h = h.astype(float).copy()
    cur_s = exact_score(h)
    rng = np.random.RandomState(seed)
    patience = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if patience > 500:
            # Random restart
            h = project_feasible(h + rng.randn(n) * 0.03, S)
            cur_s = exact_score(h)
            patience = 0
            continue
        
        tau = max(2e-4, 0.01 * cur_s)
        _, grad = compute_smooth_grad(h, tau)
        
        improved = False
        # Determine direction: move from high grad to low grad
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        
        if grad[i] - grad[j] < 1e-10:
            patience += 1
            continue
        
        # Direction: h[i] -= d, h[j] += d
        max_d = min(h[i], 1.0 - h[j])
        if max_d < 1e-12:
            # Try reverse
            i, j = j, i
            max_d = min(h[j], 1.0 - h[i])
            if max_d < 1e-12:
                patience += 1
                continue
        
        # Line search
        best_s = cur_s
        best_h = None
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05, 0.01]:
            d = alpha * max_d
            if d < 1e-14:
                continue
            h_try = h.copy()
            h_try[i] -= d
            h_try[j] += d
            s_try = exact_score(h_try)
            if s_try < best_s - 1e-15:
                best_s = s_try
                best_h = h_try
        
        if best_h is not None:
            h = best_h
            cur_s = best_s
            patience = 0
            improved = True
        else:
            # Try a few random pairs
            for _ in range(5):
                ii = rng.randint(n)
                jj = rng.randint(n)
                if ii == jj:
                    continue
                if grad[ii] < grad[jj]:
                    ii, jj = jj, ii
                md = min(h[ii], 1.0 - h[jj])
                if md < 1e-12:
                    continue
                for alpha in [0.5, 0.2, 0.05]:
                    d = alpha * md
                    h_try = h.copy()
                    h_try[ii] -= d
                    h_try[jj] += d
                    s_try = exact_score(h_try)
                    if s_try < cur_s - 1e-15:
                        h = h_try
                        cur_s = s_try
                        patience = 0
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            patience += 1
    
    return h, cur_s

def fw_step(h, tau=None):
    n = len(h)
    S = n / 2.0
    if tau is None:
        tau = max(5e-4, 0.005 * exact_score(h))
    _, grad = compute_smooth_grad(h, tau)
    order = np.argsort(grad)
    u = np.zeros(n)
    n_full = int(np.floor(S))
    frac = S - n_full
    u[order[:n_full]] = 1.0
    if frac > 1e-12:
        u[order[n_full]] = frac
    cur_s = exact_score(h)
    best_s = cur_s
    best_h = h.copy()
    for t in [0.9, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
        h_new = (1-t)*h + t*u
        h_new = project_feasible(h_new, S)
        s_new = exact_score(h_new)
        if s_new < best_s - 1e-14:
            best_s = s_new
            best_h = h_new
    return best_h, best_s

def generate_candidates(h_base, n, seed, num=10):
    """Generate diverse candidates from a base"""
    rng = np.random.RandomState(seed)
    S = n / 2.0
    cands = [h_base.copy()]
    # Complement
    cands.append(project_feasible(1.0 - h_base, S))
    # Random perturbations
    for i in range(num):
        scale = 0.02 + 0.03 * i / num
        c = project_feasible(h_base + rng.randn(n) * scale, S)
        cands.append(c)
    return cands
