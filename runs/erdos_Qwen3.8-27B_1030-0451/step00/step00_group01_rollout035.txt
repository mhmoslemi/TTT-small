    # (1-h[i-k]) if 0 <= i-k < N, minus h[i+k] if 0 <= i+k < N

    # g_k = sum_j h[j+k]*(1-h[j]) for valid j

        if 0 <= i - k < N:

import numpy as np
import time


def project_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)


def compute_lags(h):
    """Compute all lag values: C_k = sum_j h[j+k]*(1-h[j]) * dx."""
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h):
    """Compute C5 = max_k C_k."""
    return compute_lags(h).max()


def coord_optimize(h, i, all_lags):
    """Find optimal h[i] given all other coordinates fixed."""
    N = len(h)
    num_lags = 2 * N - 1
    ks = np.arange(num_lags) - (N - 1)
    
    idx1 = i - ks
    valid1 = (idx1 >= 0) & (idx1 < N)
    t1 = np.zeros(num_lags)
    t1[valid1] = 1.0 - h[idx1[valid1]]
    
    idx2 = i + ks
    valid2 = (idx2 >= 0) & (idx2 < N)
    t2 = np.zeros(num_lags)
    t2[valid2] = -h[idx2[valid2]]

    intercepts = all_lags - slopes * h[i]
    
    lo, hi = 0.0, 1.0

    return h

    """Subgradient descent to minimize C5."""
    N = len(h)
    target_sum = N / 2.0
    dx = 2.0 / N
    h = h.copy()
    ks = np.arange(2 * N - 1) - (N - 1)
    indices = np.arange(N)
    
    for it in range(n_iter):
        all_lags = compute_lags(h)
        max_val = all_lags.max()
        active = all_lags >= max_val - active_eps
        n_active = max(int(active.sum()), 1)

        active_indices = np.where(active)[0]
