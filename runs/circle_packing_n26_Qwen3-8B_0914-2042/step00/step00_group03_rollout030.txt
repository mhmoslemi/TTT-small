import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

    # and boundary circles to utilize corner and edges
    radius_estimate = 0.1
    center_estimate = []

    center_estimate = np.array(center_estimate)

    # Add boundary circles near corners and edges
    boundary_centers = [
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9],
        [0.1, 0.5], [0.5, 0.1], [0.5, 0.9], [0.9, 0.5],
        [0.5, 0.5], [0.2, 0.2], [0.8, 0.2], [0.2, 0.8], [0.8, 0.8]
    ]
    center_estimate = np.vstack([center_estimate, boundary_centers])

    # Initial radii
    radius_estimate = np.full(len(center_estimate), radius_estimate)

    # Define variables: centers (x, y) and radii
    # Final variables: x1, y1, r1, x2, y2, r2, ..., x26, y26, r26
    # Total of 26 * 3 = 78 variables
    n = 26
    x0 = np.hstack([center_estimate[:, 0], center_estimate[:, 1], radius_estimate])

    # Optimization bounds: x, y ∈ [0, 1], radii ≥ 0
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1)])  # x, y, radius for each circle

    # Distance constraints between each pair of circles (non-overlap)
    def distance_constraints(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraint = dist - (radii[i] + radii[j]) >= -1e-8
                constraints.append(constraint)
        return constraints

    # Objective: maximize the sum of radii (equivalent to minimize negative sum of radii)
    def objective(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return -np.sum(radii)

    # Optimization with constraints
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=opt.Constraint(
            lambda x: np.array(distance_constraints(x)),

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
