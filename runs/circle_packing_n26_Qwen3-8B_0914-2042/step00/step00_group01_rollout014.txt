import numpy as np
import scipy.optimize as opt

def run_packing():
    # Step 1: Create an initial hexagonal packing inside unit square
    # Let's try a 7-row hexagonal packing where each row alternates offset
    # We'll fit as many rows as we can in the vertical space of 1

    # Determine number of rows and columns
    rows = 5
    cols = 6  # Starting with 6 columns in the bottom row

    # Calculate spacing between centers
    circle_diameter = 1.0 / (rows + 1)  # The vertical spacing
    col_spacing = circle_diameter * np.sqrt(3) / 2  # Horizontal spacing in staggered hexagonal grid

    # Initialize coordinates
    centers = np.zeros((rows * cols, 2))
    radii = np.zeros(rows * cols)

    # Generate initial staggered hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * col_spacing
            y = i * circle_diameter
            if i % 2 == 1:
                x += col_spacing / 2
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = circle_diameter / 2

    # Step 2: Set up optimization problem using scipy.optimize
    # We will use a constrained optimization with bounds

    # Flatten centers and radii for optimization variables
    n = 26
    flat_centers = centers.reshape(-1)
    flat_radii = radii.reshape(-1)

    # Optimization variables: x, y, r for each circle
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 1.0))  # r

    # Objective function to maximize
    def objective(x):
        return -np.sum(x[2::3])  # We maximize by minimizing the negative sum

    # Constraints: boundary checks and non-overlap
    def constraint_func(x):
        centers = x[0::3].reshape(-1, 2)
        radii = x[2::3]

        # Check if centers are within [0, 1] and radius >= 0
        for i in range(n):
            x_i = centers[i, 0]
            y_i = centers[i, 1]
            r_i = radii[i]
            if (x_i < 0 - 1e-8 or x_i > 1 + 1e-8 or
                y_i < 0 - 1e-8 or y_i > 1 + 1e8):
                return -np.inf
            if r_i < 0:
                return -np.inf

        # Check if circles are within the square
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < 0 - 1e-8 or x_i + r_i > 1 + 1e-8 or

                return -np.inf

        # Check for non-overlapping between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf

        return 0.0

    # Use SLSQP for constrained optimization
    result = opt.minimize(
        objective,
        np.concatenate([flat_centers, flat_radii]),
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-10}
    )

    # Extract the result
    optimized = result.x
    centers_opt = optimized[0::3].reshape(-1, 2)
    radii_opt = optimized[2::3]

    # Return result
    return centers_opt, radii_opt, np.sum(radii_opt)
