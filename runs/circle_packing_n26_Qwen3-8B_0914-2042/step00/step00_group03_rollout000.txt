import scipy.optimize as optimize
import sys

def validate_packing(centers, radii):

            if dist < radii[i] + radii[j] - 1e-12:

def get_hexagonal_centers(n):

    Returns hexagonal grid of centers for n circles in a unit square.
    '''
    centers = []
    for i in range(n):
        row = i // 6
        col = i % 6
        x = col * (1.0 / 4.0) + 0.25

        centers.append([x, y])
    return np.array(centers)

def initial_packing(n_circles):
