from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import warnings

def run_packing():
    # Grid layout with 5 rows and 5 columns (including diagonals)
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # First 5 rows, first 5 columns

            x = j / 4.0

            centers[i * 5 + j] = [x, y]

    # Optimize the circle positions to maximize sum of radii

        # params: array of 2*num_circles elements

        radii_flat = params[2 * num_circles:]

            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return float('inf')  # Penalize boundary violations

        # Compute pairwise distances

        total_overlap = 0

                    # Overlap, penalize

        # Total objective is to maximize sum of radii, which is equivalent to minimizing -sum(radii)
        total_sum = np.sum(radii_flat)
        return -total_sum + total_overlap  # Minimize this to maximize sum of radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
