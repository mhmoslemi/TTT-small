import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial approach: square grid with hexagonal adjustments
    n = 26
    # Placeholder for centers: grid-based with some hexagonal corrections
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial grid with spacing to prevent overlap
    spacing = 0.2
    col_count = 5
    row_count = 5

    # Initialize centers in a grid and adjust for hexagonal packing
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col * spacing + 0.05
        y = row * spacing + 0.05
        centers[i, 0] = x
        centers[i, 1] = y
    
    # Initial uniform radius based on spacing
    # We'll optimize it later
    radii = np.full(n, spacing / 2)
    
    # Define our objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum

    # Define constraint functions
    def constraint_overlap(radii, centers):
        n = centers.shape[0]
        dist = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=-1))
        return dist - (radii[:, None] + radii[None, :]) >= -1e-10

    def constraint_boundary(radii, centers):
        return np.all(centers - radii[:, None] >= -1e-10) & np.all(centers + radii[:, None] <= 1 + 1e-10)

    # Create bounds for the radii: non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Define variables to be optimized (radii)
    variables = radii

    # Optimize radii given centers
    result = minimize(
        fun=objective,
        x0=variables,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap, 'args': (centers,)},
            {'type': 'ineq', 'fun': constraint_boundary, 'args': (centers,)}
        ],
        method='SLSQP'
    )

    # Extract centers and radii
    optimized_radii = result.x
    return centers, optimized_radii, np.sum(optimized_radii)
