import numpy as np
import cvxpy as cp

def run_packing():
    n = 26

    # Define initial center grid (equally spaced)
    x_coords = np.linspace(0.1, 0.9, n)
    y_coords = np.linspace(0.1, 0.9, n)

    # Create initial guess for centers
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = x_coords[i]
        centers[i, 1] = y_coords[i]

    # Define variables for radii
    radii = cp.Variable(n, nonneg=True)

    # Objective: Maximize sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints: non-overlapping
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Constraints: circles within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i] >= 0)
        constraints.append(x + radii[i] <= 1)
        constraints.append(y - radii[i] >= 0)
        constraints.append(y + radii[i] <= 1)

    # Create and solve the problem
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-5)
    except:
        # If solver fails, return a simple fallback (this is unlikely with 26 circles)
        return np.array([[0.5, 0.5] for _ in range(n)], dtype=np.float64), np.zeros(n), 0.0

    if prob.status != 'optimal':
        raise RuntimeError(f"Optimization failed with status: {prob.status}")

    centers = np.array(centers, dtype=np.float64)
    radii = np.array(radii.value, dtype=np.float64)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
