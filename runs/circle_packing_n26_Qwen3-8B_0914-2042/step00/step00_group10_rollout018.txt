from scipy.spatial.distance import pdist, squareform
from itertools import combinations

    def hexagonal_layout(n_circles):

            max_row += 1
        max_row = int(np.ceil(np.sqrt(2 * n_circles)))

                grid.append((i, j))
                if i < max_row - 1:
                    grid.append((i + 1, j))
        return grid

    def normalize_grid(grid):

        y_min, y_max = min(y), max(y)
        scale = 1.0 / max(x_max - x_min, y_max - y_min)
        return np.array([[x * scale, y * scale] for x, y in grid])
