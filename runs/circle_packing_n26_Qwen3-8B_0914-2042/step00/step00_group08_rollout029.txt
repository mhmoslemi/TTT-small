import numpy as np
import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26

    # Define variables for center coordinates (x, y) and radius r
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: All circles fully inside [0,1]^2
    constraints = []

    # For each circle, ensure its center and radius are within [0,1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Ensure non-overlapping between each pair of circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Define the problem
    prob = cp.Problem(objective, constraints)

    # Solve the problem
    prob.solve(solver=cp.SCS, eps=1e-6)

    # Extract the optimal solution
    centers_opt = centers.value
    radii_opt = radii.value

    # Compute the total sum of radii
    total_sum = np.sum(radii_opt)

    return (centers_opt, radii_opt, total_sum)
