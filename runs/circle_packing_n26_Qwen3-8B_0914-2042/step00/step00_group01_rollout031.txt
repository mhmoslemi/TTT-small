import cvxpy as cp

def run_packing():

    grid_points.append([0.5, 0.5])
    centers = np.array(grid_points)
    radii = np.zeros(num_circles)

    # Set up the optimization problem
    optimization_vars = [cp.Variable(2) for _ in range(num_circles)]
    radii_vars = cp.Variable(num_circles)

    # Constraints
    constraints = []
    for i in range(num_circles):
        x_i, y_i = optimization_vars[i]
        r_i = radii_vars[i]
        # Ensure circle stays within the square
        constraints += [
            x_i - r_i >= 0,
            x_i + r_i <= 1,
            y_i - r_i >= 0,
            y_i + r_i <= 1
        ]

    # Non-overlap constraints between all pairs
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            x_i, y_i = optimization_vars[i]
            x_j, y_j = optimization_vars[j]
            r_i = radii_vars[i]
            r_j = radii_vars[j]
            # Distance between centers
            dist_sq = (x_i - x_j) ** 2 + (y_i - y_j) ** 2
            # Ensure non-overlap
            constraints.append(dist_sq >= (r_i + r_j) ** 2 - 1e-8)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii_vars))

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, eps=1e-6)

    # Extract the results
    centers_opt = np.array([var.value for var in optimization_vars])
    radii_opt = radii_vars.value

    def validate_packing(centers, radii):

        if np.isnan(centers).any() or np.isnan(radii).any():

            if (x - r < -1e-12 or x + r > 1 + 1e-12
