import numpy as np

def initial_approximation(n):
    """Generate an approximate hexagonal grid of n circles with radii r."""
    radii = np.full(n, 0.05)
    centers = []
    for i in range(n):
        x = (i % 7) * 0.2 + 0.1
        y = (i // 7) * (np.sqrt(3)/2 * 0.2) + 0.1
        centers.append([x, y])
    return np.array(centers), radii

def distance(a, b):
    """Compute Euclidean distance between two points."""
    return np.sqrt((a[0] - b[0])**2 + (a[1] - b[1])**2)

def find_conflicts(centers, radii, tolerance=1e-8):
    """Find index pairs with overlapping circles."""
    n = centers.shape[0]
    conflicts = []
    for i in range(n):
        for j in range(i + 1, n):
            d = distance(centers[i], centers[j])
            if d < radii[i] + radii[j] - tolerance:
                conflicts.append((i, j))
    return conflicts

def optimize_centers(centers, radii, max_iterations=10000, cooling_rate=0.995, temp=1000):
    """Perform simulated annealing to optimize circle positions."""
    n = centers.shape[0]
    for _ in range(max_iterations):
        # Random move: shift a circle slightly
        i = np.random.randint(n)
        old_x, old_y = centers[i]
        dx, dy = np.random.uniform(-0.01, 0.01, size=2)
        new_x = np.clip(old_x + dx, 0, 1)
        new_y = np.clip(old_y + dy, 0, 1)
        centers[i] = [new_x, new_y]

        # Check conflicts
        conflicts = find_conflicts(centers, radii)
        if not conflicts:
            continue

        # Compute energy (sum of overlaps)
        energy = 0.0
        for (i, j) in conflicts:
            d = distance(centers[i], centers[j])
            energy += (radii[i] + radii[j] - d) ** 2

        # Accept move with probability based on temperature
        prob = np.exp(-energy / temp)
        if np.random.rand() < prob:
            continue
        else:
            # Revert the move
            centers[i] = [old_x, old_y]

        # Cool down
        temp *= cooling_rate
    return centers

def run_packing():

    centers, radii = initial_approximation(26)
    
    # Optimize center positions
    centers = optimize_centers(centers, radii)

        x, y = centers[i]
        max_r = 0.0
        # Try moving outward in 4 directions
        for dx, dy in [(1, 0), (0, 1), (-1, 0), (0, -1)]:
            new_x = x + dx * 0.01
            new_y = y + dy * 0.01
            if not (0 <= new_x <= 1 and 0 <= new_y <= 1):
                continue
            new_r = 0.01
            # Check for overlaps with other circles
            overlap = False
            for j in range(n):
                if i == j:
                    continue

                if d < final_radii[i] + final_radii[j] - 1e-8:
                    overlap = True
                    break
            if not overlap:
                max_r = max(max_r, 0.01)
        final_radii[i] = max(final_radii[i], max_r)

    # Final check for boundary and overlaps
    for i in range(n):
        x, y = centers[i]
        r = final_radii[i]
