import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial placement in hexagonal packing
    num_circles = 26
    rows = 6
    cols = 5
    spacing = 0.5  # initial spacing to determine the grid
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Create an initial hexagonal grid in a square
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (i % 2) * spacing / 2
            y = i * (np.sqrt(3) / 2) * spacing
            if y > 1:
                y = 1 - (y - 1) % 1
            centers[i * cols + j] = [x, y]

    # Rescale centers to fit the unit square
    centers /= np.max(centers, axis=0)  # scale to unit square
    sizes = np.ones(num_circles) * 0.05  # initial radii

    # Define a function to compute the objective (sum of radii)
    def objective(radii):
        return -np.sum(radii)  # maximize sum of radii

    # Define a function to compute constraints: non-overlapping and within bounds
    def constraints(radii):
        # Ensure circles are entirely within the unit square
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if not (x - r >= 0 and x + r <= 1 and y - r >= 0 and y + r <= 1):
                return False, f"Circle {i} is outside the square"
        # Ensure circles do not overlap

                x1, y1 = centers[i]
                x2, y2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2 - 1e-10:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    # Optimization: Use scipy.optimize to maximize sum of radii
    result = optimize.minimize(
        fun=lambda r: objective(r),
        x0=sizes,
        bounds=[(0, 1) for _ in range(num_circles)],
        constraints=[{
            'type': 'ineq',
            'fun': lambda r: 0.0,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
