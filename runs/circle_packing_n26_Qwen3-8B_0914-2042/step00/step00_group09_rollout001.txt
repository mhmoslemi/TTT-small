import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.optimize.constraints import Constraint

def run_packing():
    # Initial configuration: place 26 circles in a regular grid with some gaps
    # The idea is to form a hybrid of corner-based and center-based packing
    # Create a 5x5 grid, remove 3 circles for better spacing, and re-distribute
    
    # Step 1: Create a 5x5 grid of circles for an initial configuration
    # Reference: https://www.math.ucdavis.edu/~saito/data/152/IndianPacking.pdf
    
    # Create 6 rows and 5 columns with some spacing
    # We'll use a grid that allows for some flexibility in input
    
    # Create initial random centers within [0, 1]×[0, 1]
    # Start with a grid-like solution
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Start with regular packing
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col / 4.0 + 0.1
        y = row / 4.0 + 0.1
        centers[i] = [x, y]
    
    # Set an initial radius to allow for optimization
    radii[:] = 0.05
    
    # Define objective function
    def objective(x):
        # x is flattened version of (centers, radii)
        centers_flat = x[:52].reshape((26, 2))
        radii_flat = x[52:]
        return -np.sum(radii_flat)  # Minimize negative of sum to maximize

    # Define constraints
    def constraint1(x):
        # Boundary constraints
        centers_flat = x[:52].reshape((26, 2))
        radii_flat = x[52:]
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r = radii_flat[i]
            if x_i - r < -1e-12 or x_i + r > 1 + 1e-12 or y_i - r < -1e-12 or y_i + r > 1 + 1e-12:
                return -1
        return 0

    def constraint2(x):
        # Circle separation constraints
        centers_flat = x[:52].reshape((26, 2))
        radii_flat = x[52:]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1
        return 0

    # Combine constraints
    cons = [Constraint(lambda x: constraint1(x), name='boundary'),
            Constraint(lambda x: constraint2(x), name='no_overlap')]

    # Initial guess: x = [centers, radii]
    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds: centers between 0 and 1, radii >= 0
    bounds = Bounds([0.0, 0.0], [1.0, 1.0])  # for centers
    bounds = Bounds([0.0] * 52, [1.0] * 52)  # for centers and radii

    # Run the optimization
    result = minimize(

        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract the results
    centers_opt = result.x[:52].reshape((26, 2))
    radii_opt = result.x[52:]
    
    return centers_opt, radii_opt, np.sum(radii_opt)
