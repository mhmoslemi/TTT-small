import cvxpy as cp

def run_packing():

    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []
    for i in range(n):
        # Circle must lie within the unit square
        constraints.append(centers[i, 0] - radii[i] <= 1 + 1e-6)

    prob = cp.Problem(objective, constraints)
