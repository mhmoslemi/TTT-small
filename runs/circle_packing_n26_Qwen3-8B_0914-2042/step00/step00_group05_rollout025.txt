import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Define initial configuration: 4 rows of 6 circles, with 1 extra in the middle
    # This is a hybrid hexagonal arrangement with some circles near the corners
    centers = np.zeros((26, 2))
    
    # Row 0: 6 circles
    for i in range(6):
        centers[i, 0] = i * (1 / 6)
        centers[i, 1] = 0.05
    
    # Row 1: 6 circles
    for i in range(6):
        centers[6 + i, 0] = i * (1 / 6)
        centers[6 + i, 1] = 0.3
    
    # Row 2: 6 circles
    for i in range(6):
        centers[12 + i, 0] = i * (1 / 6)
        centers[12 + i, 1] = 0.55
    
    # Row 3: 6 circles
    for i in range(6):
        centers[18 + i, 0] = i * (1 / 6)
        centers[18 + i, 1] = 0.8
    
    # Extra circle in the middle
    centers[24, 0] = 0.5
    centers[24, 1] = 0.5
    
    # Initialize radii
    radii = np.full(26, 0.05)
    
    # Define initial bounds for center positions
    bounds = [(0.0, 1.0) for _ in range(26 * 2)]
    
    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Define constraints: non-overlapping between circles
    def constraint(radii):
        dist = squareform(pdist(centers))
        overlaps = np.sum(dist < radii[:, np.newaxis] + radii[np.newaxis, :] - 1e-8)
        return overlaps
    
    # Optimization problem
    problem = {
        'n_vars': 26,
        'x0': np.concatenate((centers.flatten(), radii)),
        'bounds': Bounds([0.0]*26*2, [1.0]*26*2),
        'constraints': [{'type': 'ineq', 'fun': lambda x: 26 - constraint(x[26*2:]), 'jac': None}]
    }
    
    # Optimization settings
    options = {'ftol': 1e-8, 'maxiter': 500, 'disp': False}
    
    # Run the optimization
    result = minimize(
        fun=lambda x: objective(x[26*2:]),
        x0=np.concatenate((centers.flatten(), radii)),
        bounds=Bounds([0.0]*26*2, [1.0]*26*2),
        constraints=[{'type': 'ineq', 'fun': lambda x: 26 - constraint(x[26*2:]), 'jac': None}],
        options=options
    )
    
    # Extract centers and radii
    optimized_centers = result.x[:26*2].reshape((26, 2))
    optimized_radii = result.x[26*2:]
    
    # Validate the packing
    valid, message = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")
    
    # Compute the total sum of radii
    total_sum = np.sum(optimized_radii)
    
    return (optimized_centers, optimized_radii, total_sum)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
