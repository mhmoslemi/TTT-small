def run_packing():

    # Split into 3 rows of 9 circles and 1 row of 8 circles to maximize compactness
    grid_rows = 3
    grid_cols = 9
    spacing = 1.0 / (grid_rows + 1)
    centers[0:grid_rows * grid_cols] = np.array([
        [j * spacing + spacing / 2, i * spacing + spacing / 2]
        for i in range(grid_rows)
        for j in range(grid_cols)
    ])

    centers[grid_rows * grid_cols:grid_rows * grid_cols + 8] = np.array([
        [j * spacing + spacing / 2, grid_rows * spacing + spacing / 2]
        for j in range(8)
    ])

    centers[-1] = [0.95, 0.95]

    # Function to validate radii and centers
    def is_valid(radii):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or

                return False

            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Objective to maximize sum of radii
    def objective(radii):
        if not is_valid(radii):
            return -np.inf
        return np.sum(radii)

    initial_guess = np.array([0.1] * num_circles)

    # Run optimization
    result = minimize(lambda r: -objective(r), initial_guess, bounds=bounds, method='L-BFGS-B', tol=1e-8)

    radii = result.x

    for i in range(num_circles):
