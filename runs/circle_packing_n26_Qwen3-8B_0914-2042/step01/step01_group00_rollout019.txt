import cvxpy as cp

def run_packing():

    # The position of each circle will be adjusted to fit within [0,1]x[0,1]

    grid = []

            # Adjust to fit within [0, 1] with some padding for boundary constraints
            x = col * 0.2887 + 0.1
            y = row * (np.sqrt(3)/2) * 0.2887 + 0.1
            grid.append((x, y))

    # Convert grid to center coordinates
    centers = np.array(grid)

    # No initial guess for radii, as we'll optimize from scratch

    radii = cp.Variable(num_circles, non_negative=True)

    # Constraints
    constraints = []
    for i in range(num_circles):

        x, y = centers[i]
