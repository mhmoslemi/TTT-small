from scipy.optimize import minimize, Bounds
from typing import Tuple

            y = i * 0.15 + 0.05
            grid_centers.append([x, y])

    # Final set of centers with additional circles in corners and boundary regions
    corners = np.array([
        [0.05, 0.05], [0.95, 0.05], [0.95, 0.95], [0.05, 0.95]
    ])

        [0.5, 0.05], [0.5, 0.95], [0.05, 0.5], [0.95, 0.5]
    ])

        [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75], [0.5, 0.5]
    ])

    # Initial guess for all radii
    x0 = np.full(26, 0.05)

    # Objective function to maximize total radius
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    def constraint_func(x):
        constraints = []

                dy = final_centers[i, 1] - final_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)

    def edge_constraint_func(x):
        edge_constraints = []
