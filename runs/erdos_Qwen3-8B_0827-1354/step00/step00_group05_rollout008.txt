    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h0 = h_initial.copy()
    
    # Optimization settings
    # Using SLSQP method with numerical gradient
    result = opt.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    # Check if optimization was successful
    if result.success:
        h_values = result.x
        c5_bound = result.fun
    else:
        # If optimization fails, return initial guess
        h_values = h0
        c5_bound = objective(h0)
    
    return (h_values, c5_bound, n_points)
