import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Pre-imported variables
initial_h_values = np.array([...])  # Placeholder for the actual initial values

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    # Compute cross-correlation
    corr = correlate(h, one_minus_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def objective_with_penalty(h, n_points, penalty_coeff=1e6):
    c5 = compute_c5(h, n_points)
    sum_h = np.sum(h)
    penalty = penalty_coeff * (sum_h - n_points / 2) ** 2
    return c5 + penalty

def run(seed=42, budget_s=100, **kwargs):
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    # Optimization parameters
    options = {'maxiter': 100, 'disp': False, 'ftol': 1e-6, 'xatol': 1e-6}
    
    # Run optimization
    result = minimize(
        objective_with_penalty,
        h_initial,
        args=(n_points,),
        method='Nelder-Mead',
        bounds=bounds,
        options=options
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # If time is exceeded, return the initial solution
        return (initial_h_values, compute_c5(initial_h_values, n_points), n_points)
    
    h_values = result.x
    c5_bound = compute_c5(h_values, n_points)
    return (h_values, c5_bound, n_points)
