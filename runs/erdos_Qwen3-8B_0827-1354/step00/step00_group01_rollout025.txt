    # Ensure the sum constraint is satisfied
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # If SLSQP didn't improve much, try differential evolution
    # Define objective for DE with penalty
    def objective_de(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_corr * dx + penalty
    
    # Run differential evolution
    result_de = differential_evolution(objective_de, bounds, strategy='best1bin',
                                      popsize=15, mutation=(0.5, 1), recombination=0.8,
                                      tol=1e-6, maxiter=50, seed=seed)
    
    best_h_de = result_de.x
    best_c5_de = objective_de(best_h_de)
    
    # Compare results and choose the better one
    if best_c5 < best_c5_de:
        best_h_final = best_h
        best_c5_final = best_c5
    else:
        best_h_final = best_h_de
        best_c5_final = best_c5_de
    
    # Ensure the final solution satisfies the sum constraint
    best_h_final = best_h_final / np.sum(best_h_final) * (n_points / 2)
    
    return best_h_final, best_c5_final, n_points
