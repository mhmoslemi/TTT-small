import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)

        c5 = np.max(corr * dx)

        penalty = 1e6 * abs(np.sum(h) - n_points / 2)
