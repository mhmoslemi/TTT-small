def run(seed=42, budget_s=100, **kwargs):

    from scipy.optimize import minimize

    # Initial guess provided

    h = initial_h.copy()

    scaling_factor = (n_points / 2.0) / sum_h
    h = h * scaling_factor

    # Define the objective function
    def objective(x):
        h_new = x.reshape(n_points)
        # Ensure h_new is within [0, 1]
        h_new = np.clip(h_new, 0.0, 1.0)
        # Normalize sum
        sum_h = np.sum(h_new)
        scaling_factor = (n_points / 2.0) / sum_h
        h_scaled = h_new * scaling_factor
        # Compute correlate
        correlate_result = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        integral_values = correlate_result * dx
        return np.max(integral_values)

    # Define constraints
    def constraint(x):

        return np.sum(h_new) - n_points / 2.0
