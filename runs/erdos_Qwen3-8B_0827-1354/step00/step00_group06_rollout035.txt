    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    h0 = initial_h.copy()
    
    # Start optimization
    start_time = time.time()
    result = minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h0, objective(h0), n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
