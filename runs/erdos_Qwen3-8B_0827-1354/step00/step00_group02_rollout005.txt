        return max_overlap + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run differential evolution with parameters tuned for speed
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    best_h = result.x
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
