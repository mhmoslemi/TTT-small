from scipy.optimize import minimize

def evaluate_c5(h, n_points):

    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=100, **kwargs):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization with L-BFGS-B
    result = minimize(
        lambda h_flat: evaluate_c5(h_flat, n_points),
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(corr) * dx
    
    return best_h, c5, n_points
