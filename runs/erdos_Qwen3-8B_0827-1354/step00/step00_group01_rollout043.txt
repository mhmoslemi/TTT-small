        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Find the maximum value
        max_corr = np.max(corr)
        return max_corr * dx

    # Constraints: sum(h) = n_points / 2
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    h_initial = initial_h.copy()

    # Perform optimization
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)

    # Extract the best solution
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
