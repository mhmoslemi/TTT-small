def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values.copy()
    target_sum = n_points / 2

    def evaluate(individual):
        dx = 2.0 / n_points
        corr = np.correlate(individual, 1 - individual, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def adjust_sum(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.full_like(h, target_sum / n_points)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def mutate(h, mutation_rate=0.1, delta=0.05):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-delta, delta)
                h[i] = np.clip(h[i], 0, 1)
        return adjust_sum(h)

    def crossover(parent1, parent2):
        child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
        return adjust_sum(child)

    def generate_population(size):
        population = [initial_h.copy()]
        for _ in range(size - 1):
            perturbed = initial_h.copy()

                    perturbed[i] = np.clip(perturbed[i], 0, 1)
            perturbed = adjust_sum(perturbed)
            population.append(perturbed)
        return population

    population = generate_population(50)
