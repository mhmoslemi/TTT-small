        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        seed=seed
    )
    
    # Extract the best solution
    best_h = result.x
    # Calculate the actual C5 value by subtracting the penalty
    c5_bound = result.fun - 1e4 * abs(np.sum(best_h) - n_points / 2)
    
    return (best_h, c5_bound, n_points)
