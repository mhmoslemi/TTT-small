    if result.success and time.time() - start_time < budget_s:
        best_h = result.x
        best_c5 = result.fun
    else:
        # Fallback to initial guess if optimization fails or exceeds time
        best_h = initial_h
        best_c5 = objective(initial_h)

    # Ensure the solution is within bounds and satisfies the constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    constraint_val = np.sum(best_h) - n_points / 2
    if abs(constraint_val) > 1e-6:
        best_h = initial_h
        best_c5 = objective(initial_h)

    return (best_h, best_c5, n_points)
