    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Objective function
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        overlap = corr * dx
        return np.max(overlap)
    
    # Gradient function using finite differences
    def objective_grad(h_vec):
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            obj_plus = objective(h_plus)
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            obj_minus = objective(h_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=objective_grad,
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        # If time is up, return the initial guess
        return h, objective(h), n_points
    
    if result.success:
        best_h = result.x
        best_c5 = objective(best_h)
    else:
        best_h = h
        best_c5 = objective(h)
    
    return best_h, best_c5, n_points
