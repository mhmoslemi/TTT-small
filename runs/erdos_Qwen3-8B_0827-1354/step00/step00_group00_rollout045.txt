def run(seed=42, budget_s=100, **kwargs):

    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Bounds for each h[i] are [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Initial guess
    h_initial = initial_h.copy()

    # Start timing
    start_time = time.time()

    # Perform optimization with COBYLA method

    # Check if we have exceeded the time budget
