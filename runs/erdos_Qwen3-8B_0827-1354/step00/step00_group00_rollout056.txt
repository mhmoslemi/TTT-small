def run(seed=42, budget_s=100, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    import random

    np.random.seed(seed)
    # Initial h values
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()

    # Define objective function
    def objective(h):
        h_array = np.array(h)
        shifted = 1 - h_array
        corr = np.correlate(h_array, shifted, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / len(h)
        return max_corr * dx

    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization method
    method = 'SLSQP'  # or 'COBYLA'

    # Run optimization with initial guess
    result_initial = minimize(objective, h_initial, method=method, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])

    # Generate perturbed initial points
    perturbations = 3  # number of perturbations to try
    best_result = result_initial
    best_c5 = best_result.fun

    for _ in range(perturbations):
        # Perturb the initial solution
        perturbed = h_initial + np.random.normal(0, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        scale_factor = (n_points / 2) / current_sum
        perturbed *= scale_factor
        # Run optimization on this perturbed solution
        result_perturbed = minimize(objective, perturbed, method=method, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])
        if result_perturbed.fun < best_c5:
            best_result = result_perturbed
            best_c5 = result_perturbed.fun

    # Also, try a random initial solution
    # Generate a random solution
    random_h = np.random.rand(n_points)
    random_h = random_h / np.sum(random_h) * (n_points / 2)
    # Clip to [0,1] and adjust sum
    random_h = np.clip(random_h, 0, 1)
    current_sum = np.sum(random_h)
    scale_factor = (n_points / 2) / current_sum
    random_h *= scale_factor
    # Run optimization
    result_random = minimize(objective, random_h, method=method, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])
    if result_random.fun < best_c5:
        best_result = result_random
        best_c5 = result_random.fun

    # Return the best result
    return (best_result.x, best_c5, n_points)
