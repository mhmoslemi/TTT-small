import numpy as np

    # Set number of points and compute dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Equality constraint: sum(h) == n_points / 2
    def constraint_func(h):
        return np.sum(h) - n_points / 2

        return np.max(corr) * dx

        corr = np.correlate(h, 1 - h, mode='full')

        k = max_index - (n_points - 1)
