    # Linear constraint for sum(h) = n_points / 2
    cons = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Perform optimization with L-BFGS-B method for efficiency
    result = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 1000}
    )
    
    # Check if optimization was successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
        return (best_h, best_c5, n_points)
    else:
        # If optimization failed, return initial guess
        return (initial_h, objective(initial_h), n_points)
