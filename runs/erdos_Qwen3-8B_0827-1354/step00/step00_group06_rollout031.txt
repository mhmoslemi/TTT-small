        )
        # Check if the optimization was successful
        if not result.success:
            break
        # Update h with the new solution
        h = result.x
        # Compute the current C5 value
        current_corr = np.correlate(h, h, mode='full')
        current_min_corr = np.min(current_corr)
        current_c5 = 1.0 - current_min_corr * dx
        # Update best_c5 if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
