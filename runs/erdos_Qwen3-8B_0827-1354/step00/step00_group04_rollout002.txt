import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):
    # Initial guess provided as global variable
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function
    def objective(h):
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Gradient approximation using finite differences
    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-6
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            obj_plus = objective(h_plus)
            h_minus = h.copy()
            h_minus[j] -= eps
            obj_minus = objective(h_minus)
            grad[j] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    # Constraints
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    ]
    
    # Bounds for each h[i]
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Optimization settings
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
