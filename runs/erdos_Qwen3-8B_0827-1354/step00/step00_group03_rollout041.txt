    cons = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    # Run optimization
    result = minimize(
        compute_c5,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 1000}
    )
    # Extract the best solution
    h_values = result.x
    c5_bound = compute_c5(h_values, n_points)
    return h_values, c5_bound, n_points
