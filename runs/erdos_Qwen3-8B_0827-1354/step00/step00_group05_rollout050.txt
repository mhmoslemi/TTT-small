        sum_h = np.sum(h_mut)
        scale_factor = (n_points / 2.0) / sum_h
        h_mut = h_mut * scale_factor
        h_mut = np.clip(h_mut, 0.0, 1.0)
        return h_mut

    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    # Number of generations
    generations = 100
    for gen in range(generations):
        # Generate a new candidate by mutating the best_h
        new_h = mutate(best_h)
        new_c5 = evaluate_c5(new_h)
        # Update best if improvement
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        # Check time
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
