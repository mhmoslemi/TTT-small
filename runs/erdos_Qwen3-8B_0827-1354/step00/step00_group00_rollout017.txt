        tol=1e-6,
        maxiter=100,
        polish=True,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return initial_h, objective(initial_h) - 1e4 * abs(np.sum(initial_h) - n_points / 2), n_points

    if result.success:
        h_values = result.x
        c5_bound = result.fun - 1e4 * abs(np.sum(h_values) - n_points / 2)
    else:
        h_values = initial_h
        c5_bound = objective(initial_h)

    return h_values, c5_bound, n_points
