        scaling_factor = target_sum / current_sum
        h *= scaling_factor
    h = np.clip(h, 0, 1)
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B optimizer
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 1000})
    
    h_opt = result.x
    c5_bound = result.fun
    
    return h_opt, c5_bound, n_points
