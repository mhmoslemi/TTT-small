def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function
    def objective(h_vec):
        h_array = np.array(h_vec)
        cross_corr = np.correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(cross_corr)
        return max_corr * dx
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = np.array(initial_h)
    
    # Optimization
    start_time = time.time()
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # Return initial solution if time is up
        return (initial_h, evaluate_erdos_solution(initial_h), n_points)
    
    optimized_h = result.x
    c5_bound = result.fun
    
    return (optimized_h, c5_bound, n_points)
