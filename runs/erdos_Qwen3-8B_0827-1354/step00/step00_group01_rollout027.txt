    dx = 2.0 / n_points
    
    # Resample initial_h_values to n_points using linear interpolation
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    h_initial = np.interp(x_new, x_initial, initial_h_values)
    
    # Scale to satisfy the integral constraint
    desired_sum = n_points / 2
    current_sum = np.sum(h_initial)
    scale_factor = desired_sum / current_sum
    h_initial *= scale_factor
    
    # Objective function: maximize the overlap integral
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    
    # Extract best solution
    best_h = result.x
    best_c5 = result.fun
    
    return (best_h, best_c5, n_points)
