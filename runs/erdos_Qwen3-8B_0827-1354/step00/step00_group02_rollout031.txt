    initial_h = initial_h_values
    
    def objective(h):
        correlate_result = np.correlate(h, 1 - h, mode='full')
        max_correlate = np.max(correlate_result)
        penalty = 1e4 * abs(np.sum(h) - n_points / 2)
        return max_correlate * dx + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
