        # Compute cross-correlation between h and 1 - h
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Define constraints: sum(h) == n_points / 2
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
