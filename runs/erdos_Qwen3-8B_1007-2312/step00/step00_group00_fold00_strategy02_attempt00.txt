    return normalized.tolist()

def initialize_random_fully_perturbed(parent=None, delta_perturbation=0.1, num_perturbations=5):
    n = N_POINTS
    half = n // 2

    # Split into two regions
    region1 = np.random.rand(half) * (delta_perturbation/2) + (1 - delta_perturbation/2)/2
    region2 = np.random.rand(half) * (delta_perturbation/2) + (1 - delta_perturbation/2)/2

    h = np.concatenate((region1, region2))

    current_sum = np.sum(h) * (1) / N_POINTS
    adjust_needed = TOTAL_INTEGRAL - current_sum

    if adjust_needed != 0:
        adjust_amount = (adjust_needed / (1 - 0)) * (1 / N_POINTS) * delta_perturbation
        factor = 1 + (np.random.randn(1)[0] * adjust_amount)
        h = h * factor
        h = np.clip(h, 0, 1)
        total_current = np.sum(h) * (1) / N_POINTS
        h = h * (TOTAL_INTEGRAL / total_current)

    return h.tolist()

def evaluate_erdos(h_values):
    """Compute the overlap integral C5 according to the provided evaluator."""
    try:
        dx = 2.0 / N_POINTS
        conv_result = np.correlate(h_values, 1 - h_values, mode='full')
        overlap = np.max(conv_result) * dx
        return overlap, conv_result
    except Exception as e:
        raise ValueError(f"evaluation error: {e}")

def h_in_interval(h):
    if np.any((h < 0) | (h > 1)).item():
        return False
    return True

def optimize_shifted_basis(points, n_iterations=100, learning_rate=0.1, step_size=0.2):
    """Generic non-linear local optimization near the current point."""
    scaled_steps = 33  # Number of directions to take
    history_best = []
    history_eval = []
    
    # Convert to normalized space if we want, but keep as [0,1] for now
    h_current = np.array(points)
    best_score = float('inf')
    best_h = None
    
    unique_directions = []
    for _ in range(scaled_steps):
        u = n_sphere(len(h_current))
        unique_directions.append(u)
    
    total_iterations = 0
    bounded = True
    while (h_in_interval(h_current) and bounded) and (total_iterations < n_iterations*5):
        # Randomly select a direction
        idx = int(np.random.rand() * len(unique_directions))
        direction = unique_directions[idx].copy()
        step_mag = learning_rate * step_size * 0.5
        
        h_candidate1 = h_current + step_mag * direction
        h_candidate2 = h_current - step_mag * direction
        
        # Clip to [0,1] and renormalize
        def project_to_feasible(x):
            if h_in_interval(x):
                return x
            s = np.sum(x)
            if np.all(x >= 0) and np.all(x <= 1):
                x = x * (TOTAL_INTEGRAL * FIXED_N_POINTS / s if s != 0 else 1.0)
                return np.clip(x, 0, 1)
            else:
                return np.mean([h_current for _ in range(3)])
        
        h_candidate1 = project_to_feasible(h_candidate1)
        h_candidate2 = project_to_feasible(h_candidate2)
        
        score1, _ = evaluate_erdos(h_candidate1)
        score2, _ = evaluate_erdos(h_candidate2)
        
        total_iterations += 1
        
        if score1 < best_score:
            best_score = score1
            best_h = h_candidate1
            history_best.append(best_score)
            history_eval.append(total_iterations)
        
        if score2 < best_score:
            best_score = score2
            best_h = h_candidate2
            history_best.append(best_score)
            history_eval.append(total_iterations)
        
        if best_score < 1.0:  # Exit condition might be too accurate actually
            bounded = False
            
            
    return best_h.tolist()

def repair_feasibility(h):
    """Repair infeasible h by truncating and adjusting integral."""
    total_sum = np.sum(h) * (1) / I_NORM
    adjustment = (TOTAL_INTEGRAL * I_NORM - total_sum * I_NORM) / I_NORM
    
    h = h * (TOTAL_INTEGRAL * I_NORM / total_sum)
    h = np.clip(h, 0, 1)
    return h.tolist()

def run_annealing(n_points, initial_h=None, max_time=100):
    N_ITERATIONS = 1000
    
    I_NORM = 2.0 / n_points  # Investigation integral normalized component
    
    best_h = None
    best_score = float('inf')
    
    if initial_h is not None:
        h0 = np.array(initial_h)
        sum0 = TOTAL_INTEGRAL * I_NORM
        if abs(sum0 - np.sum(h0)) < 1e-5:
            best_initial = optimize_shifted_basis(h0, n_iterations=200)
        else:
            best_initial = h0
    else:
        best_initial = initialize_random_fully_perturbed(parent=None)
    
    best_h = np.array(best_initial)
    best_score, _ = evaluate_erdos(best_initial)
    
    current = best_h copy
    current_score = best_score
    
    # SA Parameters
    T_initial = 1000.0
    T_curr = T_initial
    cooling_rate = 0.95
    cooling_steps = 500
    
    accepted_moves = 0
    
    start_time = time.time()
    total_evals = 0
    while time.time() - start_time + 0.5 < max_time:
        if total_evals > N_ITERATIONS or T_curr < MIN_TEMP or time.time() - start_time > max_time - 10:
            break
            
        new_T_step = T_curr * cooling_rate
        T_curr = max(MIN_TEMP, new_T_step)
        new_sample = generate_neighbor(current)
        
        try:
            if np.random.rand() < 0.3:  # Occasionally do repair
                new_sample = repair_feasibility(new_sample)
            
            def object(x):
                overlap, _ = evaluate_erdos(x)
                return overlap
            
            current_score = object(current)
            new_score = object(new_sample) * 1.001  # Perturbation bias
            
            delta = new_score - current_score
            
            #6 Alternative acceptance in scaled learning space
            if new_score < current_score:
                # Improvement move
                current = new_sample
                current_score = new_score
                accepted_moves += 1
                best_h = np.array(current)
                best_score = new_score
                
                new_T_step = abs(current_score - new_score) * 0.1
            elif np.exp(-delta / (T_curr)) > 0.5:
                # Something happened
                current = new_sample
                current_score = new_score
                accepted_moves += 1
                best_h = np.array(current)
                best_score = new_score
                
                new_T_step = abs(delta) * 0.01
                
            else:
                # Move rejected
                pass
                
        except Exception as e:
            # For debugging
            pass

    return best_h, best_score

def run(selected_sol=None, time_budget=100, parent_sol=None):
    """Main function respecting the provided inputs."""
    # If parent solution is provided, fix the point from the parent
    if parent_sol is None:
        initial_values = None
    else:
        initial_values = parent_sol
    
    # Validate input dimensions

    best_score = float(best_score)  # It's taken care of here
    
    # Project back if needed, but in normalized space it's unit interval
