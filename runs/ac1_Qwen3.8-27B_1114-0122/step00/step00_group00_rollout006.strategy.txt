import numpy as np
import time

def propose_candidate(parent_sequence: list[float]) -> list[float]:
    """Creates a candidate sequenceoptimized via sliding window in Linf norm

    Args:
      parent_sequence: Initial sequence of coefficients

    Returns:
      List of floats optimized for the given objective

    We constrain candidates to be non-negative,
    but note the evaluator requires no negative numbers.
    """
    parent_seq = np.clip(np.array(parent_sequence, dtype=float), 0.0, 1000.0)
    parent_seq = parent_seq / np.sum(parent_seq) if np.allclose(parent_seq, 0) else parent_seq
    n = parent_seq.size
    
    try:
        best = evaluate_sequence(parent_sequence)  # Initial
        for _ in range(3):  # Temporary budget limit
            candidate = np.roll(parent_seq, int(rng.randint(-7,8)))
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                score = evaluate_sequence(candidate.tolist())
                if (score < best) and (score < 1.5027):
                    best = score
            except:
                continue
    except:
        return parent_sequence  # Validate and fallback
    
    # Compute vector gradient adjustments
    def get_window_gradients(seq, center_idx):
        start = max(0, center_idx-3)
        stop = min(n, center_idx+4)
        relevant_seq = seq[start:stop]
        convolve_in_bounds = relevant_seq[:-1] + relevant_seq + relevant_seq[1:]
        # Use analytic gradients to improve move selection
        return np.gradient(convolve_in_bounds)
    
    original_best_score = np.inf
    safety_retries = 0
    
    while time.time() < time.time() + 900:  # 15分钟预算上限 (宽松版)
        center_idx = rng.randint(0, n)
        grad = get_window_gradients(parent_seq, center_idx)
        exploration = np.clip(parent_seq.copy() + grad * rng.uniform(0.1,0.3), 0.0, 1000.0)
        
        try:
            score = evaluate_sequence(exploration.tolist())
            if score < 1.5027:  # Update condition
                if score < original_best_score:
                    original_best_score = score
        except:
            # Linearity violated, compute analytical fallback
            try:
                if safety_retries >= 5:
                    break
            finally:
                safety_retries += 1
    
    return exploration.tolist()  # Final candidate sequence
