                    best_candidate = candidate

        # Check if we've lost magnitude due to clipping
        # Our clip bounds are [0, 1000] so no need to check further
        
        # If the best_candidate's score is within 1.5, we break early
        # Or if we have spent 90% of the budget, we break
        
        # We'll do a constant of 0.12 for our lambda
        
    return best_candidate.tolist()
