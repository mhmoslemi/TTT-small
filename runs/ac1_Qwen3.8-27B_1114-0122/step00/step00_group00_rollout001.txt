import numpy as np

def _fast_len(n):
    L = 2 * n - 1
    return 1 << (L - 1).bit_length()

def _fft_eval(a, N):
    n = a.size
    c = np.fft.rfft(a, N)
    conv = np.fft.irfft(c * c, N)
    conv = conv[: 2 * n - 1]
    maxb = float(np.max(conv))
    kstar = int(np.argmax(conv))
    S = float(np.sum(a))
    obj = 2.0 * n * maxb / (S * S)
    return obj, maxb, kstar, S

def _active_step(a, N, etas):
    n = a.size
    obj, maxb, kstar, S = _fft_eval(a, N)
    if S <= 1e-6:
        return a, obj
    m = np.arange(n)
    j = kstar - m
    valid = (j >= 0) & (j < n)
    g = np.zeros(n)
    g[valid] = 2.0 * a[np.clip(j[valid], 0, n - 1)]
    dC = g / (S * S) - (2.0 * maxb / (S * S * S))
    dmax = float(np.max(np.abs(dC)))
    if not np.isfinite(dmax) or dmax == 0.0:
        return a, obj
    dC = dC / dmax
    best_a, best_obj = a, obj
    for eta in etas:
        cand = a - eta * dC
        cand = np.clip(cand, 0.0, 1000.0)
        s = float(np.sum(cand))
        if s <= 1e-9:
            continue
        cand = cand * (n / s)
        try:
            obj2 = _fft_eval(cand, N)[0]
        except Exception:
            continue
        if np.isfinite(obj2) and obj2 < best_obj - 1e-12 * max(1.0, best_obj):
            best_obj = obj2
            best_a = cand
    if best_obj >= obj - 1e-12:

import numpy as np
import time

RNG = None  # not used; avoid globals needing init

def _fast_len(n):
    L = 2 * n - 1
    return 1 << (L - 1).bit_length()

def _fft_eval(a, N):
    n = a.size
    c = np.fft.rfft(a, N)
    conv = np.fft.irfft(c * c, N)
    seg = conv[0: 2 * n - 1]
    maxb = float(np.max(seg))
    kstar = int(np.argmax(seg))
    S = float(np.sum(a))
    obj = 2.0 * n * maxb / (S * S)
    return obj, maxb, kstar, S

def _rescale_mean1(a, n):
    s = float(np.sum(a))
    if s <= 1e-9:
        return None
    return a * (n / s)

def _active_step(a, N, etas):
    n = a.size
    obj, maxb, kstar, S = _fft_eval(a, N)
    if S <= 1e-6:
        return a, obj
    m = np.arange(n)
    j = kstar - m
    valid = (j >= 0) & (j < n)
    g = np.zeros(n)
    g[valid] = 2.0 * a[j[valid]]
    dC = g / (S * S) - (2.0 * maxb / (S * S * S))
    dmax = float(np.max(np.abs(dC)))
    if not np.isfinite(dmax) or dmax <= 0:
        return a, obj
    dC = dC / dmax
    best_obj = obj
    best_a = a
    for eta in etas:
        cand = a - eta * dC
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _rescale_mean1(cand, n)
        if cand is None:
            continue
        try:
            o2 = _fft_eval(cand, N)[0]
        except Exception:
            continue
        if np.isfinite(o2) and o2 < best_obj - 1e-12 * max(1.0, best_obj):
            best_obj = o2
            best_a = cand
    if best_obj >= obj - 1e-12:
        a = a * max(1e-12, n / max(S, 1e-12))
        return a, obj
    return best_a, best_obj

def _family_bank(n, N, rng):
    t = np.linspace(0.0, 1.0, n)
    qu = (2.0 * t - 1.0) ** 2 - 1.0 / 3.0
    si = np.sin(2.0 * np.pi * t)

def _active_step(a, N, etas):
    n = a.size
    obj, maxb, kstar, S = _fft_eval(a, N)
    if S <= 1e-6:
        a2 = a.copy()
        a2 = a2 * (n / max(S, 1e-12))
        return a2, obj
    m = np.arange(n)
    j = kstar - m
    valid = (j >= 0) & (j < n)
    g = np.zeros(n)
    g[valid] = 2.0 * a[j[valid]]
    dC = g / (S * S) - (2.0 * maxb / (S * S * S))
    dmax = float(np.max(np.abs(dC)))
    if (not np.isfinite(dmax)) or dmax <= 1e-30:
        a2 = a * (n / S)
        return a2, obj
    dC = dC / dmax
    best_obj = obj
    best_a = a
    for eta in etas:
        cand = a - eta * dC
        cand = np.clip(cand, 0.0, 1000.0)
        s = float(np.sum(cand))
        if s <= 1e-9:
            continue
        cand = cand * (n / s)
        try:
            o2 = _fft_eval(cand, N)[0]
        except Exception:
            continue
        if np.isfinite(o2) and o2 < best_obj - 1e-12 * max(1.0, best_obj):
            best_obj = o2
            best_a = cand
    if best_obj >= obj - 1e-12:
        a2 = a * (n / S)
        return a2, obj
    return best_a, best_obj

"""
Nonnegative coefficient search for  min  2*n*max(conv(a,a))/sum(a)^2 .

Key insight: the objective is scale- and reflection-invariant, and any
PALINDROMIC nonnegative sequence satisfies  max(conv) >= b[n-1] = sum a_i^2
>= (sum a)^2/n,  hence  obj >= 2.  Symmetrizing (as in the previous SOTA)
caps the value at 2.  So we search ASYMMETRIC profiles.

Strategy:
  Phase 0: sample a family of asymmetric tilted/wavy profiles -> good seed (~1.8).
  Phase 1: active-constraint steepest descent on the current argmax coefficient
           of a*a (a minimax algorithm) + line search + periodic random block
           kicks + restarts from diverse profile basins => drives to the
           NONNEGATIVE minimax (~1.5).
  Phase 2: polish the best; optionally up-resample to a larger length to
           squeeze the value; keep the best by the exact evaluator.

Only numpy + time are used; all helpers are top-level; no lambdas/closures.
"""

import time
import numpy as np


def _fast_len(n):
    L = 2 * n - 1
    return 1 << (L - 1).bit_length()


def _fft_eval(a, N):
    n = a.size
    c = np.fft.rfft(a, N)
    conv = np.fft.irfft(c * c, N)
    seg = conv[0:2 * n - 1]
    maxb = float(np.max(seg))
    kstar = int(np.argmax(seg))
    S = float(np.sum(a))
    obj = 2.0 * n * maxb / (S * S)
    return obj, maxb, kstar, S


def _rescale_mean1(a, n):
    s = float(np.sum(a))
    if s <= 1e-9:
        return None
    return a * (n / s)


def _active_step(a, N, etas):
    n = a.size
    obj, maxb, kstar, S = _fft_eval(a, N)
    if S <= 1e-6:
        return a * (n / max(S, 1e-12)), obj
    m = np.arange(n)
    j = kstar - m
    valid = (j >= 0) & (j < n)
    g = np.zeros(n)
    g[valid] = 2.0 * a[j[valid]]
    dC = g / (S * S) - (2.0 * maxb / (S * S * S))
    dmax = float(np.max(np.abs(dC)))
    if (not np.isfinite(dmax)) or dmax <= 1e-30:
        return a * (n / S), obj
    dC = dC / dmax
    best_obj = obj
    best_a = a
    for eta in etas:
        cand = a - eta * dC
        cand = np.clip(cand, 0.0, 1000.0)
        s = float(np.sum(cand))
        if s <= 1e-9:
            continue
        cand = cand * (n / s)
        try:
            o2 = _fft_eval(cand, N)[0]
        except Exception:
            continue
        if np.isfinite(o2) and o2 < best_obj - 1e-12 * max(1.0, best_obj):
            best_obj = o2
            best_a = cand
    if best_obj >= obj - 1e-12:
        return a * (n / S), obj
    return best_a, best_obj


def _kicks(a, rng, N, wlo, whi, sigo, sighi, ntrials):
    n = a.size
    obj0 = _fft_eval(a, N)[0]
    best_a = a.copy()
    best_obj = obj0
    for _ in range(ntrials):
        cand = a.copy()
        w = int(np.ceil(n * rng.uniform(wlo, whi)))
        w = max(1, min(n, w))
        st = int(rng.integers(0, max(1, n - w + 1)))
        st = min(st, n - w)
        sig = rng.uniform(sigo, sighi)
        r = np.exp(rng.normal(0.0, sig, size=w))
        cand[st:st + w] = cand[st:st + w] * r
        if rng.random() < 0.25:
            b = int(rng.integers(0, n))
            bw = max(1, int(np.ceil(n * rng.uniform(wlo, max(2e-3, wlo * 3)))))
            bw = min(bw, n - b)
            lvl = float(np.mean(cand[b:b + bw]))
            if lvl > 0.0:
                cand[b:b + bw] += rng.uniform(0.2, 0.8) * lvl
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _rescale_mean1(cand, n)
        if cand is None:
            continue
        o2 = _fft_eval(cand, N)[0]
        if np.isfinite(o2) and o2 < best_obj - 1e-12 * max(1.0, best_obj):
            best_obj = o2
            best_a = cand
    return best_a, best_obj


def _family_bank(n, N, rng):
    t = np.linspace(0.0, 1.0, n)
    tr = 1.0 - 2.0 * t
    qu = (2.0 * t - 1.0) ** 2 - 1.0 / 3.0
    si = np.sin(2.0 * np.pi * t)
    specs = []
    for e1 in [0.2, 0.5, 0.8, 1.0]:
        for e2 in [-0.5, -0.2, 0.0, 0.2, 0.5]:
            specs.append((e1, e2, 0.0))
    for e3 in [-0.6, -0.3, 0.3, 0.6]:
        for e1 in [0.0, 0.4]:
            specs.append((e1, 0.0, e3))
    bank = []
    for (e1, e2, e3) in specs:
        f = 1.0 + e1 * tr + e2 * qu + e3 * si
        f = np.clip(f, 0.0, None)
        f = _rescale_mean1(f, n)
        if f is None:
            continue
        o = _fft_eval(f, N)[0]
        bank.append((f, o))
    return bank


def _random_profile(n, rng):
    t = np.linspace(0.0, 1.0, n)
    e1 = rng.uniform(-1.0, 1.0)
    e2 = rng.uniform(-0.8, 0.8)
    e3 = rng.uniform(-0.8, 0.8)
    f = 1.0 + e1 * (1.0 - 2.0 * t) + e2 * ((2.0 * t - 1.0) ** 2 - 1.0 / 3.0) + e3 * np.sin(2.0 * np.pi * t)
    noise = np.zeros(n)
    for jj in range(3):
        freq = rng.uniform(2.0, 8.0) * (jj + 1)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0.0, 2 * np.pi)
        noise = noise + amp * np.sin(2.0 * np.pi * freq * t + phase)
    f = f + noise
    f = np.clip(f, 0.0, None)
    return _rescale_mean1(f, n)

            if not bool(np.all(np.isfinite(c))):
                continue
            if float(np.sum(c)) <= 0.01:
                continue
            v = float(evaluate_sequence([float(x) for x in c]))
            if np.isfinite(v) and v < final_val - 1e-12:
                final_val = v
                final_arr = c
        except Exception:
            continue
    if final_arr is None:
        final_arr = np.ones(n1, dtype=np.float64)
        final_val = np.nan

    out = [float(x) for x in final_arr]
    try:
        fv = float(evaluate_sequence(out))
    except Exception:
        fv = np.nan
    print("RESULT obj=%.6f n=%d" % (fv, len(out)))
    try:
        nf = final_arr.size
        Nf = _fast_len(nf)
        of_, mbf, kf, Sff = _fft_eval(final_arr, Nf)
        cc = np.fft.rfft(final_arr, Nf)
        segf = np.fft.irfft(cc * cc, Nf)[:2 * nf - 1]
        print("FINAL maxB=%.4f meanB=%.4f minB=%.4f argmax=%d"
              % (mbf, float(np.mean(segf)), float(np.min(segf)), kf))
    except Exception:
        pass
    return out
