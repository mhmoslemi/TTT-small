```python
"""
Minimizer for J(a) = 2n * max_k (a*a)_k / (sum a)^2 with a >= 0.

Pipeline (sequences kept symmetric to halve the search space):
  setup: exact score of the provided start, analytic-subgradient sanity
         checks (FFT cross-check + finite differences).
  P1: mirror (exponential-gradient) descent on the soft-maximum of the
      convolution, with tau annealing, momentum and restart jitter.
  P2: low-dim PCHIP-spline reparameterization polished with L-BFGS-B and
      tau-annealing (global profile shaping).
  P3: block-wise L-BFGS-B refinements (random + sweeping covers).
  P4: LP-guided "flat neighbor" pivots (improved version of the earlier
      SoTA LP: reduced constraint set from the largest lags).
  P5: length diversity: resample to power-of-two lengths and short pass.
  P6: mixed simulated-annealing finisher (jitter / subgrad / blocks / LP).
Finish: cross-check the best candidates with the authoritative evaluator.
"""
import time
import numpy as np


def _n2(m):
    p = 1
    while p < m:
        p += p
    return p


def _full_b(a, N=None):
    n = a.size
    if N is None:
        N = _n2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)
    return out[: 2 * n - 1]


def _score_parts(a):
    n = a.size
    S = float(a.sum())
    if not np.isfinite(S) or S <= 1e-12:
        return np.inf, 0.0, S
    b = _full_b(a)
    M = float(b.max())
    J = 2.0 * n * M / (S * S)
    return J, M, S


def _h_lag(a, w, N3):
    n = a.size
    q = w[::-1]
    A = np.fft.rfft(a, N3)
    Q = np.fft.rfft(q, N3)
    c = np.fft.irfft(A * Q, N3)
    return c[n - 1: 2 * n - 1][::-1]


def _vtest_direct(a, w):
    n = a.size
    H = np.zeros(n)
    for i in range(n):
        H[: min(n, 2 * n - 1 - i)] += a[i] * w[: min(n, 2 * n - 1 - i)]
    return H


def _sym_full(h):
    n = 2 * h.size - 1
    out = np.empty(n, dtype=np.float64)
    out[: h.size] = h
    out[h.size:] = h[: h.size - 1][::-1]
    return out


def _resample(a, m):
    v = np.interp(np.linspace(0.0, 1.0, m), np.linspace(0.0, 1.0, a.size), a)
    return np.maximum(v, 0.0)


def _h_from_full(a, n):
    return a[: (n + 1) // 2].copy()


def _safe_eval(x):
    try:
        v = float(evaluate_sequence(x))
        return v
    except Exception:

def _phase_subgrad(ctx, a, seconds):
    rng = ctx['rng']
    end = ctx['t0'] + max(1.0, seconds)
    best = a.copy()
    lb = float(ctx['bestJ'])
    n = ctx['n']
    N3 = ctx['N3']
    acc_total = 0
    cno = 0
    ktau = 0
    klr = 0
    while True:
        if ctx['t0'] + time.monotonic() >= end:
            break
        cno += 1
        acc = 0
        arr = best.copy()
        lb_local = lb
        M0 = float(ctx['Mref'])
        tau = tau_frac * M0
        lr = 0.12 * (0.55 ** (klr % 5)) * float(rng.uniform(0.7, 1.4))
        hmo = np.zeros(n)
        S0 = float(arr.sum())
        flo = S0 * 1e-10 / n
        itb = 160 if tau > 1e-9 * max(1.0, M0) else 240
        start_t = time.monotonic()
        for it_ in range(itb):

            Mx = float(b.max())
            if Mx <= 1e-30:
                break
            S = float(arr.sum())
            expo = np.exp(np.clip((b - Mx) / tau, -60.0, 0.0))

            H = _h_lag(arr, w, N3)

def _spl_eval(c, n, n_sym, S_tgt, tau, kgrid, xgrid, last):
    from scipy.interpolate import PchipInterpolator
    kk = PchipInterpolator(kgrid, c, extrapolate=True)
    hh = np.maximum(kk(xgrid), 0.0)
    a = _sym_full(hh)
    S = float(a.sum())
    scale = S / max(S_tgt, 1e-30)
    b = _full_b(a)
    Mx = float(b.max())
    if last:
        J = 2.0 * n * Mx / (S * S)
        return J / (scale * scale), None
    if tau > 1e-9 * max(1e-9, Mx):
        expo = np.exp(np.clip((b - Mx) / tau, -60.0, 0.0))
        LSE = Mx + tau * float(expo.sum()) / b.size
    else:
        LSE = Mx
    J = 2.0 * n * LSE / (S * S)
    return J / (scale * scale), None


def _spl_opt(x, cS):
    c = np.maximum(x, 0.0)
    J, _ = _spl_eval(c, cS['n'], cS['m'], cS['S_tgt'], cS['tau'],
                     cS['kgrid'], cS['xgrid'], cS['last'])
    return float(J)


def _spl_init(rng, kind, ref_half, K):
    x = np.linspace(0.0, 1.0, K)
    if kind == 0:
        c = np.interp(x, np.linspace(0.0, 1.0, ref_half.size), ref_half)
    elif kind == 1:
        k = int(rng.integers(1, 15))
        ph = float(rng.uniform(0.0, np.pi))
        amp = float(rng.uniform(0.15, 0.9))
        c = (1.0 + amp * np.cos(np.pi * k * x + ph)) ** 2
    elif kind == 2:
        f = float(rng.uniform(0.5, 0.95))
        w = float(rng.uniform(0.03, 0.25))
        e = float(rng.uniform(0.5, 2.5))
        t = np.abs(x - 0.5) * 2.0
        c = np.where(t < f, 1.0, np.maximum(0.0, 1.0 - (t - f) / w))
        c = np.maximum(c, 1e-9) ** e
    else:
        al = float(rng.uniform(0.4, 3.0))
        c = 1.0 - np.abs(2.0 * x - 1.0) ** al
    c = np.maximum(c, 1e-6)
    c = c * np.exp(rng.normal(0.0, 0.04, K))
    c = c / (float(c.mean()) + 1e-30)
    return c

            lb = Jc
            total += 1
            print('  [spline] accept kind=%d K=%d J=%.6f t=%.0fs' %
                  (kind, K, lb, ctx['t0'] + time.monotonic()))
            Mx = float(_full_b(best).max())
            ctx['Mref'] = min(ctx['Mref'], Mx)
        else:
            tau_frac = tau_frac * lam
    ctx['bestJ'] = min(ctx['bestJ'], lb)
    return best


def _blocks_obj(x, cB):
    a = _sym_full(cB['h'])
    L = cB['L']
    st = cB['st']
    nv = np.maximum(np.exp(x), cB['lo'])
    a[st: st + L] = nv
    a[n - 1 - (st + L - 1): n - st][::-1] = nv[::-1]
    S = float(a.sum())
    scale = S / cB['S_tgt']
    b = _full_b(a)
    Mx = float(b.max())
    tau = cB['tau']
    if tau > 1e-12 * max(1e-12, Mx):
        expo = np.exp(np.clip((b - Mx) / tau, -60.0, 0.0))
        LSE = Mx + tau * float(expo.sum()) / b.size
    else:
        LSE = Mx
    J = 2.0 * cB['n'] * LSE / (S * S)
    rs = cB['rs']
    nz = np.maximum(rs.standard_normal(2 * L), -2.5)
    nz[L:] = -nz[:L]
    a[st: st + L] += nv * (1e-6 * nz[:L])
    a[n - 1 - (st + L - 1): n - st][::-1] += nv[::-1] * (1e-6 * nz[L:])
    if cB['guard']:
        a = np.maximum(a, cB['lo'])
        S2 = float(a.sum())
        duty = S2 / cB['S_tgt']
        J = J / (scale * scale) + 1e-12 * duty * duty
    else:
        J = J / (scale * scale)
    return float(J)


def _phase_blocks(ctx, a, seconds):
    from scipy.optimize import minimize
    rng = ctx['rng']
    rs = np.random.RandomState(12345)
    end = ctx['t0'] + max(1.0, seconds)
    best = a.copy()
    lb = float(ctx['bestJ'])
    n = ctx['n']
    m = n // 2
    h = _h_from_full(a, n)
    S_tgt = float(a.sum())
    Mref = max(float(ctx['Mref']), 1e-9)
    sizes = [1024, 512, 384, 256, 192, 128, 96, 64, 48, 32, 24, 16]
    mode = 0
    st = 0
    acc = 0
    while True:
        if ctx['t0'] + time.monotonic() >= end:
            break
        L = sizes[(3 * mode) % len(sizes)]
        if st is None or st + L > m:
            st = int(rng.integers(0, max(1, m - L + 1)))
            if (mode % 3) == 2:
                st = int((np.arange(-0.4, 1.2 * m, L * 0.75))[(mode * 5 + acc) % int(np.ceil(1.6 * m / (L * 0.75)))])
                st = int(np.clip(st, 0, max(0, m - L)))
            mode += 1
        tau = Mref * [0.004, 0.0015, 0.0006, 0.0002, 0.00006, 0.0, 0.00003][(mode * 11) % 7]
        cur = h[st: st + L].copy()
        lo = max(1e-9, float(cur.min()) * 0.03)
        x0 = np.log(np.maximum(cur, lo))
        cB = {'h': h, 'n': n, 'st': st, 'L': L, 'S_tgt': S_tgt, 'tau': tau,
              'lo': lo, 'rs': rs, 'guard': True}
        try:
            res = minimize(_blocks_obj, x0, args=(cB,), method='L-BFGS-B',
                           options={'maxiter': 60, 'ftol': 1e-14, 'gtol': 1e-9})
        except Exception as ex:
            st = int(rng.integers(0, max(1, m - L + 1)))
            continue
        x1 = getattr(res, 'x', None)
        if x1 is None:
            st = int(rng.integers(0, max(1, m - L + 1)))
            continue
        cB2 = dict(cB)
        cB2['guard'] = False
        J1 = _blocks_obj(x1, cB2)
        J0 = _blocks_obj(x0, cB2)
        if np.isfinite(J1) and J1 < J0 - 1e-11:
            h[st: st + L] = np.maximum(np.exp(x1), lo)
            cand = _sym_full(h)
            Sx = float(cand.sum())
            cand *= S_tgt / max(Sx, 1e-30)
            Jc = 2.0 * n * float(_full_b(cand).max()) / (S_tgt * S_tgt)
            if Jc < lb - 1e-12:
                best = cand
                lb = Jc
                acc += 1
                Mref = min(Mref, float(_full_b(cand).max()))
        st = int((st + max(8, L // 3)) % max(1, m - L + 1))
    ctx['bestJ'] = min(ctx['bestJ'], lb)
    if acc:
        print('  [blocks] done acc=%d best=%.6f' % (acc, ctx['bestJ']))
    return best


def _lp_solve(a, b, M, T):
    from scipy.optimize import linprog
    from scipy.sparse import csc_matrix
    n = a.size
    lags = np.argsort(b)[::-1][: T]
    if (n - 1) not in lags:
        lags = np.append(lags, n - 1)
    dlist = []
    ilist = []
    ptr = [0]
    for k in lags:
        klo = max(0, k - (n - 1))
        khi = min(n - 1, k)
        idx = np.arange(klo, khi + 1)
        ilist.append(idx)
        dlist.append(a[k - idx])
        ptr.append(ptr[-1] + len(idx))
    A = csc_matrix((np.concatenate(dlist), np.concatenate(ilist),
                    np.array(ptr, dtype=np.int64)), shape=(len(lags), n))
    cvec = -np.ones(n)
    bnd = [(0.0, None)] * n
    res = linprog(cvec, A_ub=A, b_ub=np.full(len(lags), M), bounds=bnd,
                  method='hiGHS', options={'presolve': 'high', 'time_limit': 240})
    if res is not None and res.status == 0 and getattr(res, 'x', None) is not None:
        sx = res.x
        sx[~np.isfinite(sx)] = 0.0
        return np.maximum(sx, 0.0)
    return None


def _phase_lp(ctx, a, seconds):
    rng = ctx['rng']
    end = ctx['t0'] + max(1.0, seconds)
    best = a.copy()
    lb = float(ctx['bestJ'])
    n = ctx['n']
    S0 = float(a.sum())
    tlist = [256, 384, 512, 640, 800] if n <= 12288 else [160, 240, 320]
    tp = 0
    acc = 0
    while True:
        if ctx['t0'] + time.monotonic() >= end:
            break
        T = tlist[min(tp, len(tlist) - 1)]
        tp += 1
        b = _full_b(best)
        M = float(b.max())
        if M <= 1e-9:
            break
        tb = time.monotonic()
        bB = _lp_solve(best, b, M, T)
        dt = time.monotonic() - tb
        print('  [lp] T=%d dt=%.2fs' % (T, dt))
        if bB is None:
            continue
        sB = float(bB.sum())
        if sB <= 1e-9:
            continue
        move = False
        for tt in (1.0, 0.8, 0.6, 0.4, 0.25, 0.12, 0.05):
            cand = (1.0 - tt) * best + tt * bB * (S0 / sB)
            cand = np.maximum(cand, 0.0)
            cand = 0.5 * (cand + cand[::-1])
            Jc = 2.0 * n * float(_full_b(cand).max()) / (S0 * S0)
            if np.isfinite(Jc) and Jc < lb - 1e-10:
                best = cand
                lb = Jc
                acc += 1
                move = True
                print('  [lp] accept t=%.2f J=%.6f' % (tt, lb))
                break
        if dt > 30.0 and (n > 12000 or T > 300):
            if T in tlist:
                tlist = [t for t in tlist if t < T]
                if not tlist:
                    tlist = [64]
            break
    ctx['bestJ'] = min(ctx['bestJ'], lb)
    if acc:
        print('  [lp] done acc=%d best=%.6f' % (acc, ctx['bestJ']))
    return best


def _phase_alt(ctx, a, m2, seconds):
    end = ctx['t0'] + max(1.0, seconds)
    a2 = _resample(a, m2)
    N3_2 = _n2(3 * m2 - 2)
    a2 = _sym_full(_h_from_full(a2, m2))
    S0 = float(a2.sum())
    J, M, S = _score_parts(a2)
    lbC = J
    best = a2.copy()
    ctx2 = ctx
    Sfv = S0
    if Sfv > 1e-6:
        a2 = a2 * (S0 / S)
    tau = 0.03 * M
    for cno in range(6):
        if ctx['t0'] + time.monotonic() >= end or M <= 1e-9:
            break
        lr = [0.35, 0.2, 0.12, 0.07, 0.05, 0.04][cno]
        hmo = np.zeros(m2)
        S0 = float(a2.sum())
        flo = S0 * 1e-10 / m2
        for _it in range(140):
            if ctx['t0'] + time.monotonic() >= end:
                Jc = 2.0 * m2 * float(_full_b(a2).max()) / (S0 * S0)
                if Jc < lbC - 1e-10:
                    lbC = Jc
                    best = a2.copy()
                break
            b = _full_b(a2)
            Mx = float(b.max())
            if Mx <= 1e-30:
                break
            S = float(a2.sum())
            expo = np.exp(np.clip((b - Mx) / tau, -60.0, 0.0))
            LSE = Mx + tau * float(expo.sum()) / b.size
            w = expo / expo.sum()
            H = _h_lag(a2, w, N3_)
            g = S * H - LSE * a2
            gm = float(np.abs(g).max())
            if gm <= 1e-30:
                break
            hmo = 0.55 * hmo + 0.45 * (g / gm)
            hmo = np.clip(hmo, -1.5, 1.5)
            a2 = np.maximum(a2 * np.exp(-lr * hmo), flo)
            S = float(a2.sum())
            a2 *= S0 / max(S, 1e-30)
        tau *= 0.45
    if lbC < ctx['bestJ']:
        ctx['bestJ'] = lbC
        ctx['alt_best'] = best.copy()
        ctx['alt_len'] = m2
        ctx['alt_score'] = lbC
        print('  [alt n=%d] accept J=%.6f' % (m2, lbC))
    return best


def _phase_finish(ctx, a, seconds):
    rng = ctx['rng']
    end = ctx['t0'] + max(1.0, seconds)
    best = a.copy()
    n = ctx['n']
    S0 = float(a.sum())
    Mref = max(float(ctx['Mref']), 1e-9)
    last_track = 0.0
    while True:
        now = ctx['t0'] + time.monotonic()
        if now >= end:
            break
        r = float(rng.random())
        rem = end - now
        if r < 0.30 and rem > 4.0:
            J, M, S = _score_parts(best)
            tau = M * float(rng.choice([0.01, 0.02, 0.005, 0.002, 0.001, 0.0004]))
            lr = float(rng.choice([0.05, 0.1, 0.2, 0.35]))
            hmo = np.zeros(n)
            flo = S * 1e-10 / n
            for _ in range(60):
                b = _full_b(best)
                Mx = float(b.max())
                if Mx <= 1e-30:
                    break
                S = float(best.sum())
                expo = np.exp(np.clip((b - Mx) / tau, -60.0, 0.0))
                LSE = Mx + tau * float(expo.sum()) / b.size
                w = expo / expo.sum()
                H = _h_lag(best, w, ctx['N3'])
                g = S * H - LSE * best
                gm = float(np.abs(g).max())
                if gm <= 1e-30:
                    break
                hmo = 0.55 * hmo + 0.45 * (g / gm)
                hmo = np.clip(hmo, -1.5, 1.5)
                best = np.maximum(best * np.exp(-lr * hmo), flo)
                best *= S / max(float(best.sum()), 1e-30)
            Jn = 2.0 * n * float(_full_b(best).max()) / (S * S)
            if np.isfinite(Jn) and Jn < ctx['bestJ'] - 1e-10:
                ctx['bestJ'] = Jn
        elif r < 0.62 and rem > 3.0:
            L = int(rng.choice([64, 96, 128, 192, 256]))
            m = n // 2
            st = int(rng.integers(0, max(1, m - L + 1)))
            a = _phase_blocks(ctx, best, 2.5)
            best = a
        elif r < 0.82 and rem > 3.0:
            sigma = 0.02
            cand = best * np.exp(rng.normal(0.0, sigma, n))
            cand = np.maximum(cand, 1e-12)
            cand *= S0 / max(float(cand.sum()), 1e-30)
            Jc = 2.0 * n * float(_full_b(cand).max()) / (S0 * S0)
            if np.isfinite(Jc):
                ad = Jc - ctx['bestJ']
                Tsa = 0.002 * max(ctx['bestJ'], 1e-9)
                eta = min(0.5, 0.004 * (rem / max(1.0, end - ctx['t0'])))
                if ad < -1e-10:
                    best = cand
                    ctx['bestJ'] = Jc
                elif np.log(float(rng.random()) + 1e-12) > ad / Tsa:
                    best = cand
        elif rem > 15.0:
            B = _phase_lp(ctx, best, min(15.0, rem - 2.0))
            best = B
        if now - last_track > 45.0:
            last_track = now
            print('  [finish] best=%.6f t=%.0f/%.0f' %
                  (ctx['bestJ'], now - ctx['t0'], seconds))
    return best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Full search over non-negative step-height vectors. See module docstring."""
    t_start = time.monotonic()
    try:
        budget = float(kwargs.get('budget_s', budget_s) if 'budget_s' in kwargs
                       else budget_s)
    except Exception:
        budget = 2400.0
    budget = min(3540.0, max(25.0, budget))
    deadline = t_start + budget - 25.0
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    base = None
    try:
        b0 = height_sequence_1
        if isinstance(b0, (list, np.ndarray)) and len(b0) > 1:
            base = np.asarray(b0, dtype=float).copy()
    except Exception:
        base = None
    if base is None or not np.all(np.isfinite(base)):
        base = None
    if base is not None:
        base = np.clip(base, 0.0, 1000.0)
    fallback = None
    if base is None:
        base = 5.0 + 0.7 * np.cos(np.pi * np.linspace(0.0, 1.0, 6121))
        print('no valid height_sequence_1; using cosine fallback')
    n0 = int(base.size)
    base = 0.5 * (base + base[::-1])
    base = np.maximum(base, 1e-6)
    S_ev = float(base.sum())
    if 0.01 < S_ev < 9e6:
        fallback = [float(x) for x in base]
        print('base n=%d sum=%.4g' % (n0, S_ev))

    N2 = _n2(2 * n0 - 1)
    N3 = _n2(3 * n0 - 2)
    J0, M0, S0 = _score_parts(base)
    print('baseline score (fft): %.6f timings N2=%d N3=%d M0=%.6g S0=%.6g' %
          (J0, N2, N3, M0, S0))
    if J0 != J0:
        J0 = 2.0
        M0 = J0 * S0 * S0 / (2.0 * n0)

    grad_ok = False
    try:
        tg = np.random.default_rng(11)
        for _tr in range(2):
            ng = 25
            ag = tg.random(ng) + 0.2
            wref = sample_tau = None
            b_g = _full_b(ag)
            M_g = float(b_g.max())
            tau_g = 0.05 * M_g
            expo = np.exp(np.clip((b_g - M_g) / tau_g, -60.0, 0.0))
            w_g = expo / expo.sum()
            H1 = _vtest_direct(ag, w_g)
            H2 = _h_lag(ag, w_g, _n2(3 * ng - 2))
            rel = float(np.abs(H1 - H2).max()) / (float(np.abs(H1).max()) + 1e-30)
            if rel > 1e-6:
                break
            S_g = float(ag.sum())
            LSE_g = M_g + tau_g * float(expo.sum()) / b_g.size
            Jg = 2.0 * ng * LSE_g / (S_g * S_g)
            gj = 4.0 * ng * (S_g * H1 - LSE_g * ag) / (S_g ** 3)
            ok = True
            for _f in range(6):
                jf = int(tg.integers(0, ng))
                eps = 1e-6 * (1.0 + abs(ag[jf]))
                ap = ag.copy()
                ap[jf] += eps
                LSE_p = _full_b(ap)
                Mp = float(LSE_p.max())
                Spre = float(ap.sum())
                LSEp = Mp + tau_g * float(np.sum(np.exp(np.clip((LSE_p - Mp) / tau_g, -60.0, 0.0)))) / LSE_p.size
                Jp = 2.0 * ng * LSEp / (Spre * Spre)
                fd = (Jp - Jg) / eps
                drel = abs(fd - gj[jf]) / (abs(fd) + abs(gj[jf]) + 1e-12)
                if drel > 5e-3:
                    ok = False
                    break
            if not ok:
                break
        else:
            grad_ok = True
        print('gradient checks: %s' % ('PASS' if grad_ok else 'FAIL'))
    except Exception as ex:
        print('gradient check error:', ex)
        grad_ok = False

    ctx = {'n': n0, 'N3': N3, 't0': t_start, 'bestJ': J0, 'Mref': M0,
           'rng': rng, 'alt_best': None, 'alt_len': 0, 'alt_score': np.inf}
    a = base.copy()
    print('phase budget: total=%.0f deadline@ t=%.1f' % (budget, deadline - t_start))

    freq_c = 0.5 if n0 <= 8192 else 0.25
    frac = {'p1': 0.28 * freq_c + 0.02, 'p2': 0.27, 'p3': 0.17,
            'p4': 0.14, 'p5': 0.06, 'p6': 0.11 * freq_c + 0.02}
    tot = sum(frac.values())
    secs = {k: v * budget / tot for k, v in frac.items()}
    t_now = t_start

    if grad_ok and secs['p1'] > 5:
        a = _phase_subgrad(ctx, a, secs['p1'])
    t_now = max(t_now, t_start)

    if secs['p2'] > 5:
        a = _phase_spline(ctx, a, secs['p2'])

    if secs['p3'] > 5:
        a = _phase_blocks(ctx, a, secs['p3'])

    if secs['p4'] > 8:
        a = _phase_lp(ctx, a, secs['p4'])

    if ctx.get('alt_best') is not None:
        al = int(ctx['alt_len'])
        ab = ctx['alt_best']
        if al != n0:
            merged = _resample(ab, n0)
            Jm, Mm, _ = _score_parts(merged)
            if np.isfinite(Jm) and Jm < ctx['bestJ']:
                a = merged
                ctx['bestJ'] = Jm
                ctx['Mref'] = min(ctx['Mref'], Mm)
                print('merged alt-* best: %.6f' % Jm)
        else:
            a = ab
            Jm, Mm, _ = _score_parts(a)
            if np.isfinite(Jm) and Jm < ctx['bestJ']:
                ctx['bestJ'] = Jm
                ctx['Mref'] = min(ctx['Mref'], Mm)

    if secs['p5'] > 12:
        cands = []
        for p in (4096, 8192, 16384, 32768):
            if 1.3 * n0 <= p <= 6.2 * n0:
                cands.append(p)
        cands = cands[:2]
        pp = 0.72
        for p in cands:
            rem_p = (t_start + budget - 25.0) - time.monotonic()
            give = min(secs['p5'] * pp, max(0.0, rem_p - 8.0))
            if give > 10.0:
                ctx2 = dict(ctx)
                ctx2['t0'] = time.monotonic()
                ctx2['bestJ'] = ctx['bestJ']
                ctx2['Mref'] = ctx['Mref']
                ctx2['alt_best'] = None
                ctx2['alt_len'] = 0
                ctx2['alt_score'] = np.inf
                try:
                    print('  [alt] probing n=%d for %.0fs' % (p, give))
                    _phase_alt(ctx2, a, p, give)
                    if ctx2.get('alt_best') is not None:
                        print('  [alt] n=%d improved: %.6f' % (p, ctx2['alt_score']))
                        if ctx2['alt_score'] < ctx['bestJ']:
                            ctx['bestJ'] = ctx2['alt_score']
                            ctx['alt_best'] = ctx2['alt_best']
                            ctx['alt_len'] = p
                except Exception as ex:
                    print('  [alt] n=%d failed: %s' % (p, ex))

    if ctx.get('alt_best') is not None:
        al = int(ctx['alt_len'])
        if al != n0:
            merged = _resample(ctx['alt_best'], n0)
            Jm, Mm, _ = _score_parts(merged)
            if np.isfinite(Jm) and Jm < ctx['bestJ']:
                a = merged
                ctx['bestJ'] = Jm
                ctx['Mref'] = min(ctx['Mref'], Mm)
                print('merged alt-best: %.6f' % Jm)
        else:
            a = ctx['alt_best']
            Jm, Mm, _ = _score_parts(a)
            if np.isfinite(Jm) and Jm < ctx['bestJ']:
                ctx['bestJ'] = Jm
                ctx['Mref'] = min(ctx['Mref'], Mm)

    remF = deadline - time.monotonic()
    if remF > 4.0:
        a = _phase_finish(ctx, a, remF)

    sa = a * (1.0 / (float(a.sum()) / (np.sqrt(2.0 * n0) if n0 else 1.0)))
    sa = 0.5 * (sa + sa[::-1])
    sa = np.clip(sa, 0.0, 1000.0)
    Jfinal, Mfinal, _ = _score_parts(sa)
    print('final internal score: %.6f (max_b=%.6g sum=%.6g)' %
          (Jfinal, Mfinal, float(sa.sum())))
    if not np.isfinite(Jfinal):
        sa = base
        Jfinal, Mfinal, _ = _score_parts(sa)

    try:
        evval = _safe_eval([float(x) for x in sa])
        print('evaluate_sequence(final) = %s' % evval)
        if not np.isfinite(evval) and fallback is not None:
            evf = _safe_eval(fallback)
            print('evaluate_sequence(fallback) = %s' % evf)
            if np.isfinite(evf):
                sa = np.asarray(fallback, dtype=float)
                Jfinal, Mfinal, _ = _score_parts(sa)
    except Exception as ex:
        print('eval check skipped:', ex)

    return [float(x) for x in sa]
```
