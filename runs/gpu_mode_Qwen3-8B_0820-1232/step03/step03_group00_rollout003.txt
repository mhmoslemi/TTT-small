import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any
import math

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Optimized Triton implementation of the outgoing version of the TriMul operator.

    Args:
        input_tensor: Input tensor of shape [bs, seq_len, seq_len, dim]
        mask: Mask tensor of shape [bs, seq_len, seq_len]
        weights: Dictionary containing model weights
        config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define kernel parameters
    BLOCK = 128
    M = seq_len
    N = seq_len
    K = seq_len
    BS = batch_size

    # Define kernel
    @triton.jit
    def tritongemm_kernel(
        left_ptr, right_ptr, out_ptr, mask_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Compute program ID
        pid = tl.program_id(0)
        # Compute batch ID
        batch_id = pid // (seq_len * seq_len)
        # Compute row ID in the batch
        row_id = (pid // seq_len) % seq_len
        # Compute column ID in the batch
        col_id = pid % seq_len

        # Initialize output
        out = tl.zeros((BLOCK,), dtype=tl.float32)

        # Iterate over k
        for k in range(0, seq_len, BLOCK):
            # Load left and right data
            left = tl.load(left_ptr + batch_id * seq_len * seq_len * hidden_dim + row_id * seq_len * hidden_dim + k * hidden_dim, mask=(k < seq_len))
            right = tl.load(right_ptr + batch_id * seq_len * seq_len * hidden_dim + col_id * seq_len * hidden_dim + k * hidden_dim, mask=(k < seq_len))

            # Multiply and accumulate
            out += tl.dot(left, right)

        # Apply mask
        mask_val = tl.load(mask_ptr + batch_id * seq_len * seq_len + row_id * seq_len + col_id)
        out = out * mask_val

        # Store result
        tl.store(out_ptr + batch_id * seq_len * seq_len * dim + row_id * seq_len * dim + col_id * dim, out)

    # Define strides
    left_strides = (seq_len * seq_len * hidden_dim, seq_len * hidden_dim, hidden_dim, 1)
    right_strides = (seq_len * seq_len * hidden_dim, seq_len * hidden_dim, hidden_dim, 1)
    out_strides = (seq_len * seq_len * dim, seq_len * dim, dim, 1)
    mask_strides = (seq_len * seq_len, seq_len, 1)

    # Cast tensors to device
    input_tensor = input_tensor.to(torch.float32)
    
    # Compute projections
    # Linear layer: A = W * x
    # left_proj = weights["left_proj.weight"]
    left_proj = weights["left_proj.weight"]
    right_proj = weights["right_proj.weight"]
    left_gate = weights["left_gate.weight"]
    right_gate = weights["right_gate.weight"]
    out_gate = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply LayerNorm to input
    # For simplicity, LayerNorm is done in PyTorch and not in Triton

    # Compute projections
    # left = (input_tensor @ left_proj)
    # right = (input_tensor @ right_proj)
    # left_gate = (input_tensor @ left_gate).sigmoid()
    # right_gate = (input_tensor @ right_gate).sigmoid()
    # out_gate = (input_tensor @ out_gate).sigmoid()
    # out = einsum('... i k d, ... j k d -> ... i j d', left * left_gate, right * right_gate)
    # out = out * out_gate

    # Compute projections in Triton
    # Convert the projections to triton matrices
    left = torch.matmul(input_tensor, left_proj.T)
    right = torch.matmul(input_tensor, right_proj.T)
    left_gate = torch.matmul(input_tensor, left_gate.T).sigmoid()
    right_gate = torch.matmul(input_tensor, right_gate.T).sigmoid()
    out_gate = torch.matmul(input_tensor, out_gate.T).sigmoid()

    # Apply mask
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask

    # Compute the TriMul output
    left = left.to(torch.float32).contiguous()
    right = right.to(torch.float32).contiguous()
    out = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), device=input_tensor.device)

    # Launch the kernel
    grid = (triton.cdiv(seq_len * seq_len * batch_size, BLOCK),)
    tritongemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        mask.data_ptr(),
        batch_size,
        seq_len,
        hidden_dim,
        hidden_dim,
        BLOCK=BLOCK
    )

    # Apply LayerNorm and final projection
    # LayerNorm can be done in PyTorch for simplicity
    # out = out.to(torch.float32)
    out = out.to(torch.float32)
    out = out * out_gate
    out = torch.matmul(out, to_out_weight.T)

    # Apply LayerNorm
    # out = self.to_out_norm(out)
    out = torch.nn.functional.layer_norm(out, out.shape[-1], weight=to_out_norm_weight, bias=to_out_norm_bias)

    return out
