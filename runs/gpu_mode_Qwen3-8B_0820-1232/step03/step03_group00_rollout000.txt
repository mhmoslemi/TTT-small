import torch
import triton
import triton.language as tl
from torch import Tensor
from typing import Tuple, Dict

def custom_kernel(data: Tuple[Tensor, Tensor, Dict[str, Tensor], Dict]) -> Tensor:
    """
    Implement the "outgoing" version of the TriMul operation using a fully optimized Triton kernel.
    This kernel handles the forward pass of the TriMul module for the given input and weights.

    The TriMul operation involves:
    1. Applying LayerNorm to the input
    2. Projection to hidden dimensions using linear projections
    3. Masking and gating operations
    4. A pairwise multiplication across the N dimension
    5. Normalization, gating, and final projection

    The kernel uses triton for optimized performance by fusing several operations
    and avoiding the nested loops used in the reference PyTorch implementation.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Cast masks and inputs to correct types
    mask = mask.unsqueeze(-1).to(input_tensor.device)

    # Define the kernel for the TriMul operation
    @triton.jit
    def tritongemm_kernel(
        input_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr,
        out_gate_ptr, to_out_norm_ptr, to_out_ptr,
        mask_ptr, output_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr, STRIDE_IN: tl.constexpr, STRIDE_LEFT: tl.constexpr,
        STRIDE_RIGHT: tl.constexpr, STRIDE_LEFT_GATE: tl.constexpr,
        STRIDE_RIGHT_GATE: tl.constexpr, STRIDE_OUT_GATE: tl.constexpr,
        STRIDE_TO_OUT_NORM: tl.constexpr, STRIDE_TO_OUT: tl.constexpr,
        STRIDE_MASK: tl.constexpr, STRIDE_OUTPUT: tl.constexpr
    ):
        # Get the batch index
        pid = tl.program_id(0)
        b = pid

        # Determine the block's output indices
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        # Compute the indices for the output
        offs_m = pid_m * BLOCK + tl.arange(0, BLOCK)
        offs_n = pid_n * BLOCK + tl.arange(0, BLOCK)

        # Compute the indices for the input
        idx_i = offs_m
        idx_j = offs_n

        # Compute the indices for the mask
        mask_idx = tl.arange(0, BLOCK)

        # Load input tensor (x)
        x = tl.load(
            input_ptr + b * STRIDE_IN + idx_i[:, None] * seq_len * seq_len * dim + idx_j[None, :] * seq_len * dim,
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        # Apply LayerNorm weights and bias
        x = x * norm_weight + norm_bias

        # Compute left projection
        left = tl.load(
            left_proj_ptr + b * STRIDE_LEFT + idx_i[:, None] * seq_len * hidden_dim + idx_j[None, :] * hidden_dim,
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        # Apply mask to left
        left = left * tl.load(
            mask_ptr + (b * STRIDE_MASK) + idx_i[:, None] * seq_len * seq_len + mask_idx[None, :],
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        # Compute left gate
        left_gate = tl.load(
            left_gate_ptr + b * STRIDE_LEFT_GATE + idx_i[:, None] * seq_len * hidden_dim + idx_j[None, :] * hidden_dim,
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        left_gate = left_gate.sigmoid()
        left = left * left_gate

        # Compute right projection
        right = tl.load(
            right_proj_ptr + b * STRIDE_RIGHT + idx_i[:, None] * seq_len * hidden_dim + idx_j[None, :] * hidden_dim,
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        # Apply mask to right
        right = right * tl.load(
            mask_ptr + (b * STRIDE_MASK) + idx_i[:, None] * seq_len * seq_len + mask_idx[None, :],
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        # Compute right gate
        right_gate = tl.load(
            right_gate_ptr + b * STRIDE_RIGHT_GATE + idx_i[:, None] * seq_len * hidden_dim + idx_j[None, :] * hidden_dim,
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        right_gate = right_gate.sigmoid()
        right = right * right_gate

        # Compute output (tri-multiplicative update)
        out = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)
        for k in range(seq_len):
            # Load left[k] and right[k]
            left_k = tl.load(
                left_proj_ptr + b * STRIDE_LEFT + k * seq_len * hidden_dim + idx_i[:, None] * hidden_dim,
                mask=(idx_i < seq_len),
                other=0.0
            )
            right_k = tl.load(
                right_proj_ptr + b * STRIDE_RIGHT + k * seq_len * hidden_dim + idx_j[None, :] * hidden_dim,
                mask=(idx_j < seq_len),
                other=0.0
            )
            # Compute element-wise product
            out += tl.dot(left_k, right_k)

        # Apply output gate
        out_gate = tl.load(
            out_gate_ptr + b * STRIDE_OUT_GATE + idx_i[:, None] * seq_len * hidden_dim + idx_j[None, :] * hidden_dim,
            mask=(idx_i < seq_len) & (idx_j < seq_len),
            other=0.0
        )
        out_gate = out_gate.sigmoid()
        out *= out_gate

        # Apply to_out_norm
        out = out * to_out_norm_weight + to_out_norm_bias
        # Final projection
        out = tl.dot(out, to_out_weight)

        # Store output
        tl.store(
            output_ptr + b * STRIDE_OUTPUT + idx_i[:, None] * seq_len * dim + idx_j[None, :] * dim,
            out,
            mask=(idx_i < seq_len) & (idx_j < seq_len)
        )

    # Launch the kernel
    BLOCK = 128
    STRIDE_IN = input_tensor.stride(0) + input_tensor.stride(1) + input_tensor.stride(2)
    STRIDE_LEFT = left_proj_weight.stride(0) + left_proj_weight.stride(1)
    STRIDE_RIGHT = right_proj_weight.stride(0) + right_proj_weight.stride(1)
    STRIDE_LEFT_GATE = left_gate_weight.stride(0) + left_gate_weight.stride(1)
    STRIDE_RIGHT_GATE = right_gate_weight.stride(0) + right_gate_weight.stride(1)
    STRIDE_OUT_GATE = out_gate_weight.stride(0) + out_gate_weight.stride(1)
    STRIDE_TO_OUT_NORM = to_out_norm_weight.stride(0) + to_out_norm_weight.stride(1)
    STRIDE_TO_OUT = to_out_weight.stride(0) + to_out_weight.stride(1)
    STRIDE_MASK = mask.stride(0) + mask.stride(1)
    STRIDE_OUTPUT = input_tensor.stride(0) + input_tensor.stride(1)

    num_blocks = triton.cdiv(seq_len, BLOCK)
    grid = (batch_size, num_blocks, num_blocks)
    tritongemm_kernel[grid](
        input_tensor.data_ptr(), left_proj_weight.data_ptr(), right_proj_weight.data_ptr(),
        left_gate_weight.data_ptr(), right_gate_weight.data_ptr(), out_gate_weight.data_ptr(),
        to_out_norm_weight.data_ptr(), to_out_weight.data_ptr(),
        mask.data_ptr(), output_tensor.data_ptr(),
        batch_size, seq_len, dim, hidden_dim,
        BLOCK, STRIDE_IN, STRIDE_LEFT, STRIDE_RIGHT, STRIDE_LEFT_GATE,
        STRIDE_RIGHT_GATE, STRIDE_OUT_GATE, STRIDE_TO_OUT_NORM, STRIDE_TO_OUT,
        STRIDE_MASK, STRIDE_OUTPUT
    )

    # Return the computed output
    return output_tensor
