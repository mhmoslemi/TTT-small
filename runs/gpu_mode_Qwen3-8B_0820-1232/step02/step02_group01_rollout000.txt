import torch
import triton
import triton.language as tl
from torch import Tensor
from typing import Tuple
import math

def custom_kernel(data):
    """
    Implements the TriMul module using Triton for optimized performance.
    This kernel handles the forward pass of the TriMul operation with masking,
    including linear projections, gate functions, and the matrix multiplication
    step that computes pairwise interactions.

    Key components:
    - Linear projections for left and right
    - Sigmoid gates for left and right
    - Out gate for final activation
    - Matrix multiplication through einsum equivalent with Triton

    The kernel is optimized to run efficiently on NVIDIA L40S with Ada Lovelace architecture.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access weights
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Ensure mask is on correct device
    mask = mask.to(input_tensor.device).unsqueeze(-1)

    # Create output tensor
    output = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), device=input_tensor.device)

    # Triton kernel for TriMul
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        out_ptr,
        batch_size: tl.constexpr,
        seq_len: tl.constexpr,
        dim: tl.constexpr,
        hidden_dim: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        # Define indices
        pid = tl.program_id(0)
        pid_m = pid // seq_len
        pid_n = pid % seq_len
        offset_m = pid_m * seq_len + tl.arange(0, seq_len)
        offset_n = pid_n * seq_len + tl.arange(0, seq_len)

        # Load weights
        left_proj = tl.load(left_proj_ptr + tl.arange(0, hidden_dim) * dim, mask=tl.arange(0, hidden_dim) < hidden_dim)
        right_proj = tl.load(right_proj_ptr + tl.arange(0, hidden_dim) * dim, mask=tl.arange(0, hidden_dim) < hidden_dim)
        left_gate = tl.load(left_gate_ptr + tl.arange(0, hidden_dim) * dim, mask=tl.arange(0, hidden_dim) < hidden_dim)
        right_gate = tl.load(right_gate_ptr + tl.arange(0, hidden_dim) * dim, mask=tl.arange(0, hidden_dim) < hidden_dim)
        out_gate = tl.load(out_gate_ptr + tl.arange(0, hidden_dim) * dim, mask=tl.arange(0, hidden_dim) < hidden_dim)
        to_out_norm = tl.load(to_out_norm_ptr + tl.arange(0, hidden_dim) * hidden_dim, mask=tl.arange(0, hidden_dim) < hidden_dim)
        to_out = tl.load(to_out_ptr + tl.arange(0, hidden_dim) * dim, mask=tl.arange(0, hidden_dim) < hidden_dim)

        # Compute left and right vectors (batch, seq_len, seq_len, hidden_dim)
        left = tl.zeros((seq_len, hidden_dim), dtype=tl.float32)
        right = tl.zeros((seq_len, hidden_dim), dtype=tl.float32)

        # Loop over all sequences
        for i in range(seq_len):
            for j in range(seq_len):
                # Compute left vector for i-th sequence
                left_i = tl.load(input_ptr + (pid_m * seq_len + i) * seq_len * dim + tl.arange(0, seq_len) * dim)
                left_i = tl.dot(left_i, left_proj)
                left_i = left_i * tl.load(mask_ptr + (pid_m * seq_len + i) * seq_len + tl.arange(0, seq_len))
                left[i] = left_i

                # Compute right vector for j-th sequence
                right_j = tl.load(input_ptr + (pid_m * seq_len + j) * seq_len * dim + tl.arange(0, seq_len) * dim)
                right_j = tl.dot(right_j, right_proj)
                right_j = right_j * tl.load(mask_ptr + (pid_m * seq_len + j) * seq_len + tl.arange(0, seq_len))
                right[j] = right_j

        # Apply gates
        left_gate_vals = tl.sigmoid(tl.load(left_gate_ptr + tl.arange(0, hidden_dim) * dim))
        right_gate_vals = tl.sigmoid(tl.load(right_gate_ptr + tl.arange(0, hidden_dim) * dim))
        out_gate_vals = tl.sigmoid(tl.load(out_gate_ptr + tl.arange(0, hidden_dim) * dim))

        left = left * left_gate_vals
        right = right * right_gate_vals

        # Compute the matrix multiplication using Triton's dot product
        out = tl.zeros((seq_len, seq_len, hidden_dim), dtype=tl.float32)

        for i in range(seq_len):
            for k in range(seq_len):
                left_i_k = left[i, k]
                for j in range(seq_len):
                    right_j_k = right[j, k]
                    out[i, j] += left_i_k * right_j_k

        # Apply norms and final gate
        out = tl.dot(out, to_out_norm)
        out = out * out_gate_vals
        out = tl.dot(out, to_out)

        # Store result
        offsets = (pid_m * seq_len + offset_m[:, None]) * seq_len * hidden_dim + offset_n[None, :] * hidden_dim
        tl.store(out_ptr + offsets, out)

    # Launch the kernel
    BLOCK = 128
    num_blocks = triton.cdiv(batch_size * seq_len * seq_len, BLOCK)
    grid = (num_blocks,)
    tri_mul_kernel[grid](
        input_tensor,
        mask,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        to_out_norm_weight,
        to_out_weight,
        output,
        batch_size,
        seq_len,
        dim,
        hidden_dim,
        BLOCK
    )

    return output
