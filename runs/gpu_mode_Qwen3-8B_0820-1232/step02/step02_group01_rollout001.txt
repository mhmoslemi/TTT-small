import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict
from task import input_t, output_t

def custom_kernel(data: input_t) -> output_t:
    """
    Optimized Triton kernel implementing the "outgoing" version of the TriMul operator.
    The kernel performs the forward pass of the TriMul layer using Triton for efficient computation.
    It includes operations for linear projections, gate activations, and the core multiplicative update.

    Args:
        data: Tuple of (input_tensor, mask, weights, config)
    
    Returns:
        output: Processed tensor of shape [bs, seq_len, seq_len, dim]
    """

    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Device and data types
    device = input_tensor.device
    dtype = torch.float16
    mask = mask.to(device)
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]

    # Allocate output
    output = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Define Triton kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr: tl.pointer_type,
        mask_ptr: tl.pointer_type,
        left_proj_ptr: tl.pointer_type,
        right_proj_ptr: tl.pointer_type,
        left_gate_ptr: tl.pointer_type,
        right_gate_ptr: tl.pointer_type,
        out_gate_ptr: tl.pointer_type,
        to_out_norm_ptr: tl.pointer_type,
        to_out_ptr: tl.pointer_type,
        output_ptr: tl.pointer_type,
        B: tl.constexpr,
        N: tl.constexpr,
        dim: tl.constexpr,
        hidden_dim: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        # Each thread handles a single output element
        # We use a grid of (N * N) threads to compute all output elements
        # We will use 1 thread per output element to avoid memory bandwidth issues

        # Get the current thread index
        pid = tl.program_id(0)
        # Compute the output indices
        i = pid // N
        j = pid % N

        # Load input values
        x = tl.load(input_ptr + i * N * N * dim + j * N * dim, mask=tl.where(mask_ptr + i * N * N + j * N < 1, 1, 0), other=0.0)
        x = x.to(tl.float32)

        # Apply LayerNorm before projections
        # Compute mean and variance
        mean = tl.sum(x * norm_weight, axis=0) / N
        var = tl.sum((x - mean) * (x - mean) * norm_weight, axis=0) / N
        x = (x - mean) / tl.sqrt(var + 1e-5)
        x = x * norm_weight + norm_bias

        # Compute left and right projections
        left = tl.dot(x, left_proj_weight.to(tl.float32), axis=-1)
        right = tl.dot(x, right_proj_weight.to(tl.float32), axis=-1)
        left = left.to(tl.float16)
        right = right.to(tl.float16)

        # Apply mask to left and right projections
        mask_val = tl.load(mask_ptr + i * N * N + j * N, other=1.0)
        left = left * mask_val
        right = right * mask_val

        # Compute left and right gates
        left_gate = tl.dot(x, left_gate_weight.to(tl.float32), axis=-1).sigmoid()
        right_gate = tl.dot(x, right_gate_weight.to(tl.float32), axis=-1).sigmoid()
        out_gate = tl.dot(x, out_gate_weight.to(tl.float32), axis=-1).sigmoid()

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Compute the TriMul operation
        # out[i][j] = sum_k (left[i][k] * right[j][k])
        acc = tl.zeros((hidden_dim,), dtype=tl.float16)
        for k in range(N):
            l = tl.load(left_ptr + i * N * hidden_dim + k * hidden_dim, other=0.0)
            r = tl.load(right_ptr + j * N * hidden_dim + k * hidden_dim, other=0.0)
            acc += l * r
        # Apply out_gate
        acc = acc * out_gate
        # Normalize using to_out_norm
        acc = tl.dot(acc, to_out_norm_weight.to(tl.float32), axis=-1)
        acc = acc + to_out_norm_bias
        # Apply to_out
        acc = tl.dot(acc, to_out_weight.to(tl.float32), axis=-1)
        # Store result
        tl.store(output_ptr + i * N * hidden_dim + j * hidden_dim, acc)

    # Compute memory sizes
    input_size = B * N * N * dim
    mask_size = B * N * N
    left_right_size = B * N * N * hidden_dim
    gate_size = B * N * hidden_dim
    to_out_norm_size = hidden_dim
    to_out_size = B * N * N * hidden_dim
    output_size = B * N * N * hidden_dim

    # Launch the kernel
    # Use 1 thread per output element
    # So grid size is B * N * N
    grid = (B * N * N,)
    tri_mul_kernel[grid](
        input_tensor.data_ptr(),
        mask.data_ptr(),
        left_proj_weight.data_ptr(),
        right_proj_weight.data_ptr(),
        left_gate_weight.data_ptr(),
        right_gate_weight.data_ptr(),
        out_gate_weight.data_ptr(),
        to_out_norm_weight.data_ptr(),
        to_out_weight.data_ptr(),
        output.data_ptr(),
        B,
        N,
        dim,
        hidden_dim,
        1
    )

    # Convert output to float32
    output = output.to(torch.float32)

    return output
