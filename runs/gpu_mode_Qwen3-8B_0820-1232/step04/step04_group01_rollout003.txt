import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Optimized Triton implementation of the TriMul module. This kernel performs the "outgoing" version
    of the TriMul operation used in BioML models like AlphaFold3, Chai, and Protenix. The core
    operation is a matrix multiplication across all sequence pairs (i,j), with intermediate weights
    and mask-based attention.

    This kernel is optimized for performance using Triton's explicit tiling and memory management,
    especially for large sequence lengths (N), and uses mixed precision where possible to maintain
    accuracy while improving runtime performance.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define the block size for tiling (optimally tuned for L40S)
    BLOCK = 128
    # Compute number of blocks needed along each dimension
    num_block_i = triton.cdiv(seq_len, BLOCK)
    num_block_j = triton.cdiv(seq_len, BLOCK)
    num_block_k = triton.cdiv(seq_len, BLOCK)

    # Convert input tensor from float32 to float16 for efficiency
    input_tensor = input_tensor.to(torch.float16)

    # Initialize output tensor
    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device, dtype=torch.float32)

    # Apply LayerNorm to the input
    input_tensor = input_tensor * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections and gate activations
    left = input_tensor @ weights["left_proj.weight"].transpose(0, 1)
    right = input_tensor @ weights["right_proj.weight"].transpose(0, 1)

    left_gate = input_tensor @ weights["left_gate.weight"].transpose(0, 1)
    right_gate = input_tensor @ weights["right_gate.weight"].transpose(0, 1)
    out_gate = input_tensor @ weights["out_gate.weight"].transpose(0, 1)

    # Apply mask
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
    left = left * mask
    right = right * mask

    # Apply sigmoid to gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # Apply gates to left and right projections
    left = left * left_gate
    right = right * right_gate

    # This kernel computes the TriMul operation using a tiling approach.
    # For each batch and pair of sequence indices (i,j), it accumulates over k in blocks
    # using efficient memory access and parallelization.

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, output_ptr,
        seq_len, batch_size, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Compute the index of the current thread (i, j)
        pid = tl.program_id(0)
        i = pid // num_block_j
        j = pid % num_block_j

        # Compute the offset for the block of i
        offs_i = i * BLOCK + tl.arange(0, BLOCK)
        # Compute the offset for the block of j
        offs_j = j * BLOCK + tl.arange(0, BLOCK)

        # Mask for when index exceeds sequence length
        m_i = offs_i < seq_len
        m_j = offs_j < seq_len

        # Initialize accumulator
        acc = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)

        # Iterate over blocks of k
        for k_block in range(num_block_k):
            # Compute offset of k
            offs_k = k_block * BLOCK + tl.arange(0, BLOCK)

            # Load values of left and right
            # Left has shape [bs, seq_len, seq_len, hidden_dim]
            # So for each batch, we load a block of i, k and multiply by a block of k, j
            # Similarly for right

            left = tl.load(left_ptr + (i * seq_len + offs_k) * seq_len * hidden_dim + offs_i[:, None] * hidden_dim, mask=m_i[:, None], other=0.0)
            right = tl.load(right_ptr + (j * seq_len + offs_k) * seq_len * hidden_dim + offs_j[None, :] * hidden_dim, mask=m_j[None, :], other=0.0)

            # Compute the dot product
            acc += tl.dot(left, right)

        # Store the result
        tl.store(output_ptr + (i * seq_len + offs_i) * seq_len * hidden_dim + (j * seq_len + offs_j) * hidden_dim, acc, mask=m_i[:, None] & m_j[None, :])

    # Launch the kernel
    grid = (num_block_i * num_block_j,)
    tri_mul_kernel[grid](
        left.flatten(0, 2).contiguous(),
        right.flatten(0, 2).contiguous(),
        output.flatten(0, 2).contiguous(),
        seq_len,
        batch_size,
        hidden_dim,
        BLOCK=BLOCK
    )

    # Apply output gate and LayerNorm
    output = output * out_gate
    output = output @ weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    output = output @ weights["to_out.weight"]

    # Convert final result to float32
    output = output.to(torch.float32)

    return output
