import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Triton implementation of the TriMul operation for the TriMul module.
    This kernel implements the "outgoing" version of the TriMul operation,
    which is used in protein structure prediction models like AlphaFold3.
    The kernel is optimized for performance on NVIDIA L40S GPUs with Ada Lovelace architecture (sm_89).
    It performs the following steps:
    1. Applies LayerNorm to the input.
    2. Projects input through linear layers (left and right projections).
    3. Applies sigmoid gates to the projections.
    4. Multiplies the masked projections with their corresponding gates.
    5. Computes the TriMul output using einsum-like operations.
    6. Applies LayerNorm and then multiplies with the output gate.
    7. Applies the final linear projection to get the output tensor.

    The kernel is optimized for performance and uses Triton's low-level features for memory access, tiling,
    and thread scheduling to maximize GPU efficiency.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)

    Returns:
        output: Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define constants for the kernel
    BS = batch_size
    N = seq_len
    C = dim
    H = hidden_dim

    # Load weights
    norm_weight = weights["norm.weight"].view(C)
    norm_bias = weights["norm.bias"].view(C)
    left_proj_weight = weights["left_proj.weight"].reshape(H, C)
    right_proj_weight = weights["right_proj.weight"].reshape(H, C)
    left_gate_weight = weights["left_gate.weight"].reshape(H, C)
    right_gate_weight = weights["right_gate.weight"].reshape(H, C)
    out_gate_weight = weights["out_gate.weight"].reshape(H, C)
    to_out_norm_weight = weights["to_out_norm.weight"].reshape(H, H)
    to_out_norm_bias = weights["to_out_norm.bias"].reshape(H)
    to_out_weight = weights["to_out.weight"].reshape(C, H)

    # Convert mask to float
    mask = mask.unsqueeze(-1).float()

    # Allocate output
    output = torch.zeros((BS, N, N, H), device=input_tensor.device)

    # Triton kernel implementation
    @triton.jit
    def triton_trimul_kernel(
        input_ptr, mask_ptr, left_proj_weight, right_proj_weight, left_gate_weight,
        right_gate_weight, out_gate_weight, to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output_ptr, BS, N, H, BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        off_b = pid * BLOCK
        mask = (off_b < BS)
        if not mask:
            return

        # Compute indices
        off_b = pid
        off_n = tl.arange(0, BLOCK)
        off_m = tl.arange(0, BLOCK)

        # Load weights
        left_proj = tl.load(left_proj_weight + off_m[:, None] * H + off_n[None, :], mask=(off_m < H) & (off_n < C), other=0.0)
        right_proj = tl.load(right_proj_weight + off_m[:, None] * H + off_n[None, :], mask=(off_m < H) & (off_n < C), other=0.0)
        left_gate = tl.load(left_gate_weight + off_m[:, None] * H + off_n[None, :], mask=(off_m < H) & (off_n < C), other=0.0)
        right_gate = tl.load(right_gate_weight + off_m[:, None] * H + off_n[None, :], mask=(off_m < H) & (off_n < C), other=0.0)
        out_gate = tl.load(out_gate_weight + off_m[:, None] * H + off_n[None, :], mask=(off_m < H) & (off_n < C), other=0.0)
        to_out_norm_weight = tl.load(to_out_norm_weight + off_m[:, None] * H + off_n[None, :], mask=(off_m < H) & (off_n < H), other=0.0)
        to_out_norm_bias = tl.load(to_out_norm_bias + off_n[None, :], mask=(off_n < H), other=0.0)
        to_out_weight = tl.load(to_out_weight + off_m[:, None] * C + off_n[None, :], mask=(off_m < C) & (off_n < H), other=0.0)

        # Load input data
        input = tl.load(input_ptr + off_b * N * N * C + off_n[None, :, None] * N * C + off_m[:, None, None] * C + off_m[None, :, None] * N, mask=off_m < C, other=0.0)
        mask = tl.load(mask_ptr + off_b * N * N + off_n[None, :, None] * N + off_m[:, None, None], mask=off_m < N, other=0.0)

        # Apply LayerNorm
        input = input - norm_bias[off_n[None, :]]
        input = input * norm_weight[off_n[None, :]] / (tl.sqrt(tl.exp(tl.sum((input * input) * (1.0 / (N * N)), axis=1) * (1.0 / N * N * C), axis=1))) + norm_bias[off_n[None, :]]

        # Apply linear projections and gates
        left_proj = left_proj * input * mask
        right_proj = right_proj * input * mask

        left_gate = left_gate * input
        right_gate = right_gate * input

        left = left_proj * left_gate
        right = right_proj * right_gate

        # Compute TriMul
        out = tl.dot(left, right, axis=1)
        out = out * out_gate

        # Apply LayerNorm
        out = out + to_out_norm_bias[off_n[None, :]]
        out = out * to_out_norm_weight + to_out_norm_bias

        # Apply final projection
        out = tl.dot(out, to_out_weight, axis=2)

        # Store output
        tl.store(output_ptr + off_b * N * N * H + off_n[None, :, None] * N * H + off_m[:, None, None] * H + off_m[None, :, None], out, mask=off_m < H)

    # Launch kernel
    BLOCK = 128
    grid = (triton.cdiv(BS * N * N, BLOCK), )
    triton_trimul_kernel[grid](
        input_tensor.data_ptr(), mask.data_ptr(),
        left_proj_weight.data_ptr(), right_proj_weight.data_ptr(),
        left_gate_weight.data_ptr(), right_gate_weight.data_ptr(),
        out_gate_weight.data_ptr(), to_out_norm_weight.data_ptr(),
        to_out_norm_bias.data_ptr(), to_out_weight.data_ptr(),
        output.data_ptr(), BS, N, H, BLOCK=BLOCK
    )

    # Convert to float32
    output = output.to(torch.float32)

    return output
