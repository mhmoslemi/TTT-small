import torch
import triton
import triton.language as tl
import math

# Custom kernel for TriMul
@triton.jit
def trimul_kernel(
    x_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, 
    out_gate_ptr, to_out_norm_ptr, to_out_ptr, 
    out_ptr,
    batch_size, seq_len, dim, hidden_dim, 
    BLOCK: tl.constexpr, 
    num_warps: tl.constexpr, 
    group_size: tl.constexpr,
):
    # Program ID
    pid = tl.program_id(0)
    # Compute block indices
    batch_idx = pid // (seq_len * seq_len)
    seq_i = (pid // seq_len) % seq_len
    seq_j = pid % seq_len

    # Compute offsets
    offset = tl.arange(0, BLOCK)
    mask_offsets = tl.arange(0, seq_len)
    offset_batch = batch_idx * seq_len * seq_len
    offset_i = seq_i * seq_len
    offset_j = seq_j * seq_len
    offset_out = batch_idx * seq_len * seq_len * dim

    # Load x (input) and apply norm
    x = tl.load(x_ptr + offset_batch + offset_i + offset, dtype=tl.float32)
    mask = tl.load(mask_ptr + offset_batch + offset_i + offset, dtype=tl.float32)
    x = x * mask

    # Compute left and right projections
    left_proj = tl.load(left_proj_ptr + offset, dtype=tl.float32)
    left_proj = x @ left_proj
    right_proj = tl.load(right_proj_ptr + offset, dtype=tl.float32)
    right_proj = x @ right_proj

    # Compute gates
    left_gate = tl.load(left_gate_ptr + offset, dtype=tl.float32)
    left_gate = x @ left_gate
    left_gate = tl.sigmoid(left_gate)

    right_gate = tl.load(right_gate_ptr + offset, dtype=tl.float32)
    right_gate = x @ right_gate
    right_gate = tl.sigmoid(right_gate)

    out_gate = tl.load(out_gate_ptr + offset, dtype=tl.float32)
    out_gate = x @ out_gate
    out_gate = tl.sigmoid(out_gate)

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Perform the TriMul operation
    acc = tl.zeros((BLOCK, dim), dtype=tl.float32)
    for k in range(0, seq_len, group_size):
        k_offsets = tl.arange(0, group_size)
        k_offsets = k_offsets + k
        k_offsets = k_offsets < seq_len

        left_block = tl.load(left_proj_ptr + offset_batch + offset_i + k_offsets, dtype=tl.float32)
        right_block = tl.load(right_proj_ptr + offset_batch + offset_j + k_offsets, dtype=tl.float32)

        left_block = left_block * tl.load(mask_ptr + offset_batch + offset_i + k_offsets, dtype=tl.float32)
        right_block = right_block * tl.load(mask_ptr + offset_batch + offset_j + k_offsets, dtype=tl.float32)

        left_block = left_block * tl.load(left_gate_ptr + offset_batch + offset_i + k_offsets, dtype=tl.float32)
        right_block = right_block * tl.load(right_gate_ptr + offset_batch + offset_j + k_offsets, dtype=tl.float32)

        acc += tl.dot(left_block, right_block)

    # Apply normalization
    to_out_norm = tl.load(to_out_norm_ptr + offset, dtype=tl.float32)
    acc = acc @ to_out_norm

    # Apply output gate
    acc = acc * out_gate

    # Store the result
    tl.store(out_ptr + offset_out + offset_i + offset, acc, mask=offset < seq_len)

def custom_kernel(data):
    """
    Optimized Triton kernel for TriMul operation in BioML models.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [batch_size, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [batch_size, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters
    
    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, seq_len, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Initialize output tensor
    output = torch.zeros(bs, seq_len, seq_len, dim, device=input_tensor.device)

    # Constants
    BLOCK = 64
    num_warps = 4
    group_size = 16

    # Define kernel arguments
    x_ptr = input_tensor
    mask_ptr = mask
    left_proj_ptr = weights["left_proj.weight"]
    right_proj_ptr = weights["right_proj.weight"]
    left_gate_ptr = weights["left_gate.weight"]
    right_gate_ptr = weights["right_gate.weight"]
    out_gate_ptr = weights["out_gate.weight"]
    to_out_norm_ptr = weights["to_out_norm.weight"]
    to_out_ptr = weights["to_out.weight"]
    out_ptr = output

    # Launch the kernel
    grid = (bs * seq_len * seq_len,)  # One thread per (batch, i, j)
    trimul_kernel[grid](
        x_ptr, mask_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, 
        out_gate_ptr, to_out_norm_ptr, to_out_ptr, 
        out_ptr,
        bs, seq_len, dim, hidden_dim, 
        BLOCK, num_warps, group_size
    )

    return output
