import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Implements the TriMul operation with a Triton kernel to accelerate pairwise operations.
    The kernel optimizes the 'einsum' operation by fusing it with memory loading and computation,
    significantly reducing the runtime for large sequence lengths.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply norm
    x = input_tensor.to(torch.float32)
    x_norm = torch.addmm(norm_bias, torch.nn.functional.linear(x, norm_weight), torch.zeros_like(norm_bias))

    # Compute projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight).to(torch.float32)
    right = torch.nn.functional.linear(x_norm, right_proj_weight).to(torch.float32)

    # Apply mask
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask

    # Compute gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Create output buffer
    output = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), device=x.device, dtype=torch.float32)

    # Triton kernel to compute the TriMul operation
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, output_ptr,
        batch_size, seq_len, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Calculate grid dimensions
        pid = tl.program_id(0)
        batch = pid // (seq_len * seq_len)
        i = (pid // seq_len) % seq_len
        j = pid % seq_len

        # Offset for current batch, row i, column j
        batch_offset = batch * seq_len * seq_len * hidden_dim
        i_offset = i * seq_len * hidden_dim
        j_offset = j * hidden_dim
        offset = batch_offset + i_offset + j_offset

        # Load left and right vectors (of length seq_len)
        left = tl.load(left_ptr + offset, mask=tl.arange(0, seq_len) < seq_len, other=0.0)
        right = tl.load(right_ptr + offset, mask=tl.arange(0, seq_len) < seq_len, other=0.0)

        # Compute the dot product and accumulate to output
        acc = tl.zeros((hidden_dim,), dtype=tl.float32)
        for k in range(seq_len):
            l = tl.load(left_ptr + k * hidden_dim + offset, mask=k < seq_len, other=0.0)
            r = tl.load(right_ptr + k * hidden_dim + offset, mask=k < seq_len, other=0.0)
            acc += tl.dot(l, r)

        # Write the result to output
        tl.store(output_ptr + offset, acc)

    # Launch parameters
    BLOCK = 128
    num_blocks = triton.cdiv(batch_size * seq_len * seq_len, BLOCK)
    grid = (num_blocks,)
    triton_trimul_kernel[grid](
        left,
        right,
        output,
        batch_size, seq_len, hidden_dim,
        BLOCK=BLOCK
    )

    # Apply normalization and out gate
    output = torch.nn.functional.linear(output, to_out_norm_weight, to_out_norm_bias)
    output = output * out_gate

    # Final projection
    output = torch.nn.functional.linear(output, to_out_weight)

    return output.to(torch.float32)
