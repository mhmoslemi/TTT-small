import torch
import triton
import triton.language as tl
from typing import Dict, Tuple
import math
import functools

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
    """
    Triton implementation of the Triangle Multiplicative Update (TriMul) operation.
    This kernel performs the forward pass of the TriMul module from AlphaFold3 and related models.
    It computes the outgoing version of the TriMul operation by efficiently performing pairwise
    operations using triton's high-performance kernels.

    Parameters:
        data: Tuple containing the input, mask, model weights, and configuration.
        - input_tensor: [bs, seq_len, seq_len, dim]
        - mask: [bs, seq_len, seq_len]
        - weights: Dictionary containing model weights
        - config: Dictionary with model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Define block sizes
    SEQLen = seq_len
    B = batch_size
    C = dim  # input dimension
    H = hidden_dim  # hidden dimension

    # We'll implement the following steps:
    # 1. LayerNorm (input)
    # 2. left_proj, right_proj, left_gate, right_gate, out_gate
    # 3. Apply masks
    # 4. Compute the TriMul output (using einsum)
    # 5. Apply to_out_norm and out_gate
    # 6. to_out

    # We will implement each part step by step using Triton for performance

    # --- Step 1: LayerNorm (input) ---

    def run_layer_norm(input_tensor, norm_weight, norm_bias):
        # input_tensor: [bs, seq_len, seq_len, dim]
        # norm_weight: [dim]
        # norm_bias: [dim]
        # output: [bs, seq_len, seq_len, dim]
        @triton.jit
        def kernel_ln(x_ptr, y_ptr, norm_weight_ptr, norm_bias_ptr,
                      num_batch, seq_len, dim, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            n = tl.arange(0, BLOCK)
            m = pid * BLOCK + n
            mask = m < seq_len * seq_len * dim

            x = tl.load(x_ptr + m, mask=mask, other=0.0)
            w = tl.load(norm_weight_ptr, mask=mask, other=0.0)
            b = tl.load(norm_bias_ptr, mask=mask, other=0.0)
            y = x * w + b
            tl.store(y_ptr + m, y, mask=mask)

        grid = (triton.cdiv(seq_len * seq_len * dim, BLOCK),)
        kernel_ln[grid](
            input_tensor, input_tensor, norm_weight, norm_bias,
            B, SEQLen, dim, BLOCK=128
        )

        return input_tensor

    input_norm = run_layer_norm(input_tensor, norm_weight=weights["norm.weight"], norm_bias=weights["norm.bias"])

    # --- Step 2: left_proj, right_proj, left_gate, right_gate, out_gate ---

    # Project input to hidden_dim
    def run_proj(input_tensor, weight, out_tensor):
        # input_tensor: [bs, seq_len, seq_len, dim]
        # weight: [dim, hidden_dim]
        # out_tensor: [bs, seq_len, seq_len, hidden_dim]
        @triton.jit
        def kernel_proj(x_ptr, y_ptr, weight_ptr,
                        num_batch, seq_len, dim, hidden_dim, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            n = tl.arange(0, BLOCK)
            m = pid * BLOCK + n
            mask = m < seq_len * seq_len * hidden_dim

            x = tl.load(x_ptr + m, mask=mask, other=0.0)
            w = tl.load(weight_ptr, mask=mask, other=0.0)
            y = x * w
            tl.store(y_ptr + m, y, mask=mask)

        grid = (triton.cdiv(seq_len * seq_len * hidden_dim, BLOCK),)
        kernel_proj[grid](
            input_tensor, out_tensor, weight,
            B, SEQLen, dim, hidden_dim, BLOCK=128
        )

        return out_tensor

    left = torch.zeros((B, SEQLen, SEQLen, H), device=input_tensor.device, dtype=torch.float32)
    right = torch.zeros((B, SEQLen, SEQLen, H), device=input_tensor.device, dtype=torch.float32)
    left = run_proj(input_norm, weights["left_proj.weight"], left)
    right = run_proj(input_norm, weights["right_proj.weight"], right)

    # Apply mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(torch.bool)
        left = torch.where(mask, left, torch.tensor(0.0, device=left.device))
        right = torch.where(mask, right, torch.tensor(0.0, device=right.device))

    # Compute gates
    def run_gate(input_tensor, weight, out_tensor):
        # input_tensor: [bs, seq_len, seq_len, dim]
        # weight: [dim, hidden_dim]
        # out_tensor: [bs, seq_len, seq_len, hidden_dim]
        @triton.jit
        def kernel_gate(x_ptr, y_ptr, weight_ptr,
                        num_batch, seq_len, dim, hidden_dim, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            n = tl.arange(0, BLOCK)
            m = pid * BLOCK + n
            mask = m < seq_len * seq_len * hidden_dim

            x = tl.load(x_ptr + m, mask=mask, other=0.0)
            w = tl.load(weight_ptr, mask=mask, other=0.0)
            y = x * w
            tl.store(y_ptr + m, y, mask=mask)

        grid = (triton.cdiv(seq_len * seq_len * hidden_dim, BLOCK),)
        kernel_gate[grid](
            input_tensor, out_tensor, weight,
            B, SEQLen, dim, hidden_dim, BLOCK=128
        )

        return out_tensor

    left_gate = torch.zeros((B, SEQLen, SEQLen, H), device=input_tensor.device, dtype=torch.float32)
    right_gate = torch.zeros((B, SEQLen, SEQLen, H), device=input_tensor.device, dtype=torch.float32)
    out_gate = torch.zeros((B, SEQLen, SEQLen, H), device=input_tensor.device, dtype=torch.float32)

    left_gate = run_gate(input_norm, weights["left_gate.weight"], left_gate)
    right_gate = run_gate(input_norm, weights["right_gate.weight"], right_gate)
    out_gate = run_gate(input_norm, weights["out_gate.weight"], out_gate)

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Multiply by gates
    left = left * left_gate
    right = right * right_gate

    # --- Step 3: Compute the TriMul output ---

    # We'll perform this step using a kernel that computes the einsum operation efficiently.
    # The einsum is: out = einsum('...i k d, ...j k d -> ...i j d', left, right)

    # Initialize output tensor
    output = torch.zeros((B, SEQLen, SEQLen, H), device=input_tensor.device, dtype=torch.float32)
    output_grad = torch.zeros((B, SEQLen, SEQLen, H), device=input_tensor.device, dtype=torch.float32)

    # Define the kernel for the einsum-like operation
    @triton.jit
    def kernel_einsum(
        left_ptr, right_ptr, output_ptr,
        num_batch, seq_len, hidden_dim, BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)  # loop over batch

        # Get the batch index
        batch_idx = pid
        # Loop over i (row indices)
        i = tl.arange(0, BLOCK)
        # Loop over j (column indices)
        j = tl.arange(0, BLOCK)

        # Indices for batch, row and column
        off_batch = batch_idx * seq_len * seq_len
        off_i = i * seq_len
        off_j = j * seq_len

        # Offsets for left and right tensors
        off_left = off_batch + off_i + tl.arange(0, hidden_dim)
        off_right = off_batch + off_j + tl.arange(0, hidden_dim)

        # Load left and right data
        left = tl.load(left_ptr + off_left, mask=off_left < seq_len * seq_len * hidden_dim, other=0.0)
        right = tl.load(right_ptr + off_right, mask=off_right < seq_len * seq_len * hidden_dim, other=0.0)

        # Perform the einsum-like operation
        temp = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)
        for k in range(0, seq_len, BLOCK):  # loop over k
            # Compute the inner product
            # left[k, :, :] and right[k, :, :] are [seq_len, hidden_dim]
            # We'll compute for each i and j
            for m in range(0, seq_len, BLOCK):
                off_left_i = off_i + m
                off_left_j = off_j + m

                # Read the data
                a = tl.load(left_ptr + off_batch + off_left_i + tl.arange(0, hidden_dim), other=0.0)
                b = tl.load(right_ptr + off_batch + off_left_j + tl.arange(0, hidden_dim), other=0.0)

                # Compute the inner product
                temp = temp + tl.dot(a, b)

        # Save the result
        tl.store(output_ptr + off_batch + off_i + off_j, temp)

    # Launch the kernel
    grid = (B,)
    kernel_einsum[grid](
        left, right, output,
        B, SEQLen, H, BLOCK=128
    )

    # --- Step 4: Apply to_out_norm and out_gate --- 

    # Normalize the output
    def run_norm(input_tensor, norm_weight, norm_bias):
        @triton.jit
        def kernel_norm(x_ptr, y_ptr, norm_weight_ptr, norm_bias_ptr,
                        num_batch, seq_len, hidden_dim, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            n = tl.arange(0, BLOCK)
            m = pid * BLOCK + n
            mask = m < seq_len * seq_len * hidden_dim

            x = tl.load(x_ptr + m, mask=mask, other=0.0)
            w = tl.load(norm_weight_ptr, mask=mask, other=0.0)
            b = tl.load(norm_bias_ptr, mask=mask, other=0.0)
            y = x * w + b
            tl.store(y_ptr + m, y, mask=mask)

        grid = (triton.cdiv(seq_len * seq_len * hidden_dim, BLOCK),)
        kernel_norm[grid](
            input_tensor, output, norm_weight, norm_bias,
            B, SEQLen, H, BLOCK=128
        )

        return output

    output = run_norm(output, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # Multiply by the out_gate
    output = output * out_gate

    # --- Step 5: Apply to_out ---

    def run_proj(out_tensor, weight, output_tensor):
        @triton.jit
        def kernel_proj(x_ptr, y_ptr, weight_ptr,
                        num_batch, seq_len, hidden_dim, dim, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            n = tl.arange(0, BLOCK)
            m = pid * BLOCK + n
            mask = m < seq_len * seq_len * dim

            x = tl.load(x_ptr + m, mask=mask, other=0.0)
            w = tl.load(weight_ptr, mask=mask, other=0.0)
            y = x * w
            tl.store(y_ptr + m, y, mask=mask)

        grid = (triton.cdiv(seq_len * seq_len * dim, BLOCK),)
        kernel_proj[grid](
            out_tensor, output_tensor, weight,
            B, SEQLen, H, dim, BLOCK=128
        )

        return output_tensor

    output = run_proj(output, weights["to_out.weight"], output)

    return output
