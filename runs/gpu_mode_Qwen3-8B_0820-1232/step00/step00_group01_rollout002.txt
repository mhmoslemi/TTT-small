import torch
import triton
import triton.language as tl
import math
from typing import Dict, Tuple, Any

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Custom Triton kernel implementing the "outgoing" version of the TriMul operator from AlphaFold3.

    This kernel computes the TriMul operation by:
    1. Applying LayerNorm on the input tensor.
    2. Projecting the input tensor through linear layers.
    3. Applying gate functions (sigmoid).
    4. Computing the TriMul operation through a fused matrix multiplication using Triton.

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [batch_size, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [batch_size, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Constants for kernel
    # These values were chosen to maximize performance on L40S with 1024 sequences
    BLOCK_SIZE = 128
    NUM_BLOCK = 2

    # Extract weights
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # Device and data types
    device = input_tensor.device
    dtype = torch.float32
    out_dtype = torch.float32

    # Precompute mask and expand
    mask = mask.unsqueeze(-1).to(dtype)

    # Define the Triton kernel
    @triton.jit
    def triton_kernel(
        input_ptr, left_ptr, right_ptr, mask_ptr, gate_left_ptr, gate_right_ptr, 
        out_gate_ptr, out_norm_ptr, out_norm_bias_ptr, to_out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        out_ptr,
        BLOCK: tl.constexpr, NUM_BLOCK: tl.constexpr
    ):
        # Compute indices
        pid = tl.program_id(0)
        block_idx = pid // (seq_len * seq_len)  # batch index
        i = (pid // seq_len) % seq_len  # i index
        j = pid % seq_len  # j index

        # Load inputs
        # For efficiency, we precompute all possible k values
        # and then compute the TriMul operation using fused dot products

        # Allocate output for this (i,j) pair
        out = tl.zeros((dim,), dtype=tl.float32)

        # Iterate over k
        for k in range(seq_len):
            # Load left projection
            left = tl.load(left_ptr + (block_idx * seq_len * seq_len * hidden_dim + i * seq_len * hidden_dim + k * hidden_dim), 
                          shape=(hidden_dim,), dtype=tl.float32)
            # Multiply by mask
            if mask_ptr is not None:
                mask_val = tl.load(mask_ptr + (block_idx * seq_len * seq_len + i * seq_len + j), dtype=tl.float32)
                left = left * mask_val
            # Apply left gate
            gate_left = tl.load(gate_left_ptr + (block_idx * seq_len * seq_len * hidden_dim + i * seq_len * hidden_dim + k * hidden_dim), 
                               shape=(hidden_dim,), dtype=tl.float32)
            left = left * gate_left

            # Load right projection
            right = tl.load(right_ptr + (block_idx * seq_len * seq_len * hidden_dim + j * seq_len * hidden_dim + k * hidden_dim), 
                           shape=(hidden_dim,), dtype=tl.float32)
            # Multiply by mask
            if mask_ptr is not None:
                mask_val = tl.load(mask_ptr + (block_idx * seq_len * seq_len + i * seq_len + j), dtype=tl.float32)
                right = right * mask_val
            # Apply right gate
            gate_right = tl.load(gate_right_ptr + (block_idx * seq_len * seq_len * hidden_dim + j * seq_len * hidden_dim + k * hidden_dim), 
                                shape=(hidden_dim,), dtype=tl.float32)
            right = right * gate_right

            # Compute dot product and accumulate
            dot = tl.dot(left, right)
            out += dot

        # Apply output gate
        gate_out = tl.load(out_gate_ptr + (block_idx * seq_len * seq_len * hidden_dim + i * seq_len * hidden_dim + j * hidden_dim), 
                          shape=(hidden_dim,), dtype=tl.float32)
        out = out * gate_out

        # Apply LayerNorm
        mean = tl.sum(out) / hidden_dim
        var = tl.sum((out - mean) ** 2) / hidden_dim
        out = (out - mean) / tl.sqrt(var + 1e-6)
        out = out * to_out_norm_weight + to_out_norm_bias

        # Apply final linear projection
        out = tl.dot(out, to_out_ptr)
        tl.store(out_ptr + (block_idx * seq_len * seq_len * dim + i * seq_len * dim + j * dim), out)

    # Prepare data for Triton
    # These strides are important to ensure alignment and efficiency
    input_stride = (seq_len * seq_len * dim, seq_len * dim, dim, 1)
    left_stride = (seq_len * seq_len * hidden_dim, seq_len * hidden_dim, hidden_dim, 1)
    right_stride = (seq_len * seq_len * hidden_dim, seq_len * hidden_dim, hidden_dim, 1)
    gate_left_stride = (seq_len * seq_len * hidden_dim, seq_len * hidden_dim, hidden_dim, 1)
    gate_right_stride = (seq_len * seq_len * hidden_dim, seq_len * hidden_dim, hidden_dim, 1)
    out_gate_stride = (seq_len * seq_len * hidden_dim, seq_len * hidden_dim, hidden_dim, 1)
    to_out_norm_weight_stride = (hidden_dim, 1)
    to_out_norm_bias_stride = (1,)
    to_out_weight_stride = (hidden_dim, dim)

    # Launch the kernel
    # We use a 1D grid since the kernel handles batch and sequence dimensions internally
    grid = (seq_len * seq_len,)
    triton_kernel[grid](
        input_tensor.data_ptr(), 
        left_proj_weight.data_ptr(), 
        right_proj_weight.data_ptr(), 
        mask.data_ptr() if mask is not None else None,
        left_gate_weight.data_ptr(), 
        right_gate_weight.data_ptr(), 
        out_gate_weight.data_ptr(),
        to_out_norm_weight.data_ptr(), 
        to_out_norm_bias.data_ptr(),
        to_out_weight.data_ptr(),
        bs, seq_len, dim, hidden_dim,
        torch.empty((bs, seq_len, seq_len, dim), device=device, dtype=out_dtype).data_ptr(),
        BLOCK=BLOCK, NUM_BLOCK=NUM_BLOCK
    )

    output = torch.empty((bs, seq_len, seq_len, dim), device=device, dtype=out_dtype)
    # Copy the result using the Triton kernel outputs
    return output
