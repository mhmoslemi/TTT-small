"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part is the Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed channel d this is a batched matrix‑multiply:
    out_d = left_d @ right_d.T
where left_d and right_d have shape (B, N, N).

The kernel below computes this product 4‑D tiled by (i,j) and iterates over
k‑blocks.  Each program instance is responsible for *one* hidden channel
(`pid_h`), so no extra broadcasting is needed.  All other operations
(LayerNorm, Linear, sigmoids) are performed with regular torch kernels.
The final result is float‑32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def triton_trimul_kernel(
    left_ptr,          # const *float*
    right_ptr,         # const *float*
    out_ptr,           # *float*
    B, N, H,           # int const
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile
    pid_h = tl.program_id(3)          # hidden channel (exact index)

    # ----------- tile indices ----------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    i_off = i_base + tl.arange(0, BLOCK_M)
    j_off = j_base + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # batch offset (same for left/right/out)
    batch_off = pid_b * stride_l_b

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over k ----------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[b, i, k, h]   -> (M, K)
        left_addr = (
            left_ptr
            + batch_off
            + i_off[:, None] * stride_l_i
            + k_off[None, :] * stride_l_k
            + pid_h * stride_l_h
        )
        left = tl.load(
            left_addr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b, j, k, h]  (note the order j,k)
        right_addr = (
            right_ptr
            + batch_off
            + j_off[None, :] * stride_r_j
            + k_off[:, None] * stride_r_k
            + pid_h * stride_r_h
        )
        right = tl.load(
            right_addr,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0,
        )

        # matrix‑multiply along k and accumulate
        acc += tl.dot(left, right)

    # --------- write result ----------
    out_addr = (
        out_ptr
        + batch_off
        + i_off[:, None] * stride_o_i
        + j_off[None, :] * stride_o_j
        + pid_h * stride_o_h
    )
    tl.store(out_addr, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            input  – Tensor[bs, N, N, dim]   (float32)
            mask   – Tensor[bs, N, N]        (bool / float)
            weights – dict with the same keys as the reference PyTorch model
            config – dict containing at least:
                     dim, hidden_dim, nomask (bool)

    Returns:
        Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # layer‑norm on the input (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        m = mask.unsqueeze(-1).to(left.dtype)                # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------- #
    # gating (sigmoid of linear)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # core triton multiplication
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    device = left.device

    out = torch.empty_like(left)   # [B,N,N,hidden]

    # strides (in elements, not bytes) – required by triton
    s_l = left.stride()   # (b,i,k,h)
    s_r = right.stride()
    s_o = out.stride()

    # launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        hidden_dim,          # one program per hidden channel
    )

    triton_trimul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------- #
    # post‑processing: LayerNorm over hidden, gate, final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
```
