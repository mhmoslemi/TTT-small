"""
TriMul (outgoing) – Triton‑accelerated matrix‑multiply core.

The heavy N³ work is the einsum
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply per hidden dimension:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

We keep the surrounding LayerNorm / linear / gating logic in PyTorch (fast enough)
and off‑load the large batched mat‑mul to a custom Triton kernel.
The kernel works on FP16 inputs (mixed‑precision) and accumulates in FP32,
producing an FP16 result that is later cast back to FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ  (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all = seq_len)
    stride_am, stride_ak,         # A strides: (batch*head, row, col)
    stride_bk, stride_bn,         # B strides (already transposed): (batch*head, row, col)
    stride_cm, stride_cn,         # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                     # block row
    pid_n = tl.program_id(1)                     # block col
    pid_z = tl.program_id(2)                     # batch*head index

    # offsets inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for A and B (starting block)
    a_ptrs = a_ptr + pid_z * stride_am + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bk + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulation buffer (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        a = tl.load(a_ptrs,
                    mask=(offs_m < M) & (cur_k + offs_k < K),
                    other=0.0)                          # (BLOCK_M, BLOCK_K) FP16
        b = tl.load(b_ptrs,
                    mask=(cur_k + offs_k < K) & (offs_n < N),
                    other=0.0)                          # (BLOCK_K, BLOCK_N) FP16
        acc += tl.dot(a, b)                         # FP32 += FP16·FP16

        # advance pointers for the next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write out FP16 result
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_z * stride_cm + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c,
             mask=(offs_m < M) & (offs_n < N))


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # unpack weights (always float32 on device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    left_proj_weight  = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight  = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight   = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)
    to_out_weight      = weights["to_out.weight"].to(device)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)          # [B,N,N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Mask (if provided, it is already on the device)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched mat‑mul: left @ rightᵀ per hidden dimension
    # ------------------------------------------------------------------
    # cast to half for the kernel
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16).transpose(-2, -1)   # transpose N‑N axes

    # collapse batch and hidden into one leading dimension
    BH = B * hidden_dim
    left_flat  = left_h.reshape(BH, N, N).contiguous()
    right_flat = right_h.reshape(BH, N, N).contiguous()
    out_flat   = torch.empty_like(left_flat, dtype=torch.float16)

    # strides for the three‑dim tensors (batch*head, row, col)
    stride_am = left_flat.stride(0)   # = N*N
    stride_ak = left_flat.stride(2)   # = 1
    stride_bk = right_flat.stride(1)  # = N   (row stride after transpose)
    stride_bn = right_flat.stride(2)  # = 1
    stride_cm = out_flat.stride(0)
    stride_cn = out_flat.stride(2)

    # tile sizes – experimentally good for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,                         # one program per (batch * hidden)
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden] (float32)
    out = out_flat.to(torch.float32).reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate                                 # broadcast over hidden dim
    out = F.linear(out, to_out_weight)                  # [B,N,N,dim]

    return out
