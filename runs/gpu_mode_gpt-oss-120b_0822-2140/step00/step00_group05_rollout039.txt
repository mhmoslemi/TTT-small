"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The heavy work of the TriMul module is the three‑dimensional “einsum”

    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]

where ``left``/``right`` are the gated projections of the input tensor
``x ∈ ℝ^{B×N×N×C}``.  For a fixed hidden dimension ``d`` the operation is a
matrix multiplication of size ``N×N``:

    out_{d} = left_{d} @ (right_{d})ᵀ

Thus we have ``B·H`` independent matrix products (``H`` = hidden_dim).  The
kernel below performs a batched GEMM on the flattened tensors
``[B·H, N, N]`` using a classic tiling strategy (BLOCK_M×BLOCK_N tiles,
inner reduction over BLOCK_K).  All other layers (LayerNorm, linear
projections, gating) are kept in PyTorch for simplicity.

The kernel works with float32 data; the surrounding code can use mixed‑
precision if desired, but the final output is cast back to float32 as
required by the spec.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where A,B are shape
# [batch, M, K] and [batch, K, N] respectively.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    stride_Ab, stride_Am, stride_Ak,   # batch, row, col strides for A
    stride_Bb, stride_Bk, stride_Bn,   # batch, row, col strides for B
    stride_Cb, stride_Cm, stride_Cn,   # batch, row, col strides for C
    M, N, K,                           # matrix sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B.

    A : [batch, M, K]   (row‑major)
    B : [batch, K, N]   (row‑major)
    C : [batch, M, N]
    """
    pid = tl.program_id(0)                              # linear program id
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_id = pid // blocks_per_batch                  # which matrix in the batch
    block_id = pid % blocks_per_batch

    pid_m = block_id // num_blocks_n                    # tile row
    pid_n = block_id % num_blocks_n                     # tile col

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the tile
    a_ptrs = A_ptr + batch_id * stride_Ab \
                  + offs_m[:, None] * stride_Am \
                  + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + batch_id * stride_Bb \
                  + tl.arange(0, BLOCK_K)[:, None] * stride_Bk \
                  + offs_n[None, :] * stride_Bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A and B
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) &
                         (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < K) &
                         (offs_n[None, :] < N),
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write out the result tile
    c_ptrs = C_ptr + batch_id * stride_Cb \
                  + offs_m[:, None] * stride_Cm \
                  + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, dim]  (float32)
            - mask         : torch.Tensor[B, N, N]       (bool or float)
            - weights      : dict of weight tensors
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]       # H
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Helper: LayerNorm via torch.nn.functional (no bias unless supplied)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        # x: [..., C]  (float32)
        C = x.shape[-1]
        return F.layer_norm(
            x,
            normalized_shape=(C,),
            weight=weight,
            bias=bias,
            eps=1e-5,
        )

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – use F.linear for speed
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if exists)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(dtype)           # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Tri‑multiplicative update via Triton batched GEMM
    #    Compute: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Equivalent to: for each d, out_d = left_d @ right_dᵀ
    # ------------------------------------------------------------------
    # Reshape to (batch*H, N, N) with right transposed on the last two axes
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # rightᵀ: swap the two N dimensions before flattening
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Launch parameters (tuned for Hopper GPUs)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = B * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = total_batches * grid_m * grid_n
    grid = (total_programs,)

    # Strides for the flattened tensors (row‑major layout)
    # A : [batch, M, K]   -> stride_Ab (batch), stride_Am (row), stride_Ak (col)
    stride_Ab = left_flat.stride(0)   # batch stride = N*N
    stride_Am = left_flat.stride(1)   # row stride   = N
    stride_Ak = left_flat.stride(2)   # col stride   = 1

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)   # row of B (K dimension)
    stride_Bn = right_flat.stride(2)   # col of B (N dimension)

    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Launch the kernel
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        N, N, N,          # M, N, K (all equal to sequence length)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    # Ensure float32 output as required
    out = out.to(torch.float32)

    return out
