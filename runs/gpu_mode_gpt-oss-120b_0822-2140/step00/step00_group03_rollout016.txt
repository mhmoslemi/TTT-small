"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy O(N³) step
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication of
    left   : (M=N_i, K=N_k)   and
    rightᵀ : (K=N_k, N=N_j)
for every batch *and* hidden channel.  The rest (LayerNorm, linear
projections, sigmoid gates) stays in PyTorch for simplicity.

The kernel works on a single hidden channel at a time (hidden_dim ≤ 384)
and tiles the (i , j) output plane with BLOCK_M × BLOCK_N blocks while
iterating over the reduction dimension K in BLOCK_K chunks.
All arithmetic is done in float32 accumulator but the loads are in
float16 for bandwidth‑efficiency.  The final result is cast back to
float32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel : one hidden channel (d) of the einsum
# ------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # * [B, N, N]   (i,k)
    right_ptr,              # * [B, N, N]   (j,k) – we will read it as (k,j)
    out_ptr,                # * [B, N, N]   (i,j)
    B, N,
    stride_lb, stride_li, stride_lk,   # left strides
    stride_rb, stride_rj, stride_rk,   # right strides (original layout)
    stride_ob, stride_oi, stride_oj,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k] for a single
    hidden dimension.  right is accessed transposed (k,j) to turn the
    operation into a classic GEMM."""
    pid = tl.program_id(0)

    # Number of tiles along M and N
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # Decode program id into (batch, tile_i, tile_j)
    b = pid // (num_i * num_j)
    tile_id = pid % (num_i * num_j)
    tile_i = tile_id // num_j
    tile_j = tile_id % num_j

    # Offsets for the M (i) and N (j) dimensions
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the current block (no k yet)
    left_tile_ptr  = left_ptr + \
        b * stride_lb + offs_i[:, None] * stride_li + tl.arange(0, BLOCK_K)[None, :] * stride_lk

    # Right is read as (k, j) → swap the original i/k strides
    right_tile_ptr = right_ptr + \
        b * stride_rb + tl.arange(0, BLOCK_K)[:, None] * stride_rj + offs_j[None, :] * stride_rk

    # Output pointer (i,j) block
    out_tile_ptr = out_ptr + \
        b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)

        # Load tiles (convert to FP16 for bandwidth)
        l = tl.load(left_tile_ptr + k_start * stride_lk,
                    mask=mask_i[:, None] & (tl.arange(0, BLOCK_K) < cur_k),
                    other=0.0).to(tl.float16)
        r = tl.load(right_tile_ptr + k_start * stride_rj,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < cur_k) & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # Matrix multiplication: (M, K) * (K, N) → (M, N)
        acc += tl.dot(l, r, out_dtype=tl.float32)

    # Write result
    tl.store(out_tile_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------------
# Python entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.
    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: layer‑norm without bias handling (bias may be None)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        return F.layer_norm(
            x,
            normalized_shape=(x.shape[-1],),
            weight=weight,
            bias=bias,
            eps=1e-5,
        )

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)          # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy O(N³) einsum → Triton kernel per hidden channel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # [B,N,N,hidden_dim], fp32
    # tile sizes chosen to fit Hopper SM and give good occupancy
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32

    # Pre‑compute strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk = left.stride()
    stride_rb, stride_rj, stride_rk = right.stride()
    stride_ob, stride_oi, stride_oj = out.stride()

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_programs = B * num_i * num_j
    grid = (total_programs,)

    # Launch kernel for each hidden dimension slice
    for h in range(hidden_dim):
        left_h = left[..., h].contiguous()
        right_h = right[..., h].contiguous()
        out_h = out[..., h]

        tri_mul_kernel[grid](
            left_h.data_ptr(),
            right_h.data_ptr(),
            out_h.data_ptr(),
            B, N,
            stride_lb, stride_li, stride_lk,
            stride_rb, stride_rj, stride_rk,
            stride_ob, stride_oi, stride_oj,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    # out is now in fp16 (from kernel), cast back to fp32
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
