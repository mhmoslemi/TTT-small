@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # batch*hidden

    b = pid_z // H
    h = pid_z % H

    # compute offsets
    offs_i = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M,K)
        left_offset = (
            b * stride_b +
            h * stride_h +
            offs_i[:, None] * stride_i +
            offs_k[None, :] * stride_k
        )
        left = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed (K,N) where we need right[j,k] => we load right as (K,N) with k as first dim
        right_offset = (
            b * stride_b +
            h * stride_h +
            offs_k[:, None] * stride_k +
            offs_j[None, :] * stride_i  # because original right stride over j (i dimension) same as i stride
        )
        right = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot
        acc += tl.dot(left, right)  # left MxK, right KxN = MxN

    # store
    out_offset = (
        b * stride_b +
        h * stride_h +
        offs_i[:, None] * stride_i +
        offs_j[None, :] * stride_k
    )
    tl.store(out_ptr + out_offset, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing” implementation.

The heavy N³ reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that tiles the i‑ and j‑axes
and iterates over k in small blocks.  All other linear layers,
LayerNorms and sigmoid gates are kept in PyTorch (they are cheap
relative to the N³ work).  The kernel works for any batch size B,
sequence length N and hidden dimension H (called hidden_dim in the
configuration) and produces a float‑32 output identical to the reference
PyTorch version.

The kernel computes a block‑wise matrix multiplication for each
pair (i‑block, j‑block) and each (batch, hidden) slice, accumulating the
partial sums in FP32.  Masks are applied before the kernel, therefore
the kernel sees dense tensors.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out = Σₖ left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
        left_ptr,          # const *[B,N,N,H]
        right_ptr,         # const *[B,N,N,H]
        out_ptr,           # *[B,N,N,H]
        B, N, H,           # int32 scalars
        stride_b, stride_i, stride_k, stride_h,   # int32 strides of the tensors
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Tile over (i,j) and reduce over k."""
    pid_i = tl.program_id(0)          # i‑block index
    pid_j = tl.program_id(1)          # j‑block index
    pid_z = tl.program_id(2)          # combined (batch, hidden) index

    # split the combined id
    batch_id = pid_z // H
    hidden_id = pid_z % H

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in BLOCK_K sized tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,d]   -> shape (BLOCK_M, BLOCK_K)
        left_offset = (
            batch_id * stride_b +
            hidden_id * stride_h +
            offs_i[:, None] * stride_i +
            offs_k[None, :] * stride_k
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b,j,k,d] → we need it transposed to (BLOCK_K, BLOCK_N)
        #   right_offset uses stride_k for the k‑dimension and stride_i for j
        right_offset = (
            batch_id * stride_b +
            hidden_id * stride_h +
            offs_k[:, None] * stride_k +
            offs_j[None, :] * stride_i   # note: stride_i == stride over the second axis
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # acc += left @ right   (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_offset = (
        batch_id * stride_b +
        hidden_id * stride_h +
        offs_i[:, None] * stride_i +
        offs_j[None, :] * stride_k
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]   (float32)
            mask_tensor  : [B, N, N]        (bool / 0‑1 float)
            weights_dict : mapping str → torch.Tensor (all float32)
            config_dict  : {"dim": int, "hidden_dim": int, ...}
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim = last dimension)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any).  The mask is broadcast over the hidden dim.
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones_like(left[..., 0], device=device, dtype=left.dtype)
    mask = mask.unsqueeze(-1)               # [B,N,N,1]
    left = left * mask
    right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Main N³ reduction via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Triton launch parameters – these work well on Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),                 # i‑blocks
        triton.cdiv(N, BLOCK_N),                 # j‑blocks
        B * H                                     # combined (batch, hidden)
    )

    # Strides of the underlying contiguous tensor [B,N,N,H]
    stride_b, stride_i, stride_k, stride_h = left.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_b, stride_i, stride_k, stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating and final projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
