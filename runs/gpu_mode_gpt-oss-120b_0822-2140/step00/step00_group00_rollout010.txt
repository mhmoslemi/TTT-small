"""
TriMul “outgoing” kernel (AlphaFold‑3)

We implement the expensive cubic‑order einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
as a batched matrix multiplication A @ Bᵀ for each (batch, d) pair,
where A = left[..., :, :, d] and B = right[..., :, :, d].

The surrounding LayerNorm / Linear / gating ops stay in PyTorch.
The Triton kernel works on fp16 for bandwidth‑limited loads and
accumulates in fp32 for numerical safety.  It supports arbitrary batch
size (≤2), sequence length (up to 1024) and hidden dimension (≤128).

The kernel is launched with a 3‑D grid:
    pid0 → batch * hidden_dim   (one matrix pair per thread‑block)
    pid1 → block row (i)
    pid2 → block column (j)
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ, where A and B have shape
#   (B, N, N, H)   →  treated as H independent B×N×N matrices.
# -------------------------------------------------------------
@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,          # tensor pointers
    B, N, H,                               # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ---------------------------------------------------------
    # 1) identify (batch, hidden) pair handled by this block
    # ---------------------------------------------------------
    pid0 = tl.program_id(0)                     # 0 … B*H‑1
    b = pid0 // H                               # batch index
    h = pid0 % H                                # hidden‑dim index

    pid_i = tl.program_id(1)                    # block row index
    pid_j = tl.program_id(2)                    # block col index

    # ---------------------------------------------------------
    # 2) offsets for the M and N dimensions
    # ---------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # 3) pointer arithmetic base for the current (b, h) pair
    # -----------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # -----------------------------------------------------------------
    # 4) initialise accumulator (fp32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # 5) loop over the reduction dimension K
    # -----------------------------------------------------------------
    num_k_blocks = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_k_blocks):
        # linear index of the k‑block
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block [i, k]
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)

        # load B block [j, k]  (note: we will transpose on‑the‑fly)
        b_ptrs = right_base \
                 + offs_j[:, None] * stride_right_i \
                 + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        b = b.to(tl.float16)

        # acc += A @ Bᵀ   (tl.dot supports transposition)
        acc += tl.dot(a, b, trans_b=True)

    # -----------------------------------------------------------------
    # 6) write back result
    # -----------------------------------------------------------------
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Host entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Parts that are O(N³) are executed in the Triton kernel above.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # -----------------------------------------------------------------
    # unpack config
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # LayerNorm on the input (dim = C)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Apply mask if present
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Triton kernel: compute out = left @ rightᵀ for each hidden dim
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape          # (B, N, N, hidden_dim)

    # cast to fp16 for the kernel (still keep a fp32 copy for later)
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    out_f16 = torch.empty_like(left_f16, dtype=torch.float16, device=device)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # pid0 : batch * hidden
        triton.cdiv(N, BLOCK_M),           # pid1 : block rows
        triton.cdiv(N, BLOCK_N),           # pid2 : block cols
    )

    tri_mul_out_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B, N, hidden_dim,
        left_f16.stride(0), left_f16.stride(1), left_f16.stride(2), left_f16.stride(3),
        right_f16.stride(0), right_f16.stride(1), right_f16.stride(2), right_f16.stride(3),
        out_f16.stride(0), out_f16.stride(1), out_f16.stride(2), out_f16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to fp32 for the rest of the pipeline
    out = out_f16.to(torch.float32)

    # -----------------------------------------------------------------
    # Final LayerNorm, gating and projection back to original dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
