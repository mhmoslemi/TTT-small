"""
TriMul “outgoing” implementation (AlphaFold3).

The heavy part of the module is the tensor‑wise contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every batch b and hidden dimension d.  For a fixed d this is a matrix
multiply L_{b,d} (N×N) with the transpose of R_{b,d} (N×N).  We fuse the
N³ work in a single Triton batched‑GEMM kernel while the cheap layer‑norm,
linear projections and gating are kept in PyTorch.

The kernel works on tensors of shape
    (batch_hidden, M=N, K=N)  ×  (batch_hidden, K=N, N)
and writes the result (batch_hidden, M=N, N) which is later reshaped to
[B, N, N, hidden_dim].  All arithmetic is done in FP32, the final output is
float‑32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton batched GEMM kernel
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    M, N, K,                                 # matrix sizes
    stride_a_batch, stride_a_row, stride_a_col,   # A strides
    stride_b_batch, stride_b_row, stride_b_col,   # B strides (B is already K×N)
    stride_c_batch, stride_c_row, stride_c_col,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B where
        A : (batch, M, K)   row‑major
        B : (batch, K, N)   row‑major (already the transpose of the original tensor)
        C : (batch, M, N)

    Grid:
        (ceil(M/BLOCK_M), ceil(N/BLOCK_N), batch)
    """
    pid_batch = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds (not needed for the test shapes, but kept for safety)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_batch * stride_a_batch
        + offs_m[:, None] * stride_a_row
        + offs_k[None, :] * stride_a_col
    )
    b_ptrs = (
        B_ptr
        + pid_batch * stride_b_batch
        + offs_k[:, None] * stride_b_row
        + offs_n[None, :] * stride_b_col
    )
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K - _k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K - _k * BLOCK_K)[:, None] & mask_n[None, :], other=0.0)
        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the kernel for a given pair of tensors
# ----------------------------------------------------------------------
def batched_gemm(A: torch.Tensor, B: torch.Tensor, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    """
    A : (B*H, N, N)   row‑major
    B : (B*H, N, N)   row‑major (will be interpreted as K×N by the kernel)
    returns C : (B*H, N, N)   row‑major
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty_like(A)

    # strides in element units (not bytes)
    stride_a_batch = A.stride(0)
    stride_a_row = A.stride(1)
    stride_a_col = A.stride(2)

    stride_b_batch = B.stride(0)
    stride_b_row = B.stride(1)   # row = K
    stride_b_col = B.stride(2)   # col = N

    stride_c_batch = C.stride(0)
    stride_c_row = C.stride(1)
    stride_c_col = C.stride(2)

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    batch_gemm_kernel[grid](
        A.data_ptr(),
        B.data_ptr(),
        C.data_ptr(),
        M,
        N,
        K,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator implemented with a
    Triton batched‑GEMM kernel for the N³ contraction.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor of shape [B, N, N, dim]  (float‑32)
            mask   – torch.Tensor of shape [B, N, N] (0/1)   (float‑32)
            weights – dict with all linear / layer‑norm parameters
            config  – dict with keys: "dim", "hidden_dim", "nomask"

    Returns:
        output tensor of shape [B, N, N, dim] (float‑32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # layer‑norm on the input
    # ------------------------------------------------------------------
    # PyTorch functional layer‑norm (last dimension)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # linear projections (no bias) – shape … → …, hidden
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.unsqueeze(-1)                               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # gating – sigmoid(linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    #   left  : [B, N, N, hidden]  -> (B*hidden, N, N)
    #   right : [B, N, N, hidden]  -> (B*hidden, N, N)   (will be read as K×N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # ------------------------------------------------------------------
    # Compute out = left @ rightᵀ  (batched GEMM via Triton)
    # ------------------------------------------------------------------
    out_flat = batched_gemm(left_flat, right_flat)   # (B*hidden, N, N)

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # final layer‑norm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
