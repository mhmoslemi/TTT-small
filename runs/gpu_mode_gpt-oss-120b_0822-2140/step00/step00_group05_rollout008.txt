"""TriMul “outgoing” implementation with a Triton kernel.

The heavy part of the TriMul forward pass is the tensor contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is exactly a batched matrix multiplication
`left @ right.transpose(-2, -1)` for each hidden‑channel h.
The kernel below performs this multiplication for all batches and
hidden channels in FP16 (accumulating in FP32) using a tiled GEMM.
All remaining operations (LayerNorm, linear projections, gates,
and the final linear) are kept in PyTorch for simplicity.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _matmul_T_kernel(
    A,                     # left  [B*H, N, N]  (FP16)
    B,                     # right [B*H, N, N]  (FP16)
    C,                     # out   [B*H, N, N]  (FP16)
    N,                     # matrix size
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                # tile row
    pid_n = tl.program_id(1)                # tile col
    pid_b = tl.program_id(2)                # combined batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    a_batch_off = pid_b * stride_a_batch
    b_batch_off = pid_b * stride_b_batch
    c_batch_off = pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]
        a_ptr = A + a_batch_off \
                + offs_m[:, None] * stride_a_row \
                + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B_T[k, j] = B[j, k]   (notice the swapped indices)
        b_ptr = B + b_batch_off \
                + offs_k[:, None] * stride_b_row \
                + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (M,K) @ (K,N) → (M,N)

    c_ptr = C + c_batch_off \
            + offs_m[:, None] * stride_c_row \
            + offs_n[None, :] * stride_c_col
    tl.store(c_ptr, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_transpose(left: torch.Tensor,
                            right: torch.Tensor) -> torch.Tensor:
    """
    Computes `left @ right.transpose(-2,-1)` for tensors of shape
    [B, N, N, H] using the Triton kernel above.
    Returns a tensor of the same shape (float32).
    """
    B, N, _, H = left.shape

    # bring the hidden dimension next to the batch for easy flattening
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # FP16 for the kernel (accumulate in FP32)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )

    _matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        left_fp16.stride(0),  # batch stride
        left_fp16.stride(1),  # row stride
        left_fp16.stride(2),  # col stride
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H] and cast to FP32
    out = out_fp16.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core multiplicative update (Triton) ----------
    out = _batched_gemm_transpose(left, right)   # [B, N, N, H]

    # ---------- Output normalization ----------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ---------- Final linear ----------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
