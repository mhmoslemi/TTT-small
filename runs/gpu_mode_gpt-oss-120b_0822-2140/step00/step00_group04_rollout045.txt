"""
TriMul “outgoing” implementation using Triton.

The heavy N³ part is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply per hidden dimension d:
    out_{b,d} = left_{b,d} @ right_{b,d}.T
The kernel computes all (batch, hidden) pairs in parallel, tiling the
output matrix with a classic GEMM loop.  All other operations
(LayerNorm, linear projections, gating, final LayerNorm/linear) are
performed with PyTorch (they are cheap compared to the N³ work).

The kernel works with float16 for compute (accumulates in fp32) and the
final result is cast back to float32 to match the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: compute out = left @ right.T  (per batch, per head)
# ------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    out_ptr,                # *mut f16   [B, N, N, H]
    left_ptr,               # *const f16 [B, N, N, H]
    right_ptr,              # *const f16 [B, N, N, H]
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_l_b,  stride_l_i,  stride_l_k,  stride_l_h,
    stride_r_b,  stride_r_j,  stride_r_k,  stride_r_h,
    BLOCK_M: tl.constexpr,  # tile rows of output
    BLOCK_N: tl.constexpr,  # tile cols of output
    BLOCK_K: tl.constexpr,  # reduction tile
):
    pid0 = tl.program_id(0)          # encodes (batch, head)
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # decode batch and hidden dimension
    b = pid0 // H
    h = pid0 % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the beginning of the (b,*,*,h) slice
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h

    # accumulator (fp32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (i,k) ----
        # left layout: (b, i, k, h)
        left_ptrs = left_base \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- load right tile (k,j)  (note the transpose) ----
        # right layout: (b, j, k, h)   --> we address it as (b, k, j, h)
        right_ptrs = right_base \
            + offs_k[:, None] * stride_r_k \
            + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ---- GEMM reduction ----
        acc += tl.dot(left, right)

    # ---- store result ----
    out_ptrs = out_base \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Host‑side entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask          : torch.Tensor [B, N, N]      (float32 or bool)
        - weights       : dict of pre‑loaded weight tensors
        - config        : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    #################################################################
    # 1.  LayerNorm on the input (dim -> dim)
    #################################################################
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # still float32

    #################################################################
    # 2.  Linear projections (dim -> hidden_dim) – no bias
    #################################################################
    left_proj_w = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    #################################################################
    # 3.  Apply mask (if any)
    #################################################################
    if mask is not None:
        m = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * m
        right = right * m

    #################################################################
    # 4.  Gating (sigmoid) and gated multiplication
    #################################################################
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    #################################################################
    # 5.  Core N³ multiplication – Triton kernel
    #################################################################
    # Cast to float16 for the kernel (accuracy stays within spec)
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    out_f16   = torch.empty((B, N, N, hidden_dim),
                            dtype=torch.float16,
                            device=input_tensor.device)

    # strides (int) – note that .stride returns number of elements
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_f16.stride()
    stride_l_b,  stride_l_i,  stride_l_k,  stride_l_h  = left_f16.stride()
    stride_r_b,  stride_r_j,  stride_r_k,  stride_r_h  = right_f16.stride()

    # Tile sizes – chosen to fit H200 SMs and give good occupancy
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                 # program id 0  (batch * head)
        triton.cdiv(N, BLOCK_M),        # program id 1  (row tiles)
        triton.cdiv(N, BLOCK_N),        # program id 2  (col tiles)
    )

    _trimul_outgoing_kernel[grid](
        out_f16,
        left_f16,
        right_f16,
        B, N, hidden_dim,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_l_b,  stride_l_i,  stride_l_k,  stride_l_h,
        stride_r_b,  stride_r_j,  stride_r_k,  stride_r_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # back to float32 for the rest of the pipeline
    out = out_f16.to(torch.float32)

    #################################################################
    # 6.  Final LayerNorm + gating + linear (hidden_dim -> dim)
    #################################################################
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)         # [B,N,N,dim]

    return out
