"""
TriMul “outgoing” kernel (AlphaFold‑3 style) – Triton‑accelerated core
===================================================================

The forward pass consists of three stages

1️⃣  Layer‑norm + linear projections + sigmoid gates (pure PyTorch).
2️⃣  The expensive tensor contraction  

        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    is executed by a custom Triton kernel.  
    For every batch element we tile the (i, j) plane (BLOCK_M × BLOCK_N) and
    accumulate the reduction over *k* in shared registers.  The hidden
    dimension *h* is also tiled (BLOCK_H).  The kernel works for any
    ``B ∈ {1,2}``, ``N ∈ {128,…,1024}`` and ``H = hidden_dim`` (typically 128).

3️⃣  Another layer‑norm, gating and a final linear projection (PyTorch).

All tensors are assumed to be on the CUDA device already.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel – batched matmul with an outer‑product reduction
# -------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # [B, N, N, H]   (b, i, k, h)
    right_ptr,              # [B, N, N, H]   (b, j, k, h)
    out_ptr,                # [B, N, N, H]   (b, i, j, h)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile size for i
    BLOCK_N: tl.constexpr,  # tile size for j
    BLOCK_H: tl.constexpr,  # tile size for hidden dim
    BLOCK_K: tl.constexpr,  # tile size for reduction over k
):
    # ---------------------------------------------------------
    # program ids : (batch, tile_i, tile_j)
    # ---------------------------------------------------------
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # -----------------------------------------------------------------
    # compute the start of the tile in each dimension (clamped later)
    # -----------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    b0 = pid_b

    # range vectors
    i_idx = tl.arange(0, BLOCK_M)
    j_idx = tl.arange(0, BLOCK_N)
    k_idx = tl.arange(0, BLOCK_K)
    h_idx = tl.arange(0, BLOCK_H)

    # -----------------------------------------------------------------
    # iterate over the hidden dimension in blocks
    # -----------------------------------------------------------------
    for h0 in range(0, H, BLOCK_H):
        # accumulator for this (i,j,h) tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # -----------------------------------------------------------------
        # reduction over the k‑dimension
        # -----------------------------------------------------------------
        for k0 in range(0, N, BLOCK_K):
            # -----------------------
            # load left  : (M, K, H)
            # -----------------------
            offs_l_i = i0 + i_idx[:, None, None]          # (M,1,1)
            offs_l_k = k0 + k_idx[None, :, None]          # (1,K,1)
            offs_l_h = h0 + h_idx[None, None, :]          # (1,1,H)

            ptr_l = left_ptr + \
                    b0 * stride_l_b + \
                    offs_l_i * stride_l_i + \
                    offs_l_k * stride_l_k + \
                    offs_l_h * stride_l_h
            mask_i = (i0 + i_idx) < N
            mask_k = (k0 + k_idx) < N
            mask_h = (h0 + h_idx) < H
            mask_l = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
            left = tl.load(ptr_l, mask=mask_l, other=0.0)   # (M,K,H)

            # -----------------------
            # load right : (N, K, H)
            # -----------------------
            offs_r_j = j0 + j_idx[:, None, None]          # (N,1,1)
            offs_r_k = k0 + k_idx[None, :, None]          # (1,K,1)
            offs_r_h = h0 + h_idx[None, None, :]          # (1,1,H)

            ptr_r = right_ptr + \
                    b0 * stride_r_b + \
                    offs_r_j * stride_r_j + \
                    offs_r_k * stride_r_k + \
                    offs_r_h * stride_r_h
            mask_j = (j0 + j_idx) < N
            mask_r = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
            right = tl.load(ptr_r, mask=mask_r, other=0.0)   # (N,K,H)

            # ---------------------------------------------------------
            # outer‑product over the current k‑slice
            #   acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]
            # ---------------------------------------------------------
            for kk in range(0, BLOCK_K):
                # guard for tail of k
                k_cond = (k0 + kk) < N
                if not k_cond:
                    continue

                l_vec = left[:, kk, :]          # (M,H)
                r_vec = right[:, kk, :]         # (N,H)

                # broadcast to (M,N,H) and accumulate
                acc += tl.broadcast_to(l_vec[:, None, :], (BLOCK_M, BLOCK_N, BLOCK_H)) * \
                       tl.broadcast_to(r_vec[None, :, :], (BLOCK_M, BLOCK_N, BLOCK_H))

        # -------------------------------------------------------------
        # store the result tile
        # -------------------------------------------------------------
        offs_o_i = i0 + i_idx[:, None, None]
        offs_o_j = j0 + j_idx[None, :, None]
        offs_o_h = h0 + h_idx[None, None, :]

        ptr_o = out_ptr + \
                b0 * stride_o_b + \
                offs_o_i * stride_o_i + \
                offs_o_j * stride_o_j + \
                offs_o_h * stride_o_h

        mask_i = (i0 + i_idx) < N
        mask_j = (j0 + j_idx) < N
        mask_h = (h0 + h_idx) < H
        mask_o = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]

        tl.store(ptr_o, acc, mask=mask_o)


# -------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : (input, mask, weights, config)
        input  – [B, N, N, dim]  (float32)
        mask   – [B, N, N]       (float32 or bool, may be all‑ones)
        weights – dict with the ten linear / norm parameters
        config  – dict with keys "dim" and "hidden_dim"
    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # denotes H (often 128)

    # ---------------------------------------------------------
    # Helper to apply LayerNorm using supplied weight / bias
    # ---------------------------------------------------------
    def layer_norm(x, weight, bias):
        # x: [..., C]
        return F.layer_norm(x, (x.shape[-1],), weight=weight, bias=bias, eps=1e-5)

    # -----------------------------------------------------------------
    # 1️⃣  Layer‑norm, linear projections, mask and gates (PyTorch)
    # -----------------------------------------------------------------
    x = layer_norm(input_tensor,
                   weights["norm.weight"].to(device),
                   weights["norm.bias"].to(device))

    # Linear projections (no bias) → shape [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(device), bias=None
    )          # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(device), bias=None
    )

    # Apply mask if present (mask shape [B,N,N])
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"].to(device), bias=None)
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"].to(device), bias=None)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"].to(device), bias=None)
    )

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 2️⃣  Triton kernel – the huge einsum contraction
    # ---------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides for the 4‑D tensors (contiguous layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – chosen to fit comfortably in registers / shared memory
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_H = 32
    BLOCK_K = 32

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_einsum_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H, BLOCK_K=BLOCK_K,
        num_warps=4,  # heuristic – works well on Hopper
    )

    # ---------------------------------------------------------
    # 3️⃣  Final layer‑norm, output gating and linear projection
    # ---------------------------------------------------------
    out = layer_norm(out,
                     weights["to_out_norm.weight"].to(device),
                     weights["to_out_norm.bias"].to(device))

    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(device), bias=None
    )   # shape [B,N,N,dim]

    return out.to(torch.float32)
