"""
TriMul “outgoing” implementation with a fused Triton GEMM.

The heavy N³ operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched matrix multiplication:
    left_flat  = left .permute(0,3,1,2)   # (B, D, N, N) → (B*D, N, N)
    right_flat = right.permute(0,3,2,1)  # (B, D, N, N) → (B*D, N, N)
    out_flat   = left_flat @ right_flat      (batched GEMM)

The GEMM is performed by a tiled Triton kernel (float32 accumulation,
output cast to float32).  All remaining LayerNorm/Linear/gates are
handled with regular PyTorch ops – they are cheap compared to the
N³ contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: batched GEMM  (C = A @ B)
# A : (batch, M, K)   – row‑major
# B : (batch, K, N)   – row‑major (already transposed in host)
# C : (batch, M, N)
# ---------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,          # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,          # B: batch, row(K), col(N) strides
    stride_cb, stride_cm, stride_cn,          # C: batch, row, col strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    batch = tl.program_id(2)          # batch id

    # -----------------------------------------------------------
    # offsets for the current block
    # -----------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    m_mask = offs_m < M
    n_mask = offs_n < N

    # Pointers for this batch
    a_ptrs = A_ptr + batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------
    # K‑loop
    # -----------------------------------------------------------
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles, respecting out‑of‑bounds with mask and zero‑fill
        a = tl.load(a_ptrs, mask=m_mask[:, None] & (cur_k * BLOCK_K + offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=n_mask[None, :] & (cur_k * BLOCK_K + offs_k < K), other=0.0)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N)

        # Move pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -----------------------------------------------------------
    # Write back
    # -----------------------------------------------------------
    c_ptrs = C_ptr + batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=m_mask[:, None] & n_mask[None, :])


# ---------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------- LayerNorm ---------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------- Linear projections ------------------
    # left / right projections
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling (optional)
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------- Gates -------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # -----------------------------------------------------------------
    # Prepare tensors for the batched GEMM (convert to (B*D, N, N))
    # -----------------------------------------------------------------
    # left: (B, N, N, D) → (B, D, N, N) → (B*D, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    # right: (B, N, N, D) → (B, D, N, N) with k‑axis first → (B*D, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(-1, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # Launch Triton kernel
    # -----------------------------------------------------------------
    total_batches = left_flat.shape[0]          # B * hidden_dim

    # strides for A (left_flat)
    stride_ab = left_flat.stride(0)   # batch stride
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)

    # strides for B (right_flat)
    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # row stride (K)   (since we already transposed)
    stride_bn = right_flat.stride(2)   # col stride (N)

    # strides for C (out_flat)
    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        total_batches,                    # batch dimension
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim)
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden]

    # --------------------- Output LayerNorm ----------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply out‑gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
