"""
tri_mul_triton.py
-----------------
A Triton‑accelerated implementation of the *outgoing* TriMul operation
used in AlphaFold‑3 and related protein‑structure models.

The heavy part of the forward pass is the three‑dimensional contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which can be rewritten for each hidden channel *d* as a matrix
multiplication  out_d = left_d @ right_dᵀ  (shape N×N).

The kernel below evaluates this contraction with one CUDA thread per
output element (b,i,j) and tiles the reduction dimension *k*.
All surrounding linear layers, LayerNorms and sigmoid gates are
performed with PyTorch – only the N³ reduction is off‑loaded to Triton.
The final output is returned in `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: one thread per (batch, i, j) element
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const float*  [B, N, N, H]
    right_ptr,           # const float*  [B, N, N, H]
    out_ptr,             #       float*  [B, N, N, H]
    B, N, H,             # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # output strides
    BLOCK_K: tl.constexpr,                          # tile size along k
):
    """Compute out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]"""

    pid = tl.program_id(0)                     # 0‑D grid, one thread per output element
    total = B * N * N
    # decode linear thread id -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for the three tensors (without the k and hidden dim)
    left_base  = b * stride_l_b + i * stride_l_i
    right_base = b * stride_r_b + j * stride_r_j
    out_base   = b * stride_o_b + i * stride_o_i + j * stride_o_j

    # accumulator for the hidden dimension (size H)
    acc = tl.zeros((H,), dtype=tl.float32)

    # tiled loop over the k‑dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        k_mask = cur_k < N                                         # mask for out‑of‑bounds k

        # pointers for a (BLOCK_K, H) tile of left and right
        left_ptrs  = left_ptr  + left_base \
                    + cur_k[:, None] * stride_l_k \
                    + tl.arange(0, H)[None, :] * stride_l_h
        right_ptrs = right_ptr + right_base \
                    + cur_k[:, None] * stride_r_k \
                    + tl.arange(0, H)[None, :] * stride_r_h

        # load tiles, zero‑pad the tail
        left_tile  = tl.load(left_ptrs,  mask=k_mask[:, None], other=0.0)   # (BLOCK_K, H)
        right_tile = tl.load(right_ptrs, mask=k_mask[:, None], other=0.0)   # (BLOCK_K, H)

        # element‑wise product and reduction over k
        prod = left_tile * right_tile                      # (BLOCK_K, H)
        acc += tl.sum(prod, axis=0)                        # (H,)

    # write the result
    out_ptrs = out_ptr + out_base + tl.arange(0, H) * stride_o_h
    tl.store(out_ptrs, acc)


# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def triton_tri_mul(left: torch.Tensor,
                   right: torch.Tensor,
                   mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    left, right: [B, N, N, H] (float32)
    mask:        [B, N, N] (bool/float) or None – broadcasted over the hidden dim.
    Returns:    out tensor of shape [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    device = left.device

    if mask is not None:
        # broadcast mask to hidden dim and apply before the kernel to mimic the reference
        mask = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask
        right = right * mask

    out = torch.empty_like(left)

    # strides (in elements, not bytes)
    l_str = left.stride()      # (sB, sI, sK, sH)
    r_str = right.stride()
    o_str = out.stride()

    # launch configuration
    total_threads = B * N * N
    BLOCK_K = 64

    grid = (total_threads,)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_b=l_str[0],
        stride_l_i=l_str[1],
        stride_l_k=l_str[2],
        stride_l_h=l_str[3],
        stride_r_b=r_str[0],
        stride_r_j=r_str[1],         # note: we reuse the "i" stride of left as "j" for right
        stride_r_k=r_str[2],
        stride_r_h=r_str[3],
        stride_o_b=o_str[0],
        stride_o_i=o_str[1],
        stride_o_j=o_str[2],
        stride_o_h=o_str[3],
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # enough to keep the SM busy
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    N³ contraction.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]       (float32)
        - mask         : [B, N, N]            (float32 or bool) – may be all‑ones
        - weights      : dict of model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """

    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)                # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Gates (sigmoid) – also linear, no bias
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # apply gates and (optional) mask
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton‑accelerated contraction
    # ------------------------------------------------------------------
    # mask may be omitted (nomask case); if so we treat it as all‑ones.
    if mask is None:
        mask_tensor = None
    else:
        mask_tensor = mask.to(dtype)  # ensure same dtype for broadcasting
    contracted = triton_tri_mul(left, right, mask_tensor)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim) + final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    contracted = F.layer_norm(contracted,
                              normalized_shape=(hidden_dim,),
                              weight=to_out_norm_weight,
                              bias=to_out_norm_bias,
                              eps=1e-5)

    contracted = contracted * out_gate                                   # gating

    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)
    output = F.linear(contracted, to_out_weight)       # [B,N,N,dim]

    return output.to(dtype)
