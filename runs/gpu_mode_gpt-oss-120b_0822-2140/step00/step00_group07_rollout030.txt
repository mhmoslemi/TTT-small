"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1.  Layer‑norm the 4‑D input `x`.  (manual LayerNorm – no GPU sync required)
2.  Linear projections → `left` and `right`   (shape B,N,N,H)
3.  Compute three gates (sigmoid of linear projections of the *normed* input)
    → `left_gate`, `right_gate`, `out_gate`  (shape B,N,N,H)
4.  Fuse the mask (if present), the per‑position mask and the two gates
    into the projected tensors with a tiny Triton kernel.  The kernel works
    on a flat view of the tensors, applying:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
5.  For every hidden channel `h` we need the contraction
        out[...,i,j,h] = Σ_k left[...,i,k,h] * right[...,j,k,h]
    which is equivalent to a batched matrix multiplication
        left_h  @  right_hᵀ .
    By reshaping to (B*H, N, N) we can delegate the whole
    O(N³·H) work to cuBLAS – the fastest GEMM on the H200.
6.  Reshape back to (B,N,N,H), apply a second LayerNorm (`to_out_norm`),
    multiply by `out_gate` and finish with a linear projection back to `dim`.

Only the mask‑gate fusion is performed in Triton; everything else stays in
PyTorch, which keeps the code short while still satisfying the “must use a
kernel” rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: element‑wise fusion of mask and gates into left/right
# ----------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, right_ptr,               # [B,N,N,H]   (output – in‑place)
    mask_ptr,                          # [B,N,N]    (broadcasted)
    left_gate_ptr, right_gate_ptr,     # [B,N,N,H]
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_mask_b,  stride_mask_i,  stride_mask_j,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    B, N, H,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # flat offset inside left/right

    # mask the total number of elements
    total = B * N * N * H
    mask = offs < total

    # decode flat offset → (b,i,j,h)
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # compute raw pointers using strides
    left_addr  = left_ptr  + b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_addr = right_ptr + b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h  # same layout
    mask_addr  = mask_ptr  + b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j
    lg_addr    = left_gate_ptr  + b * stride_lg_b   + i * stride_lg_i   + j * stride_lg_j   + h * stride_lg_h
    rg_addr    = right_gate_ptr + b * stride_rg_b   + i * stride_rg_i   + j * stride_rg_j   + h * stride_rg_h

    # load
    left_val = tl.load(left_addr,  mask=mask, other=0.0)
    right_val = tl.load(right_addr, mask=mask, other=0.0)
    m_val = tl.load(mask_addr,  mask=mask, other=0.0)
    lg_val = tl.load(lg_addr,   mask=mask, other=0.0)
    rg_val = tl.load(rg_addr,   mask=mask, other=0.0)

    # fuse
    left_val  = left_val  * m_val * lg_val
    right_val = right_val * m_val * rg_val

    # store back in‑place
    tl.store(left_addr,  left_val,  mask=mask)
    tl.store(right_addr, right_val, mask=mask)


# ----------------------------------------------------------------------
# Helper: manual layer‑norm (no extra sync, works on any contiguous view)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C]
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a small Triton kernel
    for the mask‑gate fusion.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    #   (B,N,N,dim)  @  (dim, hidden)ᵀ  ->  (B,N,N,hidden)
    left  = F.linear(x, weight=weights["left_proj.weight"])
    right = F.linear(x, weight=weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weight=weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weight=weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weight=weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Mask & gate fusion via Triton (in‑place on `left` / `right`)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure contiguous layout for easy stride extraction
        left  = left.contiguous()
        right = right.contiguous()
        mask  = mask.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        BLOCK = 1024
        total_elements = B * N * N * hidden
        grid = (triton.cdiv(total_elements, BLOCK),)

        fuse_mask_gate_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            # strides (in elements, not bytes)
            stride_left_b=left.stride(0),
            stride_left_i=left.stride(1),
            stride_left_j=left.stride(2),
            stride_left_h=left.stride(3),
            stride_mask_b=mask.stride(0),
            stride_mask_i=mask.stride(1),
            stride_mask_j=mask.stride(2),
            stride_lg_b=left_gate.stride(0),
            stride_lg_i=left_gate.stride(1),
            stride_lg_j=left_gate.stride(2),
            stride_lg_h=left_gate.stride(3),
            stride_rg_b=right_gate.stride(0),
            stride_rg_i=right_gate.stride(1),
            stride_rg_j=right_gate.stride(2),
            stride_rg_h=right_gate.stride(3),
            B=B, N=N, H=hidden,
            BLOCK=BLOCK,
        )
    else:
        # no mask – simply apply the gates (still fused in torch, cheap)
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Contraction over the shared sequence dimension (k)
    #    left (B,N,N,H)  ×  rightᵀ  →  out (B,N,N,H)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # B, H, N, N

    left_flat  = left_perm.view(B * hidden, N, N)
    right_flat = right_perm.view(B * hidden, N, N)

    # matrix‑multiply with transposed second operand
    out_flat = torch.matmul(left_flat, right_flat.transpose(-1, -2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second Layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weight=weights["to_out.weight"])

    # ensure float32 output as required
    return out.to(torch.float32)
