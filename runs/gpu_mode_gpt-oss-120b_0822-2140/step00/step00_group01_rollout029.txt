"""
TriMul “outgoing” implementation.

The heavy 4‑D tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel.  The rest of the
operations (LayerNorm, linear projections, gating, final
normalisation and projection) are kept in PyTorch for simplicity.
The kernel tiles the i, j and k axes (32×32×32) and the hidden
dimension d (32) and accumulates in float32.  The final output is
returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel:  batch‑wise tiled matmul over the shared k‑axis for every
# hidden channel d.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # *[B,N,N,D]  left tensor
    right_ptr,         # *[B,N,N,D]  right tensor (will be read as [B,K,J,D])
    out_ptr,           # *[B,N,N,D]  output

    B, N, D,           # sizes
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out = Σ_k left[...,i,k,d] * right[...,j,k,d]"""

    pid = tl.program_id(0)

    # ----- grid decomposition ------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    b  = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_block = tmp // num_j
    j_block = tmp % num_j

    # ----- offsets -----------------------------------------------------------
    i_offset = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = tl.arange(0, BLOCK_D)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ----- loop over k -------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ------- load left[i,k,d] (shape B,I,K,D) ---------------------------
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_offset[:, None, None] * stride_li
            + k_offset[None, :, None] * stride_lk
            + d_offset[None, None, :] * stride_ld
        )
        left_mask = i_mask[:, None, None] & k_mask[None, :, None]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ------- load right transposed as (k,j,d) ----------------------------
        # original layout is (b, j, k, d); we swap the k/j axes via stride
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + k_offset[:, None, None] * stride_rk   # k as the first dim
            + j_offset[None, :, None] * stride_rj   # j as the second dim
            + d_offset[None, None, :] * stride_rd
        )
        right_mask = k_mask[:, None, None] & j_mask[None, :, None]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ------- inner product over k for each d ----------------------------
        for d in range(0, BLOCK_D):
            # (I,K) @ (K,J) -> (I,J)
            acc[:, :, d] += tl.dot(left_tile[:, :, d], right_tile[:, :, d])

    # ----- write back ---------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_offset[:, None, None] * stride_oi
        + j_offset[None, :, None] * stride_oj
        + d_offset[None, None, :] * stride_od
    )
    out_mask = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: (input, mask, weights, config)
            input  – torch.Tensor  [B, N, N, dim]  (float32/float16)
            mask   – torch.Tensor  [B, N, N]       (0/1)
            weights – dict with all learned parameters
            config – dict containing 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last channel (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask handling
    # ------------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)      # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction with Triton
    # ------------------------------------------------------------------
    # allocate output buffer for the contracted tensor (still hidden_dim)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for contiguous tensors
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tunable tile sizes – chosen to fit Hopper SMs
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32                     # hidden_dim is a multiple of 32 (128/384)

    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    total_blocks = B * grid_i * grid_j
    grid = (total_blocks,)

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6) Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])          # back to dim

    return out.to(torch.float32)
```

**Explanation of the design**

* The `trimul_kernel` tiles the three large dimensions (`i`, `j`, `k`) with
  `BLOCK_I = BLOCK_J = BLOCK_K = 32`.  The hidden dimension `d` is tiled
  with `BLOCK_D = 32`.  For each tile the kernel loads a block of `left`
  (`[I,K,D]`) and a transposed block of `right` (`[K,J,D]`) and performs a
  dot‑product per hidden channel, accumulating in a float32 buffer.
* Masks guarantee correct handling of boundary tiles.
* All other lightweight operations (LayerNorm, linear layers, sigmoid
  gating) remain in PyTorch – they are memory‑bound and cheap compared
  with the O(N³ · hidden_dim) contraction, which dominates the runtime.
* The host function builds the required strides, launches the kernel,
  and then applies the remaining transformations, finally returning a
  float‑32 tensor of shape `[B, N, N, dim]`.
