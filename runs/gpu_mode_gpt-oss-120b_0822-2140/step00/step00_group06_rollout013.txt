@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_n, stride_b_k,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_batch = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_in_batch // tl.cdiv(N, BLOCK_N)
    pid_n = pid_in_batch % tl.cdiv(N, BLOCK_N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offsets = tl.arange(0, BLOCK_K)

    a_ptrs = A_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_m + k_offsets[None, :] * stride_a_k
    b_ptrs = B_ptr + batch * stride_b_batch + offs_n[None, :] * stride_b_n + k_offsets[:, None] * stride_b_k  # because B is transposed: we want B^T (K,N) so loading K dimension contiguous.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (k_offsets + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k_offsets + k*BLOCK_K < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        b_ptrs += BLOCK_K * stride_b_k

"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of TriMul is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply left @ rightᵀ for each hidden
dimension *d*.  The implementation below

* performs the cheap LayerNorm / Linear / sigmoid parts in PyTorch,
* casts the projected tensors to FP16,
* reshapes them to (B*H, N, N) and runs a custom Triton GEMM that
  computes C = A @ Bᵀ  (A,B have shape (B*H, N, N)),
* reshapes the result back and finishes the remaining LayerNorm,
  gating and final linear projection in PyTorch.

The Triton kernel is a classic tiled‑gemm with three launch dimensions:
(m‑tile, n‑tile, batch‑*hid) and uses FP32 accumulation for numerical
stability.  The final output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ   (A : [B*H, M, K], B : [B*H, N, K])
# -------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D launch grid
    pid_m = tl.program_id(0)                     # tile over M
    pid_n = tl.program_id(1)                     # tile over N
    pid_b = tl.program_id(2)                     # batch * hidden

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current tile
    a_ptrs = a_ptr + pid_b * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_a_k

    b_ptrs = b_ptr + pid_b * stride_b_batch \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n   # note: K is first dim → Bᵀ

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) + k * BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) + k * BLOCK_K < K)[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)            # (M,K)·(K,N) → (M,N)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result (convert to fp16 for bandwidth)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------
# Entry point used by the evaluator
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    batch matrix multiplication.
    """
    # unpack -------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out = left @ rightᵀ  (batched GEMM)
    # ------------------------------------------------------------------
    # cast to fp16 for the kernel but keep a fp32 copy for the rest
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape                # B batch, N seq‑len, H = hidden
    batch_h = B * H

    # reshape to (batch_h, N, N)
    left_r  = left_h.reshape(batch_h, N, N)
    right_r = right_h.reshape(batch_h, N, N)

    out_r = torch.empty_like(left_r, dtype=torch.float16, device=left_r.device)

    # tile sizes – empirically good for Hopper (sm90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # m‑tiles
        triton.cdiv(N, BLOCK_N),   # n‑tiles
        batch_h                    # one program id per (batch, hidden)
    )

    _batched_gemm_kernel[grid](
        left_r,
        right_r,
        out_r,
        N, N, N,
        left_r.stride(0), left_r.stride(1), left_r.stride(2),
        right_r.stride(0), right_r.stride(1), right_r.stride(2),
        out_r.stride(0), out_r.stride(1), out_r.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden] and up‑cast to fp32
    out = out_r.reshape(B, N, N, H).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. post‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # out_gate was computed in fp32
    out = F.linear(out, weights["to_out.weight"])

    return out
