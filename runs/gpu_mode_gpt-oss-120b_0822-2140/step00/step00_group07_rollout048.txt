"""
TriMul “outgoing” implementation using Triton for the N³ einsum.
The heavy part
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply of shape (B*D, N, N) × (B*D, N, N)ᵀ.
All cheap ops (layer‑norms, linear projections, sigmoids & gating) are
performed with native torch, while the N³ contraction is realised by a
custom Triton kernel that works in fp16 with fp32 accumulation.
The final result is cast back to fp32 so the surrounding model receives
exactly the same type as the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  C = A @ Bᵀ   where A,B ∈ (M, N, K)  (M = B*hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix dimensions
    stride_am, stride_ak,           # strides for A (M,N,K)
    stride_bk, stride_bn,           # strides for B (M,K,N)  (B is stored transposed)
    stride_cm, stride_cn,           # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds (needed for edge tiles)
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to the start of the tile
    a_ptrs = a_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # fp16 * fp16 → fp32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def tri_mul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: shape (B, N, N, hidden)   dtype=float16
    returns:   shape (B, N, N, hidden)   dtype=float16
    """
    B, N, _, H = left.shape
    device = left.device

    # reshape to (M, N, K) where M = B*H, K = N
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    M = left_t.shape[0]
    K = N  # both dimensions are N

    # allocate output
    out = torch.empty_like(left_t)

    # strides (in elements, not bytes)
    stride_am, stride_ak = left_t.stride(0), left_t.stride(2)
    stride_bk, stride_bn = right_t.stride(1), right_t.stride(2)  # B is stored transposed
    stride_cm, stride_cn = out.stride(0), out.stride(2)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    tri_mul_kernel[grid](
        left_t,
        right_t,
        out,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back to original layout
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]   (float32)
            - mask:        torch.Tensor [B, N, N]        (float32 or bool)
            - weights:     dict of parameter tensors
            - config:      dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape -> [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy N³ contraction via Triton (compute in fp16)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = tri_mul_torch(left_fp16, right_fp16)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # back to dim
    return out
