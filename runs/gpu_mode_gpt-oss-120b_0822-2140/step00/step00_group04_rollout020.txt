"""
TriMul “outgoing” implementation with a Triton‑accelerated
batched‑matrix‑multiply (the N‑dimensional einsum).

The heavy part is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched GEMM of
    left   : (B, N, K, H)
    right  : (B, K, N, H)          # right is permuted on‑the‑fly
    out    : (B, N, N, H)

The kernel processes a 2‑D tile of (i,j) and a tile of the hidden
dimension H.  All remaining ops (LayerNorm, Linear, gating, masking) are
performed with regular PyTorch for clarity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # const float*  [B, N, K, H]
    right_ptr,         # const float*  [B, K, N, H]   (already permuted)
    out_ptr,           #       float*  [B, N, N, H]
    B, N, H,           # int32 runtime arguments
    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time (constexpr) tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute a tile of the output tensor.
    Grid layout:
        program_id(0) -> i‑tile (rows)
        program_id(1) -> j‑tile (cols)
        program_id(2) -> combined (batch, hidden‑tile) index
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # ----- decode batch & hidden‑tile ----------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_D)
    b = pid_bh // num_h_blocks                # batch index
    h_block = pid_bh % num_h_blocks            # which slice of H we are on
    h_offset = h_block * BLOCK_D + tl.arange(0, BLOCK_D)

    # ----- tile coordinates -------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # ----- bounds -------------------------------------------------------
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_h = h_offset < H

    # accumulator: (M, N, D) in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- loop over K dimension ---------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- load left   [B, N, K, H] --------------------------------
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + i_offset[:, None, None] * stride_l_i \
            + k_offset[None, :, None] * stride_l_k \
            + h_offset[None, None, :] * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0)

        # ----- load right  [B, K, N, H] (already permuted) --------------
        right_ptrs = right_ptr \
            + b * stride_r_b \
            + k_offset[:, None, None] * stride_r_k \
            + j_offset[None, :, None] * stride_r_j \
            + h_offset[None, None, :] * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                        other=0.0)

        # ----- multiply‑accumulate over K, one hidden slice at a time -----
        #   left : (M, K, D)   right : (K, N, D)
        #   For each d we perform a 2‑D dot product.
        for d in range(0, BLOCK_D):
            # guard against out‑of‑range hidden channels
            if not (h_block * BLOCK_D + d) < H:
                continue
            l_slice = left[:, :, d]          # (M, K)
            r_slice = right[:, :, d]          # (K, N)
            acc[:, :, d] += tl.dot(l_slice, r_slice)   # (M, N)

    # ----- store result -------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + i_offset[:, None, None] * stride_o_i \
        + j_offset[None, :, None] * stride_o_j \
        + h_offset[None, None, :] * stride_o_h
    tl.store(out_ptrs,
             acc.to(tl.float32),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑backed forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : Tuple[
        input_tensor : torch.Tensor   # [B, N, N, dim]
        mask         : torch.Tensor   # [B, N, N]  (may be all‑ones)
        weights      : Dict[str, torch.Tensor]
        config       : Dict[str, int]  # contains "dim" and "hidden_dim"
    ]

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core einsum → batched GEMM via Triton
    # ------------------------------------------------------------------
    # Permute right so that the shared k‑dimension is the second axis
    right_swapped = right.permute(0, 2, 1, 3).contiguous()
    left = left.contiguous()

    B, N, _, H = left.shape                      # H == hidden
    assert H == hidden, "weight hidden dim mismatch"

    # Tile sizes – chosen to fit H200 SM architecture
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32                 # hidden‑dim tile

    num_h_blocks = triton.cdiv(H, BLOCK_D)
    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * num_h_blocks,                 # combined batch‑×‑hidden‑tile
    )

    out = torch.empty_like(left)           # [B, N, N, H]

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right_swapped,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        # strides (in elements)
        stride_l_b=left.stride(0),
        stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),
        stride_l_h=left.stride(3),
        stride_r_b=right_swapped.stride(0),
        stride_r_k=right_swapped.stride(1),
        stride_r_j=right_swapped.stride(2),
        stride_r_h=right_swapped.stride(3),
        stride_o_b=out.stride(0),
        stride_o_i=out.stride(1),
        stride_o_j=out.stride(2),
        stride_o_h=out.stride(3),
        # compile‑time constants
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
