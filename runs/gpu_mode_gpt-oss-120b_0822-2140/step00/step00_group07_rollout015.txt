"""
TriMul “outgoing” implementation (AlphaFold‑3).

The heavy part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    out_d = left_d @ right_dᵀ   (d = hidden dimension)

We keep the cheap LayerNorm / Linear / gating logic in PyTorch and
off‑load the batched matmul to a small Triton kernel.  The kernel works
on a flattened batch  (batch * hidden_dim)  and computes
C = A @ Bᵀ   where A,B ∈ ℝ^{M×K} (M = N, K = N).  All tensors are stored
contiguously in FP16; the accumulator uses FP32 and the result is written
back in FP16 before being cast back to FP32 in the host code.

The kernel is launched with a three‑dimensional grid:
    (batch*hidden, ceil(N/BM), ceil(N/BN))

and uses block sizes (BM,BN,BK) = (64,64,64) which are a good fit for
the Hopper (SM‑90) architecture.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B : [B, M, K])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    out_ptr,               # *BATCH, M, N   (output)
    a_ptr,                 # *BATCH, M, K   (left)
    b_ptr,                 # *BATCH, N, K   (right)
    BATCH, M, N, K,
    stride_ao, stride_am, stride_ak,   # strides for A
    stride_bo, stride_bn, stride_bk,   # strides for B (row‑major)
    stride_co, stride_cm, stride_cn,   # strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ   for each batch.
    All pointers are assumed to be contiguous FP16 tensors.
    Accumulation is performed in FP32.
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- per‑batch pointers -------------------------------------------
    a_batch = a_ptr + batch_id * stride_ao
    b_batch = b_ptr + batch_id * stride_bo
    c_batch = out_ptr + batch_id * stride_co

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- k‑loop (blocked) ---------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[i, k]   (M × K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ[k, j]  (K × N) – we load B transposed by swapping the strides
        b_t = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b_t)                          # (M×K)·(K×N) → (M×N)

    # ----- write back (store as FP16) -----------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# Host entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
          input_tensor : [B, N, N, dim]   (float32)
          mask         : [B, N, N]        (bool/int, may be all‑ones)
          weights      : dict of tensors (all float32)
          config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    Tensor of shape [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – keep FP32 for accuracy
    # ------------------------------------------------------------------
    x_f = x.to(torch.float32)
    left = F.linear(x_f, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x_f, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also FP32
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_f, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_f, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_f, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 and reshape for the kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # bring hidden dimension forward → [B, hidden, N, N] (contiguous)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    batch_h = B * hidden
    left_flat = left.view(batch_h, N, N)
    right_flat = right.view(batch_h, N, N)

    # allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides for the three tensors (all are contiguous, row‑major)
    stride_ao, stride_am, stride_ak = left_flat.stride()
    stride_bo, stride_bn, stride_bk = right_flat.stride()
    stride_co, stride_cm, stride_cn = out_flat.stride()

    _batched_matmul_kernel[grid](
        out_flat,
        left_flat,
        right_flat,
        batch_h,
        N,
        N,
        N,
        stride_ao,
        stride_am,
        stride_ak,
        stride_bo,
        stride_bn,
        stride_bk,
        stride_co,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # element‑wise gating

    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
