"""
TriMul “outgoing” implementation with a fused Triton kernel.

The expensive part of the forward pass is the 3‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be rewritten as a batched matrix multiplication with a transpose:

    out[b,:, :,d] = left[b,:, :,d] @ right[b,:, :,d].T

The kernel below computes, for every batch `b` and hidden channel `d`,
a matrix product of shape (N, N) × (N, N)ᴛ using block‑tiling (M, N, K).
All surrounding linear layers, layer‑norms and gating are kept in PyTorch
because they are cheap compared to the O(N³) work of the contraction.
The final output is returned in float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matmul where the second operand is transposed.
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_transpose_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # dimensions
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid = tl.program_id(0)                      # 0 … B*H-1
    total = B * H
    if pid >= total:
        return

    b = pid // H                                 # batch index
    h = pid % H                                  # hidden channel

    # Base pointers for this (b,h) pair
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # ------------------------------------------------------------------
    # Tile over the output matrix (i,j)
    # ------------------------------------------------------------------
    for i_start in range(0, N, BLOCK_M):
        for j_start in range(0, N, BLOCK_N):
            # accumulator in float32
            acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

            # masks for the i/j tile
            i_mask = (i_start + tl.arange(0, BLOCK_M)) < N
            j_mask = (j_start + tl.arange(0, BLOCK_N)) < N

            for k_start in range(0, N, BLOCK_K):
                # ----------------------------------------------------------
                # Load a tile of left  : shape (BLOCK_M, BLOCK_K)
                # ----------------------------------------------------------
                k_mask = (k_start + tl.arange(0, BLOCK_K)) < N

                left_i = i_start + tl.arange(0, BLOCK_M)    # rows
                left_k = k_start + tl.arange(0, BLOCK_K)    # cols

                # pointer arithmetic (left[b, i, k, h])
                left_ptrs = left_base \
                    + left_i[:, None] * stride_l_i \
                    + left_k[None, :] * stride_l_j

                left_mask = i_mask[:, None] & k_mask[None, :]
                left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

                # ----------------------------------------------------------
                # Load a tile of right (transposed) : shape (BLOCK_K, BLOCK_N)
                # We need right[b, j, k, h]   -->   (k, j) after transposition
                # ----------------------------------------------------------
                right_j = j_start + tl.arange(0, BLOCK_N)    # columns of output
                right_k = k_start + tl.arange(0, BLOCK_K)    # rows of K dimension

                # pointer arithmetic (right[b, j, k, h])
                right_ptrs = right_base \
                    + right_j[None, :] * stride_r_i \
                    + right_k[:, None] * stride_r_j

                right_mask = k_mask[:, None] & j_mask[None, :]
                right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

                # ----------------------------------------------------------
                # Accumulate
                # ----------------------------------------------------------
                acc += tl.dot(left_tile, right_tile)   # (M,K)·(K,N) → (M,N)

            # --------------------------------------------------------------
            # Write the tile back
            # --------------------------------------------------------------
            out_i = i_start + tl.arange(0, BLOCK_M)
            out_j = j_start + tl.arange(0, BLOCK_N)

            out_ptrs = out_base \
                + out_i[:, None] * stride_o_i \
                + out_j[None, :] * stride_o_j

            out_mask = i_mask[:, None] & j_mask[None, :]
            tl.store(out_ptrs, acc.to(tl.float32), mask=out_mask)


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def batch_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (float32)
    returns      : [B, N, N, H]   (float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # block sizes – empirically good on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (in elements, not bytes)
    s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    total = B * H
    grid = (total,)

    batch_matmul_transpose_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – Tensor [B, N, N, dim] (float32, on CUDA)
        mask   – Tensor [B, N, N]   (float32 or bool, on CUDA)
        weights – dict of pre‑loaded model parameters
        config – dict with keys "dim", "hidden_dim"
    Returns
    -------
    Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Layer‑norm on the input (no bias for simplicity – we add bias below)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + eps
    )
    x = x * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden, dim)

    # shape [B,N,N,hidden]
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy contraction using the Triton kernel
    # ------------------------------------------------------------------
    out = batch_matmul_transpose(left, right)   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + eps
    )
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # Apply output gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate                      # gating
    to_out_w = weights["to_out.weight"]        # (dim, hidden)
    out = F.linear(out, to_out_w)              # (B,N,N,dim)

    return out.to(torch.float32)
