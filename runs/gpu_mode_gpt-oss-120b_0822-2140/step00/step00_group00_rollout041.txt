"""
TriMul “outgoing” implementation.
The heavy N³ part (the einsum over the third dimension) is executed with a
custom Triton kernel that performs a batched matrix multiplication per
hidden channel:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is equivalent to (left[..., :, d] @ right[..., :, d].T).
All other operations (LayerNorm, Linear projections, gating, final
normalisation/linear) are done with PyTorch for simplicity.
The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden_dim = 128 (the only value used in the tests).  It runs in fp32
(and could be switched to fp16 for extra speed).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matmul per hidden channel (no bias, no activation)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                          # problem sizes
    stride_lb, stride_li, stride_lk, stride_ld,      # left strides
    stride_rb, stride_ri, stride_rk, stride_rd,      # right strides
    stride_ob, stride_oi, stride_oj, stride_od,      # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_bc = tl.program_id(2)         # flattened (batch, hidden)

    # de‑compose batch & hidden channel
    b = pid_bc // H
    d = pid_bc % H

    # offsets inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current (b,d) slice
    left_base  = left_ptr  + b * stride_lb  + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset
        mask_k = cur_k < N

        # left: shape (M, K)
        a_ptr = left_base + offs_m[:, None] * stride_li + cur_k[None, :] * stride_lk
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right (need transpose: we want (K, N) where right[j,k] = right_T[k,j])
        b_ptr = right_base + cur_k[:, None] * stride_rk + offs_n[None, :] * stride_ri
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate block matrix product
        acc += tl.dot(a, b)          # (M,K) @ (K,N) -> (M,N)

    # write result
    out_ptrs = out_ptr + b * stride_ob + d * stride_od \
               + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def _trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, H)   fp32
    returns out  : (B, N, N, H)   fp32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (in elements, not bytes) – note torch uses row‑major order
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # block sizes – tuned for Hopper SM90 (shared‑mem ~ 64 KB)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        B * H,                             # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_ri, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # enough for fp32
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input, mask, weights, config)
        - input  : Tensor[B, N, N, dim]      (float32, on CUDA)
        - mask   : Tensor[B, N, N]          (float32 or bool, on CUDA)
        - weights: dict of pre‑loaded parameters
        - config : dict containing ``dim`` and ``hidden_dim``
    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]         # (hidden_dim, dim)

    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣  Apply optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣  Gating: sigmoid(linear(x)) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Core TriMul (N³) – Triton kernel
    # ------------------------------------------------------------------
    # right needs to be transposed on the last two axes for the kernel
    # but the kernel already loads it as (k, j) so we can pass it unchanged.
    core = _trimul_matmul(left, right)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣  Output normalisation + output gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    core = F.layer_norm(core,
                        normalized_shape=(hidden_dim,),
                        weight=to_out_norm_w,
                        bias=to_out_norm_b,
                        eps=eps)

    core = core * out_gate.unsqueeze(-1)   # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(core, to_out_w)        # (B,N,N,dim)

    return out.to(torch.float32)
