"""
TriMul “outgoing” implementation (forward only).

The expensive part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

By transposing the second N‐axis of `right` we obtain a standard batched
matrix‑multiply:
    out = left @ right.transpose(1,2)     # (B,N,N,hidden) @ (B,N,N,hidden) → (B,N,N,hidden)

The kernel below performs this batched mat‑mul in Triton, tiling the
(N × N) output with BLOCK_M×BLOCK_N tiles and accumulating over K in
BLOCK_K steps.  All other linear / LayerNorm / sigmoid work is done in
PyTorch (highly‑optimized BLAS / vectorised kernels).  The final output
is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ------------------------------------------------------------
# Triton kernel: batched matmul (A @ B) where
#   A : (batch, M, K)
#   B : (batch, K, N)
#   C : (batch, M, N)
# ------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,               # *ptr to A
    B_ptr,               # *ptr to B
    C_ptr,               # *ptr to C
    M, N, K,             # matrix sizes
    stride_Ab, stride_Ak, stride_Ad,   # strides of A (batch, row, col)
    stride_Bk, stride_Bn, stride_Bd,   # strides of B (batch, row(K), col(N))
    stride_Cm, stride_Cn, stride_Cd,   # strides of C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # off‑sets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the first block of K
    a_ptrs = A_ptr + pid_batch * stride_Ab \
                     + offs_m[:, None] * stride_Ak \
                     + offs_k[None, :] * stride_Ad

    b_ptrs = B_ptr + pid_batch * stride_Bk \
                     + offs_k[:, None] * stride_Bn \
                     + offs_n[None, :] * stride_Bd

    # accumulator in fp32 (more stable for long reductions)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)          # (M×K)·(K×N) → (M×N)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_Ad
        b_ptrs += BLOCK_K * stride_Bk

    # write result
    c_ptrs = C_ptr + pid_batch * stride_Cm \
                     + offs_m[:, None] * stride_Cn \
                     + offs_n[None, :] * stride_Cd
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ------------------------------------------------------------
# Helper: fused LayerNorm (weight & bias supplied)
# ------------------------------------------------------------
def fused_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : (..., C)
    return F.layer_norm(x, x.shape[-1:], weight, bias, eps)


# ------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (bool/float, will be broadcast)
        - weights      : dict of weight tensors (no bias except where noted)
        - config       : dict containing
            * dim        : input channel dimension
            * hidden_dim : hidden dimension inside TriMul
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over dim
    # ------------------------------------------------------------------
    x = fused_layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                     # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    #   Linear: y = x @ W.T   where W has shape (hidden, dim)
    #   torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])       # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Apply optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)    # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – also bias‑free linear + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # used after to_out_norm

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core einsum -> batched matmul via Triton
    # ------------------------------------------------------------------
    # transpose second N‑axis of right to obtain (B,N,N,hidden) @ (B,N,N,hidden)ᵀ
    right_t = right.permute(0, 2, 1, 3)      # [B, N, N, hidden]  (j,k) → (k,j)

    # fuse hidden dimension into the batch axis for the kernel
    left_view = left.reshape(B * hidden, N, N).contiguous()
    right_view = right_t.reshape(B * hidden, N, N).contiguous()
    out_view = torch.empty_like(left_view)   # will hold [B*hidden, N, N]

    # strides after the view (batch, row, col)
    stride_Ab, stride_Ak, stride_Ad = left_view.stride()      # (N*N, N, 1)
    stride_Bk, stride_Bn, stride_Bd = right_view.stride()     # (N*N, N, 1) but note K‑row stride = N
    stride_Cm, stride_Cn, stride_Cd = out_view.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_m = triton.cdiv(N, BLOCK_M)
    num_n = triton.cdiv(N, BLOCK_N)
    grid = (B * hidden, num_m, num_n)

    batched_matmul_kernel[grid](
        left_view,
        right_view,
        out_view,
        N, N, N,                                   # M, N, K
        stride_Ab, stride_Ak, stride_Ad,
        stride_Bk, stride_Bn, stride_Bd,
        stride_Cm, stride_Cn, stride_Cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    out = out_view.reshape(B, N, N, hidden)   # back to [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm + out_gate + final linear
    # ------------------------------------------------------------------
    out = fused_layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out.to(torch.float32)
