"""
TriMul “outgoing” implementation.

The heavy part of the operation is the 3‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batch of hidden_dim independent matrix
multiplications (left[...,d] @ right[...,d].T).  The surrounding
LayerNorms, linear projections and gates are cheap and are kept in
PyTorch.  The kernel below performs a single‑channel (d‑slice) matrix
multiply on the GPU using Triton.  We launch the kernel once per
hidden dimension; for the maximal hidden size (384) this is still only a
few hundred lightweight launches and yields a large speed‑up over the
pure‑PyTorch reference.

All tensors are expected to be on the CUDA device already and the final
output is returned as `torch.float32`.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for a single hidden channel
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_one_channel_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_bs, stride_left_i, stride_left_k,
    stride_right_bs, stride_right_j, stride_right_k,
    stride_out_bs, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ right.T for ONE hidden channel."""
    pid = tl.program_id(0)

    # number of blocks in the (i,j) plane per batch
    total_m = tl.cdiv(N, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = total_m * total_n

    # batch index
    b = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // total_n
    pid_n = pid_in_batch % total_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K‑tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current tile
        left_tile_ptr = (
            left_ptr
            + b * stride_left_bs
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        right_tile_ptr = (
            right_ptr
            + b * stride_right_bs
            + offs_n[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )

        # load tiles (mask out-of‑bounds elements)
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_tile_ptr,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0)

        # accumulate dot product over K
        acc += tl.dot(left, right)

    # store the result
    out_tile_ptr = (
        out_ptr
        + b * stride_out_bs
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel
    for the N³ einsum part.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if present). The mask is broadcast on the hidden dim.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                       # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the einsum (one channel at a time)
    # ------------------------------------------------------------------
    # Output tensor placeholder
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Compute strides (in elements, not bytes)
    # Tensor layout is contiguous: (B, N, N, H)
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_j, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Triton block sizes – tuned for Hopper (sm_90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid: one program per (batch, i‑tile, j‑tile)
    total_blocks_m = triton.cdiv(N, BLOCK_M)
    total_blocks_n = triton.cdiv(N, BLOCK_N)
    grid = (B * total_blocks_m * total_blocks_n,)

    # Launch kernel for each hidden channel
    for h in range(hidden_dim):
        left_ptr  = left.data_ptr()  + h * l_stride_h * left.element_size()
        right_ptr = right.data_ptr() + h * r_stride_h * right.element_size()
        out_ptr   = out.data_ptr()   + h * o_stride_h * out.element_size()

        _trimul_matmul_one_channel_kernel[grid](
            left_ptr, right_ptr, out_ptr,
            B, N,
            l_stride_b, l_stride_i, l_stride_k,
            r_stride_b, r_stride_j, r_stride_k,
            o_stride_b, o_stride_i, o_stride_j,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=4,               # good default on Hopper
        )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out  # torch.float32, same device as input
