def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    device = x.device

    # LayerNorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])   # shape bs,N,N,hidden
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    # mask
    if mask is not None:
        mask_e = mask.unsqueeze(-1)  # shape bs,N,N,1
        left = left * mask_e
        right = right * mask_e
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # Prepare for matmul
    # reshape to (B*hidden, N, N)
    left_ = left.permute(0,3,1,2).contiguous().view(bs*hidden, N, N)  # (b*hdim, N, N)
    right_ = right.permute(0,3,1,2).contiguous().view(bs*hidden, N, N) # (b*hdim, N, N)

    # Cast to fp16 for speed
    left_ = left_.to(torch.float16)
    right_ = right_.to(torch.float16)
    out_ = torch.empty_like(left_, dtype=torch.float16)

    # strides
    stride_ad, stride_am, stride_ak = left_.stride()
    stride_bd, stride_bk, stride_bn = right_.stride()
    stride_cd, stride_cm, stride_cn = out_.stride()

    BLOCK = 64  # choose 64
    grid = (bs*hidden * triton.cdiv(N, BLOCK) * triton.cdiv(N, BLOCK),)
    batched_matmul_kernel[grid](left_, right_, out_,
                                 N, N, N,

                          M, N, K,
                          stride_ad, stride_am, stride_ak,
                          stride_bd, stride_bk, stride_bn,
                          stride_cd, stride_cm, stride_cn,

    block_idx = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_idx // tl.cdiv(N, BLOCK_N)
    pid_n = block_idx % tl.cdiv(N, BLOCK_N)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

"""
TriMul “outgoing” implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (dim = C).
2. Linear projections (left / right) and three gating projections (left, right,
   out) – all done with torch.nn.functional.linear for speed.
3. Apply the optional mask, then multiply the projections by their gates.
4. The expensive O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication:
        reshape left/right → (B·D, N, N)
        out = left @ rightᵀ   (per channel d)
   → realized by a custom Triton kernel `batched_matmul_kernel`.
5. Reshape back, apply a second layer‑norm, the output gate, and the final
   linear projection to obtain a tensor of shape (B, N, N, C).

The kernel works on FP16 data for compute‑efficiency and casts the final
result back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched matmul (A @ Bᵀ) for many independent matrices
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (all equal to seq_len)
    stride_ad, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bd, stride_bk, stride_bn,   # strides for B (batch, row, col)
    stride_cd, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul where each batch entry computes
       C = A @ Bᵀ  (A,B : [batch, M, K],  C : [batch, M, N])
    """
    pid = tl.program_id(0)

    # ------------------- block indices -------------------
    batch_dim = tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    batch_idx = pid // batch_dim
    block_id = pid % batch_dim
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    # ------------------- offsets inside the block -------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- K‑loop -------------------------
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : (M, K)   B : (N, K)  (note Bᵀ ⇒ we load as (K, N))
        a_ptr = A_ptr + batch_idx * stride_ad \
                + offs_m[:, None] * stride_am \
                + offs_k[None, :] * stride_ak
        b_ptr = B_ptr + batch_idx * stride_bd \
                + offs_k[:, None] * stride_bk \
                + offs_n[None, :] * stride_bn

        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # (BM, BK)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (BK, BN)

        acc += tl.dot(a, b)                         # (BM, BN)

    # ------------------- write result -------------------
    c_ptr = C_ptr + batch_idx * stride_cd \
            + offs_m[:, None] * stride_cm \
            + offs_n[None, :] * stride_cn
    tl.store(c_ptr,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]  (float32)
            - mask         : torch.Tensor [B, N, N]    (bool/float)
            - weights      : dict of leaf weights (torch.Tensor)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, C = x.shape
    H = cfg["hidden_dim"]                     # hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (dim = C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask is always supplied, may be all‑ones)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton
    #    reshape to (B*H, N, N) and compute A @ Bᵀ per channel
    # ------------------------------------------------------------------
    # permute to bring hidden dim to the batch axis
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # FP16 for the kernel (H200 has excellent FP16 tensor cores)
    left_f16  = left_.to(torch.float16)
    right_f16 = right_.to(torch.float16)
    out_f16   = torch.empty_like(left_f16, dtype=torch.float16)

    # strides for the flat tensors
    stride_ad, stride_am, stride_ak = left_f16.stride()   # (batch, row, col)
    stride_bd, stride_bk, stride_bn = right_f16.stride()
    stride_cd, stride_cm, stride_cn = out_f16.stride()

    # block size – 64 works well on Hopper for these dimensions
    BLOCK = 64
    num_blocks_m = triton.cdiv(N, BLOCK)
    num_blocks_n = triton.cdiv(N, BLOCK)
    total_blocks = (B * H) * num_blocks_m * num_blocks_n

    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        N, N, N,                                 # M, N, K
        stride_ad, stride_am, stride_ak,
        stride_bd, stride_bk, stride_bn,
        stride_cd, stride_cm, stride_cn,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
    )

    # reshape back to [B, N, N, H]
    out = out_f16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Second layer‑norm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
