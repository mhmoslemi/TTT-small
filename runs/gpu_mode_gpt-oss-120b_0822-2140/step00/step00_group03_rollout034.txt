"""
TriMul “outgoing” implementation (forward only).

The heavy part is the 4‑D einsum

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is a batched matrix multiplication for every hidden channel d.
We reshape the tensors so that each (batch, d) pair becomes an independent
matrix‑multiply A @ Bᵀ with shapes (N, N) × (N, N).  The multiplication is
performed by a custom Triton kernel; the surrounding LayerNorms,
linear layers and mask handling stay in PyTorch.

The kernel works in FP16 for bandwidth, accumulates in FP32 and writes
the result back in FP16 (later cast to FP32).  It is launched with
tunable block sizes (128×128 tiles, 32‑way K reduction) which give good
performance on Hopper (SM‑90) GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (all matrices are square N×N)
# ----------------------------------------------------------------------
@triton.jit
def _matmul_tiled_kernel(
    a_ptr,          # *Pointer* to A   [M, K]
    b_ptr,          # *Pointer* to Bᵀ [K, N]  (original B is transposed before launch)
    c_ptr,          # *Pointer* to C   [M, N]
    M, N, K,
    stride_am, stride_ak,   # strides for A
    stride_bk, stride_bn,   # strides for Bᵀ
    stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # 2‑D grid of blocks
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n
    pid_n = pid % num_n

    # Block coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds (needed for the last partial block)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current block
    a_block_ptr = a_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_block_ptr = b_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # Load A and B tiles (with masking)
        a = tl.load(a_block_ptr,
                    mask=mask_m[:, None] & (cur_k + tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0)
        b = tl.load(b_block_ptr,
                    mask=(cur_k + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # Accumulate the dot product of the two tiles
        acc += tl.dot(a, b)

        # Move pointers to the next K‑tile
        a_block_ptr += BLOCK_K * stride_ak
        b_block_ptr += BLOCK_K * stride_bk

    # Write back (store in FP16, convert later)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton GEMM for a batch of (B*H) matrices
# ----------------------------------------------------------------------
def _batched_matmul_tiled(a: torch.Tensor, b_T: torch.Tensor) -> torch.Tensor:
    """
    a, b_T : [B*H, N, N]  (FP16, contiguous)
    returns c : [B*H, N, N] (FP16)
    """
    assert a.is_contiguous() and b_T.is_contiguous()
    BxH, N, _ = a.shape
    M = N
    K = N

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    _matmul_tiled_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b_T.data_ptr(),
        c_ptr=a.data_ptr(),               # in‑place write is fine; we will allocate a new buffer
        M=M,
        N=N,
        K=K,
        stride_am=a.stride(0),
        stride_ak=a.stride(1),
        stride_bk=b_T.stride(0),
        stride_bn=b_T.stride(1),
        stride_cm=a.stride(0),            # reuse same stride for output
        stride_cn=a.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return a  # now contains the result


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a Triton kernel for the
    N³ einsum.  All LayerNorms and Linear projections are performed with
    PyTorch (FP32).  The einsum is executed in FP16 with FP32 accumulation.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – keep everything in FP32
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if mask is provided; mask can be all‑ones)
    # --------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(dtype)          # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton‑accelerated einsum  (the N³ part)
    # --------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and merge batch+channel
    left = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]

    # transpose the second operand on the last two axes (j,k) -> (k,j)
    right_T = right.permute(0, 1, 3, 2).contiguous()     # [B, H, N, N] (k, j)

    BxH, _, _, _ = left.shape
    left_flat = left.view(BxH, N, N).to(torch.float16)
    right_T_flat = right_T.view(BxH, N, N).to(torch.float16)

    # allocate output buffer (reuse left_flat for in‑place result)
    out_flat = torch.empty_like(left_flat)

    # launch batched GEMM (C = A @ Bᵀ)
    # we cannot write directly into out_flat inside the kernel because the
    # kernel currently writes to the pointer we give it; therefore we pass a
    # temporary buffer and copy after the call.
    out_flat = _batched_matmul_tiled(left_flat.clone(), right_T_flat)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out = out.to(dtype)   # FP32 for the rest of the pipeline

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
