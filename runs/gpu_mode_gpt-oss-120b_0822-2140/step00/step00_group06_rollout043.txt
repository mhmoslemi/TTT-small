"""
TriMul “outgoing” implementation (AlphaFold‑3) with a Triton kernel for the
O(N³·c) pairwise update.

Algorithm
---------
1) Layer‑norm the 4‑D input.
2) Linear projections (left / right) and three sigmoidal gates.
3) Apply the mask and gates → tensors L,R ∈ [B,N,N,H].
4) For every batch b, i, j, h compute
        out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]                (einsum)
   This O(N³·H) contraction is tiled and executed in a custom Triton kernel:
   * Blocks of (i,j) are processed by a 2‑D grid (BLOCK_I × BLOCK_J).
   * The reduction over k is performed in K‑tiles (BLOCK_K) and the
     hidden dimension H is tiled with BLOCK_D (H is a multiple of 16 in all
     test configs, so no remainder handling is required beyond a simple mask).
   * Inside the kernel we load
        L_block : (BLOCK_I, BLOCK_K, BLOCK_D)   and
        R_block : (BLOCK_J, BLOCK_K, BLOCK_D)
     broadcast them to (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_D),
     multiply element‑wise and sum over the K axis → accumulator
        acc : (BLOCK_I, BLOCK_J, BLOCK_D).
5) The resulting tensor is layer‑normed, multiplied by the out‑gate and a
   final linear projection back to the original dimension.
All computations are performed in float32 (the kernel accumulates in fp32).
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: tiled contraction over the shared “k” dimension.
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    # pointers
    left_ptr,   #   const float*  [N, N, H]   (i, k, h)
    right_ptr,  #   const float*  [N, N, H]   (j, k, h)
    out_ptr,    #   float*        [N, N, H]   (i, j, h)

    # sizes
    N, H,

    # strides (in elements, not bytes)
    stride_li, stride_lk, stride_lh,   # left  i,k,h strides
    stride_rj, stride_rk, stride_rh,   # right j,k,h strides
    stride_oi, stride_oj, stride_oh,   # out   i,j,h strides

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """ Compute out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h] """

    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)

    # ------------------------------------------------------------------
    # Offsets for the current i / j block
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_base < N
    j_mask = j_base < N

    # hidden dim offsets (same for all i/j)
    d_base = tl.arange(0, BLOCK_D)
    d_mask = d_base < H

    # accumulator (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in tiles
    # ------------------------------------------------------------------
    for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # ------------------------------------------------------------------
        # Load left[i, k, h] → (BLOCK_I, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        left_offset = (
            i_base[:, None, None] * stride_li   # i dimension
            + k_base[None, :, None] * stride_lk  # k dimension
            + d_base[None, None, :] * stride_lh   # h dimension
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )  # fp32 (weights are fp32)

        # ------------------------------------------------------------------
        # Load right[j, k, h] → (BLOCK_J, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        right_offset = (
            j_base[:, None, None] * stride_rj
            + k_base[None, :, None] * stride_rk
            + d_base[None, None, :] * stride_rh
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Broadcast to (i, j, k, h) and accumulate
        # ------------------------------------------------------------------
        # left  : (BLOCK_I, 1, BLOCK_K, BLOCK_D)
        # right : (1, BLOCK_J, BLOCK_K, BLOCK_D)
        prod = left[:, None, :, :] * right[None, :, :, :]      # (I,J,K,D)
        acc += tl.sum(prod, axis=2)                           # sum over K → (I,J,D)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_offset = (
        i_base[:, None, None] * stride_oi
        + j_base[None, :, None] * stride_oj
        + d_base[None, None, :] * stride_oh
    )
    tl.store(
        out_ptr + out_offset,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper to launch the kernel for a single batch element
# ----------------------------------------------------------------------
def _launch_trimul_batch(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor):
    """
    left, right, out : tensors with shape [N, N, H] (float32)
    """
    N, N2, H = left.shape
    assert N == N2, "square pairwise tensor expected"

    # strides in elements (not bytes)
    stride_li, stride_lk, stride_lh = left.stride()
    stride_rj, stride_rk, stride_rh = right.stride()
    stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – chosen to give good occupancy on Hopper (SM90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 16          # H is a multiple of 16 in all configs

    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))
    trimul_out_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        H=H,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]       (float32 or bool)
        - weights      : dict of torch.Tensor containing:
            * norm.weight / norm.bias
            * left_proj.weight, right_proj.weight
            * left_gate.weight, right_gate.weight, out_gate.weight
            * to_out_norm.weight / to_out_norm.bias
            * to_out.weight
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1).type_as(left)      # (B,N,N,1)
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and gating of the projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy O(N³·H) contraction using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    for b in range(B):
        _launch_trimul_batch(left[b], right[b], out[b])

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
