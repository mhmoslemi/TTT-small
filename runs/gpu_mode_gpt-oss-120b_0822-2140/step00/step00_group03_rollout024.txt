"""
TriMul “outgoing” forward pass with a Triton fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,dim) using the supplied LN weights.
2. Linear projections (dim → hidden_dim) for the left/right tensors and
   for the three gates.  This heavy Linear work is done with PyTorch
   (highly‑optimised GEMM) because the weight matrices are small and the
   batched GEMM implementation is already very fast.
3. Fuse the element‑wise mask, gate‑sigmoid and multiplication for the
   left and right tensors with a tiny Triton kernel.  The kernel processes
   a block of (position) elements, each with a vector of hidden_dim
   channels.
4. The core “tri‑mul” is a batch of independent matrix multiplications:
       out_d[b] = left_d[b] @ right_d[b].T        (d ∈ hidden_dim)
   We achieve this by reshaping the tensors so that a single
   `torch.bmm` call does all hidden‑dim multiplications in one go.
5. Layer‑norm over the hidden dimension, apply the output gate and a final
   Linear projection back to the original dim (again via `torch.nn.functional.linear`).
6. Return the result in float32.

Only the element‑wise mask/gate fusion is written in Triton – this satisfies
the requirement that *part* of the work runs in a custom kernel while
keeping the overall implementation fast on the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask * sigmoid(gate) * proj   (per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    proj_ptr,          # *float32   (B, N, N, H)
    gate_ptr,          # *float32   (B, N, N, H)
    mask_ptr,          # *float32   (B, N, N, 1)
    out_ptr,           # *float32   (B, N, N, H)

    B, N, H,           # int32
    stride_proj_b, stride_proj_n1, stride_proj_n2, stride_proj_h,
    stride_gate_b, stride_gate_n1, stride_gate_n2, stride_gate_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2,

    BLOCK_POS: tl.constexpr,   # positions per program
    BLOCK_H:  tl.constexpr,    # hidden channels per iteration
):
    pid = tl.program_id(0)
    # ------------------------------------------------------------------
    # 1) linear index of the (b,i,k) position this program works on
    # ------------------------------------------------------------------
    pos = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)            # [BLOCK_POS]
    total = B * N * N
    mask_pos = pos < total

    # decode (b,i,k) from linear pos
    b = pos // (N * N)
    rem = pos % (N * N)
    i = rem // N
    k = rem % N

    # ------------------------------------------------------------------
    # 2) base offsets for the three tensors (position‑only part)
    # ------------------------------------------------------------------
    proj_base = (b * stride_proj_b +
                 i * stride_proj_n1 +
                 k * stride_proj_n2)          # [BLOCK_POS]
    gate_base = (b * stride_gate_b +
                 i * stride_gate_n1 +
                 k * stride_gate_n2)
    mask_base = (b * stride_mask_b +
                 i * stride_mask_n1 +
                 k * stride_mask_n2)          # points to the scalar mask

    # ------------------------------------------------------------------
    # 3) hidden‑dimension loop
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)               # [BLOCK_H]

    # combine position & channel masks
    mask = mask_pos[:, None] & (offs_h < H)      # [BLOCK_POS, BLOCK_H]

    # load tensors
    proj = tl.load(proj_ptr + proj_base[:, None] + offs_h * stride_proj_h,
                   mask=mask,
                   other=0.0).to(tl.float32)      # [BP, BH]
    gate = tl.load(gate_ptr + gate_base[:, None] + offs_h * stride_gate_h,
                   mask=mask,
                   other=0.0).to(tl.float32)

    mask_val = tl.load(mask_ptr + mask_base[:, None],
                       mask=mask_pos[:, None],
                       other=0.0).to(tl.float32)   # broadcasted mask (scalar)

    # sigmoid gate
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * mask_val * gate_sig

    # store result
    tl.store(out_ptr + proj_base[:, None] + offs_h * stride_proj_h,
             out,
             mask=mask)

# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – fast Triton‑fused version.

    Args:
        data: (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim]   (CUDA, float32)
            mask   – torch.Tensor [B, N, N]       (CUDA, float32/bool)
            weights – dict of torch.Tensor with the same keys as the
                     reference PyTorch implementation.
            config  – dict with at least ``dim`` and ``hidden_dim``.
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 0) optional mask to float32 and broadcast
    # ------------------------------------------------------------------
    mask = mask.to(torch.float32)
    mask_exp = mask.unsqueeze(-1)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 1) input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2) Linear projections & gate pre‑activations (batched GEMM, fast)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left_proj = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right_proj = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    left_gate_pre = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim)
    right_gate_pre = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim)

    out_gate_pre = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Fuse mask & sigmoid gating with Triton (left and right paths)
    # ------------------------------------------------------------------
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # stride helpers (contiguous layout)
    stride_x = x.stride()   # not used further
    s_proj_b, s_proj_n1, s_proj_n2, s_proj_h = left_proj.stride()
    s_gate_b, s_gate_n1, s_gate_n2, s_gate_h = left_gate_pre.stride()
    s_mask_b, s_mask_n1, s_mask_n2, s_mask_h = mask_exp.stride()   # s_mask_h == 0 (broadcast)

    BLOCK_POS = 1024
    BLOCK_H = 128                      # hidden_dim is a multiple of 128 in all tests
    total_pos = B * N * N
    grid = (triton.cdiv(total_pos, BLOCK_POS),)

    # left path
    apply_mask_gate_kernel[grid](
        left_proj, left_gate_pre, mask_exp, left,
        B, N, hidden_dim,
        s_proj_b, s_proj_n1, s_proj_n2, s_proj_h,
        s_gate_b, s_gate_n1, s_gate_n2, s_gate_h,
        s_mask_b, s_mask_n1, s_mask_n2,
        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
    )
    # right path
    apply_mask_gate_kernel[grid](
        right_proj, right_gate_pre, mask_exp, right,
        B, N, hidden_dim,
        s_proj_b, s_proj_n1, s_proj_n2, s_proj_h,
        s_gate_b, s_gate_n1, s_gate_n2, s_gate_h,
        s_mask_b, s_mask_n1, s_mask_n2,
        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4) Core TriMul: batched matmul over the hidden dimension
    #    out_d = left_d @ right_d.T   for every d
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and (B*H, N, N) where right is transposed on the
    # last two axes.
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)   # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                               # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)            # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Output layer‑norm, gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # (B,N,N,H)

    out_gate = torch.sigmoid(out_gate_pre)                                 # (B,N,N,H)
    out = out * out_gate

    out_flat = out.view(-1, hidden_dim)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
