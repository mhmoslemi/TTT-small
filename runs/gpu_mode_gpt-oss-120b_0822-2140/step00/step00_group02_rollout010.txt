"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The expensive part of the module is the ternary contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of H (= hidden_dim) independent matrix‑multiplications
    out_d = L_d @ R_dᵀ            (L_d,R_d : N×N)

The kernel below computes many (B*H) such GEMMs in a single Triton launch.
All cheap ops (LayerNorm, Linear, gating, masking) are performed with
torch.nn.functional for clarity.  The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A : M×K , B : N×K)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,          # A: (batch, M, K)   – float16
    b_ptr,          # Bᵀ: (batch, N, K)   – float16 (B already transposed)
    c_ptr,          # C: (batch, M, N)   – float32 accumulator
    M, N, K,
    stride_am, stride_ak, stride_az,   # strides of A
    stride_bn, stride_bk, stride_bz,   # strides of Bᵀ
    stride_cm, stride_cn, stride_cz,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute batched matrix products C = A @ Bᵀ."""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_z = tl.program_id(2)          # batch index (B*H)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current block (batch offset added)
    a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bz + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bk

    # accumulator in FP32 (more stable than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension tiles
    for off_k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, applying masks for tail handling
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (off_k * BLOCK_K + tl.arange(0, BLOCK_K) < K)),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(mask_n[None, :] & (off_k * BLOCK_K + tl.arange(0, BLOCK_K) < K)),
                    other=0.0)
        # dot: (M,K)·(K,N) → (M,N)
        acc += tl.dot(a, b)
        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write result – store only valid elements
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(a_fp16: torch.Tensor, b_fp16_T: torch.Tensor) -> torch.Tensor:
    """
    a_fp16 : (B*H, M, K)   torch.float16, contiguous
    b_fp16_T : (B*H, N, K) torch.float16, contiguous   (already transposed)
    returns : (B*H, M, N) torch.float32
    """
    batch, M, K = a_fp16.shape
    _, N, _ = b_fp16_T.shape

    out = torch.empty((batch, M, N), dtype=torch.float32, device=a_fp16.device)

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_matmul_kernel[grid](
        a_fp16,
        b_fp16_T,
        out,
        M,
        N,
        K,
        a_fp16.stride(0),
        a_fp16.stride(2),
        a_fp16.stride(1),   # batch stride
        b_fp16_T.stride(0),
        b_fp16_T.stride(2),
        b_fp16_T.stride(1), # batch stride
        out.stride(0),
        out.stride(2),
        out.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        mask_tensor  : torch.Tensor  [B, N, N]      (float32 or bool)
        weights_dict : mapping name → torch.Tensor (already on device)
        config_dict  : { "dim": int, "hidden_dim": int }

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]

    eps = 1e-5  # LayerNorm epsilon – matches nn.LayerNorm default

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])    # weight: (hidden_dim, dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any).  Mask may be all‑ones when nomask==True.
    # ------------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – use Triton GEMM over the (H = hidden_dim) batch
    # ------------------------------------------------------------------
    # bring the hidden dimension to the batch axis: (B, N, N, H) → (B*H, N, N)
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * hidden_dim, N, N)    # (B*H, N, N)
    # we need rightᵀ in the inner‑K dimension, i.e. (B*H, N, N) → (B*H, N, N)ᵀ
    right_T = right_perm.transpose(-2, -1).contiguous().view(B * hidden_dim, N, N)

    # cast to fp16 for the heavy compute
    left_fp16 = left_flat.to(torch.float16)
    right_T_fp16 = right_T.to(torch.float16)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ  → (B*H, N, N)
    out_fp32 = _batched_matmul(left_fp16, right_T_fp16)   # FP32 result

    # reshape back to original layout
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (dim_out = dim)

    return out.to(torch.float32)
