"""
TriMul “outgoing” implementation (AlphaFold3).

The heavy part is the triple‑wise multiplicative update:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply of shape
    (B*H, N, N)  @  (B*H, N, N).T  → (B*H, N, N)

The kernel below computes this batched GEMM in Triton (float16
accumulation, final result cast to float32).  All other operations
(LayerNorm, linear projections, gating, final LayerNorm + linear)
are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul (left @ right^T) where each batch is a
# separate matrix of size N×N and there are M = B*H such matrices.
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    L_ptr,            # const float*  (M, N, N) left matrix
    R_ptr,            # const float*  (M, N, N) right matrix (already transposed)
    O_ptr,            #       float*  (M, N, N) output
    N,                # matrix size (int)
    stride_l_m, stride_l_n, stride_l_k,   # strides for left
    stride_r_m, stride_r_n, stride_r_k,   # strides for right (transposed)
    stride_o_m, stride_o_n, stride_o_k,   # strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                     # one program per (M) matrix
    # base pointer for this matrix
    L_base = L_ptr + pid * stride_l_m
    R_base = R_ptr + pid * stride_r_m
    O_base = O_ptr + pid * stride_o_m

    # accumulate in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ---- load a tile from left (M x K) ----
        offs_m = tl.arange(0, BLOCK_M)
        offs_k = tl.arange(0, BLOCK_K)
        L_ptrs = L_base + (offs_m[:, None] * stride_l_n) + (k_offset + offs_k)[None, :] * stride_l_k
        L_mask = (offs_m[:, None] < N) & (k_offset + offs_k)[None, :] < N
        L_tile = tl.load(L_ptrs, mask=L_mask, other=0.0).to(tl.float32)

        # ---- load a tile from right (K x N) ----
        # right is already transposed, so its rows correspond to K
        R_ptrs = R_base + (k_offset + offs_k[:, None]) * stride_r_n + (tl.arange(0, BLOCK_N)[None, :] * stride_r_k)
        R_mask = (k_offset + offs_k[:, None] < N) & (tl.arange(0, BLOCK_N)[None, :] < N)
        R_tile = tl.load(R_ptrs, mask=R_mask, other=0.0).to(tl.float32)

        # ---- matrix multiply‑accumulate ----
        acc += tl.dot(L_tile, R_tile)

    # store result
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    O_ptrs = O_base + (offs_m[:, None] * stride_o_n) + (offs_n[None, :] * stride_o_k)
    O_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(O_ptrs, acc, mask=O_mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel
    for the N³ multiplicative update.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # torch.nn.functional.linear performs: out = x @ weight.T
    left = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply optional mask (mask shape: B,N,N)
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask_f = torch.ones_like(mask, dtype=x.dtype, device=device)
    else:
        mask_f = mask
    mask_f = mask_f.unsqueeze(-1)                       # (B,N,N,1)
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM: out = left @ right^T   (over the K axis)
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) then flatten leading two dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N_check, _ = left_perm.shape
    assert N_check == N, "unexpected shape"

    # transpose right for the inner product (K dimension becomes row)
    right_t = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    # merge batch & head dimensions for a single Triton launch
    M = B * H
    left_flat = left_perm.view(M, N, N)
    right_flat = right_t.view(M, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (M,)                      # one program per (B*H) matrix

    # strides (in elements) for the flattened tensors
    stride_l_m, stride_l_n, stride_l_k = left_flat.stride()
    stride_r_m, stride_r_n, stride_r_k = right_flat.stride()
    stride_o_m, stride_o_n, stride_o_k = out_flat.stride()

    trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_m, stride_l_n, stride_l_k,
        stride_r_m, stride_r_n, stride_r_k,
        stride_o_m, stride_o_n, stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]               # (dim, hidden_dim)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate                               # gating (broadcastable)

    # final projection back to original dim
    out = F.linear(out, to_out_w)                      # (B,N,N,dim)

    # ensure output dtype matches expectation (float32)
    return out.to(dtype=torch.float32)
