"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The implementation follows the reference PyTorch module but moves the two
LayerNorms (the one on the input tensor and the one on the hidden‑dim output)
to Triton.  All linear projections, gating, and the large 4‑D einsum are kept
in PyTorch (using bf16 for speed).  The final tensor is returned in float32.

The Triton kernels operate on a single “row’’ of the last dimension:
    * one program processes one (batch, i, j) element,
    * it loads the C‑dim slice, computes mean/variance,
    * normalises, then applies weight + bias.
Both the input‑norm (dim = C) and the hidden‑norm (dim = hidden_dim) reuse the
same kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton LayerNorm kernel (row‑wise over the last dimension)
# ------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const float*
    y_ptr,          # *float*
    weight_ptr,     # *const float*
    bias_ptr,       # *const float*
    eps,            # float32 scalar
    B, N, C,        # problem size
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    BLOCK: tl.constexpr,               # C (compile‑time)
):
    pid = tl.program_id(0)               # one program per (b,i,j)
    total = B * N * N
    mask_pid = pid < total
    if not mask_pid:
        return

    # ----- decode (b,i,j) from linear pid -------------------
    b = pid // (N * N)
    resid = pid % (N * N)
    i = resid // N
    j = resid % N

    # ----- offsets for the C‑dim slice -----------------------
    offs_c = tl.arange(0, BLOCK)
    mask_c = offs_c < C                         # handle non‑divisible C

    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    x = tl.load(x_base + offs_c * stride_xc, mask=mask_c, other=0.0).to(tl.float32)

    # ----- mean ------------------------------------------------
    sum_x = tl.sum(x, axis=0)
    mean = sum_x * (1.0 / C)

    # ----- variance --------------------------------------------
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) * (1.0 / C)

    # ----- normalise --------------------------------------------
    inv_std = 1.0 / tl.sqrt(var + eps)
    norm = diff * inv_std

    # ----- weight & bias (1‑D) ---------------------------------
    w = tl.load(weight_ptr + offs_c, mask=mask_c, other=0.0)
    b_ = tl.load(bias_ptr + offs_c, mask=mask_c, other=0.0)

    out = norm * w + b_

    # ----- write back -------------------------------------------
    tl.store(y_base + offs_c * stride_yc, out.to(tl.float32), mask=mask_c)


# ------------------------------------------------------------
# Helper: launch a row‑wise LayerNorm
# ------------------------------------------------------------
def launch_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    x      : [B, N, N, C]   (any float dtype)
    weight : [C]
    bias   : [C]
    returns y with same shape & dtype (float32 accumulator, stored as torch.float32)
    """
    assert x.ndim == 4
    B, N, _, C = x.shape
    y = torch.empty_like(x, dtype=torch.float32, device=x.device)

    # strides (in elements, not bytes)
    sx = x.stride()
    sy = y.stride()
    # launch one program per (b,i,j)
    grid = (B * N * N,)

    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        eps,
        B, N, C,
        sx[0], sx[1], sx[2], sx[3],
        sy[0], sy[1], sy[2], sy[3],
        BLOCK=C,               # C is a compile‑time constant for the kernel launch
    )
    return y


# ------------------------------------------------------------
# Custom kernel entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [bs, seq_len, seq_len, dim]   torch.float32
            - mask         : [bs, seq_len, seq_len]        torch.float32 or bool
            - weights      : dict of model parameters (torch tensors)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        output: torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------
    # unpack
    # --------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------
    # 1️⃣ Input LayerNorm (row‑wise over dim)
    # --------------------------------------------------------
    x_norm = launch_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )  # shape [bs, N, N, dim]  (float32)

    # --------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------
    # shape: [..., hidden_dim]
    left = F.linear(x_norm, weights["left_proj.weight"])      # (out_features, in_features)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------
    # 3️⃣ Apply mask (if present)
    # --------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                         # [bs,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------
    # 4️⃣ Gating tensors
    # --------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------
    # 5️⃣ Main contraction (einsum) – use bf16 for speed
    # --------------------------------------------------------
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # out shape [bs, N, N, hidden_dim]
    out = torch.einsum("...ikd,...jkd->...ijd", left_bf16, right_bf16)
    out = out.to(torch.float32)            # back to fp32 for the rest

    # --------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm
    # --------------------------------------------------------
    out_norm = launch_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )  # shape [bs, N, N, hidden_dim]

    # --------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection
    # --------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])   # → [bs, N, N, dim]

    return output
