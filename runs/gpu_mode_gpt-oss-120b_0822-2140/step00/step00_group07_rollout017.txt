"""
TriMul “outgoing” implementation for AlphaFold‑3.

The heavy part is the 4‑D contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched matrix multiplication:
    left_h  : (B,N,N)   → A
    right_h : (B,N,N)   → B
    out_h   = A @ Bᵀ   (per hidden dimension h)

We fuse this N³ work in a custom Triton kernel (fp16 input, fp32
accumulation).  All surrounding LayerNorm / Linear / gating steps are
performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batch‑first layout)
# A, B, C – shape (batch, M, K) , (batch, N, K) , (batch, M, N)
#   (the B matrix is accessed transposed, i.e. stride over K then N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr,                # *float16   – left projected tensor
    b_ptr,                # *float16   – right projected tensor
    c_ptr,                # *float16   – output
    M, N, K,              # int32      – matrix sizes (all = seq_len)
    stride_a_batch,       # int32
    stride_am, stride_ak,  # int32
    stride_b_batch,       # int32
    stride_bk, stride_bn,  # int32 (note: B is transposed)
    stride_c_batch,       # int32
    stride_cm, stride_cn,  # int32
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch index (B*hidden)
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for edge tiles
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the block
    a_base = a_ptr + pid_z * stride_a_batch
    b_base = b_ptr + pid_z * stride_b_batch
    c_base = c_ptr + pid_z * stride_c_batch

    # ------------------------------------------------------------------
    # Accumulator (float32 for precision)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension, loading tiles of A and Bᵀ
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_offset

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (cur_k[None, :] * stride_ak)
        a_mask = mask_m[:, None] & (cur_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load Bᵀ tile: because of the transpose we step over K first,
        # then over N (the original column index).
        b_ptrs = b_base + (cur_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b_mask = (cur_k[:, None] < K) & mask_n[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)

        # Advance K‑pointers for next iteration
        # (automatic via cur_k advancement)
    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=c_mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor : torch.Tensor [B, N, N, D]  (float32)
        * mask         : torch.Tensor [B, N, N]    (float32/bool)
        * weights      : dict of torch.Tensor with the model parameters
        * config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor  [B, N, N, D]  (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask shape [B,N,N] → broadcast over H)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid on linear transformations of x)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched N×N matrix multiplication via Triton
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) → contiguous → view as (B*H, N, N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * H

    left_flat  = left_t.view(batch_h, N, N).to(torch.float16)
    right_flat = right_t.view(batch_h, N, N).to(torch.float16)

    # Output buffer in fp16 (will be cast back to fp32 later)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Kernel launch parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i dimension)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j dimension)
        batch_h,                   # batch*hidden
    )

    # Strides in **elements** (not bytes)
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0),   # batch stride
        left_flat.stride(1),   # row stride (M)
        left_flat.stride(2),   # col stride (K)
        right_flat.stride(0),  # batch stride
        right_flat.stride(2),  # K stride (because we access Bᵀ)
        right_flat.stride(1),  # N stride (after transpose)
        out_flat.stride(0),    # batch stride
        out_flat.stride(1),    # M stride
        out_flat.stride(2),    # N stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to regular 4‑D layout and cast to fp32
    out = out_flat.to(torch.float32).view(B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                                   # broadcasted over H
    out = F.linear(out, weights["to_out.weight"])          # back to D

    return out
