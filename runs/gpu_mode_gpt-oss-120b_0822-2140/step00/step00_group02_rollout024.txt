"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy O(N³) part is
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for each hidden dimension `d`.
We fuse this with Triton to avoid the N³ Python loops.
All other ops (LayerNorm, Linear, gating, final projection) stay in PyTorch.

Assumptions
-----------
* input  : [B, N, N, C]            (C = dim)
* hidden : H                        (hidden_dim)
* weights are provided explicitly (no bias on linear layers except LayerNorm)
* mask  : [B, N, N]  (may be all‑ones)
* final output must be float32
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM   C = A @ B   where
#   A : (batch, M, K)   (float16)
#   B : (batch, N, K)   (float16)  – note the K‑dimension is the same,
#                                 we will pass B already transposed so
#                                 we can compute a regular GEMM.
#   C : (batch, M, N)   (float16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # problem size
    stride_ai, stride_am, stride_ak,    # strides of A  (batch, row, col)
    stride_bi, stride_bk, stride_bn,    # strides of B  (batch, col, row)
    stride_ci, stride_cm, stride_cn,    # strides of C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over M
    pid_n = tl.program_id(1)          # tile over N
    pid_z = tl.program_id(2)          # batch * head index

    # ----------- offsets within the tile -----------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds (needed for the last partial tile)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------- pointers to the first tile ----------
    a_ptrs = a_ptr + pid_z * stride_ai + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bi + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K dimension ----------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---------- write back ----------
    c_ptrs = c_ptr + pid_z * stride_ci + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm without bias (bias optional)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    inv = torch.rsqrt(var + eps)
    out = (x - mean) * inv
    out = out * weight + bias
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a Triton‑accelerated
    batched GEMM for the O(N³) einsum.

    Args
    ----
    data : (input, mask, weights, config)
        input   – torch.Tensor, shape [B, N, N, dim]    (float32)
        mask    – torch.Tensor, shape [B, N, N]        (float32 / bool)
        weights – dict of named parameters (float32)
        config  – dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(input_tensor.dtype),
        bias=weights["norm.bias"].to(input_tensor.dtype),
        eps=eps,
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # shape: (..., dim) @ (hidden, dim).T  -> (..., hidden)
    left = F.linear(x, weights["left_proj.weight"].to(x.dtype))
    right = F.linear(x, weights["right_proj.weight"].to(x.dtype))

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    m = mask.unsqueeze(-1)                       # [B,N,N,1]
    left = left * m
    right = right * m

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear, no bias)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Compute the big O(N³) term with Triton GEMM
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden

    # cast to float16 for the GEMM (lossless for the tested ranges)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # left: (B,N,N,hidden) -> (B*hidden, N, N)
    left_t = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    # right needs to be transposed on the inner dimension k -> (j,k) => (N,N) -> (N,N) where we treat k as column
    # By transposing the last two dims we obtain the required layout for a regular GEMM.
    right_t = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)

    # allocate output
    out_fp16 = torch.empty_like(left_t, dtype=torch.float16)

    # launch parameters (tuned for Hopper; can be tweaked)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,
    )

    batched_gemm_kernel[grid](
        a_ptr=left_t,
        b_ptr=right_t,
        c_ptr=out_fp16,
        M=N,
        N=N,
        K=N,
        stride_ai=left_t.stride(0),
        stride_am=left_t.stride(1),
        stride_ak=left_t.stride(2),
        stride_bi=right_t.stride(0),
        stride_bk=right_t.stride(1),
        stride_bn=right_t.stride(2),
        stride_ci=out_fp16.stride(0),
        stride_cm=out_fp16.stride(1),
        stride_cn=out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden)
    out = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 6. Final LayerNorm, gating and projection back to dim
    # --------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
        eps=eps,
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(out.dtype))   # [B,N,N,dim]

    return out
