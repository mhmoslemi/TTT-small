"""
TriMul “outgoing’’ implementation.

The heavy part of the operator is the 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix multiplication for every hidden channel `d`.
We fuse the batch (B) and hidden‑channel (D) dimensions into a single Triton
program‑id and run a classic GEMM kernel (M=N, K=N).  All LayerNorms, linear
projections and gating are done with PyTorch (they are cheap compared with the
N³ work).  The kernel works for any `B ∈ {1,2}`, `N ≤ 1024` and `D` up to 384.
The final output is returned in float32.
"""

import torch
import triton
import triton.language as tl
import math
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for one hidden channel
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, D,                                     # shape constants
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for a single (batch, hidden) pair:
        out[i, j] = Σ_k left[i, k] * right[j, k]
    where left and right are matrices of shape (N, N) (the hidden dimension
    has been fused into the program id).
    """
    pid_bn = tl.program_id(0)          # combined batch * hidden index
    pid_m  = tl.program_id(1)          # block row   (i)
    pid_n  = tl.program_id(2)          # block col   (j)

    # ------------------------------------------------------------------
    # decode combined index into (b, d)
    # ------------------------------------------------------------------
    d = pid_bn % D
    b = pid_bn // D

    # ------------------------------------------------------------------
    # base pointers for this (b, d) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (good precision, still fast on Hopper)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K‑loop (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  -> shape (BLOCK_M, BLOCK_K)
        a_ptr = left_base + \
                offs_m[:, None] * stride_left_i + \
                offs_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right[j,k] (we need the transpose, i.e. matrix (K,N))
        #   right_T[k,j] = right[j,k]
        b_ptr = right_base + \
                offs_k[:, None] * stride_right_k + \
                offs_n[None, :] * stride_right_i
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # GEMM accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = out_base + \
               offs_m[:, None] * stride_out_i + \
               offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: manual LayerNorm (no bias for speed, bias added later)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: (..., C)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the task
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing’’ TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]      (float32 or bool)
            - weights: dict of torch tensors
            - config : dict with keys dim, hidden_dim

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim channel)
    # --------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                         # shape [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)   left, right
    # --------------------------------------------------------------
    # shape: (..., hidden)
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask if present
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating of left/right
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction – Triton GEMM (batch*hidden fused)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # same dtype/shape as left/right

    # strides (int64)
    s_left = left.stride()   # (B, N, N, hidden)
    s_right = right.stride()
    s_out = out.stride()

    # tile sizes – chosen for H200 (SM90) occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                     # combined batch‑hidden dimension
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden,
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_d=s_left[3],
        stride_right_b=s_right[0],
        stride_right_i=s_right[1],
        stride_right_k=s_right[2],
        stride_right_d=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_d=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden dim) + gate + final linear
    # --------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
