"""
TriMul "outgoing" implementation.

The heavy N³ part (the einsum over the shared sequence dimension) is
implemented as a batched matrix‑multiplication in Triton:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

We keep the cheap LayerNorm / Linear / gating ops in PyTorch and cast the
mat‑mul operands to FP16 for speed.  The final result is brought back to
FP32, normalized, gated and projected back to the original channel dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B^T) over many
# independent batches (batch = batch_size * hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (per batch)
    stride_Ab, stride_Am, stride_Ak,   # strides for A: batch, row, col
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (stored transposed): batch, row(K), col(N)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) Map 1‑D program id to (batch, block_m, block_n)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    block_m = block_id // num_blocks_n
    block_n = block_id % num_blocks_n

    # --------------------------------------------------------------
    # 2) Compute offsets for this block
    # --------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the start of the block
    a_ptrs = A_ptr + batch_id * stride_Ab \
                    + offs_m[:, None] * stride_Am \
                    + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + batch_id * stride_Bb \
                    + offs_k[:, None] * stride_Bk \
                    + offs_n[None, :] * stride_Bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)

        # Advance pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # 4) Store result
    # --------------------------------------------------------------
    c_ptrs = C_ptr + batch_id * stride_Cb \
                    + offs_m[:, None] * stride_Cm \
                    + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A : (B*H, M, K)  FP16
    B : (B*H, K, N)  FP16   (already transposed)
    returns C : (B*H, M, N)  FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), device=A.device, dtype=torch.float16)

    # Compile‑time block size (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(M, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (batch * grid_m * grid_n,)

    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        M=M,
        N=N,
        K=K,
        # strides (in elements, not bytes)
        stride_Ab=A.stride(0),
        stride_Am=A.stride(1),
        stride_Ak=A.stride(2),
        stride_Bb=B.stride(0),
        stride_Bk=B.stride(1),
        stride_Bn=B.stride(2),
        stride_Cb=C.stride(0),
        stride_Cm=C.stride(1),
        stride_Cn=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing) with the core N³ operation off‑loaded to Triton.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]   (float32 on CUDA)
            - mask:         [B, N, N]        (float32/bool)
            - weights: dict of torch tensors (already on device)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        output: [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask may be all‑ones)
    # ------------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for batched matmul:
    #    left_perm:  [B, hidden, N, N]
    #    right_perm: [B, hidden, N, N] -> transpose last two dims -> [B, hidden, N, N] (K,N)
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    B, N_seq, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, hidden, N, N)

    # Cast to FP16 for the kernel
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16).transpose(-2, -1)  # (B, hidden, N, K) -> (B, hidden, K, N)

    # Merge batch and hidden dims
    left_reshape = left_fp16.view(-1, N_seq, N_seq)    # (B*hidden, M, K)
    right_reshape = right_fp16.view(-1, N_seq, N_seq)  # (B*hidden, K, N)

    # ------------------------------------------------------------------
    # 6) Triton batched matmul
    # ------------------------------------------------------------------
    out_fp16 = triton_batched_matmul(left_reshape, right_reshape)  # (B*hidden, N, N)

    # Restore batch and channel dimensions, and permute back to (B,N,N,hidden)
    out = out_fp16.view(B, hidden, N_seq, N_seq).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (over hidden dim) + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to dim
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])   # shape (B, N, N, dim)

    return output
