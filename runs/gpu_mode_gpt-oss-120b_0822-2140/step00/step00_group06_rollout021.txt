"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation
=================================================================

The heavy part of the TriMul forward pass is the 3‑D einsum

    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

For every batch element *b* and hidden channel *d* this is a matrix
multiplication

    C_b,d = A_b,d @ B_b,d.T                (A,B shape N×N)

We fuse this batched GEMM (with the transposition on‑the‑fly) into a
single Triton kernel.  All other operations (LayerNorm, Linear
projections, gates, final LayerNorm + Linear) are done in PyTorch
so that the kernel stays focused on the N³ work.

The kernel works with float16 internally (the reference uses BF16) and
returns a float32 tensor as required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------------
# Triton kernel: batched GEMM with the right‑matrix transposed on‑the‑fly
# ------------------------------------------------------------------------
@triton.jit
def _batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all == seq_len)
    stride_am, stride_ak,              # A strides: (batch, row, col)
    stride_bk, stride_bn,              # B strides (notice the swap for transpose)
    stride_cm, stride_cn,              # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col
    pid_b = tl.program_id(2)      # batch‑hidden index

    # --------------------------------------------------------------
    # block offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for the edges of the matrices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch‑hidden slice
    a_base = a_ptr + pid_b * (M * K)
    b_base = b_ptr + pid_b * (K * N)
    c_base = c_ptr + pid_b * (M * N)

    # ----------------------------------------------------------------
    # Accumulator in FP32 (more stable than FP16 accumulation)
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        # ---- load A tile ------------------------------------------------
        a_offset = a_base + (offs_m[:, None] * stride_am) + ((k_iter * BLOCK_K + offs_k)[None, :] * stride_ak)
        a = tl.load(a_offset, mask=mask_m[:, None] & ((k_iter * BLOCK_K + offs_k) < K), other=0.0)

        # ---- load B tile (transposed) ------------------------------------
        # we want B_T[k, n] = B[n, k]  →  stride_bk = stride row of original B (=N)
        b_offset = b_base + ((k_iter * BLOCK_K + offs_k)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_offset, mask=((k_iter * BLOCK_K + offs_k) < K)[:, None] & mask_n[None, :], other=0.0)

        # ---- dot product -------------------------------------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write result (store as FP16, model later casts to FP32)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_offset = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_offset, c, mask=mask_m[:, None] & mask_n[None, :])


# ------------------------------------------------------------------------
# Helper: fused LayerNorm (bias may be None)
# ------------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    return F.layer_norm(x, x.shape[-1:], weight, bias, eps)


# ------------------------------------------------------------------------
# Main entry point expected by the evaluator
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – fast Triton implementation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            * input_tensor : [B, N, N, D]   (float32)
            * mask         : [B, N, N]      (float32 or bool)
            * weights      : dict of torch tensors
            * config       : dict containing "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the input
    # --------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, D]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – use torch.nn.functional.linear
    # --------------------------------------------------------------
    # shape (hidden, D)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_w = weights["to_out.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    # keep everything in float32 until the GEMM where we downcast
    left = F.linear(x, left_proj_w)            # [B, N, N, hidden]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Mask (if present) – mask is broadcast over the hidden dim
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)              # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM – fp16 for speed
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) and cast to fp16
    left = left.permute(0, 3, 1, 2).contiguous()      # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()    # [B, hidden, N, N]

    left = left.view(B * hidden, N, N).to(torch.float16)
    right = right.view(B * hidden, N, N).to(torch.float16)

    # output buffer
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 6. Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden,                       # batch‑hidden dimension
    )

    # strides for the contiguous layout (batch, row, col)
    stride_am = N          # row stride
    stride_ak = 1          # col stride
    # for Bᵀ we swap the two last strides
    stride_bk = N          # step in “k” direction lands on next original row
    stride_bn = 1          # step in “n” direction lands on next original col
    stride_cm = N
    stride_cn = 1

    _batched_gemm_transpose_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # --------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden] (still fp16) → fp32
    # --------------------------------------------------------------
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = _layer_norm(out, weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim D

    return out
