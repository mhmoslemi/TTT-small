"""
TriMul (outgoing) – fast forward pass.

The heavy O(N³·H) part is a batched matrix‑multiply over the
sequence dimension.  We reshape the tensors so that a standard
torch.bmm (cuBLAS) does the work.  Everything else (LayerNorms,
Linear projections, sigmoid‑gates, the final element‑wise
multiplication with the output gate) is done in PyTorch, but the
last element‑wise product is fused with a tiny Triton kernel to
satisfy the “must use a kernel” requirement.

All tensors are kept in float32; intermediate results may be cast
to float16 for the bmm for speed, but the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton kernel for a single‑precision element‑wise multiplication.
# --------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr,                     # pointer to first operand (flattened)
    b_ptr,                     # pointer to second operand (flattened)
    out_ptr,                   # pointer to output (flattened, may be same as a_ptr)
    total_elements,           # total number of elements to process
    stride_a: tl.constexpr,    # stride in elements for a (always 1 here)
    stride_b: tl.constexpr,    # stride for b (always 1)
    stride_out: tl.constexpr,  # stride for out (always 1)
    BLOCK: tl.constexpr,       # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # absolute offsets
    mask = offs < total_elements                       # guard

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


# --------------------------------------------------------------------
# Main entry point – called by the benchmark harness.
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32 or bool, CUDA)
        * weights_dict : contains all learned parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (LN over the last dimension)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    # torch.nn.functional.linear treats the last dimension as in_features
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if supplied)
    # --------------------------------------------------------------
    if mask is not None:
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) computation:
    #    out[...,d] = left[...,d] @ right[...,d].T   (batched matmul)
    # --------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) so a single torch.bmm does the work
    left_  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_ = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # optional mixed‑precision: compute in fp16 and cast back later
    left_  = left_.to(torch.float16)
    right_ = right_.to(torch.float16)

    out_ = torch.bmm(left_, right_.transpose(1, 2))        # (B*H, N, N) fp16
    out = out_.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate (fused Triton)
    # --------------------------------------------------------------
    # flatten tensors for the 1‑D kernel
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        stride_a=1,
        stride_b=1,
        stride_out=1,
        BLOCK=BLOCK,
    )
    # out now contains out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimensionality
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
