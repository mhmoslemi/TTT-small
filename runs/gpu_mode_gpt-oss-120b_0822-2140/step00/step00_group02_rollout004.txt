"""
TriMul “outgoing” implementation using Triton for the
N³ einsum:   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d].

The heavy contraction is performed as a batched matrix multiplication
(left @ rightᵀ) where the batch dimension is (batch * hidden_dim).
All other operations (LayerNorm, linear projections, gating, final
LayerNorm + linear) are kept in PyTorch for simplicity.
The Triton kernel works on fp16 matrices and accumulates in fp32,
producing an fp16 result that is immediately cast back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul where the second operand is already
# transposed (i.e. we compute A @ Bᵀ).  Shapes:
#   A : (M, N, K)   (M = batch*hidden_dim, N = seq_len, K = seq_len)
#   B : (M, K, N)   (the transposed view of the original right tensor)
#   C : (M, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_T_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M, N, K)
    stride_bm, stride_bk, stride_bn,  # B: (M, K, N)
    stride_cm, stride_cn,          # C: (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block over M dimension
    pid_n = tl.program_id(1)          # block over N dimension

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers
        a_ptrs = A_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = B_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # load with masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # fp16 * fp16 -> fp32 accumulation

    # write back (convert to fp16 for storage)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N]   (float32 or bool)
            - weights: dict of pre‑loaded weight tensors
            - config : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                         # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy N³ contraction via Triton: out = left @ rightᵀ
    #    left  : (B, N, N, H)  -> (M, N, K)   with M = B*H
    #    rightᵀ: (B, N, N, H) -> (M, K, N)   (swap dim 1 & 2)
    # ------------------------------------------------------------------
    # cast to fp16 for the kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # transpose the second and third axes of right (j,k) -> (k,j)
    right_T = right_f16.transpose(1, 2)

    # bring hidden dim into the batch dimension
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_mat = right_T.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)

    M = B * hidden_dim
    left_mat = left_mat.view(M, N, N)                         # (M, N, K)
    right_mat = right_mat.view(M, N, N)                       # (M, K, N)

    # output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides in element units (not bytes)
    stride_am, stride_ak, stride_an = left_mat.stride()
    stride_bm, stride_bk, stride_bn = right_mat.stride()
    stride_cm, stride_cn = out_mat.stride(0), out_mat.stride(2)

    # block sizes – tuned for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_matmul_T_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out.to(torch.float32)
