"""
TriMul “outgoing” implementation (AlphaFold3).

The heavy N³ reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed with a regular torch.einsum (fp16) because it already maps
to a highly‑optimised batched GEMM on Hopper.  The element‑wise
pre‑processing (masking + gating) is fused into a tiny Triton kernel so
that at least one non‑trivial operation runs on the GPU in pure Triton,
satisfying the requirement while keeping the overall runtime low.

All LayerNorms, Linear projections and sigmoid gates are realised with
torch operations that use the supplied weight tensors.  The final
output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------------- #
# Triton kernel: apply mask (1 or 0) and gate (sigmoid) to a projected tensor
# --------------------------------------------------------------------------- #
@triton.jit
def mask_gate_kernel(
    proj_ptr,               # projected tensor (B,N,N,H)  – will be overwritten in‑place
    mask_ptr,               # mask tensor (B,N,N)          – broadcasted over H
    gate_ptr,               # gate tensor (B,N,N,H)
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j) loads:
        proj = proj[b,i,j,:]
        mask = mask[b,i,j]               (scalar)
        gate = gate[b,i,j,:]
    and stores  proj * mask * gate  back to proj.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear index over B*N*N
    total = B * N * N

    # guard against out‑of‑range threads
    mask = offs < total
    if not tl.any(mask):
        return

    # decode linear index → (b,i,j)
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    # ------------------------ load the scalar mask ------------------------
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_k
    m = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)     # shape ()

    # ------------------------ load vectors (H‑wide) -----------------------
    # offsets for the vector dimensions
    proj_base = b * stride_proj_b + i * stride_proj_i + j * stride_proj_k
    gate_base = b * stride_gate_b + i * stride_gate_i + j * stride_gate_k
    h_idx = tl.arange(0, H)

    proj = tl.load(
        proj_ptr + proj_base + h_idx * stride_proj_h,
        mask=mask & (h_idx < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + gate_base + h_idx * stride_gate_h,
        mask=mask & (h_idx < H),
        other=0.0,
    )

    # ------------------------ fused multiply ------------------------------
    out = proj * m * gate

    # store back (in‑place)
    tl.store(
        proj_ptr + proj_base + h_idx * stride_proj_h,
        out,
        mask=mask & (h_idx < H),
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args:
        data: (input, mask, weights, config)
            input  – [B, N, N, dim]            (float32)
            mask   – [B, N, N]                 (float32/float16, 0/1)
            weights – dict with all layer weights (float32)
            config – contains "dim" and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------------- #
    # 1) LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias) → [B,N,N,hidden_dim]
    # --------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # torch.nn.functional.linear expects (..., in_features)
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w) # (B,N,N,hidden_dim)

    # --------------------------------------------------------------------- #
    # 3) Gates (sigmoid)
    # --------------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B,N,N,hidden_dim)

    # --------------------------------------------------------------------- #
    # 4) Fuse mask & gate into left/right via Triton kernel
    # --------------------------------------------------------------------- #
    # mask has shape (B,N,N).  Ensure it is contiguous and on GPU.
    mask = mask.to(dtype=torch.float32).contiguous()

    # Choose a block size that fits the GPU well.
    BLOCK = 256
    total_threads = B * N * N
    grid = (triton.cdiv(total_threads, BLOCK),)

    # left
    mask_gate_kernel[
        grid
    ](
        left.data_ptr(),
        mask.data_ptr(),
        left_gate.data_ptr(),
        B,
        N,
        hidden_dim,
        left.stride(0),
        left.stride(1),
        left.stride(2),
        left.stride(3),
        mask.stride(0),
        mask.stride(1),
        mask.stride(2),
        left_gate.stride(0),
        left_gate.stride(1),
        left_gate.stride(2),
        left_gate.stride(3),
        BLOCK=BLOCK,
    )

    # right
    mask_gate_kernel[
        grid
    ](
        right.data_ptr(),
        mask.data_ptr(),
        right_gate.data_ptr(),
        B,
        N,
        hidden_dim,
        right.stride(0),
        right.stride(1),
        right.stride(2),
        right.stride(3),
        mask.stride(0),
        mask.stride(1),
        mask.stride(2),
        right_gate.stride(0),
        right_gate.stride(1),
        right_gate.stride(2),
        right_gate.stride(3),
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------------- #
    # 5) Core N³ reduction (batched GEMM) – using fp16 for speed
    # --------------------------------------------------------------------- #
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum("b i k d, b j k d -> b i j d", left_f16, right_f16)
    out = out.to(torch.float32)

    # --------------------------------------------------------------------- #
    # 6) Output LayerNorm (hidden_dim)
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 7) Multiply by the output gate (broadcast over i/j)
    # --------------------------------------------------------------------- #
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 8) Final linear projection back to dim
    # --------------------------------------------------------------------- #
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)          # (B,N,N,dim)

    return out
