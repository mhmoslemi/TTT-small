"""
TriMul “outgoing” kernel (AlphaFold‑3 style) – Triton + PyTorch hybrid.

The heavy N³ part (the pair‑wise outer product) is implemented as a
batched matrix‑multiply:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is a batched GEMM per hidden channel (h).  All other
operations (LayerNorm, linear projections, gating, final LayerNorm
and linear) stay in PyTorch.  The kernel works in fp16 internally and
stores the result in fp16, the surrounding code casts back to fp32
as required by the reference implementation.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with hidden dimension collapsed into the
# program ID (one program per (batch, hidden, block_i, block_j)).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    B, M, N, K, H,                           # logical sizes
    stride_ab, stride_ai, stride_ak, stride_ah,  # a strides (B,M,K,H)
    stride_bb, stride_bk, stride_bj, stride_bh,  # b strides (B,K,N,H)
    stride_cb, stride_ci, stride_cj, stride_ch,  # c strides (B,M,N,H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Decompose the linear pid into (batch, hidden, block_i, block_j)
    # --------------------------------------------------------------
    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)

    # total programs = B * H * total_m * total_n
    b = pid // (H * total_m * total_n)
    tmp = pid % (H * total_m * total_n)
    h = tmp // (total_m * total_n)
    tmp = tmp % (total_m * total_n)
    block_i = tmp // total_n
    block_j = tmp % total_n

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    mask_i = i_offset < M
    mask_j = j_offset < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (blocked)
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k_blocks):
        cur_k = kk * BLOCK_K + k_offset
        mask_k = cur_k < K

        # a[b, i, k, h]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + b * stride_ab
            + i_offset[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
            + h * stride_ah
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # b[b, k, j, h]  shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + b * stride_bb
            + cur_k[:, None] * stride_bk
            + j_offset[None, :] * stride_bj
            + h * stride_bh
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # dot over the K dimension (fp32 accumulation)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store result – cast back to fp16 (same dtype as PyTorch will use)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + b * stride_cb
        + i_offset[:, None] * stride_ci
        + j_offset[None, :] * stride_cj
        + h * stride_ch
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Host wrapper – the required `custom_kernel` entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass using a Triton GEMM for the
    (i,k) × (j,k) outer‑product.  All other ops stay in PyTorch.
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # 0) unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)                    # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if nomask:
        mask = 1.0
    else:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid(sigmoid(...))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton GEMM (the N³ part)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # cast to fp16 for the kernel (speed + less shared memory)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # right must be transposed on the two middle dimensions for the dot‑product
    right_t = right_fp16.permute(0, 2, 1, 3).contiguous()   # (B, K=N, N, H)

    out_fp16 = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # strides (int64) as expected by Triton
    a_strides = left_fp16.stride()   # (B, i, k, h)
    b_strides = right_t.stride()     # (B, k, j, h)
    c_strides = out_fp16.stride()    # (B, i, j, h)

    # ------------------------------------------------------------------
    # Launch configuration – tuned for Hopper (SM90)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    total_m = math.ceil(N / BLOCK_M)
    total_n = math.ceil(N / BLOCK_N)
    grid = (B * H * total_m * total_n,)

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_t,
        out_fp16,
        B,
        N,
        N,
        N,
        H,
        a_strides[0],
        a_strides[1],
        a_strides[2],
        a_strides[3],
        b_strides[0],
        b_strides[1],
        b_strides[2],
        b_strides[3],
        c_strides[0],
        c_strides[1],
        c_strides[2],
        c_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Rest of the pipeline (still in float32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)               # [B,N,N,hidden]

    # to_out_norm (LayerNorm over hidden dim)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # multiply by output gate (broadcast over hidden dim)
    out = out * out_gate

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"]            # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                 # [B,N,N,dim]

    return out
