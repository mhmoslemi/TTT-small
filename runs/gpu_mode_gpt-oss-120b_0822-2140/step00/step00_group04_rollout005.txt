"""
TriMul (outgoing) – Triton‑accelerated implementation
=====================================================

The forward pass consists of:

1.  Layer‑norm over the last dimension (dim).
2.  Two linear projections (left / right) and three gating projections.
3.  Optional element‑wise masking.
4.  Gated left / right tensors.
5.  **Heavy part** – for every hidden channel h we need

        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

   which is a batched matrix‑multiplication :  left[...,h] @ right[...,h].T.
   This O(N³) operation is performed by a custom Triton kernel that
   computes many (batch·hidden) matrix products in parallel.
6.  Layer‑norm on the hidden dimension, a final gating, and a linear
   projection back to the original “dim”.

Only the matrix‑multiply (step‑5) is written in Triton – everything else
uses PyTorch’s highly‑optimized kernels.  The kernel works for any
batch size (B ≤ 2), sequence length N (up to 1024) and hidden dimension
H (typically 128).  Mixed‑precision (FP16) is used inside the kernel and
the result is cast back to FP32 before the final LayerNorm / Linear.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B.T) for many
# batches * hidden channels.  All tensors are contiguous and have
# shape (B*H, N, N) with strides (stride_batch, stride_m, stride_n).
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A,                     # ptr to left  tensor  (batch, M, K)
    B,                     # ptr to right tensor  (batch, N, K)  (will be read as K,N)
    C,                     # ptr to output tensor (batch, M, N)
    M, N, K,
    stride_am, stride_ak, stride_ab,   # strides for A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides for B (batch, K, N)   <-- transposed view
    stride_cm, stride_cn, stride_cb,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    #  Compute tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current batch
    a_ptrs = A + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (FP16 → FP32 while accumulating)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)

        # GEMM of the tiles
        acc += tl.dot(a, b)

        # Increment pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    #  Write back
    # ------------------------------------------------------------------
    c_ptrs = C + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
#  Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              input_tensor : [B, N, N, dim]   (float32, on CUDA)
              mask_tensor  : [B, N, N]       (float32/bool, on CUDA)
              weights_dict : {"norm.weight":..., "norm.bias":..., ...}
              config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool}
    Returns:
        output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # keep everything in FP32 for the linear layers – they are already fast
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating projections + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM kernel
    # ------------------------------------------------------------------
    # we need shape (B, H, N, N) so that each (b, h) pair is an independent matrix
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose the inner dimensions of right for the A @ B.T pattern
    right_t = right.transpose(-2, -1).contiguous()  # [B, H, N, N] (now K,N)

    # Collapse batch & hidden into a single leading dimension
    left_flat = left.view(B * hidden_dim, N, N)    # [B*H, M, K]
    right_flat = right_t.view(B * hidden_dim, N, N)  # [B*H, K, N]

    # Cast to FP16 for the kernel (still accurate enough) and allocate output
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # Strides
    stride_am = left_fp16.stride(0)
    stride_ak = left_fp16.stride(1)
    stride_ab = left_fp16.stride(2)   # batch stride
    stride_bk = right_fp16.stride(0)
    stride_bn = right_fp16.stride(2)   # after transposition we need (K,N)
    stride_bb = right_fp16.stride(1)   # batch stride
    stride_cm = out_fp16.stride(0)
    stride_cn = out_fp16.stride(1)
    stride_cb = out_fp16.stride(2)

    # Kernel launch parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * hidden_dim,                 # batch*hidden
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )
    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to FP32 and restore original layout [B, N, N, H]
    out = out_fp16.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # Hidden‑dim LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return out
