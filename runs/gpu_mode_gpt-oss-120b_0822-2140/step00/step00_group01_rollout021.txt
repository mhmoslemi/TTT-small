"""
TriMul “outgoing” implementation.
*   LayerNorm, Linear projections and gates are done in Torch (they are cheap).
*   The expensive N‑by‑N reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is realized as a batched matrix‑multiply per hidden dimension:
        out_d = left_d @ right_d.T
    This is fused in a single Triton kernel that processes a (M×N) tile for
    several hidden‑dimension channels (BLOCK_H) at once.
*   Mixed‑precision: the tensors entering the kernel are cast to FP16,
    accumulation is FP32 and the final result is cast back to FP32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# --------------------------------------------------------------
# Triton kernel: batched matmul with an extra hidden‑dim tile
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *FP16   [B, N, N, H]
    right_ptr,         # *FP16   [B, N, N, H]
    out_ptr,           # *FP16   [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    ## ------------------------------------------------------
    ##  Program IDs & tile offsets
    ## ------------------------------------------------------
    pid = tl.program_id(0)
    # total tiles per batch
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_m = tile_id // num_tiles_n
    tile_n = tile_id % num_tiles_n

    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # hidden‑dim offsets (will be masked inside the loop)
    offs_h = tl.arange(0, BLOCK_H)

    # accumulator: FP32, shape (BLOCK_M, BLOCK_N, BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    ## ------------------------------------------------------
    ##  Main K‑loop
    ## ------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- left tile (M × K × H) ---------------------------------
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + offs_m[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None, None] & mask_k[None, :, None] & (offs_h[None, None, :] < H),
            other=0.0,
        )

        # ------- right tile (N × K × H) ---------------------------------
        # we need right transposed: original right[j,k,d] → we load it as
        # (N, K, H) and later use it in a dot product where the K dimension is
        # the reduction axis.
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + offs_n[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_n[:, None, None] & mask_k[None, :, None] & (offs_h[None, None, :] < H),
            other=0.0,
        )

        # ------- accumulate over K ---------------------------------------
        # tl.dot works on 2‑D matrices; we unroll the hidden dimension.
        for h in range(0, BLOCK_H):
            # shape (BLOCK_M, BLOCK_K) @ (BLOCK_N, BLOCK_K).T → (BLOCK_M, BLOCK_N)
            acc[:, :, h] += tl.dot(left[:, :, h], right[:, :, h])

    ## ------------------------------------------------------
    ##  Write back
    ## ------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + offs_m[:, None, None] * stride_out_i
        + offs_n[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None, None] & mask_n[None, :, None] & (offs_h[None, None, :] < H),
    )


# --------------------------------------------------------------
# Host‑side wrapper
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input, mask, weights, config)
          input   : Tensor [B, N, N, dim]  (float32, on CUDA)
          mask    : Tensor [B, N, N]       (float32 or bool, on CUDA)
          weights : dict of parameters (float32, on CUDA)
          config  : dict with keys "dim" and "hidden_dim"
    Returns:
        Tensor [B, N, N, dim] (float32, on CUDA)
    """
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    H = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the heavy kernel ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)

    # ---------- Triton launch ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 32                 # hidden‑dim tile (must divide H or be masked)

    grid = (
        B
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )  # 1‑D grid, batch & tile are folded into a single id

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # back to FP32
    out = out_fp16.to(torch.float32)

    # ---------- Output norm, gating and final linear ----------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
