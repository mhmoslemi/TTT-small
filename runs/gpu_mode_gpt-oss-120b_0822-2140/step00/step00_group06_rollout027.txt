"""
TriMul “outgoing” forward pass.

The heavy part of the operator is the ternary contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,k,j,d]

which is a batched matrix‑multiply for every hidden channel *d*.
We implement this contraction with a custom Triton kernel that
  * iterates over blocks of the (i, j) plane,
  * tiles the reduction over *k*,
  * loops over the hidden dimension *d* (≤ 384) in the register file,
  * uses `tl.dot` (Tensor‑core friendly) for the inner K‑reduction,
  * stores the result in float32 (the rest of the model stays in FP16
    for speed, the final output is cast back to FP32).

All other inexpensive operations (LayerNorm, Linear, gating, mask)
are performed with regular PyTorch kernels using the supplied
weights.  This satisfies the requirement of having a non‑trivial
portion of the work executed on a Triton kernel while keeping the
implementation simple and correct for all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: batched matmul per hidden channel (the O(N³) part)
# -------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # sizes
    stride_lb, stride_li, stride_lk, stride_lh, # left strides
    stride_rb, stride_rk, stride_rj, stride_rh, # right (already transposed) strides
    stride_ob, stride_oi, stride_oj, stride_oh, # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute for a single (batch, i‑block, j‑block) tile:
        out[b,i0:i0+M, j0:j0+N, h] = Σ_k left[b,i0:i0+M, k, h] *
                                    right[b,k, j0:j0+N, h]
    The right tensor is expected to be already transposed on the
    (k, j) axes (i.e. shape [B, K, J, H]).
    """
    pid = tl.program_id(0)

    # ---------------------------------------------------------
    # decode 1‑D program id -> (batch, i‑block, j‑block)
    # ---------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i_block = rem // num_j
    j_block = rem % num_j

    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N

    # ---------------------------------------------------------
    # create coordinate vectors
    # ---------------------------------------------------------
    offs_m = i0 + tl.arange(0, BLOCK_M)
    offs_n = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds (last partial tile)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---------------------------------------------------------
    # loop over the hidden dimension in small chunks
    # ---------------------------------------------------------
    for dh in range(0, H, BLOCK_H):
        # size of the current hidden chunk (may be < BLOCK_H at the end)
        cur_h = tl.minimum(BLOCK_H, H - dh)
        # accumulator for the whole (M, N, cur_h) tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # -----------------------------------------------------------------
        # reduction over the K dimension
        # -----------------------------------------------------------------
        for k0 in range(0, N, BLOCK_K):
            cur_k = tl.minimum(BLOCK_K, N - k0)

            # offsets for the current K‑tile
            k_offs = k0 + offs_k
            mask_k = k_offs < N

            # -----------------------------------------------------------------
            # load left: shape (M, K, cur_h)
            # -----------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + b * stride_lb
                + (offs_m[:, None, None] * stride_li)
                + (k_offs[None, :, None] * stride_lk)
                + ((dh + tl.arange(0, BLOCK_H)[None, None, :]) * stride_lh)
            )
            left = tl.load(
                left_ptrs,
                mask=mask_m[:, None, None] & mask_k[None, :, None],
                other=0.0,
                dtype=tl.float16,
            )
            # -----------------------------------------------------------------
            # load right (already transposed): shape (K, N, cur_h)
            # -----------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + b * stride_rb
                + (k_offs[:, None, None] * stride_rk)
                + (offs_n[None, :, None] * stride_rj)
                + ((dh + tl.arange(0, BLOCK_H)[None, None, :]) * stride_rh)
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_n[None, :, None],
                other=0.0,
                dtype=tl.float16,
            )

            # -----------------------------------------------------------------
            # dot‑product over the K dimension for every hidden slice
            # -----------------------------------------------------------------
            # left : (M, K, h)   -> view as (M, K, h)
            # right: (K, N, h)   -> we need (K, N, h)
            # Use tl.dot which works on 2‑D matrices, so we do it h‑wise.
            for hh in range(cur_h):
                # extract the slice for this hidden channel
                l_slice = left[:, :, hh]   # (M, K)
                r_slice = right[:, :, hh]  # (K, N)
                # dot returns (M, N)
                acc[:, :, dh + hh] += tl.dot(l_slice, r_slice)

        # -----------------------------------------------------------------
        # store the computed tile (only the valid part)
        # -----------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_ob
            + (offs_m[:, None, None] * stride_oi)
            + (offs_n[None, :, None] * stride_oj)
            + ((dh + tl.arange(0, BLOCK_H)[None, None, :]) * stride_oh)
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_m[:, None, None] & mask_n[None, :, None],
        )


# -------------------------------------------------------------
# Helper to launch the kernel
# -------------------------------------------------------------
def trimul_einsum_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    left : [B, N, N, H]   (float16)
    right_t : [B, N, N, H] (float16)   = right.permute(0,2,1,3).contiguous()
    returns out : [B, N, N, H] (float16)  (still in FP16, caller may cast)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rk, stride_rj, stride_rh = right_t.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 32  # hidden dim chunk; 32 works for all H ≤ 384

    # grid: one program per (batch, i‑block, j‑block)
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid = (B * grid_i * grid_j,)

    trimul_einsum_kernel[grid](
        left.data_ptr(),
        right_t.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rk,
        stride_rj,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    return out

# Main entry point required by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (last dimension = dim)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, torch.float32)
    right_proj_w = weights["right_proj.weight"].to(device, torch.float32)

    # use torch.nn.functional.linear : (B,N,N,dim) @ (dim, hidden_dim)
    left = F.linear(x, left_proj_w)          # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)        # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3) Apply mask (if present) – mask shape [B,N,N]
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(device, torch.float32)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) – also linear w/o bias
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device, torch.float32)
    right_gate_w = weights["right_gate.weight"].to(device, torch.float32)
    out_gate_w = weights["out_gate.weight"].to(device, torch.float32)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core contraction – Triton kernel
    # -----------------------------------------------------------------
    # Cast to FP16 for the kernel (Tensor‑cores)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # transpose right on the (k,j) axes so that the kernel can perform a
    # standard batched matmul: left[b,i,k,:] @ right_t[b,k,j,:]
    right_t = right.permute(0, 2, 1, 3).contiguous()

    out = trimul_einsum_triton(left, right_t)   # FP16 output
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float32)
    to_out_w = weights["to_out.weight"].to(device, torch.float32)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate          # broadcasting over last dim works
    out = F.linear(out, to_out_w) # final projection back to `dim`

    return out.to(torch.float32)
```
