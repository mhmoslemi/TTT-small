"""
TriMul “outgoing” kernel (AlphaFold3 style) with a fused Triton batched
matrix‑multiply that replaces the O(N³) einsum.
The heavy computation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is rewritten as a batch of N×N matrix multiplications per hidden channel:
    left_h  : (B, H, N, N)   = left.permute(0,3,1,2)
    right_T : (B, H, N, N)   = right.permute(0,3,2,1)   # transpose k‑j
    out_h    = left_h @ right_T      (B*H independent matmuls)

The Triton kernel processes each (batch, hidden) pair as a separate
“batch” dimension, using FP16 loads and FP32 accumulation.
All remaining ops (LayerNorm, Linear, gating) stay in PyTorch for
readability and to keep the implementation compact.

The final output is returned in FP32 with the same shape as the input:
    [B, N, N, dim].
"""

from typing import Tuple, Dict, Any

    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched FP16 mat‑mul with FP32 accumulation."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # pointers to the first tile of the current block
    a_tile_ptr = a_base + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_tile_ptr = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # mask out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + cur_k < K)
        mask_b = (tl.arange(0, BLOCK_K)[:, None] + cur_k < K) & (offs_n[None, :] < N)

        a = tl.load(a_tile_ptr, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_tile_ptr, mask=mask_b, other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

        a_tile_ptr += BLOCK_K * stride_a_k
        b_tile_ptr += BLOCK_K * stride_b_k

    # write back
    c = acc.to(tl.float16)
    c_tile_ptr = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_tile_ptr, c, mask=mask_c)


# ---------------------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input  : [B, N, N, dim]   (float32 on CUDA)
        - mask   : [B, N, N]        (float32/ bool, already on CUDA)
        - weights: dict of torch tensors
        - config : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # == c_z in the paper

    # -----------------------------------------------------------------------
    # layer‑norm on the input (dim‑wise)
    # -----------------------------------------------------------------------
    x = F.layer_norm(x,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # -----------------------------------------------------------------------
    # linear projections (no bias)
    # -----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------------
    # gating (sigmoid) and gating‑application
    # -----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # prepare tensors for the Triton mat‑mul:
    #   left  -> (B, hidden, N, N)
    #   right -> (B, hidden, N, N)   with the last two axes transposed
    # -----------------------------------------------------------------------
    # (B,N,N,hidden) -> (B,hidden,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # transpose the inner (k,j) axes of right before flattening
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # cast to FP16 for the kernel (memory‑bandwidth friendly)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    B, H, N, _ = left_fp16.shape          # H == hidden_dim

    # flatten batch‑and‑hidden into a single “batch” dimension for the kernel
    left_flat = left_fp16.view(B * H, N, N)
    right_flat = right_fp16.view(B * H, N, N)

    # allocate output tensor (FP16)
    out_flat = torch.empty_like(left_flat)

    # -----------------------------------------------------------------------
    # strides for the kernel (row‑major contiguous tensors)
    # -----------------------------------------------------------------------
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()   # (M*K, K, 1)
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()  # (K*N, N, 1)
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # -----------------------------------------------------------------------
    # launch the Triton kernel
    # -----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
