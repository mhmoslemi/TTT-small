"""
TriMul “outgoing” implementation with a fused Triton kernel.

The heavy part of TriMul is the 3‑D contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for every batch *head* pair.  This is exactly a batched matrix‑multiply
(left @ rightᵀ) for each hidden dimension *h*.
The kernel below computes a block‑wise GEMM for every (batch, hidden)
pair, while the rest of the module (LayerNorms, linear projections,
sigmoids, and the final linear) is performed with PyTorch.

The kernel works with FP16 inputs and accumulates in FP32, returning
FP16 results that are later cast back to FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel: batch‑wise GEMM for one hidden dimension (head)
# -------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh,   # left  [b, i, k, h]
    stride_rb, stride_rj, stride_rk, stride_rh,   # right [b, j, k, h]
    stride_ob, stride_oi, stride_oj, stride_oh,   # out   [b, i, j, h]
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a single GEMM tile:
        out[b, i:i+M, j:j+N, h] += left[b, i:i+M, k:k+K, h] @ right[b, j:j+N, k:k+K, h].T
    The kernel is launched with a 3‑D grid:
        (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    where the third dimension encodes (batch, head).
    """
    pid_z = tl.program_id(2)          # encodes batch and head
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile

    # decode batch and head from pid_z
    batch = pid_z // H
    head  = pid_z %  H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i dimension
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j dimension
    offs_k = tl.arange(0, BLOCK_K)                    # reduction dimension

    # masks for out‑of‑bounds accesses
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # pointer arithmetic (convert element offsets to byte offsets)
    left_base  = left_ptr  + batch * stride_lb + head * stride_lh
    right_base = right_ptr + batch * stride_rb + head * stride_rh
    out_base   = out_ptr   + batch * stride_ob + head * stride_oh

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k‑offsets for this iteration
        k_offs = k * BLOCK_K + offs_k
        # masks for this k‑slice
        cur_mask_k = k_offs < N

        # -----------------------------------------------------------------
        # left tile: shape (M, K)  ->  stride_li (i) , stride_lk (k)
        # -----------------------------------------------------------------
        left_ptrs = left_base \
                    + (offs_m[:, None] * stride_li) \
                    + (k_offs[None, :] * stride_lk)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)

        # -----------------------------------------------------------------
        # right tile: need (K, N) because we will multiply left (M,K) @ rightᵀ (K,N)
        # right is stored as (b, j, k, h).  We load a (K,N) tile by swapping the axes.
        # -----------------------------------------------------------------
        right_ptrs = right_base \
                     + (k_offs[:, None] * stride_rk) \
                     + (offs_n[None, :] * stride_rj)
        right = tl.load(right_ptrs, mask=cur_mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # GEMM accum
        acc += tl.dot(left, right)

    # -----------------------------------------------------------------
    # write‑back
    # -----------------------------------------------------------------
    out_ptrs = out_base \
               + (offs_m[:, None] * stride_oi) \
               + (offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------
# Helper: apply LayerNorm without creating a Module
# -------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C] , weight/bias : C
    return F.layer_norm(
        x,
        normalized_shape=(x.shape[-1],),
        weight=weight,
        bias=bias,
        eps=eps,
    )


# -------------------------------------------------------------
# Entry point required by the evaluator
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a fused Triton kernel
    for the O(N³) contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = torch.float32

    B, N, _, dim = input_tensor.shape          # input shape  [B, N, N, dim]
    hidden_dim = config["hidden_dim"]          # = H

    # -------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    # x : [B, N, N, dim]   ->   [B, N, N, hidden_dim]
    left  = F.linear(x, weights["left_proj.weight"])   # (hidden_dim, dim)
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Apply mask (if any)
    # -------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] -> [B, N, N, 1]
        mask = mask.unsqueeze(-1).to(dtype)
        left  = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4. Compute gating signals (sigmoid)
    # -------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Triton GEMM: out = left @ rightᵀ   (per hidden dimension)
    # -------------------------------------------------
    # Cast to fp16 for the kernel (memory & speed)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides in **elements**
    stride_lb, stride_li, stride_lk, stride_lh = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_fp16.stride()

    # block sizes (tuned for Hopper SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim                    # (batch, head) dimension
    )

    triton_trimul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # enough warps for the chosen block size on H200
    )

    # back to FP32 for the rest of the pipeline
    out = out_fp16.to(dtype)

    # -------------------------------------------------
    # 7. Post‑processing: LayerNorm, out‑gate, final linear
    # -------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                         # broadcast over hidden_dim
    # final linear: hidden_dim -> dim
    out = F.linear(out, weights["to_out.weight"])

    return out
