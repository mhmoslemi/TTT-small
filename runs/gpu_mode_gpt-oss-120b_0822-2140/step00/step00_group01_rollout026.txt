"""
TriMul “outgoing” implementation with a custom Triton batched‑GEMM kernel.

The heavy part of the forward pass is the Einstein‑summation
    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]

For each batch * hidden dimension* (B·H) this is a standard matrix
multiplication:
    out_d = left_d @ right_dᵀ
where left_d/right_d ∈ ℝ^{N×N}.  The kernel below computes many
independent GEMMs in parallel (one per (b, d) pair).  All other
operations (LayerNorm, linear projections, gates) are performed in
PyTorch for simplicity.

The kernel works on FP32 data and stores the result in FP32, which is
later reshaped back to [B, N, N, C] (C = dim).  Mixed‑precision could be
added, but FP32 is safe for the required accuracy.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent batches.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_ah, stride_am, stride_ak,   # strides for A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ: batch, row (k), col (j)
    stride_ch, stride_cm, stride_cn,   # strides for C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Compute which (batch, block_m, block_n) this program handles
    # --------------------------------------------------------------
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_batch = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    pid_m = pid_in_batch // tl.cdiv(N, BLOCK_N)
    pid_n = pid_in_batch % tl.cdiv(N, BLOCK_N)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicate for out‑of‑bounds (needed for the tail)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A pointer:   batch*stride_ah + row*stride_am + col*stride_ak
        a_ptrs = a_ptr + batch * stride_ah \
                       + offs_m[:, None] * stride_am \
                       + offs_k[None, :] * stride_ak
        # Bᵀ pointer: batch*stride_bh + row(k)*stride_bk + col(j)*stride_bn
        b_ptrs = b_ptr + batch * stride_bh \
                       + offs_k[:, None] * stride_bk \
                       + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product for the inner dimension
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the result
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch * stride_ch \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def batched_gemm(a: torch.Tensor, b_t: torch.Tensor, block_m=128, block_n=128, block_k=32):
    """
    a : (B*H, M, K)  FP32, contiguous
    b_t : (B*H, K, N) FP32, contiguous (already transposed)
    returns c : (B*H, M, N) FP32
    """
    assert a.is_contiguous() and b_t.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b_t.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    grid = (
        batch * triton.cdiv(M, block_m) * triton.cdiv(N, block_n),
    )

    batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b_t,
        c_ptr=c,
        M=M,
        N=N,
        K=K,
        stride_ah=a.stride(0),
        stride_am=a.stride(1),
        stride_ak=a.stride(2),
        stride_bh=b_t.stride(0),
        stride_bk=b_t.stride(1),
        stride_bn=b_t.stride(2),
        stride_ch=c.stride(0),
        stride_cm=c.stride(1),
        stride_cn=c.stride(2),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    pairwise update (the einsum).  The function receives:
        - input tensor   : [B, N, N, dim]   (float32)
        - mask tensor    : [B, N, N]        (float32 or bool)
        - weights dict   : model parameters
        - config dict    : {"dim": ..., "hidden_dim": ...}
    Returns:
        - output tensor  : [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Apply mask if present (mask is broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))      # [B, N, N, hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton‑accelerated einsum: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We reshape to (B*H, N, N) and compute many independent GEMMs.
    # ------------------------------------------------------------------
    # move hidden dimension to the batch axis
    left_ = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] -> reshape
    right_ = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_flat = right_.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # rightᵀ for each batch: shape (B*H, N, N)
    right_t = right_flat.transpose(-2, -1).contiguous()

    # batched GEMM (FP32)
    out_flat = batched_gemm(left_flat, right_t)    # (B*H, N, N)

    # reshape back to original layout: [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return out.to(torch.float32)   # guarantee fp32 output
