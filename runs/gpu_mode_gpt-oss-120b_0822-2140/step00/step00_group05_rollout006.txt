"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The heavy N³ tensor contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is realised as a batched matrix‑multiply:
    out[b,d] = left[b,d] @ right[b,d].T
where left/right ∈ ℝ^{B×H×N×N} (H = hidden_dim).

All cheap operations (LayerNorm, Linear, gating, mask) are done with
torch.nn.functional for simplicity.  The matrix‑multiply is off‑loaded to a
custom Triton kernel that works on the H200 (fp16 compute, fp32 accumulation)
and returns fp16 which is finally cast back to fp32.

The kernel processes one (batch, head) pair per program instance and tiles the
output with BLOCK_M×BLOCK_N tiles while streaming over K in BLOCK_K steps.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : out = A @ Bᵀ   (A,B : [M,N] float16, accumulate in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,                     # *float16  A  (M×N)
    b_ptr,                     # *float16  B  (M×N)   (will be read transposed)
    c_ptr,                     # *float16  C  (M×N)   output
    N,                         # int32    matrix size (square)
    stride_am, stride_ak,      # int32    strides for A (row, col)
    stride_bn, stride_bk,      # int32    strides for B (row, col) – we will swap them
    stride_cm, stride_cn,      # int32    strides for C (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # (batch * head) index

    # ------------------------------------------------------------------
    # block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers for the current (batch*head)
    a_ptr = a_ptr + pid_z * stride_am * N   # stride_am already includes the whole row‑stride, we just offset by whole matrix
    b_ptr = b_ptr + pid_z * stride_bn * N

    # ------------------------------------------------------------------
    # Load A panel  (M × K)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

    # ------------------------------------------------------------------
    # Load B panel transposed  (K × N)
    #   original layout is B[j,k]; we need Bᵀ[k,j] → swap row/col strides.
    # ------------------------------------------------------------------
    b_ptrs = b_ptr + (offs_k[:, None] * stride_bn) + (offs_n[None, :] * stride_bk)
    b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

    # ------------------------------------------------------------------
    # Compute block product
    # ------------------------------------------------------------------
    acc = tl.dot(a, b)                     # (M,N) in fp32

    # ------------------------------------------------------------------
    # Store result (convert back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point – called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor  [B, N, N, dim] (float32, on CUDA)
            mask   – torch.Tensor  [B, N, N]   (bool/int, on CUDA)
            weights – dict of model parameters (all float32, on CUDA)
            config – dict with keys:
                     dim, hidden_dim, nomask (bool)

    Returns:
        torch.Tensor  [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-5
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply (optional) mask
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).type_as(left)            # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy contraction – batched matmul via Triton (fp16 compute)
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B*H, N, N) where H = hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * hidden_dim, N, N)
    right = right.view(B * hidden_dim, N, N)

    out = torch.empty_like(left, dtype=torch.float16)

    # launch parameters – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    _tri_mul_kernel[
        grid
    ](
        left,
        right,
        out,
        N,
        left.stride(0),
        left.stride(2),          # column stride = 1
        right.stride(0),
        right.stride(2),
        out.stride(0),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm & linear
    # ------------------------------------------------------------------
    # LayerNorm on hidden dimension
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # Gating
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
