"""
TriMul “outgoing” implementation with a fused Triton kernel.

The heavy part of the module is the triple‑product
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].transpose(-2,-1)

The kernel below computes this operation for all batch‑hidden slices
(B * hidden_dim) in parallel, tiling the output matrix with
BLOCK_M × BLOCK_N tiles and looping over the K‑dimension in blocks of
size BLOCK_K.  The rest of the TriMul pipeline (LayerNorms, linear
projections and gates) are performed in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the triple‑product
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr,               # left  [B,N,N,H]
    b_ptr,               # right [B,N,N,H] (will be read as transposed on the last two axes)
    c_ptr,               # out   [B,N,N,H]
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program instance computes a BLOCK_M × BLOCK_N tile of the
    output matrix for a single (batch, hidden) pair.
    """
    pid = tl.program_id(0)                 # linear id for (batch, hidden)
    batch_id = pid // H
    h_id     = pid % H

    pid_m = tl.program_id(1)               # i‑tile
    pid_n = tl.program_id(2)               # j‑tile

    # Offsets in the i / j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the tile (k = 0)
    a_base = a_ptr + batch_id * stride_a_batch + h_id * stride_a_h
    b_base = b_ptr + batch_id * stride_b_batch + h_id * stride_b_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Compute actual k indices within this block
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile.  B is accessed as (j, k) because we need the
        # transposed view (right[j, k, d]).
        b_ptrs = b_base \
                 + offs_j[None, :] * stride_b_j \
                 + offs_k[:, None] * stride_b_k
        b = tl.load(b_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store the tile to C
    c_ptrs = c_ptr + batch_id * stride_c_batch + h_id * stride_c_h \
             + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]   (may be all‑ones)
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input: normalize over last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # left = x @ left_proj_weight.T
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)                # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projection, no bias)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: compute the triple‑product
    # ------------------------------------------------------------------
    out = torch.empty_like(left)               # shape [B,N,N,hidden], dtype float32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # program id 0  (batch * hidden)
        triton.cdiv(N, BLOCK_M),                # program id 1  (i‑tiles)
        triton.cdiv(N, BLOCK_N),                # program id 2  (j‑tiles)
    )

    tri_mul_kernel[grid](
        a_ptr=left.data_ptr(),
        b_ptr=right.data_ptr(),
        c_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_a_batch=left.stride(0),
        stride_a_i=left.stride(1),
        stride_a_k=left.stride(2),
        stride_a_h=left.stride(3),
        stride_b_batch=right.stride(0),
        stride_b_j=right.stride(1),   # note: we treat the first axis after batch as “j”
        stride_b_k=right.stride(2),
        stride_b_h=right.stride(3),
        stride_c_batch=out.stride(0),
        stride_c_i=out.stride(1),
        stride_c_j=out.stride(2),
        stride_c_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm + gate + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate shape: [B,N,N,hidden] after unsqueeze
    out = out * out_gate

    # final linear: hidden -> dim
    out = F.linear(out, weights["to_out.weight"])

    return out
