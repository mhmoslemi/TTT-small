"""
TriMul (outgoing) – fast Triton implementation.

The heavy part of TriMul is the cubic‑scaled einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We implement this reduction in a fused Triton kernel that computes a
batched GEMM for each hidden dimension `d`.  All LayerNorms, Linear
projections and gating are kept in PyTorch (they are cheap compared to the
N³ work), while the N‑by‑N multiply‑accumulate is off‑loaded to Triton.
The kernel works for any batch size B (1‑2), any sequence length N
(≤1024) and any hidden dimension H (≡ hidden_dim).  The final result is
returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out = Σ_k left[b,i,k,d] * right[b,j,k,d]   (d is hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  (b,i,k,d)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right (b,k,j,d)   (note j & k swapped for transposed view)
    stride_right_b, stride_right_k, stride_right_j, stride_right_d,
    # strides for out   (b,i,j,d)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)       # combined batch * hidden_dim
    pid1 = tl.program_id(1)       # tile over i
    pid2 = tl.program_id(2)       # tile over j

    # decode batch and hidden index
    b = pid0 // H
    d = pid0 % H

    # block offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer base for this (b,d)
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, d]   shape (M, K)
        left_ptrs = left_base \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[b, k, j, d]  shape (K, N)  (note k first, j second → transposed view)
        right_ptrs = right_base \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_base \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – Triton‑accelerated cubic reduction.
    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            mask         : torch.Tensor [B, N, N]      (float32 or bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : {"dim": C, "hidden_dim": H}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_view = mask.unsqueeze(-1)                 # [B,N,N,1]
        left  = left * mask_view
        right = right * mask_view

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the cubic reduction
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (accumulates in fp32)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty_like(left_fp16)  # shape [B,N,N,H]

    # Strides (convert to int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_d = right_fp16.stride()
    # right layout is (B, N, N, H) = (b, j, k, d) in the original tensor.
    # For the transposed view we need stride over k first then j:
    stride_right_k = stride_right_i          # k is the original second dimension
    stride_right_j = stride_right_j          # j is the original first dimension (same as i)

    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_fp16.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * hidden_dim, 1),               # pid0 dimension (batch*hidden)
        triton.cdiv(N, BLOCK_M),                     # i tiles
        triton.cdiv(N, BLOCK_N),                     # j tiles
    )

    _trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b,
        stride_right_k,
        stride_right_j,
        stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output layernorm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])    # [B,N,N,dim]

    return out
