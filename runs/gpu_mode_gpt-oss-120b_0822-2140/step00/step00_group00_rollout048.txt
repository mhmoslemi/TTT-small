"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the three‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an N×N matrix multiplication for every hidden channel *and* batch.
We fuse the batch and hidden dimensions (B·H) and run a high‑performance
batched GEMM kernel written in Triton.  All other operations (LayerNorm,
linear projections, gating, final LayerNorm/Linear) are kept in PyTorch
because they are cheap compared with the N³ reduction.

The Triton kernel is a classic tiled GEMM that processes a tile of size
[BLOCK_M, BLOCK_N] and reduces over K in blocks of size BLOCK_K.
It works for any batch size (B·H) and any sequence length N that fits in
GPU memory.  The kernel uses fp16 arithmetic for speed and the result is
cast back to fp32 before the final linear layer, matching the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,                     # dimensions
    stride_ab, stride_am, stride_ak,    # A strides (batch, row, col)
    stride_bb, stride_bk, stride_bn,    # B strides (batch, row, col)
    stride_cb, stride_cm, stride_cn,    # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A,B,C have shape (batch, M, K), (batch, K, N), (batch, M, N)."""
    pid = tl.program_id(0)

    # tiles per matrix
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # decode batch and tile indices
    batch_id = pid // (num_m * num_n)
    tile_id = pid % (num_m * num_n)
    pid_m = tile_id // num_n
    pid_n = tile_id % num_n

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointer offsets for the current tile
    a_ptrs = (
        A_ptr
        + batch_id * stride_ab
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + batch_id * stride_bb
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M)
            & (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < K)
            & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N) accumulate
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = (
        C_ptr
        + batch_id * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operation.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: Tensor [B, N, N, dim] (float32)
            - mask : Tensor [B, N, N] (float32 or bool)
            - weights: dict of model parameters (all float32)
            - config : dict with keys
                * dim, hidden_dim, nomask (bool)

    Returns:
        Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM:
    #   left  : [B, N, N, H]  ->  (B*H, N, N)
    #   right : [B, N, N, H]  ->  (B*H, N, N) with K dimension transposed
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dim next to batch
    left_t = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_t = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (k axis is last)

    batch_h = B * hidden_dim                                 # effective batch size

    A = left_t.view(batch_h, N, N).to(torch.float16)         # (B*H, N, N)
    Bmat = right_t.view(batch_h, N, N).to(torch.float16)    # (B*H, N, N)
    C = torch.empty_like(A, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides for the kernel (must be in element counts, not bytes)
    # ------------------------------------------------------------------
    stride_ab = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bb = Bmat.stride(0)
    stride_bk = Bmat.stride(1)
    stride_bn = Bmat.stride(2)

    stride_cb = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # ------------------------------------------------------------------
    # Launch tiled batched GEMM
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        A,
        Bmat,
        C,
        batch_h,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating, and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
