"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast Triton implementation.

The expensive part of the module is the 4‑D tensor einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension *d* this is a matrix multiplication
    L_d (N×K) @ R_dᵀ (K×N) → (N×N)

The kernel tiles the (i, j, d) output block (M×N×D) and iterates over
the reduction dimension *k* in blocks (size BLOCK_K).  All arithmetic is
performed in float32 and the final tensor is returned in FP32.

Only the reduction is done in Triton; the surrounding LayerNorms,
Linear projections and sigmoids stay in PyTorch (fast BLAS + fused
element‑wise ops).  This gives a huge speed‑up for the N³ term while
keeping the code short and easy to understand.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ------------------------------------------------------------
# Triton kernel
# ------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const float*  [B,N,N,D]   (i,k,d)
    right_ptr,         # const float*  [B,N,N,D]   (j,k,d)
    out_ptr,           # float*        [B,N,N,D]   (i,j,d)
    B: tl.constexpr,
    N: tl.constexpr,
    D: tl.constexpr,
    stride_left_b: tl.constexpr,
    stride_left_i: tl.constexpr,
    stride_left_k: tl.constexpr,
    stride_left_d: tl.constexpr,
    stride_right_b: tl.constexpr,
    stride_right_j: tl.constexpr,
    stride_right_k: tl.constexpr,
    stride_right_d: tl.constexpr,
    stride_out_b: tl.constexpr,
    stride_out_i: tl.constexpr,
    stride_out_j: tl.constexpr,
    stride_out_d: tl.constexpr,
    BLOCK_M: tl.constexpr,   # tile size for i
    BLOCK_N: tl.constexpr,   # tile size for j
    BLOCK_D: tl.constexpr,   # tile size for d (hidden_dim)
    BLOCK_K: tl.constexpr,   # tile size for reduction k
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(0)

    # map 1‑D program index to (b,i_block,j_block)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i_block = rem // num_j
    j_block = rem % num_j

    # offsets inside the tile
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator (M, N, D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left[b,i,k,d] -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )   # (M, K, D)

        # ----- load right[b,j,k,d] and transpose the k dimension -----
        # we need right_T[k,j,d] so that later we can do dot( M×K , K×N )
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None, None] * stride_right_k
            + offs_j[None, :, None] * stride_right_j
            + offs_d[None, None, :] * stride_right_d
        )
        right_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )   # (K, N, D)

        # ----- accumulate for each d in the tile -----
        for d in range(0, BLOCK_D):
            l = left[:, :, d]          # (M, K)
            r = right_T[:, :, d]       # (K, N)
            acc[:, :, d] += tl.dot(l, r)   # (M, N)

    # ----- write back -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )

# ------------------------------------------------------------
# Helper: launch wrapper
# ------------------------------------------------------------
def launch_trimul_kernel(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, D)  float32
    returns out : (B, N, N, D)  float32
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, D = left.shape

    out = torch.empty_like(left)

    # strides in element units
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # tiling parameters – tuned for Hopper SM90 (128 KB shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_D = 32          # hidden_dim is a multiple of 32 in all tests
    BLOCK_K = 32

    grid = (B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=D,
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_d=s_left[3],
        stride_right_b=s_right[0],
        stride_right_j=s_right[1],   # j is the second spatial dim
        stride_right_k=s_right[2],
        stride_right_d=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_d=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )
    return out

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: (input, mask, weights, config)
            input  – Tensor[B, N, N, dim]   (float32 on CUDA)
            mask   – Tensor[B, N, N]        (bool/float, broadcastable)
            weights – dict with all linear / norm parameters
            config  – {"dim": dim, "hidden_dim": hidden_dim, "nomask": bool}
    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # The O(N³) Tensor contraction – Triton kernel
    # ------------------------------------------------------------------
    # left/right are (B,N,N,hidden_dim).  The kernel returns out of same shape.
    out = launch_trimul_kernel(left, right)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out.float()
