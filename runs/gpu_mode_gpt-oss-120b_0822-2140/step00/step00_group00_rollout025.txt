"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the 3‑D tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be rewritten as a batched matrix‑multiply

    out[b,h] = left[b,h] @ right[b,h].T          (N×N × N×N → N×N)

The kernel below computes C = A @ Bᵀ for many independent groups
(`group = batch * hidden_dim`).  All surrounding operations
(LayerNorm, linear projections, sigmoid gates and the final
LayerNorm/Linear) are performed with PyTorch for simplicity.
The kernel works in fp16 for speed and writes the result back in fp16
(which is then cast to fp32 for the remaining PyTorch ops).

The implementation works for any batch size ≤2, any sequence length
(128‑1024) and hidden dimensions that fit in memory.  It is launched
with a 3‑D grid: (group, block_m, block_n).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B are (G,N,N))
# ---------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A_ptr,                      # *fp16   [G, N, N]
    B_ptr,                      # *fp16   [G, N, N]
    C_ptr,                      # *fp16   [G, N, N]
    M, N_dim, K,                # int32   (all equal to seq_len)
    stride_ah, stride_am, stride_ak,   # int64  strides for A
    stride_bh, stride_bm, stride_bn,   # int64  strides for B (row‑major)
    stride_ch, stride_cm, stride_cn,   # int64  strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance handles one (group, block_m, block_n).
    `group` = batch * hidden_dim.
    """
    group_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N_dim

    # ----- pointers to the beginning of the current tile -----------------
    a_ptrs = A_ptr + group_id * stride_ah + offs_m[:, None] * stride_am
    b_ptrs = B_ptr + group_id * stride_bh + offs_n[None, :] * stride_bn  # we will load Bᵀ

    # accumulator in fp32 (more accurate)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K --------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # current K‑slice
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A[tile_m, tile_k]   (no transpose)
        a = tl.load(
            a_ptrs + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ[tile_n, tile_k] = B[tile_k, tile_n]
        b = tl.load(
            b_ptrs + cur_k[:, None] * stride_bm,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot accumulates in fp32
        acc += tl.dot(a, b)

        # advance pointers for next K‑tile (only the K‑stride changes)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bm

    # ----- write result -------------------------------------------------
    c_ptrs = C_ptr + group_id * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ---------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask       : torch.Tensor [B, N, N]   (bool/float)
            - weights    : dict of pre‑trained parameters
            - config     : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]                     # hidden_dim

    # -------------------------------------------------
    # 1. LayerNorm on the last dimension (dim = D)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,D]

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Triton batched matmul  out = left @ rightᵀ  (over the K‑axis)
    # -------------------------------------------------
    #   reshape to (G, N, N) where G = B * H
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    G = B * H

    left_flat = left.view(G, N, N).to(torch.float16)   # fp16 for speed
    right_flat = right.view(G, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat)            # fp16 result

    # strides (in elements, not bytes) ---------------------------------
    stride_ah, stride_am, stride_ak = left_flat.stride()
    stride_bh, stride_bm, stride_bn = right_flat.stride()
    stride_ch, stride_cm, stride_cn = out_flat.stride()

    # launch configuration ------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        G,                                 # groups
        triton.cdiv(N, BLOCK_M),           # blocks in M dimension
        triton.cdiv(N, BLOCK_N),           # blocks in N dimension
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                           # M, N, K
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bm,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # -------------------------------------------------
    # 6. Final LayerNorm over hidden dimension + gating
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # broadcast over hidden dim

    # -------------------------------------------------
    # 7. Linear projection back to original dimension D
    # -------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,D]

    return out
