"""
TriMultiplicative Update (outgoing) – Triton‐accelerated core.

The heavy part of the TriMul forward pass is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an N×N × N matrix multiplication for every hidden channel *d*.
The kernel below computes, for each (b,i,j) pair, a whole vector of
`hidden_dim` results in one thread, reducing over the *k* dimension in
BLOCK_K sized tiles.  The rest of the module (LayerNorms, linear
projections, gates, final LayerNorm+Linear) remains in PyTorch – this
keeps the code simple while still off‑loading the O(N³·C) work to
Triton.

The kernel works for any batch size B∈{1,2}, sequence length N up to
1024 and hidden_dim that is a multiple of (or smaller than) BLOCK_H
(we mask the tail).  Input tensors are expected to be already on the
CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes the 3‑D “tri‑mul” contraction.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # const float16*   [B,N,N,H]
    right_ptr,           # const float16*   [B,N,N,H]
    out_ptr,             #       float16*   [B,N,N,H]

    B, N, H,             # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides

    BLOCK_K: tl.constexpr,   # tile size along reduction dimension K
    BLOCK_H: tl.constexpr,   # tile size along hidden dimension H
):
    """
    Each program instance processes a single (b,i,j) triplet and produces
    a vector of length H (in BLOCK_H chunks) that is the sum over k.
    """

    # ------------------------------------------------------------------
    # 1) Decode the 1‑D program id into (b,i,j)
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    total = B * N * N
    active = pid < total
    if not active:
        return

    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # ------------------------------------------------------------------
    # 2) Offsets for the hidden dimension (processed in blocks)
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # accumulator for the hidden block
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over K in tiles, load left & right blocks, accumulate
    # ------------------------------------------------------------------
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_k = cur_k < N

        # ---- left[b,i,k,:]  shape (BLOCK_K, BLOCK_H) ----
        left_off = (
            b * stride_l_b +
            i * stride_l_i +
            cur_k[:, None] * stride_l_k +
            offs_h[None, :] * stride_l_h
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float32)

        # ---- right[b,j,k,:] shape (BLOCK_K, BLOCK_H) ----
        right_off = (
            b * stride_r_b +
            j * stride_r_j +
            cur_k[:, None] * stride_r_k +
            offs_h[None, :] * stride_r_h
        )
        right = tl.load(
            right_ptr + right_off,
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float32)

        # element‑wise product + reduction over k
        prod = left * right                     # (BLOCK_K, BLOCK_H)
        acc += tl.sum(prod, axis=0)             # (BLOCK_H,)

        k += BLOCK_K

    # ------------------------------------------------------------------
    # 4) Write the hidden block back
    # ------------------------------------------------------------------
    out_off = (
        b * stride_o_b +
        i * stride_o_i +
        j * stride_o_j +
        offs_h * stride_o_h
    )
    tl.store(out_ptr + out_off, acc.to(tl.float16), mask=mask_h)


# ----------------------------------------------------------------------
# Python entry point – combines PyTorch pre/post‑processing with the kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : Tuple[
        input      : torch.Tensor   shape [B, N, N, dim]      (float32)
        mask       : torch.Tensor   shape [B, N, N]           (float32 / bool)
        weights    : dict of torch.Tensor (float32) – model parameters
        config     : dict with keys "dim", "hidden_dim"
    ]

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = H in the paper

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (no bias in the reference code)
    # ------------------------------------------------------------------
    eps = 1e-5
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free) – keep in float32 for precision
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and gating of the projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to float16 before Triton (half‑precision is fast on H200)
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape               # H == hidden_dim

    # allocate output for the contraction
    out_f16 = torch.empty((B, N, N, hidden_dim),
                          dtype=torch.float16,
                          device=left_f16.device)

    # ------------------------------------------------------------------
    # 6) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_K = 32          # tile size for reduction over k
    BLOCK_H = 16          # tile size for hidden dimension (must be <= hidden_dim)

    total_threads = B * N * N
    grid = (triton.cdiv(total_threads, 1),)   # 1‑D grid, one thread per (b,i,j)

    # strides for each tensor (int64 → Python int is fine)
    s_l = left_f16.stride()   # (Bstride, iStride, kStride, hStride)
    s_r = right_f16.stride()
    s_o = out_f16.stride()

    trimul_outgoing_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B, N, hidden_dim,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    out = out_f16.to(torch.float32)           # back to fp32 for the rest

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating, and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
