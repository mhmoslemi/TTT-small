"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

  * LayerNorm on the 4‑D input.
  * Linear projections + sigmoid gates.
  * Core O(N³·C) operation:
        out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    which is a batched matrix‑multiply of shape
        (B·C, N, N)  @  (B·C, N, N)ᵀ .
  * The heavy mat‑mul is executed in Triton (block‑wise, FP32 accumulator,
    FP16 inputs for bandwidth).  All other ops stay in PyTorch.

The kernel works for any batch size (1‑2), sequence lengths up to 1024 and
hidden dimensions up to 384 (the reference hidden_dim is 128).  It returns a
float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel – batched mat‑mul left @ rightᵀ  (left/right: (B·C, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # *float16   (B*C, N, N)
    right_ptr,                    # *float16   (B*C, N, N)   (will be read transposed)
    out_ptr,                      # *float32   (B*C, N, N)
    BATCH_H, N,                   # runtime constants
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile col
    pid_bh = tl.program_id(2)              # (batch * hidden) id

    # --------------------------------------------------------------
    # Compute offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # batch‑hidden index
    bh = pid_bh
    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + bh * stride_left_batch
    right_base = right_ptr + bh * stride_right_batch
    out_base   = out_ptr   + bh * stride_out_batch

    # accumulator in FP32 (more numerically stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  tile  (M x K)
        left_ptrs = (left_base
                     + offs_m[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right[j,k] needs to be read transposed → we load right[k,j] (K x N)
        right_ptrs = (right_base
                      + offs_k[:, None] * stride_right_i   # rows = k
                      + offs_n[None, :] * stride_right_k)  # cols = j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot‑product over the K dimension
        acc += tl.dot(left, right)          # (M,N)

    # write result
    out_ptrs = (out_base
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim]   (float32)
        mask    – Tensor[B, N, N]        (bool / 0‑1 float)
        weights – dict with all linear / layer‑norm parameters
        config  – dict with ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor[B, N, N, dim]   (float32)
    """
    device = torch.device("cuda")
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    input_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep them in FP16 for bandwidth
    # ------------------------------------------------------------------
    # shape: (..., hidden)
    left  = F.linear(input_norm, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(input_norm, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – stay in FP32 for stability, then cast back
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(input_norm, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(input_norm, weights["right_gate.weight"])).to(right.dtype)
    out_gate   = torch.sigmoid(F.linear(input_norm, weights["out_gate.weight"])).to(torch.float32)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·C) operation via Triton
    #    left, right : [B, N, N, hidden]  →  permute to [B, hidden, N, N]
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # flatten batch & hidden for the kernel
    left_flat  = left.view(B * hidden, N, N)
    right_flat = right.view(B * hidden, N, N)

    # output buffer (FP32 accumulator)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # strides for the flattened tensors (row‑major)
    stride_batch = N * N
    stride_i = N
    stride_k = 1

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden,                       # batch‑hidden dimension
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden, N,
        stride_batch, stride_i, stride_k,
        stride_batch, stride_i, stride_k,
        stride_batch, stride_i, stride_j := stride_k,   # out strides (same layout)
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.unsqueeze(-1)                         # broadcast over hidden
    out = F.linear(out, weights["to_out.weight"]).to(torch.float32)

    return out
