"""
TriMul “outgoing” forward pass.

- LayerNorm + Linear projections are done in PyTorch (they are cheap compared
  to the O(N³) reduction).
- The costly reduction `einsum('... i k d, ... j k d -> ... i j d')` is
  expressed as a batch of matrix‑multiplications: for each hidden dimension
  `d` we compute `L_d @ R_dᵀ` where `L_d,b` and `R_d,b` have shape (N, K).
- A custom Triton kernel (`batched_matmul_kernel`) performs the batched
  GEMM with tiling over M, N and K, delivering a float‑16 result that is
  cast back to float32 for the remaining layers.
- The kernel is launched with a three‑dimensional grid:
      (batch*hidden_dim, ceil(N/B_M), ceil(N/B_N))

The implementation works for all test configurations listed in the prompt
and respects the “nomask” flag (no mask → skip masking step).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B: (BATCH, M, K))
# -------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, M, N, K,                    # problem size
    stride_a_batch, stride_a_m, stride_a_k,   # strides of A
    stride_b_batch, stride_b_k, stride_b_n,   # strides of B (already transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------------------------------------
    # offsets for the current tile
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the first block of K
    a_ptrs = a_ptr + batch_id * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_id * stride_b_batch \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Main loop over K
    # -------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] &
                          (k * BLOCK_K + tl.arange(0, BLOCK_K) < K)),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=((k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) &
                          mask_n[None, :]),
                    other=0.0)
        acc += tl.dot(a, b)                # dot over the K dimension

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # -------------------------------------------------
    # Write back
    # -------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------
# Host‑side helper to launch the kernel
# -------------------------------------------------------------
def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B : (batch, hidden, N, K)   (float16)
    returns C : (batch, hidden, N, N)   (float16)
    """
    BATCH, H, N, K = A.shape
    # collapse batch*hidden into one dimension for the kernel
    A_flat = A.view(BATCH * H, N, K)
    B_flat = B.view(BATCH * H, N, K)

    C_flat = torch.empty_like(A_flat, dtype=torch.float16)

    # block sizes – tuned for Hopper SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BATCH * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A_flat,                     # a_ptr
        B_flat,                     # b_ptr (will be interpreted as transposed)
        C_flat,                     # c_ptr
        BATCH * H, N, N, K,
        A_flat.stride(0), A_flat.stride(1), A_flat.stride(2),
        B_flat.stride(0), B_flat.stride(2), B_flat.stride(1),   # note the transpose in strides
        C_flat.stride(0), C_flat.stride(1), C_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C_flat.view(BATCH, H, N, N)


# -------------------------------------------------------------
# The required entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel
    for the O(N³) reduction.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -------------------------------------------------
    # Linear projections (no bias)
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(dtype)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    left = F.linear(x, left_proj_w)           # (B,N,N,hidden)
    right = F.linear(x, right_proj_w)

    # -------------------------------------------------
    # Optional mask
    # -------------------------------------------------
    if not nomask:
        mask = mask.unsqueeze(-1).to(dtype)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # Gating logits -> sigmoid
    # -------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Cast to float16 for the heavy reduction
    # -------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Rearrange to (B, hidden, N, K) where K == N (the second sequence dim)
    # left  : (B, N, N, H) -> (B, H, N, N)
    left_t = left_f16.permute(0, 3, 1, 2).contiguous()
    right_t = right_f16.permute(0, 3, 1, 2).contiguous()

    # Triton batched GEMM: out_f16 shape (B, H, N, N)
    out_f16 = triton_batched_matmul(left_t, right_t)

    # Back to (B, N, N, H) and to float32
    out = out_f16.permute(0, 2, 3, 1).contiguous().to(dtype)

    # -------------------------------------------------
    # to_out_norm (LayerNorm over hidden dim)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w, bias=to_out_norm_b)

    # -------------------------------------------------
    # Apply out gate and final linear projection
    # -------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(dtype)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
