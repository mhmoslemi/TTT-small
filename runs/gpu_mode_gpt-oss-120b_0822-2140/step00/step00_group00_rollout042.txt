"""
TriMul "outgoing" implementation with a fused Triton kernel.

The heavy part of the module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication (left @ rightᵀ) for every
hidden dimension `d`.  We:
  * perform LayerNorms, Linear projections and the element‑wise gates in PyTorch,
  * cast the projected tensors to fp16,
  * permute them to (B, H, N, N) and flatten the first two dimensions,
  * run a custom Triton kernel that computes B·H independent N×N matrix
    multiplications (no explicit transpose needed – the kernel multiplies
    L[i,k] with R[j,k] over the reduction axis k),
  * cast the result back to fp32 and finish the remaining LayerNorm / Linear
    steps in PyTorch.

The kernel processes tiles of size BLOCK_M×BLOCK_N with a reduction tile
BLOCK_K and accumulates in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM (L @ Rᵀ) for many heads at once
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    L_ptr, R_ptr, O_ptr,                     # pointers
    BATCH_HEAD, N,                           # sizes
    stride_L_batch, stride_L_i, stride_L_k,  # L strides
    stride_R_batch, stride_R_j, stride_R_k,  # R strides
    stride_O_batch, stride_O_i, stride_O_j,  # O strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute O = L @ Rᵀ for each (batch, head) pair."""
    bh_id = tl.program_id(0)          # 0 … BATCH_HEAD-1
    pid_i = tl.program_id(1)          # block row
    pid_j = tl.program_id(2)          # block column

    # Offsets in the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers for the current batch/head
    L_base = L_ptr + bh_id * stride_L_batch
    R_base = R_ptr + bh_id * stride_R_batch
    O_base = O_ptr + bh_id * stride_O_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of L (i,k) and R (j,k)
        L = tl.load(
            L_base
            + (offs_i[:, None] * stride_L_i)
            + (offs_k[None, :] * stride_L_k),
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        R = tl.load(
            R_base
            + (offs_j[:, None] * stride_R_j)
            + (offs_k[None, :] * stride_R_k),
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Accumulate the dot product over k
        acc += tl.dot(L, R)   # (BLOCK_M, BLOCK_N)

    # Write back the result
    O = acc.to(tl.float16)
    tl.store(
        O_base
        + (offs_i[:, None] * stride_O_i)
        + (offs_j[None, :] * stride_O_j),
        O,
        mask=mask_i[:, None] & mask_j[None, :],
    )

# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask : torch.Tensor [B, N, N]    (float32/bool, on CUDA)
            - weights: dict of pre‑loaded weight tensors (float32, on CUDA)
            - config : dict with keys:
                * dim, hidden_dim (int)
                * nomask (bool) – if True the mask may be all‑ones

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    # ----------------------------------------------------------
    # unpack
    # ----------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # ----------------------------------------------------------
    # LayerNorm on the input (dim)
    # ----------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])        # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])      # [B,N,N,hidden]

    # ----------------------------------------------------------
    # Apply mask if present
    # ----------------------------------------------------------
    if not cfg.get("nomask", True):
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------
    # Gates (sigmoid)
    # ----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # Cast to fp16 for the heavy multiplication
    # ----------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ----------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the first two dims
    # ----------------------------------------------------------------
    #   shape after permute: (B, hidden_dim, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    # flatten batch & head
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)   # fp16 output

    # --------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------
    BLOCK = 64                # good trade‑off for Hopper SM90
    grid = (
        BH,                               # batch*head dimension
        triton.cdiv(N, BLOCK),            # i‑blocks
        triton.cdiv(N, BLOCK),            # j‑blocks
    )

    # strides for the flattened tensors
    stride_L_batch, stride_L_i, stride_L_k = left_flat.stride()
    stride_R_batch, stride_R_j, stride_R_k = right_flat.stride()
    stride_O_batch, stride_O_i, stride_O_j = out_flat.stride()

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_L_batch,
        stride_L_i,
        stride_L_k,
        stride_R_batch,
        stride_R_j,
        stride_R_k,
        stride_O_batch,
        stride_O_i,
        stride_O_j,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim) and continue in PyTorch
    # --------------------------------------------------------------
    out_fp32 = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # final LayerNorm over hidden dimension
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # gate and final linear projection back to dim
    out_norm = out_norm * out_gate
    out = F.linear(out_norm, weights["to_out.weight"])

    return out
