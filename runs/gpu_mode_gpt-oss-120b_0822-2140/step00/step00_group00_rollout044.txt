"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of matrix multiplications for every hidden dimension *d*:
    out_d = left_d @ right_dᵀ   (left_d,right_d : [B,N,N])

The implementation below
* applies the LayerNorm / Linear / gating steps in native PyTorch,
* casts the projected tensors to fp16 (the H200 works best in fp16),
* launches a Triton kernel that performs a blocked batched GEMM for all
  hidden channels in the same kernel (no extra synchronisation is required),
* converts the result back to fp32 and finishes the remaining LayerNorm / Linear
  steps in PyTorch.

The kernel processes a tile (BLOCK_M × BLOCK_N) of the output matrix for a
single batch element and a whole hidden‑channel tile (BLOCK_D).  Inside the
kernel we loop over the reduction axis *k* in blocks of size BLOCK_K, loading
tiles of ``left`` (shape M×K) and the *transposed* tile of ``right`` (shape
K×N).  For each sub‑channel we call ``tl.dot`` to accumulate the product.
All arithmetic is performed in fp32, while the loads are fp16 → fp32 casts.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # const fp16*
    right_ptr,         # const fp16*
    out_ptr,           # * fp16
    B, N, H,           # int32
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  """
    pid_b = tl.program_id(0)              # batch dimension
    pid_i = tl.program_id(1)              # output i‑tile
    pid_j = tl.program_id(2)              # output j‑tile

    batch = pid_b
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # loop over hidden‑channel tiles
    # ------------------------------------------------------------------
    for d_off in range(0, H, BLOCK_D):
        # ------------------------------------------------------------------
        # build index vectors for the current tile
        # ------------------------------------------------------------------
        offs_i = i_base + tl.arange(0, BLOCK_M)
        offs_j = j_base + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)
        offs_d = d_off + tl.arange(0, BLOCK_D)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_d = offs_d < H

        # accumulator for this (i,j,d) tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # reduction over k
        # ------------------------------------------------------------------
        for k_off in range(0, N, BLOCK_K):
            cur_k = k_off + offs_k
            mask_k = cur_k < N

            # ------------------------------------------------------------------
            # load left tile   shape (M, K, D)   -> indices (i, k, d)
            # ------------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + batch * stride_lb
                + offs_i[:, None, None] * stride_li
                + cur_k[None, :, None] * stride_lk
                + offs_d[None, None, :] * stride_ld
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            ).to(tl.float32)

            # ------------------------------------------------------------------
            # load right tile transposed to (K, N, D)
            #   we need right[b, j, k, d] → we load as (k, j, d)
            # ------------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + batch * stride_rb
                + cur_k[:, None, None] * stride_rk   # k dimension becomes first
                + offs_j[None, :, None] * stride_rj   # j dimension becomes second
                + offs_d[None, None, :] * stride_rd
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                other=0.0,
            ).to(tl.float32)

            # ------------------------------------------------------------------
            # per‑channel dot product: (M,K) @ (K,N) → (M,N)
            # ------------------------------------------------------------------
            for dd in range(0, BLOCK_D):
                # slices have shape (M,K) and (K,N)
                l = left[:, :, dd]
                r = right[:, :, dd]
                acc[:, :, dd] += tl.dot(l, r)   # tl.dot does (M,K)·(K,N)

        # ------------------------------------------------------------------
        # store the computed tile (respecting bounds)
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + batch * stride_ob
            + offs_i[:, None, None] * stride_oi
            + offs_j[None, :, None] * stride_oj
            + offs_d[None, None, :] * stride_od
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
        )


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]       (bool / float)
            - weights: dict with all linear / layer‑norm parameters
            - config : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    device = inp.device

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    eps = 1e-5
    # compute mean / var manually to avoid a second kernel
    mean = inp.mean(-1, keepdim=True)
    var = ((inp - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (inp - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"][None, None, None, :] + weights["norm.bias"][None, None, None, :]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Mask handling (mask may be omitted)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy matmul, allocate output buffer
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    out_f16 = torch.empty_like(left_f16, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Prepare strides (int64 → python int)
    # ------------------------------------------------------------------
    stride_lb, stride_li, stride_lk, stride_ld = left_f16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_f16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_f16.stride()

    # ------------------------------------------------------------------
    # Triton launch parameters (tuned for Hopper)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 8                     # hidden_dim is a multiple of 8 in all test configs

    grid = (B, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    trimul_einsum_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B, N, hidden,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # Convert back to fp32
    # ------------------------------------------------------------------
    out = out_f16.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden) -> Linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
