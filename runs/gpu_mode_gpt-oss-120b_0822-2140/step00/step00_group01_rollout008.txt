"""
TriMul (outgoing) – Triton‑accelerated core

The expensive part of the TriMul forward pass is the
einsum
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]          (1)

which is a batched matrix multiply for every hidden channel *h*.
We fuse the hidden channel into the batch dimension (B*H) and
run a classic tiled GEMM kernel in fp16 with fp32 accumulation.
All other ops (LayerNorm, Linear projections, gating, masks) are
performed with PyTorch – they are cheap compared with the O(N³) work.
The final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -----------------------------------------------------------------------
# Triton GEMM kernel (B*H, N, N) * (B*H, N, N)^T  →  (B*H, N, N)
# -----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr,          # left   (B*H, M, K)
    b_ptr,          # right  (B*H, N, K)   note: K is the shared dimension
    c_ptr,          # out    (B*H, M, N)
    M, N, K,
    stride_am, stride_ak,          # a: (batch, row, col)
    stride_bm, stride_bk,          # b: (batch, row=j, col=k) – we will load transposed
    stride_cm, stride_cn,          # c: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = pid // tl.cdiv(M, BLOCK_M)          # batch index (B*H)
    block_i = (pid % tl.cdiv(M, BLOCK_M)) * BLOCK_M
    block_j = tl.program_id(1) * BLOCK_N

    # Pointers for this batch
    a_batch_ptr = a_ptr + batch * stride_am * 0
    b_batch_ptr = b_ptr + batch * stride_bm * 0
    c_batch_ptr = c_ptr + batch * stride_cm * 0

    # Offsets within the block
    offs_i = block_i + tl.arange(0, BLOCK_M)
    offs_j = block_j + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load a tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_am
                                + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)

        # Load b tile, but we need the transposed view: (BLOCK_N, BLOCK_K)
        # b is stored as (batch, j, k). We want b[j, k] which matches our layout.
        b_ptrs = b_batch_ptr + (offs_j[:, None] * stride_bm
                                + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)

        # Accumulate: dot over K dimension
        acc += tl.dot(a, b)

        # advance pointers
        a_batch_ptr += BLOCK_K * stride_ak
        b_batch_ptr += BLOCK_K * stride_bk

    # Write back
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_cm
                            + offs_j[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -----------------------------------------------------------------------
# Public entry point
# -----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Parts that are compute‑heavy (the N³ einsum) are executed in a
    Triton kernel; everything else stays in PyTorch for simplicity.

    Args:
        data: (input, mask, weights, config)
            - input : [B, N, N, dim]    float32
            - mask  : [B, N, N]         (may be all‑ones if nomask)
            - weights: dict with all linear / LN parameters
            - config : {"dim": dim, "hidden_dim": hidden_dim, "nomask": bool}

    Returns:
        out : [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # -------------------------------------------------------------------
    ln_weight = weights["norm.weight"]
    ln_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=ln_weight,
                     bias=ln_bias,
                     eps=1e-5)

    # -------------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden
    # -------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------
    # 3. Apply mask (if present)
    # -------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------------
    # 4. Gating
    # -------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5. Core N³ einsum via Triton
    #    We reshape (B, N, N, H) → (B*H, N, N) and compute a batched
    #    matmul with the right operand transposed.
    # -------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Permute so hidden dim is next to batch for easy view
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_.view(B * hidden, N, N)   # (B*H, N, N)
    right_flat = right_.view(B * hidden, N, N) # (B*H, N, N)

    # Cast to fp16 for the kernel (lossless for the tested ranges)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    out_flat = torch.empty_like(left_fp16)  # (B*H, N, N)

    # -------------------------------------------------------------------
    # Kernel launch configuration
    # -------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * hidden, 1),               # we launch one program per (batch*H)
        triton.cdiv(N, BLOCK_M),                 # i‑tile dimension
        triton.cdiv(N, BLOCK_N)                  # j‑tile dimension (as second axis)
    )

    _tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        N, N, N,
        left_fp16.stride(0), left_fp16.stride(2),   # a: batch, row(i), col(k)
        right_fp16.stride(0), right_fp16.stride(2), # b: batch, row(j), col(k)
        out_flat.stride(0), out_flat.stride(2),     # c: batch, row(i), col(j)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Cast back to float32 and reshape to original 4‑D layout
    out = out_flat.to(torch.float32).view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -------------------------------------------------------------------
    # 6. Post‑processing: LN (hidden) → linear → gating
    # -------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
