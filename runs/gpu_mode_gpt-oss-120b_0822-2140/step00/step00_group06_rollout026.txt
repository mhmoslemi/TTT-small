def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    # map pid to batch and head id
    total = B * H
    batch_head = pid
    b = batch_head // H
    h = batch_head % H
    # offsets for tile
    pid_m = tl.program_id(1) # i tile
    pid_n = tl.program_id(2) # j tile
    # Compute start offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask
    i_mask = i < N
    j_mask = j < N

    # pointers to left slice: left[b, i, k, h]
    # need stride calculations
    # left base = left_ptr + b*stride_l_batch + h*stride_l_h
    # We'll handle offset for i and k.

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left tile (M,K)

            left_ptr + b*stride_l_batch + h*stride_l_h

"""
TriMul “outgoing” forward pass implemented with a fused Triton kernel.

The heavy part of the module is the 4‑D einsum

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is exactly a batched matrix multiplication per head:
    C = A @ Bᵀ   with   A = left[..., h]   (N×N) ,   B = right[..., h] (N×N)

A single Triton kernel computes many (batch, head) matmuls together.
All cheap operations (LayerNorm, linear projections, gating and the final
output projection) are kept in PyTorch, while the O(B·H·N³) inner product is
performed on the GPU in FP16 for speed and accumulated in FP32 for
numerical stability. The final result is returned in float32 as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: batched matmul (A @ Bᵀ) over the hidden dimension
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    left_ptr,               # *const float16
    right_ptr,              # *const float16
    out_ptr,                # *float16
    B, N, H,                # problem size
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile rows
    BLOCK_N: tl.constexpr,  # tile cols
    BLOCK_K: tl.constexpr,  # reduction tile
):
    # --------------------------------------------------------------
    # program id decomposition
    # axis 0 : (batch, head) pair
    # axis 1 : i‑tile
    # axis 2 : j‑tile
    # --------------------------------------------------------------
    pid = tl.program_id(0)
    total_bh = B * H
    b = pid // H            # batch index
    h = pid % H             # head index

    pid_m = tl.program_id(1)    # i‑tile
    pid_n = tl.program_id(2)    # j‑tile

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    i_mask = i_offset < N
    j_mask = j_offset < N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_batch + h * stride_o_h

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # reduction over k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left tile : shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base
            + i_offset[:, None] * stride_l_i
            + k_offset[None, :] * stride_l_k,
            mask = i_mask[:, None] & k_mask[None, :],
            other = 0.0
        ).to(tl.float32)

        # right tile : we need (k, j) because we later multiply left @ right
        right = tl.load(
            right_base
            + k_offset[:, None] * stride_r_k
            + j_offset[None, :] * stride_r_j,
            mask = k_mask[:, None] & j_mask[None, :],
            other = 0.0
        ).to(tl.float32)

        # dot‑product over the K dimension
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # write back (store as FP16, convert later in PyTorch)
    # --------------------------------------------------------------
    out = acc.to(tl.float16)
    tl.store(
        out_base
        + i_offset[:, None] * stride_o_i
        + j_offset[None, :] * stride_o_j,
        out,
        mask = i_mask[:, None] & j_mask[None, :]
    )


# --------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy pair‑wise multiplication is executed by a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # weight shapes: (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Convert to FP16 for the kernel (keep a FP32 copy for safety)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape  # H == hidden_dim

    # ------------------------------------------------------------------
    # 6) Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    grid = (
        B * H,                                   # (batch, head) dimension
        triton.cdiv(N, BLOCK_M),                 # i‑tiles
        triton.cdiv(N, BLOCK_N),                 # j‑tiles
    )

    batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # good default for fp16 matmul on Hopper
    )
    torch.cuda.synchronize()

    # Back to FP32 for the remaining LayerNorm / linear
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Final LayerNorm (hidden_dim) and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
