"""
TriMul “outgoing” implementation – Triton accelerated einsum.

The heavy part of TriMul is the 3‑D contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is equivalent to, for every batch element *b* and hidden channel *d*,
a matrix multiplication L_{b,d} @ R_{b,d}ᵀ where L,R ∈ ℝ^{N×N}.
We fuse the batch and hidden dimensions (B·H independent GEMMs) and
execute the GEMM with a custom Triton kernel.  All other operations
(LayerNorm, Linear, gating, final LayerNorm/Linear) are done in PyTorch.

The kernel works on fp16 inputs, accumulates in fp32 and writes fp16
results that are later cast back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : (batch, M, K) ,  C : (batch, M, N) )
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    A_ptr,               # *fp16   (batch, M, K)
    B_ptr,               # *fp16   (batch, N, K)   (note: we treat K as row dim)
    C_ptr,               # *fp16   (batch, M, N)
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)          # which (batch*hidden) matrix
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile col

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # -------------------- pointers --------------------
    a_ptrs = (A_ptr + pid_batch * stride_a_batch
              + offs_m[:, None] * stride_a_row
              + offs_k[None, :] * stride_a_col)

    b_ptrs = (B_ptr + pid_batch * stride_b_batch
              + offs_k[:, None] * stride_b_row   # k as *row* in B
              + offs_n[None, :] * stride_b_col)  # n as *col* in B

    # -------------------- accumulation --------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot: (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row

    # -------------------- store result --------------------
    c_ptrs = (C_ptr + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_row
              + offs_n[None, :] * stride_c_col)

    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
              input  – torch.Tensor [B, N, N, dim]   (float32)
              mask   – torch.Tensor [B, N, N]       (float32 or bool)
              weights – dict of torch.Tensor (all float32)
              config – dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------- LayerNorm on last dim --------------------
    eps = 1e-5
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # -------------------- Linear projections --------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------- optional mask --------------------
    if not nomask:
        mask = mask.unsqueeze(-1)                     # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -------------------- gates (sigmoid) --------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM:
    #   left, right : (B, N, N, H)  →  (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    A = left_perm.view(B * hidden_dim, N, N).to(torch.float16)
    Bmat = right_perm.view(B * hidden_dim, N, N).to(torch.float16)

    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # -------------------- strides --------------------
    stride_a_batch, stride_a_row, stride_a_col = A.stride()
    stride_b_batch, stride_b_row, stride_b_col = Bmat.stride()
    stride_c_batch, stride_c_row, stride_c_col = C.stride()

    # -------------------- launch Triton kernel --------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim                    # batch*hidden
    )
    _batched_matmul_transpose_kernel[grid](
        A, Bmat, C,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -------------------- reshape back --------------------
    out = C.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # -------------------- final LayerNorm + out_gate --------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate

    # -------------------- final Linear (hidden -> dim) --------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
