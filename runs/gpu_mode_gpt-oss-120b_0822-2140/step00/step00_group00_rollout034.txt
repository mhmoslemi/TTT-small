"""
TriMul “outgoing” implementation in Triton.

The heavy part of the module is the following einsum
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be seen as a batch of H independent matrix‑multiplications:
    out_h = left_h @ right_h.T      (left_h,right_h : (B,N,N))

The kernel below performs this batched GEMM with three‑dimensional tiling
(M‑tiles × N‑tiles × K‑tiles).  The surrounding LayerNorm / Linear /
sigmoid‑gating are kept in PyTorch – they are cheap compared with the
O(B·H·N³) work of the contraction.

The kernel works for any
    B ∈ {1,2}, N ∈ {128,…,1024}, dim ∈ {128,384}, hidden_dim = 128
and for both masked / un‑masked inputs.
All tensors are expected to be on CUDA already.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batch‑wise matrix multiplication for each hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N, H,                                 # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid:
        program_id(0) -> combined batch * hidden index (bh)
        program_id(1) -> row tile (i)
        program_id(2) -> column tile (j)
    """
    bh_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # decode batch (b) and hidden (h) from the first dimension
    # --------------------------------------------------------------
    b = bh_id // H
    h = bh_id % H

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # base pointers for this (b,h) pair
    # --------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = (left_base
                     + offs_m[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile needs to be transposed (K, N) because the einsum
        # sums over the *same* k index for left[i,k] and right[j,k].
        # We therefore load right at (k, j) order.
        right_ptrs_T = (right_base
                        + offs_k[:, None] * stride_right_k
                        + offs_n[None, :] * stride_right_i)
        right_T = tl.load(right_ptrs_T,
                          mask=mask_k[:, None] & mask_n[None, :],
                          other=0.0)

        # matrix‑multiply the tile
        acc += tl.dot(left, right_T)   # (M,K)·(K,N) → (M,N)

    # --------------------------------------------------------------
    # write back the tile
    # --------------------------------------------------------------
    out_ptrs = (out_base
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator, with the heavy
    contraction performed by a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # ensure FP32 for the linear layers (the reference uses float32)
    x = x.to(torch.float32)

    # ------------------------------------------------------------------
    # 2) Projections & gating (performed with PyTorch)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is None:
        mask = torch.ones(input_tensor.shape[:3], device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)                               # (B,N,N,1)
    left = left * mask
    right = right * mask

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape  # H == hidden_dim

    # ------------------------------------------------------------------
    # 3) Core Triton contraction
    # ------------------------------------------------------------------
    # allocate output tensor
    out = torch.empty_like(left, device=device, dtype=torch.float32)

    # make tensors contiguous for simple stride extraction
    left_c = left.contiguous()
    right_c = right.contiguous()
    out_c = out

    # strides (in elements, not bytes) for the contiguous layout
    stride_left = left_c.stride()
    stride_right = right_c.stride()
    stride_out = out_c.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * H, 1),                 # bh dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_out_kernel[grid](
        left_c, right_c, out_c,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # a reasonable default for H200
    )

    # ------------------------------------------------------------------
    # 4) Post‑processing (LayerNorm → gating → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(dtype)   # match the original input dtype (usually float32)
