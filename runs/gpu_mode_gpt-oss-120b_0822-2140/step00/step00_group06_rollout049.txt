"""
TriMul “outgoing” implementation (forward only).

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every hidden channel *d*.
We keep the cheap operations (LayerNorm, linear projections, gating) in
pure PyTorch – they are bandwidth bound and already highly optimized.
The reduction itself is performed with a small Triton kernel that
fuses the two loads, the element‑wise multiplication, and the reduction
over *k* for a whole block of (i,j) pairs.  The kernel works for any
batch size *B*, sequence length *N* and hidden dimension *H* (the hidden
dimension is treated as a vectorised dimension inside the kernel).

The kernel returns a float32 tensor; the surrounding code keeps the rest
of the computation in float16/BFloat16 for best throughput and finally
casts the result back to float32 as required by the specification.
"""

import torch
import triton
import triton.language as tl
from torch import nn, einsum


# ----------------------------------------------------------------------
# Triton kernel: fused reduction over the k‑dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,          # [B, N, N, H]   (float16 or bfloat16)
    right_ptr,         # [B, N, N, H]   (float16 or bfloat16)
    out_ptr,           # [B, N, N, H]   (float32)
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled reduction over k and a small inner loop over the hidden
    dimension (BLOCK_H).  The kernel is launched with a 3‑D grid
    (B, ceil(N/BLOCK_I), ceil(N/BLOCK_J)).
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for the i, j and k dimensions
    # ------------------------------------------------------------------
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k_off = tl.arange(0, BLOCK_K)

    # Masks for boundary handling
    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator for the (i,j,h) block
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k‑dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        k = k_start + k_off
        mask_k = k < N

        # ------------------------------------------------------------------
        # Load a small slice of the hidden dimension (BLOCK_H) at a time.
        # The hidden dimension is processed in chunks to keep shared memory
        # usage modest and to allow the use of TL's vectorised loads.
        # ------------------------------------------------------------------
        h_start = 0
        while h_start < H:
            cur_h = tl.minimum(BLOCK_H, H - h_start)

            # pointers for left: shape (BLOCK_I, BLOCK_K, cur_h)
            left_ptrs = (
                left_ptr
                + pid_b * stride_lb
                + i_off[:, None, None] * stride_li
                + k[None, :, None] * stride_lk
                + (h_start + tl.arange(0, cur_h))[None, None, :] * stride_lh
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None],
                other=0.0,
            ).to(tl.float32)          # cast to fp32 for accumulation

            # pointers for right: shape (BLOCK_J, BLOCK_K, cur_h)
            right_ptrs = (
                right_ptr
                + pid_b * stride_rb
                + j_off[:, None, None] * stride_ri
                + k[None, :, None] * stride_rk
                + (h_start + tl.arange(0, cur_h))[None, None, :] * stride_rh
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None],
                other=0.0,
            ).to(tl.float32)

            # ------------------------------------------------------------------
            # Broadcast to (i, j, k, h) → multiply and reduce over k.
            # left:  (i, k, h) → (i, 1, k, h)
            # right: (j, k, h) → (1, j, k, h)
            # product → sum over k.
            # ------------------------------------------------------------------
            # Expand dimensions via view‑trick (adding a dummy axis) – Triton
            # broadcasting handles the rest.
            prod = left[:, None, :, :] * right[None, :, :, :]   # (i, j, k, h)
            acc[:, :, h_start : h_start + cur_h] += tl.sum(prod, axis=2)   # sum over k

            h_start += cur_h

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_ob
        + i_off[:, None, None] * stride_oi
        + j_off[None, :, None] * stride_oj
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_oh
    )
    # mask for i/j boundaries (h is always inside because we loop to H)
    store_mask = mask_i[:, None, None] & mask_j[None, :, None]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]   (float16 / bf16)
    returns       : [B, N, N, H]   (float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tunable block sizes – chosen to fit in shared memory on H200.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 32   # must divide hidden dim (we loop over remainder)

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )
    tri_mul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out

# Main entry‑point matching the evaluation harness signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16                     # we keep the bulk in fp16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm (dim) – use torch implementation (very cheap)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias          # shape [B,N,N,dim]
    x = x.to(dtype)

    # ------------------------------------------------------------------
    # Linear projections (no bias) – fused into a single matmul for speed
    # ------------------------------------------------------------------
    def linear_no_bias(inp, w):
        # inp [B,N,N,dim] → view as (B*N*N, dim)
        B, N, _, _ = inp.shape
        inp_flat = inp.reshape(B * N * N, dim)
        out_flat = torch.nn.functional.linear(inp_flat, w)   # (B*N*N, hidden_dim)
        return out_flat.view(B, N, N, hidden_dim)

    left = linear_no_bias(x, weights["left_proj.weight"].to(dtype))
    right = linear_no_bias(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(dtype)        # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(dtype)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(dtype)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(dtype)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core reduction – Triton kernel
    # ------------------------------------------------------------------
    # Cast to bfloat16 if the hardware prefers it; here we keep fp16.
    left = left.to(dtype)
    right = right.to(dtype)
    out = tri_mul_einsum(left, right)     # float32 result, shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Output projection + final LayerNorm
    # ------------------------------------------------------------------
    # to_out_norm (hidden_dim)  (float32)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # Gate the output
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float32)
    B, N, _, _ = out.shape
    out_flat = out.reshape(B * N * N, hidden_dim)
    out_flat = torch.nn.functional.linear(out_flat, to_out_w)   # (B*N*N, dim)
    out = out_flat.view(B, N, N, dim)

    return out.to(torch.float32)
