"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of TriMul is the bilinear contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every batch b and hidden channel d.  For a fixed (b,d) the operation is
exactly a matrix multiplication `L @ Rᵀ` where L,R ∈ ℝ^{N×N}.  We therefore
flatten the (batch,hidden) dimensions into a single “batch” dimension Z = B·H
and launch a 3‑D Triton kernel that computes many (M×N) blocks of
`L @ Rᵀ` in FP16 while accumulating in FP32.  All other ops
(LayerNorm, linear projections, gating) are done with PyTorch which is cheap
compared with the O(N³) contraction.

The kernel works for any N≤1024, B∈{1,2}, H∈{128,384,768} and respects an
optional mask.  The final tensor is returned in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul L @ Rᵀ  (Z x N x N)  → (Z x N x N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,           # const float16 *
    right_ptr,          # const float16 *
    out_ptr,            # float16 *
    N,                  # seq_len (int)
    stride_lz, stride_ly, stride_lx,   # left strides
    stride_rz, stride_ry, stride_rx,   # right strides
    stride_oz, stride_oy, stride_ox,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_z = tl.program_id(2)          # flattened (batch * hidden)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (the summed axis)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bz, i, k]  (M × K)
        a_ptr = (left_ptr
                 + pid_z * stride_lz
                 + offs_m[:, None] * stride_ly
                 + offs_k[None, :] * stride_lx)
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # right[bz, j, k]  → we need (K × N) for the dot
        b_ptr = (right_ptr
                 + pid_z * stride_rz
                 + offs_k[:, None] * stride_ry   # k as row
                 + offs_n[None, :] * stride_rx)  # j as column
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot produces (M × N) in fp16, cast‑accumulate to fp32
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    out_ptrs = (out_ptr
                + pid_z * stride_oz
                + offs_m[:, None] * stride_oy
                + offs_n[None, :] * stride_ox)
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (no bias for speed – bias added later)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x : [*, C]  float32
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel
    for the O(N³) contraction.
    Input:
        data = (input_tensor, mask, weights, config)
    Output:
        Tensor of shape [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H
    eps = 1e-5

    # ---------- LayerNorm on the last dim ----------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=eps,
    )   # [B,N,N,dim]  float32

    # ---------- Linear projections (no bias) ----------
    # use torch.nn.functional.linear for speed
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device),   # [hidden, dim]
    )   # [B,N,N,hidden]
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device),
    )

    # ---------- Apply mask (optional) ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device))
    )

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton contraction ----------
    # reshape to (Z, N, N) where Z = B*hidden_dim
    Z = B * hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(Z, N, N)   # [Z,N,N]
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(Z, N, N)

    # cast to half for the kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)   # same shape, fp16

    # strides (in elements)
    lz, ly, lx = left_fp16.stride()
    rz, ry, rx = right_fp16.stride()
    oz, oy, ox = out_fp16.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        Z,
    )
    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        lz, ly, lx,
        rz, ry, rx,
        oz, oy, ox,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 matmul on H200
    )

    # reshape back to [B, N, N, hidden]
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ---------- Output LayerNorm ----------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=eps,
    )

    # ---------- Apply final gate and linear ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(device))   # [B,N,N,dim]

    return out
