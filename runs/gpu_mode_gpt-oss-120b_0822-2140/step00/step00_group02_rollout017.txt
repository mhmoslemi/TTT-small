    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    # LayerNorm: (dim)
    # compute norm: (x - mean) / sqrt(var+eps) * weight + bias
    # we can use torch.nn.functional.layer_norm
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)

    # linear projections
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])
    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask
    # gates
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # reshape for matmul

    right_t = right.permute(0,3,1,2).contiguous()

"""
TriMul “outgoing” kernel (AlphaFold‑3 style) – Triton‑accelerated N³ reduction.

The heavy part of TriMul is the einsum  
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* this is a batched matrix‑multiply:
    left_d   : (B, N, N)          (i‑k)
    right_d  : (B, N, N)  →  right_dᵀ : (B, N, N)  (k‑j)

    out_d = left_d @ right_dᵀ    → (B, N, N)

We flatten the batch and hidden dimensions into a single “batch’’ axis
and launch one Triton kernel that computes many independent matrix‑multiplies
with a K‑reduction loop.  All other operations (LayerNorm, Linear, gates)
remain in PyTorch, which on Hopper is already highly optimized.

The kernel works for arbitrary batch size (1‑2), sequence length N (≤1024)
and hidden dimension (typically 128).  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A: (B,H,N,K), B: (B,H,K,N))
# ----------------------------------------------------------------------
@triton.jit
def _matmul_reduce_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers
    M, N, K,                        # matrix sizes (all = seq_len)
    stride_ah, stride_am, stride_ak,   # strides of A (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # strides of B (batch, row, col)
    stride_ch, stride_cm, stride_cn,   # strides of C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A : (batch, M, K)   (row‑major)
    B : (batch, K, N)
    C : (batch, M, N)
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decode a 3‑D grid: (batch, block_row, block_col)
    # ------------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    block_index = pid % (num_blocks_m * num_blocks_n)
    block_row = block_index // num_blocks_n
    block_col = block_index % num_blocks_n

    # ------------------------------------------------------------------
    # Offsets inside the current block
    # ------------------------------------------------------------------
    offs_m = block_row * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_col * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + batch * stride_ah
    b_batch_ptr = B_ptr + batch * stride_bh
    c_batch_ptr = C_ptr + batch * stride_ch

    # Pointers to the start of the tile (M‑K and K‑N)
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                            + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over the K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] &
                         (tl.arange(0, BLOCK_K) < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K)[:, None] &
                         (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < M)[:, None] &
                  (offs_n < N)[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            • input_tensor : [B, N, N, dim]  (float32)
            • mask         : [B, N, N]      (bool/float)
            • weights      : dict of torch tensors (see code)
            • config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm + linear projections (remain in PyTorch)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Optional mask – broadcast on the hidden dimension
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gating scalars (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for batched matmul:
    #   left  : (B, hidden, N, N)
    #   right : (B, hidden, N, N)  →  transposed on the last two axes
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    right_t = right.transpose(-2, -1).contiguous()   # (B, hidden, N, N)

    # Flatten batch & hidden into a single dimension for the kernel
    BH = B * hidden
    A = left.view(BH, N, N)          # (BH, M=N, K=N)
    B_mat = right_t.view(BH, N, N)   # (BH, K=N, N)

    # Output buffer
    C = torch.empty_like(A)

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for Hopper)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M) *
        triton.cdiv(N, BLOCK_N) *
        BH,
    )

    _matmul_reduce_kernel[grid](
        A_ptr=A,
        B_ptr=B_mat,
        C_ptr=C,
        M=N,
        N=N,
        K=N,
        stride_ah=A.stride(0),
        stride_am=A.stride(1),
        stride_ak=A.stride(2),
        stride_bh=B_mat.stride(0),
        stride_bk=B_mat.stride(1),
        stride_bn=B_mat.stride(2),
        stride_ch=C.stride(0),
        stride_cm=C.stride(1),
        stride_cn=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original 4‑D layout
    # ------------------------------------------------------------------
    out = C.view(B, hidden, N, N).permute(0, 2, 3, 1)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # Final LayerNorm, output gating and linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out
