"""
TriMul “outgoing” kernel (AlphaFold3) – Triton‑accelerated N³ pairwise update.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C).
2. Linear projections (left / right) and three sigmoid gates (left, right,
   out) – done in PyTorch (these are cheap compared to the N³ step).
3. Fuse the mask and the two gating multiplications.
4. Compute the heavy contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix‑multiply L @ Rᵀ for every hidden
   channel *d*.  The kernel treats every (batch, hidden) pair as an
   independent GEMM of shape (N×N)·(N×N)ᵀ.
5. Apply the output Layer‑norm, the output gate and a final linear
   projection – also in PyTorch.
All computations are performed in float32; the kernel loads in fp32
(and could be changed to fp16 for extra speed) and accumulates in fp32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# -------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    lhs_ptr, rhs_ptr, out_ptr,               # pointers
    M, N, K,                                 # matrix sizes (all == seq_len)
    stride_lhs_batch, stride_lhs_m, stride_lhs_k,
    stride_rhs_batch, stride_rhs_m, stride_rhs_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = lhs @ rhsᵀ   (lhs, rhs : [batch*hid, M, K] )
    where each batch*hid slice is a separate GEMM.
    """

    batch_id   = tl.program_id(0)               # combined (batch * hidden)
    pid_m      = tl.program_id(1)               # block row
    pid_n      = tl.program_id(2)               # block col

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of L (shape BLOCK_M x BLOCK_K)
        lhs = tl.load(
            lhs_ptr
            + batch_id * stride_lhs_batch
            + offs_m[:, None] * stride_lhs_m
            + offs_k[None, :] * stride_lhs_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load a tile of Rᵀ.
        # Original RHS layout is (batch*hid, M=row, K=col).
        # To get Rᵀ we load with K as the first dimension and N (the
        # output‑column) as the second.
        rhs = tl.load(
            rhs_ptr
            + batch_id * stride_rhs_batch
            + offs_k[:, None] * stride_rhs_k          # K dimension becomes rows
            + offs_n[None, :] * stride_rhs_n,         # N dimension becomes columns
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Block‑matrix multiply‑accumulate
        acc += tl.dot(lhs, rhs)  # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the result back
    # ------------------------------------------------------------------
    tl.store(
        out_ptr
        + batch_id * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# -------------------------------------------------------------
# Helper that wraps the Triton kernel
# -------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (H = hidden_dim)
    Returns out : [B, N, N, H]
    """
    B, N, _, H = left.shape

    # bring hidden dim to the second axis for easy reshaping
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # combine batch and hidden -> each GEMM works on (N x N) × (N x N)ᵀ
    left = left.view(-1, N, N)                     # [B*H, N, N]
    right = right.view(-1, N, N)
    out = torch.empty_like(left)

    # -------------------------------------------------------------
    # launch parameters (tuned for Hopper – 64×64 tiles, 64‑wide K)
    # -------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left.shape[0],                     # batch*hidden
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,                     # M
        N,                     # N (output cols)
        N,                     # K (inner dim)
        left.stride(0),
        left.stride(1),
        left.stride(2),
        right.stride(0),
        right.stride(1),
        right.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # reasonable default for Hopper
    )

    # reshape back to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Entrypoint required by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating values (sigmoid of linears)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Heavy N³ contraction via Triton
    # -----------------------------------------------------------------
    out = tri_mul_triton(left, right)                 # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 6) Output Layer‑norm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)          # guarantee float32 output
