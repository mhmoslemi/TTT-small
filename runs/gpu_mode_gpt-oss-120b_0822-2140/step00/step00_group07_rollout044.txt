"""
TriMul “outgoing” forward pass with a Triton kernel for the
core N×N×N tensor contraction.

The heavy part is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden dimension `d` this is a batched matrix‑multiply
(L @ Rᵀ) where L,R ∈ ℝ^{B×N×N}.  The kernel computes one
hidden slice at a time (kernel launch per `d`).  All LayerNorm,
linear projections, and gating are kept in PyTorch – they are
memory‑bound and inexpensive compared with the O(N³·H) contraction.

The kernel works for any batch size (1 or 2), any sequence length
(≤ 1024) and any hidden dimension (≤ 128).  It correctly masks
the inputs when a mask is supplied.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel – computes one hidden‑dimension slice:
#   out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_slice_kernel(
    left_ptr,          # *float32   left slice   (B,N,N)
    right_ptr,         # *float32   right slice  (B,N,N)
    out_ptr,           # *float32   output slice (B,N,N)

    B, N,                         # scalar dimensions
    stride_l_b, stride_l_i, stride_l_k,   # left strides (no hidden stride)
    stride_r_b, stride_r_j, stride_r_k,   # right strides
    stride_o_b, stride_o_i, stride_o_j,   # out strides

    BLOCK_M: tl.constexpr,        # tile size for i
    BLOCK_N: tl.constexpr,        # tile size for j
    BLOCK_K: tl.constexpr,        # tile size for reduction over k
):
    # ------------------------------------------------------------------
    # 1)  Identify the (batch, i‑block, j‑block) this program works on
    # ------------------------------------------------------------------
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    pid_i = pid_rem // num_j          # i‑tile index
    pid_j = pid_rem % num_j           # j‑tile index

    # ------------------------------------------------------------------
    # 2)  Compute absolute coordinates for the tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_base < N
    j_mask = j_base < N

    # ------------------------------------------------------------------
    # 3)  Accumulate Σ_k left * right
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # load left[b, i, k]  -> (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + batch * stride_l_b
            + i_base[:, None] * stride_l_i
            + k_base[None, :] * stride_l_k,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # load right[b, j, k] -> (BLOCK_N, BLOCK_K)  (need transpose for dot)
        right = tl.load(
            right_ptr
            + batch * stride_r_b
            + j_base[:, None] * stride_r_j
            + k_base[None, :] * stride_r_k,
            mask=j_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # right is (BLOCK_N, BLOCK_K); we need (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, tl.trans(right))

    # ------------------------------------------------------------------
    # 4)  Write the tile to the output tensor
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_o_b
        + i_base[:, None] * stride_o_i
        + j_base[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
#   Public entry point – called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for
    the O(N³·H) contraction.

    Args:
        data: (input, mask, weights, config)
            - input : [B, N, N, dim]  (float32)
            - mask  : [B, N, N]       (bool/float, may be all‑ones)
            - weights: dict with all required parameters
            - config : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0) unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = torch.float32

    # --------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑last)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (applied before gates)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate          # [B,N,N,H]
    right = right * right_gate

    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    # --------------------------------------------------------------
    # 5) Triton kernel – compute out_slice for every hidden channel
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides (in elements, not bytes)
    stride_l = left.stride()   # (b,i,k,h)
    stride_r = right.stride()  # (b,j,k,h)
    stride_o = out.stride()    # (b,i,j,h)

    # tile sizes – tuned for Hopper SM90 (large shared‑memory, 128‑bit loads)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # number of tiles per batch
    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_tiles = B * num_i * num_j

    # launch one kernel per hidden dimension slice
    for h in range(hidden_dim):
        # pointers for the current hidden slice
        left_ptr = left.data_ptr() + h * stride_l[3] * left.element_size()
        right_ptr = right.data_ptr() + h * stride_r[3] * right.element_size()
        out_ptr = out.data_ptr() + h * stride_o[3] * out.element_size()

        tri_mul_slice_kernel[
            (total_tiles,)
        ](
            left_ptr,
            right_ptr,
            out_ptr,
            B,
            N,
            stride_l[0],
            stride_l[1],
            stride_l[2],
            stride_r[0],
            stride_r[1],
            stride_r[2],
            stride_o[0],
            stride_o[1],
            stride_o[2],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # --------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out‑gate, final linear
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate  # broadcast over hidden dim – out_gate shape matches

    out = F.linear(out, weights["to_out.weight"])
    return out
