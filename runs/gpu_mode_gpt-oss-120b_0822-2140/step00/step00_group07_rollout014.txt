"""
TriMul “outgoing” implementation (forward only).

The expensive part of TriMul is the contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

for every batch (b) and hidden channel (h).  After permuting the tensors
to (B, H, N, K) this is a batched matrix‑multiply:

    out_flat[b*h] = left_flat[b*h] @ right_flat[b*h].T

where left_flat and right_flat have shape (B*H, N, K) and K = N.
We implement this GEMM with a Triton kernel (tiling in M, N, K) and keep
the surrounding LayerNorms / Linear / gating logic in PyTorch for
simplicity and numerical stability.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# -------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B.T)  (float32 accumulator)
# -------------------------------------------------------------
@triton.jit
def _gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix sizes (all = seq_len)
    stride_az, stride_ak, stride_aa,    # A strides: batch, row, col
    stride_bz, stride_bk, stride_bn,    # B strides: batch, row(k), col(n)
    stride_cz, stride_cm, stride_cn,    # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)       # batch * hidden dimension
    pid_m = tl.program_id(0)       # block row
    pid_n = tl.program_id(1)       # block col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointer arithmetic
    a_tile_ptr = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_ak + offs_k[None, :] * stride_aa
    b_tile_ptr = b_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)                 # matmul of the tile

        a_tile_ptr += BLOCK_K * stride_aa   # advance K‑tile for A
        b_tile_ptr += BLOCK_K * stride_bk   # advance K‑tile for B

    # write result
    c_tile_ptr = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr,
             acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# -------------------------------------------------------------
# Public entry point used by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N]       (float32/bool)
            weights      : dict of torch.Tensor (all float32)
            config       : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------
    # 1. LayerNorm on the input tensor (no bias folding)
    # -------------------------------------------------
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + eps
    )
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # -------------------------------------------------
    # 2. Linear projections (no bias, weight‑only)
    # -------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4. Gating (sigmoid(linear))
    # -------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # -------------------------------------------------
    # 5. Rearrange for batched GEMM: (B, H, N, N)
    # -------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, K]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*hidden → (B*H, N, N)
    left_flat  = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)

    # -------------------------------------------------
    # 6. Triton GEMM launch
    # -------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim,                   # batch*head
    )

    _gemm_out_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # -------------------------------------------------
    # 7. Final LayerNorm, gating and linear projection
    # -------------------------------------------------
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + eps
    )
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(dtype)
