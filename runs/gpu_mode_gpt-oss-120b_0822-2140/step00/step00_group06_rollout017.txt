"""
TriMul (outgoing) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input (dim = C).
2. Linear projections → left/right  (shape B,N,N,H)  (H = hidden_dim)
3. Optional mask (B,N,N,1) is applied.
4. Gating sigmoid (left/right/out) and element‑wise gating of left/right.
5. **Heavy step**:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of H independent matrix‑multiplications
   `out_d = left_d @ right_d.T`.
   Implemented with a Triton kernel that tiles the (i,j) output plane
   (BLOCK_M × BLOCK_N) and reduces over K in chunks (BLOCK_K).  The hidden
   dimension is split into small blocks (BLOCK_H) to keep register pressure
   low.
6. Layer‑norm over the hidden dimension, multiply by the out‑gate and a final
   linear projection back to the original dimension C.

The kernel works for any batch size ≤2, sequence length up to 1024 and
hidden dimension up to 384.  All tensors are assumed to be on CUDA and the
final output is returned as `torch.float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: computes   out = Σ_k left[...,k,:] * right[...,k,:]
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # sizes
    B, N, H,
    # strides in elements (not bytes!)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # tile over row  (i)
    pid_j = tl.program_id(1)          # tile over col  (j)

    # start offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # hidden‑dim block offsets (looped over inside the kernel)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension (the sequence length) in BLOCK_K steps
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # -----------------------------------------------------------------
        # Load left  : shape (BLOCK_M, BLOCK_K, BLOCK_H)
        # Load right : shape (BLOCK_N, BLOCK_K, BLOCK_H)
        # -----------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_b
            + offs_i[:, None, None] * stride_l_i
            + cur_k[None, :, None] * stride_l_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_l_h
        )
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_b
            + offs_j[:, None, None] * stride_r_j
            + cur_k[None, :, None] * stride_r_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_r_h
        )

        # masks for out‑of‑bounds loads (zero‑pad)
        m_left = mask_i[:, None, None] & mask_k[None, :, None] & (tl.arange(0, BLOCK_H)[None, None, :] < H)
        m_right = mask_j[:, None, None] & mask_k[None, :, None] & (tl.arange(0, BLOCK_H)[None, None, :] < H)

        left = tl.load(left_ptrs, mask=m_left, other=0.0).to(tl.float32)   # (M,K,H)
        right = tl.load(right_ptrs, mask=m_right, other=0.0).to(tl.float32)  # (N,K,H)

        # Broadcast to (M,N,K,H) and reduce over K
        # left  : (M,K,1,H)   -> expand dim for j
        # right : (1,N,K,H)   -> expand dim for i
        left_exp = left[:, :, None, :]          # (M,K,1,H)
        right_exp = right[None, :, :, :]        # (1,N,K,H)
        prod = left_exp * right_exp             # (M,N,K,H)
        acc += tl.sum(prod, axis=2)             # sum over K → (M,N,H)

    # -----------------------------------------------------------------
    # Store the accumulated result
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_o_h
    )
    # mask for final store (handles edge tiles)
    mask_store = mask_i[:, None, None] & mask_j[None, :, None] & (tl.arange(0, BLOCK_H)[None, None, :] < H)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)


# -------------------------------------------------------------
# Python entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        * input_tensor – torch.Tensor [B, N, N, C] (float32)
        * mask         – torch.Tensor [B, N, N]   (bool/float) – may be all‑ones
        * weights_dict – mapping str → torch.Tensor containing all model weights
        * config_dict  – contains "dim", "hidden_dim", and optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension C
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float32)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    # x  : (B,N,N,C)   weight : (H,C)  → result (B,N,N,H)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.to(x.dtype).unsqueeze(-1)          # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Heavy reduction – Triton kernel
    # -----------------------------------------------------------------
    # Cast to float16 for the kernel (Hopper TensorCores)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # allocate output buffer (float16, will be cast back later)
    out_h = torch.empty((B, N, N, H), device=left.device, dtype=torch.float16)

    # strides in *elements* (not bytes)
    s_l = left_h.stride()
    s_r = right_h.stride()
    s_o = out_h.stride()

    # Tile sizes – chosen to fit in registers on Hopper
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 8   # hidden‑dim tile (H ≤ 384 → at most 48 tiles)

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B,                         # batch
    )
    tri_mul_kernel[grid](
        left_h.data_ptr(),
        right_h.data_ptr(),
        out_h.data_ptr(),
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    # Cast back to FP32 for the remaining ops
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Post‑processing: to_out_norm, out_gate, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
