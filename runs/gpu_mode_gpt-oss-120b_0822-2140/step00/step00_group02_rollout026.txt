"""
TriMul “outgoing” implementation.

The heavy N³ work is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We rewrite it as a batched matrix‑multiply per hidden dimension d:

    out_slice = left_slice @ right_sliceᵀ   (where right_sliceᵀ[b,k,j,d] = right[b,j,k,d])

The kernel tiles (i,j) and the hidden dimension d, loading a BLOCK_K chunk of
the k‑dimension at a time and accumulating with tl.dot.  All other ops
(LayerNorm, Linear, gates, masking) are performed with PyTorch for simplicity.

The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides (in elements)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where `right` is accessed with its k‑dimension first (i.e. we compute
    left @ rightᵀ per hidden slice.
    """
    pid = tl.program_id(0)

    # ----- decode 1‑D pid into (batch, block_i, block_j) -----
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_i * num_j)
    block_id = pid % (num_i * num_j)
    block_i = block_id // num_j
    block_j = block_id % num_j

    # ----- tile offsets -----
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator: (M, N, H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----- loop over k dimension in BLOCK_K chunks -----
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (M, K, H)
        left_ptrs = (
            left_ptr
            + batch * stride_l_batch
            + offs_i[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # right transposed on (k, j): shape (K, N, H)
        # we swap the i/k strides to get right_T[k, j, h] = right[j, k, h]
        right_ptrs = (
            right_ptr
            + batch * stride_r_batch
            + offs_k[:, None, None] * stride_r_i   # <-- i‑stride gets k
            + offs_j[None, :, None] * stride_r_k   # <-- k‑stride gets j
            + offs_h[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----- per‑hidden‑slice dot product -----
        # left: (M, K, H)   right: (K, N, H)
        for h in range(0, BLOCK_H):
            # slice dimensions for this h
            left_h = left[:, :, h]   # (M, K)
            right_h = right[:, :, h] # (K, N)
            acc[:, :, h] += tl.dot(left_h, right_h)  # (M, N)

    # ----- store the result -----
    out_ptrs = (
        out_ptr
        + batch * stride_o_batch
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_h[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input:   torch.Tensor[B, N, N, dim]   (float32)
            - mask:    torch.Tensor[B, N, N]        (float32 or bool)
            - weights: dict with all linear / norm parameters (float32)
            - config:  dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------- 1. LayerNorm on input (dim) ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------- 3. Optional mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)           # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------- 4. Gates (sigmoid) ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Triton kernel: sum_k left * right (with rightᵀ)
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # will hold the intermediate result (B,N,N,hidden_dim)

    # strides in elements (not bytes)
    s_l_batch, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_batch, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_batch, s_o_i, s_o_j, s_o_h = out.stride()

    # tile sizes – tuned for Hopper (SM90) H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # hidden_dim is a multiple of 8 (e.g., 128)

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    total_blocks = B * grid_i * grid_j
    grid = (total_blocks,)

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_batch=s_l_batch,
        stride_l_i=s_l_i,
        stride_l_k=s_l_k,
        stride_l_h=s_l_h,
        stride_r_batch=s_r_batch,
        stride_r_i=s_r_i,
        stride_r_k=s_r_k,
        stride_r_h=s_r_h,
        stride_o_batch=s_o_batch,
        stride_o_i=s_o_i,
        stride_o_j=s_o_j,
        stride_o_h=s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------- 6. Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # --------- 7. Apply out gate ----------
    out = out * out_gate

    # --------- 8. Final linear ----------
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
