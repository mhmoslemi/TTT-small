"""
TriMul “outgoing” implementation with a custom Triton kernel.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply of the form
    (B*D, N, N) @ (B*D, N, N).T .
A lightweight Triton kernel performs this GEMM (transposed second operand)
while all LayerNorm / Linear / gating work is done with native PyTorch
operations.  The final result is returned as float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ B^T   (A,B : (batch, M, K) with M==K==N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_transpose_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes (M=N, K=N)
    stride_ah, stride_ak, stride_ab,    # A: row, col, batch strides
    stride_bh, stride_bk, stride_bb,    # B: row, col, batch strides
    stride_ch, stride_ck, stride_cb,    # C: row, col, batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B.T for a single batch (B*D)."""
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # ------------------- offsets & masks -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the block in A and B (B is accessed transposed)
    a_ptrs = A_ptr + batch_id * stride_ab \
                     + offs_m[:, None] * stride_ah \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B^T : we swap the row/col strides of the original B
    b_ptrs = B_ptr + batch_id * stride_bb \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_bh \
                     + offs_n[None, :] * stride_bk

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- main loop over K -------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bh

    # ------------------- write back -------------------
    c_ptrs = C_ptr + batch_id * stride_cb \
                     + offs_m[:, None] * stride_ch \
                     + offs_n[None, :] * stride_ck
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – Tensor[bs, N, N, dim] (float32, on CUDA)
        mask    – Tensor[bs, N, N] (bool/float, on CUDA)
        weights – mapping of all linear / layer‑norm parameters
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor[bs, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm expects normalized_shape = (dim,)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(input_tensor.dtype),
        bias=weights["norm.bias"].to(input_tensor.dtype),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(x.dtype), bias=None)   # (B,N,N,hd)
    right = F.linear(x, weights["right_proj.weight"].to(x.dtype), bias=None)  # (B,N,N,hd)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                     # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype), bias=None))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype), bias=None))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype), bias=None))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Reshape for batched GEMM:
    #    (B, N, N, hd) -> (B*hd, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    batch_hd = B * hidden_dim

    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(batch_hd, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch_hd, N, N)

    # ------------------------------------------------------------------
    # 6. Allocate output buffer (float32)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 7. Launch Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_hd,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (all tensors are contiguous in dim order)
    stride_ah, stride_ak, stride_ab = left_flat.stride()
    stride_bh, stride_bk, stride_bb = right_flat.stride()
    stride_ch, stride_ck, stride_cb = out_flat.stride()

    batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_ah, stride_ak, stride_ab,
        stride_bh, stride_bk, stride_bb,
        stride_ch, stride_ck, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hd)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hd)

    # ------------------------------------------------------------------
    # 9. Final LayerNorm + gating + output linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(out.dtype), bias=None)  # (B,N,N,dim)

    return out.type(torch.float32)
