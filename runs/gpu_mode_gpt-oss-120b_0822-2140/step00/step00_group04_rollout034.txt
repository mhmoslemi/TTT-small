"""
TriMul “outgoing” implementation with a custom Triton kernel.

The expensive operation in TriMul is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication of shape (N,N) for every
(batch, hidden_dim) pair.  The kernel tiles the i‑ and j‑axes (BLOCK_M,
BLOCK_N) and loops over k in BLOCK_K tiles, accumulating the result with
tl.dot.  All other linear / LayerNorm / gating steps are performed in
PyTorch – they are cheap compared to the O(N³) contraction.

The kernel is launched with a 3‑D grid:
    (ceil(N/BLOCK_M), ceil(N/BLOCK_N), batch * hidden_dim)

so each program instance computes one (i‑tile, j‑tile) for a single
(b,d) pair.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,                          # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The kernel assumes tensors are contiguous in memory (the strides are
    passed explicitly).  Computation is performed in FP32.
    """
    pid_i = tl.program_id(0)               # tile along the i dimension
    pid_j = tl.program_id(1)               # tile along the j dimension
    pid_z = tl.program_id(2)               # combined (batch, hidden) index

    # --------------------------------------------------------------
    # decode batch and hidden dim from pid_z
    # --------------------------------------------------------------
    b = pid_z // D                         # batch index
    d = pid_z % D                          # hidden‑dim index

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over k in BLOCK_K steps
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_base = k_iter * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for the current tile
        left_tile_ptr = left_ptr + \
            b * stride_left_b + \
            d * stride_left_d + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k

        right_tile_ptr = right_ptr + \
            b * stride_right_b + \
            d * stride_right_d + \
            j_offset[:, None] * stride_right_j + \
            k_offset[None, :] * stride_right_k

        # load tiles (zero‑pad out‑of‑bounds)
        left_tile = tl.load(
            left_tile_ptr,
            mask=(mask_i[:, None] & mask_k[None, :]),
            other=0.0
        )
        right_tile = tl.load(
            right_tile_ptr,
            mask=(mask_j[:, None] & mask_k[None, :]),
            other=0.0
        )

        # accumulate: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_tile_ptr = out_ptr + \
        b * stride_out_b + \
        d * stride_out_d + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j

    tl.store(
        out_tile_ptr,
        acc,
        mask=(mask_i[:, None] & mask_j[None, :])
    )


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim] (float32)
        - mask         : torch.Tensor, [B, N, N] (bool or float)
        - weights_dict : dict of torch.Tensor containing all learnable weights
        - config_dict  : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel: the tri‑multiplicative contraction
    # ------------------------------------------------------------------
    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=l_strides[0],
        stride_left_i=l_strides[1],
        stride_left_k=l_strides[2],
        stride_left_d=l_strides[3],
        stride_right_b=r_strides[0],
        stride_right_j=r_strides[1],
        stride_right_k=r_strides[2],
        stride_right_d=r_strides[3],
        stride_out_b=o_strides[0],
        stride_out_i=o_strides[1],
        stride_out_j=o_strides[2],
        stride_out_d=o_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # adequate for fp32 on Hopper
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
