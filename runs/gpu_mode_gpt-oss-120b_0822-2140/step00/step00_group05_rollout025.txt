"""
TriMul “outgoing” kernel (AlphaFold3).

Given an input tensor X ∈ ℝ^{B×N×N×C} the forward pass is:

    X̂ = LayerNorm_C (X)
    L = (X̂·W_l) * σ(X̂·W_gl)          # left projection  (B,N,N,H)
    R = (X̂·W_r) * σ(X̂·W_gr)          # right projection (B,N,N,H)
    if mask is provided:   L,R ← L,R ⊙ mask[...,None]

    Out[b,i,j,:] = Σ_k L[b,i,k,:] * R[b,j,k,:]          (H‑dim kept)
                     └─────────────────────┘
                     contraction over the “k” axis

    Out ← LayerNorm_H (Out)
    Out ← Out * σ(X̂·W_go)                # output gate
    Out ← Out · W_out                     # linear back to C

The costly N‑wise contraction (N³·H work) is implemented in Triton.
All other operations (LayerNorm, Linear, Sigmoid, masking) are performed
with PyTorch for simplicity.
The kernel works on fp16 data for the contraction and returns fp32.

The kernel launches a 3‑D grid:
    dim0 : batch × hidden_dim   (each “channel” of the hidden dim is independent)
    dim1 : blocks over the first sequence axis i
    dim2 : blocks over the second sequence axis j
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *const float16
    right_ptr,              # *const float16  (will be read transposed)
    out_ptr,                # *float16
    B, N, H,                # int32 scalars
    # stride information (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) identify the (batch, hidden) slice this program instance works on
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)                     # batch * hidden_dim index
    total_bc = B * H
    mask_total = pid0 < total_bc
    b = pid0 // H                               # batch index
    h = pid0 % H                                # hidden‑dim index
    # early‑exit for out‑of‑range programs (grid may be slightly larger)
    if not mask_total:
        return

    # ------------------------------------------------------------------
    # 2) 2‑D tiling over the i and j axes
    # ------------------------------------------------------------------
    pid_m = tl.program_id(1)                    # block over i
    pid_n = tl.program_id(2)                    # block over j

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left[b, i, k, h]  (M × K) --------------------------------
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- load right[b, j, k, h] transposed to (K × N) --------------------
        # we need right_T[k, j] = right[b, j, k, h]
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_n[None, :] * stride_right_j
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_n[None, :],
                          other=0.0).to(tl.float32)

        # ----- inner‑product over K ------------------------------------------
        acc += tl.dot(left, right_T)   # (M,K) @ (K,N) → (M,N)

    # ------------------------------------------------------------------
    # 4) write back, cast to fp16
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a Triton kernel.

    Args:
        data: (input, mask, weights, config)
            input   – torch.Tensor of shape [B, N, N, C] (float32)
            mask    – torch.Tensor of shape [B, N, N] (bool/float) or None
            weights – dict mapping the parameter names used in the reference
                      PyTorch implementation to torch.Tensors.
            config  – dict with keys "dim" (C) and "hidden_dim" (H).

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim = C)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(C,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    # Helper: linear on the last dimension
    def linear_last_dim(t, w):
        # t: [B,N,N,C] , w: [out_dim, C]
        B_, N_, N_, C_ = t.shape
        t_flat = t.reshape(-1, C_)
        out_flat = torch.nn.functional.linear(t_flat, w)   # (out_dim, C)
        return out_flat.view(B_, N_, N_, -1)

    left = linear_last_dim(x, weights["left_proj.weight"])
    right = linear_last_dim(x, weights["right_proj.weight"])

    # Apply mask if present – mask shape [B,N,N] → [B,N,N,1]
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(linear_last_dim(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_last_dim(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_last_dim(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Triton contraction over k
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)   # shape [B,N,N,H]

    # strides in elements (not bytes)
    s_lb, s_li, s_lk, s_lh = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rh = right_fp16.stride()
    s_ob, s_oi, s_oj, s_oh = out_fp16.stride()

    # Choose block sizes – tuned for Hopper (SM90) and the given N ranges
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * H, 1),                 # dim0 : batch * hidden
        triton.cdiv(N, BLOCK_M),               # dim1 : i‑axis
        triton.cdiv(N, BLOCK_N),               # dim2 : j‑axis
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,                           # enough warps for fp16 matmul
    )

    # back to fp32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 4) Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(H,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate

    # final linear : [B,N,N,H] → [B,N,N,C]
    out = torch.nn.functional.linear(out.reshape(-1, H),
                                      weights["to_out.weight"]).view(B, N, N, C)

    return out
