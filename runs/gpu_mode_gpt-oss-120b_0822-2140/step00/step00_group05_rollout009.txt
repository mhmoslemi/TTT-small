"""
TriMul “outgoing” forward pass (AlphaFold3) with a Triton kernel for the
heavy N‑by‑N matrix multiplication across the hidden channel dimension.

The kernel computes, for every batch b and hidden channel h,

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is exactly a batched matmul `A @ Bᵀ` with A = left[..., :, :, h] and
B = right[..., :, :, h].  All other operations (LayerNorm, linear
projections, gates, final LayerNorm and linear) are carried out in PyTorch.
The kernel works with float16 inputs and accumulates in float32, then stores
the result as float32.

The implementation supports the full range of test shapes (B∈{1,2},
N∈{128,256,512,1024}, dim∈{128,384}, hidden_dim=128) and handles the
presence or absence of a mask.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _tri_mul_kernel(
    a_ptr,                     # left   [B,N,N,H]  (float16)
    b_ptr,                     # right  [B,N,N,H]  (float16)
    c_ptr,                     # out    [B,N,N,H]  (float32)
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a blocked GEMM.  The kernel is launched with a 3‑D grid:
        pid0 = batch * hidden
        pid1 = block row (i)
        pid2 = block col (j)
    """
    # ---- identify which (batch, hidden) we are working on -----------------
    pid0 = tl.program_id(0)                     # 0 … B*H-1
    total = B * H
    if pid0 >= total:
        return
    b = pid0 // H                               # batch index
    h = pid0 % H                                # hidden channel index

    # ---- block indices for the output matrix (i, j) -----------------------
    pid_i = tl.program_id(1)                    # block row
    pid_j = tl.program_id(2)                    # block col

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---- compute base pointers for this (b,h) slice -----------------------
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the reduction dimension k -----------------------------
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape [BLOCK_M, BLOCK_K]  (i, k)
        a = tl.load(
            a_base
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right (transposed view): we need B[j,k] -> accessed as (k, j)
        # shape [BLOCK_K, BLOCK_N]  (k, j)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # ---- write result ----------------------------------------------------
    tl.store(
        c_base
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” module.

    Args:
        data: Tuple containing
            - input   : torch.Tensor of shape [B, N, N, dim]
            - mask    : torch.Tensor of shape [B, N, N]  (may be all‑ones)
            - weights : dict of model parameters (see docs)
            - config  : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm over the input dim ------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B,N,N,dim)  float32

    # ----- Linear projections (no bias) ------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"]) # (B,N,N,hidden)

    # ----- Apply mask if present -------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)     # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ----- Gates (sigmoid) -------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ----- Cast to half for the kernel (accumulate in fp32) ----------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ----- Output buffer ---------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # ----- Strides (required by the Triton kernel) ------------------------
    a_ptr = left_h
    b_ptr = right_h
    c_ptr = out

    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = (
        a_ptr.stride(0), a_ptr.stride(1), a_ptr.stride(2), a_ptr.stride(3)
    )
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = (
        b_ptr.stride(0), b_ptr.stride(1), b_ptr.stride(2), b_ptr.stride(3)
    )
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = (
        c_ptr.stride(0), c_ptr.stride(1), c_ptr.stride(2), c_ptr.stride(3)
    )

    # ----- Kernel launch parameters ----------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total = B * hidden
    grid = (
        total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        B,
        N,
        hidden,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Second LayerNorm (over hidden) ----------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ----- Apply output gate ------------------------------------------------
    out = out * out_gate

    # ----- Final linear projection back to dim -----------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
