"""
TriMul “outgoing” implementation (forward only).

The heavy part is the 3‑dimensional contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For a fixed batch `b` and hidden channel `d` this is simply a matrix
multiplication:
    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T

The kernel below performs many independent GEMMs (one per (b, d)) in a
single Triton launch.  All other linear / LayerNorm / gating
operations are done with regular PyTorch kernels because they are
memory‑bound and much cheaper than the O(N³) contraction.

The final result is cast back to float32 and passed through the output
normalisation and linear projection as in the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B^T   (A,B : [B, N, N, D])
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, D,
    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld,   # left  : (b,i,k,d)
    stride_rb, stride_rj, stride_rk, stride_rd,   # right : (b,j,k,d)  (we use j as row)
    stride_ob, stride_oi, stride_oj, stride_od,   # out   : (b,i,j,d)
    # BLOCK sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
        The loop over `k` is tiled with BLOCK_K. """
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_bd = tl.program_id(2)         # combined batch & hidden dim

    # decode batch and hidden index
    b = pid_bd // D
    d = pid_bd % D

    # -------------------- offsets --------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the required tiles
    # left:  (b, i, k, d)
    left_tile_ptr = left_ptr \
        + b * stride_lb \
        + d * stride_ld \
        + offs_i[:, None] * stride_li \
        + offs_k[None, :] * stride_lk

    # right (transposed on‑the‑fly):
    # we need right[b, j, k, d] but treat it as (k, j) matrix,
    # so we load with k as the “row” dimension.
    right_tile_ptr = right_ptr \
        + b * stride_rb \
        + d * stride_rd \
        + offs_k[:, None] * stride_rj \
        + offs_j[None, :] * stride_rk

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- loop over k dimension (tiled) -------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k
        cur_mask_k = cur_k < N

        # load tiles, apply k‑mask
        a = tl.load(left_tile_ptr,
                    mask=mask_i[:, None] & cur_mask_k[None, :],
                    other=0.0).to(tl.float32)

        b_tile = tl.load(right_tile_ptr,
                         mask=cur_mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        # inner product
        acc += tl.dot(a, b_tile)

        # advance pointers to next k‑block
        left_tile_ptr  += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rj

    # write result (store as float16 to save bandwidth)
    out_tile_ptr = out_ptr \
        + b * stride_ob \
        + d * stride_od \
        + offs_i[:, None] * stride_oi \
        + offs_j[None, :] * stride_oj

    tl.store(out_tile_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]       (bool or float, 1 for keep)
        - weights      : dict of learned parameters
        - config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    output : torch.Tensor
        [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]  fp32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present).  mask is broadcast on the hidden dim.
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction with Triton
    # ------------------------------------------------------------------
    # Cast to float16 for the kernel (still accumulate in fp32)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float16,
                      device=device)

    # Triton launch parameters
    BLOCK_M = 64   # rows  of A
    BLOCK_N = 64   # rows  of B (i.e. cols of output)
    BLOCK_K = 64   # reduction tile size

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        B * hidden_dim,                   # (batch, hidden) combined
    )

    # stride extraction (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left_f16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_f16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    tri_mul_kernel[grid](
        left_f16, right_f16, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # back to fp32
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final Linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    out = out * out_gate                       # broadcast on hidden dim
    out = F.linear(out, weights["to_out.weight"])   # back to `dim`

    return out
