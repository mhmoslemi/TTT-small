    pid2 = tl.program_id(2)  # col tile

"""
TriMul (outgoing) forward pass – fast Triton implementation

The heavy part of the TriMul operator is the 3‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication: for every batch `b` and hidden
dimension `d` we need `out_d = left_d @ right_dᵀ` (both are N×N matrices).
The kernel below performs a fused batched GEMM on the flattened
(batch*hidden) dimension using fp16 inputs and a fp32 accumulator.
All other operations (layer‑norms, linear projections, gating) are carried
out with regular PyTorch kernels for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape: [B*H, N, N], fp16)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,                      # matrix size (N×N)
    stride_batch,           # N*N  (distance between two batches)
    BLOCK_M: tl.constexpr,  # tile size in M dimension
    BLOCK_N: tl.constexpr,  # tile size in N dimension
    BLOCK_K: tl.constexpr,  # tile size in K dimension
):
    pid_batch = tl.program_id(0)      # batch*hidden id
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointer for this batch (flattened batch*hidden dimension)
    batch_offset = pid_batch * stride_batch
    a_base = a_ptr + batch_offset
    b_base = b_ptr + batch_offset
    c_base = c_ptr + batch_offset

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load a tile of A (row‑major) and B (we need Bᵀ, so we load B with
        # swapped indices: B[k, n] becomes Bᵀ[n, k]).
        a = tl.load(
            a_base + (offs_m[:, None] * N) + offs_k[None, :],
            mask=(offs_m[:, None] < N) & (offs_k[None, :] < N),
            other=0.0,
        )
        b = tl.load(
            b_base + (offs_k[:, None] * N) + offs_n[None, :],
            mask=(offs_k[:, None] < N) & (offs_n[None, :] < N),
            other=0.0,
        )
        # fp16 input, fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_base + (offs_m[:, None] * N) + offs_n[None, :],
        c,
        mask=(offs_m[:, None] < N) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        - mask          : torch.Tensor  [B, N, N]      (bool or float)
        - weights       : dict of torch.Tensor (model parameters)
        - config        : dict with keys
                "dim", "hidden_dim", "nomask" (bool)
    Returns
    -------
    torch.Tensor   [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply optional mask (mask is broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape                     # B, N, N, hidden_dim
    total_batches = B * H                       # flatten batch & hidden

    # reshape to [B*H, N, N] – need contiguous storage
    left = left.reshape(total_batches, N, N).contiguous()
    right = right.reshape(total_batches, N, N).contiguous()

    # output buffer
    out_fp16 = torch.empty_like(left)

    # ------------------------------------------------------------------
    # Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    # stride_batch = N * N (distance between two batch*hidden slices)
    _batched_gemm_kernel[grid](
        left,
        right,
        out_fp16,
        N,
        N * N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden] and cast to FP32
    # ------------------------------------------------------------------
    out = out_fp16.reshape(B, N, N, H).to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim) + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
