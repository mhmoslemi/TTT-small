"""
TriMul (outgoing) – Triton‑fused implementation
================================================

The heavy part of the TriMul forward pass is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every channel *d*.
The kernel below computes this contraction in a fused GEMM‑style
loop over *k*, tiling the (i,j) plane and the reduction dimension.
All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
are kept in PyTorch for simplicity.

The kernel processes one ``(batch, hidden_channel)`` pair per program
instance, iterating over K‑tiles and accumulating a ``BLOCK_M × BLOCK_N``
tile of the output.  The final result is stored in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the 4‑D einsum (the “core” of TriMul)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # tensor meta‑data
    B, N, H,          # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    One program instance works on a fixed (b, h) pair and a tile
    (i0:i0+BLOCK_M, j0:j0+BLOCK_N) of the output matrix.
    """
    pid0 = tl.program_id(0)               # 2‑D grid over i‑tiles × j‑tiles
    pid1 = tl.program_id(1)               # linear over (batch * hidden)

    # ------------------------------------------------------------------
    # decode program ids
    # ------------------------------------------------------------------
    # (b, h) for this thread
    b = pid1 // H
    h = pid1 % H

    # i‑tile and j‑tile indices
    tiles_i = tl.cdiv(N, BLOCK_M)
    tiles_j = tl.cdiv(N, BLOCK_N)

    i_tile = pid0 // tiles_j
    j_tile = pid0 % tiles_j

    i0 = i_tile * BLOCK_M
    j0 = j_tile * BLOCK_N

    # ------------------------------------------------------------------
    # range vectors
    # ------------------------------------------------------------------
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)
    k_idx = tl.arange(0, BLOCK_K)

    # mask for boundary handling
    mask_i = i_idx < N
    mask_j = j_idx < N

    # ------------------------------------------------------------------
    # accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k‑tiles
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k = k0 + k_idx
        mask_k = k < N

        # --------------------------------------------------------------
        # load left:  [BLOCK_M, BLOCK_K]
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_idx[:, None] * stride_left_i
            + k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # load right (transposed):  [BLOCK_K, BLOCK_N]
        #   we need right[j,k,h] → we load as right_T[k,j,h]
        # --------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k[:, None] * stride_right_k
            + j_idx[None, :] * stride_right_j
            + h * stride_right_h
        )
        right_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # GEMM over the K‑tile
        # --------------------------------------------------------------
        acc += tl.dot(left, right_T)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_idx[:, None] * stride_out_i
        + j_idx[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (weight/bias are 1‑D)
# ----------------------------------------------------------------------
def fused_layernorm(x, weight, bias, eps=1e-5):
    # x: (..., C)  float32
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]        (bool/float)
        - weights      : dict of parameter tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    out : torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = fused_layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # Linear = x @ W.T   (W shape [hidden, dim])
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply pairwise mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core Triton kernel:  out_core[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # allocate result tensor
    out_core = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements) for contiguous tensors
    # left/right/out have shape [B, N, N, H]  (contiguous C‑order)
    stride_left = left.stride()   # (N*N*H, N*H, H, 1)
    stride_right = right.stride()
    stride_out = out_core.stride()

    # tile sizes – tuned for Hopper (can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),   # i‑tile × j‑tile
        B * hidden,                                         # (b, h) dimension
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_core,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden) + gating
    # ------------------------------------------------------------------
    out_norm = fused_layernorm(
        out_core,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out_norm, weights["to_out.weight"])

    return out
