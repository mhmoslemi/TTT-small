    left_ptr, right_ptr, out_ptr,
    B, N, D,

    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    pid_batch = tl.program_id(1)  # index over B*D

    # compute batch and channel indices
    b = pid_batch // D
    d = pid_batch % D

    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_n = pid % num_pid_n

    mask_m = offs_m < N
    mask_n = offs_n < N

    out_base = out_ptr + b*stride_out_b + d*stride_out_d

"""
TriMul “outgoing” implementation (AlphaFold‑3).

The heavy N³ component
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication per hidden channel:
    out[b, d] = left[b, d] @ right[b, d].T    (N×N) @ (N×N).

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(because they are cheap) and fuse the N³ GEMM in a single Triton kernel.
The kernel works for any batch (B≤2) and hidden dimension (D≤384) and
supports both masked / un‑masked inputs.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for each (batch, hidden_channel)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, D,                                     # logical sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # 1‑D grid (program_id 0) encodes the (M‑block, N‑block) pair,
    # 2‑D grid (program_id 1) encodes (batch, hidden channel) pair.
    # --------------------------------------------------------------
    pid = tl.program_id(0)          # combines M‑block & N‑block
    pid_batch = tl.program_id(1)    # indexes B*D

    # batch / channel index
    b = pid_batch // D
    d = pid_batch % D

    # block indices for output matrix (M = i, N = j)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this (b, d) slice
    # left : [B, N, N, D]   with layout (b, i, k, d)
    # right: [B, N, N, D]   with layout (b, j, k, d)
    # out  : [B, D, N, N]   (we store as (b, d, i, j) for easier strides)
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in BLOCK_K tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:  (i,k)  → strides (i, k)
        ptrs_left = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(ptrs_left,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right block needs to be transposed: we want (k,j)
        # right layout is (j, k) → swap strides when loading
        ptrs_right = right_base + offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_k
        b = tl.load(ptrs_right,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul forward pass with a fused Triton kernel for the cubic
    pairwise multiplication.

    Args
    ----
    data : (input, mask, weights, config)
        * input   – [B, N, N, dim]  (float32 or bfloat16)
        * mask    – [B, N, N]       (bool / 0‑1 float)
        * weights – dict of all linear / layer‑norm parameters
        * config  – {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    def linear_last_dim(t, w):
        # w: [out, in]
        return F.linear(t, w)

    left  = linear_last_dim(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = linear_last_dim(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Transpose to [B, hidden, N, N] for the kernel
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # allocate output buffer in the same layout
    out_buf = torch.empty_like(left, dtype=torch.float32, device=device)  # (B,H,N,N)

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute strides (in elements) for each tensor
    stride_left_b, stride_left_h, stride_left_i, stride_left_j = left.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_j = right.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out_buf.stride()

    # The kernel expects layout (b, i, k, d) for left, (b, j, k, d) for right.
    # Our tensors are (b, h, i, j).  We map:
    #   i  ↔ i  (row index)
    #   k  ↔ j  (column index of the *second* tensor)
    #   d  ↔ h  (hidden channel)
    # Therefore:
    #   left strides: (b, i, k, d) → (stride_left_b, stride_left_i, stride_left_j, stride_left_h)
    #   right strides (with transpose) need (b, k, j, d) → (stride_right_b, stride_right_j, stride_right_i, stride_right_h)
    # We'll pass them accordingly.

    # Launch a 2‑D grid: (num_blocks_M * num_blocks_N, B*hidden_dim)
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n, B * hidden_dim)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_buf,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_j,   # k ↔ j
        stride_left_d=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_j, # k ↔ j  (we want (k,j) order)
        stride_right_k=stride_right_i, # j ↔ i
        stride_right_d=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Re‑transpose back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_buf.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 8. Final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # gating
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out.to(torch.float32)
