"""
tri_mul_outgoing.py
-------------------

Implementation of the “outgoing” TriMul operator (AlphaFold‑3) with the
heavy pair‑wise multiplication (einsum) off‑loaded to a Triton kernel.
All other linear / layer‑norm / gating steps stay in PyTorch for
readability and to keep the launch overhead low.

The kernel computes, for each batch * hidden_dim slice,
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
which is exactly a batched matrix‑multiply of shape (N,N) per hidden
dimension.  The computation is carried out in FP16 inside the kernel
and accumulated in FP32 for numerical stability; the result is stored
back as FP16 and cast to FP32 before the final layer‑norm and linear
projection.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,            # [B, N, N, D]  (FP16)
    right_ptr,           # [B, N, N, D]  (FP16)
    out_ptr,             # [B, N, N, D]  (FP16)
    B, N, D,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Compute one tile of the output matrix for a given (batch, hidden_dim)
    slice.  The operation performed is

        out[i, j] = Σₖ left[i, k] * right[j, k]   (dot over k)

    where left and right are 2‑D slices of shape (N, N) for a fixed
    hidden dimension d.
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_bd = tl.program_id(2)         # packed (batch, d)

    batch = pid_bd // D
    d = pid_bd % D

    # ------------------------------------------------------------------
    # Offsets within the N×N matrix
    # ------------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)   # i indices
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)   # j indices
    offs_k = tl.arange(0, TILE_K)                   # k indices (inside the loop)

    # mask for out‑of‑range rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the first tile of A and B (both FP16)
    #   A : left[batch, i, k, d]
    #   B : right[batch, j, k, d]   (note the j‑first layout)
    # ------------------------------------------------------------------
    a_ptrs = (left_ptr
              + batch   * stride_left_batch
              + d       * stride_left_d
              + offs_m[:, None] * stride_left_i
              + offs_k[None, :] * stride_left_k)

    b_ptrs = (right_ptr
              + batch   * stride_right_batch
              + d       * stride_right_d
              + offs_n[:, None] * stride_right_j
              + offs_k[None, :] * stride_right_k)

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for start_k in range(0, tl.cdiv(N, TILE_K)):
        # load tiles – mask out-of‑range elements with 0.0
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k + start_k * TILE_K < N),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & (offs_k + start_k * TILE_K < N),
                    other=0.0)

        # b is (TILE_N, TILE_K); we need (TILE_K, TILE_N) for tl.dot
        b_T = tl.transpose(b, (1, 0))

        # GEMM tile
        acc += tl.dot(a, b_T)          # (TILE_M,TILE_K)·(TILE_K,TILE_N) → (TILE_M,TILE_N)

        # advance pointers to the next K‑tile
        a_ptrs += TILE_K * stride_left_k
        b_ptrs += TILE_K * stride_right_k

    # ------------------------------------------------------------------
    # Write back the result (still FP16)
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + batch * stride_out_batch
                + d     * stride_out_d
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper – a thin wrapper around torch.nn.functional.layer_norm that
# works with pre‑extracted weight / bias tensors.
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    return F.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32, on CUDA)
            - mask         : [B, N, N]        (float32 or bool, on CUDA)
            - weights      : dict name → torch.Tensor
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) – mask is broadcasted over the hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.unsqueeze(-1)                       # [B, N, N, 1]
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of a linear) → same shape
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the expensive einsum
    # ------------------------------------------------------------------
    # Cast to FP16 for the kernel (the kernel accumulates in FP32)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, device=device)   # [B,N,N,hidden_dim]

    # strides (in elements, not bytes) for the 4‑D tensors
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out_fp16.stride()

    # Block sizes – tuned for Hopper (SM90) and the given N range
    TILE_M = 32
    TILE_N = 32
    TILE_K = 32

    grid = (
        triton.cdiv(N, TILE_M),          # program_id(0) – rows i
        triton.cdiv(N, TILE_N),          # program_id(1) – cols j
        B * hidden_dim,                  # program_id(2) – (batch, hidden)
    )

    _trimul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        TILE_M=TILE_M, TILE_N=TILE_N, TILE_K=TILE_K,
    )

    # Convert back to FP32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Final layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Multiply by the output gate (same shape)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
