"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be rewritten as a batched matrix‑multiply after transposing the
second pairwise dimension of `right`:

    out[b,i,j,d] = ( left[b,:, :, d] ) @ ( right[b,:, :, d].transpose(1,2) )

The kernel below computes this product for a single batch element *and* a
single hidden channel (dimension `d`).  The outer loops over batches and
channels are expressed in the launch grid, while each kernel tiles the
`i` and `j` axes (M×N) and iterates over the reduction dimension `k` (K) in
small blocks.  All computations are done in float16 for bandwidth,
accumulated in float32 and finally written back as float16.  The rest of the
TriMul module (layer‑norms, linear projections, gating, final projection) is
performed with regular PyTorch operations.

The final output is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: one batch element × one hidden channel × a tile of (i,j)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shape parameters
    B, N, H,
    # strides for left: (b, i, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (original layout b, j, k, h)
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    # strides for out (b, i, j, h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i:j, h] = left[b,i:k, h] @ right[b,k:j, h] (k summed)."""
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decode the 4‑D grid: (batch, hidden, block_i, block_j)
    # ------------------------------------------------------------------
    num_pid_h = H                     # one program per hidden channel
    num_pid_m = tl.cdiv(N, BLOCK_M)   # tiles along i
    num_pid_n = tl.cdiv(N, BLOCK_N)   # tiles along j

    h = pid % num_pid_h
    tmp = pid // num_pid_h
    b = tmp // (num_pid_m * num_pid_n)
    tmp2 = tmp % (num_pid_m * num_pid_n)
    block_i = tmp2 // num_pid_n
    block_j = tmp2 % num_pid_n

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left[b, i, k, h]  (shape M×K) -------------------------
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- right[b, k, j, h]  (shape K×N) – note the transpose -----
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k   # k dimension (original third dim)
            + offs_j[None, :] * stride_r_j   # j dimension (original second dim)
            + h * stride_r_h
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- accumulate ------------------------------------------------
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Store the result (convert back to FP16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with the heavy
    pairwise multiplication performed by a Triton kernel.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N] (bool or float) – may be all‑ones
        - weights      : dict of torch tensors (already on device)
        - config       : dict with keys “dim”, “hidden_dim”, “nomask”

    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 for the kernel (bandwidth saving)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Allocate output buffer (still FP16)
    # ------------------------------------------------------------------
    out_fp16 = torch.empty((B, N, N, hidden),
                           device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 7️⃣ Prepare strides for kernel launch
    # ------------------------------------------------------------------
    # left:  (B, N, N, H) -> strides (b,i,k,h)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    # right: (B, N, N, H) original layout (b, j, k, h)
    s_r_b, s_r_j, s_r_k, s_r_h = right.stride()
    # out:   (B, N, N, H) -> strides (b,i,j,h)
    s_o_b, s_o_i, s_o_j, s_o_h = out_fp16.stride()

    # ------------------------------------------------------------------
    # 8️⃣ Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32                         # fits well in shared memory on H200

    num_blocks_i = triton.cdiv(N, BLOCK_M)
    num_blocks_j = triton.cdiv(N, BLOCK_N)

    # each program = one (batch, hidden) pair × one (i‑tile, j‑tile)
    total_programs = B * hidden * num_blocks_i * num_blocks_j
    grid = (total_programs,)

    tri_mul_kernel[grid](
        left, right, out_fp16,
        B, N, hidden,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_k, s_r_j, s_r_h,   # note swapped K/J order for transposed view
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Convert back to FP32 for the remaining ops
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 10️⃣ Final LayerNorm + gating + linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate               # broadcast over (B,N,N,hidden)
    out = F.linear(out, weights["to_out.weight"])

    return out
