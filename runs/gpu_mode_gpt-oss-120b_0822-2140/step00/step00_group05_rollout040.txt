"""
TriMul “outgoing” implementation.
The heavy part is the three‑dimensional einsum:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each batch b and hidden dimension d this is a matrix multiply
(L = left[..., :, :, d]) (N×N)  ×  (R = right[..., :, :, d])ᵀ (N×N).
We fuse all those multiplications in a single Triton kernel that
processes (b,d) together (M = B*H independent mat‑muls) with
typical block sizes (64×64) and accumulates in fp32 for accuracy.
All other ops (LayerNorm, Linear, gating, final LayerNorm & Linear)
are performed with PyTorch – they are inexpensive compared with the
N³ work of the einsum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched matmul L (N×N) × Rᵀ (N×N) for every (batch, hidden)
# ------------------------------------------------------------------
@triton.jit
def batched_matmul_T_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # dimensions
    B, N, H,
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program ids
    # pid0: combined batch*hidden index (M = B*H)
    # pid1: block row (i coordinate)
    # pid2: block col (j coordinate)
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # decode batch and hidden dim
    b = pid0 // H          # batch index
    d = pid0 % H           # hidden dimension index

    # block offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundary handling
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the (b,d) slice
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base = out_ptr + b * stride_out_b + d * stride_out_d

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # Right matrix is accessed transposed, i.e. we need right[j, k] → we
    # swap the k and j strides.
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑tile offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load L block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
            (offs_i[:, None] * stride_left_i) + \
            (offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # load R block transposed: we need Rᵀ[k, j] → original R[j, k]
        # original strides: (stride_right_j for j, stride_right_k for k)
        # after transposition we treat k as first dim, j as second:
        right_ptrs = right_base + \
            (offs_k[:, None] * stride_right_j) + \
            (offs_j[None, :] * stride_right_k)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float32)

        # dot product on K dimension
        acc += tl.dot(left, right)

    # write back, cast to fp16 (the rest of the pipeline expects fp16)
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------------
# Host‑side entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast forward‑only implementation of the TriMul “outgoing” operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N]   (bool/float)
            - weights: dict of torch.Tensor containing all learnable params
            - config:  dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (binary 0/1)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched MatMul via Triton (L × Rᵀ)
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (still accurate enough because we
    # accumulate in fp32 inside the kernel)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # same shape [B,N,N,hidden]

    # strides (contiguous layout)
    s_left = left_fp16.stride()   # (N*N*H, N*H, H, 1)
    s_right = right_fp16.stride()
    s_out = out_fp16.stride()

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden_dim, 1),          # pid0 = batch*hidden
        triton.cdiv(N, BLOCK_M),                # pid1 = row blocks
        triton.cdiv(N, BLOCK_N),                # pid2 = col blocks
    )

    batched_matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Convert back to fp32 for the remaining pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Post‑processing LayerNorm, gating and final Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
