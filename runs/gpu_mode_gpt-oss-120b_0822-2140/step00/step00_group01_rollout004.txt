"""
TriMul “outgoing” implementation with a fused Triton kernel.

The heavy part of the module is the Einstein‑summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix‑multiply for every hidden dimension
d (and every batch).  We rewrite the three‑dimensional tensors as
    L  = left.permute(0,3,1,2)   # (B, D, N, N)  – rows = i , cols = k
    R  = right.permute(0,3,2,1)  # (B, D, N, N)  – rows = k , cols = j
and compute   out_perm = matmul(L, R)   # (B, D, N, N)

The kernel `batched_gemm_kernel` performs many independent N×N → N×N
matrix multiplications (one per (batch, hidden_dim) pair) using
tiling (BLOCK_M × BLOCK_N) and a K‑tile of size BLOCK_K.
All other linear‑layers, LayerNorms and gates are carried out in
regular PyTorch – they are cheap compared with the N³ work.
The final output is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel : batched GEMM (C = A @ B)  with A,B,C : (B*D, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix sizes (all = seq_len)
    stride_ablk, stride_am, stride_ak,   # strides of A
    stride_bblk, stride_bk, stride_bn,   # strides of B
    stride_cblk, stride_cm, stride_cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch*hidden dimension

    # ---------------------- tile offsets -----------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds (necessary for the last partial block)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------- pointers -------------------------------
    a_ptrs = A + pid_b * stride_ablk + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B + pid_b * stride_bblk + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ---------------------- accumulation ----------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        # dot‑product accumulation
        acc += tl.dot(a, b)

        # advance pointers in K dimension
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---------------------- write back -------------------------------
    c_ptrs = C + pid_b * stride_cblk + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑fused
    batched matrix multiplication.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor, shape [B, N, N, dim]
            mask   – torch.Tensor, shape [B, N, N]  (may be all‑ones)
            weights – dict with the ten model parameters
            config  – dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
        eps=eps,
    )

    # ---------- Linear projections (bias‑free) ----------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(dtype),   # (hidden_dim, dim)
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(dtype),  # (hidden_dim, dim)
        bias=None,
    )

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype), bias=None)
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype), bias=None)
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype), bias=None)
    )

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for the batched GEMM ----------
    #   left : (B,N,N,hidden) -> (B,hidden,N,N) rows=i, cols=k
    #   right: (B,N,N,hidden) -> (B,hidden,N,N) rows=k, cols=j  (transpose k<->j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, D, N, N)

    B, D, N, _ = left_perm.shape
    total_batches = B * D                                    # number of independent GEMMs

    # reshape to (total_batches, N, N)
    A = left_perm.reshape(total_batches, N, N)
    B_mat = right_perm.reshape(total_batches, N, N)
    C = torch.empty_like(A)

    # ------ launch Triton kernel ---------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),            # rows
        triton.cdiv(N, BLOCK_N),            # cols
        total_batches,                      # batch*hidden
    )

    batched_gemm_kernel[grid](
        A, B_mat, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),    # A strides
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),  # B strides
        C.stride(0), C.stride(1), C.stride(2),    # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, D, N, N) then to original layout (B,N,N,D)
    out_perm = C.reshape(B, D, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # ---------- Output LayerNorm, gate and final linear ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
        eps=eps,
    )
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype),    # (dim, hidden_dim)
        bias=None,
    )
    return out.to(torch.float32)
