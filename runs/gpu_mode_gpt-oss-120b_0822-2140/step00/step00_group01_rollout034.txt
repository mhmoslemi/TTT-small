"""
TriMul “outgoing” kernel – highly‑optimized Triton implementation of the
core N³ einsum:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The surrounding LayerNorm / Linear / gating operations are kept in
PyTorch for clarity, but the expensive O(N³·D) step is off‑loaded to a
custom Triton kernel.  The kernel works for arbitrary batch size (B≤2),
sequence length N (up to 1024) and hidden dimension D (≤128).  It
processes N in 32‑wide tiles and D in 32‑wide tiles and uses `tl.dot`
to perform the inner‑k reduction efficiently.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # *B,N,N,D   left tensor
    right_ptr,         # *B,N,N,D   right tensor
    out_ptr,           # *B,N,N,D   output tensor
    B, N, D,
    # left strides
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    # right strides (same layout as left)
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    # out strides
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Compute the TriMul einsum for a tile (i_tile, j_tile, batch)."""
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # ------------------------------------------------------------------- #
    # tile offsets
    # ------------------------------------------------------------------- #
    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator  (BLOCK_N, BLOCK_N, BLOCK_D)
    acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # loop over k dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------- #
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------- #
        # load left: shape (BLOCK_N, BLOCK_K, BLOCK_D)
        # --------------------------------------------------------------- #
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_batch
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (i,k,d)

        # --------------------------------------------------------------- #
        # load right transposed: shape (BLOCK_K, BLOCK_N, BLOCK_D)
        # we want right[j,k,d] → (k,j,d) for the dot product
        # --------------------------------------------------------------- #
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_batch
            + offs_k[:, None, None] * stride_right_k   # k as first dim
            + offs_j[None, :, None] * stride_right_i   # j as second dim
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None],
            other=0.0,
        )  # (k,j,d)

        # --------------------------------------------------------------- #
        # For each d‑slice compute a matrix‑matrix product:
        #   acc[:,:,d] += left[:,:,d] @ right[:,:,d].T
        # --------------------------------------------------------------- #
        for d in range(0, BLOCK_D):
            # slice out the d‑th channel
            l = left[:, :, d]          # (BLOCK_N, BLOCK_K)
            r = right[:, :, d]         # (BLOCK_K, BLOCK_N)
            # standard GEMM over the K dimension
            acc[:, :, d] += tl.dot(l, r)   # (BLOCK_N, BLOCK_N)

    # ------------------------------------------------------------------- #
    # store the result
    # ------------------------------------------------------------------- #
    d_off = tl.arange(0, BLOCK_D)
    mask_d = d_off < D

    out_ptrs = (
        out_ptr
        + pid_b * stride_out_batch
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + d_off[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )

# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton kernel
    for the N³ matrix multiplication.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]   (float32)
            - mask:         [B, N, N]        (float32 or bool)
            - weights: dict of all Linear/LayerNorm parameters
            - config:  dict with keys:
                * dim, hidden_dim, nomask (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free) -> shape [B,N,N,hidden]
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the last dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear + element‑wise multiplication)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Cast to FP16 for the heavy N³ computation
    # ------------------------------------------------------------------- #
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, D = left.shape                     # D == hidden_dim
    out = torch.empty_like(left, dtype=torch.float16)  # [B,N,N,D]

    # ------------------------------------------------------------------- #
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------- #
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 32

    # strides in the contiguous layout (B,N,N,D)
    stride_left = left.stride()    # (N*N*D, N*D, D, 1)
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (
        B,
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        # left strides
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        # right strides
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        # out strides
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------- #
    # 7. Remaining post‑processing (LayerNorm, gating, final Linear)
    # ------------------------------------------------------------------- #
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
