def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, H,
                  stride_left_bn, stride_left_bk, stride_left_nk, stride_left_h,
                  stride_right_bn, stride_right_bk, stride_right_nk, stride_right_h,
                  stride_out_bn, stride_out_bi, stride_out_bj, stride_out_h,
                  BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute offsets
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k_offset = tl.arange(0, BLOCK_K)
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N

    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + k_offset
        cur_mask_k = cur_k < N
        # load left: shape (BLOCK_I, BLOCK_K, H)
        left_ptrs = left_ptr + pid_b * stride_left_bn \
                    + i_offset[:, None, None] * stride_left_bi \
                    + cur_k[None, :, None] * stride_left_nk \
                    + tl.arange(0, H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0)

        # load right: shape (BLOCK_J, BLOCK_K, H)
        right_ptrs = right_ptr + pid_b * stride_right_bn \
                     + j_offset[:, None, None] * stride_right_bi \
                     + cur_k[None, :, None] * stride_right_nk \
                     + tl.arange(0, H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None] & cur_mask_k[None, :], other=0.0)

        # left: (I, K, H), right: (J, K, H)
        # we want sum over K: acc[i,j,h] += sum_k left[i,k,h] * right[j,k,h]

"""
TriMul “outgoing” kernel (AlphaFold‑3)

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where  left/right ∈ ℝ^{B×N×N×H}.  For each hidden channel *d* this is an
N×N Gram‑matrix product over the *k* dimension.  The kernel below computes the
reduction over *k* while keeping the hidden dimension in registers, i.e. it
evaluates the above equation without materialising the O(N³) intermediate
tensor.

The surrounding LayerNorms and linear projections are performed in PyTorch
(because they are cheap compared to the O(N³) work).  The kernel works for any
batch size B∈{1,2}, sequence length N up to 1024 and hidden size H up to 384.
All tensors are assumed to be on the same CUDA device and the final result is
returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel
# -------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          #  B,N,N,H  (input after projection & gating)
    right_ptr,         #  B,N,N,H
    out_ptr,           #  B,N,N,H  (output before final norm/out‑proj)
    B, N, H,
    # strides of the three tensors (contiguous layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # program IDs : (batch, tile_i, tile_j)
    # ---------------------------------------------------------------
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # ---------------------------------------------------------------
    # tile offsets
    # ---------------------------------------------------------------
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_base < N
    j_mask = j_base < N

    # accumulator: (BLOCK_I, BLOCK_J, H) in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # ----------------------------------------------------------------
    # loop over the reduction dimension k
    # ----------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # ----------------------------------------------------------------
        # load left  : shape (BLOCK_I, BLOCK_K, H)
        # ----------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + i_base[:, None, None] * stride_left_i
            + cur_k[None, :, None] * stride_left_k
            + tl.arange(0, H)[None, None, :] * stride_left_h
        )
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)                     # fp32

        # ----------------------------------------------------------------
        # load right : shape (BLOCK_J, BLOCK_K, H)
        # ----------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + j_base[:, None, None] * stride_right_i
            + cur_k[None, :, None] * stride_right_k
            + tl.arange(0, H)[None, None, :] * stride_right_h
        )
        right = tl.load(right_ptrs,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0)                    # fp32

        # ----------------------------------------------------------------
        # acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]
        #   -> broadcast over i/j then reduce over k
        # ----------------------------------------------------------------
        # left  : (I, K, H) → (I, 1, K, H)
        # right : (J, K, H) → (1, J, K, H)
        prod = left[:, None, :, :] * right[None, :, :, :]      # (I,J,K,H)
        acc += tl.sum(prod, axis=2)                           # sum over K

    # ----------------------------------------------------------------
    # write result
    # ----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + i_base[:, None, None] * stride_out_i
        + j_base[None, :, None] * stride_out_j
        + tl.arange(0, H)[None, None, :] * stride_out_h
    )
    tl.store(out_ptrs,
             acc,
             mask=i_mask[:, None, None] & j_mask[None, :, None])


# -------------------------------------------------------------------------
# Host entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the “outgoing” TriMul operator.
    Args
    ----
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor, shape [B, N, N, D]       (float32)
        mask    – torch.Tensor, shape [B, N, N] (bool/int) or None
        weights – dict with all model parameters
        config  – dict with entries “dim”, “hidden_dim”
    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (called “hidden_dim”)

    device = input_tensor.device
    dtype = torch.float32

    # -----------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim = D)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑free) + mask
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        # mask shape [B,N,N] → broadcast over hidden dim
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid) + element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core TriMul – Triton kernel
    # -----------------------------------------------------------------
    # allocate output buffer (float32)
    out = torch.empty((B, N, N, H), dtype=dtype, device=device)

    # strides of contiguous tensors (B,N,N,H) in elements
    stride_b, stride_i, stride_k, stride_h = (
        N * N * H,
        N * H,
        H,
        1,
    )

    # kernel launch parameters
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 16                     # controls k‑loop granularity

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        # strides (in elements)
        stride_left_b=stride_b,
        stride_left_i=stride_i,
        stride_left_k=stride_k,
        stride_left_h=stride_h,
        stride_right_b=stride_b,
        stride_right_i=stride_i,
        stride_right_k=stride_k,
        stride_right_h=stride_h,
        stride_out_b=stride_b,
        stride_out_i=stride_i,
        stride_out_j=stride_k,        # note: j dimension steps like k‑dim
        stride_out_h=stride_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 5. Final LayerNorm + out‑gate + linear projection back to D
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure final dtype is float32 (the kernel already wrote fp32)
    return out.to(torch.float32)
