"""
TriMul outgoing kernel (AlphaFold3 style)

We perform the heavy N³ “einsum”:
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

The surrounding LayerNorm / Linear / gating logic stays in PyTorch.
The kernel is a blocked batch‑matrix‑multiply where every (batch, hidden)
pair is treated as an independent matrix‑multiply of shape (N, N)·(N, N)ᵀ.
Only the result (float32) is written back, everything else runs in FP16/BF16
for higher throughput on Hopper (sm_90).
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# --------------------------------------------------------------
# Triton kernel ------------------------------------------------
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    Grid dimensions:
        pid0 : batch * hidden   (b * H + h)
        pid1 : block row i
        pid2 : block col j
    """
    # ------------------------------------------------------------------
    # decode batch & hidden from pid0
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid0 // H          # batch
    h = pid0 % H           # hidden channel

    # ------------------------------------------------------------------
    # block offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # k‑loop (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (i, k)  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

        # right: (k, j)  -> shape (BLOCK_K, BLOCK_N)  (note k first!)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # block‑dot over the K dimension
        acc += tl.dot(left, right)      # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Host helper --------------------------------------------------
# --------------------------------------------------------------
def _launch_trimap(out, left, right):
    """
    Launch the Triton kernel.  All tensors must be contiguous and on CUDA.
    """
    B, N, _, H = left.shape
    # strides in element counts (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[2], s_right[1], s_right[3],   # note: right uses j‐axis stride as stride_right_j
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )


# --------------------------------------------------------------
# Main entry point ---------------------------------------------
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for
    the N³ contraction.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6  # LayerNorm epsilon

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask handling (optional)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # allocate output of the contraction (still hidden_dim)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Cast to fp16 for the kernel (Hopper is very fast in fp16)
    left_k = left.to(torch.float16).contiguous()
    right_k = right.to(torch.float16).contiguous()
    out_k = out.to(torch.float16)

    _launch_trimap(out_k, left_k, right_k)

    # Cast back to fp32
    out = out_k.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
