def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden = config["hidden_dim"]; B,N,N2, C = input_tensor.shape
    assert N == N2, etc

    # LayerNorm
    x = (input_tensor - (input_tensor * 0))  # actually use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # No bias in linear layers.

    # linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # mask
    if mask is None:
        mask = torch.ones(B, N, N, device=input_tensor.device, dtype=left.dtype)
    mask_ = mask.unsqueeze(-1)
    left = left * mask_
    right = right * mask_

    # gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # transpose right for outgoing
    right_t = right.permute(0,2,1,3).contiguous()

    # allocate output tensor
    out = torch.empty(B, N, N, hidden, dtype=torch.float32, device=input_tensor.device)

    # launch kernel
    stride_left = left.stride()
    stride_right = right_t.stride()
    stride_out = out.stride()
    BLOCK_M=64; BLOCK_N=64; BLOCK_K=64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)
    tri_mul_outgoing_kernel[grid](
        left, right_t, out,
        B, N, hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, D_CH=hidden
    )

    # post processing
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights['to_out.weight'])
    return out

"""
TriMul “outgoing” implementation.

The heavy part of the module is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,k,j,d]

which is a batched matrix‑multiply for every hidden channel *d*.
We compute the LayerNorms, linear projections and gating in PyTorch and
off‑load the contraction to a fused Triton kernel that iterates over the
k‑dimension in blocks, accumulates with `tl.dot` for each hidden channel and
stores the result.  The final Linear + LayerNorm + gating are again done in
PyTorch.  The kernel works for any batch size ≤2, sequence length up to 1024
and hidden dimension up to 384 (the hidden dimension is a compile‑time
constant for the kernel).
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# -------------------------------------------------------------
# Triton kernel: batched “outer” matrix multiply over hidden dim
# -------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # [B, N, N, D]   (i, k, d)
    right_ptr,         # [B, N, N, D]   (k, j, d) – already transposed
    out_ptr,           # [B, N, N, D]   (i, j, d)

    B, N, D,

    # strides (in elements) of the three tensors
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bs, stride_right_k, stride_right_j, stride_right_d,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d,

    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, D_CH: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch dimension
    pid_m = tl.program_id(0)          # i‑block
    pid_n = tl.program_id(1)          # j‑block

    # -----------------------------------------------------------------
    # block offsets for the i and j axes
    # -----------------------------------------------------------------
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator for the output block – shape (M, N, D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_CH), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)

    # -----------------------------------------------------------------
    # iterate over the k‑dimension in blocks
    # -----------------------------------------------------------------
    for kb in range(0, num_k_blocks):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # -------------------------------------------------------------
        # left[b, i, k, d]  → shape (M, K, D)
        # -------------------------------------------------------------
        left_ptrs = left_ptr \
            + pid_b * stride_left_bs \
            + i_offset[:, None, None] * stride_left_i \
            + k_offset[None, :, None] * stride_left_k \
            + tl.arange(0, D_CH)[None, None, :] * stride_left_d

        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )

        # -------------------------------------------------------------
        # right[b, k, j, d]  → shape (K, N, D)    (already transposed)
        # -------------------------------------------------------------
        right_ptrs = right_ptr \
            + pid_b * stride_right_bs \
            + k_offset[:, None, None] * stride_right_k \
            + j_offset[None, :, None] * stride_right_j \
            + tl.arange(0, D_CH)[None, None, :] * stride_right_d

        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None],
            other=0.0,
        )

        # -------------------------------------------------------------
        # For each hidden channel d perform a small matrix multiply:
        #   left[d] : (M, K)   dot   right[d] : (K, N)   → (M, N)
        # -------------------------------------------------------------
        for d in range(0, D_CH):
            left_d  = left[:, :, d]   # (M, K)
            right_d = right[:, :, d]  # (K, N)
            acc[:, :, d] += tl.dot(left_d, right_d)

    # -----------------------------------------------------------------
    # store the accumulated block
    # -----------------------------------------------------------------
    out_ptrs = out_ptr \
        + pid_b * stride_out_bs \
        + i_offset[:, None, None] * stride_out_i \
        + j_offset[None, :, None] * stride_out_j \
        + tl.arange(0, D_CH)[None, None, :] * stride_out_d

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# -------------------------------------------------------------
# Entry point used by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton kernel.

    Args:
        data: (input, mask, weights, config)
            - input : Tensor[B, N, N, dim]  (float32)
            - mask  : Tensor[B, N, N]      (bool / float)
            - weights: dict of all linear / layer‑norm parameters
            - config : dict with keys `dim` and `hidden_dim`

    Returns:
        Tensor[B, N, N, dim]  (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) Layer‑norm on the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, device=input_tensor.device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)   # [B,N,N,1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid of linear maps)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core contraction – Triton kernel
    # -----------------------------------------------------------------
    # transpose right so that its first two axes are (k, j)
    right_t = right.permute(0, 2, 1, 3).contiguous()   # [B, N, N, hidden]

    out = torch.empty(B, N, N, hidden, dtype=torch.float32, device=input_tensor.device)

    # strides in element units (not bytes)
    s_left = left.stride()
    s_right = right_t.stride()
    s_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    tri_mul_outgoing_kernel[grid](
        left,
        right_t,
        out,
        B,
        N,
        hidden,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        D_CH=hidden,
    )

    # -----------------------------------------------------------------
    # 6) Post‑processing: layer‑norm → gating → final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
