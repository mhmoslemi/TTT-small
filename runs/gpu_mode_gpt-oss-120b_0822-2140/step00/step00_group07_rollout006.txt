"""
TriMul “outgoing” implementation (forward only).

The heavy part is the 3‑D contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of matrix multiplications per channel d.
We fuse this contraction in a single Triton kernel while keeping
LayerNorms / Linear / Gates in PyTorch for readability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------
# Triton kernel: batched per‑channel matmul (left @ right.T)
# -------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for a fixed batch element `b` and channel `d`:
        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    The kernel is launched with a 3‑D grid:
        dim0 = B * H          (each program handles a (b,d) pair)
        dim1 = ceil(N / BLOCK_M)   (tiles over i)
        dim2 = ceil(N / BLOCK_N)   (tiles over j)
    """
    pid = tl.program_id(0)                # encodes (b, d)
    b = pid // H
    d = pid % H

    pid_i = tl.program_id(1)              # tile over i dimension
    pid_j = tl.program_id(2)              # tile over j dimension

    # start offsets of the current tile
    i_off = pid_i * BLOCK_M
    j_off = pid_j * BLOCK_N

    # thread‑local range vectors
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # predicates for out‑of‑bounds (needed for the last partial tile)
    i_mask = (i_off + offs_i) < N
    j_mask = (j_off + offs_j) < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over the reduction dimension k
    # ---------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K
        k_mask = (k_off + offs_k) < N

        # left: shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr \
            + b * stride_l_b \
            + d * stride_l_h \
            + (i_off + offs_i)[:, None] * stride_l_i \
            + (k_off + offs_k)[None, :] * stride_l_k
        a = tl.load(a_ptr,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # right (need the transpose over the two N dims):
        # we want right[b, j, k, d] -> index i=j, k stays k
        b_ptr = right_ptr \
            + b * stride_r_b \
            + d * stride_r_h \
            + (j_off + offs_j)[None, :] * stride_r_i \
            + (k_off + offs_k)[:, None] * stride_r_k
        b = tl.load(b_ptr,
                    mask=j_mask[None, :] & k_mask[:, None],
                    other=0.0,
                    dtype=tl.float16)

        # matrix multiply‑accumulate (FP16 inputs → FP32 accumulation)
        acc += tl.dot(a, b).to(tl.float32)

    # ---------------------------------------------------------
    # Store the result tile
    # ---------------------------------------------------------
    c_ptr = out_ptr \
        + b * stride_o_b \
        + d * stride_o_h \
        + (i_off + offs_i)[:, None] * stride_o_i \
        + (j_off + offs_j)[None, :] * stride_o_j
    tl.store(c_ptr, acc, mask=i_mask[:, None] & j_mask[None, :])

# -------------------------------------------------------------
# Helper for the Triton launch
# -------------------------------------------------------------
def _launch_triton_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (FP16/FP32)
    Returns out of shape [B, N, N, H] in float32.
    """
    B, N, _, H = left.shape
    # ensure contiguous and use fp16 for the kernel (good performance on H200)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in element units (not bytes) for each dimension
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # block sizes – tuned for Hopper (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _trimul_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # good default for these shapes
    )
    return out

# -------------------------------------------------------------
# Entry point expected by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module implemented with a
    custom Triton kernel for the core contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input  (dim -> dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim) – no bias
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (shape [B, N, N]) – broadcast on channel dim
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – also dim -> hidden_dim
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton: out = left @ right.T  (per channel)
    # -----------------------------------------------------------------
    out = _launch_triton_matmul(left, right)   # shape [B, N, N, hidden_dim], float32

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim -> hidden_dim)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 7. Apply output gate and final projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
