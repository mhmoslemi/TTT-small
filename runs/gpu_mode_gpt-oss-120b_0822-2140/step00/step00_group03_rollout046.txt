"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be written as a batched matrix‑multiply per hidden channel:
    out[b,i,j,d] = (L[b,d,i,k] @ R[b,d,k,j]).T

We keep the LayerNorms, linear projections and sigmoids in PyTorch
(because they are cheap) and off‑load the reduction over *k* to a
custom Triton kernel.  Each program computes one (batch, i, j) tile and
reduces over the sequence dimension *k* in blocks.  The hidden dimension
is processed in a compile‑time vector (`BLOCK_D`).  The kernel accumulates
in FP32 and writes the result back as FP32.

The final linear projection and output LayerNorm are again done with
torch (they are small compared to the N³ reduction).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # *B*N*N*H   left tensor
    right_ptr,         # *B*N*N*H   right tensor
    out_ptr,           # *B*N*N*H   output
    B, N, H,           # problem size
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_K: tl.constexpr,   # k‑tile size
    BLOCK_D: tl.constexpr,   # hidden‑channel vector size
):
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i index
    pid_j = tl.program_id(2)          # j index

    # --------------------------------------------------------------
    # bounds
    # --------------------------------------------------------------
    offs_i = pid_i
    offs_j = pid_j
    mask_i = offs_i < N
    mask_j = offs_j < N
    # early exit if out of range
    if ~(mask_i & mask_j):
        return

    # vector of hidden channels
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < H

    # accumulator
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension in tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ----- load left[b,i,k,d] -------------------------------------------------
        #   left_ptr + b*lb + i*li + k*lk + d*ld
        left_base = left_ptr \
            + pid_b * stride_lb \
            + offs_i * stride_li \
            + cur_k[:, None] * stride_lk \
            + offs_d[None, :] * stride_ld
        left = tl.load(left_base, mask=mask_k[:, None] & mask_d[None, :], other=0.0)
        # ----- load right[b,j,k,d] ------------------------------------------------
        #   right_ptr + b*rb + j*rj + k*rk + d*rd
        right_base = right_ptr \
            + pid_b * stride_rb \
            + offs_j * stride_rj \
            + cur_k[:, None] * stride_rk \
            + offs_d[None, :] * stride_rd
        right = tl.load(right_base, mask=mask_k[:, None] & mask_d[None, :], other=0.0)

        # element‑wise product & reduction over k
        acc += tl.sum(left * right, axis=0)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_base = out_ptr \
        + pid_b * stride_ob \
        + offs_i * stride_oi \
        + offs_j * stride_oj \
        + offs_d * stride_od
    tl.store(out_base, acc, mask=mask_d)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : (input, mask, weights, config)
        * input   – Tensor [B, N, N, C]  (C == dim)
        * mask    – Tensor [B, N, N]    (binary 0/1, may be all‑ones)
        * weights – dict with all linear / layer‑norm parameters
        * config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor [B, N, N, C]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the feature dimension (C == dim)
    # ------------------------------------------------------------------
    # LayerNorm uses weight & bias (both 1‑D of size C)
    ln_weight = weights["norm.weight"]
    ln_bias   = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=ln_weight, bias=ln_bias)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = torch.einsum('bijk, hk -> bijh', x, weights["left_proj.weight"])
    right = torch.einsum('bijk, hk -> bijh', x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Apply mask if present (mask is B×N×N, broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.unsqueeze(-1)                     # B×N×N×1
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣  Gating scalars (sigmoid of linear transforms of the *original* x)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.einsum('bijk, hk -> bijh', x,
                                           weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.einsum('bijk, hk -> bijh', x,
                                           weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.einsum('bijk, hk -> bijh', x,
                                           weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Core reduction:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    # allocate output tensor (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) – required by Triton
    l_strides = left.stride()          # (B,N,N,H)
    r_strides = right.stride()
    o_strides = out.stride()

    # constants for the kernel
    BLOCK_K = 64          # tile size over the summed dimension k
    BLOCK_D = 32          # vector size for hidden channels (must divide H or be masked)

    grid = (B,
            triton.cdiv(N, 1),   # one program per i
            triton.cdiv(N, 1))   # one program per j

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=hidden_dim,
        stride_lb=l_strides[0], stride_li=l_strides[1],
        stride_lk=l_strides[2], stride_ld=l_strides[3],
        stride_rb=r_strides[0], stride_rj=r_strides[1],
        stride_rk=r_strides[2], stride_rd=r_strides[3],
        stride_ob=o_strides[0], stride_oi=o_strides[1],
        stride_oj=o_strides[2], stride_od=o_strides[3],
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        # important: tell Triton to treat pointers as void*
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣  Output LayerNorm (hidden_dim) + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate                         # gating after LN

    # final linear: hidden_dim → dim
    out = torch.einsum('bijk, hk -> bijh', out,
                       weights["to_out.weight"])

    return out.to(torch.float32)
