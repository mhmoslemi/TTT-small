"""
TriMul “outgoing” implementation with a fused Triton batched matrix‑multiply.

The heavy O(B·C·N³) einsum
    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]
is converted to a batched GEMM per channel:
    left  → (B·C, N, K)   with K = N
    right → (B·C, K, N)   (right is transposed on the last two axes)
    out   → (B·C, N, N)

A single Triton kernel computes the GEMM for all (B·C) batches in one launch.
All other operations (LayerNorm, linear projections, gating, final
normalisation) are performed with PyTorch for simplicity.
The final output is float‑32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply (A @ B)   A:(M,K) B:(K,N) → C:(M,N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,          # strides for A (batch, M, K)
    stride_bk, stride_bn,          # strides for B (batch, K, N)
    stride_cm, stride_cn,          # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets for the block of C we are writing to
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the block in A and B
    a_ptrs = A_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)

        # cast to fp32 for the dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = C_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A : (B*H, N, K)   fp16
    B : (B*H, K, N)   fp16   (already transposed)
    returns C : (B*H, N, N)  fp32
    """
    assert A.is_contiguous() and B.is_contiguous()
    B_dim, N, K = A.shape
    _, K2, N2 = B.shape
    assert K == K2 and N == N2

    C = torch.empty((B_dim, N, N), dtype=torch.float32, device=A.device)

    # strides (in elements)
    stride_am = A.stride(0)
    stride_ak = A.stride(2)
    stride_bk = B.stride(1)
    stride_bn = B.stride(2)
    stride_cm = C.stride(0)
    stride_cn = C.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n,)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass.
    Args:
        data: (input, mask, weights, config)
            input  – [B, N, N, dim]   (float32)
            mask   – [B, N, N]        (float32/float16, 0 or 1)
            weights – dict with all linear / layer‑norm parameters
            config – {'dim': int, 'hidden_dim': int}
    Returns:
        output – [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm expects normalized_shape to be last dim
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # -> (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring channel dimension to second axis and cast to fp16 for speed
    left_t = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, hidden, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous().to(torch.float16) # (B, hidden, N, N) = transposed on last two axes

    # merge batch and channel dimensions
    left_flat = left_t.view(B * hidden, N, N)
    right_flat = right_t.view(B * hidden, N, N)

    # Triton GEMM (fp16 input, fp32 accumulator)
    out_flat = batched_matmul(left_flat, right_flat)   # (B*hidden, N, N) fp32

    # reshape back to original layout
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
