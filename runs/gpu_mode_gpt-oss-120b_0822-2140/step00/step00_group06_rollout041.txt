"""
TriMul (outgoing) kernel implementation.

The heavy part of the TriMul forward is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of H (= hidden_dim) matrix‑multiplications of size N×N.
We fuse this contraction in Triton (fp16 → fp32 accumulation) and keep the
remaining linear / layer‑norm / gating logic in regular PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_fp16(
    a_ptr, b_ptr, c_ptr,                     # pointers
    a_batch_stride, a_row_stride, a_col_stride,  # a strides
    b_batch_stride, b_row_stride, b_col_stride,  # b strides
    c_batch_stride, c_row_stride, c_col_stride,  # c strides
    M, N, K,                                 # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B.T for each batch element.
    A, B, C are 3‑D tensors of shape (batch, M, K), (batch, N, K), (batch, M, N)
    using fp16 loads and fp32 accumulation.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds (needed for the last partial tiles)
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the first element of the tile in A and B
    a_ptrs = (
        a_ptr
        + batch_id * a_batch_stride
        + offs_m[:, None] * a_row_stride
        + offs_k[None, :] * a_col_stride
    )
    b_ptrs = (
        b_ptr
        + batch_id * b_batch_stride
        + offs_k[:, None] * b_row_stride
        + offs_n[None, :] * b_col_stride
    )

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # a : (BLOCK_M, BLOCK_K)   b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)   # fp16→fp32 dot

        # Advance K‑dimension pointers
        a_ptrs += BLOCK_K * a_col_stride
        b_ptrs += BLOCK_K * b_row_stride

    # --------------------------------------------------------------
    # Write back the result (still fp32)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_id * c_batch_stride
        + offs_m[:, None] * c_row_stride
        + offs_n[None, :] * c_col_stride
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing variant) with the heavy contraction
    executed on a Triton kernel.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (already on CUDA)
        - mask        : torch.Tensor [B, N, N]   (bool / float, may be all‑ones)
        - weights     : dict of torch.Tensor containing all model weights
        - config      : dict with keys:
            * "dim"          – input channel dimension
            * "hidden_dim"   – hidden dimension (proj/gate width)
            * "nomask" (opt) – True → mask is ignored
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton matmul:
    #   reshape to (B*hidden, N, N) and cast to fp16 for bandwidth.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, hidden) → (B, hidden, N, N) → (B*hidden, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    left_fp16 = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)   # will hold fp16 result (still fp32 after kernel)

    # ------------------------------------------------------------------
    # Compute strides (in element units, not bytes) for the kernel
    # ------------------------------------------------------------------
    a_batch_stride, a_row_stride, a_col_stride = left_fp16.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_fp16.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_fp16.stride()

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden,                     # batch dimension (each (hidden) slice)
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )

    _batched_matmul_fp16[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_row_stride,
        b_col_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        N,          # M
        N,          # N
        N,          # K (square matrices)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Back‑to original layout, cast to float32 for the remaining ops
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32).view(B, hidden, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
