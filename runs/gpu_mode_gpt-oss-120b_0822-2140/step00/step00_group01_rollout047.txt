"""
TriMul “outgoing” implementation (forward only).

The computational bottleneck is the triple‑wise tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication per hidden dimension:

    out_d = left_d @ right_d.T      (left_d , right_d : N×N)

We fuse this reduction into a single Triton kernel that operates on the
flattened batch×hidden dimension (M = B·H).  All cheap operations
(LayerNorm, linear projections, gating) are performed with PyTorch
(either FP32 or mixed‑precision) and only the O(N³·H) matmul is off‑loaded
to Triton.  The final output is cast back to float32.

The kernel works for the full range of test configurations (B∈{1,2},
N∈{128,…,1024}, dim∈{128,384}, hidden_dim=128) on an NVIDIA H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A,B,C are 3‑D)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                # matrix sizes (M = B*H, N = seq_len, K = seq_len)
    stride_am, stride_ak,   # strides for A (M,K)
    stride_bk, stride_bn,   # strides for B (K,N)
    stride_cm, stride_cn,   # strides for C (M,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col

    # offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + (offs_m[:, None] * stride_am
                          + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + (offs_k[:, None] * stride_bk
                          + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm
                      + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Host‑side helper that launches the kernel
# --------------------------------------------------------------
def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : (M, K, N)   – we treat the last two dims as a matrix (K,N)
    b : (M, K, N)   – already transposed on the last two dims
    returns C : (M, N, N)
    """
    assert a.is_contiguous() and b.is_contiguous()
    M, K, N = a.shape
    # output buffer
    c = torch.empty((M, N, N), dtype=torch.float32, device=a.device)

    # strides (in elements, not bytes)
    stride_am, stride_ak, stride_an = a.stride()
    stride_bk, stride_bn, stride_bb = b.stride()
    stride_cm, stride_cn, stride_cc = c.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    contraction.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : Tensor[bs, N, N, dim]   (float32, on CUDA)
            - mask  : Tensor[bs, N, N]        (float32 or bool)
            - weights: dict with all linear / layer‑norm parameters
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor[bs, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # denoted H in the paper

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on input (dim‑wise)
    # ------------------------------------------------------------------
    # LayerNorm: mean & var are computed over the last dimension only
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after proj: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) – mask is broadcasted on the hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triton‑accelerated contraction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    ↔ out_d = left_d @ right_d.T   for each hidden dimension d
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                           # B, seq_len, hidden_dim
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs to be transposed on the last two axes before flattening
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Triton kernel computes C = left_flat @ right_flat   (both (M,N,N))
    out_flat = _batched_matmul(left_flat, right_flat)   # shape (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm (hidden_dim) + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # out_gate has shape [B,N,N,H]; broadcast multiplication
    out = out * out_gate

    # final linear back to model dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
