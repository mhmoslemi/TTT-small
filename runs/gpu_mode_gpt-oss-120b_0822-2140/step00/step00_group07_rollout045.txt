"""
TriMul – outgoing version
-------------------------

This implementation follows the reference PyTorch module and moves the
dominant O(N³) part (the “einsum” that multiplies the projected pair
representations) to a custom Triton kernel.  All other linear,
normalisation and gate operations stay in PyTorch for simplicity.

The kernel computes, for each (batch, hidden) slice, a matrix product

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is exactly a batched GEMM of shape (B·H, N, N) × (B·H, N, N).
The kernel is written for fp16 compute (the inputs are cast to fp16) and
stores the result in fp32 to keep the final precision identical to the
reference implementation.

The surrounding Python code:
* Layer‑norms and linear layers are performed with `torch.nn.functional`.
* Optional masking is applied.
* Gates are computed with a sigmoid.
* The Triton kernel is launched with a 3‑D grid: (M‑tiles, N‑tiles,
  B·H) where each tile computes a BLOCK_M×BLOCK_N sub‑matrix.
* After the kernel, a second LayerNorm, an output gate and a final
  linear projection finish the forward pass.

The code works for all shapes listed in the statement (batch size 1‑2,
sequence length up to 1024, dim = 128/384, hidden_dim = 128) on an
NVIDIA H200 (sm_90) using Triton 3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication of (B·H, N, N) × (B·H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr,            # pointer to left tensor   (B·H, N, N)
    right_ptr,           # pointer to right tensor  (B·H, N, N)
    out_ptr,             # pointer to output tensor (B·H, N, N)

    B_H, N,              # dimensions
    stride_lbh, stride_lm, stride_lk,   # left strides
    stride_rbh, stride_rk, stride_rn,   # right strides
    stride_obh, stride_om, stride_on,   # output strides

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a single (BLOCK_M, BLOCK_N) tile of the output matrix."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_bh = tl.program_id(2)         # which (batch * hidden) slice

    # ------------------------------------------------------------------
    # Offsets within the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask out-of-bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the three tensors
    # ------------------------------------------------------------------
    left_tile_ptr  = left_ptr \
        + pid_bh * stride_lbh \
        + offs_m[:, None] * stride_lm \
        + offs_k[None, :] * stride_lk

    right_tile_ptr = right_ptr \
        + pid_bh * stride_rbh \
        + offs_k[:, None] * stride_rk \
        + offs_n[None, :] * stride_rn

    out_tile_ptr   = out_ptr \
        + pid_bh * stride_obh \
        + offs_m[:, None] * stride_om \
        + offs_n[None, :] * stride_on

    # ------------------------------------------------------------------
    # Initialise accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Load a BLOCK_M×BLOCK_K block of left and a BLOCK_K×BLOCK_N block of right
        left = tl.load(left_tile_ptr, mask=mask_m[:, None] & (offs_k[None, :] < N), other=0.0)
        right = tl.load(right_tile_ptr, mask=(offs_k[:, None] < N) & mask_n[None, :], other=0.0)

        # FP16→FP32 dot product (triton's dot works on FP16/F32)
        acc += tl.dot(left, right)

        # Increment pointers to the next K‑block
        left_tile_ptr  += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # Write result back
    # ------------------------------------------------------------------
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: (B, N, N, H)  fp32 tensor
    returns: (B, N, N, H) fp32 tensor
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) and cast to fp16 for the kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # strides in elements (not bytes)
    stride_lbh, stride_lm, stride_lk = left_flat.stride()
    stride_rbh, stride_rk, stride_rn = right_flat.stride()
    stride_obh, stride_om, stride_on = out_flat.stride()

    # block sizes – chosen to fit in shared memory on H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    _trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_lbh,
        stride_lm,
        stride_lk,
        stride_rbh,
        stride_rk,
        stride_rn,
        stride_obh,
        stride_om,
        stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    batch‑matrix‑multiply core.

    Args:
        data: (input, mask, weights, config)
            input  – Tensor[B, N, N, dim]   (fp32)
            mask   – Tensor[B, N, N]        (bool/float, may be None)
            weights – dict of pre‑initialised weight tensors
            config – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden, dim)

    left  = F.linear(x, left_proj_w)   # shape [B, N, N, hidden]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – each is a linear layer + sigmoid
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N^3 operation – Triton batched GEMM
    # ------------------------------------------------------------------
    # out shape : [B, N, N, hidden] (float32)
    out = _batched_matmul_fp16(left, right)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden dim), output gate and final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # apply output gate (broadcast over hidden)
    out = out * out_gate

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"]      # (dim, hidden)
    out = F.linear(out, to_out_w)            # [B, N, N, dim]

    return out.to(dtype)
