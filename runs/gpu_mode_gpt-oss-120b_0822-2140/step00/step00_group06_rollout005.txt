"""
TriMul “outgoing” implementation (forward only).

The costly part of the module is the tensor contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden channel *d*.
We fuse the mask, projection and gating in a tiny Triton kernel and
use a single batched‑gemm (torch.bmm) on the reshaped tensors to
perform the N³ contraction efficiently.

The kernel operates on 1‑D flattened views of the tensors:
    out = left * left_gate * mask   (and the same for “right”).
All other linear / layer‑norm steps are performed with the highly
optimized PyTorch primitives.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel: element‑wise fuse = a * b * mask
# --------------------------------------------------------------------------- #
@triton.jit
def _fuse_mul_kernel(
    a_ptr,               # left or right projection          (float32*)
    b_ptr,               # corresponding gate (sigmoid)     (float32*)
    m_ptr,               # mask (already broadcast)         (float32*)
    out_ptr,             # result                           (float32*)
    N_ELEMENTS: tl.constexpr,   # total number of elements
    BLOCK_SIZE: tl.constexpr,   # work per program
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMENTS

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    m = tl.load(m_ptr + offs, mask=mask, other=0.0)

    out = a * b * m
    tl.store(out_ptr + offs, out, mask=mask)


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor [B, N, N, dim]   (float32)
        - mask  : torch.Tensor [B, N, N]       (bool/float, may be all‑ones)
        - weights: dict of pre‑loaded weight tensors
        - config : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack
    # ------------------------------------------------------------------ #
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = inp.shape

    # ------------------------------------------------------------------ #
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    inp_norm = F.layer_norm(inp,
                            normalized_shape=(dim,),
                            weight=norm_weight,
                            bias=norm_bias,
                            eps=1e-5)

    # ------------------------------------------------------------------ #
    # Linear projections (no bias) – use torch.nn.functional.linear
    # ------------------------------------------------------------------ #
    left_proj  = F.linear(inp_norm, weights["left_proj.weight"])    # [B,N,N,H]
    right_proj = F.linear(inp_norm, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(inp_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(inp_norm, weights["right_gate.weight"]))

    # ------------------------------------------------------------------ #
    # Prepare mask: broadcast to [B,N,N,H] (float32)
    # ------------------------------------------------------------------ #
    if nomask:
        mask_b = torch.ones((B, N, N, hidden_dim),
                            dtype=inp.dtype,
                            device=inp.device)
    else:
        mask_b = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
    # ------------------------------------------------------------------ #
    # Fuse projection, gating and mask with Triton (left & right separately)
    # ------------------------------------------------------------------ #
    def _fuse(tensor_proj, tensor_gate):
        out = torch.empty_like(tensor_proj)
        total_el = tensor_proj.numel()
        BLOCK = 131072                     # 128‑K elements per program (tunable)
        grid = (triton.cdiv(total_el, BLOCK),)

        _fuse_mul_kernel[grid](
            a_ptr=tensor_proj.data_ptr(),
            b_ptr=tensor_gate.data_ptr(),
            m_ptr=mask_b.data_ptr(),
            out_ptr=out.data_ptr(),
            N_ELEMENTS=total_el,
            BLOCK_SIZE=BLOCK,
        )
        return out

    left  = _fuse(left_proj,  left_gate)   # [B,N,N,H]
    right = _fuse(right_proj, right_gate)  # [B,N,N,H]

    # ------------------------------------------------------------------ #
    # Core contraction: batched matmul over the “k” dimension.
    # We flatten (B, H) → M and use torch.bmm which is backed by cuBLAS.
    # ------------------------------------------------------------------ #
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # right needs to be transposed on the k‑axis (second dim)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_t    = right_flat.transpose(1, 2)

    out_flat = torch.bmm(left_flat, right_t)          # (B*H, N, N)

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # Final LayerNorm over the hidden dimension
    # ------------------------------------------------------------------ #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    # ------------------------------------------------------------------ #
    # Out gating (sigmoid) and final linear projection to original dim
    # ------------------------------------------------------------------ #
    out_gate = torch.sigmoid(F.linear(inp_norm, weights["out_gate.weight"]))
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
