"""
TriMul “outgoing” implementation (AlphaFold‑3) with a fused
N³‑einsum performed in Triton.

The heavy part:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is rewritten as a batched matrix‑multiply per hidden dimension d:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

The kernel computes many (B·hidden_dim) independent matrix‑multiplications
using block‑wise GEMM (BLOCK_M×BLOCK_N tiles, BLOCK_K inner reduction).
All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
are kept in PyTorch – they are cheap compared to the O(N³) work.
The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimensions 128 (or any multiple thereof) on Hopper (SM‑90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   (per batch and per hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides of left  (B,N,N,H) -> (batch, i, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides of right (B,N,N,H) -> (batch, j, k, h)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    # strides of out   (B,N,N,H) -> (batch, i, j, h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid = tl.program_id(0)                     # combines batch and hidden dim
    pid_m = tl.program_id(1)                   # block row (i)
    pid_n = tl.program_id(2)                   # block col (j)

    batch_id = pid // H
    hid_id   = pid % H

    # ---- block offsets -------------------------------------------------
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # mask for out‑of‑bounds rows / cols
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer base for this (batch, hid) slice
    left_base  = left_ptr  + batch_id * stride_l_b + hid_id * stride_l_h
    right_base = right_ptr + batch_id * stride_r_b + hid_id * stride_r_h
    out_base   = out_ptr   + batch_id * stride_o_b + hid_id * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over k ---------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[i,k]
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right[j,k]   (note: we need right[j,k] not transposed because we
        # will use tl.dot which does the “k‑reduction” on the second axis)
        right_ptrs = right_base + offs_j[:, None] * stride_r_j + offs_k[None, :] * stride_r_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot: (M,K) x (N,K) -> (M,N)  (reduction over K)
        acc += tl.dot(left, right)

    # ---- store ---------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, on CUDA)
        - mask        : [B, N, N]        (float32 0/1, on CUDA)
        - weights     : dict of parameter tensors
        - config      : dict with keys:
            * dim          : input channel dimension
            * hidden_dim   : hidden projection dimension
            * nomask (bool): if True ignore the mask (mask is all‑ones)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm with supplied weight/bias
    # ------------------------------------------------------------------
    def layer_norm(x, w, b):
        return F.layer_norm(x, (x.shape[-1],), weight=w, bias=b, eps=1e-5)

    # ------------------------------------------------------------------
    # 1️⃣  Input norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (B,N,N,dim) @ (dim, H).T -> (B,N,N,H)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Apply mask (if not nomask)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Triton fused GEMM over the k‑dimension (the N³ part)
    # ------------------------------------------------------------------
    # work in FP16 inside the kernel for speed
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out_h = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides (contiguous layout)
    l_str = left_h.stride()   # (B*N*N*H, N*H, H, 1) -> (batch, i, k, h)
    r_str = right_h.stride()
    o_str = out_h.stride()

    # block sizes – tuned for Hopper SM90 (shared‑memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                              # batch * hidden dim
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_gemm_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        # left strides
        l_str[0], l_str[1], l_str[2], l_str[3],
        # right strides
        r_str[0], r_str[1], r_str[2], r_str[3],
        # out strides
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to FP32 for the remaining ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣  Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
