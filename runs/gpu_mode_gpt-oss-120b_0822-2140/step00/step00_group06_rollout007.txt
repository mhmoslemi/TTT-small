"""
TriMul “outgoing” implementation (forward only).

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiplication for every
hidden channel *d*:  out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

The kernel below performs this batched GEMM in FP16 (the
intermediate product) and writes the result in FP32.
All other operations (layer‑norms, linear projections,
gating, masking) are carried out in PyTorch – they are cheap
relative to the N³ reduction and keep the code simple.
The kernel works for any batch size B∈{1,2}, sequence length N,
and hidden dimension H (called *hidden_dim* in the config).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B.T) where A and B have shape
# (batch, M, K) and (batch, N, K) respectively.
# The kernel is launched with a 2‑D grid:
#   dim 0 : tiles over (M, N)
#   dim 1 : batch dimension (B*H)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ak,              # A: (batch, M, K) strides
    stride_bk, stride_bn,              # B: (batch, N, K) strides (note: we treat B as transposed)
    stride_cm, stride_cn,              # C: (batch, M, N) strides
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) Identify tile indices.
    # ------------------------------------------------------------------
    pid = tl.program_id(0)                     # tile index over M & N
    batch_id = tl.program_id(1)                # which (B*H) matrix we are processing

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n                  # tile row
    pid_n = pid % num_pid_n                   # tile col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds tiles
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2) Compute pointer offsets including the batch dimension.
    # ------------------------------------------------------------------
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Pointers to the start of the tile in A and B
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over K dimension in blocks.
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a block of A and B (FP16)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K), other=0.0).to(tl.float32)

        # dot‑product accumulation
        acc += tl.dot(a, b)

        # Advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # 4) Write the result back to C (FP32).
    # ------------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel for a batch of (B*H) matrices.
# ----------------------------------------------------------------------
def tri_mul_gemm_tiled(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.Tensor of shape (B, N, N, H)  (float16/float32)
    returns out  : torch.Tensor of shape (B, N, N, H) (float32)
    """
    B, N, _, H = left.shape
    device = left.device

    # ------------------------------------------------------------------
    # 1) Move channel dimension to be inside the batch so we can treat
    #    each (b, h) pair as an independent GEMM.
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # collapse batch & channel
    batch = B * H
    left = left.reshape(batch, N, N)
    right = right.reshape(batch, N, N)

    # Output buffer
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 2) Strides (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_a_batch = left.stride(0)   # batch stride = N*N
    stride_am = left.stride(1)        # stride over M (rows) = N
    stride_ak = left.stride(2)        # stride over K (cols) = 1

    # For B we need it transposed: we will read B as B_T[n, k]
    # Hence: stride_bk = stride over K (the row index after transpose) = 1
    #        stride_bn = stride over N (the column index after transpose) = N
    stride_b_batch = right.stride(0)
    stride_bk = right.stride(2)   # = 1
    stride_bn = right.stride(1)   # = N

    stride_c_batch = out.stride(0)
    stride_cm = out.stride(1)      # rows
    stride_cn = out.stride(2)      # cols

    # ------------------------------------------------------------------
    # 3) Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = triton.cdiv(N, BLOCK_M)
    num_n = triton.cdiv(N, BLOCK_N)
    grid = (num_m * num_n, batch)   # 2‑D grid

    tri_mul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4) Reshape back to original layout (B, N, N, H)
    # ------------------------------------------------------------------
    out = out.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Main entry point required by the challenge.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: (input, mask, weights, config)
            input  – Tensor [B, N, N, dim] (float32)
            mask   – Tensor [B, N, N]   (bool/float) – may be omitted (nomask=True)
            weights – dict with the exact keys used in the reference module
            config  – dict containing "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-5
    # Compute mean / var manually to avoid torch.nn.LayerNorm overhead
    x_flat = x.reshape(-1, dim)
    mean = x_flat.mean(-1, keepdim=True)
    var = x_flat.var(-1, unbiased=False, keepdim=True)
    x_norm = (x_flat - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    x = x_norm.reshape_as(x)                   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # Using torch.nn.functional.linear: y = x @ W.T
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask_ = torch.ones_like(mask, dtype=left.dtype) if mask is not None else 1.0
    else:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core tensor contraction – Triton kernel
    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy compute, keep final result fp32.
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    out = tri_mul_gemm_tiled(left, right)        # (B,N,N,hidden_dim) fp32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (hidden_dim) + gating
    # ------------------------------------------------------------------
    # LayerNorm over hidden_dim
    out_flat = out.reshape(-1, hidden_dim)
    mean_o = out_flat.mean(-1, keepdim=True)
    var_o = out_flat.var(-1, unbiased=False, keepdim=True)
    out_norm = (out_flat - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out_norm.reshape_as(out)

    out = out * out_gate  # gating after norm

    # ------------------------------------------------------------------
    # 7) Final linear back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out.to(torch.float32)
