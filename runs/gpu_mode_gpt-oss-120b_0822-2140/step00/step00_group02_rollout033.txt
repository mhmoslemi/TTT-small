"""
TriMul “outgoing” implementation.
The heavy N³ contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed with a custom Triton kernel that behaves like a batched
matrix‑multiply (left @ rightᵀ) for every hidden channel d.
All cheap operations (LayerNorm, Linear, gating, final projection) are
handled in PyTorch, while the core contraction is done in Triton.
The kernel works on a flattened batch×hidden dimension (P = B·HD) and
expects contiguous tensors of shape [P, N, N] (float32).  The final
output is returned as a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel : batched mat‑mul (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,           # const float*   [P, M, K]
    right_ptr,          # const float*   [P, N, K] (we read transposed -> K,N)
    out_ptr,            # float*         [P, M, N]

    M, N, K,            # matrix dimensions
    stride_left_batch,  # stride between different P
    stride_left_m,      # stride for rows (i)
    stride_left_k,      # stride for cols (k)

    stride_right_batch, # stride between different P
    stride_right_n,     # stride for cols (j)   (original j dimension)
    stride_right_k,     # stride for rows (k)   (original k dimension)

    stride_out_batch,   # stride between different P
    stride_out_m,       # stride for rows (i)
    stride_out_n,       # stride for cols (j)

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Compute 3‑D block identifiers   (batch, block_i, block_j)
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_i = block_idx // num_n          # row tile
    block_j = block_idx % num_n           # col tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch
    left_batch_ptr  = left_ptr  + batch * stride_left_batch
    right_batch_ptr = right_ptr + batch * stride_right_batch
    out_batch_ptr   = out_ptr   + batch * stride_out_batch

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # left:   shape (BLOCK_M, BLOCK_K)   (i,k)
        a_ptr = left_batch_ptr + \
                offs_m[:, None] * stride_left_m + \
                cur_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right: we need rightᵀ -> rows are k, cols are j
        # stored layout: (P, N, K)   -> stride_right_n (j), stride_right_k (k)
        b_ptr = right_batch_ptr + \
                offs_n[None, :] * stride_right_n + \
                cur_k[:, None] * stride_right_k
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix‑multiply tile
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_batch_ptr + \
               offs_m[:, None] * stride_out_m + \
               offs_n[None, :] * stride_out_n
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [P, N, N]  (contiguous, float32)
    returns    : [P, N, N]  (float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    P, N, _ = left.shape
    K = N

    out = torch.empty_like(left)

    # Tile sizes – empirically good for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_blocks = P * grid_m * grid_n

    trimul_matmul_kernel[ (total_blocks,) ](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        M=N, N=N, K=K,
        stride_left_batch=left.stride(0),
        stride_left_m=left.stride(1),
        stride_left_k=left.stride(2),
        stride_right_batch=right.stride(0),
        stride_right_n=right.stride(1),
        stride_right_k=right.stride(2),
        stride_out_batch=out.stride(0),
        stride_out_m=out.stride(1),
        stride_out_n=out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul.
    Args:
        data: (input, mask, weights, config)
            input   : [B, N, N, dim]       float32/float16 on CUDA
            mask    : [B, N, N]            (0/1)  on CUDA
            weights : dict of all linear / ln parameters (torch tensors)
            config  : dict with keys "dim", "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = hidden_dim in paper

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    ln_weight = weights["norm.weight"]
    ln_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=ln_weight,
                     bias=ln_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]     # (hidden_dim, dim)

    left  = F.linear(x, left_proj_w)               # [B,N,N,HD]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,HD]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction – Triton batched matmul
    #    left  : (B,N,N,HD) -> (B,HD,N,N) -> (P,N,N)
    #    right : same, but we need rightᵀ inside the kernel
    # ------------------------------------------------------------------
    B = input_tensor.shape[0]
    N = input_tensor.shape[1]

    # bring hidden dim to the second axis, then flatten batch & hidden
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,HD,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    P = B * hidden_dim
    left_flat  = left_perm.view(P, N, N)
    right_flat = right_perm.view(P, N, N)

    # Triton kernel (float32 accumulation)
    out_flat = trimul_matmul(left_flat, right_flat)      # [P,N,N]

    # reshape back to [B,N,N,HD]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑norm, gating and final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate                     # element‑wise gating

    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim)
    out = F.linear(out, to_out_w)            # [B,N,N,dim]

    return out
```
