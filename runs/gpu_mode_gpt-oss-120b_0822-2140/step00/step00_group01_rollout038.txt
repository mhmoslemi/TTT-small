"""
TriMul “outgoing” kernel (AlphaFold‑3 style) – Triton‑accelerated contraction.

The forward pass consists of:
1. Layer‑norm over the last channel (dim).
2. Linear projections (left/right) and three sigmoid‑gated “out” linear layers.
3. A dense pairwise contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply (B·D independent GEMMs of shape N×N).
4. Layer‑norm over the hidden dimension, gating and final linear projection.

Only the N³ contraction (step 3) is written in Triton; everything else is
performed with regular PyTorch (high‑performance fused kernels already
provided by PyTorch).  Mixed‑precision is used for the GEMM (fp16
accumulated in fp32) and the final result is cast back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) where A, B ∈ ℝ^{batch×M×K},
# B is supplied already transposed (i.e. shape batch×N×K).
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # matrix sizes
    M, N, K,
    # strides (batch, row, col) for each tensor
    stride_aa, stride_am, stride_ak,          # A: batch‑row‑col
    stride_bb, stride_bk, stride_bn,          # Bᵀ: batch‑col‑row (original Bᵀ layout)
    stride_cc, stride_cm, stride_cn,          # C: batch‑row‑col
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_base = a_ptr + batch_id * stride_aa
    b_base = b_ptr + batch_id * stride_bb
    c_base = c_ptr + batch_id * stride_cc

    # ------------------------------------------------------------------
    # Accumulator (fp32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # offsets inside the K‑tile
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # mask for out‑of‑range in K
        k_mask = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am
                         + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & k_mask[None, :], other=0.0)
        a = a.to(tl.float32)          # fp16 → fp32

        # Load Bᵀ tile: shape (BLOCK_N, BLOCK_K) because B is already transposed
        b_ptrs = b_base + (offs_n[:, None] * stride_bn
                         + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=(offs_n[:, None] < N) & k_mask[None, :], other=0.0)
        b = b.to(tl.float32)

        # Dot‑product accumulation
        acc += tl.dot(a, b)

        # advance pointers (handled implicitly by the offset arithmetic)

    # ------------------------------------------------------------------
    # Write back result (cast back to fp16 – later we will cast to fp32 in PyTorch)
    # ------------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm
                     + offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Host side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float)
        - weight_dict  : dict of torch.Tensor parameters
        - config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------- 1. LayerNorm over `dim` -----------------------
    # Using functional layer_norm to avoid creating a nn.Module.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------- 2. Linear projections ------------------------
    # No bias, weight shape: (hidden, dim)
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------- 3. Apply mask (if any) -----------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------- 4. Gating ------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------- 5. Batched contraction (Trimesh) -------------
    # Convert to fp16 for the GEMM.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape  # H == hidden
    batch_dim = B * H

    # reshape to (batch_dim, N, N)
    A = left_fp16.reshape(batch_dim, N, N)
    B_t = right_fp16.reshape(batch_dim, N, N).transpose(-2, -1)   # already transposed

    # allocate output tensor
    C = torch.empty_like(A, dtype=torch.float16)   # (batch_dim, N, N)

    # strides (in elements) for each tensor
    stride_aa, stride_am, stride_ak = A.stride()      # (batch, row, col)
    stride_bb, stride_bk, stride_bn = B_t.stride()
    stride_cc, stride_cm, stride_cn = C.stride()

    # Triton block sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        A, B_t, C,
        N, N, N,
        stride_aa, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cc, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # enough warps for fp16
    )

    # reshape back to (B, N, N, hidden)
    out = C.reshape(B, N, N, H).to(torch.float32)   # back to fp32 for the rest

    # ------------------- 6. Post‑norm, gating, final linear ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out.to(torch.float32)
