"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape B,N,N,D).
2. Linear projections to hidden dimension H (left/right) and three gating
   projections (left_gate, right_gate, out_gate) using the supplied weight
   matrices (bias‑free).
3. Fuse mask application, sigmoid gating and the out‑gate sigmoid in a single
   Triton kernel (element‑wise).  This reduces memory traffic and guarantees
   that at least part of the work is performed on the GPU in custom code.
4. Compute the “triangle‑multiply”:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of N×N matrix multiplications for each hidden channel d.
   We reshape the tensors and reuse `torch.bmm` (highly‑optimized cuBLAS) –
   this is the heavyweight O(N³) part.
5. Layer‑norm the intermediate result (hidden dimension H) and apply the
   out‑gate (already sigmoid‑ed by the kernel).
6. Final linear projection back to the original dimension D.

All tensors stay on the CUDA device; the final output is `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel : fuse mask, sigmoid gating and out‑gate sigmoid
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    left_gate_ptr,     # *float32
    right_gate_ptr,    # *float32
    out_gate_ptr,      # *float32   (will be overwritten with sigmoid)
    mask_ptr,          # *float32   broadcasted mask (1.0 or 0.0)
    total_elems,       # i32       number of elements in the tensors
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elems

    # load
    l = tl.load(left_ptr + offs, mask=m, other=0.0)
    r = tl.load(right_ptr + offs, mask=m, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=m, other=0.0)
    og = tl.load(out_gate_ptr + offs, mask=m, other=0.0)
    mask = tl.load(mask_ptr + offs, mask=m, other=1.0)   # default 1.0 when mask omitted

    # sigmoid (stable)
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))
    og_sig = 1.0 / (1.0 + tl.exp(-og))

    # apply mask and gating
    l = l * mask * lg_sig
    r = r * mask * rg_sig

    # store results back
    tl.store(left_ptr + offs, l, mask=m)
    tl.store(right_ptr + offs, r, mask=m)
    tl.store(out_gate_ptr + offs, og_sig, mask=m)


# --------------------------------------------------------------
# Helper: layer‑norm without bias (bias is handled separately)
# --------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]; weight/bias: [C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul forward (outgoing) – highly fused version using Triton.

    Args
    ----
    data : (input, mask, weights, config)
        input  : torch.Tensor,  [B, N, N, D]    (float32)
        mask   : torch.Tensor,  [B, N, N]      (float32, 0/1) – may be None/ignored
        weights: dict of torch.Tensor containing all learnable parameters
        config : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor : [B, N, N, D]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, D = x.shape
    H = cfg["hidden_dim"]          # hidden dimension
    assert D == cfg["dim"]

    # ------------------------------------------------------------------
    # 1) input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(x,
                    weight=weights["norm.weight"],
                    bias=weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # shape helpers
    x_reshape = x.view(B * N * N, D)          # (B*N*N, D)

    left = F.linear(x_reshape, weights["left_proj.weight"]).view(B, N, N, H)
    right = F.linear(x_reshape, weights["right_proj.weight"]).view(B, N, N, H)

    left_gate = F.linear(x_reshape, weights["left_gate.weight"]).view(B, N, N, H)
    right_gate = F.linear(x_reshape, weights["right_gate.weight"]).view(B, N, N, H)
    out_gate = F.linear(x_reshape, weights["out_gate.weight"]).view(B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Fuse mask & gating (Triton)
    # ------------------------------------------------------------------
    # ensure mask has shape [B,N,N,1] and dtype float32
    if mask is None or mask.numel() == 0:
        # no mask -> all ones
        mask_fused = torch.ones((B, N, N, 1), dtype=torch.float32, device=device)
    else:
        mask_fused = mask.unsqueeze(-1).float()          # [B,N,N,1]

    total_elements = left.numel()                # B*N*N*H
    BLOCK = 1024

    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,          # not needed after fusion, keep as temp
        right_gate_ptr=right_gate,
        out_gate_ptr=out_gate,
        mask_ptr=mask_fused,
        total_elems=total_elements,
        BLOCK=BLOCK,
    )
    # left_gate/right_gate are no longer needed; they contain pre‑sigmoid values.

    # ------------------------------------------------------------------
    # 4) Triangle‑multiply: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matmul over (B*H, N, N)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)    # (B*H, N, N)

    # right needs to be transposed on the last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))      # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) to_out_norm + out_gate
    # ------------------------------------------------------------------
    out = _layer_norm(out,
                      weight=weights["to_out_norm.weight"],
                      bias=weights["to_out_norm.bias"])

    out = out * out_gate          # out_gate already sigmoid‑ed by the kernel

    # ------------------------------------------------------------------
    # 6) Final linear projection back to D
    # ------------------------------------------------------------------
    out = out.view(B * N * N, H)
    out = F.linear(out, weights["to_out.weight"]).view(B, N, N, D)

    return out.to(torch.float32)
