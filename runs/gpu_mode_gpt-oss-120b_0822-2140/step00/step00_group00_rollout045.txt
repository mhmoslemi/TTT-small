"""
TriMul (outgoing) – fast Triton implementation of the N³ pairwise interaction.

The heavy work is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is exactly a batched matrix‑multiply per hidden channel:
    (left[...,d]) @ (right[...,d].T)

All LayerNorms, linear projections and gates stay in PyTorch (FP32).
The core N³ reduction is performed with a fused Triton GEMM that works on
FP16 for the two inputs and accumulates in FP32, then writes FP16 back.
The final result is cast back to FP32 and passed through the remaining
linear+norm layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: C = A @ B   where A,B are (batch, M, K) and (batch, K, N)
# ------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (int32)
    stride_az, stride_ak, stride_am,   # strides for A (batch, K, M)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM (A @ B) for the TriMul core."""
    pid_z = tl.program_id(2)               # batch index (B*H)
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col

    # --- compute block offsets ---
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch_ptr = a_ptr + pid_z * stride_az
    b_batch_ptr = b_ptr + pid_z * stride_bz
    c_batch_ptr = c_ptr + pid_z * stride_cz

    # load A tiles (M x K) and B tiles (K x N)
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back in fp16 (the caller will cast to fp32 later)
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ------------------------------------------------------------------
# Host entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: Tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim]  (float32, on CUDA)
            mask   – torch.Tensor [B, N, N]      (float32/bool, on CUDA)
            weights – dict of leaf tensors (all float32, on CUDA)
            config – dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply optional mask (mask is broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction using Triton (batched GEMM per hidden channel)
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) and transpose right for GEMM
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)      # (batch, M, K)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)    # (batch, K, N)

    # cast to fp16 for the kernel
    left_fp16 = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)   # same shape (B*H, N, N)

    # launch parameters (tuned for Hopper, but reasonable defaults)
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    trimul_gemm_kernel[
        grid
    ](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        N,
        N,
        left_fp16.stride(0),    # stride_az (batch stride)
        left_fp16.stride(2),    # stride_ak (K stride)
        left_fp16.stride(1),    # stride_am (M stride)
        right_fp16.stride(0),   # stride_bz
        right_fp16.stride(1),   # stride_bk (K stride)
        right_fp16.stride(2),   # stride_bn (N stride)
        out_fp16.stride(0),     # stride_cz
        out_fp16.stride(1),     # stride_cm (M stride)
        out_fp16.stride(2),     # stride_cn (N stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden_dim]
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # gate after norm
    out = F.linear(out, weights["to_out.weight"])  # final projection back to dim

    return out
