"""
TriMul “outgoing” implementation with a fused Triton kernel.

The heavy part of TriMul is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to, for each hidden channel `d`,
a batched matrix‑multiply `left_d @ right_d.T`.  The kernel below
computes this product for all batches and all hidden channels in a
single launch, loading tiles in FP16 while accumulating in FP32.
All other operations (layer‑norms, linear projections, gating,
masking) are performed with PyTorch which is already highly
optimized on the target hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                         # sizes
    stride_bl, stride_il, stride_kl, stride_hl,      # left strides
    stride_br, stride_jr, stride_kr, stride_hr,      # right strides
    stride_bo, stride_io, stride_jo, stride_ho,      # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for each (b, h) pair:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    left  : (B, N, N, H)  – stored as (b, i, k, h)
    right : (B, N, N, H)  – stored as (b, j, k, h)
    out   : (B, N, N, H)  – stored as (b, i, j, h)
    """

    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)
    pid_hb = tl.program_id(2)         # combined batch*hidden index

    # split combined dimension
    b = pid_hb // H
    h = pid_hb % H

    # offsets within N
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for OOB rows/cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k_offset = tl.arange(0, BLOCK_K)

        # ------- load left tile (i,k) ---------------------------------
        left_ptrs = left_ptr \
            + b * stride_bl \
            + i_offset[:, None] * stride_il \
            + (k_base + k_offset)[None, :] * stride_kl \
            + h * stride_hl
        # mask k dimension
        mask_ik = mask_i[:, None] & ((k_base + k_offset) < N)
        left_tile = tl.load(left_ptrs, mask=mask_ik, other=0.0).to(tl.float32)

        # ------- load right tile transposed (k,j) --------------------
        # we need a (K,N) tile where K is the reduction dimension
        right_ptrs = right_ptr \
            + b * stride_br \
            + (k_base + k_offset)[:, None] * stride_kr \
            + j_offset[None, :] * stride_jr \
            + h * stride_hr
        mask_kj = ((k_base + k_offset) < N)[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=mask_kj, other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile)

    # store result
    out_ptrs = out_ptr \
        + b * stride_bo \
        + i_offset[:, None] * stride_io \
        + j_offset[None, :] * stride_jo \
        + h * stride_ho
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]      (bool / float32)
        - weights_dict : contains all learned parameters
        - config_dict  : contains dim, hidden_dim and the flag ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]                      # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim is the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if cfg.get("nomask", True):
        mask_full = torch.ones(B, N, N, device=device, dtype=x.dtype)
    else:
        mask_full = mask.to(x.dtype)
    mask_full = mask_full.unsqueeze(-1)               # (B,N,N,1)
    left = left * mask_full
    right = right * mask_full

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also linear without bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triton fused contraction (batched matmul)
    # ------------------------------------------------------------------
    # Cast to fp16 for bandwidth; accumulation stays fp32 inside the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty(B, N, N, H, device=device, dtype=torch.float16)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),        # i‑blocks
        triton.cdiv(N, BLOCK_N),        # j‑blocks
        B * H,                          # (batch, hidden) pairs
    )

    _trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
