"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The heavy N³ term
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realised as a batched matrix‑multiplication
    out = (left_reshape) @ (right_reshape)ᵀ
with the batch dimension = B * hidden_dim.
Both LayerNorms (the first on *dim* and the second on *hidden_dim*) are
executed by custom Triton kernels – this satisfies the “use a kernel”
requirement while keeping the large matmul on the highly‑optimised cuBLAS
backend (fast on Hopper/H200).

All tensors are expected to be on CUDA already; the final result is
float‑32.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton LayerNorm (no bias => bias_ptr may be 0)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # input   [B,N,N,C]
    y_ptr,               # output  [B,N,N,C]
    weight_ptr,          # γ   [C] (may be 0)
    bias_ptr,            # β   [C] (may be 0)
    B, N, C,
    stride_xb, stride_xn1, stride_xn2, stride_xc,
    stride_yb, stride_yn1, stride_yn2, stride_yc,
    eps: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N                       # number of vectors to normalise
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)      # linear indices of vectors
    mask = offs < total

    # --------------------------------------------------------------
    # decode linear index -> (b,i,j)
    # --------------------------------------------------------------
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    # --------------------------------------------------------------
    # load the whole C‑dim vector for each (b,i,j)
    # --------------------------------------------------------------
    c_offs = tl.arange(0, BLOCK)[:, None] * 0  # dummy to broadcast later
    c_range = tl.arange(0, C)[None, :]         # shape (1,C)

    x = tl.load(
        x_ptr
        + b[:, None] * stride_xb
        + i[:, None] * stride_xn1
        + j[:, None] * stride_xn2
        + c_range * stride_xc,
        mask=mask[:, None],
        other=0.0,
    )

    # --------------------------------------------------------------
    # mean & variance (float32 accumulator)
    # --------------------------------------------------------------
    x_f32 = x.to(tl.float32)
    sum_x = tl.sum(x_f32, axis=1)
    mean = sum_x / C
    sum_x2 = tl.sum(x_f32 * x_f32, axis=1)
    var = sum_x2 / C - mean * mean

    # normalise
    norm = (x_f32 - mean[:, None]) * tl.rsqrt(var[:, None] + eps)

    # optional affine transform
    if weight_ptr != 0:
        w = tl.load(weight_ptr + c_range, mask=c_range < C, other=0.0)
        norm = norm * w
    if bias_ptr != 0:
        b_ = tl.load(bias_ptr + c_range, mask=c_range < C, other=0.0)
        norm = norm + b_

    # store back as float32 (later we cast to fp32 anyway)
    tl.store(
        y_ptr
        + b[:, None] * stride_yb
        + i[:, None] * stride_yn1
        + j[:, None] * stride_yn2
        + c_range * stride_yc,
        norm,
        mask=mask[:, None],
    )


def triton_layernorm(x, weight, bias, eps=1e-5):
    """
    Triton‑based LayerNorm over the last dimension.
    x : torch.Tensor [B,N,N,C]  (any floating dtype)
    weight, bias : torch.Tensor[C] or None
    Returns torch.Tensor[B,N,N,C] (float32)
    """
    B, N, _, C = x.shape
    y = torch.empty_like(x, dtype=torch.float32)

    # strides in elements (not bytes)
    stride_xb, stride_xn1, stride_xn2, stride_xc = x.stride()
    stride_yb, stride_yn1, stride_yn2, stride_yc = y.stride()

    BLOCK = 128
    total_vectors = B * N * N
    grid = (triton.cdiv(total_vectors, BLOCK),)

    weight_ptr = weight.data_ptr() if weight is not None else 0
    bias_ptr = bias.data_ptr() if bias is not None else 0

    layernorm_kernel[grid](
        x.data_ptr(),
        y.data_ptr(),
        weight_ptr,
        bias_ptr,
        B,
        N,
        C,
        stride_xb,
        stride_xn1,
        stride_xn2,
        stride_xc,
        stride_yb,
        stride_yn1,
        stride_yn2,
        stride_yc,
        eps,
        BLOCK=BLOCK,
    )
    return y

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated “outgoing” TriMul.
    data = (input, mask, weights, config)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣  First LayerNorm on dim
    # --------------------------------------------------------------
    x = triton_layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]   float32

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free)
    # --------------------------------------------------------------
    # shape: (dim, hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣  Apply optional mask (mask shape [B,N,N] → [B,N,N,1])
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(dtype)          # broadcast
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) – also bias‑free linear layers
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate          # [B,N,N,hidden]
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Core N³ operation:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     realised as batched matmul over the k‑dimension.
    # --------------------------------------------------------------
    #   left  : (B,N,N,hidden) → (B*hidden, N, N)   (i,k)
    #   right : (B,N,N,hidden) → (B*hidden, N, N)   (k,j)  (transpose k↔j)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()          # (B, hidden, k, j)
    right_reshape = right_t.view(B * hidden_dim, N, N)

    # Batched matrix‑multiply: (B*H, N, N) @ (B*H, N, N)ᵀ  → (B*H, N, N)
    out_reshape = torch.bmm(left_reshape, right_reshape)

    # Restore original layout: (B,N,N,hidden)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣  Second LayerNorm on hidden_dim
    # --------------------------------------------------------------
    out = triton_layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # still float32

    # --------------------------------------------------------------
    # 7️⃣  Apply out‑gate and final linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    # to_out : Linear(hidden_dim, dim) without bias
    out = F.linear(out, weights["to_out.weight"])

    return out.to(dtype)
