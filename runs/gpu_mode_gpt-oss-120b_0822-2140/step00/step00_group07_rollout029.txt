"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the ternary multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be written as a batch of matrix multiplications
    out_{b,d} = left_{b,d} @ right_{b,d}.T
where left_{b,d}, right_{b,d} ∈ ℝ^{N×N}.
We fuse this GEMM with a custom Triton kernel (fp16 accumulation) and
keep the surrounding linear / LayerNorm logic in PyTorch.
The final tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  C = A @ B.T   where A,B shape = (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,              # pointers
    N,                                # sequence length
    stride_ah, stride_am, stride_ak,  # A strides: (batch*hid, i, k)
    stride_bh, stride_bm, stride_bk,  # B strides (original layout: (batch*hid, j, k))
    stride_ch, stride_cm, stride_cn,  # C strides (same as A)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ B.T   (A,B : [B*H, N, N])"""
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_h = tl.program_id(2)          # batch*hidden index

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the block
    a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # B is accessed as B.T : (k, j)
    b_ptrs = b_ptr + pid_h * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bm

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & mask_n[None, :], other=0.0)

        # a : (BM, BK)   b : (BK, BN)  ->  (BM, BN)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right:  [B, N, N, H]   (H = hidden_dim)
    returns out: same shape, computed as Σ_k left[...,i,k,:] * right[...,j,k,:]
    """
    B, N, _, H = left.shape
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) and make contiguous for simple strides
    # --------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 for the kernel (accumulated in fp32)
    left_t = left_t.to(torch.float16)
    right_t = right_t.to(torch.float16)

    # Output buffer
    out_t = torch.empty_like(left_t, dtype=torch.float16, device=left.device)

    # Strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = left_t.stride()
    stride_bh, stride_bm, stride_bk = right_t.stride()
    stride_ch, stride_cm, stride_cn = out_t.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * H,                        # batch*hidden
    )
    _tri_mul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bm,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Convert back to (B, N, N, H) and to float32
    out = out_t.permute(0, 2, 3, 1).contiguous().to(torch.float32)
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – Tensor[B, N, N, dim]   (float32)
            mask   – Tensor[B, N, N]        (bool/float)
            weights – dict with the nine weight tensors used by the module
            config – dict with keys:
                     "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor[B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask_unsq = mask.unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gating scalars (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core ternary multiplication (fused GEMM via Triton)
    # ------------------------------------------------------------------
    out = _tri_mul_gemm(left, right)          # shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply out‑gate and final linear projection back to dim
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
