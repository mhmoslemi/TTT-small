    B,N,N,H = left.shape
    G = B * H
    left_ = left.permute(0,3,1,2).contiguous().view(G, N, N)

                                 BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

"""
TriMul “outgoing” implementation with a Triton kernel for the cubic einsum.

The heavy part of the forward pass is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a per‑channel matrix multiplication:
    (L_d)_{i,k}  @  (R_d)_{j,k}ᵀ  →  (out_d)_{i,j}

We flatten the batch and hidden dimension into a single “group” dimension
and launch a Triton GEMM kernel that computes all groups in parallel.
All other operations (LayerNorm, linear projections, gates, final projection)
are performed with ordinary PyTorch (they are cheap compared to the N³
multiplication).

The kernel works in float32, stores float32, and the final output is
float32 as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for groups = batch * hidden_dim
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr,          # left  [G, M, K]
    B_ptr,          # rightᵀ [G, K, N]
    C_ptr,          # out   [G, M, N]
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_g = tl.program_id(2)   # group (batch*hidden)

    # ------------------- offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to the start of the current block
    a_ptrs = A_ptr + pid_g * stride_a_batch \
                     + offs_m[:, None] * stride_am \
                     + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_g * stride_b_batch \
                     + offs_k[:, None] * stride_bk \
                     + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- main loop over K -------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)
        acc += tl.dot(a, b)               # (M,K)·(K,N) → (M,N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------- write back -------------------
    c_ptrs = C_ptr + pid_g * stride_c_batch \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=(mask_m[:, None] & mask_n[None, :]))

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _batched_gemm(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    left : [G, M, K]   (contiguous)
    right_t : [G, K, N] (contiguous, already transposed)
    returns C : [G, M, N] (float32)
    """
    G, M, K = left.shape
    _, Kb, N = right_t.shape
    assert K == Kb

    # output buffer
    out = torch.empty((G, M, N), dtype=left.dtype, device=left.device)

    # strides (in elements, not bytes)
    stride_a_batch, stride_am, stride_ak = left.stride()
    stride_b_batch, stride_bk, stride_bn = right_t.stride()
    stride_c_batch, stride_cm, stride_cn = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        G,
    )
    _tri_mul_gemm_kernel[grid](
        left,
        right_t,
        out,
        M, N, K,
        stride_am, stride_ak, stride_a_batch,
        stride_bk, stride_bn, stride_b_batch,
        stride_cm, stride_cn, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel for the
    cubic multiplication.

    Args:
        data: (input, mask, weights, config)
            input   – Tensor[bs, N, N, dim] (float32/float16)
            mask    – Tensor[bs, N, N] (bool/float)
            weights – dict containing all linear / LN parameters
            config  – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = inp.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)               # [bs, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating streams
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton‑accelerated cubic multiplication
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to group dimension G = bs * hidden_dim
    left_g = left.permute(0, 3, 1, 2).contiguous()   # [bs, H, N, N]
    right_g = right.permute(0, 3, 1, 2).contiguous()
    G = bs * hidden_dim
    left_g = left_g.view(G, N, N)                    # [G, M, K]
    right_g = right_g.view(G, N, N)                  # [G, J, K]
    right_g_t = right_g.transpose(1, 2).contiguous()  # [G, K, N]

    # GEMM via Triton
    out_g = _batched_gemm(left_g, right_g_t)        # [G, N, N]

    # reshape back to [bs, N, N, hidden_dim]
    out = out_g.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final normalization, gating and linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [bs, N, N, dim]

    # ensure float32 output
    return out.to(torch.float32)
