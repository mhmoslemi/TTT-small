"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
with left/right already masked and gated.
We fuse this reduction in a Triton kernel that tiles the
i‑ and j‑ dimensions (BLOCK_I × BLOCK_J) and processes the k‑dimension
in blocks (BLOCK_K).  Hidden_dim (hidden_dim) is fixed to 128 (the
only value used in the tests), so we keep the whole channel vector
in registers.

All other operations (layer‑norms, linear projections, gates and the
final linear+norm) are performed with PyTorch which is already
highly optimized on the GPU.  The kernel returns a FP32 tensor of
shape [B, N, N, hidden_dim]; the rest of the pipeline finishes in
float32 as required.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core N×N×hidden_dim reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,           # *const float*   [B, N, N, H]
    right_ptr,          # *const float*   [B, N, N, H]
    out_ptr,            # *float*         [B, N, N, H]
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,          # = hidden_dim (128)
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Compute 3‑D grid index (batch, i‑tile, j‑tile)
    # --------------------------------------------------------------
    grid_i = tl.cdiv(N, BLOCK_I)
    grid_j = tl.cdiv(N, BLOCK_J)

    b = pid // (grid_i * grid_j)
    rem = pid % (grid_i * grid_j)
    i0 = (rem // grid_j) * BLOCK_I
    j0 = (rem % grid_j) * BLOCK_J

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator for the tile (I,J,D)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k in blocks
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left[b, i, k, d] and right[b, j, k, d]
        left_base = left_ptr + \
            b * stride_left_batch + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_d[None, None, :] * stride_left_d

        right_base = right_ptr + \
            b * stride_right_batch + \
            offs_j[:, None, None] * stride_right_j + \
            offs_k[None, :, None] * stride_right_k + \
            offs_d[None, None, :] * stride_right_d

        # Load tiles (I,K,D) and (J,K,D)
        left_tile = tl.load(
            left_base,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )
        right_tile = tl.load(
            right_base,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # For each k in the block we need the outer product over i and j:
        #    acc[i,j,:] += left[i,k,:] * right[j,k,:]
        # ------------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            k_valid = mask_k[kk]
            # left[i,kk,:]  -> (I,D)
            # right[j,kk,:] -> (J,D)
            l = left_tile[:, kk, :]          # (I,D)
            r = right_tile[:, kk, :]         # (J,D)
            prod = l[:, None, :] * r[None, :, :]   # (I,J,D)
            acc += tl.where(k_valid, prod, 0.0)

    # --------------------------------------------------------------
    # Write the accumulated tile back to global memory
    # --------------------------------------------------------------
    out_base = out_ptr + \
        b * stride_out_batch + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_d[None, None, :] * stride_out_d

    tl.store(
        out_base,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – Tensor[B, N, N, dim]   (float32, on CUDA)
            mask   – Tensor[B, N, N]        (float32 or bool) or None
            weights – dict of parameter tensors
            config  – dict with keys:
                      dim, hidden_dim, nomask (bool)

    Returns:
        Tensor[B, N, N, dim]  (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on dim
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core reduction via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides
    stride_left = left.stride()          # (N*N*H, N*H, H, 1)
    stride_right = right.stride()
    stride_out = out.stride()

    # launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = hidden_dim   # hidden_dim is 128 in all tests

    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid = (B * grid_i * grid_j,)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_batch=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_d=stride_left[3],
        stride_right_batch=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_d=stride_right[3],
        stride_out_batch=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_d=stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6. to_out_norm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate   # broadcast (B,N,N,hidden_dim)

    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    return out
