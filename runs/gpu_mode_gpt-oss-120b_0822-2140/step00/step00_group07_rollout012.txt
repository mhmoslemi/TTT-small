"""
TriMul “outgoing” kernel – Triton‑accelerated contraction.

The heavy part of the TriMul forward pass is the 3‑tensor einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix multiplication of
    A_{b,d} = left[b,:,:,d]          (shape N×N, index i‑k)
    B_{b,d} = right[b,:,:,d].T       (shape N×N, index k‑j)

so that   out_{b,d} = A_{b,d} @ B_{b,d} .
The kernel below computes many (B·hidden_dim) matrix multiplies in
FP16 with a classic tiling strategy (BLOCK_M × BLOCK_N × BLOCK_K).
All surrounding LayerNorm/Linear/Gate work is done in PyTorch – this
keeps the implementation short while still giving a large speed‑up
for the N³‑like contraction.

The final output is returned in FP32 with the same shape as the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A and B are 3‑D tensors:
#   A : (B*H, M, K)
#   B : (B*H, K, N)  (already transposed)
#   C : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix dimensions
    stride_ah, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_h = tl.program_id(2)          # batch * hidden dim

    # ------------------------------------------------------------------
    # Compute base pointers for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = (
        A_ptr
        + pid_h * stride_ah
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_h * stride_bh
        + offs_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )
    # accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles (mask out‑of‑bounds)
        a = tl.load(
            a_ptrs,
            mask=(offs_m < M)[:, None] & (offs_k + k * BLOCK_K < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k + k * BLOCK_K < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # (M,K)·(K,N) → (M,N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result back to C
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m < M)[:, None] & (offs_n < N)[None, :],
    )


# ----------------------------------------------------------------------
# Helper: LayerNorm (no learnable bias‑shift, handled in PyTorch)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]   (float32)
            mask_tensor  : [B, N, N]       (float32 or bool)
            weights_dict : all linear / layer‑norm weights & biases
            config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        out : torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – keep in FP32 for stability
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])   # weight: [hidden_dim, dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and gating of the projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate          # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM:
    #    left  : (B, N, N, H)  -> (B*H, N, N)   (i, k)
    #    right : (B, N, N, H)  -> (B*H, N, N)   (k, j) by transposing last two axes
    # ------------------------------------------------------------------
    # Move hidden dim to axis 1 for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where inner dim is k

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    # Flatten batch*hidden into one dimension
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # Cast to FP16 for the kernel (still keep a FP32 copy for later)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Allocate output buffer in FP16
    out_fp16 = torch.empty_like(left_fp16, device=device)

    # ------------------------------------------------------------------
    # 6) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_h,                   # batch*hidden
    )

    # Strides for the flattened tensors
    stride_ah = left_fp16.stride(0)   # batch * N * N ?
    stride_am = left_fp16.stride(1)   # row stride = N
    stride_ak = left_fp16.stride(2)   # col stride = 1

    stride_bh = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)  # row stride (k dimension) = N
    stride_bn = right_fp16.stride(2)  # col stride = 1

    stride_ch = out_fp16.stride(0)
    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,               # M
        N,               # N
        N,               # K
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Reshape the GEMM result back to [B, N, N, hidden_dim] (FP32)
    # ------------------------------------------------------------------
    out_hidden = out_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 8) Final LayerNorm, gating, and projection back to `dim`
    # ------------------------------------------------------------------
    out_ln = _layer_norm(
        out_hidden,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )  # [B,N,N,hidden_dim]

    out_gated = out_ln * out_gate  # broadcast over hidden_dim

    # Linear back to original dim (no bias)
    out = F.linear(out_gated, weights["to_out.weight"])

    return out  # float32, shape [B, N, N, dim]
```

**Explanation of key choices**

* **Tiling parameters** – `BLOCK_M/N = 128` matches the Hopper SM size and gives good occupancy for N up to 1024. `BLOCK_K = 32` balances shared‑memory usage and the number of loop iterations.
* **Data layout** – We permute the hidden dimension in front, flatten `B·H` and treat each slice as an independent matrix‑multiply. This avoids any explicit synchronization across the hidden dimension.
* **Precision** – The heavy GEMM runs in FP16 for speed; the surrounding LayerNorm / gating / final Linear stay in FP32 to retain numerical stability. The final output is cast back to FP32 as required.
* **Mask handling** – Mask is multiplied before gating; when `nomask` is true the mask tensor will be all‑ones, so the same code works without extra conditionals.
* **No fallback** – The implementation always launches the Triton kernel; if it fails the error will surface immediately, satisfying the “no silent fallback” rule.
