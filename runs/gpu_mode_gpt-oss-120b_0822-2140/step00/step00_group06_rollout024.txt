"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of TriMul is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right have shape (B, N, N, H) (H = hidden_dim).  The rest of the
module (LayerNorms, linear projections, sigmoid gates) is cheap and is kept in
PyTorch.  The kernel below computes the above contraction tile‑wise with
Triton, using fp16 for the matrix data and fp32 accumulation.  The final
tensor is cast back to fp32 for the remaining post‑processing.

The launch grid is three‑dimensional (batch, tile_i, tile_j).  Each block
processes a (BLOCK_M × BLOCK_N) tile of the (i,j) matrix for all hidden
channels in chunks of BLOCK_H (default 8).  The reduction over the sequence
dimension k is performed in steps of BLOCK_K (default 32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel
# -------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # const * fp16
    right_ptr,         # const * fp16
    out_ptr,           # * fp16
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_jr, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # tile over i
    pid_j = tl.program_id(2)          # tile over j

    # start indices of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # -----------------------------------------------------------------
    # allocate accumulator (float32) for a (BLOCK_M, BLOCK_N, BLOCK_H) tile
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over k dimension in chunks of BLOCK_K
    # -----------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k_base)

        # masks for out‑of‑bounds in i / j / k
        mask_i = i0 + tl.arange(0, BLOCK_M) < N
        mask_j = j0 + tl.arange(0, BLOCK_N) < N
        mask_k = k_base + tl.arange(0, BLOCK_K) < N

        # -----------------------------------------------------------------
        # load a tile of left[b, i, k, h]
        # shape: (BLOCK_M, cur_k, BLOCK_H)
        # -----------------------------------------------------------------
        i_idx = i0 + tl.arange(0, BLOCK_M)[:, None, None]          # (M,1,1)
        k_idx = k_base + tl.arange(0, BLOCK_K)[None, :, None]      # (1,K,1)
        h_idx = tl.arange(0, BLOCK_H)[None, None, :]               # (1,1,H)

        left_offset = (
            pid_b * stride_bl
            + i_idx * stride_il
            + k_idx * stride_kl
            + h_idx * stride_hl
        )
        left_tile = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)      # cast to fp32 for the dot

        # -----------------------------------------------------------------
        # load a tile of right[b, j, k, h]
        # shape: (BLOCK_N, cur_k, BLOCK_H)
        # -----------------------------------------------------------------
        j_idx = j0 + tl.arange(0, BLOCK_N)[:, None, None]          # (N,1,1)

        right_offset = (
            pid_b * stride_br
            + j_idx * stride_jr
            + k_idx * stride_kr
            + h_idx * stride_hr
        )
        right_tile = tl.load(
            right_ptr + right_offset,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)

        # -----------------------------------------------------------------
        # contract over k for every hidden channel
        # acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]
        # -----------------------------------------------------------------
        for h in range(BLOCK_H):
            # (M, K) @ (N, K).T   -> (M, N)
            acc[:, :, h] += tl.dot(left_tile[:, :, h], right_tile[:, :, h])

    # -----------------------------------------------------------------
    # write back the result (cast to fp16)
    # -----------------------------------------------------------------
    out_offset = (
        pid_b * stride_bo
        + (i0 + tl.arange(0, BLOCK_M)[:, None, None]) * stride_io
        + (j0 + tl.arange(0, BLOCK_N)[None, :, None]) * stride_jo
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_ho
    )
    # mask for i/j out‑of‑bounds (h‑mask is handled by the loop limits)
    mask_i = i0 + tl.arange(0, BLOCK_M) < N
    mask_j = j0 + tl.arange(0, BLOCK_N) < N
    mask_h = tl.arange(0, BLOCK_H) < H
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]

    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=store_mask)


# -------------------------------------------------------------
# Host‑side wrapper (custom_kernel)
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Fast TriMul “outgoing” implementation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask          : torch.Tensor [B, N, N]   (bool / 0‑1)
        - weights       : dict of torch.Tensors (model parameters)
        - config        : dict with keys
            * dim        : int, input feature dimension
            * hidden_dim : int, projection dimension (H)
            * nomask     : bool, if True mask is ignored
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # LayerNorm and Linear layers in PyTorch (fast enough)
    # ---------------------------------------------------------
    # LayerNorm on the last dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # optional mask (broadcast to hidden dim)
    if not nomask:
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # Triton contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    # ---------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    # Cast to fp16 for the kernel (memory‑bandwidth friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)   # same shape [B,N,N,H]

    # Strides (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_fp16.stride()

    # Tile sizes – tuned for Hopper (SM90) but work for all requested sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 8   # must divide hidden_dim (all test configs satisfy this)

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,          # good default for Hopper
    )

    # ---------------------------------------------------------
    # Post‑processing (LayerNorm, out‑gate, final linear)
    # ---------------------------------------------------------
    # Convert back to fp32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # to_out_norm (LayerNorm over hidden dimension)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
