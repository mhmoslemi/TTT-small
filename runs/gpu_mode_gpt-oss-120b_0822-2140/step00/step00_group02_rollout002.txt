"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of H (= hidden_dim) independent matrix‑multiplications
(left[...,d] @ right[...,d].T).  The implementation below
* performs LayerNorm and the linear / gating projections in PyTorch,
* casts the projected tensors to fp16,
* launches a Triton kernel that computes the above einsum in fp32
  (accumulating in fp32 for numerical stability) for every batch &
  hidden channel, and finally
* finishes the remaining LayerNorm, gating and output projection in PyTorch.

The Triton kernel works on tiles of the (i,j) output matrix (BLOCK_M×BLOCK_N)
and iterates over the k dimension in blocks (BLOCK_K).  One program id in the
third dimension encodes a (batch, hidden) pair, so the kernel processes all
B·H independent matrix multiplications in parallel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel performing the einsum over the k dimension.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single (b, h) pair.
    """
    pid_bh = tl.program_id(2)                # combined batch * hidden
    batch_idx = pid_bh // H
    h_idx     = pid_bh %  H

    pid_m = tl.program_id(0)                 # tile row index (i)
    pid_n = tl.program_id(1)                 # tile col index (j)

    # ------------------------------------------------------------------
    # Offsets for the i, j, k axes
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers for the current tile (fixed b and h)
    # ------------------------------------------------------------------
    left_tile_ptr = (
        left_ptr
        + batch_idx * stride_l_b
        + h_idx * stride_l_h
        + offs_i[:, None] * stride_l_i            # i axis
        + offs_k[None, :] * stride_l_k            # k axis
    )
    right_tile_ptr = (
        right_ptr
        + batch_idx * stride_r_b
        + h_idx * stride_r_h
        + offs_j[:, None] * stride_r_i            # j axis (acts as row for right)
        + offs_k[None, :] * stride_r_k            # k axis
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k_blocks):
        # Load tiles (mask out‑of‑bounds elements)
        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & (offs_k[None, :] < N),
                       other=0.0).to(tl.float32)
        right = tl.load(right_tile_ptr,
                        mask=mask_j[:, None] & (offs_k[None, :] < N),
                        other=0.0).to(tl.float32)

        # right is (BLOCK_N, BLOCK_K); we need its transpose for dot
        acc += tl.dot(left, tl.trans(right))

        # Advance k pointers
        left_tile_ptr  += BLOCK_K * stride_l_k
        right_tile_ptr += BLOCK_K * stride_r_k

    # ------------------------------------------------------------------
    # Write the result tile back
    # ------------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + batch_idx * stride_o_b
        + h_idx * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_tile_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input (dim) ----------
    # Manual LayerNorm to avoid extra modules; use float32 for stability.
    x_f32 = input_tensor.to(torch.float32)
    mean = x_f32.mean(-1, keepdim=True)
    var = x_f32.var(-1, unbiased=False, keepdim=True)
    eps = 1e-5
    x_norm = (x_f32 - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ---------- Linear projections ----------
    # All linear layers are bias‑free in the reference code.
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        # mask shape: [B, N, N]; expand to last dim
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Cast to fp16 for the heavy kernel ----------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty_like(left_fp16)   # shape [B,N,N,hidden_dim]

    # ---------- Compute the einsum with Triton ----------
    # Retrieve strides (they match the contiguous layout)
    stride_l = left_fp16.stride()   # (B,N,N,H) -> (N*N*H, N*H, H, 1)
    stride_r = right_fp16.stride()
    stride_o = out_fp16.stride()

    # Tile sizes – tuned for Hopper (SM90) memory hierarchy.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim                    # one program per (batch, hidden)
    )
    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ---------- to_out LayerNorm ----------
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ---------- Apply out_gate and final linear ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
