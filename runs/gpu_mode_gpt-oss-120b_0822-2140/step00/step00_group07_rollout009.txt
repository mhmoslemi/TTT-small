    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

"""
TriMul “outgoing” kernel (AlphaFold3)

Given tensors
    left, right ∈ ℝ^{B×N×N×H}
the core operation is
    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]          (H‑wise)

We keep all pre‑ and post‑processing (LayerNorms, Linear projections,
gates, final Linear) in PyTorch and only off‑load the heavy N³ reduction
to a Triton kernel.  Each program instance computes a single (b,i,j)
output vector (size H).  The reduction over k is performed block‑wise
with a small K‑tile (default 32) to keep register pressure low while
maintaining good coalescing.  The kernel works for any batch size ≤2,
any sequence length up to 1024 and hidden dimension up to 128 (or larger
with a remainder mask).

The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    One program instance = one (b,i,j) output vector of length H.
    """
    pid = tl.program_id(0)

    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # ------------------------------------------------------------------
    # registers for the output vector
    # ------------------------------------------------------------------
    h_off = tl.arange(0, BLOCK_H)
    mask_h = h_off < H
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over the k dimension, tiled by BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile in range(num_k_tiles):
        k_off = tile * BLOCK_K + tl.arange(0, BLOCK_K)           # (K,)
        mask_k = k_off < N

        # pointers for the left slice (i, k, h)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i * stride_left_i
            + k_off * stride_left_k
            + h_off * stride_left_h
        )
        # pointers for the right slice (j, k, h)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j * stride_right_j
            + k_off * stride_right_k
            + h_off * stride_right_h
        )

        # load (K,H) tiles, mask out‑of‑range k / h
        left = tl.load(left_ptrs,
                       mask=(mask_k[:, None] & mask_h[None, :]),
                       other=0.0)                               # (K, H)
        right = tl.load(right_ptrs,
                        mask=(mask_k[:, None] & mask_h[None, :]),
                        other=0.0)                              # (K, H)

        # element‑wise product and reduction over k
        prod = left * right                                      # (K, H)
        acc += tl.sum(prod, axis=0)                              # (H,)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h_off * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_h)


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    N³ reduction.

    Args:
        data: (input, mask, weights, config)
            - input : [B, N, N, dim]   (float32, on CUDA)
            - mask  : [B, N, N]        (float32/bool, on CUDA)
            - weights: dict with the following keys
                * norm.weight, norm.bias
                * left_proj.weight, right_proj.weight
                * left_gate.weight, right_gate.weight, out_gate.weight
                * to_out_norm.weight, to_out_norm.bias
                * to_out.weight
            - config: dict with
                * dim, hidden_dim (int)
                * nomask (bool, optional)
    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)       # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton kernel: out = Σ_k left[...,k,:] * right[...,k,:]
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # ensure contiguous layout (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # kernel launch configuration
    BLOCK_K = 32                      # tile size along the reduction axis
    BLOCK_H = 32                      # vectorised hidden‑dim tile
    total = B * N * N
    grid = (triton.cdiv(total, 1),)   # 1‑D grid, one program per (b,i,j)

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_j=right.stride(1),   # note: right’s second axis is *j*
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
```
