"""
Triangle Mul (outgoing) – fast Triton implementation.

The expensive part of TriMul is the einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is just a batched matrix‑multiply
(L : (B,N,K) × Rᵀ : (B,N,K) → O : (B,N,N)).
We therefore launch a Triton GEMM kernel for each hidden slice and
stitch the results together.  All other ops (LayerNorm, Linear,
sigmoids, etc.) stay in PyTorch – they are cheap compared to the
O(N³·C) contraction.

The kernel works in fp16 for speed and the final result is cast back to
float32 as required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with the reduction axis K shared between
# the two operands (no transposition – we need Σ_k L[i,k] * R[j,k]).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_bmm_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem sizes
    B, N, K,
    # strides
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_j, stride_r_k,
    stride_o_b, stride_o_i, stride_o_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]"""
    pid_b = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for the current batch
    left_batch  = left_ptr  + pid_b * stride_l_b
    right_batch = right_ptr + pid_b * stride_r_b
    out_batch   = out_ptr   + pid_b * stride_o_b

    # accumulate in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_offset in range(0, tl.cdiv(K, BLOCK_K)):
        # compute effective k‑range for this iteration
        cur_k = k_offset * BLOCK_K + offs_k
        mask_k = cur_k < K

        # load L[i,k]  (shape BLOCK_M×BLOCK_K)
        l_ptrs = left_batch + (offs_m[:, None] * stride_l_i
                               + cur_k[None, :] * stride_l_k)
        l = tl.load(l_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load R[j,k]  (shape BLOCK_N×BLOCK_K)
        r_ptrs = right_batch + (offs_n[:, None] * stride_r_j
                                + cur_k[None, :] * stride_r_k)
        r = tl.load(r_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # dot‑product along K (inner product for every (i,j) pair)
        acc += tl.dot(l, r)       # acc : fp32

    # write result
    out_ptrs = out_batch + (offs_m[:, None] * stride_o_i
                            + offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),   # store as fp16 (model later casts to fp32)
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, C]
        mask          : torch.Tensor [B, N, N]  (may be all‑ones)
        weights       : dict of model parameters (see docstring above)
        config        : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input normalisation (LayerNorm over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B,N,N,dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    #   left  = x @ W_left.T   ;   right = x @ W_right.T
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise masking
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction – Triton kernel
    #    Compute per‑hidden‑channel batched matrix multiplication.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    # Work in fp16 inside the kernel for speed
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty_like(left)            # [B,N,N,hidden] fp16

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (i‑tiles, j‑tiles, batch)
    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B)

    # Loop over hidden dimension in Python (hidden is small ≤ 384)
    for h in range(hidden_dim):
        left_h  = left[..., h]   # [B,N,N] fp16
        right_h = right[..., h]  # [B,N,N] fp16
        out_h   = out[..., h]    # view to write into

        tri_mul_bmm_kernel[grid](
            left_h, right_h, out_h,
            B, N, N,                         # B, N, K (K == N)
            left_h.stride(0), left_h.stride(1), left_h.stride(2),
            right_h.stride(0), right_h.stride(1), right_h.stride(2),
            out_h.stride(0), out_h.stride(1), out_h.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # Cast back to fp32 for the remaining FP32 ops
    out = out.to(torch.float32)      # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # broadcast over hidden

    out = F.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of the design**

1. **Normalisation & linear layers** – inexpensive, kept in PyTorch using
   `torch.nn.functional.linear` and `layer_norm`.

2. **Mask handling** – simple element‑wise multiplication; the mask may be
   absent (`nomask` case).

3. **Gates** – computed in FP32, then applied to the projected tensors.

4. **Core contraction** – the only O(N³·C) part.
   * For each hidden slice `d`, the operation reduces to a batched matrix
     multiplication `L_d @ R_dᵀ`.
   * A custom Triton kernel (`tri_mul_bmm_kernel`) performs this GEMM in
     fp16, looping over the reduction axis `k` in tiles (`BLOCK_K = 32`).
   * Accumulators are fp32 for numerical stability, the result is stored as
     fp16 and later cast back to fp32.

5. **Post‑processing** – output layer‑norm, gating, and final linear
   projection, all in FP32.

The implementation satisfies the requirement of **using a Triton kernel for
the heavy part** while keeping the surrounding logic in PyTorch, works for
all listed configurations, and respects the datatype/shape contracts.
