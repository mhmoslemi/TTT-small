"""
TriMul “outgoing” implementation (AlphaFold‑3).

The heavy part of the module is the N‑N‑K Einstein summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for each hidden dimension *d*.
We reshape the tensors so that the hidden dimension becomes an extra batch
dimension and launch a custom Triton GEMM kernel that works on the
batched tensors (B·D, N, N).  All other operations (LayerNorm, Linear,
gates, final LayerNorm/Linear) are performed with regular PyTorch
operators – they are cheap compared to the N³ work.

The kernel is written for float‑16 inputs with a float‑32 accumulator,
matching the precision used by the reference implementation.
The final output is cast back to float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM (C = A @ B) where A, B are of shape (B, M, K) and (B, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    B, M, N, K,                         # logical sizes
    stride_ab, stride_am, stride_ak,    # A strides (batch, row, col)
    stride_bb, stride_bk, stride_bn,    # B strides (batch, row, col)
    stride_cb, stride_cm, stride_cn,    # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)          # batch dimension (actually B*D)
    pid_m = tl.program_id(1)          # row tile
    pid_n = tl.program_id(2)          # col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the first block of the current batch
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles (float16)
        a = tl.load(a_ptrs, mask=mask_m[:, None], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :], other=0.0)

        # dot product for the tile (acc stays float32)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back the result
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------

    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N]        (bool or 0/1, may be None)
            weights      : dict of weight tensors (all float32)
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """

    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]

    # ---------- Linear projections ----------
    # Linear without bias: out = x @ W.T
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the heavy contraction ----------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------- Reshape for batched GEMM ----------
    # we want: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # treat d as an extra batch dimension.
    #   left  -> (B, D, N, N)  (i,k) matrix
    #   right -> (B, D, N, N)  (j,k) matrix, need transpose on the last two axes.
    left_ = left.permute(0, 3, 1, 2).contiguous()          # (B, D, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous()        # (B, D, N, N)  (j,k) -> (k,j)

    # collapse batch and hidden dimensions
    left_flat  = left_.view(B * hidden_dim, N, N)   # (B*D, M=N, K=N)
    right_flat = right_.view(B * hidden_dim, N, N)  # (B*D, K=N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # strides for the kernel (converted to python ints)
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bk, stride_bn = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    # launch parameters (tuned for H200 – 64‑thread blocks)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                # batch (B*D)
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
    )
    batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        B * hidden_dim, N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,D)

    # ---------- to_out_norm ----------
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---------- gate & final linear ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])          # (B,N,N,dim)

    return out.to(torch.float32)
