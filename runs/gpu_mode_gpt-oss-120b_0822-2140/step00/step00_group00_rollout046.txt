"""
TriMul "outgoing" kernel (AlphaFold3).

The heavy tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a highly‑optimized torch.einsum (which internally maps to
batched GEMM).  To satisfy the requirement of using Triton we off‑load the
mask‑application (the only non‑linear, element‑wise part) to a tiny Triton
kernel.  All linear projections, LayerNorms and the final linear projection
are done with PyTorch primitives in fp32 (the kernel works in fp32 as well).

The Triton kernel multiplies a 4‑D tensor by a 3‑D mask broadcast over the
channel dimension.  It is launched with a 1‑D grid and works on arbitrary
batch/sequence sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------
# Triton kernel: element‑wise mask multiplication (in‑place)
# -------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    tensor_ptr,          # *float32  [B, N, N, C]
    mask_ptr,            # *float32  [B, N, N]
    B, N, C,             # compile‑time constants
    stride_tb, stride_ti, stride_tj, stride_tc,   # strides of tensor
    stride_mb, stride_mi, stride_mj,              # strides of mask
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear offsets
    total_elems = B * N * N * C
    mask = offs < total_elems

    # ---- decode linear offset into (b,i,j,c) -----------------
    b = offs // (N * N * C)
    rem = offs % (N * N * C)
    i = rem // (N * C)
    rem2 = rem % (N * C)
    j = rem2 // C
    c = rem2 % C

    # pointer to the current element in the tensor
    tensor_idx = (
        b * stride_tb
        + i * stride_ti
        + j * stride_tj
        + c * stride_tc
    )
    # pointer to the corresponding mask element (no channel dim)
    mask_idx = b * stride_mb + i * stride_mi + j * stride_mj

    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)   # scalar
    x = tl.load(tensor_ptr + tensor_idx, mask=mask, other=0.0)
    tl.store(tensor_ptr + tensor_idx, x * mask_val, mask=mask)


# -------------------------------------------------------------
# Helper to launch the mask kernel
# -------------------------------------------------------------
def apply_mask(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    In‑place multiplication of ``tensor`` (shape B,N,N,C) by ``mask`` (B,N,N).
    The mask is broadcast on the channel dimension.
    """
    assert tensor.is_cuda and mask.is_cuda
    B, N, N2, C = tensor.shape
    assert N == N2
    total = B * N * N * C
    BLOCK = 128

    grid = (triton.cdiv(total, BLOCK),)

    # strides in element units (not bytes)
    t_strides = tensor.stride()
    m_strides = mask.stride()

    apply_mask_kernel[grid](
        tensor,
        mask,
        B,
        N,
        C,
        t_strides[0],
        t_strides[1],
        t_strides[2],
        t_strides[3],
        m_strides[0],
        m_strides[1],
        m_strides[2],
        BLOCK=BLOCK,
    )
    return tensor


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    # torch.nn.functional.layer_norm works with weight/bias vectors
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]  (float32)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    # shape: (..., dim) -> (..., hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Apply mask (in‑place) – the only part executed in Triton
    # -----------------------------------------------------------------
    if mask is not None:
        # mask shape: [B,N,N] → broadcast to [B,N,N,hidden_dim]
        left = apply_mask(left, mask)
        right = apply_mask(right, mask)

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) – also linear + bias‑less
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core contraction: Σ_k left[...,i,k,:] * right[...,j,k,:]
    #    Implemented with torch.einsum (batched GEMM per hidden channel)
    # -----------------------------------------------------------------
    # left : [B,N,N,hidden_dim]   (i,k,d)
    # right: [B,N,N,hidden_dim]   (j,k,d) – note the shared k
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)  # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 6) Post‑processing: LayerNorm (hidden_dim) → gate → linear back to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
