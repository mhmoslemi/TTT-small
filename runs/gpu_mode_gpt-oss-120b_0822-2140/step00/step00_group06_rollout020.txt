"""
TriMul “outgoing” forward pass with a custom Triton kernel.

The heavy part is the einsum
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
which is a batched matrix multiplication for every hidden dimension *d*.
We implement it as a fused GEMM kernel that iterates over the *k* dimension
in tiles.  The kernel works on a 3‑D grid (batch*hidden, i‑tile, j‑tile)
and supports arbitrary batch size, sequence length and hidden dimension.
All remaining operations (LayerNorm, linear projections, sigmoids, final
LayerNorm + linear) are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: computes   out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  (b,i,k,d)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right (b,j,k,d)
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides for out   (b,i,j,d)
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid_bh = tl.program_id(0)          # batch * hidden
    pid_i  = tl.program_id(1)          # i‑tile
    pid_j  = tl.program_id(2)          # j‑tile

    # decode batch index and hidden index
    b = pid_bh // H
    h = pid_bh %  H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers with the fixed batch and hidden offsets
    left_base  = left_ptr  + b * stride_lb + h * stride_ld
    right_base = right_ptr + b * stride_rb + h * stride_rd
    out_base   = out_ptr   + b * stride_ob + h * stride_od

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + offs_k
        mask_k = cur_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = (
            left_base
            + (offs_i[:, None] * stride_li)   # i dimension
            + (cur_k[None, :] * stride_lk)    # k dimension
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ----- load right tile transposed (K x N) -----
        # we load as (k, j) so that a dot product gives (i, j)
        right_ptrs = (
            right_base
            + (cur_k[:, None] * stride_rk)    # k dimension first
            + (offs_j[None, :] * stride_rj)   # j dimension second
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # ----- store result -----
    out_ptrs = (
        out_base
        + (offs_i[:, None] * stride_oi)   # i dimension
        + (offs_j[None, :] * stride_oj)   # j dimension
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.
    The expensive einsum is executed by a custom Triton kernel.
    All tensors are assumed to be on the CUDA device.
    """
    # unpack inputs -------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]            # input feature dimension (C)
    hidden = cfg["hidden_dim"]  # projected hidden dimension (C_z)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (per‑position, per‑pair)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided).  mask shape: [B, N, N]
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel:  out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B,N,N,hidden
    out = torch.empty_like(left)                # same dtype & layout (float32)

    # make tensors contiguous for simple stride calculations
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides (in elements) for contiguous tensors of shape (B,N,N,hidden)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()   # (b,i,k,d)
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # kernel launch configuration ------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # batch * hidden dimension
        triton.cdiv(N, BLOCK_M),        # i‑tiles
        triton.cdiv(N, BLOCK_N),        # j‑tiles
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
