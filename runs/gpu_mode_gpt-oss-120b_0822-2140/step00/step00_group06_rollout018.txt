"""
triton TriMul “outgoing” kernel
--------------------------------
The heavy part of the TriMul forward pass is the 3‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension *d* this is a matrix multiplication
    (N×N) @ (N×N)^T .
We fuse the hidden loop with the batch dimension (B·H) and compute the
product completely in a single Triton kernel (fp16 accumulation, fp32 output).
All other operations (LayerNorm, Linear, sigmoid gates) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel : C = A @ B^T    (A,B,C : [M, N, N]), M = B*H
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    N, K,                               # matrix sizes (both = seq_len)
    stride_am, stride_ak, stride_an,    # strides for A (M,N,K)
    stride_bm, stride_bk, stride_bn,    # strides for B (M,N,K)   **will be used transposed**
    stride_cm, stride_cn, stride_ck,    # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)          # block row of C (N dimension)
    pid_n = tl.program_id(1)          # block col of C (N dimension)
    pid_z = tl.program_id(2)          # fused (batch * hidden) index

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds blocks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer arithmetic for the current block
    a_ptrs = a_ptr + pid_z * stride_am + offs_m[:, None] * stride_ak + offs_k[None, :] * stride_an
    # B is accessed as transposed: we need B_T[k, n] = B[n, k]
    b_ptrs = b_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    accum = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        accum += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_an
        b_ptrs += BLOCK_K * stride_bk

    # write back (fp16 output)
    c = accum.to(tl.float16)
    c_ptrs = c_ptr + pid_z * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_ck
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        * input_tensor : [B, N, N, D]   float32
        * mask_tensor  : [B, N, N]      (may be all‑ones)
        * weight_dict  : contains all Linear/LayerNorm weights & biases
        * config_dict  : {"dim": D, "hidden_dim": H}
    Returns
    -------
    output : torch.Tensor  [B, N, N, D]   float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, D = inp.shape
    H = cfg["hidden_dim"]
    device = inp.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (no bias in the reference code)
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,D]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left  = left  * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    # --------------------------------------------------------------
    # use fp16 for the heavy multiplication
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # reshape to (M, N, N) where M = B*H  (hidden fused with batch)
    left_mat  = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_mat   = torch.empty_like(left_mat)          # fp16 output

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H,                            # fused batch‑hidden dimension
    )

    trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        # strides for A (M,N,K)
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        # strides for B (M,N,K) – we will read it transposed, so swap last two strides
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1),
        # strides for C (M,N,N)
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and projection back to D
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
