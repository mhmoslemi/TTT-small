"""
TriMul “outgoing” kernel (AlphaFold3 style).

The expensive part of the forward pass is the bilinear reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be written per‑channel as a matrix multiplication
    out_d = left_d @ right_d.T .
We fuse this batch of GEMMs with a custom Triton kernel that works on
FP16 inputs and accumulates in FP32. All other ops (LayerNorm, Linear,
sigmoid, masking) are performed with plain PyTorch for simplicity.
The final output is cast back to float32 as required.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# --------------------------------------------------------------
# Triton batched GEMM: C[b] = A[b] @ B[b].T   (A,B,C are (B*H,N,N))
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,   # batch, row, col strides for A
    stride_Bb, stride_Br, stride_Bc,   # batch, row, col strides for B
    stride_Cb, stride_Cr, stride_Cc,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch*head index
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the current tile of A and B
    a_ptrs = A + pid * stride_Ab + offs_m[:, None] * stride_Ar + tl.arange(0, BLOCK_K)[None, :] * stride_Ac
    b_ptrs = B + pid * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Br + offs_n[None, :] * stride_Bc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                     # FP32 accumulate
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Br

    # write‑back
    c_ptrs = C + pid * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    bilinear reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32, on CUDA)
            - mask:         [B, N, N]       (float32/float16, on CUDA)
            - weights: dict of Parameter tensors (float32)
            - config:  dict with keys “dim” and “hidden_dim”

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (no bias handling needed beyond weight/bias)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gating (all bias‑free as per the reference)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask (optional – if the test case sets nomask=True the mask will be all ones)
    mask_ = mask.unsqueeze(-1)                # [B, N, N, 1]
    left  = left * mask_
    right = right * mask_

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Triton‑accelerated batch GEMM over the hidden dimension
    #    left  : [B, N, N, H] → (B*H, N, N)
    #    right : [B, N, N, H] → (B*H, N, N)
    #    out   : (B*H, N, N)  → [B, N, N, H]
    # ------------------------------------------------------------------
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # (B*H, N, N)

    # FP16 for speed, keep FP32 accumulator inside kernel
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16   = torch.empty_like(left_fp16)

    batch_dim = left_fp16.shape[0]                # B * hidden_dim

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N, N, N,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_fp16.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 4️⃣ Output normalization, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
