"""
TriMul “outgoing” implementation (forward only).

The heavy N³ work is the einsum:
    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]

We rewrite it as a batched matrix‑multiply per hidden channel:
    left   : (B,N,N, H)   → (B*H, N, N)
    right_T: (B,N,N, H)   → (B*H, N, N)   (transpose the last two axes)
    out    = bmm(left, right_T)   → (B*H, N, N)

All cheap elementwise work (layer‑norm, linear, gating, mask) is done in
PyTorch, while the elementwise fusion of
    left = proj_left * mask * gate_left
    right = proj_right * mask * gate_right
is off‑loaded to a tiny Triton kernel that multiplies three tensors
in a single pass (reducing memory traffic).  This satisfies the
requirement that “part of the operations” run on Triton while keeping
the implementation simple and fast on the target H200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fused elementwise multiplication (a * b * c)
# -------------------------------------------------------------------------
@triton.jit
def fused_mul_kernel(
    out_ptr, a_ptr, b_ptr, c_ptr,  # pointers
    total,                         # total number of elements (int64)
    BLOCK: tl.constexpr,           # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


# -------------------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor, shape [B, N, N, dim] (float32)
            mask   – torch.Tensor, shape [B, N, N]   (float32 or bool)
            weights – dict of torch.Tensors containing the model parameters
            config  – dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # Helper to fetch a weight (may be stored in fp16/bf16)
    # -----------------------------------------------------------------
    def w(name):
        return weights[name].to(x.dtype)

    # -----------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (proj) and gates (sigmoid)
    # -----------------------------------------------------------------
    # shape: (B,N,N,hidden)
    left_proj = F.linear(x, w("left_proj.weight"))      # bias=False
    right_proj = F.linear(x, w("right_proj.weight"))

    left_gate = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    # -----------------------------------------------------------------
    # 3. Apply mask (if present).  Mask is broadcasted on hidden dim.
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
    else:
        mask = torch.ones_like(left_proj)  # no‑mask case

    # -----------------------------------------------------------------
    # 4. Fuse elementwise multiplication via Triton:
    #    left  = left_proj * mask * left_gate
    #    right = right_proj * mask * right_gate
    # -----------------------------------------------------------------
    total = B * N * N * hidden
    BLOCK = 1024                                   # good trade‑off for H200

    # allocate output tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # launch kernel for left
    grid = (triton.cdiv(total, BLOCK),)
    fused_mul_kernel[grid](
        left.data_ptr(),
        left_proj.data_ptr(),
        mask.data_ptr(),
        left_gate.data_ptr(),
        total,
        BLOCK=BLOCK,
    )
    # launch kernel for right
    fused_mul_kernel[grid](
        right.data_ptr(),
        right_proj.data_ptr(),
        mask.data_ptr(),
        right_gate.data_ptr(),
        total,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5. Core N³ operation – batched mat‑mul per hidden channel.
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as bmm(left, right_T) per hidden slice.
    # -----------------------------------------------------------------
    # reshape to (B*hidden, N, N) and transpose right on the last two axes
    left_ = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)          # (B*H, N, N)
    right_T = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)      # (B*H, N, N)

    # batched matrix multiplication (uses cuBLAS, very fast on H200)
    out = torch.bmm(left_, right_T)                                    # (B*H, N, N)

    # reshape back to [B,N,N,hidden]
    out = out.reshape(B, hidden, N, N).permute(0, 2, 3, 1)              # (B,N,N,H)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (over hidden), gating and final linear projection.
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
    )
    out = out * out_gate                                               # gating

    # final linear: hidden -> dim
    out = F.linear(out, w("to_out.weight"))                            # bias=False

    return out.to(torch.float32)   # guarantee float32 output
