"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the module is the 4‑D tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]          (h = hidden_dim)

which is exactly a batched matrix multiplication
    out_h = left_h @ right_h.T   for each hidden dimension h.

The Triton kernel below computes this contraction tile‑wise over the
(i, j) output plane while looping over k.  All other operations
(LayerNorm, Linear, gating, final projection) are performed with
high‑performance PyTorch primitives.

The kernel works for any batch size B ∈ {1,2}, sequence length N up to
1024, and hidden dimension H (typically 128).  The final output is
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for the N‑N contraction per hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr,          # const float *
    right_ptr,         # const float *
    out_ptr,           # float *
    B, N, H,           # sizes (int32)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute a block of the output tensor:
        out[b, i0:i0+BM, j0:j0+BN, h] +=
            Σ_k left[b, i0:i0+BM, k, h] * right[b, j0:j0+BN, k, h]
    """
    pid = tl.program_id(0)

    # ----------------------------------------------
    # Decode the 4‑D grid: (b, h, block_i, block_j)
    # ----------------------------------------------
    total_blocks_i = tl.cdiv(N, BLOCK_M)
    total_blocks_j = tl.cdiv(N, BLOCK_N)

    block_i = pid % total_blocks_i
    pid_div = pid // total_blocks_i
    block_j = pid_div % total_blocks_j
    pid_div //= total_blocks_j
    h = pid_div % H
    b = pid_div // H

    # ----------------------------------------------
    # Offsets for this block
    # ----------------------------------------------
    i_off = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    i_mask = i_off < N
    j_mask = j_off < N

    # Base pointers for the selected batch & hidden channel
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Pointers for the first tile of k
    left_ptrs  = left_base  + i_off[:, None] * stride_l_i + k_off[None, :] * stride_l_k
    right_ptrs = right_base + j_off[:, None] * stride_r_j + k_off[None, :] * stride_r_k

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over k in tiles of size BLOCK_K
    # -------------------------------------------------
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # Load tiles (mask out‑of‑bounds entries)
        left_tile  = tl.load(left_ptrs,
                             mask=i_mask[:, None] & (k_off < N),
                             other=0.0)
        right_tile = tl.load(right_ptrs,
                             mask=j_mask[:, None] & (k_off < N),
                             other=0.0)

        # tl.dot performs Σ_k left[m,k] * right[n,k]  → (M,N)
        acc += tl.dot(left_tile, right_tile)

        # Advance k pointers
        left_ptrs  += BLOCK_K * stride_l_k
        right_ptrs += BLOCK_K * stride_r_k
        k_off += BLOCK_K

    # -------------------------------------------------
    # Store the result
    # -------------------------------------------------
    out_ptrs = out_base + i_off[:, None] * stride_o_i + j_off[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm (no bias/weight fusion needed)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    return F.layer_norm(x, x.shape[-1:], weight, bias, eps)


# ----------------------------------------------------------------------
# Entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy N³ contraction is off‑loaded to Triton.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over dim
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(dtype)            # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (int64 -> int32 for Triton)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for Hopper (SM90) 128‑KB shared memory per block
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * hidden_dim * total_blocks_i * total_blocks_j,)

    trimul_contract_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=hidden_dim,
        stride_l_b=tl.constexpr(stride_l_b), stride_l_i=tl.constexpr(stride_l_i),
        stride_l_k=tl.constexpr(stride_l_k), stride_l_h=tl.constexpr(stride_l_h),
        stride_r_b=tl.constexpr(stride_r_b), stride_r_j=tl.constexpr(stride_r_j),
        stride_r_k=tl.constexpr(stride_r_k), stride_r_h=tl.constexpr(stride_r_h),
        stride_o_b=tl.constexpr(stride_o_b), stride_o_i=tl.constexpr(stride_o_i),
        stride_o_j=tl.constexpr(stride_o_j), stride_o_h=tl.constexpr(stride_o_h),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # enough to keep the pipeline busy on Hopper
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out.to(dtype)   # preserve original dtype (usually float32)
