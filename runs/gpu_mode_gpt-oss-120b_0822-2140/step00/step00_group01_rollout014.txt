def custom_kernel(data):

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B,N,_,_ = input_tensor.shape

    # LayerNorm on dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
    else:
        mask = torch.ones_like(left)
    left = left * mask
    right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for batched gemm
    left_t = left.permute(0,3,1,2).contiguous()   # [B, hidden, N, N]
    right_t = right.permute(0,3,2,1).contiguous() # transpose last two dims for B^T

    B_h = B * hidden_dim
    A = left_t.view(B_h, N, N)
    Bmat = right_t.view(B_h, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

"""
TriMul “outgoing” forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm + 4 linear projections are done with PyTorch (fast and
   easy to express).
2. After applying the mask and the three sigmoid‑gates we obtain
   tensors ``left`` and ``right`` of shape
   (B, N, N, H) (H = hidden_dim).  The core operation is

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   which is exactly a batched matrix multiplication
        (left_per_h) @ (right_per_h).T
   for each (batch, h) pair.

3. We reshape the tensors to (B*H, N, N) and launch a Triton kernel that
   computes a tiled GEMM for each batch‑head.  The kernel works in FP16
   (tensor‑core friendly) while accumulating in FP32 for numeric
   stability.

4. The result is reshaped back to (B, N, N, H) and passed through the
   output LayerNorm, multiplied by the output‑gate and a final linear
   projection back to ``dim``.  The final tensor is returned in FP32.

The heavy N³ work is fully performed by the Triton kernel; the rest of
the module is cheap PyTorch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape: [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _triton_gemm_outgoing(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                     # all equal to seq_len
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N)  (already transposed)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # --------------------------------------------------------------
    # offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # init accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # A block: (m, k)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # B block: (k, n) – B already transposed
        b_ptrs = (
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator with a Triton GEMM.

    Args:
        data: Tuple containing
              - input tensor    : [B, N, N, dim]      (torch.float32)
              - mask tensor     : [B, N, N]           (torch.float32 or bool)
              - weights dict    : model parameters (tensors on the same device)
              - config dict     : {"dim": int, "hidden_dim": int}
    Returns:
        Tensor of shape [B, N, N, dim] (torch.float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask may be omitted → treat as all‑ones)
    # ------------------------------------------------------------------
    if mask is None:
        mask_exp = torch.ones_like(left, dtype=left.dtype, device=device)
    else:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]

    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy N³ GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Rearrange to (batch*head, N, N) and transpose right for Bᵀ
    # ------------------------------------------------------------------
    #   left  : [B, N, N, H] -> [B, H, N, N] -> (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    #   right : [B, N, N, H] -> [B, H, N, N] -> transpose last two dim
    right_t = right.permute(0, 3, 2, 1).contiguous()

    left_flat = left_t.view(B * hidden, N, N)
    right_flat = right_t.view(B * hidden, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 7. Launch Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _triton_gemm_outgoing[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out_hidden = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_hidden = out_hidden * out_gate  # element‑wise

    # final linear back to dim (bias‑free)
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out.to(torch.float32)
