"""
TriMul “outgoing” implementation (AlphaFold‑3) with a custom Triton kernel.

The expensive part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of hd (= hidden_dim) matrix‑multiplications:
    left_b,d   (N×N)  @  right_b,d.T (N×N)   →  out_b,d (N×N)

The kernel below performs this batched mat‑mul for all
batch * hidden_dim matrices in a single launch, using
float32 accumulation and float16 traffic (the H200’s
Tensor‑Core pathways are exercised implicitly by cuBLAS‑like
dot‑products inside the kernel).

All surrounding operations (LayerNorm, linear projections,
gating, final LayerNorm + linear) are kept in PyTorch – they
are cheap compared with the N³ work of the einsum.

The function `custom_kernel` follows the required signature:
    data = (input_tensor, mask, weights, config)
and returns a float‑32 tensor of shape [B, N, N, dim].
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul  (left @ right.T)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # dimensions
    N,
    # strides (batch, row, col)
    stride_lbatch, stride_lrow, stride_lcol,
    stride_rbatch, stride_rrow, stride_rcol,
    stride_obatch, stride_orow, stride_ocol,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes out = left @ right.T   for each “batch” (B * hidden_dim).

    left, right, out : shape (BHD, N, N)   (row‑major)
    """
    pid = tl.program_id(0)                     # linear index over (batch, block_m, block_n)

    # ------------------------------------------------------------------
    # decode pid → batch, block row, block col
    # ------------------------------------------------------------------
    batch = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    rem   = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_m = rem // tl.cdiv(N, BLOCK_N)
    block_n = rem % tl.cdiv(N, BLOCK_N)

    # ------------------------------------------------------------------
    # compute base pointers for this batch
    # ------------------------------------------------------------------
    left_batch_ptr  = left_ptr  + batch * stride_lbatch
    right_batch_ptr = right_ptr + batch * stride_rbatch
    out_batch_ptr   = out_ptr   + batch * stride_obatch

    # ------------------------------------------------------------------
    # offsets inside the M×N tile
    # ------------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the current tile
    a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_lrow
                               + offs_k[None, :] * stride_lcol)
    b_ptrs = right_batch_ptr + (offs_k[:, None] * stride_rrow
                                + offs_n[None, :] * stride_rcol)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K‑loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < N),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < N) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                     # (M,K)·(K,N) → (M,N)

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_lcol
        b_ptrs += BLOCK_K * stride_rrow

    # ------------------------------------------------------------------
    # store the result
    # ------------------------------------------------------------------
    c_ptrs = out_batch_ptr + (offs_m[:, None] * stride_orow
                              + offs_n[None, :] * stride_ocol)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]   (float32)
            mask_tensor  : [B, N, N]       (float32 or bool)
            weights_dict : contains all linear / LN parameters
            config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the *dim* channel
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – all in float32
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (shape [B,N,N,1])
    # ------------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)          # broadcast to hidden dim
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul (batched mat‑mul) – Triton kernel
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)  → each “batch” is one hidden channel
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # strides for the contiguous layout (batch, row, col)
    stride_lbatch, stride_lrow, stride_lcol = left_flat.stride()
    stride_rbatch, stride_rrow, stride_rcol = right_flat.stride()
    stride_obatch, stride_orow, stride_ocol = out_flat.stride()

    # block sizes – tuned for H200 (64 is a good compromise)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_blocks = (triton.cdiv(N, BLOCK_M) *
                    triton.cdiv(N, BLOCK_N) *
                    (B * hidden_dim))

    grid = (total_blocks,)

    _trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_lbatch,
        stride_lrow,
        stride_lcol,
        stride_rbatch,
        stride_rrow,
        stride_rcol,
        stride_obatch,
        stride_orow,
        stride_ocol,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden_dim and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                                 # gating after LN
    out = F.linear(out, weights["to_out.weight"])        # final projection

    return out
