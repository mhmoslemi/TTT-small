"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the module is the contraction

    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected‑and‑gated pairwise tensors.
The kernel evaluates this contraction directly on the GPU.
All other linear / layer‑norm / gating operations are done with
regular PyTorch kernels – they are cheap compared with the O(N³·C) work.

The kernel works on blocks of the (i,j) plane (BLOCK_M × BLOCK_N) and
processes a small block of the hidden dimension (H_BLOCK).  Inside the
kernel we iterate over the k‑dimension in tiles of size BLOCK_K,
load the two tiles, and accumulate a matrix‑multiply (tl.dot) for each
hidden‑channel slice.  The implementation uses fp16 inside the kernel
and casts the final result back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,          # const * fp16   [B,N,N,H]
    right_ptr,         # const * fp16   [B,N,N,H]
    out_ptr,           #       * fp16   [B,N,N,H]
    B, N, H,           # tensor sizes (int32)
    # strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H_BLOCK: tl.constexpr,
):
    """Compute out = Σ_k left[...,i,k,d] * right[...,j,k,d].

    Grid dimensions:
        dim0 – batch index
        dim1 – i‑tile
        dim2 – j‑tile
        dim3 – hidden‑channel block
    """
    pid_b   = tl.program_id(0)        # batch
    pid_i   = tl.program_id(1)        # i‑tile
    pid_j   = tl.program_id(2)        # j‑tile
    pid_h   = tl.program_id(3)        # hidden‑channel block

    # offsets for i, j and hidden channels within the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * H_BLOCK + tl.arange(0, H_BLOCK)

    # masks for boundary handling
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # pointers that already include the batch offset
    left_batch_ptr  = left_ptr  + pid_b * stride_left_b
    right_batch_ptr = right_ptr + pid_b * stride_right_b
    out_batch_ptr   = out_ptr   + pid_b * stride_out_b

    # accumulator for the (i,j) tile, one accumulator per hidden channel
    acc = tl.zeros((BLOCK_M, BLOCK_N, H_BLOCK), dtype=tl.float32)

    # iterate over k in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------- load left tile -------------------
        # left[i,k,h]  -> shape (BLOCK_M, BLOCK_K, H_BLOCK)
        left_ptrs = (
            left_batch_ptr
            + (offs_i[:, None, None] * stride_left_i)
            + (offs_k[None, :, None] * stride_left_k)
            + (offs_h[None, None, :] * stride_left_h)
        )
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                            other=0.0).to(tl.float32)          # (M,K,Hb)

        # ------------------- load right tile ------------------
        # right[j,k,h]  -> shape (BLOCK_N, BLOCK_K, H_BLOCK)
        right_ptrs = (
            right_batch_ptr
            + (offs_j[:, None, None] * stride_right_j)
            + (offs_k[None, :, None] * stride_right_k)
            + (offs_h[None, None, :] * stride_right_h)
        )
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                             other=0.0).to(tl.float32)         # (N,K,Hb)

        # ------------------- matmul per hidden channel ----------
        # we need Σ_k left[i,k,h] * right[j,k,h]
        # we can use tl.dot for each channel slice
        for hb in range(H_BLOCK):
            # (M,K) dot (K,N) -> (M,N)
            acc[:, :, hb] += tl.dot(left_tile[:, :, hb], right_tile[:, :, hb].T)

    # ------------------- store result --------------------
    out_ptrs = (
        out_batch_ptr
        + (offs_i[:, None, None] * stride_out_i)
        + (offs_j[None, :, None] * stride_out_j)
        + (offs_h[None, None, :] * stride_out_h)
    )
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input:  [B, N, N, dim]    (float32)
            - mask :  [B, N, N]        (float32 or bool)
            - weights: dict with the nine weight tensors
            - config : {"dim": dim, "hidden_dim": hidden_dim}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections ----------
    # Linear without bias: y = x @ W.T
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)                # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ---------- Cast to fp16 for the contraction ----------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # ---------- Strides (int64) ----------
    # left/right/out are [B,N,N,H]; we use row‑major (C) layout
    stride_b = left_fp16.stride(0)
    stride_i = left_fp16.stride(1)
    stride_k = left_fp16.stride(2)
    stride_h = left_fp16.stride(3)

    # launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    H_BLOCK = 8                     # must divide hidden_dim (8 works for 128/384)

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden_dim, H_BLOCK),
    )

    trimul_out_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,   # left strides
        stride_b, stride_i, stride_k, stride_h,   # right strides (j uses same i‑stride)
        stride_b, stride_i, stride_k, stride_h,   # out strides
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, H_BLOCK=H_BLOCK,
    )

    # ---------- Back to fp32 ----------
    out = out_fp16.to(torch.float32)

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ---------- Final linear ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
