"""
TriMul “outgoing” implementation with a custom Triton kernel.

The most expensive part of the module is the 4‑D Einstein sum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply per hidden channel:

    left  : (B, D, N, N)   – (i,k)   axes
    right : (B, D, N, N)   – (k,j)   axes (i.e. right transposed)

We fuse the reduction over *k* with a tiled GEMM written in Triton.
All other linear layers, gate‑sigmoids and LayerNorms are performed with
torch (they are cheap compared with the O(B·D·N³) work of the GEMM).

The kernel works on fp16 inputs and accumulates in fp32 for numerical
stability.  Final output is cast back to fp32 as required by the spec.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton kernel:  per‑batch‑channel GEMM  (left @ rightᵀ)
# --------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,               # * (B, D, N, N)   left  – (i,k)
    right_ptr,              # * (B, D, N, N)   right – (k,j)  (already transposed)
    out_ptr,                # * (B, D, N, N)   output – (i,j)
    B, D, N,
    stride_left_b, stride_left_d, stride_left_i, stride_left_k,
    stride_right_b, stride_right_d, stride_right_k, stride_right_j,
    stride_out_b, stride_out_d, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_bd = tl.program_id(0)               # flattened (batch, hidden)
    pid_i  = tl.program_id(1)               # row tile
    pid_j  = tl.program_id(2)               # col tile

    b = pid_bd // D
    d = pid_bd % D

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary handling
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the (b,d) slice
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (i,k)
        left_ptrs = (left_base +
                     offs_i[:, None] * stride_left_i +
                     offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (k,j)   (right is already transposed)
        right_ptrs = (right_base +
                      offs_k[:, None] * stride_right_k +
                      offs_j[None, :] * stride_right_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # GEMM on the tile
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back
    out_ptrs = (out_base +
                offs_i[:, None] * stride_out_i +
                offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------------
# Helper: LayerNorm without bias/weight handling (fast torch call)
# --------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: (..., C)   weight/bias: (C,)
    return F.layer_norm(x, x.shape[-1:], weight, bias, eps)

# --------------------------------------------------------------------
# Entry point required by the evaluator
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton GEMM.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (float32, 0/1)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = layer_norm(x,
                   weight=weights["norm.weight"],
                   bias=weights["norm.bias"])

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # shape: (B,N,N,hidden)
    left  = F.linear(x, weights["left_proj.weight"])   # (dim, hidden) -> result (..., hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask if present (mask is always supplied, may be all‑ones)
    # --------------------------------------------------------------
    mask_f = mask.unsqueeze(-1)          # (B,N,N,1)
    left  = left * mask_f
    right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of left/right
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the Triton GEMM
    # --------------------------------------------------------------
    #   left  : (B, N, N, hidden) -> (B, hidden, N, N)
    #   right : (B, N, N, hidden) -> transpose k<->j then (B, hidden, N, N)
    left_t   = left.permute(0, 3, 1, 2).contiguous()          # (B, D, N, i)
    right_t  = right.permute(0, 3, 2, 1).contiguous()         # (B, D, k, j)  (already transposed)

    # Cast to fp16 for the kernel (Hopper excels at fp16 math)
    left_t  = left_t.to(torch.float16)
    right_t = right_t.to(torch.float16)

    # Output buffer (fp16)
    out_t = torch.empty_like(left_t, dtype=torch.float16, device=x.device)

    # --------------------------------------------------------------
    # 6. Launch Triton kernel
    # --------------------------------------------------------------
    # Block sizes – tunable; 64 works well on Hopper for these shapes.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden, 1),          # one program per (batch, channel)
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides for the four‑dim tensors (after the permutes)
    l_strides = left_t.stride()   # (stride_b, stride_d, stride_i, stride_k)
    r_strides = right_t.stride()
    o_strides = out_t.stride()

    trimul_gemm_kernel[grid](
        left_t, right_t, out_t,
        B, hidden, N,
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # H200 supports many warps; 4 is a safe default
    )

    # Convert back to fp32 and restore original layout (B,N,N,hidden)
    out = out_t.to(torch.float32).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Final LayerNorm + out‑gate + projection to original dim
    # --------------------------------------------------------------
    out = layer_norm(out,
                     weight=weights["to_out_norm.weight"],
                     bias=weights["to_out_norm.bias"])

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (hidden, dim) → (B,N,N,dim)

    return out
