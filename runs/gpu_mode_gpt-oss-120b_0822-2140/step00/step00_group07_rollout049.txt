def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # apply mask if exists
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_
        right = right * mask_

    left = left * left_gate
    right = right * right_gate

    # allocate out tensor
    out = torch.empty((bs, N, N, hidden), dtype=torch.float32, device=device)

    # compute strides
    left_ptr = left
    right_ptr = right
    out_ptr = out

    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # launch triton kernel
    import triton
    import triton.language as tl

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), bs)

    tri_mul_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        bs, N, hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul “outgoing” kernel (AlphaFold‑3 style) – fast contraction.

The heavy O(N³·C) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel.  Everything else
(LayerNorm, linear projections, gating) stays in PyTorch – the
projected tensors are only (B,N,N,C_hidden) and the kernel
reduces over the “k” dimension with a tiled broadcast‑multiply
+ sum.  The kernel works for any batch size ≤2, any sequence
length up to 1024 and hidden dimensions that are multiples of
32 (the last tile is masked).  The final output is returned in
float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel:   out = Σ_k left[i,k,:] * right[j,k,:]
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                   # const float*   left  [B,N,N,C]
    right_ptr,                  # const float*   right [B,N,N,C]
    out_ptr,                    # float*         out   [B,N,N,C]
    B, N, C,                    # int32 scalars
    # strides for left   (B,N,K,C)
    stride_l_b, stride_l_i, stride_l_k, stride_l_c,
    # strides for right  (B,N,K,C)
    stride_r_b, stride_r_i, stride_r_k, stride_r_c,
    # strides for out    (B,N,N,C)
    stride_o_b, stride_o_i, stride_o_j, stride_o_c,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,     # i‑tile
    BLOCK_N: tl.constexpr,     # j‑tile
    BLOCK_K: tl.constexpr,     # k‑tile
    BLOCK_D: tl.constexpr,     # hidden‑tile
):
    pid_b = tl.program_id(2)                 # batch dimension
    pid_i = tl.program_id(0)                 # i block
    pid_j = tl.program_id(1)                 # j block

    # ------------------------------------------------------------------
    # index ranges for this block
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # ------------------------------------------------------------------
    # accumulate over hidden tiles
    # ------------------------------------------------------------------
    for d_start in range(0, C, BLOCK_D):
        d = d_start + tl.arange(0, BLOCK_D)
        d_mask = d < C

        # accumulator for this (i,j,d) tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # reduction over k
        # ------------------------------------------------------------------
        for k_start in range(0, N, BLOCK_K):
            k = k_start + tl.arange(0, BLOCK_K)
            k_mask = k < N

            # ------------ left tile  (M,K,D) ---------------------------
            left_off = (
                pid_b * stride_l_b
                + i[:, None, None] * stride_l_i
                + k[None, :, None] * stride_l_k
                + d[None, None, :] * stride_l_c
            )
            left = tl.load(
                left_ptr + left_off,
                mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                other=0.0,
            )   # shape (M,K,D)

            # ------------ right tile (N,K,D) ---------------------------
            right_off = (
                pid_b * stride_r_b
                + j[:, None, None] * stride_r_i
                + k[None, :, None] * stride_r_k
                + d[None, None, :] * stride_r_c
            )
            right = tl.load(
                right_ptr + right_off,
                mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                other=0.0,
            )   # shape (N,K,D)

            # ------------ broadcast‑multiply & sum over k -------------
            # for each k we add left[:,k,:] * right[:,k,:]^T
            for kk in range(0, BLOCK_K):
                # safety mask for the last k‑tile
                kk_valid = (k_start + kk) < N
                if not kk_valid:
                    continue

                l = left[:, kk, :]          # (M,D)
                r = right[:, kk, :]         # (N,D)

                # expand dims to (M,1,D) and (1,N,D) → broadcast to (M,N,D)
                l_exp = l[:, None, :]       # (M,1,D)
                r_exp = r[None, :, :]       # (1,N,D)

                acc += l_exp * r_exp        # (M,N,D)

        # ------------------------------------------------------------------
        # write out the accumulated tile
        # ------------------------------------------------------------------
        out_off = (
            pid_b * stride_o_b
            + i[:, None, None] * stride_o_i
            + j[None, :, None] * stride_o_j
            + d[None, None, :] * stride_o_c
        )
        tl.store(
            out_ptr + out_off,
            acc,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
        )


# --------------------------------------------------------------
# Host‑side entry point expected by the grader
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim]   (float32, on CUDA)
            mask   – torch.Tensor [B, N, N] (may be None)
            weights – dict of model parameters (all float32 tensors)
            config – dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections + gating
    # --------------------------------------------------------------
    # (B,N,N,hidden) = (B,N,N,dim) @ (hidden, dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # optional mask – broadcasted to the hidden dim
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Heavy contraction – Triton kernel
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (int64 -> python int)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32   # hidden dimension tile

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )

    tri_mul_kernel[
        grid
    ](
        left, right, out,
        B, N, hidden,
        # left strides
        s_l[0], s_l[1], s_l[2], s_l[3],
        # right strides
        s_r[0], s_r[1], s_r[2], s_r[3],
        # out strides
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 4. Post‑processing: LayerNorm, output gate, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
