"""
TriMul “outgoing” kernel (AlphaFold‑3)

The heavy part of TriMul is the three‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden channel *d*:
    out[d] = left[d] @ right[d].T   (B × N × N → B × N × N)

The kernel below tiles the (i,j) plane (BLOCK_M × BLOCK_N) and the
k‑reduction (BLOCK_K).  The hidden dimension is tiled with BLOCK_H
(≤ 32) and accumulated in fp32 before being stored as fp16.
All layer‑norms, linear projections and gating are kept in PyTorch –
only the O(N³·H) contraction is executed in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    L_ptr,               # * left  [B,N,N,H]
    R_ptr,               # * right [B,N,N,H]
    O_ptr,               # * out   [B,N,N,H]
    B, N, H,
    stride_LB, stride_Li, stride_Lk, stride_Lh,   # strides of left
    stride_RB, stride_Ri, stride_Rk, stride_Rh,   # strides of right
    stride_OB, stride_Oi, stride_Oj, stride_Oh,   # strides of out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)

    # ----- 2‑D tiling over (i,j) and batch ----------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch_id = pid // tiles_per_batch
    tile_id  = pid %  tiles_per_batch
    tile_m   = tile_id // num_tiles_n
    tile_n   = tile_id % num_tiles_n

    # offsets inside the current tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator: (BLOCK_M, BLOCK_N, BLOCK_H) in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----- reduction over k --------------------------------------------
    for k_base in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_base * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a block of left   : (BM, BK, BH)
        L = tl.load(
            L_ptr
            + batch_id * stride_LB
            + offs_m[:, None, None] * stride_Li
            + offs_k[None, :, None] * stride_Lk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_Lh,
            mask=(mask_m[:, None, None] & mask_k[None, :, None]),
            other=0.0,
        )
        # load a block of right  : (BN, BK, BH)
        R = tl.load(
            R_ptr
            + batch_id * stride_RB
            + offs_n[:, None, None] * stride_Ri
            + offs_k[None, :, None] * stride_Rk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_Rh,
            mask=(mask_n[:, None, None] & mask_k[None, :, None]),
            other=0.0,
        )

        # broadcast to (BM, BN, BK, BH) and accumulate over k
        # L : (BM, 1, BK, BH)   R : (1, BN, BK, BH)
        prod = (L[:, None, :, :] * R[None, :, :, :])          # (BM, BN, BK, BH)
        acc += tl.sum(prod, axis=2)                           # sum over k -> (BM, BN, BH)

    # ----- write result -------------------------------------------------
    h_offs = tl.arange(0, BLOCK_H)
    mask_h = h_offs < H

    # pointer arithmetic for the output
    out_ptrs = (
        O_ptr
        + batch_id * stride_OB
        + offs_m[:, None, None] * stride_Oi
        + offs_n[None, :, None] * stride_Oj
        + h_offs[None, None, :] * stride_Oh
    )

    # store as fp16 (final tensor will be cast back to fp32 in host code)
    tl.store(out_ptrs, acc.to(tl.float16), mask=(mask_m[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :]))


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Optimized forward pass of the “outgoing” TriMul operator.
    Args:
        data: (input, mask, weights, config)
            - input:  [B, N, N, dim]      (float32)
            - mask :  [B, N, N]           (float32 or bool)
            - weights: dict with all Linear/LayerNorm parameters
            - config : {dim, hidden_dim, nomask}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left  * mask
        right = right * mask

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Triton kernel: batched matmul over the “k” dimension for every hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16   = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides of the tensors (B, N, N, H) – PyTorch layout is contiguous
    stride_L = left_fp16.stride()   # (B*N*N*H, N*N*H, N*H, H)
    stride_R = right_fp16.stride()
    stride_O = out_fp16.stride()

    # tiling parameters (tuned for Hopper – feel free to adjust)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 32                     # hidden dim is ≤ 384, fits well

    grid = (
        triton.cdiv(B, 1)              # batch dimension is handled inside kernel, we launch one grid per batch tile
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )

    trimul_einsum_kernel[grid](
        L_ptr=left_fp16,
        R_ptr=right_fp16,
        O_ptr=out_fp16,
        B=B,
        N=N,
        H=hidden_dim,
        stride_LB=stride_L[0],
        stride_Li=stride_L[1],
        stride_Lk=stride_L[2],
        stride_Lh=stride_L[3],
        stride_RB=stride_R[0],
        stride_Ri=stride_R[1],
        stride_Rk=stride_R[2],
        stride_Rh=stride_R[3],
        stride_OB=stride_O[0],
        stride_Oi=stride_O[1],
        stride_Oj=stride_O[2],
        stride_Oh=stride_O[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # Convert back to FP32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
