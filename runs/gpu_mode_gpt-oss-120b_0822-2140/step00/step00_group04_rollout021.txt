"""
TriMul “outgoing” implementation with a Triton kernel for the costly
einsum (batch‑wise GEMM over the sequence dimension).

The heavy part computes, for each batch b, hidden channel h and
positions i, j:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is equivalent to a batched matrix multiplication
    out_h = left_h @ right_h.T
with left_h, right_h ∈ ℝ^{N×N}.  The kernel tiles over i, j and
accumulates over k using fp16 loads and fp32 accumulation.
All other operations (LayerNorm, Linear, gating, final projection)
are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled GEMM for the “einsum” part
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,        # pointers
    B, N, H,                             # shapes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    with left, right ∈ fp16 and out ∈ fp32.
    Grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    """
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # combined batch‑head index

    # ------------------------------------------------------------------
    # decode batch and hidden‑channel indices
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # compute base pointers for this (b, h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # ------------------------------------------------------------------
    # pointers that will be incremented over the K‑dimension
    # ------------------------------------------------------------------
    left_ptrs  = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    right_ptrs = right_base + offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k_tiles):
        # load with mask, cast to fp16 (original data are fp16)
        left = tl.load(left_ptrs,
                       mask=(mask_i[:, None] & (offs_k[None, :] < N)),
                       other=0.0).to(tl.float16)
        right = tl.load(right_ptrs,
                        mask=((offs_k[:, None] < N) & mask_j[None, :]),
                        other=0.0).to(tl.float16)

        # GEMM on this tile
        acc += tl.dot(left, right)                     # fp32 accumulation

        # advance pointers to the next K‑tile
        left_ptrs  += BLOCK_K * stride_l_k
        right_ptrs += BLOCK_K * stride_r_i

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask_tensor  : [B, N, N]      (float32, 0/1)
            - weights_dict : contains the eight weight/bias tensors
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to fp16 for the GEMM kernel (memory‑bandwidth saving)
    # --------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 6. Triton GEMM (the heavy N³ part)
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable block sizes (empirically good for Hopper)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,
    )

    _trimul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden,
        stride_l_b, stride_l_i, stride_l_j, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7. Post‑processing (LayerNorm, gating, final linear)
    # --------------------------------------------------------------
    # to_out_norm over the hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    # multiply by out_gate (shape broadcasted)
    out = out * out_gate

    # final projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of key design choices**

* **Kernel focus** – The `einsum` is the only O(N³) operation; it is rewritten as a
  batched GEMM and implemented in Triton. All other linear layers, LayerNorms,
  and sigmoid gates stay in PyTorch (they are cheap compared to N³ work).

* **Memory layout** – The input tensors are in the standard PyTorch layout
  (contiguous, stride order `B, i, j, h`).  The kernel loads `left[i,k]` and
  `right_T[k,j]` by swapping the stride for the `j`‑dimension, avoiding any
  explicit transposition.

* **Precision** – `left` and `right` are cast to `fp16` before the kernel;
  accumulation is performed in `fp32` (`tl.dot` returns float32), matching the
  reference implementation while reducing memory bandwidth.

* **Tiling** – 32×32×32 tiles fit well in the shared memory of Hopper SM90 and
  give high occupancy.  The grid is `(ceil(N/32), ceil(N/32), B*hidden)`,
  covering all batches and hidden channels.

* **Correctness** – The kernel mirrors exactly the mathematical definition of
  the einsum, and the surrounding PyTorch code reproduces the original
  `TriMul` forward pass (including optional masking).

The function `custom_kernel` complies with the required signature, launches the
Triton kernel, and returns a float‑32 tensor of shape `[B, N, N, dim]`.
