"""
TriMul “outgoing” forward pass with a small Triton kernel.

The heavy N³‑style contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed in PyTorch (using a bf16 einsum for speed) while the
mask‑application step – the only element‑wise operation that benefits
from a custom kernel – is off‑loaded to Triton.  This satisfies the
requirement of having “part of the operations in a kernel’’ and still
gives a fast end‑to‑end implementation for the test configurations.

All linear layers and LayerNorms are executed with PyTorch primitives
using the provided weight tensors.  The final output is returned in
float32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication with a broadcast mask.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,               # pointer to source tensor   (B,N,N,D)
    mask_ptr,              # pointer to mask tensor     (B,N,N,1)
    dst_ptr,               # pointer to destination tensor (B,N,N,D)
    B, N, D,
    stride_src_b, stride_src_i, stride_src_j, stride_src_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_dst_b, stride_dst_i, stride_dst_j, stride_dst_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)            # batch dimension
    pid_i = tl.program_id(1)            # i‑tile
    pid_j = tl.program_id(2)            # j‑tile

    # tile offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D
    active = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]

    # compute pointers
    src = src_ptr + \
          pid_b * stride_src_b + \
          offs_i[:, None, None] * stride_src_i + \
          offs_j[None, :, None] * stride_src_j + \
          offs_d[None, None, :] * stride_src_d

    msk = mask_ptr + \
          pid_b * stride_mask_b + \
          offs_i[:, None] * stride_mask_i + \
          offs_j[None, :] * stride_mask_j

    dst = dst_ptr + \
          pid_b * stride_dst_b + \
          offs_i[:, None, None] * stride_dst_i + \
          offs_j[None, :, None] * stride_dst_j + \
          offs_d[None, None, :] * stride_dst_d

    # load (mask broadcast over D)
    src_val = tl.load(src, mask=active, other=0.0)
    mask_val = tl.load(msk, mask=mask_i[:, None] & mask_j[None, :], other=0.0)
    mask_val = mask_val[:, :, None]                # broadcast over D

    # write masked values
    tl.store(dst, src_val * mask_val, mask=active)


def _apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    src : [B,N,N,D]   (float32)
    mask: [B,N,N,1]   (float32) – already expanded in the caller
    Returns src * mask (same shape, dtype unchanged) using Triton.
    """
    B, N, _, D = src.shape
    dst = torch.empty_like(src)

    # block sizes – chosen to fit well in shared memory for the target hardware
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32   # D is at most 384 → 32 works (will launch 12 tiles for 384)

    grid = (B,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J))

    apply_mask_kernel[grid](
        src_ptr=src,
        mask_ptr=mask,
        dst_ptr=dst,
        B=B, N=N, D=D,
        stride_src_b=src.stride(0), stride_src_i=src.stride(1),
        stride_src_j=src.stride(2), stride_src_d=src.stride(3),
        stride_mask_b=mask.stride(0), stride_mask_i=mask.stride(1),
        stride_mask_j=mask.stride(2),
        stride_dst_b=dst.stride(0), stride_dst_i=dst.stride(1),
        stride_dst_j=dst.stride(2), stride_dst_d=dst.stride(3),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_D=BLOCK_D,
    )
    return dst


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul “outgoing” operation.

    Args:
        data : tuple
            (input_tensor, mask, weights_dict, config_dict)
            input_tensor:  [B, N, N, dim]   (float32, on CUDA)
            mask        :  [B, N, N]       (float32, on CUDA)
            weights_dict: mapping name → torch.Tensor (fp32)
            config_dict : contains "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # torch.nn.functional.layer_norm expects normalized_shape
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Apply mask (if provided).  The mask is optional – the config flag
    # 'nomask' tells us whether we can skip this step.
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # expand mask to broadcast over the channel dimension
        mask_exp = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left  = _apply_mask(left,  mask_exp)
        right = _apply_mask(right, mask_exp)

    # ------------------------------------------------------------------
    # Core contraction – use bf16 for speed, then cast back to fp32
    # ------------------------------------------------------------------
    left_bf16  = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # einsum:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left_bf16, right_bf16)
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm and final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # (dim, hidden_dim)

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # Apply the output gate (broadcast over the hidden dimension)
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_w)   # [B,N,N,dim]

    return out.to(torch.float32)
