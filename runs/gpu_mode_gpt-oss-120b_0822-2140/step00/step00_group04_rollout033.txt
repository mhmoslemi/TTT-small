"""
TriMul “outgoing” implementation with a fused Triton kernel.

The heavy N³ part of TriMul is the einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is simply a batched matrix multiply per hidden dimension:
    out_d = left_d @ right_dᵀ   (shape B×H × N × N)

The kernel below computes the product for every (batch, hidden) pair in
parallel, loading tiles of the K‑dimension (the summed axis) and using
`tl.dot`.  All other operations (LayerNorms, Linear projections, gates,
and the final linear projection) stay in PyTorch so that we can reuse the
provided weight tensors directly.  The kernel works with FP16
intermediate storage (the model is normally trained in FP16/BF16) and the
final result is cast back to FP32 as required by the spec.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   (batch‑wise, per hidden)
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # sizes
    # strides for left (b, i, k, h)
    stride_l_bn, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (b, j, k, h)
    stride_r_bn, stride_r_j, stride_r_k, stride_r_h,
    # strides for out (b, i, j, h)
    stride_o_bn, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ----------------------------------------------------------
    # grid dimensions:
    #   program_id(0) : batch * hidden   (B*H)
    #   program_id(1) : block row i (M)
    #   program_id(2) : block col j (N)
    # ----------------------------------------------------------
    bh_idx = tl.program_id(0)          # 0 … B*H-1
    pid_m = tl.program_id(1)           # block row
    pid_n = tl.program_id(2)           # block col

    # decode batch and hidden index
    b = bh_idx // H
    h = bh_idx % H

    # offsets inside the N×N matrix
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer base for this (b, h)
    base_l = left_ptr + b * stride_l_bn + h * stride_l_h
    base_r = right_ptr + b * stride_r_bn + h * stride_r_h
    base_o = out_ptr + b * stride_o_bn + h * stride_o_h

    # accumulator in FP32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left tile  (M × K) -------
        ptr_l = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(ptr_l, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ------- load right tile (K × N) -------
        # right is indexed as (b, j, k, h); we need K as first dim, N as second
        ptr_r = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(ptr_r, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # ----------------------------------------------------------
    # write result (cast back to FP16 for storage)
    # ----------------------------------------------------------
    ptr_o = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(ptr_o, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    core multiplication.

    Args:
        data: tuple (input, mask, weights, config)
            - input : Tensor[B, N, N, dim]  (float32 or float16, already on CUDA)
            - mask  : Tensor[B, N, N]        (bool/float, already on CUDA)
            - weights: dict with all Linear/LayerNorm parameters
            - config : dict with keys dim, hidden_dim, nomask (bool)

    Returns:
        Tensor[B, N, N, dim]  (float32)
    """
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) -> shape [B,N,N,hidden]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel: out = left @ rightᵀ   (B×H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Cast to FP16 for the kernel (most H200 kernels are FP16‑first)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty((B, N, N, hidden), dtype=torch.float16, device=x.device)

    # stride helpers (torch uses (B,N,N,H) ordering)
    s_l = left_fp16.stride()   # (BNH, NH, H, 1)
    s_r = right_fp16.stride()
    s_o = out_fp16.stride()

    # tile sizes – chosen to fit H200 SMEM (128×128 work per block)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden,                     # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden,
        # strides for left
        s_l[0], s_l[1], s_l[2], s_l[3],
        # strides for right
        s_r[0], s_r[1], s_r[2], s_r[3],
        # strides for out
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out_gate (elementwise), final Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
