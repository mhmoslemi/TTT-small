"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the 4‑D input   X ∈ R[B,N,N,D].
2. Two linear projections → left/right   (shape [B,N,N,H]).
3. Optional element‑wise mask applied to both projections.
4. Gated versions of the projections:
        left  ← left  * sigmoid(Linear_g(x))
        right ← right * sigmoid(Linear_g(x))
5. Core “tri‑multiplicative” update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication for every hidden channel d.
   We compute it with a custom Triton kernel that fuses the
   (B·H) independent mat‑muls (N×N)·(N×N)ᵀ.
6. Layer‑norm + output gate + final linear projection back to D.

The kernel works in fp16 (fast tensor cores) and accumulates in fp32.
The final output is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched N×N matmul for each hidden channel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
        a_ptr,                     # left  [B,N,N,H]  (i,k,d)
        b_ptr,                     # right_T [B,N,N,H] (k,j,d) = right.permute(0,2,1,3)
        c_ptr,                     # out   [B,N,N,H]  (i,j,d)
        B, N, H,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
        stride_b_batch, stride_b_k, stride_b_j, stride_b_d,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    # ------------------------------------------------------------------
    # 0) identify which (batch, hidden) slice we are computing
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)                     # combined (b, h)
    b = pid0 // H                               # batch index
    h = pid0 % H                                # hidden channel index

    pid_m = tl.program_id(1)                    # block row   (i)
    pid_n = tl.program_id(2)                    # block col   (j)

    # ------------------------------------------------------------------
    # 1) offsets within the block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2) accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) loop over k dimension (also blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to A[i,k] and B[k,j] for this (b,h) slice
        a_ptrs = (a_ptr
                  + b * stride_a_batch
                  + h * stride_a_d
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        b_ptrs = (b_ptr
                  + b * stride_b_batch
                  + h * stride_b_d
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_j)

        # load (fp16), apply masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # dot product (fp16 inputs → fp32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4) write the result (store fp32)
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr
              + b * stride_c_batch
              + h * stride_c_d
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a Triton kernel for the
    N³‑core (left/right multiplication).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32, cuda)
        - mask         : torch.Tensor [B, N, N]   (bool/float, cuda)
        - weights      : dict of torch.Tensor (all float32, cuda)
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32, cuda)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) layer‑norm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                           # [B,N,N,D]  fp32

    # ------------------------------------------------------------------
    # 2) linear projections (no bias) → left/right  (B,N,N,H)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # fp32
    right = F.linear(x, weights["right_proj.weight"])        # fp32

    # ------------------------------------------------------------------
    # 3) optional mask (mask may be omitted → all‑ones)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) gates (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # fp32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # fp32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) core TriMul via Triton
    # ------------------------------------------------------------------
    # right needs to be transposed on the middle axes: (k,j) instead of (j,k)
    right_t = right.permute(0, 2, 1, 3).contiguous()

    # Cast to fp16 for the kernel (tensor cores)
    left_h = left.to(torch.float16)
    right_h = right_t.to(torch.float16)

    out_h = torch.empty_like(left_h)           # allocate [B,N,N,H] fp16

    # strides (in elements, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d = left_h.stride()
    stride_b_batch, stride_b_k, stride_b_j, stride_b_d = right_h.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d = out_h.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                               # each (batch, hidden) pair
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, H,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
        stride_b_batch, stride_b_k, stride_b_j, stride_b_d,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to fp32 for the remaining layers
    out = out_h.to(torch.float32)                     # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                              # broadcast over H
    out = F.linear(out, weights["to_out.weight"])    # back to D

    return out
