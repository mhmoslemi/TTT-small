"""
TriMul (outgoing) – fast Triton implementation
================================================

The heavy part of the TriMul forward pass is the 3‑dimensional contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

for each batch `b` and hidden channel `d`.  For a fixed `d` this is
a matrix multiplication   L (N×N)  @  Rᵀ (N×N) → (N×N).

We:
1. apply the LayerNorms, linear projections and sigmoid gates in PyTorch
   (these are cheap compared with the N³ work);
2. reshape the tensors so that each *batch × hidden* pair becomes an
   independent matrix‑multiply problem;
3. launch a Triton kernel that computes many small (N×N) @ (N×N) products
   in parallel, using FP32 accumulation and FP16 storage;
4. restore the original layout and finish the remaining LayerNorm,
   gating and the final linear projection – all in PyTorch.

The kernel is written for the Hopper H200 (SM‑90) and uses block sizes
that give good occupancy for N up to 1024 while keeping register pressure
low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape = [batch*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # all equal to seq_len
    stride_am, stride_ak,        # A: (batch*H, M, K)
    stride_bm, stride_bk,        # Bᵀ: (batch*H, K, N)
    stride_cm, stride_cn,        # C: (batch*H, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Block‑wise FP32 accumulation, FP16 store."""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch*hidden index

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers with the batch offset baked in
    a_batch = a_ptr + pid_b * stride_am
    b_batch = b_ptr + pid_b * stride_bm
    c_batch = c_ptr + pid_b * stride_cm

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A block (M×K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # load Bᵀ block (K×N)  – note the transposition in the host code
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)          # FP32 dot

    # store FP16 result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def _triton_matmul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]   (float16)
    returns out    : [B, N, N, H] (float16)
    """
    B, N, _, H = left.shape
    # (B, H, N, N)  – each (b,h) pair is an independent matrix
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # we need rightᵀ on the last two dims
    right_t = right_perm.transpose(-2, -1).contiguous()

    # flatten batch*hidden
    a = left_perm.view(-1, N, N)
    b = right_t.view(-1, N, N)
    c = torch.empty_like(a)                     # FP16 output

    # strides (row‑major)
    stride_am, stride_ak, _ = a.stride()
    stride_bm, stride_bk, stride_bn = b.stride()
    stride_cm, stride_cn, _ = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        a.shape[0],                     # batch*hidden
    )
    _matmul_outgoing_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bm=stride_bm,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    N³ contraction.

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float) – may be all‑ones
            - weights: dict of model parameters (float32)
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – keep FP16 for the heavy path
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden]
    left = torch.einsum("...d,hd->...dh", x_norm, weights["left_proj.weight"]).to(torch.float16)
    right = torch.einsum("...d,hd->...dh", x_norm, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present) – broadcast on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – done in FP32 for numerical stability
    # ------------------------------------------------------------------
    left_gate = torch.einsum("...d,hd->...dh", x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.einsum("...d,hd->...dh", x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.einsum("...d,hd->...dh", x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Heavy N³ contraction via Triton
    # ------------------------------------------------------------------
    out = _triton_matmul_outgoing(left, right)           # FP16, shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden) – back to FP32
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # 7) Final gating and linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.einsum("...h,dh->...d", out, weights["to_out.weight"])
    return out
