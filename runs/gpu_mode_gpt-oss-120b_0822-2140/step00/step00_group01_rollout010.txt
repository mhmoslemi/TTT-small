"""
TriMul “outgoing” implementation.

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each (batch, d) this is a standard matrix multiplication
    A = left[b,:,:,d]   (N×N)
    B = right[b,:,:,d]  (N×N)
    out[b,:,:,d] = A @ Bᵀ
We fuse the batch and hidden dimension and execute a batched GEMM in
Triton.  All other linear / layer‑norm / gating work is done with
torch (the overhead is negligible compared with the N³ work).

The kernel works in fp16 for the two inputs, accumulates in fp32,
and writes fp16 results that are later cast back to fp32 for the final
linear layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,        # *fp16   [B,N,N,H]
    right_ptr,       # *fp16   [B,N,N,H]
    out_ptr,         # *fp16   [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh_id = tl.program_id(0)          # fused (batch, hidden)
    pid_m = tl.program_id(1)          # block row   (i)
    pid_n = tl.program_id(2)          # block col   (j)

    # de‑compose bh_id
    b = bh_id // H
    h = bh_id % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the summed axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile : (M, K)
        left_tile = tl.load(
            left_base
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile : we need (K, N) so we load transposed
        right_tile = tl.load(
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # GEMM on the tile
        acc += tl.dot(left_tile, right_tile)

    # write result
    out_ptrs = out_base \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing variant) with a Triton‑accelerated einsum.
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]      (bool/float)
        - weights_dict : contains all linear / LN weights & biases
        - config_dict  : {"dim":..., "hidden_dim":..., "nomask": bool}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] , float32

    # ------------------------------------------------------------------
    # 2. Linear projections + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        m = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 3. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Triton GEMM (the costly einsum)
    # ------------------------------------------------------------------
    # Convert to fp16 for the kernel (accumulates in fp32)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)

    # Strides (as ints) for the original layout [B,N,N,H]
    ls = left_fp16.stride()   # (N*N*H, N*H, H, 1)
    rs = right_fp16.stride()
    os = out_fp16.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                 # fused batch * hidden
        triton.cdiv(N, BLOCK_M),    # rows
        triton.cdiv(N, BLOCK_N),    # cols
    )

    trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden,
        ls[0], ls[1], ls[2], ls[3],
        rs[0], rs[1], rs[2], rs[3],
        os[0], os[1], os[2], os[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_fp16.to(torch.float32)   # back to fp32

    # ------------------------------------------------------------------
    # 5. Post‑processing: LayerNorm, out‑gate and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
