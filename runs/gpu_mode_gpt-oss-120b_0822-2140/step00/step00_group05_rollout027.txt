"""
TriMul “outgoing” implementation.

The expensive part of the module is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch element and hidden channel `d` it is a matrix
multiplication `L_d @ R_d.T` where `L_d,R_d` are (N×N) matrices.
The kernel below computes this batched GEMM with Triton, while the
surrounding LayerNorm / linear / gating operations are performed in
pure PyTorch.  The result is returned in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched matmul L @ R^T  (L,R shapes: [B*D, N, N])
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    l_ptr, r_ptr, out_ptr,          # pointers
    N,                              # sequence length
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)               # combined batch*hidden
    pid_m = tl.program_id(1)                   # row block
    pid_n = tl.program_id(2)                   # col block

    # -----------------------------------------------------------------
    # offsets
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds (necessary for N not divisible by block size)
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # pointers with the right stride ordering
    # left  : (batch, i, k)
    l_ptrs = l_ptr + pid_batch * stride_l_batch \
                  + offs_m[:, None] * stride_l_m \
                  + offs_k[None, :] * stride_l_k
    # right : we need R^T, i.e. (k, j)
    r_ptrs = r_ptr + pid_batch * stride_r_batch \
                  + offs_k[:, None] * stride_r_k \
                  + offs_n[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # main K loop
    # -----------------------------------------------------------------
    for _ in range(tl.cdiv(N, BLOCK_K)):
        a = tl.load(l_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(r_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # (M,K) • (K,N) → (M,N)

        # advance K dimension
        l_ptrs += BLOCK_K * stride_l_k
        r_ptrs += BLOCK_K * stride_r_k

    # -----------------------------------------------------------------
    # store
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + pid_batch * stride_o_batch \
                       + offs_m[:, None] * stride_o_i \
                       + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32, on CUDA)
        - mask         : [B, N, N]      (float32, 0/1)
        - weights      : dict of tensors (already on CUDA)
        - config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    out : torch.Tensor [B, N, N, dim] (float32)
    """
    # -------------------------------------------------
    # unpack
    # -------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------
    # LayerNorm + linear projections (PyTorch)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask (mask may be all‑ones when nomask==True)
    mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
    left  = left * mask_
    right = right * mask_

    # Gating values
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Triton batched GEMM: out = left @ right^T  (per hidden dim)
    # -------------------------------------------------
    # reshape to (B, hidden_dim, N, N) → (B*hidden_dim, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BHD = B * hidden_dim
    out_perm = torch.empty_like(left_perm)

    # strides in elements (not bytes)
    stride_l_batch, stride_l_m, stride_l_k, _ = left_perm.stride()
    stride_r_batch, stride_r_j, stride_r_k, _ = right_perm.stride()
    stride_o_batch, stride_o_i, stride_o_j, _ = out_perm.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        BHD,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to original layout [B,N,N,hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # Final LayerNorm, gating and projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
