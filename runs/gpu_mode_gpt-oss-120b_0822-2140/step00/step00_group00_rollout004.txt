"""
TriMul “outgoing” implementation optimized with:
*   Torch linear layers for the heavy matrix‑multiplications (the N³ einsum)
    – interpreted as a batch of hidden‑dim GEMMs and thus run by cuBLAS.
*   A tiny Triton kernel that fuses the three element‑wise products
    (proj * mask * gate) for the left and right tensors.
*   All other operations (LayerNorm, sigmoid, final linear) are done with
    Torch’s native CUDA kernels (fast and numerically stable).

The kernel works on flattened contiguous tensors, so it can be reused for
both left and right branches.  The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = a * b * c   (all tensors have the same shape)
# ----------------------------------------------------------------------
@triton.jit
def mul3_kernel(
    out_ptr, a_ptr, b_ptr, c_ptr,  # pointers
    N,                             # total number of elements
    BLOCK: tl.constexpr,           # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # (BLOCK)
    mask = offs < N

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)

    tl.store(out_ptr + offs, a * b * c, mask=mask)


# ----------------------------------------------------------------------
# Helper: run the three‑way elementwise product via Triton
# ----------------------------------------------------------------------
def triton_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """a, b, c must be contiguous and have identical shape."""
    assert a.shape == b.shape == c.shape
    total = a.numel()
    out = torch.empty_like(a)

    BLOCK = 65536  # 64 Ki elements per program; fits in shared memory
    grid = (triton.cdiv(total, BLOCK),)

    mul3_kernel[grid](
        out.data_ptr(),
        a.data_ptr(),
        b.data_ptr(),
        c.data_ptr(),
        total,
        BLOCK=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    # flatten to (B*N*N, dim) for a single GEMM
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                      # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3️⃣  Mask handling (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if nomask:
        mask_exp = torch.ones(B, N, N, 1, dtype=left.dtype, device=device)
    else:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden)
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden)
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden)

    # Cast to half for the elementwise kernel (speed / memory)
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    mask_exp = mask_exp.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣  Triton fused element‑wise multiplication
    # ------------------------------------------------------------------
    left = triton_mul3(left, mask_exp, left_gate)     # left = proj * mask * gate
    right = triton_mul3(right, mask_exp, right_gate)  # right = proj * mask * gate

    # ------------------------------------------------------------------
    # 6️⃣  Core N³ operation ⇒ batched GEMM over hidden dimension
    #     out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) so we can use torch.matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # GEMM for each hidden channel
    # (B, H, N, N) @ (B, H, N, N)^T  -> (B, H, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Restore original layout [B, N, N, hidden]
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣  to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8️⃣  Gate the output and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                       # elementwise gate
    out = out.to(torch.float32)                # back to FP32 for final linear

    out_flat = out.view(-1, hidden)            # (B*N*N, hidden)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
