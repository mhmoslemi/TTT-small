"""
TriMul “outgoing” forward pass with a fused batched matrix‑multiply kernel.

The heavy part of the module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is mathematically a batch of hidden‑dimension matrix
multiplications:  out[...,d] = left[...,d] @ right[...,d].T .
We flatten the batch and hidden dimensions and compute the
multiplications with a custom Triton kernel (fp16 inputs, fp32
accumulator).  All other operations (LayerNorm, linear projections,
gates, final linear) are performed with PyTorch/F.linear for
convenience.

The kernel works for any sequence length N (up to the limits of the
test suite) and any hidden dimension H (=config["hidden_dim"]).  The
final tensor is returned in float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply C = A @ B   (A: [B, M, K], B: [B, K, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: (batch, row, col)
    stride_bb, stride_bk, stride_bn,   # B: (batch, row, col)
    stride_cb, stride_cm, stride_cn,   # C: (batch, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)      # which (batch * hidden) matrix
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds tiles
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_ptrs = a_ptr + pid_batch * stride_ab \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bb \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # load tiles, zero‑pad out‑of‑range elements
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # inner‑product accumulation
        acc += tl.dot(a, b)

    # write back results
    c_ptrs = c_ptr + pid_batch * stride_cb \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: apply LayerNorm without bias handling extra logic
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]; weight/bias: (C,)
    return F.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a fused
    Triton batched‑matmul kernel.
    """
    # unpack inputs ------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Normalisation of the input
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) -> left / right
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication across the hidden dimension
    #    out[d] = left[d] @ right[d].T   for every hidden channel d
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) and put the hidden dim as leading axis
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # we need the transpose of right on the last two axes
    right_T = right_flat.transpose(1, 2).contiguous()

    # allocate output (fp16 for storage, fp32 accumulator inside kernel)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=left_flat.device)

    # launch parameters (tuned for Hopper; can be changed if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden_dim, 1),          # one program per (batch*hidden)
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[
        grid
    ](
        a_ptr=left_flat,
        b_ptr=right_T,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_ab=left_flat.stride(0),
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bb=right_T.stride(0),
        stride_bk=right_T.stride(1),
        stride_bn=right_T.stride(2),
        stride_cb=out_flat.stride(0),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout and cast to float32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and output linear
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise gate (same shape)

    out = F.linear(out, weights["to_out.weight"])

    return out
