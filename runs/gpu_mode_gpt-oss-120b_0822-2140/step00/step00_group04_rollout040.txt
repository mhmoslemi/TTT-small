"""
TriMul “outgoing” implementation.

The heavy N³ work is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch element *hidden channel* (d) this is a standard
matrix‑multiply: (N×N) @ (N×N)ᵀ.
We keep the LayerNorms, linear projections and gating in PyTorch (they
are cheap) and fuse the batched matrix multiplication with a single
Triton kernel.  The kernel works on FP16 for bandwidth‑limited work and
accumulates in FP32, then the result is cast back to FP32 for the final
LayerNorm / linear layer.

The kernel receives the left tensor in (i,k) order and the right tensor
in (j,k) order (already transposed), both flattened as (batch*H, N, N)
so that each program instance processes a tile (BLOCK_M × BLOCK_N)
of the output matrix for a given (batch*H) index.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    L_ptr, R_ptr, O_ptr,
    M, N, K,
    stride_l_bc, stride_l_i, stride_l_k,
    stride_r_bc, stride_r_j, stride_r_k,
    stride_o_bc, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: (M×K) @ (N×K)ᵀ → (M×N)

    Arguments
    ----------
    L_ptr, R_ptr, O_ptr : pointers to the flattened left, right and output tensors
    M, N, K               : problem size (all are N in TriMul)
    stride_*              : strides for the three‑dimensional tensors
    BLOCK_*               : tile sizes (compile‑time constants)
    """
    bc = tl.program_id(2)                     # combined batch*hidden index
    pid_m = tl.program_id(0)                  # block row
    pid_n = tl.program_id(1)                  # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers into the current tile
    a_ptrs = L_ptr + bc * stride_l_bc + offs_m[:, None] * stride_l_i + tl.arange(0, BLOCK_K)[None, :] * stride_l_k
    b_ptrs = R_ptr + bc * stride_r_bc + offs_n[None, :] * stride_r_j + tl.arange(0, BLOCK_K)[:, None] * stride_r_k

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M·BLOCK_K)·(BLOCK_K·BLOCK_N) → (BLOCK_M·BLOCK_N)

        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    c_ptrs = O_ptr + bc * stride_o_bc + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask        : torch.Tensor [B, N, N] (may be all‑ones)
        - weights     : dict of pre‑loaded model parameters
        - config      : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. LayerNorm and linear projections (pure PyTorch – cheap)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if required
    if not nomask:
        mask_exp = mask.unsqueeze(-1)                    # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ----------------------------------------------------------------------
    # 2. Batched N×N matrix multiplication per hidden channel (Triton)
    # ----------------------------------------------------------------------
    # cast to FP16 for bandwidth, accumulate in FP32 inside the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange tensors so that the reduction dimension (k) is the innermost
    # and we can flatten (B,H) into one leading dimension.
    # left_perm : (B, H, N, N)   – i (row), k (col)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # right_perm : (B, H, N, N) – j (col), k (col)  → we need (j,k) order
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()

    # Flatten the first two dimensions (batch * hidden)
    left_flat = left_perm.view(B * hidden_dim, N, N)     # (BC, i, k)
    right_flat = right_perm.view(B * hidden_dim, N, N)   # (BC, j, k)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # strides for the contiguous layout (BC, row, col)
    stride_l_bc, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_bc, stride_r_j, stride_r_k = right_flat.stride()
    stride_o_bc, stride_o_i, stride_o_j = out_flat.stride()

    BLOCK_M = 64      # tuneable; fits well in shared memory on Hopper
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim,                   # batch*hidden
    )

    _batched_matmul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N, N, N,
        stride_l_bc, stride_l_i, stride_l_k,
        stride_r_bc, stride_r_j, stride_r_k,
        stride_o_bc, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and convert to FP32 for the remaining ops
    out = (
        out_flat.view(B, hidden_dim, N, N)         # (B, H, N, N)
        .permute(0, 2, 3, 1)                       # (B, N, N, H)
        .to(torch.float32)
    )

    # ----------------------------------------------------------------------
    # 3. Final LayerNorm, gating and output linear projection (PyTorch)
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
