"""
TriMul (outgoing) – Triton‑accelerated core contraction.

The heavy part of the TriMul forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be rewritten as a batched matrix multiplication per hidden
dimension:

    left_d   : (B, N, N)   with axes (i,k)
    right_dᵀ : (B, N, N)   with axes (k,j)
    out_d    = left_d @ right_dᵀ          → (B, N, N)

We fuse the `hidden_dim` dimension into the batch dimension
(`B*hidden_dim` independent GEMMs) and implement a simple tiled
GEMM kernel in Triton (fp16 compute, fp32 accumulation).  All other
operations (LayerNorm, Linear projections, gating) are performed with
PyTorch – they are cheap compared to the O(N³) contraction.

The final output is cast back to float32 to match the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton GEMM kernel (fp16 compute, fp32 accumulator)
# --------------------------------------------------------------
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M, K)
    stride_bk, stride_bn,          # B: (K, N)
    stride_cm, stride_cn,          # C: (M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B  where
        A: (M, K)   stride_am, stride_ak
        B: (K, N)   stride_bk, stride_bn
        C: (M, N)   stride_cm, stride_cn
    """
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid %  num_pid_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the tiles
    a_ptrs = a_ptr + (offs_m[:, None] * stride_am
                     + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                     + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) &
                         (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < K) &
                         (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)          # fp16 * fp16 -> fp32 accumulate
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back (convert to fp16 for storage)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm
                      + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def _triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: for each batch element compute A @ B.
    A – (batch, M, K)   fp16
    B – (batch, K, N)   fp16
    Returns C – (batch, M, N) fp16
    """
    assert A.is_cuda and B.is_cuda
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # output buffer
    C = torch.empty((batch, M, N), device=A.device, dtype=torch.float16)

    # tiling parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N), batch)

    # launch one kernel per batch slice (grid dim 1 = batch)
    _matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(1), A.stride(2),
        B.stride(1), B.stride(2),
        C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Helper: LayerNorm without torch.nn (faster than calling the module)
# --------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]   weight/bias: (C,)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------
# Core TriMul contraction using Triton
# --------------------------------------------------------------
def _tri_mul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, H)   fp32
    Returns out: (B, N, N, H)   fp32
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)   # transpose (k,j)

    # fp16 for the kernel
    left_h = left_t.to(torch.float16)
    right_h = right_t.to(torch.float16)

    out_h = _triton_matmul(left_h, right_h)                       # (B*H, N, N)
    out = out_h.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (B, N, N, dim) -> (B, N, N, hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction – Triton
    # ------------------------------------------------------------------
    out = _tri_mul_contraction(left, right)          # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])    # back to dim
    return out
