    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    # layernorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # reshape for batched matmul
    left = left.permute(0,3,1,2).contiguous()   # (B,H,N,K)
    right = right.permute(0,3,1,2).contiguous()
    B, H, N, K = left.shape
    left_2d = left.view(B*H, N, K)
    right_2d = right.view(B*H, N, K)
    out_2d = torch.empty((B*H, N, N), dtype=torch.float32, device=device)
    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B*H)
    matmul_transpose_kernel[grid](
        left_2d, right_2d, out_2d,
        N, N, K,
        left_2d.stride(0), left_2d.stride(1), left_2d.stride(2),
        right_2d.stride(0), right_2d.stride(1), right_2d.stride(2),
        out_2d.stride(0), out_2d.stride(1), out_2d.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_2d.view(B, H, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

"""
TriMul “outgoing” implementation.

The heavy N³ part (einsum `'... i k d , ... j k d -> ... i j d'`) is
re‑written as a batched matrix multiplication C = A @ Bᵀ where

    A[b, h, i, k] = left  [b, i, k, h]
    B[b, h, j, k] = right [b, j, k, h]

For each (b, h) we compute a Gram matrix of size (N, N) over the
sequence dimension K.  The kernel is a classic tiled GEMM written in
Triton; all other operations (LayerNorm, linear projections and gates)
are performed with regular PyTorch (they are cheap compared with the
N³ work).  The kernel works for any batch size ≤ 2, any sequence length
up to 1024 and any hidden dimension (typically 128).  The final output
is returned in float32 as required.

The kernel launches a 3‑D grid (M‑blocks, N‑blocks, batch‑elements) and
uses fp16 loads with fp32 accumulation for speed on Hopper (SM‑90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape = (batch, M, K))
# --------------------------------------------------------------
@triton.jit
def _matmul_transpose_kernel(
    a_ptr,                # *f16   A   (batch, M, K)
    b_ptr,                # *f16   B   (batch, N, K)
    c_ptr,                # *f32   C   (batch, M, N)
    M, N, K,              # dimensions (int32)
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bn, stride_bk,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute one tile of C = A @ Bᵀ.
    A is (batch, M, K)   B is (batch, N, K)   C is (batch, M, N)
    The K‑dimension is reduced, so we load B transposed (K, N).
    """
    pid_z = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)          # block in M dimension
    pid_n = tl.program_id(1)          # block in N dimension

    # ---- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- mask out‑of‑bounds -------------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ---- pointers for the current tile --------------------------------
    a_ptrs = a_ptr + pid_z * stride_az + \
             offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # B is loaded transposed: (K, N)
    b_ptrs = b_ptr + pid_z * stride_bz + \
             offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ---- accumulator (fp32) -------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension -----------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)               # (M,K)·(K,N) -> (M,N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---- write back ----------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_cz + \
             offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Python entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, D] (float32)
            mask         : torch.Tensor [B, N, N]   (bool/float, may be all‑ones)
            weights      : dict of parameter tensors (float32)
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ----------------------------------------------------------
    # 1. LayerNorm on the input
    # ----------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------
    # 3. Optional mask (mask is broadcasted over the hidden dim)
    # ----------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask
        right = right * mask

    # ----------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Prepare tensors for the N³ GEMM
    #    left/right : (B, N, N, H) --> (B, H, N, K)
    # ----------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, K)

    B, H, N, K = left.shape
    # reshape to (batch*head, N, K) for a single GEMM batch dimension
    left  = left.view(B * H, N, K).to(torch.float16)
    right = right.view(B * H, N, K).to(torch.float16)

    # output buffer (float32)
    out = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ----------------------------------------------------------
    # 6. Launch Triton kernel
    # ----------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i dimension)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j dimension)
        B * H,                     # batch * heads
    )
    _matmul_transpose_kernel[grid](
        left,
        right,
        out,
        N, N, K,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------
    # 7. Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ----------------------------------------------------------
    # 8. Output LayerNorm, gating, final linear
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,D)

    return out
