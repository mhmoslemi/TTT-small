"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation of the
core N³ einsum.

The heavy part is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiply of left[..., :, :, d] with the
transpose of right[..., :, :, d].  We flatten (batch, hidden) into a
single “batch” dimension and launch a 3‑D Triton kernel that computes
C = A @ Bᵀ for each (b,d) pair.
All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
are kept in PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B, M, K], C : [B, M, N])
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
        a_ptr, b_ptr, c_ptr,        # pointers
        B, M, N, K,                 # int64 sizes
        stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
        stride_bb, stride_bn, stride_bk,   # strides for B (batch, N, K)
        stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr):
    # ---------------------------------------------------------------
    # program IDs
    pid_m = tl.program_id(0)      # M tile
    pid_n = tl.program_id(1)      # N tile
    pid_b = tl.program_id(2)      # batch (actually batch * hidden)

    # ---------------------------------------------------------------
    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---------------------------------------------------------------
    # mask for out‑of‑bounds (handles non‑divisible sizes)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # pointers for the first tile of A and B
    a_ptrs = a_ptr + pid_b * stride_ab \
                + offs_m[:, None] * stride_am \
                + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb \
                + offs_n[None, :] * stride_bn \
                + offs_k[:, None] * stride_bk   # note B is (batch,N,K) → we need K as first dim

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, applying masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K - k * BLOCK_K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K - k * BLOCK_K)[:, None] & mask_n[None, :],
                    other=0.0)
        # dot: (M,K) x (K,N) → (M,N)
        acc += tl.dot(a, b)

        # advance K‑dim pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---------------------------------------------------------------
    # write back
    c_ptrs = c_ptr + pid_b * stride_cb \
                + offs_m[:, None] * stride_cm \
                + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),      # store as fp16 for memory bandwidth
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    N³ update.

    Args:
        data: (input, mask, weights, config)
            input  – Tensor[B, N, N, dim] (float32)
            mask   – Tensor[B, N, N] (float32/bool) – may be all‑ones
            weights – dict with all linear / layer‑norm parameters
            config  – dict with keys:
                       dim (int), hidden_dim (int)

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    inp = F.layer_norm(inp,
                       normalized_shape=(dim,),
                       weight=norm_weight,
                       bias=norm_bias,
                       eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(inp, weights["left_proj.weight"])
    right = F.linear(inp, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(inp, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(inp, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(inp, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare for Triton: cast to FP16 and reshape to (B*hidden, N, N)
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # move hidden dim to second axis and flatten batch*hidden
    left  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # [B*H, N, N]
    right = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)  # [B*H, N, N]

    out = torch.empty_like(left)   # fp16 buffer for C

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # M‑tiles
        triton.cdiv(N, BLOCK_N),   # N‑tiles
        B * hidden                 # batch*hidden (program_id(2))
    )

    # strides in elements (not bytes) – Triton expects byte strides
    a_ptr = left.data_ptr()
    b_ptr = right.data_ptr()
    c_ptr = out.data_ptr()

    stride_ab, stride_am, stride_ak = left.stride()   # (batch, M, K)
    stride_bb, stride_bn, stride_bk = right.stride()
    stride_cb, stride_cm, stride_cn = out.stride()

    # launch
    trimul_gemm_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        B * hidden, N, N, N,
        stride_ab * left.element_size(),
        stride_am * left.element_size(),
        stride_ak * left.element_size(),
        stride_bb * right.element_size(),
        stride_bn * right.element_size(),
        stride_bk * right.element_size(),
        stride_cb * out.element_size(),
        stride_cm * out.element_size(),
        stride_cn * out.element_size(),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden] and cast to fp32
    # ------------------------------------------------------------------
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)    # now [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and Linear (to original dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # shape -> [B,N,N,dim]

    return out
