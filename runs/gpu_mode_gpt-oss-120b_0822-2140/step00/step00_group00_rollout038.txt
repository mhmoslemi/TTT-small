"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy O(N³·C) part is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of hidden‑dim (C) matrix multiplications:
    left[b,d] @ right[b,d].T   with shape (N,N)

The kernel below performs a fused batched GEMM for all
(batch * hidden_dim) pairs using Triton.  All other
operations (LayerNorm, linear projections, gating, final
normalisation) are kept in PyTorch for readability and
to avoid unnecessary engineering of tiny kernels.

The final output has shape [B, N, N, dim] and is returned in
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: batched matmul   C = A @ B.T
# A, B, C are stored as (batch, M, K) , (batch, N, K) , (batch, M, N)
# where batch = B * hidden_dim, M = N = seq_len, K = seq_len.
# ---------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,               # *float32   A
    b_ptr,               # *float32   B
    c_ptr,               # *float32   C
    M, N, K,             # int32      matrix sizes
    stride_ab, stride_ak, stride_a_batch,   # strides for A
    stride_bb, stride_bk, stride_b_batch,   # strides for B
    stride_cb, stride_ck, stride_c_batch,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute one block of C = A @ B.T for a given batch element."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    #--- coordinate offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #--- loop over K --------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        mask_k = cur_k + k_range < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ak
                               + (cur_k + k_range)[None, :] * stride_ab)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B block: (BLOCK_N, BLOCK_K)   note the transpose (B.T)
        b_ptrs = b_batch_ptr + ((cur_k + k_range)[:, None] * stride_bb
                               + offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot‑product over the K dimension
        acc += tl.dot(a, b)

    #--- write back ---------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ck
                            + offs_n[None, :] * stride_cb)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul block.

    Args
    ----
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        (input, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------
    # shape -> [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------
    # 4) Gating (sigmoid)
    # -----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------
    # 5) Core O(N³·C) operation – Triton batched GEMM
    # -----------------------------------------------------------
    B, N, _, _ = left.shape   # B, seq_len, seq_len, hidden_dim
    # reshape to (B*hidden_dim, N, N) for a single‑batch GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    out_flat = torch.empty_like(left_flat)

    # launch parameters (tuned for Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                       # M
        N,                       # N
        N,                       # K
        # strides for A
        left_flat.stride(2),     # stride_ab   (fastest dim = K)
        left_flat.stride(1),     # stride_ak   (M stride)
        left_flat.stride(0),     # stride_a_batch
        # strides for B
        right_flat.stride(2),    # stride_bb
        right_flat.stride(0),    # stride_bk   (N stride, after transpose)
        right_flat.stride(0),    # stride_b_batch  (same as batch stride)
        # strides for C
        out_flat.stride(2),      # stride_cb
        out_flat.stride(1),      # stride_ck
        out_flat.stride(0),      # stride_c_batch
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear
    # -----------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
