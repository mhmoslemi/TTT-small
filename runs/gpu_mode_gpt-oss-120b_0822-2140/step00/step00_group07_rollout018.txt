"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an N×N matrix multiplication for every batch and hidden channel.
The kernel below computes this batched mat‑mul with Triton, while the
surrounding LayerNorm / Linear / gating logic stays in PyTorch for simplicity.
All tensors are assumed to live on CUDA and the final result is returned as
float‑32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,            # const float*  [B, N, N, H]
    right_ptr,           # const float*  [B, N, N, H]
    out_ptr,             #       float*  [B, N, N, H]

    B, N, H,                         # meta‑parameters
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_om, stride_on, stride_oh,   # out strides

    BLOCK_M: tl.constexpr,            # tile size along i
    BLOCK_N: tl.constexpr,            # tile size along j
    BLOCK_K: tl.constexpr,            # tile size along k
):
    # program id layout:
    #   pid0 : (batch * hidden)   -> defines b and h
    #   pid1 : tile over i (M)
    #   pid2 : tile over j (N)
    pid0 = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Decode batch and hidden channel from pid0
    # --------------------------------------------------------------
    b = pid0 // H          # batch index
    h = pid0 % H           # hidden channel index

    # --------------------------------------------------------------
    # Offsets for the M/N tiles
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left block (M x K) ----
        # shape: (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ---- right block (K x N) ----
        # we load it transposed so that its shape is (K, N)
        # shape: (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rj
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # ---- dot product across K ----
        # left : (M, K)   right : (K, N)  -> (M, N)
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_m[:, None] * stride_om
        + offs_n[None, :] * stride_on
    )
    tl.store(out_ptrs, acc, mask=mask_out)


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      [B, N, N, dim]
        - mask         : torch.Tensor      [B, N, N]   (bool / float)
        - weights      : dict of torch.Tensor, keys:
            norm.weight, norm.bias,
            left_proj.weight, right_proj.weight,
            left_gate.weight, right_gate.weight, out_gate.weight,
            to_out_norm.weight, to_out_norm.bias,
            to_out.weight
        - config : dict with fields
            dim (int) – input channel size,
            hidden_dim (int) – projection size.
    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # → [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present (mask is broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and element‑wise application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core tensor contraction via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=x.device)

    # strides (as ints)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_om, stride_on, stride_oh = out.stride()

    # tile sizes – tuned for Hopper (sm90) SMEM 64KB, FP32 compute
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid shape: (B*H, ceil(N/BM), ceil(N/BN))
    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_lb=stride_lb, stride_li=stride_li, stride_lk=stride_lk, stride_lh=stride_lh,
        stride_rb=stride_rb, stride_rj=stride_rj, stride_rk=stride_rk, stride_rh=stride_rh,
        stride_ob=stride_ob, stride_om=stride_om, stride_on=stride_on, stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
