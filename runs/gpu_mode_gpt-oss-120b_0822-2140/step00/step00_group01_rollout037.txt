"""
TriMul “outgoing” implementation (AlphaFold3).

The expensive part of the forward pass is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden channel *d*.
We fuse all B×D matrix multiplications into a single Triton kernel.
All other operations (LayerNorm, Linear, gating, final projection) are
performed with regular PyTorch kernels.

The kernel computes for a given (block_i, block_j) tile of the output
matrix (size BLOCK_M×BLOCK_N) and a combined head index h = b*D + d:

    acc += dot( left[b,i:k, d] , right[b,j:k, d].T )   (k over BLOCK_K)

The final tensor is accumulated in float32 and stored back in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel for the ternary einsum
# ------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, D,                               # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 2‑D grid:
    #   program_id(0) -> tiles over (i,j)
    #   program_id(1) -> combined head index h = b*D + d
    pid_tile = tl.program_id(0)
    pid_head = tl.program_id(1)

    # ----- decode head -------------------------------------------------
    b = pid_head // D               # batch index
    d = pid_head % D                # hidden channel index

    # ----- decode tile coordinates --------------------------------------
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_i = pid_tile // num_blocks_n      # tile row (i)
    block_j = pid_tile % num_blocks_n       # tile col (j)

    # ----- offsets ------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        # k indices for this block
        k_valid = (k_base + offs_k) < N

        # ------- load left tile  (shape BLOCK_M × BLOCK_K) -------------
        # left[b, i, k, d]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + d * stride_left_d
            + offs_i[:, None] * stride_left_i
            + (k_base + offs_k)[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & k_valid[None, :],
            other=0.0,
        )

        # ------- load right tile (shape BLOCK_K × BLOCK_N) -------------
        # we need right transposed: right[b, j, k, d] → (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + d * stride_right_d
            + (k_base + offs_k)[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        right = tl.load(
            right_ptrs,
            mask=k_valid[:, None] & mask_j[None, :],
            other=0.0,
        )

        # ----- dot accumulation -----------------------------------------
        acc += tl.dot(left, right)

    # ----- store result -------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + d * stride_out_d
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Entry point called by the evaluator
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]    (float32)
            - mask         : torch.Tensor [B, N, N] (bool or 0/1) or None
            - weights      : dict of torch.Tensor parameters
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # --------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float32),
        bias=weights["norm.bias"].to(torch.float32),
    )

    # --------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(torch.float32),  # (hidden, dim)
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(torch.float32),
    )  # shape [B,N,N,hidden]

    # --------------------------------------------------------
    # 3. Mask (if present)
    # --------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------
    # 4. Gating + sigmoid
    # --------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(
            x,
            weights["left_gate.weight"].to(torch.float32),
        )
    )
    right_gate = torch.sigmoid(
        F.linear(
            x,
            weights["right_gate.weight"].to(torch.float32),
        )
    )
    out_gate = torch.sigmoid(
        F.linear(
            x,
            weights["out_gate.weight"].to(torch.float32),
        )
    )
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------
    # 5. Triton fused einsum (batched matmul per hidden channel)
    # --------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides for the three tensors (contiguous layout)
    stride_left = left.stride()   # (B_stride, i_stride, k_stride, d_stride)
    stride_right = right.stride()
    stride_out = out.stride()

    # Choose block sizes (tuned for Hopper, fp32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # grid: (tiles over i,j) × (B*hidden)
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid = (grid_i * grid_j, B * hidden)

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_d=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],   # note: right tensor dims (B,i,k,d) → we treat j as second N
        stride_right_k=stride_right[2],
        stride_right_d=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_d=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final Linear
    # --------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float32),
        bias=weights["to_out_norm.bias"].to(torch.float32),
    )
    out = out * out_gate

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float32),   # (dim, hidden)
    )
    return out
