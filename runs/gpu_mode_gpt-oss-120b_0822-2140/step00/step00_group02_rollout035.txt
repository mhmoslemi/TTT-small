#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
TriMul “outgoing” implementation with a Triton‑accelerated
batched matrix‑multiply.  All cheap ops (LayerNorm, Linear, gating,
final LayerNorm+Linear) are done in PyTorch; the cubic‑scale
einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched GEMM for each head `d` and launched with a
custom Triton kernel.  The kernel works on FP32 tiles (accumulated in
float32, loaded as float16 when possible) and supports any batch size
and sequence length that fit the launch grid.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B^T) where A,B are (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,                     # *ptr, shape (batch, M, K)
    B_ptr,                     # *ptr, shape (batch, N, K)   (note: K dimension is last)
    C_ptr,                     # *ptr, shape (batch, M, N)
    M, N, K,                   # problem size
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B.T  for each batch entry.
    A : (batch, M, K)
    B : (batch, N, K)   (we treat the second dim as N and third as K)
    C : (batch, M, N)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_tile_ptr = A_ptr + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + tl.arange(0, BLOCK_K)[None, :] * stride_a_k

    b_tile_ptr = B_ptr + batch_id * stride_b_batch \
                 + offs_n[None, :] * stride_b_n \
                 + tl.arange(0, BLOCK_K)[:, None] * stride_b_k

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main loop over K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Mask to avoid out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[None, :]
        b_mask = (offs_n < N)[None, :] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[:, None]

        a = tl.load(a_tile_ptr, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_tile_ptr, mask=b_mask, other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_tile_ptr += BLOCK_K * stride_a_k
        b_tile_ptr += BLOCK_K * stride_b_k

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_tile_ptr = C_ptr + batch_id * stride_c_batch \
                 + offs_m[:, None] * stride_c_m \
                 + offs_n[None, :] * stride_c_n

    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_tile_ptr, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Helper: Fold a 4‑D tensor (B,N,N,H) into (B*H,N,N)
# ----------------------------------------------------------------------
def _fold_head(tensor: torch.Tensor) -> torch.Tensor:
    """reshape (B,N,N,H) -> (B*H,N,N)  (contiguous)"""
    B, N, _, H = tensor.shape
    return tensor.permute(0, 3, 1, 2).reshape(B * H, N, N)


def _unfold_head(tensor: torch.Tensor, B: int, N: int, H: int) -> torch.Tensor:
    """reshape (B*H,N,N) -> (B,N,N,H)"""
    return tensor.reshape(B, H, N, N).permute(0, 2, 3, 1)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            input : torch.Tensor [B, N, N, dim]      (float16/bfloat16/float32)
            mask  : torch.Tensor [B, N, N] (or None)
            weights: dict of all linear / LN parameters (float32)
            config : dict with keys “dim”, “hidden_dim”

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    eps = 1e-5
    w_ln = weights["norm.weight"]
    b_ln = weights["norm.bias"]
    # compute mean / var across dim
    mean = input_tensor.mean(-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x_norm * w_ln + b_ln

    # --------------------------------------------------------------
    # Linear projections (no bias) – use torch.nn.functional.linear
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Apply mask if present (mask shape B,N,N)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # Gating (sigmoid of linear)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    # --------------------------------------------------------------
    # left : (B,N,N,hidden) -> (B*hidden, N, N)
    # right: we need the K‑dimension on the *second* axis for the kernel,
    # i.e. we will feed right as (B*hidden, N, N) and let the kernel
    # compute A @ B.T.
    left_fold = _fold_head(left)
    right_fold = _fold_head(right)

    B, N, _, H = left.shape
    total_batches = left_fold.shape[0]          # B*H
    M = N
    K = N
    # Output buffer
    out_fold = torch.empty_like(left_fold)

    # --------------------------------------------------------------
    # Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,                     # batch * head
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides for contiguous tensors (batch, row, col)
    stride_a_batch, stride_a_m, stride_a_k = left_fold.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_fold.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_fold.stride()

    _batched_gemm_kernel[grid](
        left_fold,
        right_fold,
        out_fold,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to (B,N,N,hidden)
    # --------------------------------------------------------------
    out = _unfold_head(out_fold, B, N, H)          # (B,N,N,hidden)

    # --------------------------------------------------------------
    # Final LayerNorm + gating + projection
    # --------------------------------------------------------------
    # to_out_norm over hidden_dim
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    eps_out = 1e-5
    mean_o = out.mean(-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps_out)
    out_norm = out_norm * w_outnorm + b_outnorm

    out_norm = out_norm * out_gate

    # final projection back to dim (no bias)
    out_final = F.linear(out_norm, weights["to_out.weight"])

    # Ensure float32 output
    return out_final.to(torch.float32)
