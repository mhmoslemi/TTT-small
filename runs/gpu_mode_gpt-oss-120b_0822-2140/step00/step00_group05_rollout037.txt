"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part of the TriMul module is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is exactly a batch of H independent N×N matrix multiplications
(L_d @ R_dᵀ).  The kernel below performs this batched mat‑mul on GPU.
All other lightweight operations (layer‑norm, linear projections, gating)
are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ
#   A : (BATCH, M, K)
#   B : (BATCH, N, K)   (we treat B as transposed inside the kernel)
#   C : (BATCH, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_transpose_kernel(
    A_ptr,               # *ptr to A
    B_ptr,               # *ptr to B (will be accessed as Bᵀ)
    C_ptr,               # *ptr to output
    BATCH, M, N, K,      # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices."""
    pid_batch = tl.program_id(2)               # batch index
    pid_m = tl.program_id(0)                    # block row
    pid_n = tl.program_id(1)                    # block col

    # --------------------------------------------------------------
    # offsets for the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch
    a_batch_ptr = A_ptr + pid_batch * stride_a_batch
    b_batch_ptr = B_ptr + pid_batch * stride_b_batch
    c_batch_ptr = C_ptr + pid_batch * stride_c_batch

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main K‑loop
    # --------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = tl.minimum(K - k, BLOCK_K)

        # A tile: (M, K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m
                                + (k + offs_k)[None, :].clip(0, K - 1) * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < cur_k),
                    other=0.0)

        # B tile (treated as transposed): we want Bᵀ → (K, N)
        b_ptrs = b_batch_ptr + (k + offs_k)[:, None] * stride_b_k \
                                + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < cur_k) & mask_n[None, :],
                    other=0.0)

        # dot‑product over K
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # store the result
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m \
                               + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float32),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the kernel
# ----------------------------------------------------------------------
def batched_matmul_transpose(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B : (batch, N, N)   (float32)
    returns C = A @ Bᵀ  with shape (batch, N, N)
    """
    assert A.is_contiguous() and B.is_contiguous()
    BATCH, M, K = A.shape
    _, N, _ = B.shape
    assert K == N

    C = torch.empty_like(A)

    # block sizes – chosen for Hopper (SM90)  (tuned for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BATCH,
    )

    batched_matmul_transpose_kernel[grid](
        A,
        B,
        C,
        BATCH,
        M,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),   # note: B stride order (batch, N, K)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input, mask, weights, config)
            input   – Tensor[B, N, N, dim]   (float32)
            mask    – Tensor[B, N, N]        (float32 or bool)
            weights – dict of all learnable parameters
            config  – {"dim": int, "hidden_dim": int}
    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.nn.functional.layer_norm works on the last dimension.
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden_dim, dim)

    # Linear: out = x @ Wᵀ
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Apply mask (if present)
    # --------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1).to(left.dtype)          # (B,N,N,1)
    left = left * mask_unsq
    right = right * mask_unsq

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core einsum -> batched matmul per hidden dimension
    #    out_d = left_d @ right_dᵀ   for each d in hidden_dim
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    A = left_perm.view(B * H, N, N)       # (B*H, N, K)   K == N
    B_ = right_perm.view(B * H, N, N)     # (B*H, N, K)

    # Triton kernel computes C = A @ Bᵀ
    C = batched_matmul_transpose(A, B_)

    # reshape back to (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate                       # gating after norm

    to_out_weight = weights["to_out.weight"]    # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)         # final projection, no bias

    return out.to(torch.float32)
