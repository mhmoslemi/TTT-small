"""
TriMul (outgoing) – Triton‑accelerated core

The expensive part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every hidden channel d.
We implement it with a tiled GEMM kernel (A @ Bᵀ) in Triton.  All
LayerNorms, Linear projections and gating are kept in PyTorch because
they are already highly optimized; only the O(N³·D) contraction is
off‑loaded to the GPU.

The kernel works for any batch size B (≤2), sequence length N (≤1024) and
hidden dimension D (≤384).  It aggregates in float32 and stores the
result as float16 (the remainder of the pipeline works in float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B : [N,N], one channel)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, N, D,                           # sizes (batch, seq_len, hidden)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_k, stride_b_j, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for a fixed (batch, hidden) pair,
        C[i, j] = Σ_k A[i, k] * B[j, k]      (i,j,k ∈ [0, N))
    i.e. C = A @ Bᵀ.
    """
    pid = tl.program_id(0)          # tile index over (i,j)
    bh  = tl.program_id(1)          # combined batch*hidden index

    # ------------------------------------------------------------------
    # decode batch and hidden indices
    # ------------------------------------------------------------------
    b = bh // D
    d = bh % D

    # ------------------------------------------------------------------
    # tile coordinates
    # ------------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    pid_m = pid // num_tiles_m          # i‑tile
    pid_n = pid %  num_tiles_m          # j‑tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # pointers for the current (b,d) slice
    # ------------------------------------------------------------------
    a_base = a_ptr + b * stride_a_batch + d * stride_a_d
    b_base = b_ptr + b * stride_b_batch + d * stride_b_d
    c_base = c_ptr + b * stride_c_batch + d * stride_c_d

    # A: [i,k]  (row = i, col = k)
    a_ptrs = a_base + offs_m[:, None] * stride_a_i + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    # Bᵀ: we need B[j,k] → treat as Bᵀ: rows = j, cols = k
    # original B layout: (batch, i, k, d)  → stride over i = stride_b_j, over k = stride_b_k
    b_ptrs = b_base + offs_n[None, :] * stride_b_j + tl.arange(0, BLOCK_K)[:, None] * stride_b_k

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        acc += tl.dot(a, b)                         # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # write result (store as float16 for memory bandwidth)
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, D] (float32)
    returns out  : [B, N, N, D] (float32)
    """
    B, N, _, D = left.shape

    # output allocation (float16, later cast)
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # strides (in elements)
    sa = left.stride()   # (batch, i, k, d)
    sb = right.stride()  # (batch, i, k, d) – we will use i as j when transposed
    sc = out.stride()

    # launch configuration (tuned for Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n, B * D)

    _trimul_gemm_kernel[grid](
        left, right, out,
        B, N, D,
        sa[0], sa[1], sa[2], sa[3],
        sb[0], sb[2], sb[1], sb[3],      # notice swapped i/k for Bᵀ
        sc[0], sc[1], sc[2], sc[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.to(torch.float32)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   torch.float32 on CUDA
            - mask         : [B, N, N]        torch.float32 (0/1) on CUDA
            - weights      : dict of torch tensors (already on CUDA)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        out : torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask handling (mask may be all‑ones when nomask==True)
    # ------------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1)               # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid, no bias)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    out = _trimul_gemm(left, right)   # [B, N, N, hidden_dim] float32

    # ------------------------------------------------------------------
    # 6) Final LayerNorm (hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Output linear projection back to dim (no bias)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
