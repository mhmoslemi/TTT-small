"""
TriMul “outgoing” implementation (forward only).

The expensive part of the module is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We keep the cheap LayerNorm / Linear / gating steps in PyTorch and
accelerate the 𝑁³ reduction with a Triton kernel.  The kernel treats each
pair (batch, hidden‑dim) as an independent GEMM:

    out_hd = left_hd @ right_hdᵀ          # (N,N) = (N,N) × (N,N)ᵀ

The kernel loops over the reduction dimension `k` in tiles and uses
`tl.dot` for the inner product.  It works for any batch size (1‑2) and
any sequence length up to 1024.  All tensors are assumed to be on CUDA.
The final output is returned as `torch.float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # *f32  [B, N, N, H]
    right_ptr,         # *f32  [B, N, N, H]
    out_ptr,           # *f32  [B, N, N, H]
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_m = tl.program_id(0)                # tile over i
    pid_n = tl.program_id(1)                # tile over j
    pid_hb = tl.program_id(2)               # fused (batch, hidden)

    b = pid_hb // H                         # batch index
    h = pid_hb % H                          # hidden‑dim index

    # tile coordinates
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # thread‑local row / column indices
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # predicates for out‑of‑bounds rows / cols
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in BLOCK_K steps
    for k_base in range(0, N, BLOCK_K):
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------------------------------------------------------------------
        # left[b, i, k, h]   shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # right[b, j, k, h]ᵀ   we need (k, j) layout for dot
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + k[:, None] * stride_r_k   # k as first dimension
            + j[None, :] * stride_r_j   # j as second dimension
            + h * stride_r_h
        )
        right_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate dot product over k
        acc += tl.dot(left, right_T)          # (M,N) += (M,K)·(K,N)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    ternary einsum.  All weights are provided explicitly in `weights`.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, C = input_tensor.shape                 # C == dim
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    assert C == dim

    # ------------------------------------------------------------------
    # Helper to apply linear layers without bias
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: [..., dim]   w: [out_dim, dim]
        return F.linear(x, w)

    # ------------------------------------------------------------------
    # LayerNorms – we implement them with functional calls
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # Projections
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask is bool or 0/1 float)
    # ------------------------------------------------------------------
    if mask_tensor is not None and mask_tensor.dtype != torch.bool:
        mask_tensor = mask_tensor.to(dtype)
    if mask_tensor is not None:
        mask_unsq = mask_tensor.unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: left @ rightᵀ over the k‑dimension, per (batch, head)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)                     # [B,N,N,hidden]

    # strides (in elements) – PyTorch stores tensors in contiguous layout
    stride_l = left.stride()   # (b,i,k,h)
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l[0],
        stride_l_i=stride_l[1],
        stride_l_k=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_b=stride_r[0],
        stride_r_j=stride_r[2],          # note: we need stride over “j” (the second N)
        stride_r_k=stride_r[1],
        stride_r_h=stride_r[3],
        stride_o_b=stride_o[0],
        stride_o_i=stride_o[1],
        stride_o_j=stride_o[2],
        stride_o_h=stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # apply out_gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])   # weight shape [dim, hidden]

    return out.to(dtype)
