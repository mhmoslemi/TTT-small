"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy part is the 4‑D contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched “row‑wise” matrix multiplication over the
summation dimension *k*.  The kernel tiles the i‑ and j‑ dimensions
with BLOCK_M×BLOCK_N tiles and walks over k in BLOCK_K steps,
using fp16 for the loads and fp32 accumulation.  The rest of the
operations (LayerNorm, Linear, gating, final projection) are
performed with regular PyTorch kernels – they are cheap compared
to the O(N³·C) contraction.

The kernel is launched with a 3‑D grid:
    dim0 : batch × hidden_dim   (together identify a sub‑tensor)
    dim1 : ceil(N/BLOCK_M)       (tile of i)
    dim2 : ceil(N/BLOCK_N)       (tile of j)
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel that computes the contraction over the k‑dimension.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # *float16  [B, N, N, H]
    right_ptr,         # *float16  [B, N, N, H]
    out_ptr,           # *float16  [B, N, N, H]

    B, N, H,                           # meta‑data
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    bh_id   = tl.program_id(0)                     # batch * hidden
    i_tile  = tl.program_id(1)                     # i‑tile
    j_tile  = tl.program_id(2)                     # j‑tile

    # decode batch and hidden dim
    b = bh_id // H
    h = bh_id % H

    # offsets inside the current tile
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑range rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialise accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left[b,i,k,h] ----------
        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # -------- load right[b,j,k,h] ----------
        # we need it transposed to (BLOCK_K, BLOCK_N) for tl.dot
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        # shape (BLOCK_N, BLOCK_K) → transpose
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)
        right_T = tl.trans(right)               # (BLOCK_K, BLOCK_N)

        # accumulate
        acc += tl.dot(left, right_T)

    # -------- store result ----------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a fused Triton kernel
    for the N³·C contraction.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N]      (float32 or bool)
            - weights: dict of pre‑loaded model parameters (float32)
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]          # = hidden_dim in the paper

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply optional mask (mask is broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                  # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # allocate output buffer (fp16)
    out_f16 = torch.empty_like(left_f16, device=device)

    # ------------------------------------------------------------------
    # 6) Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64

    # strides for the contiguous tensors (B,N,N,H)
    s_left  = left_f16.stride()   # (N*N*H, N*H, H, 1)
    s_right = right_f16.stride()
    s_out   = out_f16.stride()

    grid = (
        B * hidden,                                 # dim0 : batch × hidden
        triton.cdiv(N, BLOCK_M),                    # dim1 : i‑tiles
        triton.cdiv(N, BLOCK_N),                    # dim2 : j‑tiles
    )

    _trimul_kernel[grid](
        left_f16, right_f16, out_f16,
        B, N, hidden,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # back to float32 for the remaining ops
    out = out_f16.to(torch.float32)                     # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
