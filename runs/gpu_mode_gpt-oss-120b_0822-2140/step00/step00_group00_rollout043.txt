"""
TriMul “outgoing” implementation (forward only).

The heavy N³ operation 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched matrix multiplication per hidden channel:
    left  : (B, N, N, H) → (B*H, N, N)
    right : (B, N, N, H) → (B*H, N, N)ᵀ
    out   : (B*H, N, N) = left @ rightᵀ

The matrix multiplication is performed by a custom Triton kernel (fp16
accumulation, fp32 final result).  All other operations – LayerNorm,
Linear projections, gating and the final linear layer – are done with
plain PyTorch (they are negligible compared to the N³ step).

The kernel works for any batch size (B∈{1,2}) and sequence length
(N∈{128…1024}), hidden dimension H (often 128) and model dimension D
(128 or 384).  Masks are optional; when `nomask` is True the mask is
treated as all‑ones.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ---------------------------------------------------------
# Triton kernel: batched matrix multiplication (fp16)
# ---------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dims
    stride_am, stride_ak, stride_an,    # A strides (batch, row, col)
    stride_bk, stride_bn, stride_bb,    # B strides (batch, row(K), col)
    stride_cm, stride_cn, stride_cc,    # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.
    A, B, C are stored as (batch, M, K) , (batch, N, K) , (batch, M, N)
    but B is interpreted transposed (so we load it as (K, N)).
    """
    pid = tl.program_id(0)                      # linear program id
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_batch = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    pid_m = pid_in_batch // tl.cdiv(N, BLOCK_N)  # block row within batch
    pid_n = pid_in_batch %  tl.cdiv(N, BLOCK_N)  # block col within batch

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the current block (include batch offset)
    a_ptrs = A_ptr + batch * stride_an \
             + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = B_ptr + batch * stride_bb \
             + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)              # dot → fp32 accumulation
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = C_ptr + batch * stride_cc \
             + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# ---------------------------------------------------------
# Entry point used by the evaluator
# ---------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel
    for the O(N³) contraction.

    Args
    ----
    data : (input, mask, weights, config)
        input  – Tensor[B, N, N, D] (float32)
        mask   – Tensor[B, N, N]    (float32 or bool, may be all‑ones)
        weights – dict with all linear / layer‑norm parameters
        config  – dict with 'dim', 'hidden_dim', optional 'eps'

    Returns
    -------
    Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -------------------------------------------------
    # 1) LayerNorm on the input (dim = D)
    # -------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    # manual layer‑norm: mean & var over last dim
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b                   # [B,N,N,D]

    # -------------------------------------------------
    # 2) Linear projections (no bias)
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (H, D)
    right_proj_w = weights["right_proj.weight"].to(device) # (H, D)

    left = F.linear(x_norm, left_proj_w)   # [B,N,N,H]
    right = F.linear(x_norm, right_proj_w) # [B,N,N,H]

    # -------------------------------------------------
    # 3) Apply mask (if any)
    # -------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)           # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4) Gates (sigmoid)
    # -------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5) Triton batched matmul: out = Σ_k left[...,k,d] * right[...,k,d]
    # -------------------------------------------------
    # reshape to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)            # (batch’, M, K)
    # right needs to be transposed on the last two dims before multiplication
    right_flat_T = right_perm.view(B * H, N, N).transpose(-2, -1).contiguous()

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # fp16 result

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_batches = B * H
    total_grid = total_batches * grid_m * grid_n

    # strides (batch, row, col) for each tensor
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat_T.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    batched_matmul_kernel[total_grid](
        left_flat,
        right_flat_T,
        out_flat,
        N, N, N,
        stride_a_row, stride_a_col, stride_a_batch,
        stride_b_row, stride_b_col, stride_b_batch,
        stride_c_row, stride_c_col, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 and reshape to original layout
    out = out_flat.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # -------------------------------------------------
    # 6) Final LayerNorm + gating + linear projection back to D
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    # LayerNorm over hidden dimension H
    mean_out = out.mean(-1, keepdim=True)
    var_out = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply output gate (broadcast over H)
    out_norm = out_norm * out_gate

    # Final linear projection (H -> D)
    to_out_w = weights["to_out.weight"].to(device)     # (D, H)
    out_final = F.linear(out_norm, to_out_w)           # [B,N,N,D]

    return out_final
