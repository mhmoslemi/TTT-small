    B,N,_,dim = input_tensor.shape

    norm_bias = weights["norm.bias"]

    right = right * right_gate

    right_h = right.to(torch.float16)

    BLOCK_N = 64
    BLOCK_K = 32

        b = right_h[..., d].contiguous()

"""
TriMul (outgoing) – fast Triton implementation
================================================
The heavy N³ part of the TriMul forward pass is the einsum

    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]

For each hidden dimension *d* this is exactly a batched matrix‑multiply:

    out_d = left_d @ right_d.T        (B, N, N) × (B, N, N)ᵀ → (B, N, N)

We keep the light‑weight operations (LayerNorm, linear projections,
gate‑sigmoids) in PyTorch and only replace the inner product with a
custom Triton kernel that performs a **batched GEMM** in FP16 and
accumulates in FP32.  The kernel works for any batch size (B≤2), any
sequence length N (up to 1024) and any hidden dimension (usually ≤ 384).

The kernel is launched once per hidden channel; the loop over ``d`` lives
in Python (the hidden dimension is tiny compared with N, so the launch
overhead is negligible).  All tensors are assumed to already reside on
the CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (B, M, K)    B : (B, N, K)   (note: we treat B as already transposed)
#   C : (B, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,              # pointers
    B, M, N, K,                       # sizes
    stride_abl, stride_abm, stride_abk,   # A strides
    stride_bbk, stride_bbk2, stride_bbn,  # B (transposed) strides
    stride_cbl, stride_cbm, stride_cbn,  # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch index
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # ------------------------------------------------------------------
    # Block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers for the current block
    a_ptrs = a_ptr + pid_b * stride_abl + (offs_m[:, None] * stride_abm) + (offs_k[None, :] * stride_abk)
    b_ptrs = b_ptr + pid_b * stride_bbk + (offs_k[:, None] * stride_bbk2) + (offs_n[None, :] * stride_bbn)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # dot: FP16 inputs → FP32 accumulation
        acc += tl.dot(a, b)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_abk
        b_ptrs += BLOCK_K * stride_bbk2

    # ------------------------------------------------------------------
    # Write results
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_cbl + (offs_m[:, None] * stride_cbm) + (offs_n[None, :] * stride_cbn)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    batched matrix multiplication.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor:  Tensor of shape [B, N, N, dim]   (float32)
            - mask:          Tensor of shape [B, N, N]       (bool / float)
            - weights:       Dict of model parameters (all float32)
            - config:        Dict with keys ``dim``, ``hidden_dim`` and
                              optional ``nomask`` (bool).

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton‑accelerated batched GEMM (per hidden channel)
    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy kernel, keep FP32 accumulator.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Output tensor (FP32)
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # Tune block sizes – good defaults for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Pre‑compute grid dimensions (batch, row blocks, col blocks)
    grid = (B, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    for d in range(hidden_dim):
        # Slice the d‑th hidden channel (contiguous view)
        a = left_h[..., d].contiguous()
        b = right_h[..., d].contiguous()
        c = out[..., d]

        # Pointers
        a_ptr = a.data_ptr()
        b_ptr = b.data_ptr()
        c_ptr = c.data_ptr()

        # Strides (in element units)
        stride_abl = a.stride(0)          # batch stride
        stride_abm = a.stride(1)          # M (i) stride
        stride_abk = a.stride(2)          # K (k) stride

        # For B we need it transposed: we access (k, n) which corresponds
        # to original (n, k) in the tensor.
        stride_bbk  = b.stride(0)         # batch stride
        stride_bbk2 = b.stride(2)         # k stride (after transpose)
        stride_bbn  = b.stride(1)         # n stride (after transpose)

        stride_cbl = c.stride(0)          # batch stride
        stride_cbm = c.stride(1)          # M stride
        stride_cbn = c.stride(2)          # N stride

        # Launch kernel
        batched_matmul_kernel[grid](
            a_ptr,
            b_ptr,
            c_ptr,
            B,
            N,
            N,
            N,
            stride_abl,
            stride_abm,
            stride_abk,
            stride_bbk,
            stride_bbk2,
            stride_bbn,
            stride_cbl,
            stride_cbm,
            stride_cbn,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
