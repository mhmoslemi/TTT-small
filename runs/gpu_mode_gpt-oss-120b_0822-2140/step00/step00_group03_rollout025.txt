"""
TriMul “outgoing” kernel (AlphaFold3) – fast forward pass.

The heavy work inside TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication per hidden channel:
    out_d = left_d @ right_dᵀ   where   left_d,right_d ∈ ℝ^{B·N×N}.

We keep the linear projections, LayerNorms and the final linear layer in PyTorch
(cuBLAS / torch.nn.functional are already heavily optimized).  The only part
implemented in Triton is the element‑wise masking + gating that follows the
two projections – this fuses three element‑wise operations per entry and
removes an extra memory pass.

The algorithm
--------------
1.  Layer‑norm the input (last dimension = `dim`).
2.  Linear projections → `left`, `right`   (shape B×N×N×H).
3.  Linear projections → gates and sigmoid them.
4.  Triton kernel:
        left = left * left_gate * mask
        right = right * right_gate * mask
   (`mask` is broadcasted to the hidden dimension, omitted if `nomask`).
5.  Reshape to (B·H, N, N) and perform a batched GEMM:
        out = left @ rightᵀ      → shape (B·H, N, N)
   Reshape back to B×N×N×H.
6.  Layer‑norm on the hidden dimension, multiply by `out_gate`,
   final linear projection back to `dim`.
All tensors are kept on the GPU and the final result is `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel : combine mask, left/right gating (per hidden channel)
# --------------------------------------------------------------------------- #
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, right_ptr,          # B,N,N,H
    left_gate_ptr, right_gate_ptr,  # B,N,N,H (already after sigmoid)
    mask_ptr,                     # B,N,N   (may be NULL)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)

    # linear index over (b,i,j)
    total = B * N * N
    b = pid // (N * N)
    idx = pid % (N * N)
    i = idx // N
    j = idx % N

    # mask element (scalar) – if mask_ptr==0 we treat it as 1.0
    mask = 1.0
    if mask_ptr != 0:
        mask = tl.load(mask_ptr + b * stride_m_b + i * stride_m_i + j * stride_m_j)

    # loop over hidden dimension in tiles
    d = 0
    while d < H:
        offs_h = tl.arange(0, BLOCK_H) + d
        hmask = offs_h < H

        # pointers for this tile
        l_ptr = (left_ptr + b * stride_l_b + i * stride_l_i + j * stride_l_j
                 + offs_h * stride_l_h)
        r_ptr = (right_ptr + b * stride_r_b + i * stride_r_i + j * stride_r_j
                 + offs_h * stride_r_h)

        lg_ptr = (left_gate_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
                  + offs_h * stride_lg_h)
        rg_ptr = (right_gate_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
                  + offs_h * stride_rg_h)

        # load
        left = tl.load(l_ptr, mask=hmask, other=0.0)
        right = tl.load(r_ptr, mask=hmask, other=0.0)
        left_gate = tl.load(lg_ptr, mask=hmask, other=0.0)
        right_gate = tl.load(rg_ptr, mask=hmask, other=0.0)

        # apply mask & gates
        left = left * left_gate * mask
        right = right * right_gate * mask

        # store back
        tl.store(l_ptr, left, mask=hmask)
        tl.store(r_ptr, right, mask=hmask)

        d += BLOCK_H

# Entry point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a small Triton kernel
    for the mask‑gate fusion.  All other linear / norm operations are done
    with PyTorch (cuBLAS / torch.nn.functional) which is already optimal.
    """
    # ------------------------------------------------------------------- #
    # unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]            # input channel dimension
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # layer‑norm on the input (last dimension = dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # linear projections (no bias) → left/right (B,N,N,H)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # compute gates (sigmoid after linear) – still B,N,N,H
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------- #
    # optional mask – shape (B,N,N) → broadcasted later
    # ------------------------------------------------------------------- #
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
    else:
        mask_f = None

    # ------------------------------------------------------------------- #
    # Triton kernel: fuse mask + left/right gates
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    BLOCK_H = 32                     # 32 works for H∈{128,384,768}
    total_elements = B * N * N
    grid = (total_elements,)

    # Strides
    l_strides = left.stride()              # (B,N,N,H) → tuple of 4 ints
    r_strides = right.stride()
    lg_strides = left_gate.stride()
    rg_strides = right_gate.stride()
    if mask_f is None:
        mask_ptr = 0                       # null pointer for the kernel
    else:
        mask_ptr = mask_f.contiguous().data_ptr()

    fuse_mask_gate_kernel[
        grid
    ](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask_ptr,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=l_strides[0],
        stride_l_i=l_strides[1],
        stride_l_j=l_strides[2],
        stride_l_h=l_strides[3],
        stride_r_b=r_strides[0],
        stride_r_i=r_strides[1],
        stride_r_j=r_strides[2],
        stride_r_h=r_strides[3],
        stride_lg_b=lg_strides[0],
        stride_lg_i=lg_strides[1],
        stride_lg_j=lg_strides[2],
        stride_lg_h=lg_strides[3],
        stride_rg_b=rg_strides[0],
        stride_rg_i=rg_strides[1],
        stride_rg_j=rg_strides[2],
        stride_rg_h=rg_strides[3],
        stride_m_b=mask_f.stride(0) if mask_f is not None else 0,
        stride_m_i=mask_f.stride(1) if mask_f is not None else 0,
        stride_m_j=mask_f.stride(2) if mask_f is not None else 0,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------- #
    # Batched GEMM per hidden channel: left @ rightᵀ
    # ------------------------------------------------------------------- #
    # reshape to (B*H, N, N)
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # matmul (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # ------------------------------------------------------------------- #
    # hidden‑dim LayerNorm, multiply by out_gate, final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(dtype)
