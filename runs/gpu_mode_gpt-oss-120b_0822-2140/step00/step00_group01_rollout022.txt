"""
TriMul “outgoing” forward pass (AlphaFold3) with a custom Triton GEMM.

The heavy part of the module is the einsum
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
which is exactly a batched matrix‑multiply:
    out_d = left_d @ right_d.T   (for each hidden channel d)

We implement this product with a fused Triton kernel that works on the
flattened batch‑×‑hidden dimension (B·H).  All other operations (LayerNorm,
linear projections, gating, mask handling and final linear) are performed
with PyTorch which is already highly optimized.  The kernel works in fp16
and accumulates in fp32 to keep numerical quality; the final result is
converted back to fp32 before the output LayerNorm and linear projection.

Supported shapes:
    input          : [B, N, N, D]   (B∈{1,2}, N up to 1024, D = dim)
    hidden_dim (H) : 128
    mask (optional): [B, N, N] (broadcasted over the channel dim)
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# -------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B.T) for tensors of shape
#   A : [batch, M, K]   (fp16)
#   B : [batch, N, K]   (fp16)
#   C : [batch, M, N]   (fp16)   ->  C = A @ B.T
# -------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,            # pointers
    M, N, K,            # matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # A strides (batch, row, col)
    stride_Bb, stride_Bn, stride_Bk,   # B strides (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)          # 2‑D grid: (pid_m * num_n + pid_n)
    batch_id = tl.program_id(1)     # each batch * hidden channel

    # ---------------------------------------------------------
    # Compute block coordinates
    # ---------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of this block (including batch offset)
    a_ptrs = A + batch_id * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B + batch_id * stride_Bb + offs_n[None, :] * stride_Bn + offs_k[:, None] * stride_Bk

    # Accumulator in fp32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over K dimension
    # ---------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles (fp16)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (k * BLOCK_K + offs_k)[None, :] < K,
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=(k * BLOCK_K + offs_k)[:, None] < K & (offs_n[None, :] < N),
                    other=0.0).to(tl.float16)

        # Matrix‑multiply the tile and accumulate
        acc += tl.dot(a, b)   # result is fp32 because acc is fp32

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---------------------------------------------------------
    # Store the result (back to fp16)
    # ---------------------------------------------------------
    c_ptrs = C + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# -------------------------------------------------------------
# Python entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
                input_tensor : torch.Tensor[B, N, N, dim]   (float32)
                mask          : torch.Tensor[B, N, N]      (bool/float, may be None)
                weights       : dict of torch.Tensor with the model parameters
                config        : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor[B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ---------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim]

    # ---------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])     # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ---------------------------------------------------------
    if mask is None:
        mask_exp = torch.ones_like(left[..., :1])
    else:
        mask_exp = mask.unsqueeze(-1).type_as(left)

    left = left * mask_exp
    right = right * mask_exp

    # ---------------------------------------------------------
    # 4. Gating (linear + sigmoid) and apply
    # ---------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Batched GEMM: out_d = left_d @ right_d.T   (for each hidden channel)
    # ---------------------------------------------------------
    B, N, _, H = left.shape                      # H == hidden
    # Cast to fp16 for the kernel (still keep a copy in fp32 for later ops)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Flatten batch & channel dimension => [B*H, N, N]
    left_flat = left_fp16.view(B * H, N, N)
    right_flat = right_fp16.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides (in element counts, not bytes)
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bn, stride_Bk = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # Block sizes (tuned for Hopper H200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (num_blocks_M * num_blocks_N, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),
        B * H,
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bn, stride_Bk,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden] and cast to fp32
    out = out_flat.view(B, N, N, H).to(torch.float32)

    # ---------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
