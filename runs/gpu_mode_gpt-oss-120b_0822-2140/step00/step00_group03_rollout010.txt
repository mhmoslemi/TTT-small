    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

"""
TriMul "outgoing" kernel (AlphaFold3)

We implement the heavy N³ einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The surrounding LayerNorm / Linear / gating ops stay in PyTorch.
The core reduction is fused into a Triton kernel that computes, for every
(batch * hidden_dim) “head”, a matrix‑matrix product L @ Rᵀ where
L and R have shape (seq_len, seq_len).  This is exactly the
einsum above and dominates runtime, so accelerating it yields large
speed‑ups on Hopper GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr,          # *float32   [H, N, N]   (i,k)
    right_ptr,         # *float32   [H, N, N]   (k,j)  – already transposed
    out_ptr,           # *float32   [H, N, N]   (i,j)
    H, N,
    stride_lh, stride_li, stride_lk,
    stride_rh, stride_rk, stride_rj,
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute out = left @ rightᵀ for each head H.
        left  : (H, i, k)
        right : (H, k, j)   (right already transposed)
        out   : (H, i, j)
    """
    pid_h = tl.program_id(0)        # head dimension (B * hidden_dim)
    pid_i = tl.program_id(1)        # rows  (i)
    pid_j = tl.program_id(2)        # cols  (j)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M, K)
        left_ptrs = (left_ptr
                     + offs_h * stride_lh
                     + offs_i[:, None] * stride_li
                     + offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs,
                       mask=(mask_i[:, None] & mask_k[None, :]),
                       other=0.0)

        # right tile (K, N)
        right_ptrs = (right_ptr
                      + offs_h * stride_rh
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rj)
        right = tl.load(right_ptrs,
                        mask=(mask_k[:, None] & mask_j[None, :]),
                        other=0.0)

        # inner‑product over K
        acc += tl.dot(left, right)

    # write back the result
    out_ptrs = (out_ptr
                + offs_h * stride_oh
                + offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.
    Input:
        data = (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   float32
            - mask         : [B, N, N]         (may be all‑ones)
            - weights      : dict of tensors (see docstring)
            - config       : {"dim":int, "hidden_dim":int, "nomask":bool}
    Output:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm on the last dim ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections (no bias) ----------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        mask_u = mask.unsqueeze(-1).to(dtype)
        left = left * mask_u
        right = right * mask_u

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for the Triton kernel ----------
    # Combine batch and hidden_dim → heads = B * hidden_dim
    # We need left as (H, i, k) and right as (H, k, j)
    # Permute to bring the hidden_dim in front.
    left_hik = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k) -> (B*H, i, k)
    right_hjk = right.permute(0, 3, 1, 2).contiguous()  # (B, H, j, k)

    # Collapse batch & hidden_dim
    left_hik = left_hik.view(B * hidden_dim, N, N)       # (H, i, k)
    right_hjk = right_hjk.view(B * hidden_dim, N, N)     # (H, j, k)

    # Transpose right to (H, k, j) so that a dot product gives the desired sum over k
    right_hkj = right_hjk.permute(0, 2, 1).contiguous()  # (H, k, j)

    out_hij = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_lh, stride_li, stride_lk = left_hik.stride()
    stride_rh, stride_rk, stride_rj = right_hkj.stride()
    stride_oh, stride_oi, stride_oj = out_hij.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _trimul_kernel[grid](
        left_hik,
        right_hkj,
        out_hij,
        B * hidden_dim,
        N,
        stride_lh, stride_li, stride_lk,
        stride_rh, stride_rk, stride_rj,
        stride_oh, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_hij.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm + gating + final linear ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
