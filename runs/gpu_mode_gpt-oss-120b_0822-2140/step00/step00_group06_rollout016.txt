"""
TriMul “outgoing” implementation.

The heavy O(N³·C) reduction
    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
is implemented with a fused Triton kernel that
* loads tiles of the K‑dimension,
* multiplies the two tiles element‑wise (the hidden dimension `c` stays inside a register),
* accumulates the partial sums,
* writes the result tile.

All the surrounding LayerNorm / Linear / gating work is performed with
torch.nn.functional which is already highly optimized on the H200.
The kernel therefore accelerates the only cubic‑time part while keeping the
overall API identical to the reference PyTorch version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication + reduction over K.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # LEFT  [B, N, K, C]
    right_ptr,              # RIGHT [B, N, K, C]
    out_ptr,                # OUT   [B, N, N, C]
    B, N, C,                # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_I: tl.constexpr,  # tile size for the first output dimension (i)
    BLOCK_J: tl.constexpr,  # tile size for the second output dimension (j)
    BLOCK_K: tl.constexpr,  # reduction tile size over k
):
    """
    Compute   out[b, i, j, c] = Σₖ left[b, i, k, c] * right[b, j, k, c]

    The kernel is launched with a 2‑D grid (i_tile, j_tile) and a 1‑D
    program‑id loop over the batch dimension that is fused into the
    pointer arithmetic.
    """
    pid_i = tl.program_id(0)          # tile index over i
    pid_j = tl.program_id(1)          # tile index over j

    # ------------------------------------------------------------------
    # 1) Compute the global coordinates of the output tile.
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_offset = i_base + tl.arange(0, BLOCK_I)
    j_offset = j_base + tl.arange(0, BLOCK_J)

    # mask out-of‑range rows / cols
    i_mask = i_offset < N
    j_mask = j_offset < N

    # ------------------------------------------------------------------
    # 2) Allocate the accumulator (float32 for numerical stability).
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, C), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over the K dimension in tiles.
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        # ----- load a tile from LEFT  (shape: BLOCK_I x BLOCK_K x C) -----
        k_offset = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        left_ptrs = left_ptr + \
            i_offset[:, None, None] * stride_left_i + \
            k_offset[None, :, None] * stride_left_k + \
            tl.arange(0, C)[None, None, :] * stride_left_c

        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # ----- load a tile from RIGHT (shape: BLOCK_J x BLOCK_K x C) -----
        right_ptrs = right_ptr + \
            j_offset[:, None, None] * stride_right_j + \
            k_offset[None, :, None] * stride_right_k + \
            tl.arange(0, C)[None, None, :] * stride_right_c

        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # ----- element‑wise multiply and accumulate over K -----
        # left : (I, K, C)   right : (J, K, C)
        # we need Σₖ left[i,k,c] * right[j,k,c]
        # -> broadcast K dimension and reduce.
        prod = left[:, None, :, :] * right[None, :, :, :]      # (I, J, K, C)
        acc += tl.sum(prod, axis=2)                           # (I, J, C)

    # ------------------------------------------------------------------
    # 4) Write the result tile.
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        i_offset[:, None, None] * stride_out_i + \
        j_offset[None, :, None] * stride_out_j + \
        tl.arange(0, C)[None, None, :] * stride_out_c

    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None, None] & j_mask[None, :, None])


# ----------------------------------------------------------------------
# Helper that launches the above kernel.
# ----------------------------------------------------------------------
def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, C]   (float16)
    returns      : [B, N, N, C]   (float16)
    """
    B, N, _, C = left.shape
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # strides in elements (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # tile sizes – empirically good for Hopper
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        C=C,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_c=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[2],          # note: right has shape (B,N,K,C) where K is the 2nd N
        stride_right_k=stride_right[1],
        stride_right_c=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_c=stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor  : [B, N, N]      (float32 or bool)
        - weights_dict : model parameters
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim, hidden = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden, dim)

    left = F.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    # mask may be boolean; convert to float for multiplication
    mask_f = mask.to(left.dtype).unsqueeze(-1)      # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of a bias‑less Linear)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Trilinear reduction – the cubic part.
    #    Cast to float16 for speed, Triton kernel returns float16.
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out_f16 = trimul_triton(left_f16, right_f16)          # [B,N,N,hidden] FP16

    out = out_f16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out_gate, final Linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    out = F.linear(out, to_out_w)                # [B,N,N,dim]

    return out
