"""
TriMul “outgoing” implementation.
The expensive 4‑D einsum  (sum_k left[b,i,k,d] * right[b,j,k,d])
is off‑loaded to a Triton kernel that computes, for every
batch, i‑block × j‑block tile and for a block of hidden channels,
a batched matrix‑multiply across the k‑dimension:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

All other operations (LayerNorms, linear projections, gates,
masking, final LayerNorm + linear) are performed with PyTorch
(FP32).  The kernel works for any batch size ≤2, any sequence
length N up to 1024, and hidden_dim that is a multiple of
BLOCK_D (we default to 8).  It returns a FP32 tensor of shape
[B, N, N, hidden_dim] which is then fed through the remaining
PyTorch layers.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batch‑wise tiled matmul for each hidden channel block
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                      # static sizes
    stride_lb, stride_li, stride_lk, stride_ld,  # left strides
    stride_rb, stride_ri, stride_rk, stride_rd,  # right strides
    stride_ob, stride_oi, stride_oj, stride_od,  # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # total number of i‑tiles and j‑tiles
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # decode batch, i‑tile and j‑tile from pid
    batch = pid // (num_i * num_j)
    pid_ij = pid % (num_i * num_j)
    block_i = pid_ij // num_j
    block_j = pid_ij % num_j

    # tile offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # loop over hidden‑dim blocks
    d = 0
    while d < H:
        # accumulator for the (M,N) result of this d‑block
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over k in BLOCK_K chunks
        k = 0
        while k < N:
            offs_k = k + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ---------- load left ----------
            # shape (M, K)
            left_ptrs = (
                left_ptr
                + batch * stride_lb
                + offs_i[:, None] * stride_li
                + offs_k[None, :] * stride_lk
                + d * stride_ld
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            )

            # ---------- load right ----------
            # shape (N, K)
            right_ptrs = (
                right_ptr
                + batch * stride_rb
                + offs_j[:, None] * stride_ri
                + offs_k[None, :] * stride_rk
                + d * stride_rd
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None] & mask_k[None, :],
                other=0.0,
            )

            # accumulate dot product over k
            acc += tl.dot(left, right)   # (M, N)

            k += BLOCK_K

        # ---------- store result ----------
        out_ptrs = (
            out_ptr
            + batch * stride_ob
            + offs_i[:, None] * stride_oi
            + offs_j[None, :] * stride_oj
            + d * stride_od
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None] & mask_j[None, :],
        )
        d += BLOCK_D


# ----------------------------------------------------------------------
# Helper: layer‑norm using given weight & bias (no bias for LN without bias)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]  weight/bias: (C,)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_hat = (x - mean) * torch.rsqrt(var + eps)
    return x_hat * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the forward pass of the “outgoing” TriMul operator.
    At least the N³ einsum is executed on the GPU via Triton.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                 # original channel dim
    hidden = config["hidden_dim"]       # projection dim (H)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )
    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) -> shape [B,N,N,hidden]
    # ------------------------------------------------------------------
    # (B,N,N,dim) @ (dim, hidden).T  ==  torch.nn.functional.linear
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Compute gates (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the einsum over k:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    assert hidden % 8 == 0, "hidden_dim must be a multiple of BLOCK_D (8)"

    # allocate output
    out = torch.empty((B, N, N, hidden), device=input_tensor.device, dtype=torch.float32)

    # strides (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    BLOCK_D = 8

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    grid = (B * num_i * num_j,)

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_ri, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm (hidden) → gate → final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate                                 # gate
    out = F.linear(out, weights["to_out.weight"])        # back to dim

    return out
