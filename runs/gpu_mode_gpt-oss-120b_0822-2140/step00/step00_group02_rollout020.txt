"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be written as a batch of matrix‑multiplications per hidden
dimension (d).  In this submission we keep the surrounding linear
layers, LayerNorms and gating in pure PyTorch (they are cheap) and
off‑load the O(N³·H) tensor contraction to a custom Triton kernel.
The kernel works on blocks of the (i, j) plane and accumulates over the
k‑dimension in tiles.  The final result is cast back to float32.

The kernel is deliberately written to be independent of the exact
sequence length, batch size or hidden dimension – all of those are
passed as arguments and the block sizes are chosen to give good
occupancy on Hopper GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                     # *[B, N, N, H] left tensor
    right_ptr,                    # *[B, N, N, H] right tensor
    out_ptr,                      # *[B, N, N, H] output tensor
    B, N, H,                      # sizes (runtime)
    stride_lb0, stride_lb1, stride_lb2, stride_lb3,
    stride_rb0, stride_rb1, stride_rb2, stride_rb3,
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,
    BLOCK_M: tl.constexpr,        # tile size in dim i
    BLOCK_N: tl.constexpr,        # tile size in dim j
    BLOCK_K: tl.constexpr,        # tile size in dim k
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The kernel iterates over k in tiles of size BLOCK_K.  For each tile we
    load a (BLOCK_M, BLOCK_K, H) slice of ``left`` and a (BLOCK_N,
    BLOCK_K, H) slice of ``right`` and accumulate the outer-product over
    the k‑dimension.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) map program id to (batch, block_i, block_j)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_i * num_j)
    rem   = pid % (num_i * num_j)
    block_i = rem // num_j
    block_j = rem % num_j

    # --------------------------------------------------------------
    # 2) base offsets for the i/j dimensions
    # --------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # 3) accumulator: (BLOCK_M, BLOCK_N, H)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4) loop over k tiles
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile -------------------------------------------------
        # shape (BLOCK_M, BLOCK_K, H)
        left_ptrs = (
            left_ptr
            + batch * stride_lb0
            + offs_i[:, None, None] * stride_lb1
            + offs_k[None, :, None] * stride_lb2
            + tl.arange(0, H)[None, None, :] * stride_lb3
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )

        # ----- load right tile ------------------------------------------------
        # shape (BLOCK_N, BLOCK_K, H)
        right_ptrs = (
            right_ptr
            + batch * stride_rb0
            + offs_j[:, None, None] * stride_rb1
            + offs_k[None, :, None] * stride_rb2
            + tl.arange(0, H)[None, None, :] * stride_rb3
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )

        # ----- accumulate -----------------------------------------------------
        # For each hidden dimension h we need:
        #   acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]
        #   i.e. a matrix‑multiply of (BLOCK_M, BLOCK_K) × (BLOCK_N, BLOCK_K)ᵀ
        for h in range(0, H):
            # (BLOCK_M, BLOCK_K) and (BLOCK_N, BLOCK_K)
            l = left[:, :, h]          # (M, K)
            r = right[:, :, h]         # (N, K)

            # rᵀ has shape (K, N)
            # tl.dot works on 2‑D matrices; the result is (M, N)
            prod = tl.dot(l, r, trans_b=True)   # (M, N)

            # accumulate into the slice for this hidden dim
            acc[:, :, h] += prod

    # --------------------------------------------------------------
    # 5) write back
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_ob0
        + offs_i[:, None, None] * stride_ob1
        + offs_j[None, :, None] * stride_ob2
        + tl.arange(0, H)[None, None, :] * stride_ob3
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor : torch.Tensor  [B, N, N, dim]
        * mask         : torch.Tensor  [B, N, N]  (may be all‑ones)
        * weights      : dict of torch.Tensor – model parameters
        * config       : dict – contains ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    # Linear layers – we use F.linear to keep everything on‑device
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask broadcasting
    mask_b = mask.unsqueeze(-1)                # [B, N, N, 1]
    left  = left * mask_b
    right = right * mask_b

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Triton contraction
    # ------------------------------------------------------------------
    # Allocate output buffer (still in hidden dimension)
    out_hidden = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32   # tile size along the summed dimension

    # grid: one block per (batch, i‑tile, j‑tile)
    grid = (
        B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4) Output projection (LayerNorm + gated linear)
    # ------------------------------------------------------------------
    # to_out_norm
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```
