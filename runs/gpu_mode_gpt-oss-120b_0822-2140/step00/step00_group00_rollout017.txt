"""
TriMul “outgoing” implementation (forward only).

The heavy part of the module is the ternary multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension `d` this is a matrix multiplication between
`left[b,:, :,d]` (N×N) and the **transpose** of `right[b,:, :,d]` (N×N).
We fuse all hidden dimensions and batches into a single Triton GEMM kernel
that computes

    C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, k, j]

where `batch = batch_size * hidden_dim`.  The kernel works on `fp32`
accumulators and stores the result back in `fp32`.  All surrounding
operations (LayerNorm, Linear, gating, mask) are performed with
torch.nn.functional for simplicity.

The kernel is launched with a 1‑D grid:
    total_programs = batch * ceil(N / BLOCK_M) * ceil(N / BLOCK_N)

The implementation targets NVIDIA H200 (SM‑90) and uses block sizes that
fit comfortably in shared memory while providing good occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where B is already
# transposed (i.e. we multiply A (M×K) with B (K×N) for each batch).
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B for many independent batches."""
    pid = tl.program_id(0)

    # grid decomposition:  (batch, grid_m, grid_n) -> 1‑D pid
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (grid_m * grid_n)
    block_idx = pid % (grid_m * grid_n)
    pid_m = block_idx // grid_n
    pid_n = block_idx % grid_n

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch
    a_ptrs = A_ptr + batch_idx * stride_a_batch
    b_ptrs = B_ptr + batch_idx * stride_b_batch
    c_ptrs = C_ptr + batch_idx * stride_c_batch

    # accumulate into fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for the K‑tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load A and B tiles
        a = tl.load(
            a_ptrs
            + (offs_m[:, None] * stride_a_m)
            + (k_offset[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs
            + (k_offset[:, None] * stride_b_k)
            + (offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product (adds fp32 to fp32 accumulator)
        acc += tl.dot(a, b)

        # advance pointers for next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back
    c = acc.to(tl.float32)
    tl.store(
        c_ptrs
        + (offs_m[:, None] * stride_c_m)
        + (offs_n[None, :] * stride_c_n),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A : (batch, M, K)   fp32 contiguous
    B : (batch, K, N)   fp32 contiguous   (already transposed on the host)
    returns C : (batch, M, N) fp32
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # output allocation
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in number of elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # tuning parameters (work well on Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch configuration
    grid_m = triton.cdiv(M, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = batch * grid_m * grid_n
    grid = (total_programs,)

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton GEMM.

    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        - input   : torch.Tensor, shape [B, N, N, dim]
        - mask    : torch.Tensor, shape [B, N, N] (binary 0/1)
        - weights : dict of torch.Tensors (model parameters)
        - config  : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor  shape [B, N, N, dim] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # unpack and basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if present).  mask is broadcast over the hidden dim.
    # ------------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1)                # (B,N,N,1)
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the fused GEMM.
    # We need A : (B, hidden, N, N)   (i,k)
    #      B : (B, hidden, N, N)   (j,k)  → transpose to (B, hidden, N, N) = (batch, K, N)
    # Afterwards we treat each (b, h) pair as an independent batch.
    # ------------------------------------------------------------------
    # bring hidden dimension next to batch for easy reshaping
    A = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    B = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k,j) order

    # flatten batch × hidden → single batch dimension
    batch = B * hidden_dim
    A_flat = A.view(batch, N, N)   # (batch, M=N, K=N)
    B_flat = B.view(batch, N, N)   # (batch, K=N, N)

    # fused GEMM via Triton
    C_flat = _batched_gemm(A_flat, B_flat)   # (batch, N, N)

    # restore original layout: (B, H, N, N) → (B, N, N, H)
    C = C_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm (hidden), gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        C,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                         # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out.to(dtype)
