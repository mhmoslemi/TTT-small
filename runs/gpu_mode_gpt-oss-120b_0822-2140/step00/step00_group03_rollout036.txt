"""
TriMul “outgoing” implementation.

The heavy part of the module is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden channel `d`.  The kernel below
computes this contraction in pure Triton, looping over the reduction
dimension `k` in blocks.  All surrounding operations (LayerNorm, linear
projections, gating, the final LayerNorm + linear) are performed with
torch (they are cheap compared to the N³ reduction).  The kernel works on
tensors of shape `[B, N, N, H]` stored in row‑major order and produces an
output of the same shape.  The final result is cast back to `float32` to
match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,                # *[B,N,N,H] left tensor
    right_ptr,               # *[B,N,N,H] right tensor
    out_ptr,                 # *[B,N,N,H] output tensor
    B, N, H,                 # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,   # tile rows   (i)
    BLOCK_N: tl.constexpr,   # tile cols   (j)
    BLOCK_K: tl.constexpr,   # reduction tile (k)
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Launch grid:
        (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    Each program instance handles one (batch, head) pair and a tile of
    (i,j) positions.
    """
    pid = tl.program_id(0)                     # encodes (b, d)
    b = pid // H                               # batch index
    d = pid % H                                # hidden‑dim index

    # tile coordinates for the (i,j) plane
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_base < N
    mask_j = j_base < N

    # accumulator in FP32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in BLOCK_K steps
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # -------------------- load left[b,i,k,d] --------------------
        # layout: (B,N,N,H) → strides: (stride_left_b, stride_left_i,
        #                               stride_left_k, stride_left_h)
        left_offset = (
            b * stride_left_b
            + d * stride_left_h
            + i_base[:, None] * stride_left_i
            + k_base[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # (BLOCK_M, BLOCK_K)

        # ------------------- load right[b,j,k,d] --------------------
        # right is indexed as (b, j, k, d)
        right_offset = (
            b * stride_right_b
            + d * stride_right_h
            + j_base[:, None] * stride_right_i
            + k_base[None, :] * stride_right_k
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # (BLOCK_N, BLOCK_K)

        # -------------------------- dot -----------------------------
        # left: (M,K)   right: (N,K) → we need Σ_k left * right  so we
        # compute (M,K) @ (N,K).T = (M,N)
        acc += tl.dot(left, right)           # (BLOCK_M, BLOCK_N)

    # ------------------------ store result -------------------------
    out_offset = (
        b * stride_out_b
        + d * stride_out_h
        + i_base[:, None] * stride_out_i
        + j_base[None, :] * stride_out_j
    )
    tl.store(
        out_ptr + out_offset,
        acc.to(tl.float16),                 # store as fp16 to save bandwidth
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the outgoing TriMul operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – [B, N, N, dim]   torch.float32 (already on CUDA)
        mask    – [B, N, N] (optional, may be all‑ones)
        weights – dict with all linear / layer‑norm parameters
        config  – dict containing “dim” and “hidden_dim”
    Returns
    -------
    out : torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input (dim) ----------
    # weight/bias are 1‑D tensors of length `dim`
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections (no bias) ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask if it exists ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating (sigmoid) ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to fp16 for the heavy contraction ----------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # ---------- Compute strides (row‑major) ----------
    # shape : [B, N, N, H]
    stride_left = left_fp16.stride()   # (sB, sI, sK, sH)
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()

    # ---------- Triton launch parameters ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                    # one program per (batch, head)
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------- Convert back to float32 ----------
    out = out_fp16.to(torch.float32)

    # ---------- Final LayerNorm + gating + linear ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
