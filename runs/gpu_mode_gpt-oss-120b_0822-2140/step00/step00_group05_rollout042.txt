"""
TriMul “outgoing” implementation with a custom Triton kernel for the
N‑by‑N matrix multiplication that sums over the intermediate sequence
dimension (the einsum in the reference implementation).

The kernel computes, for every batch b and hidden channel h,
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is equivalent to a batched GEMM (left @ rightᵀ) for each hidden
channel.  All other operations (LayerNorm, Linear projections, gates,
final LayerNorm + Linear) are performed with PyTorch.

The kernel works on FP16 tensors (the heavy work) and accumulates in
FP32 for numerical stability.  The final result is cast back to FP32
before the remaining PyTorch layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# -------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             #       float16 *
    N,                   # int32  (sequence length)
    H,                   # int32  (hidden dimension)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_z = tl.program_id(2)          # combined (batch, hidden)

    # decode batch and hidden channel from pid_z
    b = pid_z // H
    h = pid_z % H

    # -----------------------------------------------------------------
    # Offsets for the current block
    # -----------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialise accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over k dimension in tiles
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------------------- load A (left) -----------------------
        # A layout: (b, i, k, h)
        a_ptrs = (left_ptr
                  + b * stride_l_b
                  + h * stride_l_h
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # ---------------------- load B (rightᵀ) --------------------
        # We need right[b, j, k, h] but use it as (k, j)
        # B layout for dot: rows = k, cols = j
        b_ptrs = (right_ptr
                  + b * stride_r_b
                  + h * stride_r_h
                  + offs_k[:, None] * stride_r_k   # k‑dimension
                  + offs_j[None, :] * stride_r_i)  # j‑dimension
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # ---------------------- GEMM tile ---------------------------
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Store result back to out (still in FP16)
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + h * stride_o_h
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Host side entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a fused Triton kernel for the
    N³ matrix multiplication.
    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        (input, mask, weights, config) as described in the statement.
    Returns
    -------
    torch.Tensor : output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------------------------------------------------------
    # 1. LayerNorm over last dim (dim)
    # ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ---------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3. Apply mask if present
    # ---------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------------------------------------------------------
    # 4. Gating parameters (sigmoid)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Cast to FP16 for the heavy N³ compute
    # ---------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # Prepare output tensor (same layout as left/right)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # ---------------------------------------------------------
    # 6. Launch Triton kernel
    # ---------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # strides of the original (B, N, N, H) layout
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),            # rows
        triton.cdiv(N, BLOCK_N),            # cols
        B * hidden_dim                      # batch * hidden
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------
    # 7. Convert back to FP32
    # ---------------------------------------------------------
    out = out_fp16.to(dtype)

    # ---------------------------------------------------------
    # 8. Final LayerNorm + gating + output linear
    # ---------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
