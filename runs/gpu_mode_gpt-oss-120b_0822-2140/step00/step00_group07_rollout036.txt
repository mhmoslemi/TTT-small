"""
TriMul “outgoing” kernel – fast forward pass.

Algorithm
---------
1. Layer‑norm the input (last dimension = dim).
2. Linear projections → left/right (shape B,N,N,hidden) and the three gates.
3. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel
   (element‑wise: left = left*mask*left_gate, right = right*mask*right_gate).
4. The core “tri‑mul” is a batched matrix multiply:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly   out = left @ rightᵀ   for every (b,d).
   We reshape to (B*H,N,N) and use torch.bmm (cuBLAS) – the most efficient way
   on Hopper GPUs.
5. Layer‑norm over the hidden dimension, multiply by out‑gate,
   and a final linear projection back to dim.

Only the mask‑gate fusion is written in Triton; everything else leverages
high‑performance Torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: apply mask and the two per‑element gates to left/right tensors #
# ----------------------------------------------------------------------------- #
@triton.jit
def _mask_gate_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    mask_ptr,               # *float32 (broadcasted over channel dim)
    left_gate_ptr,          # *float32
    right_gate_ptr,         # *float32
    total_elements,         # int64  – total # of elements in left/right
    hidden_dim: tl.constexpr,  # channel dimension (H)
    BLOCK: tl.constexpr,       # thread‑block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elements

    # load tensors
    l = tl.load(left_ptr + offs,  mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)

    lg = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask is only (B,N,N); broadcast over hidden_dim
    mask_idx = offs // hidden_dim                     # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # fuse
    l = l * m * lg
    r = r * m * rg

    # store back in‑place
    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------------- #
# Host entry point – the required `custom_kernel` function                        #
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward‑only TriMul implementation.
    Input:
        data = (input_tensor, mask, weights, config)
    Output:
        tensor of shape [B, N, N, dim]  (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5                     # LayerNorm epsilon

    B, N, _, _ = input_tensor.shape    # input shape = [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 1) Layer‑norm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B,N,N,dim]

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3) Gates (sigmoid after linear, no bias)
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------------- #
    # 4) Fuse mask + left/right gates with a Triton kernel
    # --------------------------------------------------------------------- #
    # make everything contiguous for the kernel
    left        = left.contiguous()
    right       = right.contiguous()
    left_gate   = left_gate.contiguous()
    right_gate  = right_gate.contiguous()
    mask_flat   = mask.contiguous().view(-1)               # (B*N*N)

    total = left.numel()                                   # B*N*N*H
    BLOCK = 1024                                            # tuneable
    grid  = (triton.cdiv(total, BLOCK),)

    _mask_gate_kernel[grid](
        left_ptr       = left.data_ptr(),
        right_ptr      = right.data_ptr(),
        mask_ptr       = mask_flat.data_ptr(),
        left_gate_ptr  = left_gate.data_ptr(),
        right_gate_ptr = right_gate.data_ptr(),
        total_elements = total,
        hidden_dim     = hidden_dim,
        BLOCK          = BLOCK,
    )

    # --------------------------------------------------------------------- #
    # 5) Core TriMul – batched gemm: out = left @ rightᵀ   (per hidden dim)
    # --------------------------------------------------------------------- #
    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # rightᵀ : (B*H, N, N) -> (B*H, N, N) with last two dims swapped
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H,N,N)

    # reshape back to [B,N,N,hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6) Hidden‑dim layer‑norm, scale by out_gate, final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # out_gate has shape [B,N,N,hidden] already
    out = out * out_gate

    # final projection back to dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out.to(torch.float32)
