"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden dimension `d` this is a matrix multiplication
    out_d = left_d @ right_dᵀ        (shape N×N @ N×N → N×N)

We implement this N‑by‑N matrix multiply in Triton (float16 compute,
float32 accumulation) and keep the surrounding LayerNorm/Linear
operations in PyTorch.  The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: one GEMM per (batch, hidden_dim) pair
# --------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                # float32 *
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr,  # tile rows of A
    BLOCK_N: tl.constexpr,  # tile cols of Bᵀ (i.e. rows of B)
    BLOCK_K: tl.constexpr,  # reduction tile
):
    # program IDs
    pid_bh = tl.program_id(0)                     # batch * hidden
    pid_m  = tl.program_id(1)                     # row tile of A
    pid_n  = tl.program_id(2)                     # col tile of Bᵀ

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh %  H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the current (b,h) matrices
    a_ptr = left_ptr + b * stride_b_l + h * stride_h_l
    b_ptr = right_ptr + b * stride_b_r + h * stride_h_r

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # -------- load A block (M x K) --------
        a_offs_i = offs_m[:, None] * stride_i_l
        a_offs_k = (k * BLOCK_K + offs_k)[None, :] * stride_k_l
        a = tl.load(a_ptr + a_offs_i + a_offs_k,
                    mask=mask_m[:, None] & ((k * BLOCK_K + offs_k) < N),
                    other=0.0).to(tl.float32)   # cast to fp32 for accumulation

        # -------- load Bᵀ block (K x N) --------
        # Bᵀ[k, j] = right[b, j, k, h]
        b_offs_k = (k * BLOCK_K + offs_k)[:, None] * stride_k_r
        b_offs_j = offs_n[None, :] * stride_i_r
        b = tl.load(b_ptr + b_offs_k + b_offs_j,
                    mask=((k * BLOCK_K + offs_k) < N)[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # -------- multiply‑accumulate --------
        acc += tl.dot(a, b)          # (M,K) @ (K,N) → (M,N)

    # write back
    out_offs_i = offs_m[:, None] * stride_i_o
    out_offs_j = offs_n[None, :] * stride_j_o
    out_ptrs = out_ptr + b * stride_b_o + h * stride_h_o + out_offs_i + out_offs_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy triple‑product is executed in Triton; the surrounding
    LayerNorm / Linear / gating logic stays in PyTorch.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helper to fetch weights
    # ------------------------------------------------------------------
    def w(name): return weights[name]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden]
    right = F.linear(x, w("right_proj.weight"))

    # ------------------------------------------------------------------
    # 3) Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).type_as(left)          # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton GEMM over the hidden dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # cast to fp16 for the kernel (still exact for the test ranges)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output in fp32 (accumulator precision)
    out = torch.empty_like(left_fp16, dtype=torch.float32, device=left.device)

    # strides (in elements, not bytes)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left_fp16.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right_fp16.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    # tiling parameters – tuned for Hopper (sm_90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # fits well into L2/shared memory

    grid = (
        B * H,                         # one program per (batch, hidden)
        triton.cdiv(N, BLOCK_M),       # tile rows of A
        triton.cdiv(N, BLOCK_N),       # tile cols of Bᵀ
    )
    _trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, H,
        stride_b_l, stride_i_l, stride_k_l, stride_h_l,
        stride_b_r, stride_i_r, stride_k_r, stride_h_r,
        stride_b_o, stride_i_o, stride_j_o, stride_h_o,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (hidden_dim) → Linear → gating
    # ------------------------------------------------------------------
    # out is FP32 already
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=1e-5,
    )
    out = out * out_gate                               # broadcast over dim
    out = F.linear(out, w("to_out.weight"))           # final projection

    return out.to(torch.float32)
