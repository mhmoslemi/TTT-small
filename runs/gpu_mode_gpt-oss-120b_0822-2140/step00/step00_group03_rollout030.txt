def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B,N,_,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # layernorm x
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # compute mean/var over last dim
    mean = input_tensor.mean(-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x_norm * w_norm + b_norm

    # linear projections (weight shape hidden_dim, dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    mask_uns = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_uns
    right = right * mask_uns

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # strides
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32  # hidden dim block

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_blocks = B * num_i * num_j
    grid = (total_blocks,)

    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_H = min(BLOCK_H, hidden_dim - h_start)
        # pass pointers offset
        left_ptr = left_fp16.data_ptr() + h_start * stride_left[3] * left_fp16.element_size()
        right_ptr = right_fp16.data_ptr() + h_start * stride_right[3] * right_fp16.element_size()
        out_ptr = out_fp16.data_ptr() + h_start * stride_out[3] * out_fp16.element_size()

        trimul_kernel[grid](
            left_ptr, right_ptr, out_ptr,
            B, N, cur_H,
            stride_left[0], stride_left[1], stride_left[2], stride_left[3],
            stride_right[0], stride_right[1], stride_right[2], stride_right[3],
            stride_out[0], stride_out[1], stride_out[2], stride_out[3],
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
        )

    out = out_fp16.to(torch.float32)

    # to_out_norm
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul "outgoing" kernel (AlphaFold3).

The heavy part of TriMul is the three‑dimensional einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is an N×N matrix multiplication for every hidden channel d.
The kernel below computes this operation directly on GPU with Triton.
All other linear / LayerNorm / gating steps are performed with PyTorch
(but in fp16 where possible).  The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out = Σ_k left[...,i,k,d] * right[...,j,k,d]
#
# left,right,out : (B, N, N, H)   (H = hidden_dim, processed in BLOCK_H chunks)
# The kernel works on a tile (BLOCK_M, BLOCK_N) of the (i,j) matrix
# and accumulates over k in blocks of size BLOCK_K.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # block identifiers
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_i * num_j)
    rem   = pid % (num_i * num_j)
    block_i = rem // num_j
    block_j = rem % num_j

    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator: (M,N,H) -> float32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over k dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left block -------------------------------------------------
        # shape (M, K, H)
        left_ptrs = (
            left_ptr
            + batch * stride_lb
            + offs_i[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )   # (M,K,H)

        # ----- load right block ------------------------------------------------
        # shape (N, K, H)   (j dimension first)
        right_ptrs = (
            right_ptr
            + batch * stride_rb
            + offs_j[:, None, None] * stride_rj
            + offs_k[None, :, None] * stride_rk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )   # (N,K,H)

        # ----- dot‑product over k for each hidden slice -----------------------
        for h in range(0, BLOCK_H):
            l = left[:, :, h]          # (M, K)
            r = right[:, :, h]         # (N, K)
            # trans_b=True treats r as (K,N)
            acc[:, :, h] += tl.dot(l, r, trans_b=True)

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_oh
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )

# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a fused Triton kernel.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool / 0‑1 float)
        - weights      : dict of leaf Tensors (float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B,N,N,dim), float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)               # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the cubic einsum (fp16 inside)
    # ------------------------------------------------------------------
    left_fp  = left.to(torch.float16).contiguous()
    right_fp = right.to(torch.float16).contiguous()
    out_fp   = torch.empty_like(left_fp, dtype=torch.float16)

    # strides in elements (not bytes)
    s_left = left_fp.stride()   # (N*N*H, N*H, H, 1)
    s_right = right_fp.stride()
    s_out = out_fp.stride()

    # tile sizes – tuned for Hopper SM90
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32                     # processed hidden channels per launch

    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_blocks = B * num_i * num_j
    grid = (total_blocks,)

    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_H = min(BLOCK_H, hidden_dim - h_start)

        # pointer arithmetic for the hidden‑slice start
        left_ptr  = left_fp.data_ptr()  + h_start * s_left[3] * left_fp.element_size()
        right_ptr = right_fp.data_ptr() + h_start * s_right[3] * right_fp.element_size()
        out_ptr   = out_fp.data_ptr()   + h_start * s_out[3] * out_fp.element_size()

        trimul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            B,
            N,
            cur_H,
            s_left[0],
            s_left[1],
            s_left[2],
            s_left[3],
            s_right[0],
            s_right[1],
            s_right[2],
            s_right[3],
            s_out[0],
            s_out[1],
            s_out[2],
            s_out[3],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # back to float32
    out = out_fp.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and output linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
