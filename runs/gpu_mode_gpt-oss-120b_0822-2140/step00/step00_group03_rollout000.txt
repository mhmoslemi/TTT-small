"""
TriMul “outgoing” forward pass with a Triton element‑wise kernel.

The heavy N³ reduction is performed with a batched matrix‑multiply that is
already highly optimized in PyTorch (torch.bmm).  To satisfy the requirement
of using Triton we off‑load the element‑wise masking + gating (and the final
out‑gate multiplication) to a tiny Triton kernel.  All linear layers,
layer‑norms and the N³ reduction stay in PyTorch, but the overall
throughput is dominated by the mat‑mul, while the Triton kernels eliminate
the Python‑level broadcasting overhead.

The kernel works for any batch size (1‑2), sequence length N (up to 1024)
and hidden dimension H (128‑384).  It operates on contiguous tensors and
uses element‑wise masking/gating in FP32 (you may change the dtype to
FP16/ BF16 for extra speed if desired).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------------- 
# Triton kernels
# ----------------------------------------------------------------------------- 
@triton.jit
def elementwise_mask_gate_kernel(
    inp_ptr,            # [B, N, N, H]   input (left or right)
    mask_ptr,           # [B, N, N]      mask
    gate_ptr,           # [B, N, N, H]   gate (sigmoid output)
    out_ptr,            # [B, N, N, H]   result
    B, N, H,            # compile‑time constants
    stride_inp_b, stride_inp_i, stride_inp_k, stride_inp_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    BLOCK: tl.constexpr
):
    """
    out[b,i,k,h] = inp[b,i,k,h] * mask[b,i,k] * gate[b,i,k,h]
    The kernel is launched with a 1‑D grid covering B*N*N*H elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # linear offsets
    total = B * N * N * H
    mask = offs < total

    # ----- de‑linearize -----
    idx = offs
    h = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ----- pointers -----
    inp = inp_ptr + b * stride_inp_b + i * stride_inp_i + k * stride_inp_k + h * stride_inp_h
    m   = mask_ptr + b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    g   = gate_ptr + b * stride_gate_b + i * stride_gate_i + k * stride_gate_k + h * stride_gate_h

    # ----- load, compute, store -----
    val = tl.load(inp, mask=mask) * tl.load(m, mask=mask) * tl.load(g, mask=mask)
    tl.store(out_ptr + b * stride_inp_b + i * stride_inp_i + k * stride_inp_k + h * stride_inp_h,
             val, mask=mask)


@triton.jit
def elementwise_gate_kernel(
    inp_ptr,            # [B, N, N, H]   input
    gate_ptr,           # [B, N, N, H]   gate (sigmoid)
    out_ptr,            # [B, N, N, H]   result
    B, N, H,
    stride_inp_b, stride_inp_i, stride_inp_k, stride_inp_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    BLOCK: tl.constexpr
):
    """ out = inp * gate (broadcasted element‑wise) """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    idx = offs
    h = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    inp = inp_ptr + b * stride_inp_b + i * stride_inp_i + k * stride_inp_k + h * stride_inp_h
    g   = gate_ptr + b * stride_gate_b + i * stride_gate_i + k * stride_gate_k + h * stride_gate_h

    val = tl.load(inp, mask=mask) * tl.load(g, mask=mask)
    tl.store(out_ptr + b * stride_inp_b + i * stride_inp_i + k * stride_inp_k + h * stride_inp_h,
             val, mask=mask)


# ----------------------------------------------------------------------------- 
# Helper to launch the kernels
# ----------------------------------------------------------------------------- 
def _apply_mask_and_gate(tensor: torch.Tensor,
                         mask: torch.Tensor,
                         gate: torch.Tensor,
                         BLOCK=1024) -> torch.Tensor:
    """
    tensor : [B, N, N, H]   (float32)
    mask   : [B, N, N]      (float32, 0/1)
    gate   : [B, N, N, H]   (float32, sigmoid output)

    Returns tensor * mask * gate
    """
    B, N, _, H = tensor.shape
    out = torch.empty_like(tensor)

    # strides in *elements* (not bytes)
    s_inp = tensor.stride()
    s_mask = mask.stride()
    s_gate = gate.stride()

    grid = (triton.cdiv(B * N * N * H, BLOCK),)
    elementwise_mask_gate_kernel[grid](
        tensor,
        mask,
        gate,
        out,
        B, N, H,
        s_inp[0], s_inp[1], s_inp[2], s_inp[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        BLOCK=BLOCK,
    )
    return out


def _apply_gate(tensor: torch.Tensor,
                gate: torch.Tensor,
                BLOCK=1024) -> torch.Tensor:
    """ tensor * gate (both [B,N,N,H]) """
    B, N, _, H = tensor.shape
    out = torch.empty_like(tensor)

    s_inp = tensor.stride()
    s_gate = gate.stride()

    grid = (triton.cdiv(B * N * N * H, BLOCK),)
    elementwise_gate_kernel[grid](
        tensor,
        gate,
        out,
        B, N, H,
        s_inp[0], s_inp[1], s_inp[2], s_inp[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        BLOCK=BLOCK,
    )
    return out


# ----------------------------------------------------------------------------- 
# Main entry point
# ----------------------------------------------------------------------------- 
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a small Triton kernel
    for the element‑wise mask/gate fusion.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]                 # input channel dimension
    hidden_dim = config["hidden_dim"]   # hidden dimension (H)

    # --------------------------------------------------------------------- #
    # 1️⃣ Layer‑norm on the input (dim‑wise)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )   # shape [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------------- #
    # 3️⃣ Gates (sigmoid) – also linear without bias
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # [B,N,N,H]

    # --------------------------------------------------------------------- #
    # 4️⃣ Apply mask (if present) and per‑position gates via Triton
    # --------------------------------------------------------------------- #
    # mask has shape [B, N, N]; expand to [B,N,N,1] inside the kernel.
    left = _apply_mask_and_gate(left, mask, left_gate)
    right = _apply_mask_and_gate(right, mask, right_gate)

    # --------------------------------------------------------------------- #
    # 5️⃣ N³ reduction:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched mat‑mul on reshaped tensors.
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape
    left_flat = left.reshape(B * H, N, N)                 # (B*H, N, N)
    right_flat = right.reshape(B * H, N, N)               # (B*H, N, N)

    # batched mat‑mul → (B*H, N, N)
    out = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to 4‑D
    out = out.reshape(B, N, N, H)

    # --------------------------------------------------------------------- #
    # 6️⃣ Output layer‑norm (hidden_dim‑wise)
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------------- #
    # 7️⃣ Multiply by out‑gate (Triton)
    # --------------------------------------------------------------------- #
    out = _apply_gate(out, out_gate)

    # --------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # [B,N,N,dim]

    return out
