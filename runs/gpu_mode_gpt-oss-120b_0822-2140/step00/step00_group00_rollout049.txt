"""
TriMul “outgoing” version – forward only.
-------------------------------------------------
We keep the cheap LayerNorm / Linear / gating steps in PyTorch and
off‑load the only O(N³) portion:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched matrix‑multiply per head (hidden_dim).  By reshaping
(b,h) → B*H we can reuse a standard Triton GEMM kernel (A @ Bᵀ) with
FP16 arithmetic and write the result back in FP32.

The kernel is deliberately simple: tile M×N using BLOCK_M/N and
accumulate over K‑tiles (BLOCK_K).  All strides are passed from the host
so the same kernel works for any batch size, sequence length and head
dimension.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel (A @ Bᵀ) where A,B are (batch, N, K) and K == N.
# The kernel works on FP16 data and accumulates in FP32 for numerical
# stability, then casts back to FP16 for the output tensor.
# ----------------------------------------------------------------------
@triton.jit
def _matmul_transpose_kernel(
    a_ptr,                # *ptr, left  tensor   (batch, M, K)
    b_ptr,                # *ptr, right tensor   (batch, N, K) – we read it transposed
    c_ptr,                # *ptr, output tensor  (batch, M, N)
    M, N, K,
    stride_am, stride_ak,        # A stride over M and K
    stride_bk, stride_bn,        # B stride over K and N   (note the order → transposed)
    stride_cm, stride_cn,        # C stride over M and N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch * head index
    pid_m = tl.program_id(1)              # tile over M
    pid_n = tl.program_id(2)              # tile over N

    # ------------------------------------------------------------------
    # Compute the start offsets of the block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (only needed for the final store)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulation buffer in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute actual slice sizes (last tile may be smaller)
        cur_k = tl.minimum(BLOCK_K, K - k * BLOCK_K)

        # Offsets for the A block (M x cur_k)
        a_offs_k = k * BLOCK_K + tl.arange(0, cur_k)
        a_ptrs = a_ptr + (
            pid_batch * stride_am * M * K          # batch offset
            + offs_m[:, None] * stride_am          # M offset
            + a_offs_k[None, :] * stride_ak         # K offset
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (a_offs_k[None, :] < K), other=0.0).to(tl.float32)

        # Offsets for the B block (N x cur_k) – we read it transposed
        b_offs_k = k * BLOCK_K + tl.arange(0, cur_k)
        b_ptrs = b_ptr + (
            pid_batch * stride_bk * K * N          # batch offset (note stride_bk = stride over K)
            + b_offs_k[:, None] * stride_bk        # K offset (now first dim)
            + offs_n[None, :] * stride_bn          # N offset (second dim)
        )
        b = tl.load(b_ptrs, mask=(b_offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back (cast to FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + (
        pid_batch * stride_cm * M * N
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (weight/bias supplied) – kept in PyTorch for
# simplicity and because its cost is negligible compared to O(N³).
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    return F.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # --------------------------------------------------------------
    # unpack config
    # --------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # ----- 1. LayerNorm on input ------------------------------------
    # --------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                         # (B,N,N,dim)  float32

    # --------------------------------------------------------------
    # ----- 2. Linear projections (no bias) -------------------------
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # ----- 3. Apply mask if present --------------------------------
    # --------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → broadcast over hidden_dim
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # ----- 4. Gating (sigmoid) --------------------------------------
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # ----- 5. Core O(N³) operation – batched GEMM over hidden_dim ----
    # --------------------------------------------------------------
    # Cast to FP16 for the kernel (gain both bandwidth & compute)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape  # B, seq_len, seq_len, hidden_dim
    total_batches = B * H

    # Reshape to (batch*head, N, N) so we can treat each (b,h) as an
    # independent GEMM.
    left_mat = left_h.permute(0, 3, 1, 2).reshape(total_batches, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).reshape(total_batches, N, N)
    out_mat = torch.empty_like(left_mat)   # fp16 output

    # Get strides (in elements, not bytes)
    stride_am, stride_ak = left_mat.stride()[1], left_mat.stride()[2]
    stride_bk, stride_bn = right_mat.stride()[1], right_mat.stride()[2]
    stride_cm, stride_cn = out_mat.stride()[1], out_mat.stride()[2]

    # Triton launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        total_batches,                       # program id 0 → batch*head
        triton.cdiv(N, BLOCK_M),             # program id 1 → M‑tiles
        triton.cdiv(N, BLOCK_N),             # program id 2 → N‑tiles
    )

    _matmul_transpose_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                              # M,N,K (all equal seq_len)
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to original layout (B,N,N,hidden_dim) and up‑cast to fp32
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # ----- 6. Post‑processing (LayerNorm, gating, final linear) -----
    # --------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out
