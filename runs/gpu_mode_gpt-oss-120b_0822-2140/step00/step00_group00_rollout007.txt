"""
TriMul “outgoing” implementation.

The costly part of the module is the 3‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be rewritten for each (b,d) as a matrix multiplication
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ          (shape N×N)

The kernel below performs a batched FP16 GEMM on the
(B * hidden_dim) matrices of size N×N.  All surrounding
operations (layer‑norms, linear projections, gates, final
linear) are kept in PyTorch for simplicity and numerical
stability.  The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A : (B*H, M, K)   B : (B*H, N, K)   C : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix dimensions
    stride_am, stride_ak, stride_ab,   # A strides (row, col, batch)
    stride_bm, stride_bn, stride_bb,   # B strides (row, col, batch)
    stride_cm, stride_cn, stride_cb,   # C strides (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch index = b * hidden
    pid_m = tl.program_id(1)              # block-row index
    pid_n = tl.program_id(2)              # block-col index

    # ---- offsets within the current block --------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- compute pointers --------------------------------------------------
    a_ptrs = (A + pid_batch * stride_ab
                 + offs_m[:, None] * stride_am
                 + (offs_k)[None, :] * stride_ak)
    b_ptrs = (B + pid_batch * stride_bb
                 + (offs_k)[:, None] * stride_bm
                 + offs_n[None, :] * stride_bn)

    # ---- accumulation ------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension --------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute the actual K‑offset for this iteration
        k_offset = k * BLOCK_K

        # masks for loading valid elements
        a_mask = (offs_m[:, None] < M) & (k_offset + offs_k[None, :] < K)
        b_mask = (k_offset + offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bm

    # ---- write back --------------------------------------------------------
    c_ptrs = (C + pid_batch * stride_cb
                 + offs_m[:, None] * stride_cm
                 + offs_n[None, :] * stride_cn)

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‐accelerated
    batched matrix multiplication.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm over the last dim --------------------------------
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------- Linear projections (no bias) --------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates --------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for the Triton GEMM -------------------------
    # reshape to (B*hidden, N, N) and cast to fp16 for the kernel
    left_ = left.permute(0, 3, 1, 2).contiguous()   # B, hidden, N, N
    right_ = right.permute(0, 3, 1, 2).contiguous() # B, hidden, N, N

    left_ = left_.view(B * hidden, N, N).to(torch.float16)
    # we need rightᵀ for the matmul, i.e. transpose the last two dims
    right_T = right_.transpose(-2, -1).contiguous()
    right_T = right_T.view(B * hidden, N, N).to(torch.float16)

    out_ = torch.empty_like(left_)                  # (B*hidden, N, N), fp16

    # ---------- Triton launch ------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_, right_T, out_,
        N, N, N,
        left_.stride(1), left_.stride(2), left_.stride(0),
        right_T.stride(1), right_T.stride(2), right_T.stride(0),
        out_.stride(1), out_.stride(2), out_.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ---------- Reshape back to original layout -------------------------------
    out = out_.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ---------- Final LayerNorm, gating and projection -----------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim C, no bias
    out = out.to(torch.float32)

    return out
