"""
TriMul (outgoing) – fast Triton implementation of the N³ einsum.

The heavy part of the module is the following contraction:

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where ``left`` and ``right`` have shape (B, N, N, H) (H = hidden_dim).
For each batch and hidden dimension the operation is a matrix‐matrix
multiplication of size (N×N) · (N×N)ᵀ, i.e. a batched GEMM.
The kernel below fuses the inner product over ``k`` using tiled
matrix‑multiplication (standard Triton matmul pattern) and writes the
result back to ``out``.  All other linear layers, LayerNorms and gates are
performed with PyTorch – they are cheap compared to the N³ term, but the
GEMM is the bottleneck and is therefore off‑loaded to Triton.

The kernel works for any batch size (≤2), any sequence length N (≤1024)
and any hidden dimension H (128 or 384).  It uses FP32 accumulation and
stores the final tensor in FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel – batched GEMM over the ``k`` dimension.
# -------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,           # const float*  (B, N, N, H)   left  [i,k,d]
    right_ptr,          # const float*  (B, N, N, H)   right [j,k,d]  (will be read as (k,j))
    out_ptr,            # float*        (B, N, N, H)   output[i,j,d]

    B, N, H,            # problem sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides (b,i,k,h)
    stride_rb, stride_rk, stride_rj, stride_rh,   # right strides (b,k,j,h) – note swapped order!
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides  (b,i,j,h)

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)                     # tile over i
    pid_j = tl.program_id(1)                     # tile over j
    pid_hb = tl.program_id(2)                    # combined batch * hidden dim

    batch_id = pid_hb // H
    hid_id   = pid_hb % H

    # -------------------- offsets for the current tile --------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)   (i,k)
        left_ptrs = left_ptr + \
            batch_id * stride_lb + \
            hid_id   * stride_lh + \
            offs_i[:, None] * stride_li + \
            offs_k[None, :] * stride_lk

        # right tile: we want (k,j) layout → shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            batch_id * stride_rb + \
            hid_id   * stride_rh + \
            offs_k[:, None] * stride_rk + \
            offs_j[None, :] * stride_rj

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # standard tiled GEMM accumulation
        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # write back
    out_ptrs = out_ptr + \
        batch_id * stride_ob + \
        hid_id   * stride_oh + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Host side entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input : Tensor[B, N, N, dim] (float32/float16 on CUDA)
            - mask  : Tensor[B, N, N] (bool/int/float) – may be all‑ones.
            - weights: dict of learned parameters (already on CUDA)
            - config : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    torch.cuda.set_device(torch.cuda.current_device())

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # torch.nn.functional.layer_norm expects normalized_shape = dim
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -----------------------------------------------------------------
    # 2. Linear projections  (no bias)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3. Mask handling (if a mask is provided)
    # -----------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] – cast to float and broadcast over hidden dim
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton GEMM: out = Σₖ left[...,k,:] * right[...,k,:]^T
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim

    # Make sure tensors are contiguous for simple stride extraction
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty_like(left)   # will hold the intermediate result

    # Strides (in number of elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    # Right is accessed as (k, j) → we need strides for that order
    # Original order: (B, N_j, N_k, H)
    #   stride over j = stride_rj, stride over k = stride_rk
    # When we treat right as (k, j) we keep the same numeric strides:
    #   rows (k) use stride_rk, cols (j) use stride_rj
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – experimentally good for H200 (FP32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i tiles
        triton.cdiv(N, BLOCK_N),          # j tiles
        B * H                             # batch * hidden
    )

    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rk=stride_rk,   # rows = k
        stride_rj=stride_rj,   # cols = j
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate       # gating
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)         # final projection

    return out.to(torch.float32)
