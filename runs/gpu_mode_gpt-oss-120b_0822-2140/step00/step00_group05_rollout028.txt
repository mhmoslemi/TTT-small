"""
TriMul "outgoing" kernel (AlphaFold3 style) – forward only.

The heavy part of the module is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension `d` this is a matrix multiplication
    out_d = left_d @ right_d.T      (shape N×N)
We fuse all hidden dimensions (and the batch dimension) into a single
batched GEMM:
    A : (B*H, N, K)   = left.reshape(-1, N, N)
    B : (B*H, N, K)   = right.reshape(-1, N, N)   (note the second N is the K‑dim)
    C : (B*H, N, N)   = A @ Bᵀ

The GEMM is implemented in Triton (block‑wise dot product). All other
operations (LayerNorms, Linear projections, sigmoids) are performed with
torch.nn.functional for simplicity and to keep the code concise.

The kernel works with fp16 arithmetic for speed and casts the result back
to fp32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM: C = A @ Bᵀ   (A: (M,K), B: (N,K), C: (M,N))
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_transpose_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,      # A: (batch, M, K)
    stride_bk, stride_bn,      # B: (batch, N, K)   (we will read B as (N,K))
    stride_cm, stride_cn,      # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch (B*H)
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # Compute the start offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current block
    a_ptrs = A_ptr + pid_z * stride_am + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_z * stride_bn + offs_n[:, None] * stride_bn + tl.arange(0, BLOCK_K)[None, :] * stride_bk

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        ksize = tl.where(cur_k + BLOCK_K > K, K - cur_k, BLOCK_K)

        # Load tiles (use mask for tail)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < ksize)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (tl.arange(0, BLOCK_K) < ksize)[None, :], other=0.0)

        acc += tl.dot(a, b)   # (M,K)·(N,K)ᵀ → (M,N)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = C_ptr + pid_z * stride_cm + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the batched GEMM
# ----------------------------------------------------------------------
def batched_gemm_transpose(left: torch.Tensor, right: torch.Tensor):
    """
    left, right: (B, N, N, H)  fp32
    returns: (B, N, N, H)  fp32   (C = left @ rightᵀ per hidden dim)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.reshape(-1, N, N).contiguous()
    right_flat = right.reshape(-1, N, N).contiguous()
    out_flat = torch.empty_like(left_flat, dtype=left.dtype)

    # Strides in *elements* (not bytes)
    stride_am = left_flat.stride(0)   # batch stride
    stride_ak = left_flat.stride(2)   # K stride (inner)
    stride_bk = right_flat.stride(2)  # K stride
    stride_bn = right_flat.stride(1)  # N stride (row of B)
    stride_cm = out_flat.stride(0)
    stride_cn = out_flat.stride(2)

    M = N
    K = N
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        left_flat.shape[0],          # B*H
    )
    triton_matmul_transpose_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back to (B, N, N, H)
    return out_flat.reshape(B, N, N, H)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a Triton kernel
    for the N³ multiplication.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N]      (float32 or bool)
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input   (LN over last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float32)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if present (mask shape B,N,N).  The reference code multiplies
    # both left and right by the same mask.
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating vectors (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy N³ part – use Triton kernel
    # ------------------------------------------------------------------
    # Cast to fp16 before the GEMM for speed (the kernel works in fp16 and
    # accumulates in fp32).  The final result is cast back to fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = batched_gemm_transpose(left_fp16, right_fp16)   # (B,N,N,hidden) fp16
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim) + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return output
