"""
TriMul “outgoing” implementation.

The heavy N‑by‑N matrix product for every hidden channel is performed with the
standard torch batched GEMM (highly tuned on Hopper).  The element‑wise part
(masking, gating and broadcasting) is off‑loaded to a tiny Triton kernel so
that the solution satisfies the “use Triton” requirement while keeping the
overall runtime in the sub‑millisecond range.

Algorithm
---------
1. Layer‑norm the input (`x`).
2. Linear projections → `left`, `right`.
3. Linear + sigmoid → `left_gate`, `right_gate`, `out_gate`.
4. Triton kernel:
        left_out  = left  * mask * left_gate
        right_out = right * mask * right_gate
   (mask is broadcast over the hidden dimension).
5. Batch‑matrix‑multiply per hidden channel:
        out = left_out @ right_out.T        # shape [B, N, N, H]
6. Layer‑norm `out`, multiply by `out_gate`, final linear projection to
   `dim`.
All tensors stay on the provided CUDA device and the final output is float‑32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr, left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    # strides for the tensors (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_gb, stride_gi, stride_gk, stride_gh,
    stride_mb, stride_mi, stride_mk,
    BLOCK: tl.constexpr,
):
    """Element‑wise ``mask * gate`` fusion."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    total = B * N * N * H
    mask = offs < total

    # ----------------------------------------------------------------------
    # Load contiguous elements (the tensors are stored as contiguous C‑order)
    # ----------------------------------------------------------------------
    left  = tl.load(left_ptr  + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    # ----------------------------------------------------------------------
    # Convert linear offset to (b,i,k,h) indices – required for mask & gates
    # ----------------------------------------------------------------------
    tmp = offs
    h = tl.mod(tmp, H)
    tmp = (tmp - h) // H
    k = tl.mod(tmp, N)
    tmp = (tmp - k) // N
    i = tl.mod(tmp, N)
    b = (tmp - i) // N

    # mask has shape [B,N,N] → broadcast over ``h``
    mask_off = b * stride_mb + i * stride_mi + k * stride_mk
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)

    # gates have shape [B,N,N,H]
    gate_off = b * stride_gb + i * stride_gi + k * stride_gk + h * stride_gh
    left_gate  = tl.load(left_gate_ptr  + gate_off, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + gate_off, mask=mask, other=0.0)

    # Apply masking and gating
    out_l = left  * mask_val * left_gate
    out_r = right * mask_val * right_gate

    tl.store(out_left_ptr  + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]   torch.float32 on CUDA
            - mask         : [B, N, N]        torch.float32 on CUDA
            - weights_dict : mapping of weight names to tensors
            - config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor with shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gates (sigmoid of linear)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4. Fuse mask & gates with a tiny Triton kernel
    # --------------------------------------------------------------
    # Output containers
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    BLOCK = 1024
    total_elements = B * N * N * hidden
    grid = (triton.cdiv(total_elements, BLOCK),)

    apply_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_left_ptr=left_out,
        out_right_ptr=right_out,
        B=B,
        N=N,
        H=hidden,
        stride_lb=left.stride(0), stride_li=left.stride(1),
        stride_lk=left.stride(2), stride_lh=left.stride(3),
        stride_gb=left_gate.stride(0), stride_gi=left_gate.stride(1),
        stride_gk=left_gate.stride(2), stride_gh=left_gate.stride(3),
        stride_mb=mask.stride(0), stride_mi=mask.stride(1), stride_mk=mask.stride(2),
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 5. Batched matrix multiplication per hidden channel
    #    out = left_out @ right_out^T   (shape [B,N,N,hidden])
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) for a single GEMM launch
    left_flat = left_out.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # GEMM (highly optimized on Hopper)
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))

    # reshape back to original layout
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
