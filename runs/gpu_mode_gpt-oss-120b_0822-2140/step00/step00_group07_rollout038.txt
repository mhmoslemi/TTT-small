@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_lb, stride_li, stride_lk, stride_lh,
                   stride_rb, stride_ri, stride_rk, stride_rh,
                   stride_ob, stride_oi, stride_oj, stride_oh,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # load left: (i,k)
        left_ptrs = left_ptr + b*stride_lb + i_off[:,None]*stride_li + k_off[None,:]*stride_lk + h*stride_lh
        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float16)

        # load right: (j,k) where right index order (j,k) but memory (i=j, k)
        right_ptrs = right_ptr + b*stride_rb + j_off[:,None]*stride_ri + k_off[None,:]*stride_rk + h*stride_rh

"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input tensor x ∈ (B,N,N,D).
2. Linear projections → left/right of shape (B,N,N,H) (bias‑free).
3. Optional mask (broadcast over the hidden dimension).
4. Gating: sigmoid(linear(x)) applied element‑wise to left/right.
5. Core “TriMul” operation:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply of shape (B·H,N,N) and is
   computed with a custom Triton kernel (float16 compute, FP32 accumulation).
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   a final linear projection back to D.
All tensors are kept on the device; only the cubic‑cost reduction
(einsum) is executed in Triton.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# --------------------------------------------------------------------
# Triton kernel for the N×N×H tensor contraction.
# --------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # static sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] """
    pid_i = tl.program_id(0)          # tile over i (first N)
    pid_j = tl.program_id(1)          # tile over j (second N)
    pid_bh = tl.program_id(2)         # fused batch‑head dimension

    b = pid_bh // H                  # original batch index
    h = pid_bh % H                   # hidden‑dim index

    # ----------------------------------------------------------------
    # Tile offsets
    # ----------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator in FP32 (more stable for long reductions)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------
    # Loop over K dimension in blocks
    # ----------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left[b,i,k,h]  (i dimension is stride_li, k is stride_lk)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_offset[:, None] * stride_li
            + k_offset[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # right[b,j,k,h] – note that in memory the “j” index is the second
        # dimension (same stride as the first N dimension of left)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + j_offset[:, None] * stride_ri
            + k_offset[None, :] * stride_rk
            + h * stride_rh
        )
        right = tl.load(right_ptrs,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0)

        # dot over K (fp16 × fp16 → fp32 accumulation)
        acc += tl.dot(left, right)

    # ----------------------------------------------------------------
    # Write the result back (still fp16 for storage, later cast to fp32)
    # ----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_offset[:, None] * stride_oi
        + j_offset[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a Triton kernel
    for the cubic‑cost einsum.

    Args:
        data: tuple (input, mask, weights, config)
              - input : Tensor[B, N, N, D] (float32)
              - mask  : Tensor[B, N, N]   (bool/int, may be all‑ones)
              - weights: dict of model parameters (all float32)
              - config : dict with keys
                  * dim        – D
                  * hidden_dim – H
                  * nomask     – bool (True ⇒ ignore mask)

    Returns:
        Tensor[B, N, N, D] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, D = x.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------
    # 1. Input LayerNorm
    # ----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dim)
    # ----------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)               # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ----------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5. Core TriMul (batched N×N matrix multiply per hidden dim)
    #    Compute in float16 for speed, accumulate in FP32.
    # ----------------------------------------------------------------
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    out_f16 = torch.empty_like(left_f16)          # (B,N,N,H)

    # Triton launch parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),                # i‑tiles
        triton.cdiv(N, BLOCK_N),                # j‑tiles
        B * H,                                   # fused batch‑hidden dimension
    )

    _tri_mul_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B,
        N,
        H,
        # strides for left
        left_f16.stride(0),
        left_f16.stride(1),
        left_f16.stride(2),
        left_f16.stride(3),
        # strides for right
        right_f16.stride(0),
        right_f16.stride(1),
        right_f16.stride(2),
        right_f16.stride(3),
        # strides for out
        out_f16.stride(0),
        out_f16.stride(1),
        out_f16.stride(2),
        out_f16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_f16.to(torch.float32)              # back to FP32

    # ----------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear projection
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
