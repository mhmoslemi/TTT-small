"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the module is the 4‑D einsum

    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

which is equivalent to a batched matrix multiplication per hidden channel:
for each batch b and hidden dimension d

    O_bd = A_bd @ B_bd.T   with  A_bd = left[b,d]   (N×N)
                           B_bd = right[b,d]  (N×N)

We fuse this multiplication with Triton.  All other ops (LayerNorm,
linear projections, sigmoids) are cheap and stay in PyTorch.

The kernel works on a flattened leading dimension `B * H` (batch × hidden)
and computes a dense mat‑mul with tiling (BLOCK_M, BLOCK_N, BLOCK_K).  The
result is reshaped back to `[B, N, N, H]`.  All arithmetic is performed in
float32; the final output is cast to float32 as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ B^T   where A,B : [B*H, N, N]
# ----------------------------------------------------------------------
@triton.jit
def tri_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N,                                 # matrix size (square)
    stride_ah, stride_ai, stride_ak,   # A: batch, i, k strides
    stride_bh, stride_bi, stride_bk,   # B: batch, i, k strides (i == row index)
    stride_ch, stride_ci, stride_cj,   # C: batch, i, j strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # flattened (batch * hidden)
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # starting offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers with the batch offset
    A_batch_ptr = A_ptr + pid_batch * stride_ah
    B_batch_ptr = B_ptr + pid_batch * stride_bh
    C_batch_ptr = C_ptr + pid_batch * stride_ch

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # A[i, k]
        a_ptrs = A_batch_ptr + (offs_m[:, None] * stride_ai
                                 + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B[j, k]   (note: we need B^T, so we read row=j, col=k)
        b_ptrs = B_batch_ptr + (offs_n[:, None] * stride_bi
                                 + cur_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # dot over k
        acc += tl.dot(a, b)

        # advance k
    # write result
    c_ptrs = C_batch_ptr + (offs_m[:, None] * stride_ci
                             + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def tri_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : tensors of shape [B, H, N, N]   (contiguous)
    returns torch.Tensor of same shape = a @ b^T per (B,H) slice
    """
    B, H, N, _ = a.shape
    # flatten leading dimensions
    a_flat = a.reshape(B * H, N, N)
    b_flat = b.reshape(B * H, N, N)

    # ensure contiguous and get strides (in elements)
    a_flat = a_flat.contiguous()
    b_flat = b_flat.contiguous()
    out_flat = torch.empty_like(a_flat)

    # strides (in elements)
    stride_ah, stride_ai, stride_ak = a_flat.stride()
    stride_bh, stride_bi, stride_bk = b_flat.stride()
    stride_ch, stride_ci, stride_cj = out_flat.stride()

    # tiling parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_matmul_kernel[grid](
        a_flat,
        b_flat,
        out_flat,
        N,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bi,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out_flat.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input, mask, weights, config)
            input : torch.Tensor [B, N, N, dim]   (float32 or bf16)
            mask  : torch.Tensor [B, N, N]       (bool/float)
            weights: dict of model parameters (see prompt)
            config : dict with keys "dim" and "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # c_z in paper

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core einsum via Triton: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    -> batch matmul per hidden channel
    # ------------------------------------------------------------------
    # reshape to [B, hidden, N, N] and make contiguous for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    out = tri_matmul(left_perm, right_perm)            # [B, H, N, N]
    # back to original layout [B, N, N, H]
    out = out.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden dim) and final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])      # [B,N,N,dim]

    # ensure float32 output
    return out.to(torch.float32)
