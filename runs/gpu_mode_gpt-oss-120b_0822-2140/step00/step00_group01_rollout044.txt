"""
TriMul “outgoing” implementation for AlphaFold‑3.

The heavy part – the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reformulated as a batched matrix‑multiply:
    left_slice  = left[b,:, :, d]          # (N, N)
    right_slice = right[b,:, :, d]          # (N, N)
    out_slice   = left_slice @ right_slice.T   # (N, N)

All LayerNorm / Linear / sigmoid work in regular PyTorch (fast on HW),
while the *N³* contraction is executed by a custom Triton kernel which
computes many independent (batch * hidden_dim) matrix products in parallel.
The kernel is a classic tiled GEMM (BLOCK_M × BLOCK_N × BLOCK_K) with
the second operand accessed transposed, i.e. we compute A @ Bᵀ.

The final tensor is again LayerNorm‑ed, gated and linearly projected back
to the original `dim`.  The output is required to be float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   where A,B shape (M, N, K) and K == N
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix sizes
    stride_am, stride_ak,        # strides for A (M dim, K dim)
    stride_bm, stride_bk, stride_bn,  # strides for B (M dim, K dim, N dim)
    stride_cm, stride_cn,        # strides for C (M dim, N dim)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # 2‑D tiling over (M, N)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicates
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to A and B⁺ᵀ
    a_ptrs = a_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)

    # B is accessed transposed: we need B[m, k, n] where original layout is (M,N,K)
    # original strides: (stride_bm, stride_bn, stride_bk)
    # after transpose we read B at (m, k, n) → offset = m*stride_bm + k*stride_bk + n*stride_bn
    b_ptrs = b_ptr + (
        offs_m[:, None] * stride_bm
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K)), other=0.0)
        b = tl.load(b_ptrs, mask=((k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :]), other=0.0)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float32), mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    left / right : (B, N, N, hidden_dim)  float32, contiguous
    Returns tensor of same shape containing left @ rightᵀ for each hidden dim.
    """
    B, N, _, H = left.shape
    assert H == hidden_dim and right.shape == left.shape

    # (B, H, N, N) -> (M, N, K) where M = B*H, K = N
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim

    A = left_perm.view(M, N, N)
    B_ = right_perm.view(M, N, N)

    C = torch.empty_like(A)

    # tiling sizes – chosen to fit in shared memory on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    _tri_mul_kernel[grid](
        A, B_, C,
        M, N, N,
        A.stride(0), A.stride(2),          # stride_am , stride_ak
        B_.stride(0), B_.stride(2), B_.stride(1),   # stride_bm , stride_bk , stride_bn
        C.stride(0), C.stride(1),          # stride_cm , stride_cn
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, N, N, hidden_dim)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args:
        data: tuple (input, mask, weights, config)
            input  – Tensor [B, N, N, dim] (float32, on CUDA)
            mask   – Tensor [B, N, N] (bool/float) or None
            weights – dict with the eight weight tensors
            config – dict with keys dim, hidden_dim, nomask (bool)

    Returns:
        Tensor [B, N, N, dim] (float32, on the same device)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input channel dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – shape (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; unsqueeze to [...,1] to broadcast over hidden_dim
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy contraction via Triton (A @ Bᵀ per hidden dim)
    # ------------------------------------------------------------------
    contracted = _tri_mul_triton(left, right, hidden_dim)          # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6) Output normalization, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        contracted,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])                # back to dim

    return out
