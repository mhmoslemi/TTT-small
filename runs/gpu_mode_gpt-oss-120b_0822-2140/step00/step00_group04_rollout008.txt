    device = input_tensor.device
    dtype = input_tensor.dtype

    hidden = config["hidden_dim"]
    # LayerNorm on input
    # using torch.nn.functional.layer_norm

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    right = right * right_gate

    right_perm = right.permute(0,3,2,1).contiguous()

    out_perm = torch.empty_like(left_perm)  # same shape

    BLOCK_N = 64

"""
TriMul “outgoing” implementation.

The expensive part of TriMul is the Einstein‑summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every hidden dimension *d*.
We fuse this N³ reduction with a custom Triton kernel that computes a
batched GEMM of shape (B·D, N, N) × (B·D, N, N)ᵀ → (B·D, N, N).
All other operations (layer‑norms, linear projections, gating, mask)
are performed with PyTorch (using the provided weight tensors).  The final
output is returned in float‑32.

The kernel works on three‑dimensional tensors that are contiguous in memory
(batch, row, col).  It tiles the output with BLOCK_M × BLOCK_N tiles and
accumulates over the shared K dimension in BLOCK_K‑sized chunks with
tl.dot.  The implementation is fully compatible with Triton 3.3.1 and the
H200 accelerator.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# --------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) where A,B are (B*D, N, N)
# --------------------------------------------------------------------
@triton.jit
def tri_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    N,                                 # sequence length
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute (batch, N, N) @ (batch, N, N)ᵀ -> (batch, N, N)
    where each batch index corresponds to a distinct hidden channel.
    """
    pid_batch = tl.program_id(0)                     # which (b,d) pair
    pid_m = tl.program_id(1)                         # row tile
    pid_n = tl.program_id(2)                         # col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + \
                 pid_batch * stride_a_batch + \
                 offs_m[:, None] * stride_a_row + \
                 offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B block (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + \
                 pid_batch * stride_b_batch + \
                 offs_k[:, None] * stride_b_row + \
                 offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate dot product
        acc += tl.dot(a, b)

    # store the result
    c_ptrs = c_ptr + \
             pid_batch * stride_c_batch + \
             offs_m[:, None] * stride_c_row + \
             offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, dim]
            - mask         : torch.Tensor, shape [B, N, N] (float 0/1) or None
            - weights      : dict of torch.Tensor with the model parameters
            - config       : dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ----------------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------------
    # 2) Linear projections and gating
    # ----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # optional mask
    if mask is not None and mask.numel() > 0:
        mask_f = mask.unsqueeze(-1).to(dtype)          # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 3) Batched GEMM via Triton (compute the N³ reduction)
    # ----------------------------------------------------------------
    # reshape to (B*hidden, N, N) and make the right operand transposed
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N).contiguous()
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N).contiguous()

    out_mat = torch.empty_like(left_mat)

    # launch parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * hidden,                     # batch dimension (B·D)
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
    )

    tri_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------
    # 4) Output LayerNorm, gating and final linear
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
