"""
TriMul “outgoing” implementation.

The heavy 3‑D tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reduced to a batched matrix multiply per hidden channel d.
We fuse the N×N ×‑N reduction with a Triton kernel (block‑wise dot product)
and keep the cheap surrounding ops (LayerNorm, linear projections,
gates) in PyTorch.

The kernel works on a flattened view of the tensors:
    (B*H, N, N)  where H = hidden_dim.
For each (batch*head) the kernel computes C = A @ Bᵀ with
    A[i,k] = left[i,k]      (i‑row, k‑col)
    B[j,k] = right[j,k]     (j‑row, k‑col)
i.e. a standard batched matmul but with the “right” operand not transposed
in memory – we just load it with the j‑stride.

The kernel accumulates in FP32, stores in FP16 (saved memory / bandwidth) and
the outer code casts back to FP32 for the final LayerNorm / linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N×N×N contraction (per hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,              # A: row‑stride, col‑stride
    stride_bn, stride_bk,              # B: row‑stride (j), col‑stride (k)
    stride_cm, stride_cn,              # C: row‑stride, col‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)           # block on M (i dimension)
    pid_n = tl.program_id(1)           # block on N (j dimension)
    batch = tl.program_id(2)           # combined (batch * hidden)

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch = a_ptr + batch * (stride_am * M + stride_ak * K)   # stride to next batch
    b_batch = b_ptr + batch * (stride_bn * N + stride_bk * K)
    c_batch = c_ptr + batch * (stride_cm * M + stride_cn * N)

    # Pointer arithmetic
    a_ptrs = a_batch + (offs_m[:, None] * stride_am
                        + (offs_k[None, :] * stride_ak))
    b_ptrs = b_batch + (offs_n[:, None] * stride_bn
                        + (offs_k[None, :] * stride_bk))

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (mask out‑of‑range elements)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k[None, :] < K), other=0.0)
        # Dot‑product across k dimension
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Cast back to FP16 for storage
    c = acc.to(tl.float16)

    # Store the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def _batched_matmul_triton(left_fp16: torch.Tensor,
                           right_fp16: torch.Tensor,
                           seq_len: int,
                           BLOCK_M: int = 64,
                           BLOCK_N: int = 64,
                           BLOCK_K: int = 32) -> torch.Tensor:
    """
    left_fp16/right_fp16 : (B*H, N, N)  – contiguous, dtype half
    returns out_fp16 of same shape
    """
    BxH, N, _ = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16)

    # Strides in element counts (not bytes)
    stride_am = left_fp16.stride(1)          # i‑stride
    stride_ak = left_fp16.stride(2)          # k‑stride
    stride_bn = right_fp16.stride(1)         # j‑stride
    stride_bk = right_fp16.stride(2)         # k‑stride
    stride_cm = out_fp16.stride(1)           # i‑stride
    stride_cn = out_fp16.stride(2)           # j‑stride

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        BxH,                       # batch dimension
    )
    _trimul_kernel[grid](
        left_fp16.data_ptr(),
        right_fp16.data_ptr(),
        out_fp16.data_ptr(),
        N, N, N,
        stride_am, stride_ak,
        stride_bn, stride_bk,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out_fp16


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim] (float32, on CUDA)
            mask   – torch.Tensor [B, N, N]    (float32 or bool)
            weights – dict of all linear / ln parameters (float32)
            config  – dict with keys: dim, hidden_dim, nomask (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gates (sigmoid) – also linear without bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare for the Triton kernel
    # ------------------------------------------------------------------
    # Cast to FP16 for higher throughput, keep a contiguous layout
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Merge batch and hidden dimensions -> (B*H, N, N)
    left_flat = left_fp16.view(B * hidden, N, N)
    right_flat = right_fp16.view(B * hidden, N, N)

    # Batched matmul via Triton: out = left @ rightᵀ  (summation over k)
    out_fp16 = _batched_matmul_triton(left_flat, right_flat, seq_len=N)

    # Reshape back to [B, N, N, hidden] and convert to FP32 for later ops
    out = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden dim) and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast on hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
