"""
TriMul “outgoing’’ kernel (AlphaFold‑3).

Given 4‑D tensor X ∈ ℝ^{B×N×N×C} we compute

    X̂ = LayerNorm(X)                                          (C = dim)
    L   = X̂·W_left   ,   R = X̂·W_right        (→ B×N×N×H)
    L   = L * mask * sigmoid(X̂·W_left_gate)
    R   = R * mask * sigmoid(X̂·W_right_gate)

    O_{i,j,d} = Σ_k L_{i,k,d} · R_{j,k,d}        (for every batch b and d∈[H))

    Ô = LayerNorm(O) * sigmoid(X̂·W_out_gate)
    Y  = Ô·W_out        → B×N×N×C
where H = hidden_dim.

The heavy N³ “einsum’’ is implemented as a batched matrix‑multiply
L_{b,d} @ R_{b,d}ᵀ using a single Triton kernel.  The kernel treats each
pair (b, d) as an independent mat‑mul of size N×N, tiled with
BLOCK_M×BLOCK_N×BLOCK_K blocks.  Input/outputs are kept in fp16 inside the
kernel for higher throughput and cast back to fp32 afterwards.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = A @ Bᵀ   where A,B ∈ ℝ^{(B*H)×N×N}
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, N, H,                           # shapes
    stride_b_a, stride_i_a, stride_k_a, stride_h_a,
    stride_b_b, stride_i_b, stride_k_b, stride_h_b,
    stride_b_c, stride_i_c, stride_j_c, stride_h_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------- block decomposition -------------------------
    # total number of (batch*head) blocks:
    blocks_i = tl.cdiv(N, BLOCK_M)
    blocks_j = tl.cdiv(N, BLOCK_N)
    total_heads = B * H
    total_blocks = total_heads * blocks_i * blocks_j

    # exit if launch space larger than needed (safety)
    if pid >= total_blocks:
        return

    # map pid -> (head_index, block_i, block_j)
    head_idx = pid // (blocks_i * blocks_j)                 # 0 … B*H-1
    rem = pid % (blocks_i * blocks_j)
    block_i = rem // blocks_j
    block_j = rem % blocks_j

    batch_idx = head_idx // H
    head = head_idx % H

    # offsets within A & B
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this (batch, head)
    a_base = a_ptr + batch_idx * stride_b_a + head * stride_h_a
    b_base = b_ptr + batch_idx * stride_b_b + head * stride_h_b
    c_base = c_ptr + batch_idx * stride_b_c + head * stride_h_c

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # global k offset
        k_offset = k * BLOCK_K
        # A block: (i, k)
        a_ptrs = a_base + (offs_i[:, None] * stride_i_a) + ((k_offset + k_range)[None, :] * stride_k_a)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & ((k_offset + k_range) < N), other=0.0).to(tl.float32)

        # B block: (j, k) – we need Bᵀ, so we load B[j, k] as is
        b_ptrs = b_base + (offs_j[:, None] * stride_i_b) + ((k_offset + k_range)[None, :] * stride_k_b)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & ((k_offset + k_range) < N), other=0.0).to(tl.float32)

        # dot‑product over k
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # store result
    c_ptrs = c_base + (offs_i[:, None] * stride_i_c) + (offs_j[None, :] * stride_j_c)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing’’ TriMul operator.

    Args:
        data : tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N]        (float32/bool)
            weights      : dict of torch.Tensor containing all learned parameters
            config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # Linear expects weight of shape (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask if required
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B,N,N] → [B,N,N,1] for broadcasting
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ operation – batched matmul (fp16 inside kernel)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # Cast to fp16 for the kernel (much faster on Hopper)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)

    # strides (in element units) for the original layout (contiguous)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left_fp16.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right_fp16.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out_fp16.stride()

    # tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_heads = B * hidden_dim
    grid = (total_heads * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_b_l, stride_i_l, stride_k_l, stride_h_l,
        stride_b_r, stride_i_r, stride_k_r, stride_h_r,
        stride_b_o, stride_i_o, stride_j_o, stride_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    torch.cuda.synchronize()   # ensure kernel completion

    out = out_fp16.to(torch.float32)  # back to fp32

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final [B,N,N,dim]

    return out
