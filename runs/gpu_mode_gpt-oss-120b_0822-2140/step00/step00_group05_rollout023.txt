"""
TriMul "outgoing" implementation with a custom Triton kernel.

The expensive part of the forward pass is the
    out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]          (Eq. 1)

For every hidden channel `d` this is a matrix multiplication
    (left[..., d]) @ (right[..., d].T)
with shape (N, N) @ (N, N) → (N, N).  By folding the batch dimension `B`
and the hidden dimension `H` into a single composite batch `M = B*H` we
obtain a batched GEMM of shape (M, N, N) × (M, N, N)ᵀ → (M, N, N).

The kernel below implements this batched matmul in FP16 and accumulates
in FP32 for numerical stability.  All other operations (LayerNorm,
Linear, gating, final Linear) are performed with PyTorch.  The final
output is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                           # matrix sizes (rows, cols, reduction)
    stride_ab, stride_am, stride_ak,   # A: (batch, row, col) strides
    stride_bb, stride_bk, stride_bn,   # B: (batch, row(K), col) strides
    stride_cb, stride_cm, stride_cn,   # C: (batch, row, col) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ B.T  (A,B shape = (batch,M,N))."""
    pid_b = tl.program_id(0)        # batch index   (M = B*H)
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # --- block coordinates -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the first tile of A and B for this batch
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- iterate over K ----------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles (zero‑pad edges)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) + k * BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) + k * BLOCK_K < K)[:, None] & mask_n[None, :], other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result (convert back to FP16)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operator.

    Args:
        data: Tuple (x, mask, weights, config)
            x      – tensor [B, N, N, D] (float32)
            mask   – tensor [B, N, N]   (bool/int)  (may be all‑ones)
            weights – dict of model parameters (torch tensors)
            config  – dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, D = x.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension D
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy part: batched matmul via Triton (Eq. 1)
    # ------------------------------------------------------------------
    # cast to FP16 for the kernel (accumulator is FP32)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B, H, N, N) and pack batch & hidden dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (transpose K dimension)

    M_batch = B * H                                     # fused batch size

    A = left_perm.view(M_batch, N, N)   # (M, N, K)   K == N
    B_ = right_perm.view(M_batch, N, N) # (M, K, N)

    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_ab = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bb = B_.stride(0)
    stride_bk = B_.stride(1)
    stride_bn = B_.stride(2)

    stride_cb = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # block sizes – tuned for H200 (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (M_batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batch_matmul_kernel[grid](
        A.data_ptr(),
        B_.data_ptr(),
        C.data_ptr(),
        N, N, N,                     # M, N, K (all equal to sequence length)
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure final dtype is float32
    return out.to(torch.float32)
