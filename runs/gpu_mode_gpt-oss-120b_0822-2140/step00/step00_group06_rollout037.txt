"""
TriMul “outgoing” kernel (AlphaFold‑3)

The heavy part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right ∈ ℝ^{B×N×N×H}.  The kernel below computes this reduction
with a tiled GEMM per hidden channel H (H is a compile‑time constant).  The
rest of the module (LayerNorms, linear projections, gates, final Linear) is
implemented with PyTorch which is cheap compared to the O(N³·H) work.

The kernel works for any batch size ≤2, sequence length up to 1024 and hidden
dimension 128/384.  It is written for Triton 3.3.1 and targeted at NVIDIA
H200 (SM90).  All tensors are expected to be on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute the einsum for all hidden channels.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    out_ptr, left_ptr, right_ptr,                     # pointers
    B, N, H,                                          # sizes
    stride_ob, stride_oi, stride_oj, stride_od,       # out strides
    stride_lb, stride_li, stride_lk, stride_ld,       # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,       # right strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H_DIM:   tl.constexpr,                            # hidden dim (compile‑time)
):
    """   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   """
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # ------------------------------- tile indices & masking
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator: (BLOCK_I, BLOCK_J, H_DIM)
    acc = tl.zeros((BLOCK_I, BLOCK_J, H_DIM), dtype=tl.float32)

    # ------------------------------- reduction over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile (i, k, h)
        left_ptrs = left_ptr + \
            pid_b * stride_lb + \
            offs_i[:, None, None] * stride_li + \
            offs_k[None, :, None] * stride_lk + \
            tl.arange(0, H_DIM)[None, None, :] * stride_ld
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )   # shape (Bi, Bk, H)

        # ----- right tile (j, k, h)
        right_ptrs = right_ptr + \
            pid_b * stride_rb + \
            offs_j[:, None, None] * stride_rj + \
            offs_k[None, :, None] * stride_rk + \
            tl.arange(0, H_DIM)[None, None, :] * stride_rd
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )   # shape (Bj, Bk, H)

        # ----- per‑channel GEMM: (Bi,Bk) × (Bk,Bj)ᵀ → (Bi,Bj)
        for h in range(H_DIM):
            left_h  = left[:, :, h]          # (Bi, Bk)
            right_h = right[:, :, h]         # (Bj, Bk)
            # dot expects (M,K) @ (K,N); we transpose right_h
            acc[:, :, h] += tl.dot(left_h, tl.trans(right_h))

    # ------------------------------- store result
    out_ptrs = out_ptr + \
        pid_b * stride_ob + \
        offs_i[:, None, None] * stride_oi + \
        offs_j[None, :, None] * stride_oj + \
        tl.arange(0, H_DIM)[None, None, :] * stride_od
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with the heavy einsum
    implemented in Triton.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   torch.float32 on CUDA
        - mask         : [B, N, N]        (ignored when config["nomask"] is True)
        - weights      : dict of torch tensors (no bias except for LayerNorms)
        - config       : dict with keys
              "dim"         : input channel size
              "hidden_dim"  : projection size H
              "nomask"      : bool
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – shape B×N×N×H
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).type_as(x)           # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and gating of the projections
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel – the core O(N³·H) operation
    # ------------------------------------------------------------------
    # allocate output buffer for the intermediate hidden representation
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # contiguous tensors → simple stride layout
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides (in elements, not bytes)
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()

    # launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), B)

    _trimul_einsum_kernel[grid](
        out_ptr=out,
        left_ptr=left,
        right_ptr=right,
        B=B,
        N=N,
        H=H,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        H_DIM=H,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: final LayerNorm, gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
