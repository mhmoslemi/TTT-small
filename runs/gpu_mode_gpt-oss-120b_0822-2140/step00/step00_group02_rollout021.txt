"""
TriMul “outgoing” implementation (AlphaFold‑3) with a fused batched
matrix‑multiply kernel written in Triton.

The heavy part of the forward pass is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel *d* this is a matrix multiplication
    (N × N) @ (N × N)ᵀ
which we compute in a single Triton kernel that processes all
batch‑*hidden* blocks together (B·hidden matrices of size N×N).

All other operations (LayerNorm, linear projections, gating, final
linear) stay in PyTorch for simplicity.  The kernel works with
float‑32 tensors, returns a float‑32 result and supports the
various sequence lengths and hidden dimensions required by the test
suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply (A @ Bᵀ) for many batches at once
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes (M=N=seq_len, K=seq_len)
    stride_ab, stride_am, stride_ak,  # strides for A  (batch, M, K)
    stride_bb, stride_bm, stride_bk,  # strides for B  (batch, M, K)
    stride_cb, stride_cm, stride_cn,  # strides for C  (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_n = tl.program_id(0)          # block index in N dimension
    pid_m = tl.program_id(1)          # block index in M dimension
    pid_b = tl.program_id(2)          # batch index (B*hidden)

    # ------------------------------------------------------------------
    # Compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the start of the tile
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A and B (B is accessed transposed)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        # Dot product over the K dimension
        acc += tl.dot(a, b)

        # Increment pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write the result back to C
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32, on CUDA)
            - mask  : torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of parameter tensors (all float32, on CUDA)
            - config : {'dim': int, 'hidden_dim': int, ...}
    Returns:
        torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present (mask is broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication: out = left @ rightᵀ   (per hidden)
    #    Reshape to (B*hidden, N, N) to treat each hidden channel as a batch.
    # ------------------------------------------------------------------
    #   left  : [B, N, N, hidden] -> [B*hidden, N, N]
    #   right : [B, N, N, hidden] -> [B*hidden, N, N]
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_t.view(B * hidden, N, N)
    right_flat = right_t.view(B * hidden, N, N)

    out_flat = torch.empty_like(left_flat)

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # N‑tiles
        triton.cdiv(N, BLOCK_M),   # M‑tiles
        B * hidden,                # batch‑*‑hidden dimension
    )

    # Strides for the three‑dimensional tensors (batch, M, K) etc.
    a_batch, a_m, a_k = left_flat.stride()
    b_batch, b_m, b_k = right_flat.stride()
    c_batch, c_m, c_n = out_flat.stride()

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_m, a_k,
        b_batch, b_m, b_k,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
