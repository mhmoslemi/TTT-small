"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the 4‑D input   x ∈ (B,N,N,dim).
2. Linear projections → left, right ∈ (B,N,N,hidden_dim).
3. Optional pair‑wise mask (broadcast on the channel dim).
4. Gating (sigmoid) on the *original* x and element‑wise multiply
   left ← left * left_gate ,   right ← right * right_gate.
5. Core cubic operation:
        out_hidden[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for each hidden channel *h*.
   Implemented as a Triton kernel that tiles the (i,j) output plane and
   reduces over k.  The kernel works on FP16 data (cast from FP32) and
   accumulates in FP32 for numerical stability.
6. Layer‑norm over hidden_dim, gate with out_gate (sigmoid), and final
   linear projection back to ‘dim’.
7. Return the final tensor in float32.

The heavy O(N³·hidden_dim) work is carried out by the Triton kernel,
while the lightweight linear / norm / sigmoid work stays in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul for each hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,               # const * pointer to left tensor
    right_ptr,              # const * pointer to right tensor
    out_ptr,                # * pointer to output tensor
    B, N, H,                # problem sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,  # tile rows   (i dimension)
    BLOCK_N: tl.constexpr,  # tile cols   (j dimension)
    BLOCK_K: tl.constexpr,  # tile reduction over k
):
    # --------------------------------------------------------------
    # 1) identify which (batch, hidden) pair this program instance works on
    # --------------------------------------------------------------
    pid = tl.program_id(0)                     # linear id over (B * H)
    total_bh = B * H
    if pid >= total_bh:
        return
    b = pid // H                               # batch index
    h = pid % H                                # hidden channel index

    pid_i = tl.program_id(1)                   # tile along i (seq dimension)
    pid_j = tl.program_id(2)                   # tile along j (seq dimension)

    # --------------------------------------------------------------
    # 2) compute global offsets for the i‑ and j‑tiles
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) reduction loop over k
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[b, j, k, h]   (note the j index first)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_j[:, None] * stride_ri
            + offs_k[None, :] * stride_rk
            + h * stride_rh
        )
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # dot‑product accumulate: (M,K)·(K,N) → (M,N)
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # 4) write result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              input_tensor : [B, N, N, dim]   (torch.float32, cuda)
              mask         : [B, N, N]        (torch.float32 or bool, cuda)
              weights      : dict of torch tensors (see README)
              config       : dict with keys:
                               - dim (int)
                               - hidden_dim (int)

    Returns:
        output tensor : [B, N, N, dim] (torch.float32, cuda)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # helpers to fetch weight tensors (all are 1‑D or 2‑D as in the
    # reference PyTorch module)
    # ------------------------------------------------------------------
    def linear(x, w):
        # w shape: (out_features, in_features)
        return F.linear(x, w)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                 # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on channel dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core cubic operation via Triton
    # ------------------------------------------------------------------
    # Cast to FP16 for the kernel (works well on Hopper)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_hidden = torch.empty_like(left_fp16)         # (B,N,N,hidden)

    # strides of the (B,N,N,H) layout (contiguous)
    stride_lb, stride_li, stride_lk, stride_lh = left_fp16.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_hidden.stride()

    # tiling parameters – chosen to fit H200 SM resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden_dim, 1),          # program id 0  → (B * H)
        triton.cdiv(N, BLOCK_M),                # program id 1  → i‑tiles
        triton.cdiv(N, BLOCK_N),                # program id 2  → j‑tiles
    )

    trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # back to FP32 for the remaining layers
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                         # broadcast over hidden_dim
    out = linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
