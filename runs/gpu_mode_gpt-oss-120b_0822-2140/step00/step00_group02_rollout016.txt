"""
TriMul (outgoing) – fast Triton implementation
================================================

The heavy part of the TriMul forward pass is the three‑dimensional
einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix multiply `A @ B.T` for every
`(b,d)` pair where `A` and `B` are `N×N` matrices (the hidden dimension
is the “channel”).  The surrounding operations (LayerNorms, linear
projections, gating, final LayerNorm + linear) are cheap relative to the
`N³` work and are kept in PyTorch.

The kernel below computes the batched GEMM in FP16 with an FP32
accumulator and writes the result in FP16.  Afterwards the result is
cast back to FP32 and the remaining PyTorch post‑processing is applied.

The launch grid is
    (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B * hidden_dim)

so each program instance handles a tile `(i0:i0+BM, j0:j0+BN)` for a
specific `(batch, channel)` pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # tensor sizes
    N, hidden_dim,
    # strides (int64)
    stride_abl, stride_ai, stride_ak, stride_ad,   # A: (batch, i, k, d)
    stride_bbl, stride_bi, stride_bk, stride_bd,   # B: (batch, j, k, d)   (we will load B^T)
    stride_cbl, stride_ci, stride_cj, stride_cd,   # C: (batch, i, j, d)
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # tile over i
    pid_n = tl.program_id(1)               # tile over j
    pid_z = tl.program_id(2)               # combined batch * channel

    # ------------------------------------------------------------------
    # decode batch and channel from pid_z
    # ------------------------------------------------------------------
    batch_id = pid_z // hidden_dim
    chan_id  = pid_z %  hidden_dim

    # ------------------------------------------------------------------
    # coordinate ranges for the tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_max = N

    # masks for out‑of‑bounds (only needed for the last tile)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # compute base pointers for this (batch, channel)
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_id * stride_abl + chan_id * stride_ad
    b_base = b_ptr + batch_id * stride_bbl + chan_id * stride_bd
    c_base = c_ptr + batch_id * stride_cbl + chan_id * stride_cd

    # ------------------------------------------------------------------
    # allocate accumulator (FP32 for accuracy)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(k_max, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   -> shape (BM, BK)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B[j, k]   we need B^T, i.e. load B at (j, k) where j corresponds to
        # the column tile.  We therefore use the same stride pattern as A
        # but swap the role of i and j.
        b_ptrs = b_base + (offs_n[:, None] * stride_bi
                           + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # multiply‑accumulate
        acc += tl.dot(a, b)        # dot does FP16*FP16 → FP32 accumulation

    # ------------------------------------------------------------------
    # write result (FP16) – only for in‑bounds positions
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper for LayerNorm using the supplied weight/bias (no running stats)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]  (C = dim)
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    normalized = (x - mean) / torch.sqrt(var + eps)
    return normalized * weight + bias


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast forward pass for the outgoing TriMul operation.
    Arguments:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N]    (bool/float)
            - weights: dict of parameter tensors (float32)
            - config : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor, norm_w, norm_b)          # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in FP16 for speed
    # ------------------------------------------------------------------
    x16 = x.to(torch.float16)
    left = F.linear(x16, left_proj_w).to(torch.float16)    # (B,N,N,hidden)
    right = F.linear(x16, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Mask (optional)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w)).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(right.dtype)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w)).to(left.dtype)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # allocate output buffer (FP16)
    out = torch.empty_like(left)   # (B,N,N,hidden)  FP16

    # strides for the 4‑D tensors
    stride_abl, stride_ai, stride_ak, stride_ad = left.stride()
    stride_bbl, stride_bi, stride_bk, stride_bd = right.stride()
    stride_cbl, stride_ci, stride_cj, stride_cd = out.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim,                   # batch * channel
    )
    tri_mul_gemm_kernel[grid](
        a_ptr=left.data_ptr(),
        b_ptr=right.data_ptr(),
        c_ptr=out.data_ptr(),
        N=N,
        hidden_dim=hidden_dim,
        stride_abl=stride_abl,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_ad=stride_ad,
        stride_bbl=stride_bbl,
        stride_bi=stride_bi,
        stride_bk=stride_bk,
        stride_bd=stride_bd,
        stride_cbl=stride_cbl,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        stride_cd=stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # cast back to FP32 for the remaining operations
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = _layer_norm(out, to_out_norm_w, to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)               # (B,N,N,dim)

    return out
