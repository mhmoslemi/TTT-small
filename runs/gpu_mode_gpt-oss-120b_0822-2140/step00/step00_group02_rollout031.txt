"""
TriMul "outgoing" implementation with a fused Triton kernel.

The heavy N³ part    
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per head:
    out = left @ rightᵀ   (shape B×H, N, N)

The kernel computes this matmul for every (batch, hidden) pair,
while the surrounding LayerNorm / Linear / gating logic remains in
regular PyTorch (highly‑optimized and simple).  The final tensor is
float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   (per (batch, head) pair)
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,           # *f32   [B,N,N,H] (contiguous)
    right_ptr,          # *f32   same layout as left
    out_ptr,            # *f32   same layout as left
    N,                  # i32    sequence length
    H,                  # i32    hidden dimension (per‑batch)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul left @ rightᵀ where each head works on (N×N) matrices."""
    pid_m = tl.program_id(0)          # row block
    pid_n = tl.program_id(1)          # col block
    pid_head = tl.program_id(2)       # combined (batch, hidden)

    # ------------------------------------------------------------------
    # Decode batch and hidden indices from the flattened head dimension
    # ------------------------------------------------------------------
    b = pid_head // H                 # batch index
    h = pid_head % H                  # hidden index

    # Base pointer for this (b,h) matrix
    off = b * stride_left_b + h * stride_left_h
    left  = left_ptr  + off
    right = right_ptr + off
    out   = out_ptr   + off

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:   (M, K)  rows i, inner k
        left_idx = (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        l = tl.load(left + left_idx,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # rightᵀ: (K, N)  we need right[j,k] → transpose to (k,j)
        # Load as (K, N) so tl.dot can be used (M×K)·(K×N)
        right_idx = (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)
        r = tl.load(right + right_idx,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Accumulate dot product over K
        acc += tl.dot(l, r)

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    out_idx = (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(out + out_idx, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry point used by the autograder
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor [B, N, N] (float32 or bool) – may be all‑ones
            - weights: dict of tensors for all linear / norm parameters
            - config: dict with keys:
                * dim          – input channel dimension
                * hidden_dim   – hidden dimension (proj / gate width)
                * nomask (bool) – True ⇒ mask can be ignored
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections + gates (no bias, as in the reference)
    # ------------------------------------------------------------------
    # Shape helpers: torch.nn.functional.linear expects weight (out_features, in_features)
    left  = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_view = mask.unsqueeze(-1).to(x.dtype)           # [B,N,N,1]
        left = left * mask_view
        right = right * mask_view

    # ------------------------------------------------------------------
    # 4. Apply per‑head gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul – fused Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tile sizes (tuned for Hopper – 64 is a good trade‑off)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (rows, cols, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # Compute strides for the contiguous tensors
    # left / right layout: (B, N, N, H) -> strides (N*N*H, N*H, H, 1)
    s_b, s_i, s_k, s_h = left.stride()
    # out has the same layout
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Launch kernel
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        s_b,
        s_i,
        s_k,
        s_h,
        s_b,                 # right batch stride is identical
        s_i,                 # right "j" stride == left "i" stride (both = N*H)
        s_k,
        s_h,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: normalization, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # gating after norm
    out = F.linear(out, weights["to_out.weight"])   # final projection -> [B,N,N,dim]

    return out
