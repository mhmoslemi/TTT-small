"""
TriMul “outgoing” implementation with a fused Triton kernel for the heavy
N³ einsum.

Algorithm
---------

1. Layer‑norm the input ``x`` (shape [B,N,N,D]).
2. Linear projections (no bias) → ``left`` and ``right`` of shape
   [B,N,N,H] where H = hidden_dim.
3. (Optional) mask the two tensors.
4. Compute three sigmoid gates and apply them element‑wise.
5. **Fused kernel**: for each batch **b** and hidden channel **h**, compute

       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   which is a batched matrix‑multiply of the (i,k) matrix with the
   transpose of the (j,k) matrix.  The kernel tiles the i‑ and j‑dimensions
   (BLOCK_M×BLOCK_N) and iterates over k in BLOCK_K steps, accumulating
   in FP32.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   apply the final linear projection back to D.

Only the O(N³) part is executed on the GPU via Triton; the remaining
operations stay in PyTorch for simplicity.  The final tensor is returned
in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,                     # *const float *
    right_ptr,                    # *const float *
    out_ptr,                      # *float *
    B, N, H,                      # int scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute one (block_m × block_n) tile of the output for a given
    batch *b* and hidden *h*."""
    pid = tl.program_id(0)

    # ----- decode program id ------------------------------------------------
    total_i = tl.cdiv(N, BLOCK_M)          # tiles along i
    total_j = tl.cdiv(N, BLOCK_N)          # tiles along j
    total_h = H
    total_b = B

    # layout: b  -> h -> i_tile -> j_tile
    b = (pid // (total_h * total_i * total_j))
    tmp = pid % (total_h * total_i * total_j)
    h = (tmp // (total_i * total_j))
    tmp2 = tmp % (total_i * total_j)
    i_tile = tmp2 // total_j
    j_tile = tmp2 % total_j

    # ----- offsets ----------------------------------------------------------
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over k dimension -------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # left[i,k]  -> shape (M, K)
        left_ptrs = left_base \
            + offs_i[:, None] * stride_l_i \
            + cur_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[j,k] needs to be transposed to (K, N) for tl.dot
        # right_trans[k, j] = right[j, k]
        right_ptrs = right_base \
            + cur_k[:, None] * stride_r_k \
            + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiply (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)        # FP32 accumulation

    # ----- write back --------------------------------------------------------
    out_ptrs = out_base \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, D] (float32)
            mask   – torch.Tensor [B, N, N] (bool/float). May be all‑ones when
                     ``config['nomask']`` is True.
            weights – dict with the same keys as the reference PyTorch module.
            config  – dict containing at least ``dim`` and ``hidden_dim``.
    Returns:
        Tensor of shape [B, N, N, D] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(D,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask = torch.ones(B, N, N, device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)               # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton fused N³ product  (output shape [B,N,N,hidden])
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # allocate FP32 output

    # strides (in elements, not bytes) required by the kernel
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # kernel launch parameters – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_i = triton.cdiv(N, BLOCK_M)
    total_j = triton.cdiv(N, BLOCK_N)
    total_blocks = B * hidden * total_i * total_j
    grid = (total_blocks,)

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_l_b=s_l[0], stride_l_i=s_l[1], stride_l_k=s_l[2], stride_l_h=s_l[3],
        stride_r_b=s_r[0], stride_r_j=s_r[1], stride_r_k=s_r[2], stride_r_h=s_r[3],
        stride_o_b=s_o[0], stride_o_i=s_o[1], stride_o_j=s_o[2], stride_o_h=s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate            # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # back to D

    return out.to(torch.float32)
