"""
TriMul “outgoing’’ implementation.

The heavy part of the module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]      (d = hidden_dim)

For every batch `b` and hidden channel `d` this is exactly a matrix‑multiply
   (i × k)  ×  (j × k)ᵀ  → (i × j).

The kernel below computes this reduction in FP32 (inputs are FP16 for bandwidth)
using a classic tiling algorithm (BLOCK_M × BLOCK_N tiles,
reduced over BLOCK_K).  The remaining LayerNorm, linear projections and gating
operations are performed with PyTorch, which is negligible compared to the
O(N³) reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: tiled matmul over the `k` dimension for every (b,d) pair
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # FP16, shape [B, N, N, D]
    right_ptr,         # FP16, shape [B, N, N, D]
    out_ptr,           # FP32, shape [B, N, N, D]
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --- program IDs ----------------------------------------------------
    pid_i = tl.program_id(0)                     # block row  (i)
    pid_j = tl.program_id(1)                     # block col  (j)
    pid_z = tl.program_id(2)                     # combined batch * channel

    # decode batch and hidden channel from pid_z
    d = pid_z % D
    b = pid_z // D

    # start offsets for this block
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    k_offset = 0

    # offsets inside the block
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer base for this (b,d)
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- k‑reduction loop ------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k_blocks):
        # left tile: (BLOCK_M, BLOCK_K)
        left_tile = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + (k_offset + offs_k)[None, :] * stride_left_k,
            mask=mask_i[:, None] & ((k_offset + offs_k)[None, :] < N),
            other=0.0,
        ).to(tl.float32)

        # right tile: (BLOCK_K, BLOCK_N)  – note the order so we can use tl.dot
        right_tile = tl.load(
            right_base
            + (k_offset + offs_k)[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j,
            mask=((k_offset + offs_k)[:, None] < N) & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate dot products over the K dimension
        acc += tl.dot(left_tile, right_tile)

        # advance k
        k_offset += BLOCK_K

    # store result (still FP32)
    out_ptrs = out_base \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing’’ TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]       (bool/float, may be all‑ones)
            - weights: dict with the LayerNorm and Linear parameters
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → left / right
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise modulation
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the reduction kernel (good bandwidth on H200)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer in FP32
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        bs * hidden_dim,                  # (b,d) pairs
    )

    # strides of the tensors
    s_lb, s_li, s_lk, s_ld = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rd = right_fp16.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        bs,
        N,
        hidden_dim,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # H200 can profit from a few warps per block
    )

    # ------------------------------------------------------------------
    # 7) post‑processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final projection back to original dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
