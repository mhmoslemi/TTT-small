"""
TriMul “outgoing” kernel (forward only).

The heavy part of the module is the three‑tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
We fuse this reduction into a Triton kernel that iterates over tiles of
(i, j, d) and accumulates over k.  All other operations (layer‑norms,
linear projections, gates, final norm/linear) are performed with
torch.nn.functional for simplicity.

The kernel works for arbitrary batch size (1‑2), sequence length up to
1024 and hidden dimensions that are multiples of the block size (we
mask the remainder).  The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,            # [B, N, N, H] left  (i,k,d)
    right_ptr,           # [B, N, N, H] right (j,k,d)
    out_ptr,             # [B, N, N, H] output (i,j,d)

    # sizes
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,

    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld,   # left
    stride_rb, stride_rj, stride_rk, stride_rd,   # right
    stride_ob, stride_oi, stride_oj, stride_od,   # out

    # tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute a tiled outer‑product reduction over the k dimension.
    Each program instance works on a tile (i_tile, j_tile, d_tile) of
    shape (BLOCK_I, BLOCK_J, BLOCK_D) for a single batch element.
    """
    pid0 = tl.program_id(0)          # combined batch * i_tile * j_tile
    pid1 = tl.program_id(1)          # d_tile

    # ------------------------------------------------------------------
    # 0‑D grid layout
    # ------------------------------------------------------------------
    n_i_tiles = tl.cdiv(N, BLOCK_I)
    n_j_tiles = tl.cdiv(N, BLOCK_J)
    n_d_tiles = tl.cdiv(H, BLOCK_D)

    # batch index
    batch = pid0 // (n_i_tiles * n_j_tiles)
    tile_ij = pid0 % (n_i_tiles * n_j_tiles)
    i_tile = tile_ij // n_j_tiles
    j_tile = tile_ij % n_j_tiles
    d_tile = pid1

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    i_off = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    d_off = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)

    i_mask = i_off < N
    j_mask = j_off < N
    d_mask = d_off < H

    # Accumulator (float32 for precision)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # --------------------------------------------------------------
        # Load left[i, k, d]   shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # --------------------------------------------------------------
        left_ptrs = (left_ptr
                     + batch * stride_lb
                     + i_off[:, None, None] * stride_li
                     + k_off[None, :, None] * stride_lk
                     + d_off[None, None, :] * stride_ld)

        left = tl.load(left_ptrs,
                       mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                       other=0.0)

        # --------------------------------------------------------------
        # Load right[j, k, d]  shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # --------------------------------------------------------------
        right_ptrs = (right_ptr
                      + batch * stride_rb
                      + j_off[:, None, None] * stride_rj
                      + k_off[None, :, None] * stride_rk
                      + d_off[None, None, :] * stride_rd)

        right = tl.load(right_ptrs,
                        mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                        other=0.0)

        # --------------------------------------------------------------
        # Outer‑product over the k dimension:
        #   for each k: acc[i,j,d] += left[i,k,d] * right[j,k,d]
        # --------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            k_valid = (k_off[kk] < N)
            if not k_valid:
                continue

            l = left[:, kk, :]          # (BLOCK_I, BLOCK_D)
            r = right[:, kk, :]         # (BLOCK_J, BLOCK_D)

            # broadcast to (i, j, d)
            prod = l[:, None, :] * r[None, :, :]   # (BLOCK_I, BLOCK_J, BLOCK_D)
            acc += prod

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + batch * stride_ob
                + i_off[:, None, None] * stride_oi
                + j_off[None, :, None] * stride_oj
                + d_off[None, None, :] * stride_od)

    tl.store(out_ptrs,
             acc.to(tl.float16),                     # store as fp16 for bandwidth
             mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])

# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton kernel
    for the N³ reduction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm (no bias => bias may be present)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        return F.layer_norm(x,
                            normalized_shape=(x.shape[-1],),
                            weight=weight,
                            bias=bias,
                            eps=1e-5)

    # ------------------------------------------------------------------
    # Load model parameters (all are stored as float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor, norm_w, norm_b)          # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (..., dim) @ (hidden_dim, dim).T -> (..., hidden_dim)
    left = F.linear(x, left_proj_w)                      # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)                    # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_ = torch.ones(B, N, N, 1, device=device, dtype=left.dtype)
    else:
        mask_ = mask.unsqueeze(-1).to(left.dtype)        # [B,N,N,1]

    left = left * mask_
    right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the kernel (still keep a fp32 copy for later)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)               # placeholder for result

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    # Tile sizes (tuned for Hopper, can be adjusted)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32

    n_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    n_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    n_d_tiles = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    # 2‑D grid: (batch * i_tiles * j_tiles, d_tiles)
    grid = (B * n_i_tiles * n_j_tiles, n_d_tiles)

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=4,   # Hopper works well with 4 warps per block
    )

    # ------------------------------------------------------------------
    # 7. Convert back to fp32 for the remaining ops
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # 8. Output layer‑norm, gate and final projection
    out = layer_norm(out, to_out_norm_w, to_out_norm_b)          # [B,N,N,hidden]
    out = out * out_gate                                        # broadcast over hidden
    out = F.linear(out, to_out_w)                               # [B,N,N,dim]

    return out
