@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_batch = tl.program_id(2)  # flattened matrix index

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # pointers for given batch
    a_batch_ptr = a_ptr + pid_batch * stride_am  # move to start of matrix
    b_batch_ptr = b_ptr + pid_batch * stride_bm
    c_batch_ptr = c_ptr + pid_batch * stride_cm

    # load a block
    a_ptrs = a_batch_ptr + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_j[None, :] * stride_bi + offs_k[:, None] * stride_bk  # note switched

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # a,b are float16
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing” implementation.

The heavy O(N³) part (the einsum) is reduced to a batch of
matrix‑multiplications:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is exactly (left @ rightᵀ) for every (b,h) pair.
We pack the (batch, hidden) dimensions into a single “matrix” dimension
M = B * hidden_dim and launch a Triton kernel that computes a
block‑wise GEMM for each of the M independent matrices.

All other operations (layer‑norms, linear projections, gating) are kept in
native PyTorch for simplicity and numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [M, N, K] , C : [M, N, N])
# -------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    a_ptr,                # const * half
    b_ptr,                # const * half
    c_ptr,                # * half
    M, N, K,              # matrix sizes (int32)
    stride_am, stride_ai, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bi, stride_bk,   # strides for B (M,N,K)
    stride_cm, stride_ci, stride_ck,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row of C
    pid_n = tl.program_id(1)          # block column of C
    pid_mat = tl.program_id(2)        # which (batch, hidden) matrix

    # -------------------- offsets --------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # move to the correct matrix (M dimension)
    a_batch_ptr = a_ptr + pid_mat * stride_am
    b_batch_ptr = b_ptr + pid_mat * stride_bm
    c_batch_ptr = c_ptr + pid_mat * stride_cm

    # pointers for the current tile
    a_ptrs = a_batch_ptr + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    # B is used transposed, i.e. we need B[j,k] while iterating over k → we
    # simply swap the order of the offsets.
    b_ptrs = b_batch_ptr + offs_j[None, :] * stride_bi + offs_k[:, None] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- loop over K (reduction) -------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # a,b are fp16; dot returns fp32 accumulation
        acc += tl.dot(a, b)

        # advance K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store C (fp16)
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Python entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor : [B, N, N, dim]    (float32)
        * mask         : [B, N, N]        (float32 or bool)
        * weights      : dict of pre‑loaded parameters
        * config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = H

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # shape [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in fp32 for stability
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Mask (if present)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul: out = left @ rightᵀ  (batched GEMM)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                       # B = batch, H = hidden_dim

    # reshape to (M, N, K) where M = B*H, K = N
    left_reshaped = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_reshaped = right.permute(0, 3, 1, 2).contiguous()
    M = B * H
    left_flat = left_reshaped.view(M, N, N)   # [M, N, N]  (fp32)
    right_flat = right_reshaped.view(M, N, N)

    # cast to fp16 for the kernel (more bandwidth, still accurate enough)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)    # will hold result

    # strides (in elements) for the flattened tensors
    stride_am = left_fp16.stride(0)   # M‑stride = N*N
    stride_ai = left_fp16.stride(1)   # row‑stride = N
    stride_ak = left_fp16.stride(2)   # col‑stride = 1

    stride_bm = right_fp16.stride(0)
    stride_bi = right_fp16.stride(1)
    stride_bk = right_fp16.stride(2)

    stride_cm = out_fp16.stride(0)
    stride_ci = out_fp16.stride(1)
    stride_ck = out_fp16.stride(2)

    # Tiling parameters – tuned for Hopper (sm90) L2 cache size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        M,                            # independent matrices
    )

    _trimul_matmul_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_fp16,
        M=M,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bm=stride_bm,
        stride_bi=stride_bi,
        stride_bk=stride_bk,
        stride_cm=stride_cm,
        stride_ci=stride_ci,
        stride_ck=stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_fp16.to(torch.float32)                     # back to fp32
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)       # [B,N,N,H]

    # -----------------------------------------------------------------
    # 6. Final LayerNorm + gating + output projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])        # [B,N,N,dim]

    return out
