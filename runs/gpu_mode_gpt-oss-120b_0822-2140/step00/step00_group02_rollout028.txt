    input_tensor, mask, weights, config = data
    B,N,N2,dim = input_tensor.shape, ensure N2==N.
    dim = config['dim']; hidden_dim = config['hidden_dim']
    # LayerNorm
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True)+1e-5)
    # Apply learned weight/bias
    x = x * weights['norm.weight'] + weights['norm.bias']

    # Linear projections
    # using einsum: x @ weight.T
    # shape weight (hidden_dim, dim)
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # move to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for batched mul
    B, N, _, H = left.shape
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)
    stride_b_n = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

"""
TriMul “outgoing” implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,D) using the supplied weight/bias.
2. Linear projections → left / right (B,N,N,H) and gated element‑wise
   multiplication with mask (if present) and learned sigmoid gates.
3. The expensive part is the einsum
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   which is a batched matrix‑multiply per hidden dimension:
        out_h = left_h @ right_hᵀ   (h = 0 … H‑1)
   We fuse this with Triton:
   * reshape left/right to (B*H, N, N)  (batch dimension = B*H)
   * launch a tiled GEMM kernel that computes C = A @ Bᵀ in fp16,
     accumulating in fp32 for numerical safety.
4. Reshape the result back to (B,N,N,H) (fp32).
5. Apply a second LayerNorm, the output gate and the final linear
   projection to obtain the (B,N,N,D) output.

The kernel works for arbitrary batch‑size (1‑2), sequence lengths
(≤ 1024) and hidden dimensions (128).  Block sizes are tuned for
Hopper‑class GPUs (NVIDIA H200) but remain correct for any size.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix sizes (all = seq_len)
    stride_Ab, stride_Am, stride_Ak,   # A strides: batch, row(i), col(k)
    stride_Bb, stride_Bk, stride_Bn,   # B strides: batch, row(k), col(j)
    stride_Cb, stride_Cm, stride_Cn,   # C strides: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ   (batch‑wise)   A,B,C are fp16, accumulator fp32."""
    pid_b = tl.program_id(2)          # batch index (B*H)
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers for the first K‑tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + tl.arange(0, BLOCK_K)[:, None] * stride_Bk
        + offs_n[None, :] * stride_Bn
    )

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑dimension
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_iter * BLOCK_K

        # mask for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + k_offset < K)
        b_mask = (tl.arange(0, BLOCK_K)[:, None] + k_offset < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # dot‑product over the K‑dimension (produces a scalar per (m,n) pair)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write back (cast to fp16)
    c = acc.to(tl.float16)
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, D]   (float32)
            - mask         : torch.Tensor[B, N, N]       (bool/int, may be None)
            - weights      : dict of torch.Tensors containing all learned parameters
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ----------- 1. Input LayerNorm ---------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # (B,N,N,D)  float32

    # ----------- 2. Linear projections ------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ----------- 3. Optional mask ----------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)       # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ----------- 4. Gating ------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------- 5. Core einsum via Triton -------------------------------
    # Cast to fp16 for the kernel (H200 fp16 performance is excellent)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to batch = B*H  : (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_Ab = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)   # K‑dimension of B (the “k” we sum over)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Tile sizes – empirically good for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # along M (i)
        triton.cdiv(N, BLOCK_N),   # along N (j)
        B * H,                     # batch dimension (B * hidden_dim)
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout and datatype (float32)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)   # (B,N,N,H)

    # ----------- 6. Output LayerNorm ------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ----------- 7. Output gate & final linear ---------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
