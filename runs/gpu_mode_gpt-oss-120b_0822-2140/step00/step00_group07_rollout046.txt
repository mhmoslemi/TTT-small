"""
TriMul “outgoing” implementation.
The expensive part is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiply per hidden channel:
    left_d  : (N,N)  = left[...,d]
    right_d : (N,N)  = right[...,d]
    out_d   = left_d @ right_d.T

We fuse this N³ work with a Triton GEMM kernel.  All lightweight
operations (LayerNorm, linear projections, gates, final LayerNorm & linear)
stay in PyTorch (torch.nn.functional) because they are bandwidth bound and
very fast on Hopper.  The kernel works for any batch size B∈{1,2},
any sequence length N (≤1024) and any hidden dimension H (128 or 384).

The kernel launches a 3‑D grid:
    (M‑blocks, N‑blocks, B*H)  where each (b,h) pair is an independent
matrix‑multiply of size (N×N)·(N×N).  FP16 is used inside the kernel and the
result is cast back to FP32 for the remaining PyTorch pipeline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent batches.
# ------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dims (all int32)
    stride_am, stride_ak, stride_ab,    # A strides: (row, col, batch)
    stride_bk, stride_bn, stride_bb,    # B strides: (row, col, batch)
    stride_cm, stride_cn, stride_cb,    # C strides: (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ Bᵀ for many independent batches.
        A : (batch, M, K)   B : (batch, N, K)   C : (batch, M, N)
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    batch_id = tl.program_id(2)       # which (b, hidden) pair

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for edge handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # compute base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    # pointers to the start of the block
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_n[None, :] * stride_bn + offs_k[:, None] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        # dot: (M,K)·(K,N) -> (M,N)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ------------------------------------------------------------------
# Helper to launch the batched GEMM
# ------------------------------------------------------------------
def batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A : (batch, M, K)   FP16
    B : (batch, N, K)   FP16
    returns C : (batch, M, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, _ = B.shape

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements (not bytes) for each dimension
    stride_am, stride_ak, stride_ab = A.stride()
    stride_bk, stride_bn, stride_bb = B.stride()
    stride_cm, stride_cn, stride_cb = C.stride()

    # reasonable block sizes for Hopper (empirically good)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # enough for FP16 work‑efficient kernels
    )
    return C

# ------------------------------------------------------------------
# Main entry point -------------------------------------------------
# ------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass with a fused Triton GEMM for the
    N³ contraction.
    data = (input, mask, weights, config)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm over the last dimension (dim) ----------
    # x_norm = (x - μ) / √(σ²+ε) * w + b
    x = input_tensor
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * w_norm + b_norm

    # ---------- Linear projections (no bias) ----------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)           # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- Gates (sigmoid) ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Convert to FP16 for the heavy contraction ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N) and fuse B*hidden as the batch dimension
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    left_reshape = left_perm.view(B * hidden_dim, N, N)     # (B*H, M, K)
    right_reshape = right_perm.view(B * hidden_dim, N, N)   # (B*H, N, K)

    # Triton batched GEMM: C = A @ Bᵀ  (shape (B*H, N, N))
    out_fp16 = batched_gemm_fp16(left_reshape, right_reshape)

    # Restore original layout: (B, N, N, H)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)
    out = out.to(torch.float32)       # back to FP32 for the remaining ops

    # ---------- Final LayerNorm over hidden dimension ----------
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) * torch.rsqrt(var_o + eps)
    out = out * w_outnorm + b_outnorm

    # ---------- Multiply by out_gate ----------
    out = out * out_gate

    # ---------- Final linear projection back to dim ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
