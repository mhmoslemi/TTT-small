"""
TriMul "outgoing" implementation (forward only).

Algorithm
---------
1. Layer‑norm + linear projections (left/right) and three sigmoid gates are done
   with native PyTorch – they are cheap compared to the N³ contraction.
2. The contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply for each hidden channel *d*:
        out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].transpose(-2,-1)
   We fuse the batch and hidden dimensions (B·H matrices of size N×N) and
   compute the mat‑mul with a custom Triton kernel.  The kernel works in
   FP16 for speed and casts the result back to FP32.
3. The resulting tensor is reshaped, Layer‑norm‑ed and linearly projected
   back to the original channel dimension (dim).

The kernel is written for the Hopper/SM‑90 architecture (H200) and uses
static block sizes.  It is launched with a 3‑D grid: (block_m, block_n,
batch_total) where *batch_total = B * hidden_dim*.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (FP16 → FP32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ak,              # A strides: (batch, M, K)
    stride_bk, stride_bn,              # B strides: (batch, K, N)
    stride_cm, stride_cn,              # C strides: (batch, M, N)
    batch_idx: tl.constexpr,           # which batch this block works on
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C[batch_idx] = A[batch_idx] @ B[batch_idx]"""
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col

    # offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (in case M/N not multiple of block size)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the batch
    a_batch = a_ptr + batch_idx * stride_am * M * K
    b_batch = b_ptr + batch_idx * stride_bk * K * N
    c_batch = c_ptr + batch_idx * stride_cm * M * N

    # pointer arithmetic for the tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am
                        + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                        + offs_n[None, :] * stride_bn)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a tile of A and B in FP16, promote to FP32
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (tl.arange(0, BLOCK_K)[:, None] < K - k * BLOCK_K), other=0.0)
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side helper for the batched mat‑mul
# ----------------------------------------------------------------------
def batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : (B*H, N, N)   FP16
    b : (B*H, N, N)   FP16   (already transposed on the second dim)
    returns c : (B*H, N, N) FP16
    """
    assert a.is_contiguous() and b.is_contiguous()
    Btotal, N, _ = a.shape
    K = N
    dtype = a.dtype
    # output allocation
    c = torch.empty_like(a)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        Btotal,
    )

    # strides in element units
    stride_am, stride_ak = a.stride(0), a.stride(2)   # (batch, M, K)
    stride_bk, stride_bn = b.stride(0), b.stride(2)   # (batch, K, N)
    stride_cm, stride_cn = c.stride(0), c.stride(2)   # (batch, M, N)

    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=N,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_idx=tl.constexpr,   # dummy – will be specialised per block in kernel
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim]   (float32)
            mask   – torch.Tensor [B, N, N]        (float32 or bool)
            weights – dict of parameter tensors (float32)
            config  – dict with keys: dim, hidden_dim
    Returns:
        out – torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    # x_norm = (x - mean) / sqrt(var + eps) * w + b
    eps = 1e-5
    x = input_tensor
    w = weights["norm.weight"]
    b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (x - mean) * torch.rsqrt(var + eps)
    x = x * w + b

    # ---------- Linear projections ----------
    # use torch.nn.functional.linear which does: out = x @ weight.T
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Contraction (batched mat‑mul) ----------
    # right needs to be transposed on its last two axes (j,k) -> (k,j)
    right_t = right.permute(0, 2, 1, 3)          # [B, N, N, H]

    # fuse batch and hidden dims -> (B*H, N, N)
    left_flat = left.reshape(B * hidden, N, N).contiguous()
    right_flat = right_t.reshape(B * hidden, N, N).contiguous()

    # cast to FP16 for the heavy compute
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Triton batched mat‑mul (FP16 → FP16)
    out_fp16 = batched_matmul(left_fp16, right_fp16)

    # reshape back to [B, N, N, H] and promote to FP32
    out = out_fp16.to(torch.float32).reshape(B, N, N, hidden)

    # ---------- Output LayerNorm ----------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out = (out - mean_o) * torch.rsqrt(var_o + eps)
    out = out * w_out_norm + b_out_norm

    # ---------- Gate & final linear ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
