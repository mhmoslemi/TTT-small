"""
TriMul “outgoing” implementation (forward only).

The heavy part of the module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which for each hidden dimension *d* is a batched matrix‑multiply
    out_d = left_d @ right_d.T   (shape B×N×N).

We implement this matrix multiply with a custom Triton kernel
(`bmm_kernel`).  All the surrounding LayerNorms, Linear projections,
and gating are left in PyTorch (they are cheap compared with the
N³‑like contraction).  The kernel works for arbitrary batch size *B*,
sequence length *N* and hidden dimension *D_h* (here called `hidden_dim`),
and returns the result in float16 which is later cast back to float32
so the final output matches the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply C = A @ Bᵀ
# A : [B, M, K]   (left  slice)
# B : [B, N, K]   (right slice, NOT transposed – the kernel transposes it)
# C : [B, M, N]   (output slice)
# ----------------------------------------------------------------------
@triton.jit
def bmm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, M, N, K,                        # problem sizes
    stride_ab, stride_am, stride_ak,   # strides for A
    stride_bb, stride_bn, stride_bk,   # strides for B
    stride_cb, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch dimension
    pid_m = tl.program_id(1)   # output M block
    pid_n = tl.program_id(2)   # output N block

    # ------------------------------------------------------------------
    # Block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current batch
    a_batch_ptr = a_ptr + pid_b * stride_ab
    b_batch_ptr = b_ptr + pid_b * stride_bb
    c_batch_ptr = c_ptr + pid_b * stride_cb

    # ------------------------------------------------------------------
    # Initialize accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # tile offsets in K
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                 + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (note we need Bᵀ, so we load as [BLOCK_N, BLOCK_K])
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bn
                                 + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Accumulate a @ bᵀ  (b is already row‑wise in k)
        acc += tl.dot(a, b)   # (M×K)·(K×N) → (M×N)

        # advance pointers to next K‑tile (automatic via offsets)
    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                             + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host function
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            input  – Tensor[B, N, N, dim]   (float32)
            mask   – Tensor[B, N, N]        (float32 or bool)
            weights – dict with all linear / layer‑norm parameters
            config  – dict with keys "dim", "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hd]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (einsum over k)
    #    We work in float16 inside the kernel for speed.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer for the contracted tensor (float16)
    out_fp16 = torch.empty((B, N, N, hidden_dim),
                           dtype=torch.float16,
                           device=left.device)

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # Strides (in elements) for the three tensors
    # shape: [B, N, N, H] → (B, i, k, d)
    stride_b, stride_i, stride_k, stride_d = left_fp16.stride()
    # For the right tensor we need the same layout (B, j, k, d)
    stride_b_r, stride_j, stride_k_r, stride_d_r = right_fp16.stride()

    # For each hidden dimension we launch the kernel separately.
    # This keeps the kernel simple (2‑D matrix multiply).
    for hd in range(hidden_dim):
        a_ptr = left_fp16[:, :, :, hd].contiguous()
        b_ptr = right_fp16[:, :, :, hd].contiguous()
        c_ptr = out_fp16[:, :, :, hd]

        # Compute strides in elements (not bytes) for the kernel
        a_batch_stride = a_ptr.stride(0)
        a_i_stride = a_ptr.stride(1)
        a_k_stride = a_ptr.stride(2)

        b_batch_stride = b_ptr.stride(0)
        b_j_stride = b_ptr.stride(1)
        b_k_stride = b_ptr.stride(2)

        c_batch_stride = c_ptr.stride(0)
        c_i_stride = c_ptr.stride(1)
        c_j_stride = c_ptr.stride(2)

        bmm_kernel[grid](
            a_ptr,
            b_ptr,
            c_ptr,
            B, N, N, N,                         # B, M, N, K  (K == N)
            a_batch_stride, a_i_stride, a_k_stride,
            b_batch_stride, b_j_stride, b_k_stride,
            c_batch_stride, c_i_stride, c_j_stride,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=4,
        )
    # Cast back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
