"""
TriMul “outgoing’’ Triton implementation.

The heavy part of TriMul is the 3‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply for each hidden channel:
    left  : (B, H, N, N)   (i,k)
    right : (B, H, N, N)   (j,k)  →  rightᵀ : (B, H, N, N) (k,j)
    out   = left @ rightᵀ      → (B, H, N, N)

We fuse this multiplication in a Triton kernel.  All other operations
(LayerNorm, Linear projections, gating, final LayerNorm and Linear) are
performed with PyTorch – they are cheap compared with the O(B·H·N³) work
of the contraction.  The kernel works in FP16 for speed and the result
is cast back to FP32 before the remaining PyTorch layers.

The kernel is launched with a 3‑D grid:
    (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
where each program handles a tile of the output matrix (M×N) for a
single (batch, hidden) pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel (batched, per‑hidden‑channel)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                # LEFT  (B,H,N,N)  – row‑major (M,K)
    b_ptr,                # RIGHT (B,H,N,N)  – row‑major (K,N)  (already transposed)
    c_ptr,                # OUT   (B,H,N,N)  – row‑major (M,N)
    M, N, K,              # matrix dimensions (all equal to seq_len)
    stride_ah, stride_am, stride_ak,   # a: batch‑hidden, row, col strides
    stride_bh, stride_bk, stride_bn,   # b: batch‑hidden, row(K), col(N) strides
    stride_ch, stride_cm, stride_cn,   # c: batch‑hidden, row, col strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch*hidden index

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds (when N is not a multiple of BLOCK_*)
    m_mask = offs_m < M
    n_mask = offs_n < N

    # compute the flat offset for the (batch, hidden) pair
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # pointers to the start of the tile for A and B
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a and b tiles (cast to fp16 then to fp32 for accumulation)
        a = tl.load(a_ptrs, mask=m_mask[:, None], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=n_mask[None, :], other=0.0).to(tl.float32)

        # multiply‑accumulate
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=m_mask[:, None] & n_mask[None, :])


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, H, N, N]  (float16)
    returns:    [B, H, N, N]  (float16)
    """
    assert left.shape == right.shape
    B, H, N, _ = left.shape
    M = N
    K = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    # strides for contiguous tensors (after permute+contiguous)
    stride_ah = left.stride(0)   # B*H*N*N stride between (batch,hidden) pairs
    stride_am = left.stride(2)   # stride to next row (M dimension)
    stride_ak = left.stride(3)   # stride to next col (K dimension)

    stride_bh = right.stride(0)
    stride_bk = right.stride(2)  # K dimension stride
    stride_bn = right.stride(3)  # N dimension stride

    stride_ch = left.stride(0)   # output has same layout as left/right
    stride_cm = left.stride(2)
    stride_cn = left.stride(3)

    batched_gemm_kernel[grid](
        left, right, left,                 # we can reuse left’s storage as output
        M, N, K,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return left  # now holds the result


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing’’ TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32, on CUDA)
            - mask         : [B, N, N]        (float32 or bool, on CUDA)
            - weights      : dict of torch tensors
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]

    left  = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask absent when config["nomask"] is True)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM: out = left @ rightᵀ per hidden channel
    #    Cast to fp16 for the kernel, result stays fp16.
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_fp16 = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()

    # Compute batched GEMM (result stored in left_fp16 to avoid extra allocation)
    out_fp16 = batched_gemm(left_fp16, right_fp16)

    # Back to [B,N,N,hidden_dim] layout and float32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).to(torch.float32)   # B,N,N,hidden_dim

    # ------------------------------------------------------------------
    # 6. Post‑projection LayerNorm and gating
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # [B,N,N,dim]

    return out
