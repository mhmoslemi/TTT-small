"""
TriMul “outgoing” implementation.

The heavy part of the module is the ternary matrix multiply
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]      (d = hidden_dim)

For each hidden dimension d this is a regular matrix multiplication
L_d (N×N) @ R_dᵀ (N×N).  We fuse all hidden dimensions by reshaping the
tensor to (B*hidden_dim, N, N) and launch a tiled GEMM Triton kernel that
computes the product L @ Rᵀ.  All other operations (LayerNorms, linear
projections, sigmoids, masking) are done with native PyTorch kernels,
which are already highly‑optimized for the H200.

The kernel follows the classic “persistent‑workspace” matmul pattern:
* block‑wise loads of L and the *transposed* R tiles,
* an accumulation loop over the K‑dimension,
* final store with optional masking for edge tiles.

The final result is post‑processed with a second LayerNorm, an output gate
and a linear projection, reproducing the reference PyTorch behavior.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM L @ Rᵀ  (L and R have shape [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    L_ptr, R_ptr, O_ptr,                # pointers
    N,                                  # matrix size (square)
    stride_lm, stride_ln, stride_lk,    # strides for L (batch, row, col)
    stride_rm, stride_rn, stride_rk,    # strides for R (batch, row, col)
    stride_om, stride_on, stride_ok,    # strides for O (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute O = L @ Rᵀ for each batch slice.
    Grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), batch)
    """
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col
    pid_b = tl.program_id(2)      # batch index  (b * hidden_dim)

    # ------------------------------
    # Compute tile offsets
    # ------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N   # used when loading tiles

    # Pointers for the first tile of L and Rᵀ
    # L[batch, m, k]
    L_tile_ptr = L_ptr + pid_b * stride_lm + offs_m[:, None] * stride_ln + offs_k[None, :] * stride_lk
    # Rᵀ is accessed as R[batch, n, k] because we need Rᵀ[m, n] = R[n, m]
    R_tile_ptr = R_ptr + pid_b * stride_rm + offs_n[:, None] * stride_rn + offs_k[None, :] * stride_rk

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------
    # Loop over K dimension
    # ------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        # Load tiles (apply K‑mask)
        L_tile = tl.load(L_tile_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        R_tile = tl.load(R_tile_ptr, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Dot‑product accumulation
        acc += tl.dot(L_tile, R_tile)

        # Increment pointers to next K‑tile
        L_tile_ptr += BLOCK_K * stride_lk
        R_tile_ptr += BLOCK_K * stride_rk

    # ------------------------------
    # Write back
    # ------------------------------
    O_tile_ptr = O_ptr + pid_b * stride_om + offs_m[:, None] * stride_on + offs_n[None, :] * stride_ok
    tl.store(O_tile_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused linear + (optional) mask + sigmoid gate
# ----------------------------------------------------------------------
def _linear_gate(x, weight, bias=None, mask=None, activation=True):
    """
    Linear projection w/ optional element‑wise mask and sigmoid gate.
    Returns tensor of shape x.shape[:-1] + (out_dim,).
    """
    proj = F.linear(x, weight, bias)            # [..., out_dim]
    if mask is not None:
        proj = proj * mask
    if activation:
        proj = torch.sigmoid(proj)
    return proj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]       (float32 or bool)
            - weights: dict of parameter tensors
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # LayerNorm (first)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # Linear projections + gates (PyTorch, fast)
    # --------------------------------------------------------------
    # mask shape -> [B, N, N, 1] for broadcasting
    if mask is None or mask.numel() == 0:
        mask_ex = None
    else:
        mask_ex = mask.unsqueeze(-1).to(x.dtype)

    # left / right projections
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply mask and gates
    if mask_ex is not None:
        left = left * mask_ex
        right = right * mask_ex
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Triton GEMM: out = left @ rightᵀ   (over the K dimension)
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) so we can treat each hidden slice as a batch
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_resh = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # allocate output buffer in fp16 (intermediate); final cast to fp32 later
    out_resh = torch.empty_like(left_resh, dtype=torch.float16)

    # strides (in elements)
    stride_lm, stride_ln, stride_lk = left_resh.stride()
    stride_rm, stride_rn, stride_rk = right_resh.stride()
    stride_om, stride_on, stride_ok = out_resh.stride()

    # tile sizes – chosen to fit H200 SMEM (adjustable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,
    )
    tri_mul_gemm_kernel[grid](
        left_resh,
        right_resh,
        out_resh,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, hidden]
    out = out_resh.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # Second LayerNorm + output gate + final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
