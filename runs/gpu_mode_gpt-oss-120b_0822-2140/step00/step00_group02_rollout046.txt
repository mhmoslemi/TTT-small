"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every hidden channel d:
    out_d = left_d @ right_d.T      (left_d, right_d : B×N×N)

We keep the LayerNorms, linear projections and gating in PyTorch
(because they are cheap) and fuse the cubic contraction with a
custom Triton kernel.  The kernel works on tiles of (i, j, d) and
accumulates over k in blocks.  All arithmetic is performed in FP32
and the final result is returned as float‑32.

The kernel is launched with a 3‑D grid: (i‑tile, j‑tile, batch).
It supports any batch size ≤2, sequence lengths up to 1024 and hidden
dimensions that are multiples of the block size (we mask the remainder).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the cubic contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const * left  [B,N,N,D]
    right_ptr,         # const * right [B,N,N,D]
    out_ptr,           # * out          [B,N,N,D]
    N,                 # sequence length (int)
    D,                 # hidden dimension (int)
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]   """
    pid_b = tl.program_id(2)                # batch dimension
    pid_i = tl.program_id(0)                # i‑tile index
    pid_j = tl.program_id(1)                # j‑tile index

    # ------------------------------------------------------------------
    # 1) tile offsets
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = tl.arange(0, BLOCK_D)

    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < D

    # ------------------------------------------------------------------
    # 2) accumulator: (BLOCK_I, BLOCK_J, BLOCK_D)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) loop over k in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        # k indices inside this block
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ------------------------------
        # load left  : shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # ------------------------------
        left_offs = (
            pid_b * stride_left_b
            + i_offset[:, None, None] * stride_left_i
            + k_offset[None, :, None] * stride_left_k
            + d_offset[None, None, :] * stride_left_d
        )
        left_mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
        left_tile = tl.load(left_ptr + left_offs, mask=left_mask, other=0.0)  # (I,K,D)

        # ------------------------------
        # load right : shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # ------------------------------
        right_offs = (
            pid_b * stride_right_b
            + j_offset[:, None, None] * stride_right_j
            + k_offset[None, :, None] * stride_right_k
            + d_offset[None, None, :] * stride_right_d
        )
        right_mask = j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
        right_tile = tl.load(right_ptr + right_offs, mask=right_mask, other=0.0)  # (J,K,D)

        # ------------------------------
        # accumulate Σ_k left[i,k,d] * right[j,k,d]
        # ------------------------------
        # unroll over the K‑dimension inside the block
        for kk in range(0, BLOCK_K):
            lk = left_tile[:, kk, :]          # (I,D)
            rk = right_tile[:, kk, :]         # (J,D)

            # broadcast to (I,J,D) and multiply
            # (I,1,D) * (1,J,D) -> (I,J,D)
            prod = lk[:, None, :] * rk[None, :, :]
            acc += prod

    # ------------------------------------------------------------------
    # 4) write result
    # ------------------------------------------------------------------
    out_offs = (
        pid_b * stride_out_b
        + i_offset[:, None, None] * stride_out_i
        + j_offset[None, :, None] * stride_out_j
        + d_offset[None, None, :] * stride_out_d
    )
    out_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
    tl.store(out_ptr + out_offs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   (float32, on CUDA)
            mask         : [B, N, N]        (float32/bool)
            weights      : dict of pre‑trained Parameter tensors
            config       : {"dim": int, "hidden_dim": int}

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply optional mask (mask is 1.0 where valid, 0.0 otherwise)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the cubic contraction
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                # B, seq_len, seq_len, hidden_dim
    # ensure contiguous layout for simple stride math
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    sL_b, sL_i, sL_k, sL_d = left.stride()
    sR_b, sR_j, sR_k, sR_d = right.stride()
    sO_b, sO_i, sO_j, sO_d = out.stride()

    # tile sizes – tuned for Hopper (SM90) shared‑memory / registers
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32                     # hidden_dim ≤ 384, fits nicely

    grid = (
        triton.cdiv(N, BLOCK_I),     # i‑tiles
        triton.cdiv(N, BLOCK_J),     # j‑tiles
        B,                           # batch dimension
    )
    trimul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N,
        hidden_dim,
        sL_b, sL_i, sL_k, sL_d,
        sR_b, sR_j, sR_k, sR_d,
        sO_b, sO_i, sO_j, sO_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6) Output post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection back to dim

    return out
```
