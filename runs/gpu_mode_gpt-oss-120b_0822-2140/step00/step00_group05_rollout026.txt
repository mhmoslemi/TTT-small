"""
TriMul “outgoing” implementation (forward only).

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    left[b,d] @ right[b,d].T  for each (b,d).

We keep the cheap LayerNorm / Linear / gating ops in PyTorch and
fuse the N×N × N reduction in a custom Triton kernel that works on
FP16 (accumulating in FP32) and then returns FP32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ B.T) where A,B are (B, M, K) and (B, N, K)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,      # A: batch, M, K
    stride_bb, stride_bn, stride_bk,      # B: batch, N, K
    stride_cb, stride_cm, stride_cn,      # C: batch, M, N
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # row block
    pid_n = tl.program_id(1)          # col block
    pid_b = tl.program_id(2)          # batch (combined B*hidden_dim)

    # ------------------------ offsets ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # base pointers for this batch
    a_base = A_ptr + pid_b * stride_ab
    b_base = B_ptr + pid_b * stride_bb
    c_base = C_ptr + pid_b * stride_cb

    # -----------------------------------------------------------------
    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute pointers for current K tile
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + ((k * BLOCK_K + offs_k)[None, :] * stride_ak)
        b_ptrs = b_base + ((k * BLOCK_K + offs_k)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load tiles (FP16 -> compute in FP32)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # inner product
        acc += tl.dot(a, b)

        # advance pointers (handled implicitly by the k‑loop increment)
    # -----------------------------------------------------------------
    # write result
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B : (batch, N, N)  fp16 tensors
    returns C : (batch, N, N) fp16 tensor = A @ B.T
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, _ = B.shape
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides (in elements, not bytes) – torch gives them in number of elements
    stride_ab, stride_am, stride_ak = A.stride()
    stride_bb, stride_bn, stride_bk = B.stride()
    stride_cb, stride_cm, stride_cn = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            batch)

    batched_matmul_kernel[grid](
        A.data_ptr(),
        B.data_ptr(),
        C.data_ptr(),
        M, N, K,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bn, stride_bk,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # good for fp16 on Hopper
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    N³ reduction.

    Args:
        data: (input, mask, weights, config)
            - input:      [B, N, N, dim]   (torch.float32)
            - mask:       [B, N, N]        (torch.bool or float)
            - weights:    dict of pretrained tensors
            - config:     dict with keys "dim" and "hidden_dim"

    Returns:
        output: [B, N, N, dim] (torch.float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. LayerNorm over last dim (dim)
    # --------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]     # [hidden_dim, dim]

    left  = F.linear(x, left_proj_w)        # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (mask shape: [B,N,N] -> broadcast)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core einsum -> batched matmul on hidden_dim channels
    # --------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden_dim

    # reshape to (B*H, N, N) and cast to fp16 for the kernel
    left_resh  = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous().to(torch.float16)
    right_resh = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous().to(torch.float16)

    # compute A @ B.T for each (batch, channel)
    out_resh = batched_matmul_triton(left_resh, right_resh)   # fp16

    # reshape back to [B, N, N, hidden_dim] (still fp16)
    out = out_resh.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden_dim)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=eps)

    # --------------------------------------------------------------
    # 7. Apply out gate and final linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"]        # [dim, hidden_dim]
    out = F.linear(out, to_out_w)              # [B, N, N, dim]

    return out.to(torch.float32)
