"""
TriMul “outgoing” kernel – fast forward pass.

The heavy part of TriMul is the three‑dimensional contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batch of H (= hidden_dim) matrix‑multiplications
(L_d @ R_dᵀ).  The Triton kernel tiles over the i‑ and j‑axes (BLOCK_I/J)
and over the hidden dimension (BLOCK_H).  For each tile it iterates over
the contraction axis k in blocks (BLOCK_K) and accumulates the dot‑product
with `tl.dot`.  The rest of the module (LayerNorms, linear projections,
gating) is done with PyTorch to keep the implementation short while still
meeting the “must use Triton” requirement.

The kernel works for any batch size ≤2, any sequence length that is a
multiple of the tile size (or any size with proper masking), and hidden
dimensions up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """ 
    Compute 
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a GEMM‑style tiling scheme.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) 2‑D + batch decomposition of the 1‑D grid
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    blocks_per_batch = num_i * num_j

    b = pid // blocks_per_batch                     # batch index
    rem = pid % blocks_per_batch
    i_block = rem // num_j
    j_block = rem % num_j

    # --------------------------------------------------------------
    # 2) Offsets for the tile
    # --------------------------------------------------------------
    i_start = i_block * BLOCK_I
    j_start = j_block * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # hidden dimension offsets (processed whole BLOCK_H at once)
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # accumulator: (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over the contraction axis k
    # --------------------------------------------------------------
    for block_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = block_k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left : shape (I, K, H) ---------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # (I,K,H)

        # ----- right : we want (K, J, H) so that tl.dot works ------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None, None] * stride_right_k   # K first
            + offs_j[None, :, None] * stride_right_j   # J second
            + offs_h[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # (K,J,H)

        # ----- dot‑product for each hidden slice -----------------------
        for h in range(0, BLOCK_H):
            # out_{h} += left[:,:,h] @ right[:,:,h]ᵀ   (K is contracted)
            acc[:, :, h] += tl.dot(left[:, :, h], right[:, :, h])

    # --------------------------------------------------------------
    # 4) Write the tile back
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Host side – the entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a fused Triton kernel
    for the costly contraction.  All other linear / LayerNorm / gating steps
    are performed with native PyTorch for simplicity.

    Args:
        data: (input, mask, weights, config)
            - input:  Tensor [B, N, N, dim]   (float32 or bf16)
            - mask :  Tensor [B, N, N]        (bool / float)
            - weights: dict with all model parameters
            - config : dict with 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dim
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)                    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton contraction
    # ------------------------------------------------------------------
    # ensure contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)                       # same shape [B,N,N,hidden_dim]

    # strides (in elements, not bytes) needed by the kernel
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – tuned for Hopper SM90 (shared‑mem capacity ≈ 64 KB)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 32

    # Grid: one program per (batch, i‑tile, j‑tile)
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    total_programs = B * grid_i * grid_j
    grid = (total_programs,)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm → gating → final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # out currently has shape [B,N,N,hidden_dim]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    # apply the final gate (same shape as out)
    out = out * out_gate

    # final projection back to dim
    out = F.linear(out, to_out_w)                      # [B,N,N,dim]

    return out.to(torch.float32)
```
