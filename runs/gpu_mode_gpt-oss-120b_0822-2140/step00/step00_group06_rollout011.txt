"""
TriMul “outgoing” implementation (forward only).

The heavy part of the TriMul forward pass is the three‑tensor einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication per hidden channel:
for every (b,d) compute   out_bd = left_bd @ right_bd.T
with left_bd,right_bd ∈ ℝ^{N×N}.  The kernel below performs this
multiplication with Triton, while the cheap LayerNorm / Linear /
Sigmoid parts stay in PyTorch.  The final output is returned as
float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = left @ right^T   (per batch‑hidden slice)
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,               # *float32   left  [B,N,N,H]
    right_ptr,              # *float32   right [B,N,N,H]
    out_ptr,                # *float32   out   [B,N,N,H]
    N,                      # i32        sequence length
    H,                      # i32        hidden dimension
    B,                      # i32        batch size
    # strides of the tensors (contiguous layout assumed but we pass them for safety)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # constexpr block sizes
    BLOCK_M: tl.constexpr,  # rows of output (i dimension)
    BLOCK_N: tl.constexpr,  # cols of output (j dimension)
    BLOCK_K: tl.constexpr,  # reduction dimension (k)
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    # ------------------------------------------------------------------
    # 1) Identify the (batch, hidden) slice this program instance works on
    # ------------------------------------------------------------------
    pid = tl.program_id(0)                 # 0 … B*H-1
    b = pid // H                           # batch index
    h = pid % H                            # hidden index

    # base pointer for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # ------------------------------------------------------------------
    # 2) 2‑D grid over (i,j) output matrix
    # ------------------------------------------------------------------
    pid_i = tl.program_id(1)               # row block id
    pid_j = tl.program_id(2)               # col block id

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over k (reduction)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[ i , k ]  → shape (BLOCK_M , BLOCK_K)
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[ j , k ] – we need it transposed so that k is the inner dim:
        # shape (BLOCK_K , BLOCK_N)
        right_ptrs = right_base \
                     + offs_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product along the K dimension
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # ------------------------------------------------------------------
    # 4) Write results
    # ------------------------------------------------------------------
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32 or bfloat16)
        - mask         : torch.Tensor [B, N, N]       (bool/float)
        - weights      : dict of torch.Tensor containing all linear / LN weights
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (norm over the last dimension)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections + gates (no bias)
    # --------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # mask (optional)
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Core TriMul einsum using Triton (batched matmul per hidden channel)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    # output buffer
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (int64 -> python int)
    (s_b_l, s_i_l, s_k_l, s_h_l) = left.stride()
    (s_b_r, s_j_r, s_k_r, s_h_r) = right.stride()
    (s_b_o, s_i_o, s_j_o, s_h_o) = out.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * hidden_dim,                     # program id 0  → (batch, hidden)
        triton.cdiv(N, BLOCK_M),           # program id 1  → row blocks
        triton.cdiv(N, BLOCK_N),           # program id 2  → col blocks
    )
    trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        H=hidden_dim,
        B=B,
        stride_left_b=s_b_l,
        stride_left_i=s_i_l,
        stride_left_k=s_k_l,
        stride_left_h=s_h_l,
        stride_right_b=s_b_r,
        stride_right_j=s_j_r,
        stride_right_k=s_k_r,
        stride_right_h=s_h_r,
        stride_out_b=s_b_o,
        stride_out_i=s_i_o,
        stride_out_j=s_j_o,
        stride_out_h=s_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 4. Post‑processing: LayerNorm over hidden_dim, gate, final Linear
    # --------------------------------------------------------------
    # to_out_norm : LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # apply output gate (broadcast over hidden dim)
    out = out * out_gate

    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    # ensure final dtype = float32
    out = out.to(torch.float32)

    return out
