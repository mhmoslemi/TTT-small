"""
TriMul “outgoing” implementation – fast Triton kernel for the
core N×N×N contraction.

The heavy step is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T

The kernel below computes this GEMM for every (batch, hidden_dim) pair.
All LayerNorms, Linear projections, gates and the final
LayerNorm+Linear are kept in PyTorch – they are cheap compared to the
O(N³) contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (left @ right.T) over the (i,j) plane
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, D,                               # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ right.T for every (b,d) pair.
    left  : [B, N, N, D]  (i,k)
    right : [B, N, N, D]  (j,k)
    out   : [B, N, N, D]  (i,j)
    """
    pid = tl.program_id(0)                     # combines batch and hidden dim
    b = pid // D                               # batch index
    d = pid % D                                # hidden dim index

    pid_i = tl.program_id(1)                   # tile row
    pid_j = tl.program_id(2)                   # tile col

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,d]  -> shape (M, K)
        ptr_left = (
            left_ptr
            + b * stride_left_b
            + d * stride_left_d
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(ptr_left, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_T[k,j,d]  (note the transpose) -> shape (K, N)
        ptr_right_T = (
            right_ptr
            + b * stride_right_b
            + d * stride_right_d
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        right_T = tl.load(
            ptr_right_T, mask=mask_k[:, None] & mask_j[None, :], other=0.0
        )

        # accumulate dot product
        acc += tl.dot(left, right_T)

    # store result
    ptr_out = (
        out_ptr
        + b * stride_out_b
        + d * stride_out_d
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(ptr_out, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass with a Triton‑accelerated
    N×N×N contraction.

    Args:
        data: (input, mask, weights, config)
            input  – [B, N, N, dim]  (float32)
            mask   – [B, N, N] (bool/float) – will be broadcast over hidden dim
            weights – dict with the exact keys used in the reference model
            config  – {"dim": int, "hidden_dim": int}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on last dim ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections ----------
    # Linear without bias: y = x @ W.T
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------- Optional mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)                  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Triton GEMM (left @ right.T) ----------
    B, N, _, _ = left.shape                     # B,N,N,hidden_dim
    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements) for contiguous tensors [B,N,N,D]
    s_b, s_i, s_k, s_d = left.stride()          # left and right share the same layout
    # right strides are identical
    # out strides
    so_b, so_i, so_j, so_d = out.stride()

    # block sizes – tuned for Hopper, adjust if needed
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # program_id 0  -> (batch, hidden_dim)
        triton.cdiv(N, BLOCK_M),               # program_id 1  -> i‑tile
        triton.cdiv(N, BLOCK_N),               # program_id 2  -> j‑tile
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=s_b,
        stride_left_i=s_i,
        stride_left_k=s_k,
        stride_left_d=s_d,
        stride_right_b=s_b,
        stride_right_j=s_k,      # note: right has shape (B,N,N,D); j dimension stride == stride of the second N
        stride_right_k=s_i,      # k dimension corresponds to the first N stride
        stride_right_d=s_d,
        stride_out_b=so_b,
        stride_out_i=so_i,
        stride_out_j=so_j,
        stride_out_d=so_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Output LayerNorm + gating ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ---------- Final linear ----------
    output = F.linear(out, weights["to_out.weight"])

    return output
