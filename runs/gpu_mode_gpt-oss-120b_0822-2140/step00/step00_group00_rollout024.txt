"""
TriMul “outgoing” implementation (forward only).

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication for every hidden
dimension `d`:

    out_d = left_d @ right_dᵀ   with   left_d, right_d ∈ ℝ^{B×N×N}

We fuse this N³ computation in a custom Triton kernel that works on a
flattened batch‑hidden dimension (B·hidden).  All remaining linear /
layer‑norm / gating operations are performed with the native PyTorch
functions (they are cheap compared to the N³ work).  The kernel
operates in FP16 for bandwidth‑limited speed but returns FP32 as required.

The kernel layout:
    * BLOCK_M × BLOCK_N tiles for the output matrix.
    * BLOCK_K reduction over the `k` dimension.
    * 3‑D grid → (grid_m, grid_n, batch) packed into a 1‑D program ID.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for C = A @ Bᵀ  (A,B shape: [batch, M, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_t_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_am, stride_ak,               # A strides (batch, M, K)
    stride_bm, stride_bn,               # Bᵀ strides (batch, N, K) – note B is transposed
    stride_cm, stride_cn,               # C strides (batch, M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,  # stride to next batch element
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Compute 3‑D program ID → (batch, block_m, block_n)
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)

    pid_batch = pid // (grid_m * grid_n)
    pid_rem   = pid %  (grid_m * grid_n)
    pid_m = pid_rem // grid_n
    pid_n = pid_rem %  grid_n

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in A, B, C (including batch offset)
    a_tile_ptr = a_ptr + pid_batch * batch_stride_a \
               + offs_m[:, None] * stride_am \
               + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    b_tile_ptr = b_ptr + pid_batch * batch_stride_b \
               + offs_n[:, None] * stride_bn \
               + tl.arange(0, BLOCK_K)[None, :] * stride_bm   # Bᵀ has K as inner dim

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in steps of BLOCK_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a sub‑tile of A and Bᵀ (FP16)
        a = tl.load(a_tile_ptr, mask=mask_m[:, None], other=0.0).to(tl.float16)
        b = tl.load(b_tile_ptr, mask=mask_n[:, None], other=0.0).to(tl.float16)

        # dot: (M, K) • (N, K)ᵀ → (M, N)
        acc += tl.dot(a, b, trans_b=True).to(tl.float32)

        # Increment pointers to next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bm

    # ------------------------------------------------------------------
    # Write back C = acc (cast back to FP16 then to FP32 as required)
    # ------------------------------------------------------------------
    c_tile_ptr = c_ptr + pid_batch * batch_stride_c \
               + offs_m[:, None] * stride_cm \
               + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with a fused Triton kernel.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float16/32 on CUDA)
        - mask         : torch.Tensor [B, N, N] (bool/float)
        - weights      : dict of torch.Tensor containing all linear / LN params
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias) – keep FP16 where possible
    # ------------------------------------------------------------------
    # x is (B,N,N,dim)
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)
    left  = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(left.dtype).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden)
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden)
    out_gate   = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden)

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched matmul kernel
    #   out_d = left_d @ right_dᵀ   for each hidden dimension d
    # ------------------------------------------------------------------
    # Bring hidden dimension to a batch axis
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    batch = B * hidden
    left  = left.view(batch, N, N)                 # (B*hidden, N, N)
    right = right.view(batch, N, N)                # (B*hidden, N, N)

    # We need rightᵀ for the multiplication
    right_t = right.transpose(1, 2).contiguous()   # (B*hidden, N, N)

    # ------------------------------------------------------------------
    # Allocate output buffer (FP16 inside kernel, convert later)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)                   # (B*hidden, N, N), dtype matches left (fp16/32)

    # ------------------------------------------------------------------
    # Kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_grid = grid_m * grid_n * batch

    # Strides (in elements) for contiguous tensors
    stride_am = left.stride(1)   # stride over M dimension
    stride_ak = left.stride(2)   # stride over K dimension
    stride_bm = right_t.stride(1)  # stride over N (treated as M in kernel)
    stride_bn = right_t.stride(2)  # stride over K (since we stored transposed)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    batch_stride_a = left.stride(0)
    batch_stride_b = right_t.stride(0)
    batch_stride_c = out.stride(0)

    batched_matmul_t_kernel[(total_grid,)](
        left,
        right_t,
        out,
        N, N, N,
        stride_am, stride_ak,
        stride_bm, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Reshape back to original layout (B,N,N,hidden)
    # ------------------------------------------------------------------
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # to_out_norm + gating + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # Final linear: hidden -> dim
    out_flat = out.reshape(-1, hidden)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    # Ensure final dtype is float32 (as required by the spec)
    return out.to(torch.float32)
