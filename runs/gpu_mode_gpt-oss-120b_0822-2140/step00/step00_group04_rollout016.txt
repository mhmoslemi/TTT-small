"""
TriMul “outgoing” implementation.

The heavy N‑³ contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel.  All other
operations (LayerNorms, Linear projections, gates, final
LayerNorm + Linear) are kept in PyTorch – they are cheap
compared with the triple‑product and give us a clean
interface to the supplied weight dict.

The kernel computes a tiled batched GEMM for every
(batch, hidden‑dim‑block, tile‑row, tile‑col) and accumulates
in FP32, storing the result back in FP16 (later cast to FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   (batched, per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,      # const float16 *
    right_ptr,     # const float16 *
    out_ptr,       # float16 *
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_i, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # --------------------------------------------------------------
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute the Einstein‑summation over the third dimension
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    where `right` is accessed transposed in the k‑dimension.
    The kernel tiles the (i,j) matrix and also tiles the hidden
    dimension `d` to keep register pressure low.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1️⃣  Grid decomposition  (batch, d‑block, row‑block, col‑block)
    # --------------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)          # blocks in i
    num_n = tl.cdiv(N, BLOCK_N)          # blocks in j
    num_d = tl.cdiv(H, BLOCK_D)          # blocks in hidden dim
    total_blocks = B * num_m * num_n * num_d

    if pid >= total_blocks:
        return

    tmp = pid
    b = tmp // (num_m * num_n * num_d)
    tmp = tmp % (num_m * num_n * num_d)

    d_block = tmp // (num_m * num_n)
    tmp = tmp % (num_m * num_n)

    m_block = tmp // num_n
    n_block = tmp % num_n

    # --------------------------------------------------------------
    # 2️⃣  Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = m_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = n_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = d_block * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_d = offs_d < H

    # --------------------------------------------------------------
    # 3️⃣  Accumulator for each hidden‑dim slice
    # --------------------------------------------------------------
    # acc[d] has shape (BLOCK_M, BLOCK_N)
    acc = tl.zeros((BLOCK_D, BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4️⃣  Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of left:   (BLOCK_M, BLOCK_K, BLOCK_D)
        #   layout: b, i, k, d
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_m[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_h
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )  # fp16 -> fp32 automatically on load

        # Load a tile of rightᵀ: (BLOCK_K, BLOCK_N, BLOCK_D)
        #   right is stored as (b, j, k, d); we need (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None, None] * stride_right_k       # k as the first axis
            + offs_n[None, :, None] * stride_right_i       # j as the second axis
            + offs_d[None, None, :] * stride_right_h
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_n[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # 5️⃣  Inner product for each hidden‑dim slice
        # --------------------------------------------------------------
        for d in range(0, BLOCK_D):
            # left_slice: (BLOCK_M, BLOCK_K)   right_slice: (BLOCK_K, BLOCK_N)
            left_slice = left_tile[:, :, d]
            right_slice = right_tile[:, :, d]
            acc[d] += tl.dot(left_slice, right_slice)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # 6️⃣  Write the result back
    # --------------------------------------------------------------
    for d in range(0, BLOCK_D):
        out_ptrs = (
            out_ptr
            + b * stride_out_b
            + offs_m[:, None] * stride_out_i
            + offs_n[None, :] * stride_out_j
            + (offs_d[d]) * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc[d].to(tl.float16),
            mask=mask_m[:, None] & mask_n[None, :],
        )

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣  unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]  float32

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 3️⃣  Apply mask (if any)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.unsqueeze(-1).to(x.dtype)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Cast to FP16 for the heavy contraction
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape

    # --------------------------------------------------------------
    # 6️⃣  Prepare strides for the kernel (contiguous layout)
    # --------------------------------------------------------------
    left_fp16 = left_fp16.contiguous()
    right_fp16 = right_fp16.contiguous()
    out_fp16 = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    stride_left = left_fp16.stride()     # (B, i, k, h)
    stride_right = right_fp16.stride()   # same layout
    stride_out = out_fp16.stride()

    # Triton expects 64‑bit ints
    B_i, N_i, H_i = [int(v) for v in (B, N, H)]

    # --------------------------------------------------------------
    # 7️⃣  Launch kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32   # hidden‑dim tile

    grid = (
        triton.cdiv(B_i, 1)
        * triton.cdiv(N_i, BLOCK_M)
        * triton.cdiv(N_i, BLOCK_N)
        * triton.cdiv(H_i, BLOCK_D),
    )

    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B_i,
        N_i,
        H_i,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[2],   # k stride (row‑wise after transpose)
        stride_right[1],   # i (j) stride
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 8️⃣  Cast back to FP32 and continue with the remaining ops
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply out‑gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
