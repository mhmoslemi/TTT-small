"""
TriMul “outgoing” forward pass (AlphaFold3) with a fused Triton kernel for the
N³ matrix‑multiply across the sequence dimension.

The heavy part:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply L @ Rᵀ for every hidden channel d.
We compute this with a custom Triton kernel that tiles the (i, j) plane and the
hidden‑channel dimension, looping over k in blocks.  All other operations
(LayerNorm, Linear, Sigmoid, gating, final projection) are performed in
regular PyTorch – they are cheap compared to the N³ work and keep the code
readable.

The kernel works for:
    B ∈ {1,2}, N ∈ {128,…,1024}, hidden_dim ∈ {128,384,768}
and supports optional masking.
The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul L @ Rᵀ per hidden‑channel block
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,               # [B, N, N, H]   (i,k,d)
    right_ptr,              # [B, N, N, H]   (j,k,d)
    out_ptr,                # [B, N, N, H]   (i,j,d)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid0 = tl.program_id(0)   # combines batch and hidden‑channel block
    pid1 = tl.program_id(1)   # i‑tile
    pid2 = tl.program_id(2)   # j‑tile

    # --------------------------------------------------------------
    # decode batch index and hidden‑channel block
    # --------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid0 // num_h_blocks
    h_block = pid0 % num_h_blocks
    h_offset = h_block * BLOCK_H + tl.arange(0, BLOCK_H)

    # --------------------------------------------------------------
    # tile offsets for i and j
    # --------------------------------------------------------------
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds in each dimension
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_h = h_offset < H

    # ------------------------------------------------------------------
    # Pointers with proper strides
    # ------------------------------------------------------------------
    # left: (b, i, k, h)
    left_ptrs = left_ptr + \
        b * stride_l_b + \
        i_offset[:, None, None] * stride_l_i + \
        tl.arange(0, BLOCK_K)[None, :, None] * stride_l_k + \
        h_offset[None, None, :] * stride_l_h

    # right: (b, j, k, h)
    right_ptrs = right_ptr + \
        b * stride_r_b + \
        j_offset[:, None, None] * stride_r_j + \
        tl.arange(0, BLOCK_K)[None, :, None] * stride_r_k + \
        h_offset[None, None, :] * stride_r_h

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k dimension in BLOCK_K chunks
    # --------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)

        # load left & right tiles
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & (tl.arange(0, cur_k)[None, :] < cur_k)[:, None, None] & mask_h[None, None, :],
                       other=0.0)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & (tl.arange(0, cur_k)[None, :] < cur_k)[:, None, None] & mask_h[None, None, :],
                        other=0.0)

        # for each hidden channel compute dot product across k
        for h in range(BLOCK_H):
            # (M, K) dot (N, K)ᵀ → (M, N)
            a = left[:, :, h]          # (M, K)
            b = right[:, :, h]         # (N, K)
            acc[:, :, h] += tl.dot(a, b)

        # advance pointers to next k‑block
        left_ptrs += BLOCK_K * stride_l_k
        right_ptrs += BLOCK_K * stride_r_k

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        i_offset[:, None, None] * stride_o_i + \
        j_offset[None, :, None] * stride_o_j + \
        h_offset[None, None, :] * stride_o_h

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm + Linear (bias‑less) in pure torch
# ----------------------------------------------------------------------
def linear_no_bias(x, weight):
    # x: [..., C], weight: [out, C] (as in nn.Linear(out_features, in_features))
    return torch.nn.functional.linear(x, weight)


def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    return F.layer_norm(x, x.shape[-1:], weight, bias, eps)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel
    for the N³ matrix multiply.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]  float32 / bfloat16
            - mask         : [B, N, N]       (may be all‑ones if nomask)
            - weights      : dict of tensors (see top‑level skeleton)
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        out: [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                         # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left  = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (optional) – broadcast on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: out = left @ rightᵀ  (per hidden channel)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (int64) for pointer arithmetic
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (
        triton.cdiv(B * triton.cdiv(hidden, BLOCK_H), 1),   # program_id(0)
        triton.cdiv(N, BLOCK_M),                           # program_id(1)
        triton.cdiv(N, BLOCK_N),                           # program_id(2)
    )

    trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final projection back to dim
    # ------------------------------------------------------------------
    out = linear_no_bias(out, weights["to_out.weight"])

    return out.to(torch.float32)
