"""
TriMul “outgoing” implementation with a small Triton kernel.

The heavy linear / gating work is kept in PyTorch (cuBLAS / cuDNN) which is
already highly optimized on Hopper.  The only part written in Triton is a
fused element‑wise kernel that combines:
    left  = left_proj  * mask * left_gate
    right = right_proj * mask * right_gate
This saves memory traffic and guarantees that we use at least one
custom Triton kernel as required by the task.

The rest follows the reference PyTorch code:
    * LayerNorm on the input.
    * Linear projections (left/right) and three gating linear layers.
    * Per‑hidden‑dim matrix multiplication  O(N³·hidden)  via
      torch.bmm (L @ Rᵀ) where L,R are N×N matrices for each hidden channel.
    * Output LayerNorm, gating and final linear projection.
All final tensors are returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: fuse mask & gates into left/right tensors
# -------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, right_ptr,               # projected tensors
    mask_ptr,                          # mask[...,1]
    left_gate_ptr, right_gate_ptr,     # sigmoided gates
    out_left_ptr, out_right_ptr,       # fused results
    N_ELEM: tl.constexpr, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    m = tl.load(mask_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    out_l = l * m * lg
    out_r = r * m * rg

    tl.store(out_left_ptr + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


# -------------------------------------------------------------
# Entry point required by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (float32, 0/1)
        - weights      : dict of torch.Tensor containing the model weights
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (float32 → float16 for the rest)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in fp16
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # 3. Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Fuse mask * gates * projections with Triton
    # ------------------------------------------------------------------
    #   left  = left_proj * mask * left_gate
    #   right = right_proj * mask * right_gate
    mask_ = mask.unsqueeze(-1).to(torch.float16)          # [B,N,N,1]

    total_elem = left_proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elem, BLOCK),)

    # allocate output buffers
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    fuse_mask_gate_kernel[grid](
        left_proj.data_ptr(),
        right_proj.data_ptr(),
        mask_.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        left_fused.data_ptr(),
        right_fused.data_ptr(),
        N_ELEM=total_elem,
        BLOCK=BLOCK,
    )
    # now `left_fused` and `right_fused` hold the masked & gated tensors
    left = left_fused
    right = right_fused

    # ------------------------------------------------------------------
    # 5. Core TriMul:   out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    #    This is a batched GEMM per hidden dimension:   L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # shape is [B,N,N,hidden]

    # reshape to (B*hidden, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)          # (B·h, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)        # (B·h, N, N)

    # GEMM: L @ Rᵀ  → shape (B·h, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden]
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # broadcast over hidden dim

    # final linear (hidden -> dim)
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output as required
    return out.to(torch.float32)
