"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with left/right already masked and gated.
For every (batch, hidden_dim) pair this is a matmul N×N  @  N×Nᵀ.
The kernel below performs a tiled matmul over the k‑dimension
using Triton, while the surrounding LayerNorm/Linear logic is kept
in PyTorch (fast and easy to express).  The final result is
converted back to float32 as required by the specification.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (per batch and per hidden dim)
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,              # pointers
    B, N, H,                                   # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    Each program works on a tile (BLOCK_M x BLOCK_N) of the output
    for a fixed (batch, hidden) pair.
    """
    ph = tl.program_id(0)          # combined batch*hidden index
    pi = tl.program_id(1)          # block row   (i)
    pj = tl.program_id(2)          # block col   (j)

    # ------------------------------------------------------------------
    # decode batch and hidden index from the first dimension
    # ------------------------------------------------------------------
    b = ph // H
    h = ph % H

    # ------------------------------------------------------------------
    # compute the starting coordinates of the tile
    # ------------------------------------------------------------------
    i_start = pi * BLOCK_M
    j_start = pj * BLOCK_N

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # accumulator for the tile (float32 is safe for the reduction)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        cur_k = tl.minimum(offs_k, N - k_base - 1)   # handle tail
        k_valid = (k_base + cur_k) < N

        # left tile: shape (BLOCK_M, cur_K)
        left_ptrs = left_ptr + (
            b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + (k_base + cur_k)[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & k_valid[None, :], other=0.0)

        # right tile needs to be (K, BLOCK_N) because tl.dot expects (M,K)@(K,N)
        right_ptrs = right_ptr + (
            b * stride_right_b
            + (k_base + cur_k)[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(right_ptrs, mask=k_valid[:, None] & mask_j[None, :], other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # write the tile back
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + (
        b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Entry point called by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : [B, N, N, C]  (float32)
        - mask         : [B, N, N]    (float32 / bool)
        - weights      : dict of learned parameters
        - config       : dict with keys
            * dim          – input channel dimension C
            * hidden_dim   – hidden projection dimension H
    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and masking
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
    left = left * mask_
    right = right * mask_

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 4) Triton kernel for the heavy einsum:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # allocate output (float32 for safety)
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=x.device)

    # compute strides (in elements, not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # launch parameters (tuned for Hopper; can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                                 # combined batch*hidden dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        # strides
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_h=s_left[3],
        stride_right_b=s_right[0],
        stride_right_j=s_right[1],
        stride_right_k=s_right[2],
        stride_right_h=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_h=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5) Post‑processing: LayerNorm over hidden dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure final dtype
    return out.to(torch.float32)
