"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension *d* :

    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T

The kernel below computes this product in fp16 using Triton, while the
surrounding LayerNorms, linear projections, gates and the final linear
are performed with PyTorch (they are cheap compared to the N³ work).
The result is converted back to fp32 for the remaining operations.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with a transposed B (right) matrix.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute  out = left @ right.T   for each (b, h) pair."""
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Map 1‑D pid → (batch, head, i_tile, j_tile)
    # ------------------------------------------------------------------
    tiles_i = tl.cdiv(N, BLOCK_I)
    tiles_j = tl.cdiv(N, BLOCK_J)
    tiles_per_head = tiles_i * tiles_j          # number of (i,j) tiles per (b,h)

    b = pid // (H * tiles_per_head)
    tmp = pid % (H * tiles_per_head)
    h = tmp // tiles_per_head
    tile_idx = tmp % tiles_per_head
    i_tile = tile_idx // tiles_j
    j_tile = tile_idx % tiles_j

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    k_offset = tl.arange(0, BLOCK_K)

    # masks – handle tails
    i_mask = i_offset < N
    j_mask = j_offset < N
    k_mask = k_offset < N

    # Pointers that include batch and head stride
    left_base = left_ptr  + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # ------------------------------------------------------------------
    # Accumulator (fp32 for better precision)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k indices for this block
        k = kp * BLOCK_K + k_offset

        # Load A tile: shape (BLOCK_I, BLOCK_K)
        a_ptrs = (
            left_base
            + (i_offset[:, None] * stride_left_i)
            + (k[None, :]  * stride_left_k)
        )
        a = tl.load(a_ptrs, mask= i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_J)
        # right layout is (b, j, k, h) → we stride over k first, then j
        b_ptrs = (
            right_base
            + (k[:, None] * stride_right_k)
            + (j_offset[None, :] * stride_right_j)
        )
        b = tl.load(b_ptrs, mask= k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # GEMM: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) → (BLOCK_I, BLOCK_J)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the resulting tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_base
        + (i_offset[:, None] * stride_out_i)
        + (j_offset[None, :] * stride_out_j)
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask= i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32, cuda)
        - mask_tensor  : [B, N, N]       (float32/bool, cuda)
        - weights_dict : mapping from string → torch.Tensor
        - config_dict  : {"dim":int, "hidden_dim":int}
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32, cuda)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # used later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core einsum via Triton  (compute in fp16)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)

    # Strides for the 4‑D tensors (contiguous in memory, i.e. NCHW-like)
    s_b, s_i, s_k, s_h = left_fp16.stride()
    # left: (B, N, N, H) → strides: (N*N*H, N*H, H, 1)
    # right has the same layout
    stride_left_b,  stride_left_i,  stride_left_k,  stride_left_h  = s_b, s_i, s_k, s_h
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = s_b, s_i, s_k, s_h
    # output has identical layout
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()

    # Tile sizes – chosen to fit into shared memory on Hopper (SM90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    tiles_i = triton.cdiv(N, BLOCK_I)
    tiles_j = triton.cdiv(N, BLOCK_J)
    total_programs = B * hidden * tiles_i * tiles_j
    grid = (total_programs,)

    _trimul_einsum_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Back to fp32 and apply the remaining LayerNorm & linear
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # multiply by the output gate (broadcasted correctly)
    out = out * out_gate

    # final linear projection to original dim
    out = F.linear(out, weights["to_out.weight"])   # shape [B,N,N,dim]

    return out
