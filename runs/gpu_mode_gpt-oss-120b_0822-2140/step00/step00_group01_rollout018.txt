"""
TriMul “outgoing” kernel – fast Triton implementation of the N³ pairwise
matrix‑multiply step.

The heavy part of TriMul is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

For a fixed hidden channel h this is a standard matrix multiplication
    L_h (N×N) @ R_hᵀ (N×N) → (N×N)

We flatten the batch‐and‑hidden dimensions (B·H) and run a single‑program
Triton GEMM kernel on each (N×N)×(N×N) pair.  All surrounding linear,
normalisation and gating operations stay in PyTorch (FP32) – the kernel
operates in FP16 for bandwidth‑limited speed while the final result is
cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton GEMM kernel (row‑major A, row‑major B, row‑major C)
# ----------------------------------------------------------------------
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix dimensions
    stride_am, stride_ak,       # A strides
    stride_bk, stride_bn,       # B strides
    stride_cm, stride_cn,       # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                       # one program per (B*H) matrix
    # offsets for the current output tile
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (when N is not a multiple of BLOCK_*)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to current A and B tiles
        a_ptrs = a_ptr + (offs_m[:, None] * stride_am
                         + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + (offs_k[:, None] * stride_bk
                         + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

        # advance pointers for next K‑tile
        a_ptr += BLOCK_K * stride_ak
        b_ptr += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm
                      + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel for a batch of (B*H) matrices
# ----------------------------------------------------------------------
def matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : (B*H, M, K)   fp16, row‑major, contiguous
    b : (B*H, K, N)   fp16, row‑major, contiguous
    returns c : (B*H, M, N) fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_H, M, N), device=a.device, dtype=torch.float16)

    # strides in element counts
    stride_am, stride_ak = a.stride()
    stride_bk, stride_bn = b.stride()
    stride_cm, stride_cn = c.stride()

    # tile sizes – tuned for Hopper SM90 (shared memory ≈ 64 KB)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B_H,)
    _matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=M,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough for fp16 GEMM
    )
    return c


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Fast TriMul (outgoing) implementation.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]       (float32 or bool)
        - weights      : dict of pretrained parameters
        - config       : {"dim": int, "hidden_dim": int}
    Returns
    -------
    out : torch.Tensor
        [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + gating
    # ------------------------------------------------------------------
    # shape helpers – we keep everything in FP32 until the GEMM
    # Einsum replaces the weight matrix multiply
    left = torch.einsum("b i j d, h d -> b i j h", x, weights["left_proj.weight"])
    right = torch.einsum("b i j d, h d -> b i j h", x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1)                # [B, N, N, 1]
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(
        torch.einsum("b i j d, h d -> b i j h", x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.einsum("b i j d, h d -> b i j h", x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.einsum("b i j d, h d -> b i j h", x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core pairwise multiplication via Triton GEMM
    # ------------------------------------------------------------------
    # bring to (B, H, N, N) -> (B*H, N, N) layout
    left = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # we need Rᵀ for the einsum; transpose the last two dims
    right_t = right.transpose(1, 2).contiguous()

    # convert to FP16 for the kernel (better bandwidth on H200)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    # GEMM: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_fp16 = matmul_triton(left_fp16, right_fp16)

    # reshape back to [B, N, N, hidden]
    out = (
        out_fp16.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)          # back to FP32 for the remaining layers
    )

    # ------------------------------------------------------------------
    # 4. Output projection (LayerNorm + gating + linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gating after norm
    out = torch.einsum("b i j h, d h -> b i j d", out, weights["to_out.weight"])
    return out
