"""
TriMul “outgoing” kernel (AlphaFold3 style).

The heavy part of TriMul is a batched matrix‑multiply:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension *d* this is exactly a mat‑mul
    left_d (N×N) × right_dᵀ (N×N) → out_d (N×N).

We fuse the N³ work with a custom Triton GEMM that processes
*B × hidden_dim* independent mat‑muls in a single launch.
All LayerNorms, linear projections and gates are kept in PyTorch
(because they are cheap relative to the N³ contraction) and the
result of the GEMM is cast back to float32 for the remaining
operations.

The kernel works for any batch size (1‑2), sequence length up to 1024
and hidden dimensions that are multiples of the block size (we use
a modest block size that works for the provided configurations).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM: C = A @ Bᵀ   (A,B shape: (batch, M, K) )
# ----------------------------------------------------------------------
@triton.jit
def _matmul_transpose_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (b * hidden)

    # ------------------------------------------------------------------
    # block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # mask out‑of‑bounds
    # ------------------------------------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # pointers
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    # B is transposed inside the kernel, i.e. we read B[k, n]
    b_ptrs = (
        B_ptr
        + pid_b * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )
    c_ptrs = (
        C_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )

    # ------------------------------------------------------------------
    # accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main loop over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # a,b are fp16; promote to fp32 for the dot
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a fused Triton GEMM.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm and linear layers (remain in PyTorch)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)          # [B,N,N,dim]

    # projections (no bias)
    left = F.linear(x, left_proj_w)                                            # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # optional mask
    if not nomask:
        mask_exp = mask.unsqueeze(-1)                                          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the GEMM (good perf on Hopper)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape           # B=batch, N=seq_len, H=hidden_dim

    # Collapse batch and hidden into a single GEMM batch dimension
    A = left_fp16.reshape(B * H, N, N)      # (B*H, M=N, K=N)
    B_mat = right_fp16.reshape(B * H, N, N)

    # Allocate output
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # launch parameters (tuned for H200, works for all required sizes)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * H,                        # batch × hidden
    )

    _matmul_transpose_kernel[grid](
        A,
        B_mat,
        C,
        N,
        N,
        N,
        A.stride(0), A.stride(1), A.stride(2),          # a_batch, a_m, a_k
        B_mat.stride(0), B_mat.stride(2), B_mat.stride(1),  # b_batch, b_k, b_n (note transpose)
        C.stride(0), C.stride(1), C.stride(2),          # c_batch, c_m, c_n
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original 4‑D layout and cast to fp32
    out = C.view(B, N, N, H).to(torch.float32)

    # ------------------------------------------------------------------
    # post‑processing (LayerNorm, out‑gate, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)           # shape [B,N,N,dim]

    return out
