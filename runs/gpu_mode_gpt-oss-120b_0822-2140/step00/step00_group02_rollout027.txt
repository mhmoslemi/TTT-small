"""
tri_mul Triton kernel (outgoing version)

The heavy part of the TriMul forward pass is the 4‑D Einstein sum

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is equivalent to a batched matrix multiplication per channel `d` :

    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T

The kernel below computes this operation in FP16 with a classic
blocked GEMM on Hopper (SM90).  All surrounding linear layers,
layer‑norms and gating are kept in PyTorch for simplicity.
The final result is cast back to FP32 as required by the API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: blocked GEMM over (i, j) for each (batch, channel)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # tensors
    B, N, D,                               # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[..., i, k, d] * right[..., j, k, d]."""
    # ------------------------------------------------------------------
    #  program IDs : (grid_m, grid_n, grid_z)
    #  grid_z folds batch * channel together.
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_z = tl.program_id(2)          # batch*channel

    batch_id = pid_z // D
    channel_id = pid_z % D

    # starting offsets of the block
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # ------------------------------------------------------------------
    #  Pointers for the current (batch, channel)
    # ------------------------------------------------------------------
    left_base = left_ptr + batch_id * stride_l_b + channel_id * stride_l_d
    right_base = right_ptr + batch_id * stride_r_b + channel_id * stride_r_d
    out_base = out_ptr + batch_id * stride_o_b + channel_id * stride_o_d

    # ------------------------------------------------------------------
    #  Allocate accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑tile offsets
        k_offset = k * BLOCK_K

        # Load a tile of left:   [M, K]
        offs_i = i_offset + tl.arange(0, BLOCK_M)
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_i = offs_i < N
        mask_k = offs_k < N
        mask_l = mask_i[:, None] & mask_k[None, :]

        left_tile = tl.load(
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_l,
            other=0.0,
        ).to(tl.float32)

        # Load a tile of right (transposed in K): [N, K] -> we need right[j, k]
        offs_j = j_offset + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N
        mask_r = mask_j[:, None] & mask_k[None, :]

        right_tile = tl.load(
            right_base
            + offs_j[:, None] * stride_r_i   # note: right's "i" axis is the column index j
            + offs_k[None, :] * stride_r_k,
            mask=mask_r,
            other=0.0,
        ).to(tl.float32)

        # Accumulate
        acc += tl.dot(left_tile, right_tile)

        # Advance pointers (handled implicitly by offsets in next iteration)

    # ------------------------------------------------------------------
    #  Write the result tile
    # ------------------------------------------------------------------
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_o = mask_i[:, None] & mask_j[None, :]

    tl.store(
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j,
        acc,
        mask=mask_o,
    )


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.Tensor of shape [B, N, N, D] (FP16)
    returns out : torch.Tensor of same shape (FP16)
    """
    B, N, _, D = left.shape
    # strides in elements
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    out = torch.empty_like(left)

    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # block sizes – tuned for Hopper, can be adjusted
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * D,
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, D,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32, on CUDA)
            - mask         : [B, N, N]        (float32 or bool, on CUDA)
            - weights      : dict of tensors required by the module
            - config       : {"dim": dim, "hidden_dim": hidden_dim}

    Returns:
        out : Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim], float32

    # ------------------------------------------------------------------
    # Linear projections (bias = False)
    # ------------------------------------------------------------------
    # weight shape: [out, in] i.e. [hidden_dim, dim]
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask may be all‑ones when nomask=True)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones_like(input_tensor[..., 0])  # [B,N,N]
    mask_exp = mask.unsqueeze(-1).to(left.dtype)    # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy O(N³) multiplication
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Triton GEMM over the (i,j) axes per (batch, channel)
    # ------------------------------------------------------------------
    out_fp16 = tri_mul_gemm(left_fp16, right_fp16)   # [B,N,N,hidden_dim] FP16

    # Cast back to FP32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim) → gating → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
