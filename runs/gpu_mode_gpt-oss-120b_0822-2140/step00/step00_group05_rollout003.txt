"""
TriMul “outgoing” implementation.

The heavy N³ component is the tensor contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension *d* this is exactly a batched matrix‑multiply
   L_d @ R_dᵀ   (L_d,R_d : (B,N,N))
We fuse all hidden channels in one Triton kernel that computes many
independent GEMMs in parallel (group = B * hidden_dim).  The rest of the
module (LayerNorms, Linear projections, gates) is performed with
regular PyTorch – this keeps the code short while still moving the
N‑cubic work to fast Triton code.

The kernel works in float16 for bandwidth‑limited work and accumulates in
float32 to keep numerical quality.  The final result is cast back to
float32 as required by the specification.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
# A,B : (G, M, K)   where G = batch * hidden_dim
# C   : (G, M, N)   (here M == N == seq_len, K == seq_len)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers
    M, N, K,                        # matrix sizes
    stride_Ab, stride_Am, stride_An,  # strides for A (group, row, col)
    stride_Bb, stride_Bm, stride_Bn,  # strides for B (group, row, col) – note B will be read transposed
    stride_Cb, stride_Cm, stride_Cn,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_g = tl.program_id(0)               # group index (batch * hidden)
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block column

    # starting offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for the current group
    a_ptrs = A_ptr + pid_g * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_An
    b_ptrs = B_ptr + pid_g * stride_Bb + offs_k[:, None] * stride_Bm + offs_n[None, :] * stride_Bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] + k * BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] + k * BLOCK_K < K) & (offs_n[None, :] < N), other=0.0)
        # a : (BM, BK) , b : (BK, BN)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_An
        b_ptrs += BLOCK_K * stride_Bm

    # write back in FP16
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + pid_g * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input : Tensor[B, N, N, dim] (float32)
            - mask  : Tensor[B, N, N]   (bool/float) – may be None when `nomask` is True
            - weights: dict with all Linear / LayerNorm parameters
            - config : dict with ``dim`` and ``hidden_dim`` (both int)

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free) – use einsum for clarity
    # ------------------------------------------------------------------
    left = torch.einsum("b i j d, h d -> b i j h", x, weights["left_proj.weight"])
    right = torch.einsum("b i j d, h d -> b i j h", x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.einsum("b i j d, h d -> b i j h", x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.einsum("b i j d, h d -> b i j h", x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.einsum("b i j d, h d -> b i j h", x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) The N³ contraction – Triton GEMM (batch * hidden independent GEMMs)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    seq_len = N

    # work in FP16 for the matmul, accumulate in FP32 inside the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape to (G, M, K) where G = B * hidden, M = K = N
    G = B * hidden
    A = left_fp16.permute(0, 3, 1, 2).reshape(G, seq_len, seq_len).contiguous()
    B_mat = right_fp16.permute(0, 3, 1, 2).reshape(G, seq_len, seq_len).contiguous()
    C = torch.empty_like(A, dtype=torch.float16)

    # strides (in elements, not bytes) for the contiguous tensors
    stride_g, stride_m, stride_n = A.strides()  # (N*N, N, 1)

    # launch parameters – tuned for Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        G,                                 # groups (batch * hidden)
        triton.cdiv(seq_len, BLOCK_M),    # row blocks
        triton.cdiv(seq_len, BLOCK_N),    # col blocks
    )
    batched_matmul_kernel[grid](
        A, B_mat, C,
        seq_len, seq_len, seq_len,
        stride_g, stride_m, stride_n,
        stride_g, stride_m, stride_n,
        stride_g, stride_m, stride_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden]
    out = C.reshape(B, hidden, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear: weight shape (dim, hidden)
    out = torch.einsum("b i j h, d h -> b i j d", out, weights["to_out.weight"])
    return out.to(torch.float32)
