"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The expensive part of TriMul is the 3‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batch of hidden‑dimensional matrix
multiplications `left @ rightᵀ` for each hidden channel `d`.
The kernel below computes this product in Triton (float16 compute,
float32 accumulation) while the surrounding LayerNorm / Linear /
gating work is kept in PyTorch.  The final output is returned as
float32.

Supported shapes:
    B ∈ {1,2}, N ∈ {128,256,512,1024},   dim ∈ {128,384},
    hidden ∈ {128}
Both masked and un‑masked inputs are handled.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: one program instance works on a single (batch, hidden)
# channel and computes a BLOCK_M × BLOCK_N tile of the resulting matrix.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime sizes
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ  (sum over K) for a single head."""
    pid_m = tl.program_id(0)          # tile row  i
    pid_n = tl.program_id(1)          # tile col  j
    pid_h = tl.program_id(2)          # combined (batch, hidden) index

    # ----- decode batch and hidden index ---------------------------------
    batch = pid_h // H                # batch index
    h = pid_h % H                     # hidden (channel) index

    # ----- tile offsets ---------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j coordinates
    offs_k = tl.arange(0, BLOCK_K)                    # reduction axis

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ----- pointers to the current tile ------------------------------------
    # left : (M, K)  = left[batch, i, k, h]
    a_ptrs = (left_ptr
              + batch * stride_lb
              + h * stride_lh
              + offs_m[:, None] * stride_li
              + offs_k[None, :] * stride_lk)

    # right : (N, K) = right[batch, j, k, h]
    b_ptrs = (right_ptr
              + batch * stride_rb
              + h * stride_rh
              + offs_n[:, None] * stride_rj
              + offs_k[None, :] * stride_rk)

    # ----- accumulation ----------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_iter * BLOCK_K + offs_k
        cur_mask_k = cur_k < N

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & cur_mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & cur_mask_k[None, :],
                    other=0.0)

        # tl.dot does Σ_k a[i,k] * b[j,k] → (M,N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑block
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ----- write result ----------------------------------------------------
    c_ptrs = (out_ptr
              + batch * stride_ob
              + h * stride_oh
              + offs_m[:, None] * stride_oi
              + offs_n[None, :] * stride_oj)

    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm --------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ---------- Linear projections -----------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # -> hidden
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask -----------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(x.dtype)            # [B,N,N,1]
        left = left * m
        right = right * m

    # ---------- Gating ------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to fp16 for the heavy kernel -------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ---------- Allocate output tensor ---------------------------------------
    out_f16 = torch.empty(B, N, N, hidden,
                          dtype=torch.float16, device=device)

    # ---------- Strides (must match the kernel’s expectations) -------------
    stride_lb, stride_li, stride_lk, stride_lh = left_f16.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_f16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_f16.stride()

    # ---------- Triton launch ------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64   # good default for Hopper
    grid = (
        triton.cdiv(N, BLOCK_M),               # rows
        triton.cdiv(N, BLOCK_N),               # cols
        B * hidden,                            # one program per (batch, hidden)
    )
    trimul_outgoing_kernel[grid](
        left_f16, right_f16, out_f16,
        B, N, hidden,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ---------- Back to fp32 -------------------------------------------------
    out = out_f16.to(torch.float32)

    # ---------- Output LayerNorm & final linear -----------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
