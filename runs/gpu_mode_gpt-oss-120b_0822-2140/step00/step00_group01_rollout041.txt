"""
TriMul (outgoing) fused kernel.

The heavy part of the TriMul forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
operation.  For each hidden channel *d* this is a
batched matrix‑multiply of shape (N,N) × (N,N)ᵀ.
The kernel below computes this reduction for **one**
hidden channel at a time, using Triton’s tile‑based
matrix‑multiply (tl.dot).  All other cheap operations
(LayerNorm, Linear, gating, masking) are kept in PyTorch
so that they remain easy to verify and remain memory‑bound.

The launch grid is
    (⌈N/BM⌉, ⌈N/BN⌉, B*hidden_dim)
where BM/BN are the i/j tile sizes.  Each program instance
handles a single (batch, hidden) pair and a tile of the
(i,j) matrix.  The kernel respects out‑of‑bounds tiles via
masked loads/stores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel: one hidden channel per program instance
# -------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,                # *float32   left  [B,N,N,H]
    right_ptr,               # *float32   right [B,N,N,H]
    out_ptr,                 # *float32   out   [B,N,N,H]
    stride_b, stride_i, stride_k, stride_h,   # strides of left/right
    out_stride_b, out_stride_i, out_stride_j, out_stride_h,
    N, H,                    # sequence length & hidden dim (constexpr)
    BLOCK_M: tl.constexpr,   # tile size in i
    BLOCK_N: tl.constexpr,   # tile size in j
    BLOCK_K: tl.constexpr    # tile size in k (reduction)
):
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # combined batch & hidden

    batch = pid_bh // H               # batch index
    h = pid_bh % H                    # hidden channel

    # offsets for the current tile
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # -------- left: shape (M, K) --------
        # left[b, i, k, h]
        left_ptrs = left_ptr \
            + batch * stride_b \
            + i_off[:, None] * stride_i \
            + k_off[None, :] * stride_k \
            + h * stride_h
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # -------- right (transposed): shape (K, N) --------
        # right[b, j, k, h]  → we want (k, j) layout for matmul
        right_ptrs = right_ptr \
            + batch * stride_b \
            + k_off[:, None] * stride_k \
            + j_off[None, :] * stride_i \
            + h * stride_h
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0)

        # accumulate block matmul
        acc += tl.dot(left, right)   # (M,K) × (K,N) → (M,N)

    # -------- write result --------
    out_ptrs = out_ptr \
        + batch * out_stride_b \
        + i_off[:, None] * out_stride_i \
        + j_off[None, :] * out_stride_j \
        + h * out_stride_h
    tl.store(out_ptrs, acc,
             mask=i_mask[:, None] & j_mask[None, :])

# -------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   - Tensor [B, N, N, dim] (float32)
        mask    - Tensor [B, N, N] (bool/float)  or None
        weights - dict of pre‑trained weights (float32)
        config  - dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor [B, N, N, dim]  (float32)
    """
    # ---------------------------------------------------------
    # unpack
    # ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------------------------------------------------------
    # 1. LayerNorm on the input (dim = config["dim"])
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3. Apply mask (if present)
    # ---------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)          # [B,N,N,1] for broadcasting
    left = left * mask
    right = right * mask

    # ---------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise modulation
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Triton kernel: pairwise reduction over the “k” axis
    # ---------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # strides (contiguous, row‑major)
    stride_b, stride_i, stride_k, stride_h = left.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # tile sizes – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_b=stride_b,
        stride_i=stride_i,
        stride_k=stride_k,
        stride_h=stride_h,
        out_stride_b=out_stride_b,
        out_stride_i=out_stride_i,
        out_stride_j=out_stride_j,
        out_stride_h=out_stride_h,
        N=N,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------
    # 6. Output projection + final layer‑norm + gating
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                      # broadcasted over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out
