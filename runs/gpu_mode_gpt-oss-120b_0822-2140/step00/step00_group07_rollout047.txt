"""
custom_kernel
==============

Implements the *outgoing* TriMul operator (AlphaFold‑3).  
Only the heavy N³‑like contraction is moved to Triton – everything else
(LayerNorm, linear projections, sigmoids, final LayerNorm + linear) is done
with regular PyTorch which is already highly optimized on the target GPU.
The contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply of shape (B·H, N, N) × (B·H, N, N)ᵀ
where H = hidden_dim.  The Triton kernel computes this product block‑wise
using fp16 arithmetic and writes the result back in fp32 (the rest of the
pipeline expects fp32).

The kernel is launched with a 2‑D grid:
    dim0 – row‑blocks of the N×N output matrix
    dim1 – combined batch × hidden‑dim index (B·H)
which gives a very compact launch configuration and excellent occupancy on
Hopper (SM‑90) GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many (B*H) independent matrices
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_T_kernel(
    # Pointers
    a_ptr,                     # left  matrix  (B*H, N, N)
    b_ptr,                     # right matrix  (B*H, N, N)
    c_ptr,                     # output       (B*H, N, N)
    # Sizes
    N,                         # sequence length
    # Strides (packed as 6‑tuple for each tensor)
    stride_ah, stride_ai, stride_ak,   # a: (batch*H, i, k)
    stride_bh, stride_bi, stride_bk,   # b: (batch*H, i, k)
    stride_ch, stride_ci, stride_ck,   # c: (batch*H, i, j)
    # Block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ   where each operand is (B*H, N, N).
    A[i,k] and B[j,k] are multiplied and summed over k.
    """
    pid_col = tl.program_id(0)          # row‑block index (over N)
    pid_batch = tl.program_id(1)        # combined batch*hidden index

    # -----------------------------------------------------------------
    # Compute absolute batch and hidden‑dim indices from the linear pid
    # -----------------------------------------------------------------
    H = stride_ah // stride_ai           # hidden_dim = stride over hidden
    b = pid_batch // H                  # batch index
    h = pid_batch %  H                  # hidden‑dim index

    # -----------------------------------------------------------------
    # Offsets for the current tile of A and Bᵀ
    # -----------------------------------------------------------------
    offs_m = pid_col * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑dimension
    offs_n = tl.arange(0, BLOCK_N)                      # j‑dimension
    offs_k = tl.arange(0, BLOCK_K)                      # reduction (k)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers shifted to the current (b,h) slice
    a_base = a_ptr + b * stride_ah + h * stride_ai
    b_base = b_ptr + b * stride_bh + h * stride_bi

    # -----------------------------------------------------------------
    # Accumulator in FP32 (accumulate in higher precision)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # A: (i,k)   Bᵀ: (j,k) → we load B as (k,j) and later use it as transposed
        a_ptrs = a_base + (offs_m[:, None] * stride_ai) + (cur_k[None, :] * stride_ak)
        b_ptrs = b_base + (offs_n[None, :] * stride_bi) + (cur_k[:, None] * stride_bk)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # Accumulate a * b (notice b is already the transposed element)
        acc += tl.dot(a, b)

        # advance pointers (handled automatically by recomputing a_ptrs each iter)
    # -----------------------------------------------------------------
    # Write back the result tile
    # -----------------------------------------------------------------
    c_ptrs = c_ptr + b * stride_ch + h * stride_ci + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_ck)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Host‑side helper that launches the Triton kernel
# -------------------------------------------------------------------------
def gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, N, N, H)  float16
    returns       : (B, N, N, H)  float32
    """
    B, N, _, H = left.shape
    # Flatten batch & hidden dim → (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # strides for the flattened tensors
    stride_ah, stride_ai, stride_ak = left_flat.stride()
    stride_bh, stride_bi, stride_bk = right_flat.stride()
    stride_ch, stride_ci, stride_ck = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), B * H)

    batched_gemm_T_kernel[grid](
        left_flat, right_flat, out_flat,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bi, stride_bk,
        stride_ch, stride_ci, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, N, N, H) and up‑cast to fp32 for the rest of the pipeline
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()
    return out


# -------------------------------------------------------------------------
# Main entry point required by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape (batch, seq_len, seq_len, dim) in float32.
    """
    input_tensor, mask, weights, config = data

    # --------------------------------------------------------------
    # Configuration & convenient locals
    # --------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    # Use torch's functional layer_norm – fast and works on H200.
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(dtype)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    # torch.nn.functional.linear expects (…, in_features) -> (…, out_features)
    left = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)    # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (mask is [B,N,N]; we broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.unsqueeze(-1).to(dtype)        # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy N³ contraction – performed in fp16 on Triton
    # --------------------------------------------------------------
    # Cast to fp16 for the kernel (reduces memory traffic & improves
    # throughput on Hopper).  The kernel returns fp32.
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out = gemm_triton(left_f16, right_f16)   # (B,N,N,hidden_dim) fp32

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm, out‑gate, final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                     # gating

    to_out_w = weights["to_out.weight"].to(dtype)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)           # final projection -> (B,N,N,dim)

    return out
