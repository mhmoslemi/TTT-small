@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, D,
                  stride_b_left, stride_i_left, stride_k_left, stride_d_left,
                  stride_b_right, stride_i_right, stride_k_right, stride_d_right,
                  stride_b_out, stride_i_out, stride_j_out, stride_d_out,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)  # batch*D dimension
    pid_i = tl.program_id(1)  # tile i
    pid_j = tl.program_id(2)  # tile j

    b = pid_b // D
    d = pid_b % D

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left [b, i, k, d]
        left_ptrs = left_ptr + b*stride_b_left + offs_i[:,None]*stride_i_left + offs_k[None,:]*stride_k_left + d*stride_d_left

"""
TriMul “outgoing” implementation (AlphaFold3).

The heavy O(N³·C) part is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this contraction in a single Triton kernel.  The surrounding
LayerNorm / Linear / gating operations are kept in PyTorch (they are
tiny compared with the N³ work) and the kernel works in mixed precision:
inputs are cast to FP16, the accumulation is FP32 and the final tensor
remains FP32.

The kernel tiles the (i,j) plane (BLOCK_M×BLOCK_N) and accumulates over
tiles of the k‑dimension (BLOCK_K) with a classic GEMM‑style loop.
For every hidden channel *d* we launch a separate program (grid dimension
0 = batch*hidden).  The implementation is compatible with the
H200 (sm_90) and Triton 3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the 4‑D einsum
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # *FP16*  [B, N, N, D]
    right_ptr,         # *FP16*  [B, N, N, D]
    out_ptr,           # *FP32*  [B, N, N, D]

    B, N, D,           # static sizes

    # strides for left (b,i,k,d)
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,
    # strides for right (b,j,k,d)
    stride_b_r, stride_i_r, stride_k_r, stride_d_r,
    # strides for out   (b,i,j,d)
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 0‑th dimension of the grid encodes (batch, hidden)
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)          # = b * D + d
    pid_i = tl.program_id(1)          # tile over i
    pid_j = tl.program_id(2)          # tile over j

    b = pid0 // D
    d = pid0 % D

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, d]  -> (BLOCK_M, BLOCK_K)
        ptrs_l = (
            left_ptr
            + b * stride_b_l
            + offs_i[:, None] * stride_i_l
            + offs_k[None, :] * stride_k_l
            + d * stride_d_l
        )
        left = tl.load(ptrs_l,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[b, j, k, d] -> (BLOCK_N, BLOCK_K)
        ptrs_r = (
            right_ptr
            + b * stride_b_r
            + offs_j[:, None] * stride_i_r   # note: i‑dim of right is the “j” index
            + offs_k[None, :] * stride_k_r
            + d * stride_d_r
        )
        right = tl.load(ptrs_r,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # GEMM over the k‑dimension: (M,K) @ (N,K)ᵀ → (M,N)
        acc += tl.dot(left, right)   # tl.dot performs Σ_k left[i,k] * right[j,k]

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    ptrs_o = (
        out_ptr
        + b * stride_b_o
        + offs_i[:, None] * stride_i_o
        + offs_j[None, :] * stride_j_o
        + d * stride_d_o
    )
    tl.store(ptrs_o,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B,N,N,dim]  float32

    # --------------------------------------------------------------
    # 2) Linear projections  (dim → hidden)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # bias = None
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4) Gating (dim → hidden) + sigmoid
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triton contraction:  Σ_k left[...,k,d] * right[...,k,d]
    #    Result shape = [B, N, N, hidden]  (FP32)
    # --------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=input_tensor.device)

    # strides (as int64, matching Triton's expectations)
    l_str = left_f16.stride()   # (B,N,N,hidden) contiguous → (N*N*hidden, N*hidden, hidden, 1)
    r_str = right_f16.stride()
    o_str = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (
        B * hidden,                    # program_id(0)  = batch*hidden
        triton.cdiv(N, BLOCK_M),       # program_id(1)  = i‑tile
        triton.cdiv(N, BLOCK_N),       # program_id(2)  = j‑tile
    )

    trimul_kernel[grid](
        left_f16,
        right_f16,
        out,
        B, N, hidden,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6) Final LayerNorm on hidden, gate, and linear back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast over hidden dim
    output = F.linear(out, weights["to_out.weight"])   # hidden → dim, no bias

    return output.to(torch.float32)
