"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the einsum
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

For each batch b and hidden channel d this is a matrix multiplication:
    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T

We keep the surrounding LayerNorms / Linear / Sigmoid in PyTorch and
implement the batched GEMM in Triton.  The kernel works on FP16
matrices (the hidden dimension is at most 384) and accumulates in
FP32, giving a good speed/accuracy trade‑off on Hopper (sm_90).

Input:
    data = (x, mask, weights, config)
where
    x : [B, N, N, dim]        (float32)
    mask : [B, N, N]          (float32, 0/1)
    weights : dict of tensors (float32)
    config : {"dim": dim, "hidden_dim": hidden_dim}

Output:
    y : [B, N, N, dim]        (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul core
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float16 *
    B, N, H,           # problem sizes
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_j_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = left[b,i,k,h] * right[b,j,k,h] summed over k."""
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile col
    pid_bc = tl.program_id(2)              # combined batch*channel

    b = pid_bc // H                        # batch index
    h = pid_bc % H                         # hidden channel index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the block for this (b, h)
    left_base = (
        left_ptr
        + b * stride_b_left
        + h * stride_h_left
    )
    right_base = (
        right_ptr
        + b * stride_b_right
        + h * stride_h_right
    )
    out_base = (
        out_ptr
        + b * stride_b_out
        + h * stride_h_out
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left:   shape (M, K)   -> (i,k)
        a_ptrs = left_base \
                 + offs_m[:, None] * stride_i_left \
                 + k_off[None, :] * stride_k_left
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right: we need (K, N) = right[j,k] with k as first dim.
        b_ptrs = right_base \
                 + k_off[:, None] * stride_k_right \
                 + offs_n[None, :] * stride_j_right
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write result
    out_ptrs = out_base \
               + offs_m[:, None] * stride_i_out \
               + offs_n[None, :] * stride_j_out
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm (no bias in the forward reference code, but we have bias)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_hat = (x - mean) / torch.sqrt(var + eps)
    if weight is not None:
        x_hat = x_hat * weight
    if bias is not None:
        x_hat = x_hat + bias
    return x_hat


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with the core multiplication
    off‑loaded to a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (all stored as float32)
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]                # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # Normalisation of the input
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor, norm_w, norm_b, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Apply mask (if any).  mask is [B,N,N] – broadcast on hidden dim.
    # ------------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1).to(left.dtype)
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul: batched GEMM per hidden channel using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Cast to fp16 for the kernel (hidden dim ≤ 384 → safe)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, device=device)

    # stride information (contiguous layout)
    stride_b_left, stride_i_left, stride_k_left, stride_h_left = left_fp16.stride()
    stride_b_right, stride_j_right, stride_k_right, stride_h_right = right_fp16.stride()
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out_fp16.stride()

    # Tiling parameters – chosen to fit in shared memory on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H,                            # batch * hidden
    )

    trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        stride_b_left, stride_i_left, stride_k_left, stride_h_left,
        stride_b_right, stride_j_right, stride_k_right, stride_h_right,
        stride_b_out, stride_i_out, stride_j_out, stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # good default for fp16 GEMM
    )
    # Back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = _layer_norm(out, to_out_norm_w, to_out_norm_b, eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (dim, hidden_dim) weight, no bias

    return out
