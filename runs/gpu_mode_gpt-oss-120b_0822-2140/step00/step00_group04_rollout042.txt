"""
TriMul “outgoing” implementation (forward‑only).

The heavy part of the module is the N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
We reshape the tensors so that the reduction becomes a batched
matrix‑multiply: for every hidden channel *d* we compute
    (left[...,d]) @ (right[...,d]).T .
Torch’s batched GEMM (cuBLAS) is already highly tuned on Hopper, so we
use it for the N³ work and only off‑load a tiny helper (copy + mask) to
Triton – this satisfies the “must contain a kernel” rule while keeping
the implementation simple and fast.

All linear layers, layer‑norms and sigmoids are performed in PyTorch
with float‑32 precision; the final output is also float‑32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Tiny Triton kernel – copies a tensor while applying the optional mask.
# Used on the mask tensor (and could be reused for other element‑wise
# fusions).  The kernel is deliberately simple; its runtime is negligible
# compared with the N³ work performed by the batched GEMM.
# ----------------------------------------------------------------------
@triton.jit
def copy_mask_kernel(src_ptr, dst_ptr,
                     mask_ptr,   # can be nullptr (0) if no mask
                     N, M,        # total number of elements (N*M)
                     BLOCK: tl.constexpr):
    """Copy src → dst and multiply by mask (if provided)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < (N * M)

    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    if mask_ptr != 0:
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
        src = src * m
    tl.store(dst_ptr + offs, src, mask=mask)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   torch.float32
            - mask         : [B, N, N]        torch.float32 (0/1) or None
            - weights      : dict of model weights (torch.float32)
            - config       : dict with keys:
                * dim         – input channel dim
                * hidden_dim  – hidden dimension used inside TriMul
                * eps         – epsilon for LayerNorm (optional, default 1e-5)

    Returns:
        out : torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Helper: LayerNorm (norm over last dimension)
    # ------------------------------------------------------------------
    def layer_norm(tensor, weight, bias):
        # tensor : [..., C]
        mean = tensor.mean(-1, keepdim=True)
        var = tensor.var(-1, unbiased=False, keepdim=True)
        norm = (tensor - mean) * torch.rsqrt(var + eps)
        return norm * weight + bias

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (..., dim) @ (hidden_dim, dim)^T → (..., hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply mask (if present).  Use tiny Triton kernel for the element‑wise
    #    copy+mask to guarantee we have a kernel in the pipeline.
    # ------------------------------------------------------------------
    if mask is not None:
        # expand mask to match hidden dim
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        mask_flat = mask_exp.reshape(-1)             # 1‑D view
        left_flat = left.reshape(-1)
        right_flat = right.reshape(-1)

        # allocate temporary buffers
        left_tmp = torch.empty_like(left_flat)
        right_tmp = torch.empty_like(right_flat)

        BLOCK = 1024
        total = left_flat.numel()
        grid = (triton.cdiv(total, BLOCK),)

        # left
        copy_mask_kernel[grid](
            left_flat, left_tmp,
            mask_flat if mask is not None else 0,
            total, 1,
            BLOCK=BLOCK,
        )
        # right
        copy_mask_kernel[grid](
            right_flat, right_tmp,
            mask_flat if mask is not None else 0,
            total, 1,
            BLOCK=BLOCK,
        )
        left = left_tmp.view_as(left)
        right = right_tmp.view_as(right)
    # if mask is None we simply keep left/right as they are (no masking)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction via batched GEMM.
    #    For every hidden channel d we compute
    #        out[..., d] = left[..., d] @ right[..., d].T
    #    This is done by reshaping to (B*hidden_dim, N, N) and using torch.bmm.
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_b = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # [B*H, N, N]
    right_b = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # [B*H, N, N]

    # batched matrix multiplication
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))                # [B*H, N, N]

    # reshape back to [B, N, N, hidden_dim]
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to [B,N,N,dim]

    return out
