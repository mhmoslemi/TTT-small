"""
TriMul "outgoing" kernel (AlphaFold3)

The heavy part of the TriMul forward pass is the ternary reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched Gram‑matrix over the *k* dimension for every hidden
channel *d*.  The implementation below keeps the preprocessing (LayerNorm,
Linear projections and gating) in PyTorch and off‑loads the N³ reduction to a
single Triton kernel.  The kernel works on blocks of (i,j,d) while iterating
over *k* in tiles, using `tl.dot` to perform the inner‑product for each hidden
slice.  The kernel is launched per‑batch (batch size ≤ 2 in the test suite)
to keep the launch logic simple.

The final output is cast back to float32, layer‑normalised, gated and
projected back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel that computes the N³ reduction for one batch:
# out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const float*  (N, N, H)
    right_ptr,              # const float*  (N, N, H)
    out_ptr,                # float*        (N, N, H)
    N, H,                   # sizes
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size for i
    BLOCK_N: tl.constexpr,  # tile size for j
    BLOCK_K: tl.constexpr,  # reduction tile size over k
    BLOCK_D: tl.constexpr   # tile size for hidden dimension
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = pid_d * BLOCK_D

    # offsets inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left[i, k, d]  -> shape (M, K, D)
        # Load right[j, k, d] -> shape (N, K, D)
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + (
            offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        right_ptrs = right_ptr + (
            offs_j[:, None, None] * stride_right_i
            + offs_k[None, :, None] * stride_right_k
            + offs_d[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # For each hidden slice d, compute the outer‑product over k:
        #   acc[i, j, d] += Σ_k left[i,k,d] * right[j,k,d]
        # This is exactly a matrix multiply: (M×K) • (N×K)ᵀ  -> (M×N)
        # ------------------------------------------------------------------
        for d in range(0, BLOCK_D):
            # (M, K) and (N, K)
            left_slice = left[:, :, d]          # (M, K)
            right_slice = right[:, :, d]        # (N, K)
            # dot with transposed right (K, N) → (M, N)
            acc[:, :, d] += tl.dot(left_slice, right_slice, trans_b=True)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + (
        offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing’’ operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – torch.Tensor   [B, N, N, dim]
        mask   – torch.Tensor   [B, N, N]  (may be all‑ones)
        weights – dict of torch.Tensor weight/bias tensors
        config  – dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # Linear expects (…, in_features) and weight of shape (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (mask is [B,N,N]; we broadcast on hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Triton kernel: compute the N³ reduction per batch
    # ------------------------------------------------------------------
    # Prepare output container
    out = torch.empty_like(left)          # shape [B,N,N,hidden]

    # constants for tiling (tuned for Hopper SM90)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 32

    # launch per‑batch (B ≤ 2)
    for b in range(B):
        left_b = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b = out[b]

        # strides in element units (not bytes)
        s_li, s_lk, s_lh = left_b.stride()
        s_ri, s_rk, s_rh = right_b.stride()
        s_oi, s_oj, s_oh = out_b.stride()

        grid = (
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            triton.cdiv(hidden_dim, BLOCK_D),
        )
        trimul_outgoing_kernel[grid](
            left_b.data_ptr(),
            right_b.data_ptr(),
            out_b.data_ptr(),
            N,
            hidden_dim,
            s_li,
            s_lk,
            s_lh,
            s_ri,
            s_rk,
            s_rh,
            s_oi,
            s_oj,
            s_oh,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
            num_warps=4,
        )

    # ------------------------------------------------------------------
    # 6️⃣  Post‑processing: LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    # to_out_norm works on hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate.unsqueeze(-1)          # broadcast gate on hidden

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
