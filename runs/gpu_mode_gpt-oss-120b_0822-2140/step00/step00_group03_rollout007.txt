"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of TriMul is the three‑dimensional
einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where *left* and *right* have shape
[batch, seq_len, seq_len, hidden_dim].

For each hidden dimension *d* this is a batched matrix
multiplication:

    out_d = left_d @ right_d.T          # (N,N) @ (N,N) → (N,N)

The Triton kernel below computes this product block‑wise.
All other linear layers, LayerNorms and gates are performed
with regular PyTorch (they are cheap compared with the N³
multiplication).

The kernel works for any batch size (1 or 2), any sequence
length up to 1024 and any hidden dimension that is a multiple
of BLOCK_D (the kernel also silently pads the last block).
The final output is returned in float‑32 as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out = Σ_k left[...,k,d] * right[...,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const *float16
    right_ptr,         # const *float16
    out_ptr,           # *float32
    N,                 # seq_len (int)
    H,                 # hidden_dim (int)
    stride_l_i, stride_l_k, stride_l_d,   # left strides
    stride_r_j, stride_r_k, stride_r_d,   # right strides
    stride_o_i, stride_o_j, stride_o_d,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # program ids identify a tile (i‑block, j‑block, d‑block)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = pid_d * BLOCK_D

    # ----- masks for the tails -----
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N
    mask_d = offs_d < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(0, num_k_tiles):
        k0 = tile_k * BLOCK_K

        # real offsets for this tile
        cur_k = k0 + offs_k
        mask_k_tile = cur_k < N

        # ---- load left : shape (M, K, D) ----
        left_ptrs = (
            left_ptr
            + offs_i[:, None, None] * stride_l_i
            + cur_k[None, :, None] * stride_l_k
            + offs_d[None, None, :] * stride_l_d
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k_tile[None, :, None] & mask_d[None, None, :],
            other=0.0,
            eviction_policy="evict_last",
        )  # fp16 -> automatically cast to fp32 when used

        # ---- load right transposed : shape (K, N, D) ----
        # right is indexed as (..., j, k, d) ; we need (k, j, d)
        right_ptrs = (
            right_ptr
            + cur_k[:, None, None] * stride_r_k
            + offs_j[None, :, None] * stride_r_j
            + offs_d[None, None, :] * stride_r_d
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k_tile[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
            eviction_policy="evict_last",
        )  # fp16

        # ---- inner loop over hidden dimension blocks ----
        for d_idx in range(0, BLOCK_D):
            # skip if out‑of‑range in hidden dim
            if mask_d[d_idx] == 0:
                continue
            l_mat = left[:, :, d_idx]   # (M, K)
            r_mat = right[:, :, d_idx]   # (K, N)
            acc[:, :, d_idx] += tl.dot(l_mat, r_mat)   # (M,N) → fp32

    # ------------------------------------------------------------------
    # Store the result (float32)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_d[None, None, :] * stride_o_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32)
            - mask         : [B, N, N]        (float32 or bool)
            - weights      : dict of model parameters
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # still float32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – we keep them in FP16 for the
    #    heavy part
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden, dim)

    left = F.linear(x, left_proj_w)   # [B,N,N,hidden]  float32
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Apply mask if present
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to FP16 and run the Triton kernel
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output of the einsum (still FP32, will be cast later)
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements) for the 4‑D tensors (batch is handled outside)
    stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()[1:]   # i,k,d
    stride_r_j, stride_r_k, stride_r_d = right_fp16.stride()[1:]
    stride_o_i, stride_o_j, stride_o_d = out.stride()[1:]

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32      # hidden_dim is a multiple of 32 in all test configs

    for b in range(B):
        # pointers to the start of the current batch
        left_ptr  = left_fp16.data_ptr()  + b * left_fp16.stride(0) * left_fp16.element_size()
        right_ptr = right_fp16.data_ptr() + b * right_fp16.stride(0) * right_fp16.element_size()
        out_ptr   = out.data_ptr()        + b * out.stride(0) * out.element_size()

        grid = (
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            triton.cdiv(hidden, BLOCK_D),
        )
        trimul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden,
            stride_l_i, stride_l_k, stride_l_d,
            stride_r_j, stride_r_k, stride_r_d,
            stride_o_i, stride_o_j, stride_o_d,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, to_out_w)            # back to dim, no bias
    return out.to(dtype)                     # ensure same dtype as input (float32)
