"""
TriMul “outgoing” implementation with a custom Triton kernel.

The heavy N³ part is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is exactly a batch of independent matrix‑multiplications:
    left   : (B, H, N, N)  with axes (i,k)
    right  : (B, H, N, N)  with axes (j,k) → transpose to (k,j)
    out    : (B, H, N, N)  = matmul(left, rightᵀ)

We flatten the (B·H) dimension and run a batched GEMM completely in Triton.
All other ops (LayerNorm, Linear, sigmoid gates, final projection) are kept in
PyTorch for simplicity.  The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (M=N=seq_len)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N,                         # M = B*H, N = seq_len
    stride_am, stride_ak, stride_an,   # A strides  (m,k,n)
    stride_bm, stride_bk, stride_bn,   # B strides  (m,k,n)  (B is already transposed)
    stride_cm, stride_cn,              # C strides  (m,n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----- block coordinates ------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K -------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A tile   (M x K)
        a_ptrs = a_ptr + (offs_m[:, None] * stride_am
                          + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < N),
                    other=0.0)

        # load B tile   (K x N)  (right already transposed)
        b_ptrs = b_ptr + (offs_k[:, None] * stride_bk
                          + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < N) & mask_n,
                    other=0.0)

        acc += tl.dot(a, b)

    # ----- write back --------------------------------------------------------
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm
                      + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : Tensor[B, N, N, D]  (float32/float16)
            - mask  : Tensor[B, N, N]    (bool or 0/1)
            - weights: dict with the named weight tensors
            - config : dict containing "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, D] in float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden_dim

    # --------------------------------------------------------------
    # 1. LayerNorm on the input
    # --------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(device))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Move hidden dimension to leading axis for the batch GEMM
    # --------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*hidden -> M
    M = B * H
    left  = left.view(M, N, N)                               # (M,N,N)
    # right needs to be transposed on the last two dims (k,j)
    right = right.transpose(2, 3).contiguous().view(M, N, N)   # (M,N,N)

    # --------------------------------------------------------------
    # 6. Triton GEMM
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    out = torch.empty_like(left)   # same dtype & device, shape (M,N,N)

    _tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        M, N,
        left.stride(0), left.stride(1), left.stride(2),   # A strides
        right.stride(0), right.stride(1), right.stride(2),   # B strides
        out.stride(0), out.stride(2),                     # C strides (m,n)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7. Reshape back to (B,N,N,H)
    # --------------------------------------------------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gate and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=eps,
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(device))   # (B,N,N,D)

    # Ensure final dtype is float32
    return out.to(torch.float32)
