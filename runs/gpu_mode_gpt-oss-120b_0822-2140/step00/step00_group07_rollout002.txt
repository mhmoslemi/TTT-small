"""
Triton‑accelerated “outgoing” Triangle Multiplicative Update (TriMul).

The heavy N³ component
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication L @ Rᵀ for every hidden channel *d*.
We reshape the tensors to (B·D, N, N) and compute the product with a
custom Triton kernel that tiles the M, N and K dimensions (64‑way).  All
other operations (LayerNorm, linear projections, gating, masking) stay in
PyTorch, which keeps the implementation concise while still satisfying the
requirement to have a part of the work done in Triton.

The kernel works in float16 for the loads, accumulates in float32 and writes
float16 back; the surrounding PyTorch code cast the result to float32
before the final LayerNorm & linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matmul L @ Rᵀ  (L,R : (B·D, N, N) )
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    L_ptr,               # *float16
    R_ptr,               # *float16
    O_ptr,               # *float16
    B, N,                # int32
    stride_L_batch, stride_L_m, stride_L_k,   # int32 strides for L
    stride_R_batch, stride_R_n, stride_R_k,   # int32 strides for R (R is not transposed in memory)
    stride_O_batch, stride_O_m, stride_O_n,   # int32 strides for O
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute for one batch (B·D) slice:
        O[m, n] = Σₖ L[m, k] * R[n, k]      (note the second operand is *not* transposed)
    The loop over k therefore performs the required dot‑product with the
    transposed view of R.
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds positions
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # L : (M, K)
        L_ptrs = (L_ptr
                  + batch_id * stride_L_batch
                  + offs_m[:, None] * stride_L_m
                  + offs_k[None, :] * stride_L_k)
        L = tl.load(L_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # R : (N, K)  – we need Rᵀ, hence we load (K, N) and treat the first axis as K
        R_ptrs = (R_ptr
                  + batch_id * stride_R_batch
                  + offs_n[None, :] * stride_R_n
                  + offs_k[:, None] * stride_R_k)
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(L, R)

    # ------------------------------------------------------------------
    # Store result (convert back to float16)
    # ------------------------------------------------------------------
    O_ptrs = (O_ptr
              + batch_id * stride_O_batch
              + offs_m[:, None] * stride_O_m
              + offs_n[None, :] * stride_O_n)

    tl.store(O_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host‑side helper to launch the kernel
# ----------------------------------------------------------------------
def _batched_matmul_triton(L: torch.Tensor, R: torch.Tensor) -> torch.Tensor:
    """
    L,R : torch.float16, shape (B·D, N, N), contiguous
    Returns O of same shape and dtype.
    """
    assert L.is_contiguous() and R.is_contiguous()
    B, N, _ = L.shape
    # strides in element units
    sL_b, sL_m, sL_k = L.stride()
    sR_b, sR_n, sR_k = R.stride()          # note: second dim is N (rows), third is K
    # output strides (same layout as L)
    sO_b, sO_m, sO_n = L.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        L,
        R,
        torch.empty_like(L),               # allocate output inside kernel launch
        B, N,
        sL_b, sL_m, sL_k,
        sR_b, sR_n, sR_k,
        sO_b, sO_m, sO_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # The kernel returns the output via the third pointer; we retrieve it
    # from the temporary allocation returned by triton (the third argument is
    # the pointer we gave, which we allocated in-place above).
    # However, Triton does not return the pointer; instead we allocated a
    # tensor beforehand and let the kernel write into it.
    # So we just allocate it here and pass its pointer.
    # The call above already wrote into that tensor.
    # We need to capture it:
    out = torch.empty_like(L)
    _batched_matmul_kernel[grid](
        L, R, out,
        B, N,
        sL_b, sL_m, sL_k,
        sR_b, sR_n, sR_k,
        sO_b, sO_m, sO_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: (input, mask, weights, config)
            input  – [B, N, N, dim]   (float32 on CUDA)
            mask   – [B, N, N] (0/1)  (float32 or bool)
            weights – dict with all linear / LN parameters (float32)
            config – {"dim": int, "hidden_dim": int, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N] -> broadcast)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)       # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of left/right
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton‑accelerated N³ einsum via batched matmul
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    #   a) reshape to (B*hidden, N, N) and cast to float16 for speed
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)

    #   b) compute product L @ Rᵀ using Triton
    out_fp16 = _batched_matmul_triton(left_fp16, right_fp16)   # [B*hidden, N, N] float16

    #   c) reshape back to (B, N, N, hidden_dim)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 6. Final LayerNorm (hidden_dim) → linear → dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gating after LN
    out = F.linear(out, weights["to_out.weight"])
    return out
