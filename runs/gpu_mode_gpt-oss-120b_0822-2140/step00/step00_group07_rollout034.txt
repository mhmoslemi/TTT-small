"""
TriMul (outgoing) – fast Triton implementation of the N³ einsum.
The kernel computes, for every batch b, positions i, j and hidden
dimension d:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is equivalent to `out_d = left_d @ right_d.T` for each hidden
slice d.  All cheap LayerNorm / Linear / gating work is done in
PyTorch; only the heavy N³ accumulation is executed on the GPU with
Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Batched matrix‑multiply `left @ right.T` per hidden slice.

    left  : [B, N, N, H]   (float16)
    right : [B, N, N, H]   (float16)
    out   : [B, N, N, H]   (float32)
    """
    pid_b = tl.program_id(0)                 # batch dimension
    pid_i = tl.program_id(1)                 # output i‑tile
    pid_j = tl.program_id(2)                 # output j‑tile

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # ------------------------------------------------------------------
    # accumulator for a tile (i, j, h)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # iterate over the reduction dimension k
    # ------------------------------------------------------------------
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_outer * BLOCK_K

        # ----- indices -------------------------------------------------
        i_idx = i_base + tl.arange(0, BLOCK_I)[:, None, None]   # (I,1,1)
        j_idx = j_base + tl.arange(0, BLOCK_J)[None, :, None]   # (1,J,1)
        k_idx = k_base + tl.arange(0, BLOCK_K)[None, None, :]   # (1,1,K)
        h_idx = tl.arange(0, BLOCK_H)[None, None, :]           # (1,1,H)

        # ----- masks ---------------------------------------------------
        i_mask = i_idx < N
        j_mask = j_idx < N
        k_mask = k_idx < N
        h_mask = h_idx < H
        load_mask = i_mask & k_mask & h_mask               # for left
        load_mask_r = j_mask & k_mask & h_mask             # for right

        # ----- load tiles (float16) ------------------------------------
        left_off = (
            pid_b * stride_left_b
            + i_idx * stride_left_i
            + k_idx * stride_left_k
            + h_idx * stride_left_h
        )
        left_tile = tl.load(left_ptr + left_off, mask=load_mask, other=0.0)

        right_off = (
            pid_b * stride_right_b
            + j_idx * stride_right_j
            + k_idx * stride_right_k
            + h_idx * stride_right_h
        )
        right_tile = tl.load(right_ptr + right_off, mask=load_mask_r, other=0.0)

        # ----- compute dot products per hidden slice -------------------
        # left_tile : (I, K, H)
        # right_tile: (J, K, H)
        # we need Σ_k left[i,k,h] * right[j,k,h]  →  left @ right.T per h
        for dh in range(0, BLOCK_H):
            # (I, K) @ (J, K).T  → (I, J)
            # tl.dot expects (M, K) and (N, K); set trans_b=True to flip.
            dot_res = tl.dot(
                left_tile[:, :, dh],
                right_tile[:, :, dh],
                trans_b=True,
            )
            # cast to fp32 before accumulation
            acc[:, :, dh] += dot_res.to(tl.float32)

    # ------------------------------------------------------------------
    # write the result back
    # ------------------------------------------------------------------
    out_off = (
        pid_b * stride_out_b
        + i_idx * stride_out_i
        + j_idx * stride_out_j
        + h_idx * stride_out_h
    )
    store_mask = i_mask & j_mask & h_mask
    tl.store(out_ptr + out_off, acc, mask=store_mask)


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim] (float32)
            - mask         : [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (torch tensors)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy N³ accumulation via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # cast to fp16 for the kernel (still exact enough)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output in fp32 (accumulator is fp32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out.stride()

    # launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32                     # compile‑time constant; H may be larger

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],          # same as stride_i
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])

    return output
