"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 3‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    out[b,d] = left[b,d] @ right[b,d].T
for every hidden channel *d*.  The kernel below performs this
batched GEMM in Triton (float32 accumulation, float16 loads for speed
on Hopper).  All other operations – LayerNorm, linear projections,
gating and the final linear – are done with regular PyTorch kernels,
so the code stays simple while the N³ work is offloaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) over the hidden dimension
# ------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # problem size
    stride_az, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bz, stride_bk, stride_bn,   # B: (batch, K, N)  (B is stored as Bᵀ)
    stride_cz, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    """
    Each program instance works on a (BLOCK_M x BLOCK_N) tile of the
    output matrix for a specific batch index (pid_z).
    """
    pid_z = tl.program_id(2)          # batch * hidden index
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptrs = A_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_ak          # advance K dimension in A
        b_ptrs += BLOCK_K * stride_bk          # advance K dimension in B

    # write back
    c_ptrs = C_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ------------------------------------------------------------
# Helper: LayerNorm (no torch.nn overhead)
# ------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:         torch.Tensor [B, N, N]   (float32 or bool)
            - weights: dict of pre‑loaded parameters
            - config:  dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------
    # 3. Apply optional mask (mask is broadcastable)
    # --------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)                # [B, N, N, 1]
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # --------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------
    # 5. Core TriMul -> batched GEMM via Triton
    # --------------------------------------------------------
    # reshape to (B*hidden, N, N) so each “batch” in the kernel is one hidden channel
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # output buffer
    out_reshaped = torch.empty_like(left_reshaped)

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden,                       # batch*hidden
    )

    # strides (in elements) for the three tensors
    stride_az, stride_am, stride_ak = (
        left_reshaped.stride(0),
        left_reshaped.stride(1),
        left_reshaped.stride(2),
    )
    # right is stored as (batch, K, N) because we need its transpose; we simply
    # treat it as (batch, K, N) where K == N and N == N.
    stride_bz, stride_bk, stride_bn = (
        right_reshaped.stride(0),
        right_reshaped.stride(2),   # note: we want B[k, n] → stride over K
        right_reshaped.stride(1),   # stride over N (the column after transpose)
    )
    stride_cz, stride_cm, stride_cn = (
        out_reshaped.stride(0),
        out_reshaped.stride(1),
        out_reshaped.stride(2),
    )

    batched_gemm_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_reshaped.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------
    # 6. Output LayerNorm + gating
    # --------------------------------------------------------
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------
    # 7. Final linear (hidden -> dim)
    # --------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
