"""
TriMul "outgoing" implementation (AlphaFold‑3) with a fused Triton
batched‑matrix‑multiply kernel.

The heavy part of the module is the einsum:
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
which is a batched matrix multiplication for every hidden channel d.
The kernel below computes this product in FP16 with a FP32 accumulator,
while the surrounding LayerNorm / Linear / gating logic stays in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------
# Triton kernel: batched matmul over the (i,k) × (j,k) dimensions.
# One program processes a TILE_M × TILE_N block of the output for a
# specific (batch, hidden) pair.
# --------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    a_ptr,               # left  [B, N, N, H]  (FP16)
    b_ptr,               # right [B, N, N, H]  (FP16)
    c_ptr,               # out   [B, N, N, H]  (FP16)
    N, H,                # sequence length, hidden dimension
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,               # compile‑time hidden size
):
    pid_m = tl.program_id(0)               # i‑tile
    pid_n = tl.program_id(1)               # j‑tile
    pid_z = tl.program_id(2)               # (batch, hidden) linear index

    # --------------------------------------------------------------
    # Decode (batch, hidden) from pid_z
    # --------------------------------------------------------------
    batch = pid_z // HIDDEN
    hidden = pid_z % HIDDEN

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers to the start of the matrices for this (batch, hidden)
    a_batch_ptr = a_ptr + batch * stride_a_batch + hidden * stride_a_h
    b_batch_ptr = b_ptr + batch * stride_b_batch + hidden * stride_b_h
    c_batch_ptr = c_ptr + batch * stride_c_batch + hidden * stride_c_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K (the third axis of the
    # original tensor).
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + offs_k
        cur_mask_k = mask_k & (k_offset < N)

        # Load A[i, k]
        a_ptrs = a_batch_ptr + \
                 offs_m[:, None] * stride_a_i + \
                 k_offset[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & cur_mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B[j, k]  (note that we will dot with transposed B)
        b_ptrs = b_batch_ptr + \
                 offs_n[:, None] * stride_b_i + \
                 k_offset[None, :] * stride_b_k
        b = tl.load(b_ptrs, mask=mask_n[:, None] & cur_mask_k[None, :], other=0.0)
        b = b.to(tl.float32)

        # acc += A * B   (sum over k).  Use trans_b=True because B is (N, K)
        acc += tl.dot(a, b, trans_b=True)

    # --------------------------------------------------------------
    # Write the result back (FP16)
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + \
             offs_m[:, None] * stride_c_i + \
             offs_n[None, :] * stride_c_k
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :]
    )


# --------------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – pure‑forward pass.
    Arguments:
        data = (input, mask, weights, config)
            input   – tensor [B, N, N, C]  (float32)
            mask    – tensor [B, N, N]    (bool/float)
            weights – dict of all linear / layer‑norm parameters
            config  – dict containing "dim", "hidden_dim", and "nomask"
    Returns:
        output tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, C = input_tensor.shape          # C == dim
    H = config["hidden_dim"]                 # hidden_dim
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (C)
    # ----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # ----------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)      # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # ----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5️⃣ Core cubic update – batched matmul via Triton
    #    Convert to FP16 for bandwidth‑limited compute.
    # ----------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h   = torch.empty_like(left_h)          # result in FP16

    # Compute strides for the contiguous layout [B, N, N, H]
    a_s = left_h.stride()   # (B, i, k, h)
    b_s = right_h.stride()
    c_s = out_h.stride()

    # Tile sizes – can be tuned, 64 works well on Hopper.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),            # i‑tiles
        triton.cdiv(N, BLOCK_N),            # j‑tiles
        B * H,                               # (batch, hidden) pairs
    )

    trimul_matmul_kernel[grid](
        left_h, right_h, out_h,
        N, H,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=H,                           # compile‑time constant
    )
    torch.cuda.synchronize()

    out = out_h.to(torch.float32)

    # ----------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating and final linear projection
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to C dimensions

    return out
