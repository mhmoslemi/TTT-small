"""
TriMul "outgoing" kernel (AlphaFold3) – highly‑optimized Triton implementation.

The heavyweight operation is the 4‑D einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalently a batched GEMM per hidden channel:
    out_d = left_d @ right_d.T   (left_d : (B,N,N), right_d : (B,N,N))

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch (fast
autograd‑free ops) and off‑load only the N×N×N multiplication to a Triton
kernel that tiles the two spatial dimensions (i,j) and the reduction dimension
(k).  Hidden dimension (d) is tiled separately (BLOCK_H) and accumulated in
FP32.  The kernel works for any batch size (1‑2), sequence length up to 1024
and hidden dimensions that are multiples of BLOCK_H (we pad internally).

The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM across the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                 # [B, N, N, H]  (i,k,h)
    right_ptr,                # [B, N, N, H]  (k,j,h)   – k major for dot
    out_ptr,                  # [B, N, N, H]  (i,j,h)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,   # tile size for i
    BLOCK_N: tl.constexpr,   # tile size for j
    BLOCK_K: tl.constexpr,   # tile size for k (reduction)
    BLOCK_H: tl.constexpr,   # tile size for hidden dimension
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #  Grid layout: (batch, i_tile, j_tile, h_tile)
    # --------------------------------------------------------------
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    total_h = tl.cdiv(H, BLOCK_H)

    pid_b = pid // (total_i * total_j * total_h)
    rem   = pid % (total_i * total_j * total_h)

    pid_i = rem // (total_j * total_h)
    rem   = rem % (total_j * total_h)

    pid_j = rem // total_h
    pid_h = rem % total_h

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    #  K‑reduction loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left : (M, K, H) -----
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_b
            + offs_i[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----- right : (K, N, H) -----
        # note: we load with k as the first dimension so we can use tl.dot
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_b
            + offs_k[:, None, None] * stride_r_k
            + offs_j[None, :, None] * stride_r_j
            + offs_h[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----- inner product per hidden slice -----
        for hi in range(0, BLOCK_H):
            # left[:, :, hi] : (M, K)
            # right[:, :, hi] : (K, N)
            acc[:, :, hi] += tl.dot(left[:, :, hi], right[:, :, hi])

    # --------------------------------------------------------------
    #  Write back the result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_h[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
#  Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator, using a Triton kernel for
    the N×N×N reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, dim] (float32)
              - mask:          torch.Tensor [B, N, N] (bool / 0‑1)
              - weights:       dict of leaf tensors (float32)
              - config:        dict containing:
                       * dim, hidden_dim, nomask (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    #  LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    #  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    #  Mask handling (skip if nomask)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    #  Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  Triton GEMM across (i,j) with reduction over k for every hidden dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # allocate output tensor for the intermediate result (shape [B,N,N,hidden])
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides in elements (not bytes) – needed by Triton
    stride_l = left.stride()   # (b,i,k,h)
    stride_r = right.stride()  # (b,i,k,h) – we will reinterpret as (b,k,j,h)
    stride_o = out.stride()    # (b,i,j,h)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 32                     # hidden_dim is a multiple of 32 in all tests

    total_i = triton.cdiv(N, BLOCK_M)
    total_j = triton.cdiv(N, BLOCK_N)
    total_h = triton.cdiv(hidden_dim, BLOCK_H)

    grid = (B * total_i * total_j * total_h,)

    # note: we reinterpret right stride ordering to (b,k,j,h)
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l[0],
        stride_l_i=stride_l[1],
        stride_l_k=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_b=stride_r[0],
        # reinterpret: make k the second dimension
        stride_r_k=stride_r[2],
        stride_r_j=stride_r[1],
        stride_r_h=stride_r[3],
        stride_o_b=stride_o[0],
        stride_o_i=stride_o[1],
        stride_o_j=stride_o[2],
        stride_o_h=stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    #  Post‑processing: LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                               # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])      # [B,N,N,dim]

    return out.to(dtype)   # ensure float32 output
