"""
TriMul “outgoing” implementation with a Triton kernel for the heavy
pair‑wise multiplication.

The kernel computes, for every batch b and hidden channel d,

    out[b, i, j, d] = Σ_k  left[b, i, k, d] * right[b, j, k, d]

which is exactly a batched matrix multiplication of
left[..., :, :, d] (N×N) with the *transpose* of right[..., :, :, d].
The rest of the TriMul pipeline (LayerNorms, linear projections,
gates and final linear) is performed with ordinary PyTorch ops – this
keeps the code simple while still moving the O(N³·hidden) work to a
high‑performance Triton kernel.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *ptr, shape (B,N,N,hidden)
    right_ptr,              # *ptr, shape (B,N,N,hidden)
    out_ptr,                # *ptr, shape (B,N,N,hidden)
    B, N, H,                # dimensions
    # ---- strides for each tensor (4‑dim contiguous layout) ----
    stride_l_b0, stride_l_b1, stride_l_b2, stride_l_b3,
    stride_r_b0, stride_r_b1, stride_r_b2, stride_r_b3,
    stride_o_b0, stride_o_b1, stride_o_b2, stride_o_b3,
    # ---- compile‑time tile sizes -------------------------------
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Triton kernel that computes the per‑channel batched matrix product:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    """
    # ------- identify which (batch, hidden) we are processing ----------
    pid = tl.program_id(0)                     # linear id over B*H
    total = B * H
    b = pid // H                               # batch index
    h = pid % H                                # hidden‑channel index

    # ------- tile coordinates in the (i, j) output matrix -------------
    pid_m = tl.program_id(1)                   # tile row
    pid_n = tl.program_id(2)                   # tile col

    # global row / col indices for this thread block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer base for the current (b, h) slice
    left_base  = left_ptr  + b * stride_l_b0 + h * stride_l_b3
    right_base = right_ptr + b * stride_r_b0 + h * stride_r_b3
    out_base   = out_ptr   + b * stride_o_b0 + h * stride_o_b3

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over the reduction dimension k -------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]   -> shape (M, K)
        left_ptrs = (
            left_base
            + offs_m[:, None] * stride_l_b1          # i dimension
            + offs_k[None, :] * stride_l_b2          # k dimension
        )
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b, j, k, h] needs to be used as rightᵀ[k, j, h]
        # i.e. we load right[b, j, k, h] and treat the k index as the *row*
        right_ptrs = (
            right_base
            + offs_n[:, None] * stride_r_b1          # j dimension (becomes column)
            + offs_k[None, :] * stride_r_b2          # k dimension (row after transpose)
        )
        right = tl.load(right_ptrs,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0)

        acc += tl.dot(left, right)                # (M,K)·(K,N) → (M,N)

    # ---------- write back the result ---------------------------------
    out_ptrs = (
        out_base
        + offs_m[:, None] * stride_o_b1            # i dimension
        + offs_n[None, :] * stride_o_b2            # j dimension
    )
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:          torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights:       dict of torch.Tensor parameters
            - config:        dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----- unpack --------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    device = input_tensor.device

    # ----- LayerNorm on the input ----------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----- Linear projections (no bias) ----------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply optional mask -------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)           # [B,N,N,1]
        left  = left * mask_unsq
        right = right * mask_unsq

    # ----- Gates ---------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ----- Allocate output tensor for the kernel -------------------------
    out = torch.empty((B, N, N, hidden), device=device, dtype=torch.float32)

    # ----- Launch Triton kernel -----------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides for contiguous tensors
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    grid = (
        B * hidden,                     # one program per (batch, hidden)
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        # strides of left
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        # strides of right
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        # strides of out
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # make sure kernel has finished before we continue
    torch.cuda.synchronize()

    # ----- Final LayerNorm, gating and output projection -----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
