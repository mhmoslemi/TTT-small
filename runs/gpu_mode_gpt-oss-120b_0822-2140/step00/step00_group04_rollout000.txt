"""
TriMul “outgoing” implementation (forward only)

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiplication for every
head‑dimension *d*:

    L_d  = left[..., :, :, d]     # (B, N, N)   i‑k
    R_dᵀ = right[..., :, :, d].transpose(1, 2)   # (B, N, N)   k‑j
    out[..., :, :, d] = batched_matmul(L_d, R_dᵀ)

The kernel below performs this matmul with Triton.  All other
operations (LayerNorm, Linear‑projections, gates, final LayerNorm &
Linear) stay in PyTorch – they are already highly‑optimized and
require no custom kernel.  The kernel works on half‑precision
(`float16`) for speed and the result is cast back to `float32`
before the final linear projection.

The launch grid is (grid_M, grid_N, batch * hidden_dim).  Each program
handles a tile of size BLOCK_M × BLOCK_N for a single (batch, head)
pair and loops over k‑tiles of size BLOCK_K.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel that computes: out[b,i,j,d] = Σ_k left[b,i,k,d] * right_t[b,k,j,d]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shapes
    B, N, D,
    # strides (left)   [b, i, k, d]
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides (right_t) [b, k, j, d]  (right already transposed)
    stride_rb, stride_rk, stride_rj, stride_rd,
    # strides (out)    [b, i, j, d]
    stride_ob, stride_oi, stride_oj, stride_od,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_z = tl.program_id(2)          # combined (batch, head)

    # ----- decode batch and head index -----
    b = pid_z // D
    d = pid_z % D

    # ----- tile coordinates -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for OOB (i, j)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over k dimension -----
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load a tile of left  (M x K) -----
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + d * stride_ld
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- load a tile of right (K x N) -----
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + d * stride_rd
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rj
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- small matrix multiplication -----
        acc += tl.dot(left_tile, right_tile)

    # ----- write out (convert back to fp16) -----
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + d * stride_od
        + offs_m[:, None] * stride_oi
        + offs_n[None, :] * stride_oj
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel
    for the core N³ multiplication.

    Args:
        data: tuple (input, mask, weights, config)
            - input:  Tensor [B, N, N, dim] (float32)
            - mask:   Tensor [B, N, N]  (float32/bool) – may be omitted (all‑ones)
            - weights: dict of pre‑trained parameters
            - config: dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # Utility: layer‑norm (no bias handling inside Triton)
    # ------------------------------------------------------------------
    def layer_norm(tensor, weight, bias):
        # tensor: (B, N, N, C)
        return F.layer_norm(tensor, (tensor.shape[-1],), weight, bias, eps=1e-5)

    # ------------------------------------------------------------------
    # Linear helpers (bias‑free as per the reference implementation)
    # ------------------------------------------------------------------
    def linear(tensor, weight):
        # weight: (out_features, in_features)
        return F.linear(tensor, weight)

    # ------------------------------------------------------------------
    # 1. Input normalisation
    # ------------------------------------------------------------------
    x = layer_norm(
        x,
        weights["norm.weight"].to(torch.float32),
        weights["norm.bias"].to(torch.float32),
    )

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left = linear(
        x,
        weights["left_proj.weight"].to(torch.float32),  # (hidden_dim, dim)
    )
    right = linear(
        x,
        weights["right_proj.weight"].to(torch.float32),
    )

    # ------------------------------------------------------------------
    # 3. Mask (if any) – broadcast on the hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)               # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        linear(x, weights["left_gate.weight"].to(torch.float32))
    )
    right_gate = torch.sigmoid(
        linear(x, weights["right_gate.weight"].to(torch.float32))
    )
    out_gate = torch.sigmoid(
        linear(x, weights["out_gate.weight"].to(torch.float32))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ multiplication – Triton kernel
    # ------------------------------------------------------------------
    # Cast to half‑precision for the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    # transpose right on (k, j) so that we can use a normal GEMM pattern
    right_t_fp16 = right.permute(0, 2, 1, 3).contiguous().to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)  # shape [B, N, N, hidden_dim]

    # tile sizes – tuned for Hopper (SM90) / 128‑bit shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows (i)
        triton.cdiv(N, BLOCK_N),      # cols (j)
        B * hidden_dim,               # combined batch * head
    )

    # stride extraction (must be int64)
    s_lb, s_li, s_lk, s_ld = left_fp16.stride()
    s_rb, s_rk, s_rj, s_rd = right_t_fp16.stride()
    s_ob, s_oi, s_oj, s_od = out_fp16.stride()

    _trimul_kernel[grid](
        left_fp16,
        right_t_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_rk,
        s_rj,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to float32
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"].to(torch.float32),
        weights["to_out_norm.bias"].to(torch.float32),
    )
    out = out * out_gate
    out = linear(
        out,
        weights["to_out.weight"].to(torch.float32),  # (dim, hidden_dim)
    )
    return out
