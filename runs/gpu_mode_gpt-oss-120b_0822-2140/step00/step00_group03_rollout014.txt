"""
TriMul “outgoing” operator – fast Triton implementation of the costly
einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The surrounding element‑wise ops (LayerNorm, Linear, gating, mask) are
performed in PyTorch.  The triple‑product is rewritten as a batched
matrix‑multiply per hidden channel:

    left   : (B,N,N,H) → (B*H, N, N)      # L
    right  : (B,N,N,H) → (B*H, N, N)      # R
    out    : (B*H, N, N) where out = L @ Rᵀ

The matrix multiply is executed by a custom Triton kernel that tiles the
output (i‑j) space with blocks (BLOCK_M, BLOCK_N) and accumulates over the
k‑dimension in BLOCK_K steps.  The kernel works in FP16 and returns FP16;
the surrounding code casts back to FP32 for the final linear layer.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # FP16 * (B*H, N, N)
    right_ptr,         # FP16 * (B*H, N, N)
    out_ptr,           # FP16 * (B*H, N, N)

    B_H, N,            # dimensions
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ  for a single (batch * hidden) slice.
    """
    batch_id = tl.program_id(0)          # 0 … B*H-1
    pid_m = tl.program_id(1)             # tile row   (i)
    pid_n = tl.program_id(2)             # tile col   (j)

    # ------------------------------------------------------------------
    # Offsets
    # ------------------------------------------------------------------
    i_base = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_base = tl.arange(0, BLOCK_K)

    # mask for OOB rows/cols
    mask_i = i_base < N
    mask_j = j_base < N

    # pointers with proper strides
    batch_offset_left  = batch_id * stride_left_batch
    batch_offset_right = batch_id * stride_right_batch
    batch_offset_out   = batch_id * stride_out_batch

    # left block:   (BM, BK)
    a_ptrs = left_ptr + batch_offset_left \
             + i_base[:, None] * stride_left_i \
             + k_base[None, :] * stride_left_k
    # right block (transposed for dot): (BK, BN)
    b_ptrs = right_ptr + batch_offset_right \
             + k_base[:, None] * stride_right_k \
             + j_base[None, :] * stride_right_j

    # accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (k_base[None, :] < N), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(k_base[:, None] < N) & mask_j[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)               # (BM, BN)

        # advance k pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + batch_offset_out \
               + i_base[:, None] * stride_out_i \
               + j_base[None, :] * stride_out_j
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast Triton‑accelerated forward pass of the TriMul "outgoing" module.
    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]       (float32 or bool)
        weights      : dict of torch.Tensor containing all linear / norm parameters
        config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on dim
    # ------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare FP16 tensors for the kernel
    # ------------------------------------------------------------------
    # bring hidden dimension to the batch axis
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_t = left_t.view(B * hidden_dim, N, N).to(torch.float16)
    right_t = right_t.view(B * hidden_dim, N, N).to(torch.float16)

    out_t = torch.empty_like(left_t)                  # (B*H, N, N), fp16

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_t,
        right_t,
        out_t,
        B * hidden_dim,
        N,
        left_t.stride(0),    # batch stride
        left_t.stride(1),    # i stride
        left_t.stride(2),    # k stride
        right_t.stride(0),   # batch stride
        right_t.stride(1),   # j stride (after permute, dim 2 is j)
        right_t.stride(2),   # k stride
        out_t.stride(0),
        out_t.stride(1),
        out_t.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_t.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final linear
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
