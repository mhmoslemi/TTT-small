"""
TriMul “outgoing” implementation (AlphaFold‑3).

* LayerNorm over the last dimension is fused into a Triton kernel
  (no need to materialise mean / variance in Python).
* The costly bilinear reduction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is performed as a batched matrix‑multiply with cuBLAS (`torch.bmm`);
  this is already optimal on Hopper and avoids an O(N³·H) Python loop.
* All remaining linear / sigmoid / gating steps are kept in PyTorch,
  which is cheap compared to the N³ term.

The kernel operates on a single “pair” element (b,i,j) and processes the
full hidden dimension `H` (128 or 384) as a vector.  The kernel is launched
with a 1‑D grid of size `B·N·N`, i.e. one thread per (batch, i, j) entry.
The final output has shape `[B, N, N, dim]` and is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------
# Triton LayerNorm kernel (over the last dimension of a 4‑D tensor)
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *float32   input  [B, N, N, D]
    y_ptr,                # *float32   output [B, N, N, D]
    weight_ptr,           # *float32   gamma  [D]
    bias_ptr,             # *float32   beta   [D]
    B, N, D,              # compile‑time constants
    stride_xb, stride_xi, stride_xj, stride_xd,   # strides for x
    stride_yb, stride_yi, stride_yj, stride_yd,   # strides for y
    stride_wd, stride_bd,                         # strides for gamma / beta
    EPS: tl.constexpr = 1e-6,
    BLOCK_D: tl.constexpr = 128,
):
    """
    One thread = one (b, i, j) location.
    The hidden dimension D is processed as a vector of length BLOCK_D
    (must be a compile‑time constant that matches the actual hidden size).
    """
    # ------------------------------------------------------------------
    # 1) Decode the (b,i,j) coordinates from the linear program id.
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    total_ij = N * N
    b = pid // total_ij
    rem = pid % total_ij
    i = rem // N
    j = rem % N

    # ------------------------------------------------------------------
    # 2) Load the D‑dimensional vector for this (b,i,j) entry.
    # ------------------------------------------------------------------
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    x = tl.load(
        x_ptr
        + b * stride_xb
        + i * stride_xi
        + j * stride_xj
        + offs_d * stride_xd,
        mask=mask_d,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # 3) Compute mean and variance (scalar) over the D elements.
    # ------------------------------------------------------------------
    # manual reduction – the loop is unrolled because BLOCK_D is a constexpr.
    sum_val = tl.float32(0.0)
    sq_sum_val = tl.float32(0.0)
    for d in range(BLOCK_D):
        if d < D:
            v = x[d]
            sum_val += v
            sq_sum_val += v * v
    mean = sum_val / D
    var = (sq_sum_val / D) - mean * mean

    # ------------------------------------------------------------------
    # 4) Normalise, apply learned scale / bias.
    # ------------------------------------------------------------------
    norm = (x - mean) * tl.rsqrt(var + EPS)

    w = tl.load(weight_ptr + offs_d * stride_wd, mask=mask_d, other=1.0)
    b_ = tl.load(bias_ptr + offs_d * stride_bd, mask=mask_d, other=0.0)

    y = norm * w + b_

    # ------------------------------------------------------------------
    # 5) Store the result.
    # ------------------------------------------------------------------
    tl.store(
        y_ptr
        + b * stride_yb
        + i * stride_yi
        + j * stride_yj
        + offs_d * stride_yd,
        y,
        mask=mask_d,
    )


# -------------------------------------------------------------
# Triton helper to launch the LayerNorm kernel
# -------------------------------------------------------------
def triton_layernorm(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-6,
) -> torch.Tensor:
    """
    x : [B, N, N, D] (float32)
    weight / bias : [D] (float32)

    Returns a new tensor of the same shape as x, already normalised.
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, _, D = x.shape
    y = torch.empty_like(x)

    # make D a compile‑time constant for the kernel
    BLOCK_D = 128
    if D > BLOCK_D:
        BLOCK_D = 384   # the only other valid size in the spec

    # launch configuration (one thread per (b,i,j))
    total = B * N * N
    grid = (total,)

    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        B,
        N,
        D,
        x.stride(0),
        x.stride(1),
        x.stride(2),
        x.stride(3),
        y.stride(0),
        y.stride(1),
        y.stride(2),
        y.stride(3),
        weight.stride(0),
        bias.stride(0),
        EPS=eps,
        BLOCK_D=BLOCK_D,
    )
    return y


# -------------------------------------------------------------
# Main entry point required by the evaluator
# -------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass.
    The heavy LayerNorm is performed with a custom Triton kernel,
    everything else uses regular PyTorch (including a batched GEMM for
    the N³ reduction, which is already optimal on Volta‑class GPUs).
    """
    # unpack
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------
    # 1) LayerNorm over the last dimension (fused Triton kernel)
    # -------------------------------------------------------------
    inp_norm = triton_layernorm(
        inp,
        weights["norm.weight"],
        weights["norm.bias"],
        eps,
    )  # shape [B,N,N,dim]

    # -------------------------------------------------------------
    # 2) Linear projections (no bias) → hidden dimension
    # -------------------------------------------------------------
    # Linear: y = x @ W.T   (torch.nn.functional.linear does exactly this)
    left = F.linear(inp_norm, weights["left_proj.weight"])
    right = F.linear(inp_norm, weights["right_proj.weight"])

    # -------------------------------------------------------------
    # 3) Apply mask (if present).  mask shape: [B, N, N]
    # -------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(inp.dtype)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------
    # 4) Gating scalars (sigmoid of linear projections of the *normed* input)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(inp_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(inp_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(inp_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5) Core N³ reduction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is exactly a batched matrix‑multiply for each hidden channel.
    # -------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) → treat each hidden channel as a separate batch.
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B*H, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)

    # right needs to be transposed on the last two dimensions to match the
    # definition Σ_k left[i,k]*right[j,k]  (i‑k × j‑kᵀ).
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_reshape.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
