"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the 4‑D input  (dim = C).
2. Linear projections  →  left/right  of shape [B, N, N, H].
3. Compute three gates (sigmoid of linear layers) of shape [B, N, N, H].
4. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel
   (element‑wise: left  = left  * mask * left_gate,
                right = right * mask * right_gate).
5. For each hidden channel h we need
        out[..., :, :, h] = left[..., :, :, h] @ right[..., :, :, h].T
   This is realised by reshaping to a batch of (B*H) matrix‑multiplications
   and calling torch.bmm (cuBLAS) – the dominant O(N³) work.
6. Layer‑norm over the hidden dimension, multiply by the output‑gate
   (another tiny Triton kernel) and finally a linear projection back to
   the original feature dimension.
All heavy linear algebra is done by cuBLAS; Triton is used for the
element‑wise fusions which are the only parts that a pure‑PyTorch
implementation would perform with separate kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny element‑wise kernels (mask & gate fusion, final out‑gate multiplication)
# ----------------------------------------------------------------------
@triton.jit
def fuse_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    stride_b, stride_n1, stride_n2, stride_h,
    N, H,
    BLOCK: tl.constexpr,
):
    """
    left, right, mask, left_gate, right_gate are all contiguous
    with the same 4‑D layout [B, N, N, H].
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = stride_b * N * N * H   # total elements (B * N * N * H)

    mask = offs < total
    # 1‑D index → 4‑D indices
    b = (offs // (N * N * H))
    rem = offs % (N * N * H)
    i = (rem // (N * H))
    rem = rem % (N * H)
    j = (rem // H)
    h = rem % H

    # compute flat offsets
    idx = b * stride_b + i * stride_n1 + j * stride_n2 + h * stride_h

    left   = tl.load(left_ptr   + idx, mask=mask, other=0.0)
    right  = tl.load(right_ptr  + idx, mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + idx, mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr  + idx, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr + idx, mask=mask, other=0.0)

    left  = left  * m * lgate
    right = right * m * rgate

    tl.store(left_ptr  + idx, left,  mask=mask)
    tl.store(right_ptr + idx, right, mask=mask)


@triton.jit
def mul_outgate_kernel(
    out_ptr, out_gate_ptr,
    stride_b, stride_n1, stride_n2, stride_h,
    N, H,
    BLOCK: tl.constexpr,
):
    """out = out * out_gate  (both [B,N,N,H])"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = stride_b * N * N * H
    mask = offs < total

    b = (offs // (N * N * H))
    rem = offs % (N * N * H)
    i = (rem // (N * H))
    rem = rem % (N * H)
    j = (rem // H)
    h = rem % H

    idx = b * stride_b + i * stride_n1 + j * stride_n2 + h * stride_h

    out      = tl.load(out_ptr      + idx, mask=mask, other=0.0)
    out_gate = tl.load(out_gate_ptr + idx, mask=mask, other=0.0)

    out = out * out_gate
    tl.store(out_ptr + idx, out, mask=mask)


# ----------------------------------------------------------------------
# Helper to launch the element‑wise kernels
# ----------------------------------------------------------------------
def _apply_gates_triton(left: torch.Tensor,
                        right: torch.Tensor,
                        mask: torch.Tensor,
                        left_gate: torch.Tensor,
                        right_gate: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    In‑place fusion of mask & gates using Triton.
    All tensors must be contiguous, float32 and have shape [B,N,N,H].
    """
    B, N, _, H = left.shape
    # compute strides in elements (not bytes)
    stride_b, stride_n1, stride_n2, stride_h = left.stride()
    total_elems = B * N * N * H
    BLOCK = 1024  # 1‑K elements per program
    grid = (triton.cdiv(total_elems, BLOCK),)

    fuse_gate_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        mask.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        stride_b, stride_n1, stride_n2, stride_h,
        N, H,
        BLOCK=BLOCK,
    )
    return left, right


def _apply_outgate_triton(out: torch.Tensor,
                          out_gate: torch.Tensor) -> torch.Tensor:
    """In‑place out = out * out_gate  (both [B,N,N,H])"""
    B, N, _, H = out.shape
    stride_b, stride_n1, stride_n2, stride_h = out.stride()
    total_elems = B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    mul_outgate_kernel[grid](
        out.data_ptr(),
        out_gate.data_ptr(),
        stride_b, stride_n1, stride_n2, stride_h,
        N, H,
        BLOCK=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Heavy linear algebra (the N³ multiplication) is performed with
    torch.bmm (cuBLAS) while the element‑wise mask / gate fusion is
    executed by tiny Triton kernels.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]            # input feature dimension C
    hidden_dim = config["hidden_dim"]  # H
    B, N, _, C = input_tensor.shape   # input is [B,N,N,C]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after Linear: [B,N,N,hidden_dim]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply mask and gates with Triton (in‑place)
    # ------------------------------------------------------------------
    # mask is [B,N,N]; broadcast to H dimension
    mask_4d = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
    mask_4d = mask_4d.expand(-1, -1, -1, hidden_dim)  # [B,N,N,H]
    left, right = _apply_gates_triton(left.contiguous(),
                                      right.contiguous(),
                                      mask_4d.contiguous(),
                                      left_gate.contiguous(),
                                      right_gate.contiguous())

    # ------------------------------------------------------------------
    # 5. Core N³ multiplication: out = left @ right^T per hidden channel
    #    Reshape to (B*H, N, N) and use torch.bmm.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_resh  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # L
    right_resh = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # R

    # matrix multiplication: L @ R^T
    out_resh = torch.bmm(left_resh, right_resh.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_resh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim layer‑norm, multiply by out_gate (Triton), final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # out_gate is still [B,N,N,H] (same shape as out)
    out = _apply_outgate_triton(out, out_gate)

    # final linear projection back to dim C
    out = F.linear(out, weights["to_out.weight"])

    return out.to(dtype)
