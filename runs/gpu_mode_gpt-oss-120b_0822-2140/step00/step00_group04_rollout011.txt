"""
TriMul “outgoing” forward pass with a fused batched matmul implemented in Triton.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C) using the provided
   `norm.{weight,bias}`.
2. Linear projections → left/right (shape B,N,N,H) and three sigmoid gates.
3. Apply the optional mask (broadcast on the hidden dim) and the gates.
4. The core operation is
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply for every hidden dimension:
       out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
   A custom Triton kernel performs this `B*H` independent MatMul’s
   (H = hidden_dim) on fp16 for speed.
5. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original channel size.

The kernel works for any batch size (1‑2), sequence length up to 1024,
and hidden dimensions that fit in memory.  All other steps stay in PyTorch,
so the implementation is both correct and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (square N)
    stride_am, stride_ak, stride_ab,   # strides of A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides of B (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b]   (A,B,C are (M×K),(K×N),(M×N))."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the edge of the matrices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    #  Pointers to the start of the current tile for this batch
    # ------------------------------------------------------------------
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    # Accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch_ptr + \
                 (offs_m[:, None] * stride_am) + \
                 (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (K × N)
        b_ptrs = b_batch_ptr + \
                 (offs_k[:, None] * stride_bk) + \
                 (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    #  Write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + \
             (offs_m[:, None] * stride_cm) + \
             (offs_n[None, :] * stride_cn)

    tl.store(c_ptrs,
             acc.to(tl.float16),          # store as fp16 (memory‑bandwidth friendly)
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor, shape (B, N, N, C)
        - mask  : torch.Tensor, shape (B, N, N)  (may be all‑ones)
        - weights: dict of pretrained parameters
        - config : dict containing `dim`, `hidden_dim`, and optional `nomask`

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, C) (float32)
    """
    # ------------------------------------------------------------------
    #  unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape
    device = input_tensor.device

    # ------------------------------------------------------------------
    #  1️⃣ Layer‑norm on the input (over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    #  2️⃣ Linear projections (no bias) → left/right  (B,N,N,H)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    #  3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1)                # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    #  4️⃣ Gates (sigmoid) and gating of left/right
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  5️⃣ Core *einsum* → batched matmul via Triton
    # ------------------------------------------------------------------
    # Convert to the dtype used inside the kernel (fp16)
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Rearrange to (B*H, M, K) where M = N, K = N
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right needs to be transposed on its inner two axes (k,j)
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer
    out_perm = torch.empty_like(left_perm, dtype=torch.float16)

    # ------------------------------------------------------------------
    #  Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (in elements, not bytes)
    stride_am, stride_ak, stride_ab = left_perm.stride()
    stride_bk, stride_bn, stride_bb = right_perm.stride()
    stride_cm, stride_cn, stride_cb = out_perm.stride()

    _batched_matmul_kernel[grid](
        left_perm.data_ptr(),
        right_perm.data_ptr(),
        out_perm.data_ptr(),
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Cast to fp32 for the remaining ops (more stable for layer‑norm)
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    #  6️⃣ Hidden‑dim layer‑norm, multiply by out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                     # out_gate is (B,N,N,H) float32
    out = F.linear(out, weights["to_out.weight"])

    return out
