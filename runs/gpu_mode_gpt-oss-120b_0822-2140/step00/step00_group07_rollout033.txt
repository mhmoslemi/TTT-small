    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    # LayerNorm
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5)
    # apply weight/bias
    x = x * weights["norm.weight"] + weights["norm.bias"]
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

        mask_unsq = mask.unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq
    else:

        pass
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    right = right * right_gate

"""
TriMul “outgoing” kernel (AlphaFold3) – Triton implementation.

The heavy part of the module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for every hidden channel d:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

The kernel below computes this operation tile‑wise with
  * BLOCK_M × BLOCK_N output tiles
  * reduction over K in blocks of size BLOCK_K
  * processing the hidden dimension in chunks of BLOCK_D (fits in registers)

All other operations (LayerNorm, linear projections, gating, final LayerNorm
and linear) stay in PyTorch for clarity.  The kernel works on fp16 inputs
and accumulates in fp32, then writes fp16 results that are cast back to fp32
by the caller.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                          # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Batch‑wise, channel‑wise matrix multiplication over the third axis."""
    pid_b = tl.program_id(0)      # batch index
    pid_i = tl.program_id(1)      # tile row (i)
    pid_j = tl.program_id(2)      # tile col (j)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in chunks of BLOCK_D
    # ------------------------------------------------------------------
    num_d_chunks = tl.cdiv(H, BLOCK_D)
    for d_chunk in range(num_d_chunks):
        offs_d = d_chunk * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # accumulator in fp32
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ----------------------------------------------------------------
        # Reduce over the sequence dimension K
        # ----------------------------------------------------------------
        num_k_chunks = tl.cdiv(N, BLOCK_K)
        for k_chunk in range(num_k_chunks):
            offs_k = k_chunk * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ---------- left slice (M, K, D) ----------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_d[None, None, :] * stride_left_d
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & True,
                other=0.0,
            ).to(tl.float32)          # (M, K, D)

            # ---------- right slice (N, K, D) ----------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_j[:, None, None] * stride_right_j
                + offs_k[None, :, None] * stride_right_k
                + offs_d[None, None, :] * stride_right_d
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & True,
                other=0.0,
            ).to(tl.float32)          # (N, K, D)

            # ---------- multiply‑accumulate per hidden channel ----------
            #  For each d we compute a small GEMM: left[:,:,d] @ right[:,:,d].T
            for d in range(0, BLOCK_D):
                # (M, K) x (N, K).T -> (M, N)
                acc[:, :, d] += tl.dot(
                    left[:, :, d],          # (M, K)
                    right[:, :, d],         # (N, K)
                    trans_b=True
                )
        # ----------------------------------------------------------------
        # Store the computed tile (cast back to fp16)
        # ----------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_d[None, None, :] * stride_out_d
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
        )

def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim] (float32)
            mask   – torch.Tensor [B, N, N]   (bool/float) – may be absent
            weights – dict of pre‑loaded parameters (float32)
            config  – dict with keys:
                "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]  float32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (out=hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask_unsq = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul – Triton kernel
    #    Work in fp16 inside the kernel for better bandwidth,
    #    accumulate in fp32.
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)          # same shape [B,N,N,hidden]

    # tile sizes – chosen to fit H200 SMs and keep shared memory reasonable
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32                                   # hidden dimension chunk

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    out = out_fp16.to(torch.float32)               # back to fp32

    # --------------------------------------------------------------
    # 6. Final LayerNorm, gating and linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
```
