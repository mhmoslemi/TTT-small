"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a custom Triton kernel.

The heavy part of the module is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication per hidden dimension d:
    out[d] = L[d] @ R[d].T
where L[d] and R[d] have shape (B, N, N).

The kernel below computes this operation in FP16 (fast load/store) while
accumulating in FP32 for numerical stability.  The surrounding linear
projections, layer‑norms and gating are kept in PyTorch for simplicity.
The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel
# ------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *fp16   [B,N,N,H]
    right_ptr,              # *fp16   [B,N,N,H]
    out_ptr,                # *fp16   [B,N,N,H]

    B, N, H,                # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr,  # tile size for the i‑dimension
    BLOCK_N: tl.constexpr,  # tile size for the j‑dimension
    BLOCK_K: tl.constexpr,  # tile size for the reduction (k)
):
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single (b, h) pair.  The grid is launched over
    (batch * hidden * tile_m * tile_n).
    """
    pid = tl.program_id(0)

    # ----- decode 1‑D program id into (b, h, tile_i, tile_j) -----
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # total programs per batch element
    per_batch = H * num_pid_m * num_pid_n
    b = pid // per_batch
    pid_rem = pid % per_batch

    h = pid_rem // (num_pid_m * num_pid_n)
    pid_rem2 = pid_rem % (num_pid_m * num_pid_n)

    pid_m = pid_rem2 // num_pid_n
    pid_n = pid_rem2 % num_pid_n

    # ----- tile offsets -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- pointers ------------------------------------------------------
    # left  : (i, k)   row‑major as stored
    left_tile_ptr = (
        left_ptr
        + b * stride_left_b
        + h * stride_left_h
        + offs_i[:, None] * stride_left_i
        + offs_k[None, :] * stride_left_k
    )
    # right : we need right_T with shape (k, j) = right[j, k]
    right_tile_ptr = (
        right_ptr
        + b * stride_right_b
        + h * stride_right_h
        + offs_k[:, None] * stride_right_k
        + offs_j[None, :] * stride_right_i
    )

    # ----- accumulation ---------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k‑tiles
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles (mask out‑of‑bounds with zero)
        left = tl.load(
            left_tile_ptr,
            mask=mask_i[:, None] & (offs_k[None, :] < N),
            other=0.0,
        ).to(tl.float32)

        right = tl.load(
            right_tile_ptr,
            mask=(offs_k[:, None] < N) & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left, right)                     # (M,K)·(K,N) → (M,N)

        # advance k‑pointers
        left_tile_ptr += BLOCK_K * stride_left_k
        right_tile_ptr += BLOCK_K * stride_right_k

    # ----- write result ---------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_tile_ptr,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ------------------------------------------------------------
# Host‑side entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask_tensor  : [B, N, N]       (bool / 0‑1 float)
            - weights_dict : mapping of all Parameter tensors
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias) and optional masking
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating of the projections
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Triton kernel: batch‑wise, per‑hidden‑dim matrix multiplication
    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    B, N, _, _ = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16)

    # tile sizes – empirically good for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch grid: B * H * tiles_m * tiles_n
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = B * hidden_dim * grid_m * grid_n
    grid = (total_programs,)

    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        # strides for right (identical layout)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for output
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # convert back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
