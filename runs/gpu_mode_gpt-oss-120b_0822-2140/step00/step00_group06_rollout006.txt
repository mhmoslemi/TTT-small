"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the 4‑D tensor contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
for each (batch, hidden) pair:

    out_per_pair = left_per_pair @ right_per_pair.T
    left_per_pair : (N, N)   (i,k)
    right_per_pair: (N, N)   (j,k)  -> transpose on k

The kernel below implements a tiled GEMM for this operation in
Triton.  The remaining LayerNorm / Linear / gating logic is done
with plain PyTorch for simplicity and to keep the code short.
The final output is a float‑32 tensor of shape
[batch, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,            # const float*  [B, N, N, D]
    right_ptr,           # const float*  [B, N, N, D]
    out_ptr,             # float*        [B, N, N, D]
    B, N, D,             # int
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] """

    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) map 1‑D program id to (batch, hidden, block_i, block_j)
    # ------------------------------------------------------------------
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    blocks_ij = total_i * total_j                # number of (i,j) blocks per (b,d)

    pair_id = pid // blocks_ij                  # which (b,d) we are processing
    block_ij = pid % blocks_ij

    b = pair_id // D
    d = pair_id % D

    block_i = block_ij // total_j
    block_j = block_ij % total_j

    # ------------------------------------------------------------------
    # 2) tile offsets
    # ------------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds (needed for the last partial tile)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 3) pointers to the first tile of this block
    # ------------------------------------------------------------------
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base = out_ptr + b * stride_ob + d * stride_od

    left_tile_ptr = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
    right_tile_ptr = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) main K‑loop
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k_tiles):
        a = tl.load(left_tile_ptr, mask=(mask_i[:, None] & (offs_k[None, :] < N)), other=0.0)
        b_mat = tl.load(right_tile_ptr,
                        mask=((offs_k[:, None] < N) & mask_j[None, :]), other=0.0)

        acc += tl.dot(a, b_mat)               # (M,K) @ (K,N) -> (M,N)

        # advance K dimension
        left_tile_ptr += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # 5) write out
    # ------------------------------------------------------------------
    out_tile_ptr = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    mask_o = mask_i[:, None] & mask_j[None, :]
    tl.store(out_tile_ptr, acc, mask=mask_o)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    contraction.  The function expects:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))        # (B,N,N,hd)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)                     # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton GEMM: contract over the “k” axis
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=dtype)

    # strides for the contiguous [B,N,N,D] layout
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_i = triton.cdiv(N, BLOCK_M)
    total_j = triton.cdiv(N, BLOCK_N)
    grid = (B * hidden_dim * total_i * total_j,)

    _trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (hidden_dim) → gating → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
