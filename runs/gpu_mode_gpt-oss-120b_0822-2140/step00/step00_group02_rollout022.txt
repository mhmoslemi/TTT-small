"""
TriMul “outgoing” forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) →  x̂ .
2. Linear projections (no bias) → left, right  (B, N, N, H).
3. Optional pairwise mask.
4. Gated projections:
        left  = left  * sigmoid(Linear(x̂ , left_gate ))
        right = right * sigmoid(Linear(x̂ , right_gate))
5. Core “multiply‑and‑sum over K” :
        out_h[b,i,j,:] = Σ_k left [b,i,k,:] * right [b,j,k,:]
   This is a batched matrix multiplication for each hidden slice.
   It is implemented by a Triton kernel that computes
        (left_slice) @ (right_slice^T)   for a chunk of the hidden dimension.
6. Layer‑norm on the hidden dimension → out_h_norm .
7. Gated scaling with out_gate and final linear projection
   (hidden → C).

Only the O(N³·H) part (step‑5) runs on Triton; the rest is done with
high‑performance PyTorch primitives.  The kernel works for arbitrary
batch size (1‑2), sequence length (≤1024) and hidden dimension (≤384).

The kernel is fused across the K‑reduction and supports arbitrary
sequence lengths via tiling (BLOCK_M×BLOCK_N) and a K‑tile (BLOCK_K).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul for a hidden‑dimension slice
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,               # *float32   (B,N,N,chunk)
    right_ptr,              # *float32   (B,N,N,chunk)
    out_ptr,                # *float32   (B,N,N,chunk)
    B, N, K,                # i32  – batch, seq‑len, K (=N)
    stride_l_b, stride_l_i, stride_l_k,   # i32 strides of left  (no hidden stride)
    stride_r_b, stride_r_j, stride_r_k,   # i32 strides of right (note j‑stride first)
    stride_o_b, stride_o_i, stride_o_j,   # i32 strides of out
    BLOCK_M: tl.constexpr,  # tile size in the i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in the j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in the reduction (k)
):
    """Compute out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k] for one hidden slice."""
    pid_b = tl.program_id(2)                     # batch axis
    pid_m = tl.program_id(0)                     # i‑tile
    pid_n = tl.program_id(1)                     # j‑tile

    # start offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for the current batch
    left_batch  = left_ptr  + pid_b * stride_l_b
    right_batch = right_ptr + pid_b * stride_r_b
    out_batch   = out_ptr   + pid_b * stride_o_b

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in tiles
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left[i,k]   → shape (BLOCK_M, BLOCK_K)
        a_ptr = left_batch + (offs_i[:, None] * stride_l_i
                              + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[j,k]  – we need the transpose so that the reduction dimension is
        # the last of the left operand and the first of the right operand.
        # shape (BLOCK_K, BLOCK_N)
        b_ptr = right_batch + (offs_k[:, None] * stride_r_k
                               + offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot: (M,K)·(K,N) → (M,N)
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_batch + (offs_i[:, None] * stride_o_i
                            + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

        - weights      : dict of torch.Tensor containing all

        - config       : dict with keys ``dim`` (C) and ``hidden_dim`` (H)

    torch.Tensor

    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # helper to read weight tensors (already on device)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    eps = 1e-5

    # 1. LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_w)      # (B,N,N,hidden)
    right = F.linear(x, right_proj_w)

    # 3. Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------- #

        right = right * m

    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------- #
    # 5. Core TriMul: out_h[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    #    Implemented with the Triton kernel above.
    # ------------------------------------------------------------------- #
    out_h = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # tiling parameters – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    CHUNK_D = 32                         # hidden dim processed per kernel launch

    # Ensure contiguous layout for the hidden‑dimension slice pointers
    left  = left.contiguous()
    right = right.contiguous()
    out_h = out_h.contiguous()

    for d_start in range(0, hidden, CHUNK_D):
        d_end = min(d_start + CHUNK_D, hidden)
        cur_chunk = d_end - d_start

        # Sliced tensors view the current hidden chunk (no copy)
        left_slice  = left[:, :, :, d_start:d_end]
        right_slice = right[:, :, :, d_start:d_end]
        out_slice   = out_h[:, :, :, d_start:d_end]

        # Pointers (as 64‑bit integers) and strides
        left_ptr  = left_slice.data_ptr()
        right_ptr = right_slice.data_ptr()
        out_ptr   = out_slice.data_ptr()

        # Strides expressed in elements (not bytes)
        stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_slice.stride()
        stride_r_b, stride_r_j, stride_r_k, stride_r_d = right_slice.stride()
        stride_o_b, stride_o_i, stride_o_j, stride_o_d = out_slice.stride()

        grid = (triton.cdiv(N, BLOCK_M),
                triton.cdiv(N, BLOCK_N),
                B)

        _tri_mul_matmul_kernel[grid](
            left_ptr, right_ptr, out_ptr,
            B, N, N,
            stride_l_b, stride_l_i, stride_l_k,
            stride_r_b, stride_r_j, stride_r_k,
            stride_o_b, stride_o_i, stride_o_j,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------- #
    # 6. LayerNorm on the hidden dimension, gating and final linear
    # ------------------------------------------------------------------- #
    out_h_norm = F.layer_norm(out_h, (hidden,), weight=to_out_norm_w,
                              bias=to_out_norm_b, eps=eps)

    out_scaled = out_h_norm * out_gate                     # broadcast over hidden
    out = F.linear(out_scaled, to_out_w)                  # (B,N,N,dim)

    return out.to(torch.float32)
