"""
TriMul “outgoing” implementation for AlphaFold‑3.
------------------------------------------------
The heavy cubic (N³) part of the forward pass is the einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where   left = (x·W_left)   * mask * σ(x·W_lgate)
        right = (x·W_right) * mask * σ(x·W_rgate)

For a fixed batch *b* and hidden channel *d* this is a matrix‑matrix
multiplication:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
Thus the whole operator reduces to a batched GEMM over the hidden dimension.
The code below
* performs LayerNorm, linear projections and gating in PyTorch (highly
  optimized and memory‑coalesced),
* launches a custom Triton kernel that computes the batched GEMM for every
  (batch, hidden) pair,
* finishes the forward pass with an output‑norm, gating and a final
  linear projection.

The Triton kernel is deliberately simple yet fast on Hopper (SM‑90):
* 2‑D tiling of the output matrix (BLOCK_M × BLOCK_N) with a K‑tile (BLOCK_K).
* One program processes a single (batch, hidden) matrix pair.
* All arithmetic is performed in float32 (the input tensors are already
  float32); the kernel can be easily switched to FP16/BF16 if desired.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with transpose on the RHS
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    lhs_ptr,               # *[B, N, N, H] left matrix
    rhs_ptr,               # *[B, N, N, H] right matrix (to be transposed)
    out_ptr,               # *[B, N, N, H] output
    B, N, H,
    stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_h,
    stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (b, h):
        out[b, i, j, h] = Σ_k lhs[b, i, k, h] * rhs[b, j, k, h]
    i.e. lhs @ rhs.T  (both are N×N)
    """
    pid = tl.program_id(0)

    # number of programs that tile the output matrix
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # linear program id → (b, h, pid_m, pid_n)
    pid_bh = pid // (num_pid_m * num_pid_n)
    pid_mn = pid % (num_pid_m * num_pid_n)
    pid_m = pid_mn // num_pid_n
    pid_n = pid_mn % num_pid_n

    b = pid_bh // H          # batch index
    h = pid_bh % H           # hidden (channel) index

    # block start offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers with correct strides
    lhs_ptrs = lhs_ptr + \
        b * stride_lhs_b + \
        offs_i[:, None] * stride_lhs_i + \
        offs_k[None, :] * stride_lhs_k + \
        h * stride_lhs_h

    rhs_ptrs = rhs_ptr + \
        b * stride_rhs_b + \
        offs_j[:, None] * stride_rhs_j + \
        offs_k[None, :] * stride_rhs_k + \
        h * stride_rhs_h      # note: k is the *row* dimension after transpose

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load a tile of lhs (M×K) and rhs (N×K) – rhs is accessed as (K,N) to get the transpose
        lhs = tl.load(lhs_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0)
        rhs = tl.load(rhs_ptrs, mask=mask_j[:, None] & (offs_k[None, :] < N), other=0.0)

        # matrix multiplication of the tile
        acc += tl.dot(lhs, rhs)          # (M×K) • (K×N) → (M×N)

        # advance pointers to next K‑tile
        lhs_ptrs += BLOCK_K * stride_lhs_k
        rhs_ptrs += BLOCK_K * stride_rhs_k

    # write back
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (no mean/var cache – we use torch’s impl)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    """Layer‑norm over the last dimension (no batch dimensions)."""
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape helpers
    B, N, _, _ = x.shape

    # left / right projections
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # fuse gates into the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: batched GEMM (left @ right.T) over hidden dim
    # ------------------------------------------------------------------
    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # strides (in elements) for the 4‑D tensors
    stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_h = left.stride()
    stride_rhs_b, stride_rhs_i, stride_rhs_k, stride_rhs_h = right.stride()
    # rhs will be accessed as (b, j, k, h) → we need stride over the second dim (j)
    stride_rhs_j = stride_rhs_i          # because dimensions are (B,N,N,H) with i=first N, k=second N
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid dimensions
    num_pid_m = triton.cdiv(N, BLOCK_M)
    num_pid_n = triton.cdiv(N, BLOCK_N)
    total_programs = B * hidden_dim * num_pid_m * num_pid_n
    grid = (total_programs,)

    batched_gemm_kernel[grid](
        left,                               # lhs_ptr
        right,                              # rhs_ptr
        out,                                # out_ptr
        B, N, hidden_dim,
        stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_h,
        stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Output processing (norm, gating, final linear)
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                       # [B,N,N,hidden]

    out = out * out_gate    # element‑wise gating

    # final linear projection to original dim (no bias)
    out = F.linear(out, weights["to_out.weight"])    # [B,N,N,dim]

    return out
