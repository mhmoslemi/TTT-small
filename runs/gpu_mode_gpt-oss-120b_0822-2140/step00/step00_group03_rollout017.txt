"""
TriMul “outgoing” kernel (AlphaFold3)

We keep the cheap parts (LayerNorm, Linear, gating) in PyTorch and
off‑load the only O(N³·C) work – the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
– to a fused Triton kernel.

For every batch * hidden dimension (d) we compute a
matrix‑multiply   left_d (N×N)  ×  right_dᵀ (N×N) .
The kernel uses FP16 loads, FP32 accumulation and writes FP16 back.
The surrounding Python code casts the result to FP32, applies the
output LayerNorm, the output gate and the final linear projection.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const * left  [B,N,N,H]
    right_ptr,         # const * right [B,N,N,H]
    out_ptr,           # * out          [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    #  Program IDs
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)               # block row   (i)
    pid_n = tl.program_id(1)               # block col   (j)
    pid_z = tl.program_id(2)               # batch*hidden dimension

    b = pid_z // H                         # batch index
    h = pid_z % H                          # hidden dimension index

    # --------------------------------------------------------------
    #  Offsets for the current block
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (last blocks may be partial)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    #  Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left[b,i,k,h]  (shape M×K) --------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # ----- load right[b,j,k,h] as (K×N) (k first, j second) ----
        # This gives the transposed view needed for matmul.
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # ----- dot product -------------------------------------------------
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    #  Write back the result (FP16 for bandwidth, cast to FP32 later)
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (float32/bool)
        - weights      : dict of torch.Tensor (all float32)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – produce [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight (out_dim, in_dim)
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the O(N³·C) product
    # ------------------------------------------------------------------
    # Cast to FP16 before kernel (better bandwidth)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty_like(left_fp16)   # same shape [B,N,N,hidden_dim]

    # Strides (in elements, not bytes)
    stride_left = left_fp16.stride()   # (B,N,N,H)
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * hidden_dim                # batch * hidden dimension
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # good default for H200
    )

    # Cast back to float32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
