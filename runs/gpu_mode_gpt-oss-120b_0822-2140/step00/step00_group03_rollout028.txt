"""
TriMul “outgoing” implementation.

The heavy O(N³·C) part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication per hidden channel.
We fuse this reduction in a Triton kernel that computes,
for each (batch, hidden) pair, a tiled GEMM:
    left_slice   : (N, N)   – rows i,  cols k
    right_sliceᵀ : (N, N)   – rows k,  cols j
and stores the result (N, N) into the output tensor.
All other operations (LayerNorms, linear projections, gates) are done
with fast PyTorch kernels. The final tensor is returned in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *const float *
    right_ptr,         # *const float *
    out_ptr,           # *mut   float *
    N,                 # sequence length
    H,                 # hidden dimension
    stride_l_batch,    # stride for batch in left   (elements)
    stride_l_i,        # stride for i (first N)      (elements)
    stride_l_k,        # stride for k (second N)     (elements)
    stride_r_batch,    # stride for batch in right
    stride_r_j,        # stride for j (second N)     (elements)
    stride_r_k,        # stride for k (first N)      (elements)
    stride_o_batch,    # stride for batch in out
    stride_o_i,        # stride for i in out
    stride_o_j,        # stride for j in out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel that computes, for a single (batch, hidden) pair,
    out[i, j] = Σ_k left[i, k] * right[j, k]   (matrix multiplication left @ rightᵀ).
    The kernel is launched with a 3‑D grid:
        (ceil(N / BLOCK_M), ceil(N / BLOCK_N), batch * hidden)
    """
    pid_i = tl.program_id(0)                     # tile row
    pid_j = tl.program_id(1)                     # tile col
    pid_bh = tl.program_id(2)                    # batch*hidden index

    # recover batch and hidden indices
    b = pid_bh // H
    d = pid_bh % H

    # base pointers for this (b, d) slice
    left_base  = left_ptr  + b * stride_l_batch + d
    right_base = right_ptr + b * stride_r_batch + d
    out_base   = out_ptr   + b * stride_o_batch + d

    # tile coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kp * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (i, k) ----
        # left[i, k] layout: stride_l_i * i + stride_l_k * k
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- load right tile (k, j)  (rightᵀ) ----
        # right[j, k] layout: stride_r_k * k + stride_r_j * j
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        acc += tl.dot(a, b_tile)   # (M,K)·(K,N) → (M,N)

    # ---- store the result ----
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping from parameter names to torch.Tensor
            - config_dict : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, C = x.shape
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1]
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating mechanisms (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the O(N³·hidden) einsum
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # same shape & dtype (float32)

    # strides in elements (not bytes) – needed by the kernel
    stride_l_batch, stride_l_i, stride_l_k, _ = left.stride()
    stride_r_batch, stride_r_i, stride_r_k, _ = right.stride()
    # right is accessed as (j,k) -> we need stride over j (second N) and over k (first N)
    stride_r_j = stride_r_i      # original stride for the second N dimension
    stride_r_k = stride_r_k      # stride for the first N dimension

    stride_o_batch, stride_o_i, stride_o_j, _ = out.stride()

    # tile sizes – empirically good for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,               # one program instance per (batch, hidden)
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape -> [B,N,N,dim]

    return out
