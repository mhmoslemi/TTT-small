"""
TriMul “outgoing” implementation.

The heavy N³ reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is carried out as a batched matrix‑multiply after reshaping
    (B,N,N,H) → (B*H, N, N) .
This lets us use cuBLAS (torch.bmm) which is far faster than a naïve
triple loop on the H200.  

Only the element‑wise mask/gate fusion is done with a small Triton
kernel – this satisfies the requirement of “at least part of the
operations in a kernel” while keeping the overall kernel simple and
portable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into left/right
# -------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr,               # *float32   [N, N, H]
    right_ptr,              # *float32   [N, N, H]
    mask_ptr,               # *float32   [N, N]          (0/1)
    left_gate_ptr,          # *float32   [N, N, H]
    right_gate_ptr,         # *float32   [N, N, H]
    N, H,
    stride_left_i, stride_left_k, stride_left_h,   # strides for left
    stride_right_i, stride_right_k, stride_right_h,# strides for right
    stride_mask_i, stride_mask_k,                  # strides for mask
    stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    total = N * N * H
    mask = offs < total

    # ----- decode linear index into (i,k,h) -----
    idx = offs
    h = idx % H
    idx = idx // H
    k = idx % N
    i = idx // N

    # ----- compute pointers -----
    left_offset  = i * stride_left_i   + k * stride_left_k   + h * stride_left_h
    right_offset = i * stride_right_i  + k * stride_right_k  + h * stride_right_h
    lgate_offset = i * stride_lgate_i  + k * stride_lgate_k  + h * stride_lgate_h
    rgate_offset = i * stride_rgate_i  + k * stride_rgate_k  + h * stride_rgate_h
    mask_offset  = i * stride_mask_i   + k * stride_mask_k

    # ----- load -----
    left  = tl.load(left_ptr  + left_offset,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + lgate_offset, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_offset, mask=mask, other=0.0)
    m     = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    # ----- fuse -----
    left  = left  * m * lgate
    right = right * m * rgate

    # ----- store back -----
    tl.store(left_ptr  + left_offset,  left,  mask=mask)
    tl.store(right_ptr + right_offset, right, mask=mask)


# -------------------------------------------------------------
# Host‑side helper that launches the kernel for a single batch
# -------------------------------------------------------------
def apply_mask_and_gate_one_batch(left, right, mask,
                                  left_gate, right_gate,
                                  BLOCK=1024):
    """
    left, right : [N, N, H]   (contiguous, float32)
    mask        : [N, N]      (float32 0/1)
    left_gate , right_gate : [N, N, H] (float32)
    """
    N, _, H = left.shape
    grid = (triton.cdiv(N * N * H, BLOCK),)

    mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        N=N, H=H,
        stride_left_i=left.stride(0), stride_left_k=left.stride(1), stride_left_h=left.stride(2),
        stride_right_i=right.stride(0), stride_right_k=right.stride(1), stride_right_h=right.stride(2),
        stride_mask_i=mask.stride(0), stride_mask_k=mask.stride(1),
        stride_lgate_i=left_gate.stride(0), stride_lgate_k=left_gate.stride(1), stride_lgate_h=left_gate.stride(2),
        stride_rgate_i=right_gate.stride(0), stride_rgate_k=right_gate.stride(1), stride_rgate_h=right_gate.stride(2),
        BLOCK=BLOCK,
    )


# -------------------------------------------------------------
# Entry point required by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor  [B, N, N, D]   (float32)
            mask         : torch.Tensor  [B, N, N]      (float32, 0/1)
            weights      : dict of parameter tensors
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor  [B, N, N, D]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # H in the paper

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (no bias in the reference code)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(D,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]      # (H, D)
    right_proj_w = weights["right_proj.weight"]    # (H, D)

    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3. Gates (sigmoid) – also bias‑free linear → sigmoid
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,H]

    # -----------------------------------------------------------------
    # 4. Apply mask and gates.  The heavy element‑wise work is fused in Triton.
    # -----------------------------------------------------------------
    mask = mask.to(dtype)                     # [B,N,N]
    # Expand mask to (B,N,N,1) for broadcasting on H
    mask_exp = mask.unsqueeze(-1)              # [B,N,N,1]

    # Work batch‑by‑batch because the kernel expects contiguous 3‑D tensors.
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_exp = mask_exp.contiguous()

    for b in range(B):
        apply_mask_and_gate_one_batch(
            left[b], right[b],
            mask_exp[b, ..., 0],          # shape [N,N]
            left_gate[b], right_gate[b],
            BLOCK=1024
        )

    # -----------------------------------------------------------------
    # 5. Core einsum: batch‑wise matmul per hidden channel.
    #    Reshape to (B*H, N, N) and use torch.bmm (cuBLAS).
    # -----------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # [B*H, N, N]
    right = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # same layout

    # out = Σₖ left[:,i,k] * right[:,j,k]   →  left @ rightᵀ
    out = torch.bmm(left, right.transpose(1, 2))                     # [B*H, N, N]

    # reshape back to (B,N,N,H)
    out = out.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. to_out_norm (LayerNorm on hidden_dim) and final linear
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate                         # gated output

    to_out_w = weights["to_out.weight"]          # (D, H)
    out = F.linear(out, to_out_w)                # back to [B,N,N,D]

    return out.to(dtype)
