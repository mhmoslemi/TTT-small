"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D einsum
    out[..., i, j, d] = Σk left[..., i, k, d] * right[..., j, k, d]

which is a batched matrix multiplication for every hidden channel d.
We fuse this reduction with a Triton kernel: each kernel instance computes a
tile (BLOCK_M × BLOCK_N) of the (i, j) plane for a single (batch, hidden)
pair.  The K‑dimension (the sequence length) is iterated in blocks.
All other operations (layer‑norms, linear projections, gating, final
normalisation and output projection) are done with regular PyTorch
operations – they are cheap compared with the O(N³) reduction.
The kernel works in fp16 for the multiplication and accumulates in fp32,
producing a float‑32 output tensor that matches the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------
# Triton kernel for the core einsum (left @ right^T per channel)
# -------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,               # *const fp16
    right_ptr,              # *const fp16
    out_ptr,                # *mut   fp32
    B, N, H,                # int32 scalars
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    bh = tl.program_id(0)                     # combined batch*hidden index
    pid_i = tl.program_id(1)                  # tile row
    pid_j = tl.program_id(2)                  # tile col

    # coordinate of the first element of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Accumulator in fp32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension in blocks
    # -----------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + \
                 bh * stride_lh + \
                 offs_i[:, None] * stride_li + \
                 offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a block of right (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + \
                 bh * stride_rh + \
                 offs_k[:, None] * stride_rk + \
                 offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # (M,K) * (K,N) -> (M,N)

    # -----------------------------------------------------------------
    # Store the result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + \
               bh * stride_oh + \
               offs_i[:, None] * stride_oi + \
               offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Public entry point used by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input  : torch.Tensor [B, N, N, dim] (float32)
            - mask   : torch.Tensor [B, N, N]      (bool / float)
            - weights: dict of the model parameters (float32)
            - config : {'dim': int, 'hidden_dim': int, 'nomask': bool}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1. Layer‑norm over the last dimension
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]  float32

    # ---------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim
    # ---------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3. Apply mask (if any)
    # ---------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Core einsum – Triton kernel (fp16 inner mul, fp32 acc)
    # ---------------------------------------------------------
    B, N, _, _ = left.shape
    # Cast to fp16 for the kernel (memory bandwidth)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty_like(left_fp16, dtype=torch.float32)   # [B,N,N,hidden]

    # strides (int64) – contiguous tensors => can use .stride()
    stride_lb, stride_li, stride_lk, stride_lh = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – chosen to fit H200 SMEM and achieve high occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # bh dimension
        triton.cdiv(N, BLOCK_M),            # i‑tiles
        triton.cdiv(N, BLOCK_N),            # j‑tiles
    )

    _trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # good default for H200
    )

    # ---------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, gating, final linear
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out  # float32, shape [B,N,N,dim]
