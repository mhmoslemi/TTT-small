"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of the TriMul forward pass is the three‑dimensional einsum

    out[b,i,j,:] = Σ_k  left[b,i,k,:] * right[b,j,k,:]

which is an N×N matrix multiplication for every hidden channel.  This
implementation computes the above in a fused Triton kernel that
processes tiles of (i, j) while accumulating over k and looping over the
hidden dimension in blocks (BLOCK_D).  The surrounding LayerNorms,
linear projections and gating are kept in PyTorch for simplicity.

The kernel is written for float32 (the final result is float32) but
computations are performed in fp32 throughout.  Block sizes were chosen
to suit the Hopper H200 (SM‑90) – they fit into registers and shared
memory while providing good occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel that computes the einsum part of TriMul
# ------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    # strides for left (B, i, k, d)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right (B, j, k, d)
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for out (B, i, j, d)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

    Grid: (batch, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over hidden dimension in blocks
    # ------------------------------------------------------------------
    d0 = 0
    while d0 < D:
        cur_D = tl.minimum(D - d0, BLOCK_D)
        d_range = tl.arange(0, BLOCK_D) + d0                     # absolute d indices
        d_mask = d_range < D

        # accumulator for this d‑block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Accumulate over k
        # ------------------------------------------------------------------
        k0 = 0
        while k0 < N:
            cur_K = tl.minimum(N - k0, BLOCK_K)
            k_range = tl.arange(0, BLOCK_K) + k0               # absolute k indices
            k_mask = k_range < N

            # -------------------------
            # Load left tile: (M, K, D)
            # -------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + k_range[None, :, None] * stride_left_k
                + d_range[None, None, :] * stride_left_d
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & k_mask[None, :, None],
                other=0.0,
            )  # (M, K, D)

            # --------------------------
            # Load right tile: (N, K, D)
            # --------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_j[:, None, None] * stride_right_j
                + k_range[None, :, None] * stride_right_k
                + d_range[None, None, :] * stride_right_d
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & k_mask[None, :, None],
                other=0.0,
            )  # (N, K, D)

            # -------------------------------------------------
            # Outer‑product accumulation for each k in tile
            # -------------------------------------------------
            kk = 0
            while kk < cur_K:
                # left_vec: (M, D)   right_vec: (N, D)
                left_vec = left[:, kk, :]        # (M, D)
                right_vec = right[:, kk, :]      # (N, D)

                # broadcast to (M,N,D) and accumulate
                # left_vec[:, None, :] -> (M,1,D)
                # right_vec[None, :, :] -> (1,N,D)
                acc += left_vec[:, None, :] * right_vec[None, :, :]

                kk += 1

            k0 += BLOCK_K

        # -------------------------------------------------
        # Write accumulated block back to global memory
        # -------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + d_range[None, None, :] * stride_out_d
        )
        store_mask = (
            mask_i[:, None, None] & mask_j[None, :, None] & d_mask[None, None, :]
        )
        tl.store(out_ptrs, acc, mask=store_mask)

        d0 += BLOCK_D


# ------------------------------------------------------------------
# Host entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim] (float32)
            - mask        : torch.Tensor, shape [B, N, N] (bool/float)
            - weights     : dict of parameter tensors
            - config      : dict with keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = D
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Helper to perform a bias‑less linear projection
    # ------------------------------------------------------------------
    def linear_nobias(x, w):
        # x: (..., dim), w: (hidden_dim, dim)  (as stored in PyTorch Linear)
        return F.linear(x, w)                  # bias = None

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left = linear_nobias(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = linear_nobias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask.unsqueeze(-1).to(x.dtype)              # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear_nobias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_nobias(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_nobias(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the triple‑mul (einsum) -> out (B,N,N,hidden)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_out_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = linear_nobias(out, weights["to_out.weight"])     # (B,N,N,dim)

    return out
