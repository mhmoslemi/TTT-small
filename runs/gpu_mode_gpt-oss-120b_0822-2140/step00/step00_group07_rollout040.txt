"""
TriMul “outgoing” implementation (forward only).

The heavy part of the operator is the 3‑dimensional contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of H (=hidden_dim) independent N×N matrix multiplications
(left[...,d] @ right[...,d].T).  The surrounding LayerNorm/Linear/Gates are
performed with PyTorch; the N×N products are carried out by a custom Triton
kernel that mimics a block‑wise GEMM.  The kernel works on fp16 for speed and
the final result is cast back to fp32.

The kernel is launched once per hidden dimension (hidden_dim ≤ 384), which is
still much faster than the naïve Python loops and lets us keep the entire
contraction on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton block‑wise matrix multiplication (C = A @ B^T)
# ----------------------------------------------------------------------
@triton.jit
def _matmul_transpose_kernel(
    a_ptr,                # *float16   LEFT  (B,N,N,H)  slice d
    b_ptr,                # *float16   RIGHT (B,N,N,H)  slice d, but accessed as (K,N)
    c_ptr,                # *float16   OUT   (B,N,N,H)  slice d
    M, N, K,              # matrix sizes (all = seq_len)
    stride_am, stride_ak,        # A strides: (batch, i, k)
    stride_bk, stride_bn,        # B strides: (batch, k, j)   (note: B is transposed)
    stride_cm, stride_cn,        # C strides: (batch, i, j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_b = tl.program_id(2)          # batch index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers with batch offset
    a_ptrs = a_ptr + pid_b * stride_am * 0 + (offs_m[:, None] * stride_am
                                              + offs_k[None, :] * stride_ak)
    b_ptrs = b_ptr + pid_b * stride_bk * 0 + (offs_k[:, None] * stride_bk
                                              + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                 # (M,K)·(K,N) → (M,N)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write result
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_cm * 0 + (offs_m[:, None] * stride_cm
                                              + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a Triton kernel for the
    N×N contraction.  All tensors are expected to reside on the GPU.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    # manual layer‑norm to avoid creating nn.Module objects
    x = input_tensor
    x = (x - x.mean(dim=-1, keepdim=True)) / torch.sqrt(
        x.var(dim=-1, unbiased=False, keepdim=True) + eps
    )
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating (sigmoid)
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. The core contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    #    Implemented as hidden_dim independent GEMMs using Triton.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)   # (B,N,N,hidden_dim)

    # strides for the three tensors (batch, i, k, d)
    stride_left = left_fp16.stride()   # (sB, sI, sK, sD)
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # grid: (i_tiles, j_tiles, batch)
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    # launch one kernel per hidden dimension slice
    for d in range(hidden_dim):
        a_ptr = left_fp16.data_ptr() + d * stride_left[3] * left_fp16.element_size()
        b_ptr = right_fp16.data_ptr() + d * stride_right[3] * right_fp16.element_size()
        c_ptr = out_fp16.data_ptr() + d * stride_out[3] * out_fp16.element_size()

        _matmul_transpose_kernel[grid](
            a_ptr, b_ptr, c_ptr,
            N, N, N,
            stride_left[0], stride_left[2],           # am, ak (batch stride unused here)
            stride_right[2], stride_right[1],         # bk (k), bn (j) – note transpose
            stride_out[0], stride_out[1],
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = (out - out.mean(dim=-1, keepdim=True)) / torch.sqrt(
        out.var(dim=-1, unbiased=False, keepdim=True) + eps
    )
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out
