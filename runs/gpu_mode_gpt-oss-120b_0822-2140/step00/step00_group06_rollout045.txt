"""
TriMul (outgoing) – Triton‑accelerated core.

The heavy part of the TriMul forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication per hidden channel `d`:
    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T

The kernel below computes this operation in FP16 (accumulating in FP32) for
all batches and hidden channels simultaneously.  All other ops
(LayerNorms, Linear projections, gates, final LayerNorm + Linear) are kept
in PyTorch for simplicity and to keep the launch overhead low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel: batched matmul left @ right.T for each hidden dim
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # problem sizes
    stride_lb, stride_li, stride_lk, stride_ld, # left strides (B,N,N,D)
    stride_rb, stride_rj, stride_rk, stride_rd, # right strides (B,N,N,D)
    stride_ob, stride_oi, stride_oj, stride_od, # out strides (B,N,N,D)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # -----------------------------------------------------------------
    # Program ID encodes (batch, hidden) pair + block row/col tiles.
    #   pid0 = batch * D + d    (outer dimension)
    #   pid1 = block row (i)
    #   pid2 = block col (j)
    # -----------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    batch = pid0 // D
    d     = pid0 % D

    # Starting offsets for this block
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Pointers to the start of the (batch,d) slice
    left_base  = left_ptr  + batch * stride_lb + d * stride_ld
    right_base = right_ptr + batch * stride_rb + d * stride_rd
    out_base   = out_ptr   + batch * stride_ob + d * stride_od

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- load left : (M, K) -----
        left_ptrs = (left_base
                     + i_offset[:, None] * stride_li   # i dimension
                     + k_offset[None, :] * stride_lk   # k dimension
                     )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # ----- load right (need K as rows, N as cols) -----
        # right layout is (batch, j, k, d). We want tile (K, N)
        right_ptrs = (right_base
                      + k_offset[:, None] * stride_rk   # k as row
                      + j_offset[None, :] * stride_rj   # j as column
                      )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = (out_base
                + i_offset[:, None] * stride_oi
                + j_offset[None, :] * stride_oj
                )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Python entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Args
    ----
    data : (input, mask, weights, config)
        input  : [B, N, N, dim]   (float32)
        mask   : [B, N, N]        (float32 or bool) – may be omitted in the config
        weights: dict of layer‑norm & linear parameters
        config : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    output : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim = hidden_dim of model)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim]

    # ---------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ---------------------------------------------------------
    # shape: (B,N,N,hidden)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast over hidden)
    # ---------------------------------------------------------
    if not nomask:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------------------------------------------------------
    # 4️⃣ Gating scalars (sigmoid) and gating of projections
    # ---------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5️⃣ Core TriMul – Batched matmul per hidden channel (FP16 kernel)
    # ---------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # output buffer (FP16, will be cast later)
    out_fp16 = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides (in elements) – needed because Triton works with element strides
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    # Blocking parameters – tuned for Hopper (SM90) shared memory 64 KB
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    total_outer = B * hidden
    grid = (
        total_outer,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_fp16,
        B=B,
        N=N,
        D=hidden,
        stride_lb=stride_lb, stride_li=stride_li, stride_lk=stride_lk, stride_ld=stride_ld,
        stride_rb=stride_rb, stride_rj=stride_rj, stride_rk=stride_rk, stride_rd=stride_rd,
        stride_ob=stride_ob, stride_oi=stride_oi, stride_oj=stride_oj, stride_od=stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,               # empirically good for Hopper
    )

    # ---------------------------------------------------------
    # 6️⃣ Cast back to FP32 and apply final norm / gating / linear
    # ---------------------------------------------------------
    out = out_fp16.to(dtype)

    # to_out_norm (LayerNorm over hidden dimension)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # scale by out_gate and final linear projection back to `dim`
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
