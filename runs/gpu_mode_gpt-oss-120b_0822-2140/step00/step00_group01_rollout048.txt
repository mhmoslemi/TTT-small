"""
Triton‐accelerated "outgoing" TriMul (AlphaFold3) forward pass.

The heavy part of TriMul is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel d we need a matrix multiply of shape (N,N) × (N,N)^T.
We fuse the hidden dimension into the batch dimension (B*H) and run a
batched GEMM kernel written in Triton.  All other ops (layer‑norms,
linear projections, gates, final linear) stay in PyTorch (they are cheap
relative to the O(N³) contraction).

The kernel works in FP16 for speed and the result is cast back to FP32 to
match the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel : batched matrix multiplication (A @ B^T) for many batches
# ----------------------------------------------------------------------
@triton.jit
def _trmul_gemm_kernel(
    A,                     # ptr to left matrix   [Btot, M, K]
    B,                     # ptr to right matrix  [Btot, K, N] (already transposed)
    C,                     # ptr to output matrix [Btot, M, N]
    M, N, K,               # problem sizes
    stride_az, stride_am, stride_ak,   # strides of A
    stride_bz, stride_bk, stride_bn,   # strides of B
    stride_cz, stride_cm, stride_cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------  offsets within the block --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers start for this batch
    a_ptr = A + batch_id * stride_az
    b_ptr = B + batch_id * stride_bz
    c_ptr = C + batch_id * stride_cz

    # -----------------------------------------------------------------
    # accumulator
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # ---- load a block of A ----
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a_mask = (cur_k < K) & mask_m[:, None]
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak,
            mask=a_mask,
            other=0.0,
        )
        # ---- load a block of B (already transposed) ----
        b_mask = (cur_k[:, None] < K) & mask_n[None, :]
        b = tl.load(
            b_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=b_mask,
            other=0.0,
        )
        # ---- dot product ----
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # write back
    # -----------------------------------------------------------------
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(
        c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc.to(tl.float16),
        mask=c_mask,
    )


# ----------------------------------------------------------------------
# Helper: fused layer‑norm (weight & bias are 1‑D)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask        : torch.Tensor [B, N, N]  (float32 or bool)
            - weights     : dict of named parameters (torch.Tensor)
            - config      : dict containing "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim = C)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = _layer_norm(input_tensor, norm_weight, norm_bias)          # [B,N,N,C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)  ->  left / right  (shape [B,N,N,hidden])
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])                # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1)                                 # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of left/right
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batch GEMM over (B*hidden) batches.
    # ------------------------------------------------------------------
    # Convert to FP16 for the GEMM (speed)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape: merge batch and hidden dims, keep (i,k) for left and (k,j) for right
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # [B*H, M, K]
    # transpose right's last two dims before merging to get (k,j)
    right_t = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)   # [B*H, K, N]

    # output buffer (FP16)
    out_resh = torch.empty_like(left_resh)

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (row‑major contiguous tensors)
    stride_a_z, stride_a_m, stride_a_k = left_resh.stride()
    stride_b_z, stride_b_k, stride_b_n = right_t.stride()
    stride_c_z, stride_c_m, stride_c_n = out_resh.stride()

    _trmul_gemm_kernel[grid](
        left_resh,
        right_t,
        out_resh,
        N,                # M
        N,                # N
        N,                # K
        stride_a_z,
        stride_a_m,
        stride_a_k,
        stride_b_z,
        stride_b_k,
        stride_b_n,
        stride_c_z,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, hidden]
    out = out_resh.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden) and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = _layer_norm(out, to_out_norm_weight, to_out_norm_bias)            # [B,N,N,hidden]

    out = out * out_gate                                                    # gated

    # final linear: hidden -> C
    out = F.linear(out, weights["to_out.weight"])                          # [B,N,N,C]

    return out
