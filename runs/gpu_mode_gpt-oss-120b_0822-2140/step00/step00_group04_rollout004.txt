"""
TriMul “outgoing” implementation.

We keep the heavyweight tensor contractions (the 4‑D einsum) in
torch – cuBLAS / torch‑einsum is already heavily optimized for the
N³·H work‑load.  The part that is *not* covered by a library kernel is the
element‑wise masking + gating step; we fuse it with a tiny Triton kernel
that works on the flattened 4‑D view of the tensors.

The kernel reads the four tensors (left, right, left_gate,
right_gate) together with the optional mask, applies
   left  = left  * mask * left_gate
   right = right * mask * right_gate
in‑place and writes the results back.  All remaining
operations (LayerNorm, Linear, the big einsum, the final
norm/gate/linear) are performed with regular PyTorch calls.

The overall algorithm matches the reference PyTorch code while
off‑loading a non‑trivial element‑wise step to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused mask + gating
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,        # [B, N, N, H]
    lgate_ptr, rgate_ptr,       # [B, N, N, H]
    mask_ptr,                   # [B, N, N]  (or nullptr when nomask)
    B, N, H,                    # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK: tl.constexpr,
):
    """Flattened 1‑D kernel that updates left/right in‑place."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear indices
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # Convert linear index -> (b, i, k, h)
    # ------------------------------------------------------------------
    #  idx = ((b * N + i) * N + k) * H + h
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------------------------------------------------------------------
    # Load values (with out‑of‑bounds safety)
    # ------------------------------------------------------------------
    left = tl.load(
        left_ptr
        + b * stride_left_b
        + i * stride_left_i
        + k * stride_left_k
        + h * stride_left_h,
        mask=mask,
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + b * stride_right_b
        + i * stride_right_i
        + k * stride_right_k
        + h * stride_right_h,
        mask=mask,
        other=0.0,
    )
    lgate = tl.load(
        lgate_ptr
        + b * stride_lgate_b
        + i * stride_lgate_i
        + k * stride_lgate_k
        + h * stride_lgate_h,
        mask=mask,
        other=0.0,
    )
    rgate = tl.load(
        rgate_ptr
        + b * stride_rgate_b
        + i * stride_rgate_i
        + k * stride_rgate_k
        + h * stride_rgate_h,
        mask=mask,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over H)
    # ------------------------------------------------------------------
    if mask_ptr != 0:
        # mask has shape [B,N,N] → same (b,i,k) indexing, no h dimension
        m = tl.load(
            mask_ptr
            + b * stride_mask_b
            + i * stride_mask_i
            + k * stride_mask_k,
            mask=mask,
            other=0.0,
        )
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * lgate
    right = right * rgate

    # ------------------------------------------------------------------
    # Store results back
    # ------------------------------------------------------------------
    tl.store(
        left_ptr
        + b * stride_left_b
        + i * stride_left_i
        + k * stride_left_k
        + h * stride_left_h,
        left,
        mask=mask,
    )
    tl.store(
        right_ptr
        + b * stride_right_b
        + i * stride_right_i
        + k * stride_right_k
        + h * stride_right_h,
        right,
        mask=mask,
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, D]
            - mask  : torch.Tensor [B, N, N]  (may be all‑ones when nomask=True)
            - weights: dict of the model parameters
            - config : dict containing ``dim`` (D) and ``hidden_dim`` (H)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim = D)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) – keep in same dtype as projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Fuse mask + gating with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    BLOCK = 1024  # 1‑K elements per program
    total_elements = B * N * N * H
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Strides for the 4‑D tensors (as returned by .stride())
    s_left = left.stride()
    s_right = right.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()
    s_mask = mask.stride() if mask is not None else (0, 0, 0)

    # Pointer to mask – 0 if mask is a dummy all‑ones tensor (nomask case)
    mask_ptr = mask.data_ptr() if mask is not None else 0

    _mask_gate_fuse_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_ptr,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_lgate[0],
        s_lgate[1],
        s_lgate[2],
        s_lgate[3],
        s_rgate[0],
        s_rgate[1],
        s_rgate[2],
        s_rgate[3],
        s_mask[0] if mask_ptr != 0 else 0,
        s_mask[1] if mask_ptr != 0 else 0,
        s_mask[2] if mask_ptr != 0 else 0,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5) Core contraction:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Use torch.einsum (bfloat16 -> float32 later)
    # ------------------------------------------------------------------
    left = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Final projection path
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate  # gating after norm
    out = F.linear(out, weights["to_out.weight"].to(device))  # back to D

    return out
