"""
TriMul “outgoing” implementation (forward only).

The heavy part of the operator is the pairwise multiplication over the
sequence dimension:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are already linearly projected, gated and masked.
The kernel below computes this N³ reduction while keeping the hidden
dimension (d) in registers.  All surrounding LayerNorm/Linear/Gates are
performed with regular PyTorch calls – they are cheap compared with the
N³ inner product.  The kernel works for any batch size ≤2, any sequence
length up to 1024 and hidden dimension up to 384 (the max used in the
tests).  It returns a float‑32 tensor as required.

The launch grid consists of one thread per (b,i,j) triple; each thread
accumulates over the k‑dimension in blocks of `BLOCK_K` elements.
Hidden dimension is passed as a `tl.constexpr` so the kernel is compiled
once per distinct hidden size.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_K: tl.constexpr,    # size of the k‑block
    HIDDEN_DIM: tl.constexpr, # hidden dimension (compile‑time constant)
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    where left/right are already masked and gated.
    One program instance handles a single (b,i,j) triple.
    """
    pid = tl.program_id(0)                         # linear id over B*N*N
    total_ij = N * N
    b = pid // total_ij
    pid_ij = pid % total_ij
    i = pid_ij // N
    j = pid_ij % N

    # accumulator for the hidden dimension
    acc = tl.zeros([HIDDEN_DIM], dtype=tl.float32)

    # tile over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        k_mask = k_start < N                                    # (BLOCK_K,)

        # ---- load left[b,i,k,:] -------------------------------------------------
        # shape (BLOCK_K, HIDDEN_DIM)
        left_off = (
            b * stride_l_b
            + i * stride_l_i
            + (k_start[:, None] * stride_l_k)
            + (tl.arange(0, HIDDEN_DIM)[None, :] * stride_l_h)
        )
        left = tl.load(left_ptr + left_off, mask=k_mask[:, None], other=0.0)

        # ---- load right[b,j,k,:] ------------------------------------------------
        right_off = (
            b * stride_r_b
            + j * stride_r_j
            + (k_start[:, None] * stride_r_k)
            + (tl.arange(0, HIDDEN_DIM)[None, :] * stride_r_h)
        )
        right = tl.load(right_ptr + right_off, mask=k_mask[:, None], other=0.0)

        # element‑wise product and reduction over k
        prod = left * right                     # (BLOCK_K, HIDDEN_DIM)
        acc += tl.sum(prod, axis=0)             # (HIDDEN_DIM,)

    # ----------------------------------------------------------------------
    # write result
    # ----------------------------------------------------------------------
    out_off = (
        b * stride_o_b
        + i * stride_o_i
        + j * stride_o_j
        + (tl.arange(0, HIDDEN_DIM) * stride_o_h)
    )
    tl.store(out_ptr + out_off, acc)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  torch.float32 on CUDA
        - mask        : [B, N, N]        torch.float32 (0/1) on CUDA
        - weights     : dict of parameters (see docstring)
        - config      : dict with keys "dim", "hidden_dim"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]               # = H

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # optional mask handling
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Triton kernel for the N³ reduction
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides in element units (not bytes)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    # kernel launch configuration
    BLOCK_K = 64                      # reasonable trade‑off for H200
    total = B * N * N
    grid = (triton.cdiv(total, 1),)   # one thread per (b,i,j)

    trimul_outgoing_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l[0],
        stride_l_i=stride_l[1],
        stride_l_k=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_b=stride_r[0],
        stride_r_j=stride_r[2],        # note: right uses j as the second pair index
        stride_r_k=stride_r[2],
        stride_r_h=stride_r[3],
        stride_o_b=stride_o[0],
        stride_o_i=stride_o[1],
        stride_o_j=stride_o[2],
        stride_o_h=stride_o[3],
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=4,                  # the H200 can handle many warps; 4 is safe
    )

    # ------------------------------------------------------------------
    # 4) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection back to dim

    return out
