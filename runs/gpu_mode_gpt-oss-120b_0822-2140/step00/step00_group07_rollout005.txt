"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

The heavy part of TriMul is the 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply per hidden dimension

    out_d = L_d @ R_dᵀ   where L_d,R_d ∈ ℝ^{N×N}.

We implement this contraction with a Triton GEMM that works on the
flattened batch × hidden_dim dimension (B·H blocks).  All surrounding
operations (LayerNorm, Linear, gating, masking) are performed with
PyTorch – they are cheap compared with the N³ work.

The kernel works in fp16 (or bf16) for memory bandwidth while accumulating
in fp32 for numeric stability.  The final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (per batch*hidden)
# left  : [B, N, N, H]   (i,k) layout
# right : [B, N, N, H]   (j,k) layout – we read it as (k,j) to get the transpose
# out   : [B, N, N, H]   (i,j) layout
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const * fp16
    right_ptr,              # const * fp16
    out_ptr,                # * fp16
    B, H, N,                # int32
    stride_lb, stride_li, stride_lk,   # left strides (batch, i, k)
    stride_rb, stride_rk, stride_rj,   # right strides (batch, k, j)  <-- note transpose
    stride_ob, stride_oi, stride_oj,   # out strides (batch, i, j)
    BLOCK_M: tl.constexpr,  # tile size in i direction
    BLOCK_N: tl.constexpr,  # tile size in j direction
    BLOCK_K: tl.constexpr,  # tile size in reduction k
):
    pid_z = tl.program_id(2)                 # batch * hidden
    pid_m = tl.program_id(0)                 # i tile
    pid_n = tl.program_id(1)                 # j tile

    # ----- offsets & masks -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute the start of the current (batch, hidden) slice
    batch = pid_z // H                     # actual batch index
    hidden = pid_z % H                     # hidden dim index inside the batch
    # offset to the start of this slice
    batch_offset = batch * stride_lb + hidden * stride_li  # left/right/out share hidden stride = stride_i (same layout)

    # accumulators in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main reduction loop over k --------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ----- load left tile (i,k) ---------------------------------------
        # left offset: batch_offset + i*stride_li + k*stride_lk
        left_ptrs = left_ptr + batch_offset \
                    + offs_m[:, None] * stride_li \
                    + cur_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)   # upcast for dot

        # ----- load right tile (k,j)   (transposed view) -------------------
        # right offset: batch_offset + k*stride_rk + j*stride_rj
        right_ptrs = right_ptr + batch_offset \
                     + cur_k[:, None] * stride_rk \
                     + offs_n[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right = right.to(tl.float32)

        # ----- accumulate ----------------------------------------------------
        acc += tl.dot(left, right)

    # ----- write result ----------------------------------------------------
    out_ptrs = out_ptr + batch_offset \
               + offs_m[:, None] * stride_oi \
               + offs_n[None, :] * stride_oj
    out = acc.to(tl.float16)   # back to fp16
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Helper: fused LayerNorm (no bias‑variance reduction across hidden dims)
# -------------------------------------------------------------------------
def fused_layernorm(x, weight, bias, eps=1e-5):
    # x : (..., C)
    return F.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)

# -------------------------------------------------------------------------
# Main entry point required by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a Triton GEMM kernel.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # unpack weights (all stored as float32)
    # -----------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = fused_layernorm(input_tensor, norm_w, norm_b)               # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = F.linear(x, left_proj_w)          # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)        # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3) Mask (optional)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> broadcast over hidden dim
        mask = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask
    # otherwise mask == 1, no op

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core contraction via Triton (fp16 for bandwidth)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # cast to fp16 (or bf16) for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out_h = torch.empty_like(left_h)      # shape [B,N,N,hidden], fp16

    # strides (int64) – use .stride() which returns in elements, not bytes
    stride_lb = left_h.stride(0)          # batch stride
    stride_li = left_h.stride(1)          # i   stride
    stride_lk = left_h.stride(2)          # k   stride
    stride_rb = right_h.stride(0)         # batch stride (same as left)
    stride_rk = right_h.stride(2)         # k   stride (original last N)
    stride_rj = right_h.stride(1)         # j   stride (original middle N)
    stride_ob = out_h.stride(0)
    stride_oi = out_h.stride(1)
    stride_oj = out_h.stride(2)

    # Tile sizes – chosen to fit H200 SMEM and achieve high occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,                         # flatten batch*hidden as third grid dim
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, hidden, N,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_rk, stride_rj,
        stride_ob, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # a good default for fp16 GEMM on H200
    )

    # back to fp32 for the rest of the pipeline
    out = out_h.to(dtype)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm
    # -----------------------------------------------------------------
    out = fused_layernorm(out, to_out_norm_w, to_out_norm_b)

    # -----------------------------------------------------------------
    # 7) Output gating
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8) Final linear projection back to dim
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_w)        # [B,N,N,dim]

    return out
