"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We compute it with a custom Triton kernel that multiplies, for every
(batch, head) pair, the (N,N) tensor `left` with the transpose of the
(N,N) tensor `right`.  All other operations (LayerNorm, Linear, gating,
final LayerNorm/Linear) stay in PyTorch – they are cheap compared to the
O(N³) contraction.

The kernel works on tensors of shape (B, H, N, N) that are made contiguous
in memory (row‑major).  Each program instance computes a TILE_M × TILE_N
block of the output matrix for a single (batch, head) pair.
The implementation uses FP32 accumulation (the inputs are cast to FP16
for bandwidth) and writes the result back in FP32, matching the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)^T
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    N, H,                               # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Computes for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B have shape (N, N))
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ----------- offsets for the tile ----------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialise accumulator
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ----- load A : shape (TILE_M, TILE_K) -----
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # ----- load Bᵀ : we need B[j,k] -> load as (TILE_K, TILE_N) -----
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dimension
                  + offs_n[None, :] * stride_b_i)  # j as second dimension
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # dot‑product over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) -> (TILE_M,TILE_N)

    # ----------- store result ----------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32 or bfloat16)
        - mask         : torch.Tensor [B, N, N]      (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors           model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to fetch weight & bias tensors (they are already on device)
    # ------------------------------------------------------------------
    def w(name): return weights[name]
    def b(name): return weights[name]  # biases stored with same key naming

    # ------------------------------------------------------------------
    # 1. LayerNorm over ``dim`` (the last dimension of the input)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections & gates   (bias=False in the paper)
    # ------------------------------------------------------------------
    # Linear:  (B,N,N,dim) @ (hidden_dim, dim)ᵀ   => (B,N,N,hidden_dim)
    left  = F.linear(x, w("left_proj.weight"))          # [B,N,N,H]
    right = F.linear(x, w("right_proj.weight"))         # [B,N,N,H]

    # mask handling
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(dtype)            # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # gates
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))   # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Heavy contraction  (batch*head) matrix multiply using Triton
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and make contiguous
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    # allocate output
    out_t = torch.empty_like(left_t, dtype=dtype)    # [B,H,N,N]

    # strides (in elements) – needed by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)   we will load transposed
    c_s = out_t.stride()    # (batch, head, i, j)

    # launch parameters
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),
        triton.cdiv(N, TILE_N),
        B * hidden_dim,
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,
    )
    # back to original layout
    out = out_t.permute(0, 2, 3, 1)                 # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Final LayerNorm + gating + linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                      # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
