"""
TriMul “outgoing” implementation.

The costly step is the 3‑D batched matrix multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]      (d = hidden_dim)

For every batch and hidden channel this is a dense N×N × N matrix product.
We fuse the three nested loops into a single Triton kernel:
    * tile the (i,j) output plane with BLOCK_M × BLOCK_N
    * iterate over k in tiles of BLOCK_K
    * load a (BLOCK_M, BLOCK_K) tile from *left* and a (BLOCK_N, BLOCK_K) tile from *right*
    * broadcast‑multiply along the k‑dimension and sum → accumulate a (BLOCK_M, BLOCK_N) block
The rest of the TriMul pipeline (LayerNorms, Linear projections, gates, final
Linear) stays in PyTorch – this keeps the code concise while still moving the
N³ work to a fast Triton kernel.

The kernel works for any batch size (B ≤ 2), any sequence length N (≤ 1024)
and any hidden dimension H (here 128).  All tensors are expected to be on
CUDA and in float32.  The final output is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for the N³ multiplication (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,
    # strides (batch, i, j, k, channel) for each tensor
    stride_out_bn, stride_out_i, stride_out_j, stride_out_h,
    stride_left_bn, stride_left_i, stride_left_k, stride_left_h,
    stride_right_bn, stride_right_j, stride_right_k, stride_right_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[...,k,:] * right[...,k,:] for one channel."""
    pid_m = tl.program_id(0)               # tile index over i
    pid_n = tl.program_id(1)               # tile index over j
    pid_bh = tl.program_id(2)              # combined batch × hidden index

    b = pid_bh // H                        # batch id
    h = pid_bh % H                         # hidden channel id

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left block (M × K) -----------------------------------
        a_ptr = (
            left_ptr
            + b * stride_left_bn
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # -------- right block (N × K) ----------------------------------
        b_ptr = (
            right_ptr
            + b * stride_right_bn
            + h * stride_right_h
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        b_ = tl.load(b_ptr, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # -------- multiply‑accumulate over K ---------------------------
        # a : (M, K)   b_ : (N, K)  → broadcast → (M, N, K)
        prod = a[:, None, :] * b_[None, :, :]          # (M, N, K)
        acc += tl.sum(prod, axis=2)                    # (M, N)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_bn
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a fused Triton kernel.

    Args:
        data: (input, mask, weights, config)
            input  – Tensor[B, N, N, dim]   (float32, CUDA)
            mask   – Tensor[B, N, N] (bool/int, CUDA). May be all‑ones.
            weights – dict with the exact parameter names used in the reference
                      implementation.
            config  – dict containing at least:
                * dim          – input channel dimension
                * hidden_dim   – hidden dimension after projection
    Returns:
        Tensor[B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])       # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gates (sigmoid activation)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: batch‑wise, channel‑wise N³ multiplication
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # Strides (convert to int for Triton)
    stride_out = out.stride()          # (N*N*H, N*H, H, 1)
    stride_left = left.stride()
    stride_right = right.stride()

    # Kernel launch parameters
    BLOCK_M = 64   # tunable; fits well into shared memory on H200
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),               # i‑tiles
        triton.cdiv(N, BLOCK_N),               # j‑tiles
        B * hidden                              # batch × channel
    )

    _trimul_outgoing_kernel[grid](
        out_ptr=out,
        left_ptr=left,
        right_ptr=right,
        B=B,
        N=N,
        H=hidden,
        # output strides
        stride_out_bn=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        # left strides
        stride_left_bn=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        # right strides
        stride_right_bn=stride_right[0],
        stride_right_j=stride_right[2],   # note: right indices are (b,i,j,hidden) → we need j‑stride
        stride_right_k=stride_right[1],   # k corresponds to the second N axis
        stride_right_h=stride_right[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm → gate → final Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])

    return out
