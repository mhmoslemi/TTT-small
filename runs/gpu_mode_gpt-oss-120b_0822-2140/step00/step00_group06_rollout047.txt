def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    # LayerNorm on last dim dim
    # use weight and bias
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True)+1e-5)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # linear projections without bias
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply mask if given: mask shape [B,N,N]; unsqueeze last dim
    if mask is not None:
        mask_uns = mask.unsqueeze(-1)
        left = left * mask_uns
        right = right * mask_uns

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # reshape for matmul
    # combine batch and hidden dimensions for matmul
    left_ = left.permute(0,3,1,2).contiguous().view(bs*hidden, N, N)  # (B,hidden,N,N) -> (B*hidden,N,N)
    right_ = right.permute(0,3,1,2).contiguous().view(bs*hidden, N, N)

    # compute out = left_ @ right_.transpose(-2,-1) using Triton
    out_ = triton_matmul(left_, right_, transpose_b=True)  # we need to implement kernel that can optionally transpose B

    # reshape back
    out = out_.view(bs, hidden, N, N).permute(0,2,3,1).contiguous()  # shape [B,N,N,hidden]

    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True)+1e-5)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,

):

    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_n = pid % num_pid_n

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

"""
TriMul “outgoing” implementation with a Triton‑accelerated batched matrix‑multiply.

The heavy part of the forward pass is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched GEMM:
    left   : (B, N, N, H) → (B*H, N, N)
    right  : (B, N, N, H) → (B*H, N, N).T
    out    : (B*H, N, N) = left @ right.T
The rest of the operations (LayerNorm, Linear projections, gates) are
performed with regular PyTorch kernels – they are cheap compared to the
O(B·H·N³) multiplication.

The Triton kernel is a simple blocked matmul (float16 compute, float32
accumulation) that works for any batch size, sequence length and hidden
dimension that fit in GPU memory.  It is launched with a single‑dim grid
where each program instance computes a TILE‑M × TILE‑N block of C.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM (C = A @ B) – A,B are 3‑D tensors (batch, M, K) and
# (batch, K, N).  The kernel works on the last two dimensions; the batch
# dimension is fused into the program launch grid.
# ----------------------------------------------------------------------
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ak,              # A stride (batch, M, K)
    stride_bk, stride_bn,              # B stride (batch, K, N)
    stride_cm, stride_cn,              # C stride (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # 2‑D grid of blocks
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (M, K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # B: (K, N)
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # dot: sum over K dimension → fp32
        acc += tl.dot(a, b)

    # write back C in fp16 (later cast to fp32 in PyTorch)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the above Triton kernel.
    a: (B, M, K)   dtype=float16
    b: (B, K, N)   dtype=float16
    returns c: (B, M, N) dtype=float32
    """
    assert a.is_cuda and b.is_cuda
    assert a.shape[0] == b.shape[0]
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    # output allocation (float16 for kernel, cast later)
    c = torch.empty((B, M, N), device=a.device, dtype=torch.float16)

    # strides (in elements, not bytes)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    _matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Cast back to float32 for the rest of the network
    return c.to(torch.float32)


# ----------------------------------------------------------------------
# Main entry point – the user‑visible kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Operates on a 4‑D tensor [B, N, N, C] (C = dim) and returns a tensor
    of the same shape.  All linear / norm parts are done with PyTorch;
    the N³ einsum is replaced by a fused Triton GEMM.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]          # H in the paper

    # ---------- LayerNorm on the last dim ----------
    # Using functional layer_norm to avoid extra Python overhead
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ---------- Linear projections (no bias) ----------
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, H]

    # ---------- Apply optional mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for GEMM ----------
    # reshape to (B*H, N, N) and cast to fp16 for the kernel
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N).to(torch.float16)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N).to(torch.float16)

    # right needs to be transposed on the last two axes before matmul
    right_T = right_.transpose(1, 2).contiguous()

    # ---------- Triton batched GEMM ----------
    out_ = triton_matmul(left_, right_T)   # (B*H, N, N) in float32

    # ---------- Reshape back to [B, N, N, H] ----------
    out = out_.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B,N,N,H]

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear (H -> dim) ----------
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out.to(torch.float32)
