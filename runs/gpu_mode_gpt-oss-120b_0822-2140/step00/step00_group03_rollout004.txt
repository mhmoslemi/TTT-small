"""
TriMul “outgoing” kernel (AlphaFold‑3)

We implement the forward pass of the TriMul operator.  All expensive
operations are kept in FP16 inside a custom Triton kernel:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

The kernel computes a batched GEMM:
    left(b,h)   : (N, N)          (i,k)
    right(b,h)ᵀ : (N, N)          (j,k) → (k,j)
    out(b,h)    : (N, N) = left @ rightᵀ

All surrounding LayerNorm / Linear / gating logic stays in PyTorch
(but uses the supplied weight tensors, so the result is numerically
identical to the reference implementation).  The final output is
returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ B   (A: (M,K), B: (K,N))
# Batch dimension is fused into the program ID.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B: (batch, K, N)
    stride_ch, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program id == batch index
    batch_idx = tl.program_id(0)

    # ------------------------------------------------------------------
    # Pointers for the current batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + batch_idx * stride_ab
    b_batch = b_ptr + batch_idx * stride_bb
    c_batch = c_ptr + batch_idx * stride_cb

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds (when M or N not multiple of block size)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = a_batch + (offs_m[:, None] * stride_ah) + (offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bn

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_ch) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a Triton GEMM
    for the O(N³) contraction.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor[B, N, N, dim] (float32)
            mask   – torch.Tensor[B, N, N] (bool/float) – may be all‑ones
            weights – dict of tensors (see docstring)
            config  – dict with keys ["dim", "hidden_dim"]
    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = inp.device
    B, N, _, _ = inp.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating sigmoids
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    #   left  : [B, N, N, H] → [B, H, N, N]
    #   right : [B, N, N, H] → [B, H, N, N] → transpose last two dims
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_t = right_perm.transpose(-1, -2).contiguous()  # [B, H, N, N] (j,k) → (k,j)

    # Flatten batch*head dimension → (B*H, N, N)
    A = left_perm.view(B * hidden_dim, N, N).to(torch.float16)
    B_ = right_t.view(B * hidden_dim, N, N).to(torch.float16)

    # Allocate output
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    total_batches = B * hidden_dim
    grid = (total_batches,)

    # stride helpers (all tensors are contiguous in their view)
    stride_ah, stride_ak, stride_ab = A.stride()  # (M*K, K, 1)
    stride_bk, stride_bn, stride_bb = B_.stride()
    stride_ch, stride_cn, stride_cb = C.stride()

    _batched_gemm_kernel[grid](
        A.data_ptr(),
        B_.data_ptr(),
        C.data_ptr(),
        N,            # M == N == seq_len
        N,            # N
        N,            # K
        stride_ah,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_ch,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # Reshape back to original layout and cast to fp32
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N, N)                     # [B,H,N,N]
    out = out.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Final LayerNorm (hidden_dim), gate, and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
