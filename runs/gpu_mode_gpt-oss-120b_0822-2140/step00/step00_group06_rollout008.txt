"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the tensor contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is equivalent to a batched matrix multiplication
    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T

The kernel below performs this batched GEMM for all (batch, hidden_dim)
pairs in a single launch, using fp32 accumulation.  All surrounding
operations (LayerNorms, Linear projections, sigmoids, final LayerNorm
and Linear) are kept in PyTorch for simplicity and to keep the code
readable.

The kernel works for any batch size (1‑2), any sequence length N up to
1024, and any hidden dimension (e.g. 128).  It expects contiguous tensors
in the layout (B, N, N, H) – the same layout as the reference PyTorch
code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A,B shape: (B*H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,                     # *float32   [batch, M, K]
    B_ptr,                     # *float32   [batch, N, K]   (already transposed)
    C_ptr,                     # *float32   [batch, M, N]
    M, N, K,                   # sizes
    stride_Ab, stride_Am, stride_Ak,   # strides for A (batch, M, K)
    stride_Bb, stride_Bn, stride_Bk,   # strides for B (batch, N, K)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                # batch * hidden_dim index
    pid_m = tl.program_id(0)                # block row
    pid_n = tl.program_id(1)                # block col

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (happens on the last tiles)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers
        a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # advance k‑pointers (implicit via loop index)
    # ------------------------------------------------------------------
    # write back
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def _batched_gemm(A: torch.Tensor, B_T: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B_T for 3‑D tensors A,B_T of shape (B*H, N, N) using
    the Triton kernel defined above.  The result C has the same shape and
    dtype (float32).
    """
    assert A.is_contiguous() and B_T.is_contiguous()
    batch, M, K = A.shape
    _, N, _ = B_T.shape
    assert K == N, "inner dimensions must match"

    # output allocation
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides (in elements, not bytes)
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bn, stride_Bk = B_T.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    # block sizes – tuned for Hopper (SM90)  (empirically good)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_gemm_kernel[grid](
        A,
        B_T,
        C,
        M,
        N,
        K,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bn,
        stride_Bk,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough for fp32 GEMM on H200
    )
    return C


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass with a Triton‑accelerated pairwise
    contraction.

    Args:
        data: (input, mask, weights, config)
            - input : Tensor[bs, N, N, dim]   (float32)
            - mask  : Tensor[bs, N, N]        (float32/bool, 0‑1)
            - weights: dict of all linear / layer‑norm parameters
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # helpers to apply Linear + LayerNorm using the supplied weights
    # ------------------------------------------------------------------
    def linear(x, w):
        # w : [out_dim, in_dim]  (no bias)
        return F.linear(x, w)

    def layer_norm(x, w, b):
        # Normalized over the last dimension (dim)
        return F.layer_norm(x, (x.shape[-1],), weight=w, bias=b, eps=eps)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = linear(x, weights["right_proj.weight"]) # same shape

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM via Triton: out = left @ rightᵀ  (per hidden dim)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) where H = hidden_dim
    B, N, _, _ = left.shape
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_resh = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # transpose the second operand (required by the contraction)
    right_T = right_resh.transpose(1, 2).contiguous()

    # Triton GEMM
    out_resh = _batched_gemm(left_resh, right_T)   # shape (B*H, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_resh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])    # back to dim

    return out.to(torch.float32)
