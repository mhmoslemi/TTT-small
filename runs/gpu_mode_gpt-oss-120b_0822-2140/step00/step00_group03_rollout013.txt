"""
TriMul “outgoing” implementation.

The expensive part of the TriMul forward pass is the
einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be written as a batched matrix multiplication
    out[...,d] = left[...,d] @ right[...,d].T .
We fuse this operation in a Triton kernel that processes
all (batch, hidden) pairs in parallel.  All other
operations (LayerNorm, Linear projections, gates, final
Linear) are kept in PyTorch for simplicity.

The kernel works for arbitrary batch size B (≤2),
sequence length N (≤1024) and hidden dimension H (≤384)
and returns a float‑32 tensor of shape [B,N,N,dim].
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# -------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    B, N, H,                # int32  (batch, seq_len, hidden_dim)
    # strides for left  [b,i,k,d]
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right [b,j,k,d]
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for out   [b,i,j,d]
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,  # tile size in M (i dimension)
    BLOCK_N: tl.constexpr,  # tile size in N (j dimension)
    BLOCK_K: tl.constexpr   # tile size in K (k dimension)
):
    # -------------------------------------------------
    # 1) identify which (batch, hidden) matrix we are computing
    # -------------------------------------------------
    pid = tl.program_id(0)                 # combined batch*hidden index
    batch_idx = pid // H
    hidden_idx = pid % H

    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block col

    # -------------------------------------------------
    # 2) compute the global coordinates of the tile
    # -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # 3) Loop over K dimension in tiles
    # -------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:   A[b,i,k,d]   (row i, col k)
        a_ptrs = (left_ptr
                  + batch_idx * stride_left_b
                  + hidden_idx * stride_left_d
                  + offs_i[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right:  B[b,j,k,d]   we need B^T => B[j,k] accessed as (k,j)
        b_ptrs = (right_ptr
                  + batch_idx * stride_right_b
                  + hidden_idx * stride_right_d
                  + offs_j[None, :] * stride_right_j
                  + offs_k[:, None] * stride_right_k)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate dot product over the K‑tile
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # 4) Write back the result
    # -------------------------------------------------
    out_ptrs = (out_ptr
                + batch_idx * stride_out_b
                + hidden_idx * stride_out_d
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor [B, N, N, dim] (float32)
        - mask  : torch.Tensor [B, N, N]   (float32 or bool)
        - weights: dict of model parameters (already on CUDA)
        - config : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # called hidden_dim in the paper

    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm on the input (dim‑wise) ----------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ---------- Linear projections (no bias) ----------
    # shape after projection: [B, N, N, hidden]
    left = F.linear(x, weights["left_proj.weight"])   # (hidden, dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates (sigmoid) ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Triton GEMM for the core einsum
    # -------------------------------------------------
    # Prepare output buffer
    out = torch.empty((B, N, N, hidden), dtype=torch.float32,
                      device=input_tensor.device)

    # Strides (int64) needed by the kernel
    stride_left = left.stride()   # (b,i,k,d)
    stride_right = right.stride()
    stride_out = out.stride()

    # Choose tile sizes (tuned for Hopper, 128‑bit loads)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_bd = B * hidden
    grid = (total_bd,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_d=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_d=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_d=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # heuristic for Hopper
    )

    # ---------- to_out_norm (LayerNorm over hidden) ----------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection back to dim ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
