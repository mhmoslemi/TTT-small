"""
TriMul “outgoing” implementation for AlphaFold‑3 style models.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,dim).
2. Linear projections → left / right   (dim → hidden_dim).
3. Optional mask applied to the projected tensors.
4. Gating: sigmoid(linear(x)) per projection and an output gate.
5. Core operation – a batched matrix multiplication over the
   *k* axis:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a set of (B·hidden_dim) independent
   matrix‑multiplies  L @ Rᵀ  (L,R : N×N).  This is performed with a
   custom Triton kernel (`batch_matmul_kernel`) using FP16
   arithmetic and FP32 accumulation.
6. Layer‑norm over the hidden dimension, element‑wise multiplication
   with the output gate, and a final linear projection back to `dim`.

The kernel implements a standard tiled GEMM with three‑dimensional
grid (batch, row‑block, col‑block) and works for any N up to 1024 on
Hopper (sm_90).  All host‑side tensors are kept on the GPU; only the
core N³ reduction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  (L @ Rᵀ)   L,R : [B·H, N, N]  ->  C : [B·H, N, N]
# --------------------------------------------------------------------------- #
@triton.jit
def batch_matmul_kernel(
    a_ptr,               # *float16   left  tensor   [batch, M, K]
    b_ptr,               # *float16   right tensor   [batch, N, K]   (will be transposed)
    c_ptr,               # *float16   output tensor  [batch, M, N]
    batch, M, N, K,
    stride_ah, stride_am, stride_ak,   # strides for a
    stride_bh, stride_bn, stride_bk,   # strides for b  (original layout)
    stride_ch, stride_cm, stride_cn,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask to stay inside the matrix
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # shared accumulators in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # --------------------- load A tile ---------------------
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                 + (k * BLOCK_K + offs_k)[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & ((k * BLOCK_K + offs_k) < K), other=0.0)
        a = a.to(tl.float32)

        # --------------------- load B tile (transposed) ---------------------
        # we want Bᵀ, i.e. element (k, n) of Bᵀ is B[n, k] in memory.
        b_ptrs = b_batch_ptr + ((k * BLOCK_K + offs_k)[:, None] * stride_bk
                                 + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=((k * BLOCK_K + offs_k) < K)[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # --------------------- MAC ---------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Helper that launches the Triton kernel
# --------------------------------------------------------------------------- #
def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right :  float16 tensors of shape [B, N, N, H] (hidden_dim)
    returns out of same shape containing left @ rightᵀ per (b,d) pair.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    out_flat = torch.empty_like(left_flat)

    # strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak = left_flat.stride()
    stride_bh, stride_bn, stride_bk = right_flat.stride()
    stride_ch, stride_cm, stride_cn = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back to [B,N,N,H]
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# --------------------------------------------------------------------------- #
# Main entry point required by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton‑accelerated
    batched matrix multiplication for the N³ reduction.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32 or float16 on CUDA)
        - mask         : [B, N, N]        (bool / float, 1 means keep)
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------- unpack -------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------- LayerNorm (dim) ---------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # use FP16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # ---------------- Linear projections ------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"]).to(torch.float16)    # [B,N,N,hidden]
    right = F.linear(x_fp16, weights["right_proj.weight"]).to(torch.float16)

    # ---------------- optional mask -----------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x_fp16.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------------- Gates (sigmoid) ---------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------- Core N³ reduction (Triton) ---------------------------
    # out shape: [B, N, N, hidden_dim]  (still FP16)
    out = triton_batched_gemm(left, right)

    # ---------------- post‑processing ---------------------------------------
    out = out.to(torch.float32)                     # back to FP32 for layer‑norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate   # gate is still FP16 → cast
    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"])
    return out.to(torch.float32)
