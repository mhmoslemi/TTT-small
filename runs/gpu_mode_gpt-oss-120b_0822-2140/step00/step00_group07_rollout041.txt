"""
TriMul (outgoing) forward pass with a fused Triton matmul.

The expensive part of the module is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
operation (einsum "... i k d, ... j k d -> ... i j d").
For every hidden dimension *d* this is a standard matrix multiplication
between   L_d (N×N) and   R_dᵀ (N×N).  By reshaping we turn the whole
batch × hidden_dim into one big batched‑matmul problem
    (B·H, N, N) × (B·H, N, N) → (B·H, N, N)
and compute it with a custom Triton kernel that tiles the
M‑N dimensions and iterates over K.  All other cheap operations
(layer‑norms, linear projections, gates) are kept in PyTorch.

The kernel works in float32 (accumulates in fp32) and is launched with
two‑dimensional grid indices.  It runs efficiently on Hopper (sm_90)
and satisfies the required API:

    custom_kernel(data) → Tensor[B, N, N, dim]
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: (B*H, N, N) @ (B*H, N, N)  →  (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix sizes
    stride_am, stride_ak,        # A strides: (batch_h, M, K)
    stride_bk, stride_bn,        # B strides: (batch_h, K, N)
    stride_cm, stride_cn,        # C strides: (batch_h, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (necessary for the last partial tile)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile
    a_ptrs = A + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                      # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = C + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]
        - mask         : torch.Tensor[B, N, N]  (binary 0/1)
        - weights      : dict of pre‑trained tensors
        - config       : dict with "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor[B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = torch.float32

    dim       = config["dim"]
    hidden    = config["hidden_dim"]
    batch, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # shape: (B, N, N, hidden)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1)                     # (B, N, N, 1)
    left  = left * mask_exp
    right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating vectors
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton‑accelerated einsum (the N³ part)
    # --------------------------------------------------------------
    #   left : (B, N, N, H)
    #   right: (B, N, N, H)  → we need rightᵀ over the two N axes
    #   Reshape to (B*H, N, N) for a batched matmul.
    #
    #   left   -> (B, H, N, N) -> (B*H, N, N)
    #   rightᵀ -> (B, H, N, N) -> (B*H, N, N)
    #
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch * hidden, N, N)
    right_T   = right.permute(0, 3, 2, 1).contiguous().view(batch * hidden, N, N)

    out_flat = torch.empty_like(left_flat, device=device)

    # strides for the contiguous tensors
    stride_am, stride_ak = left_flat.stride()[1], left_flat.stride()[2]
    stride_bk, stride_bn = right_T.stride()[1], right_T.stride()[2]
    stride_cm, stride_cn = out_flat.stride()[1], out_flat.stride()[2]

    M = N
    K = N
    # tile sizes – tuned for Hopper (larger shared‑memory, 128‑byte lines)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))

    _matmul_kernel[grid](
        left_flat,                     # A
        right_T,                       # B
        out_flat,                      # C
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                   # good default for fp32 on Hopper
    )

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(batch, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    # ensure fp32 output as required
    return out.to(dtype)
