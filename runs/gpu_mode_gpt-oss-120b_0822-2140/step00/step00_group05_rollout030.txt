"""
TriMul “outgoing” implementation for AlphaFold‑3.

The heavy part is the pairwise contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be rewritten as a batched matrix‑multiply
    out = left @ rightᵀ   (per hidden dimension d)

A custom Triton kernel performs this GEMM for every
(batch * hidden_dim) slice, keeping the computation on‑GPU
and avoiding a costly Python‑level triple loop.
All other operations (LayerNorm, linear projections, gates)
are carried out with standard PyTorch primitives.
The final tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ where A,B are (B*H, N, N) matrices.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                     # *float32   A matrix
    b_ptr,                     # *float32   B matrix (will be read as Bᵀ)
    c_ptr,                     # *float32   output matrix
    N,                         # i32        size of the square matrices
    stride_batch,              # i32        stride between batch*H slices
    stride_row,                # i32        stride between rows (i)
    stride_col,                # i32        stride between columns (k)
    BLOCK_M: tl.constexpr,     # compile‑time tile size in M dimension
    BLOCK_N: tl.constexpr,     # compile‑time tile size in N dimension
    BLOCK_K: tl.constexpr,     # compile‑time tile size in K dimension
):
    pid_batch = tl.program_id(0)          # which (batch * hidden) slice
    pid_m = tl.program_id(1)              # tile row index
    pid_n = tl.program_id(2)              # tile column index

    # --------------------------------------------------------------
    # Compute global offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current batch*hidden slice
    # ------------------------------------------------------------------
    a_batch_ptr = a_ptr + pid_batch * stride_batch
    b_batch_ptr = b_ptr + pid_batch * stride_batch
    c_batch_ptr = c_ptr + pid_batch * stride_batch

    # ------------------------------------------------------------------
    # Accumulator for the output tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # global K offset for this iteration
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_row) + (offs_k[None, :] * stride_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ[k, j]  shape (BLOCK_K, BLOCK_N)
        #   we need B[j, k] => read with swapped strides
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_row) + (offs_n[None, :] * stride_col)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate the dot product over K
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the computed tile
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_row) + (offs_n[None, :] * stride_col)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a fused Triton GEMM.

    Args:
        data: Tuple containing
            - input: Tensor[bs, N, N, dim]   (float32/float16)
            - mask : Tensor[bs, N, N] (optional, may be all‑ones)
            - weights: dict of model parameters (LayerNorm, Linear weights/bias)
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # Helper: LayerNorm without autograd (using functional)
    # --------------------------------------------------------------
    def layer_norm(x, weight, bias):
        return F.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask is broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Triton GEMM: out = left @ rightᵀ  (per hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                         # B, N, N, hidden_dim

    # reshape to (B*H, N, N) where rows == i, cols == k for left
    # and rows == j, cols == k for right (same layout, we will read as transpose)
    left_f = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_f = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer
    out_f = torch.empty_like(left_f)

    # strides of the flattened view (int64)
    stride_batch = left_f.stride(0)   # N * N
    stride_row = left_f.stride(1)     # N
    stride_col = left_f.stride(2)     # 1

    # Tile sizes – chosen to fit H200 SM resources
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_gemm_kernel[grid](
        a_ptr=left_f,
        b_ptr=right_f,
        c_ptr=out_f,
        N=N,
        stride_batch=stride_batch,
        stride_row=stride_row,
        stride_col=stride_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_f.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output as required
    out = out.to(torch.float32)
    return out
