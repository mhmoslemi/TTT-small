"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The heavy part of the forward pass is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
L_d @ R_dᵀ for every hidden dimension *d*.  This kernel launches a
triton matmul over the (i,j) axes while iterating over *k* in blocks.
All other operations (layer‑norm, linear projections, gates, masking)
are performed with PyTorch for simplicity and precision.

The kernel works for any batch size B∈{1,2}, sequence length N up to
1024, and hidden dimension H (called *hidden_dim*).  The final output is
float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,    # [B, N, N, H]  (i,k)
    right_ptr,   # [B, N, N, H]  (j,k)
    out_ptr,     # [B, N, N, H]  (i,j)

    B, N, H,

    # strides for left  (batch, i, k, d)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    # strides for right (batch, j, k, d)
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    # strides for out   (batch, i, j, d)
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # tile over i
    BLOCK_N: tl.constexpr,   # tile over j
    BLOCK_K: tl.constexpr,   # tile over k
):
    pid = tl.program_id(0)               # combines batch * hidden
    pid_b = pid // H                     # batch index
    pid_d = pid % H                      # hidden index

    # block indices for the output matrix (i,j)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # -----------------------------------------------------------------
    # Coordinate offsets
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over k
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left[b,i,k,d]  (M × K) --------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_b
            + pid_d * stride_l_d
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- load right[b,j,k,d]  (K × N)  (note transposed view) -----
        # we load it as (K,N) so that tl.dot(left, right) works directly.
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_b
            + pid_d * stride_r_d
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- accumulate -------------------------------------------------
        acc += tl.dot(left, right)

    # -----------------------------------------------------------------
    # Store the result out[b,i,j,d]
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + pid_d * stride_o_d
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Host entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple containing
            - input Tensor                shape [B, N, N, dim]
            - mask  Tensor                shape [B, N, N]  (may be all‑ones)
            - weights dict (torch.Tensor) with the keys used in the reference model
            - config  dict with keys
                * dim          (int)  – input channel dimension
                * hidden_dim   (int)  – hidden dimension
                * nomask (bool)   – if True the mask can be ignored

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(left.dtype)            # [B,N,N,1]
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Triton matmul: out = Σ_k left[...,k,:] * right[...,k,:]
    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (H200 has strong fp16 support)
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    out_f16 = torch.empty_like(left_f16)   # same shape [B,N,N,hidden]

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # one program per (batch, hidden) pair
        triton.cdiv(N, BLOCK_M),        # i‑tiles
        triton.cdiv(N, BLOCK_N),        # j‑tiles
    )

    _trimul_outgoing_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B,
        N,
        hidden,
        # strides for left
        left_f16.stride(0),
        left_f16.stride(1),
        left_f16.stride(2),
        left_f16.stride(3),
        # strides for right
        right_f16.stride(0),
        right_f16.stride(1),
        right_f16.stride(2),
        right_f16.stride(3),
        # strides for out
        out_f16.stride(0),
        out_f16.stride(1),
        out_f16.stride(2),
        out_f16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    out = out_f16.to(torch.float32)

    # -----------------------------------------------------------------
    # Output normalization, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
