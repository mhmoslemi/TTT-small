"""
TriMul “outgoing” implementation with a custom Triton kernel.

The expensive part of TriMul is the 3‑dimensional einsum
    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
which can be expressed as a batched matrix‑multiply for each hidden
dimension d:
    out[..., d] = left[..., d] @ right[..., d].T

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(because they are cheap) and fuse the large N×N×N multiplication
into a single Triton kernel that computes many independent matrix
multiplications in parallel:
    (B * hidden_dim) matrices of size (N, N)  ×  (N, N)ᵀ → (N, N)

The kernel works on float16 inputs, accumulates in float32 and writes
float16 results, which are later cast back to float32 for the final
output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    A,               # ptr to left  matrix batch   (num_mats, M, K)
    B,               # ptr to right matrix batch   (num_mats, N, K)   (note: K is the shared dim)
    C,               # ptr to output matrix batch   (num_mats, M, N)
    M, N, K,         # matrix dimensions
    stride_aba, stride_am, stride_ak,   # strides for A   (batch, row, col)
    stride_bba, stride_bk, stride_bn,   # strides for B   (batch, col, row) – note B is already transposed
    stride_cba, stride_cm, stride_cn,   # strides for C   (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_mat = tl.program_id(0)          # which (batch * hidden) matrix
    pid_m   = tl.program_id(1)          # block row index
    pid_n   = tl.program_id(2)          # block column index

    # --------------------------------------------------------------
    # Offsets within the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks – out‑of‑bounds rows / cols are ignored
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the start of the current block (including batch offset)
    a_ptr = A + pid_mat * stride_aba \
            + offs_m[:, None] * stride_am \
            + offs_k[None, :] * stride_ak

    b_ptr = B + pid_mat * stride_bba \
            + offs_k[:, None] * stride_bk \
            + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main loop over the K dimension
    # --------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)               # FP32 accumulation

        # Advance pointers to the next K‑tile
        a_ptr += BLOCK_K * stride_ak
        b_ptr += BLOCK_K * stride_bk

    # Cast back to FP16 for storage
    c = acc.to(tl.float16)

    # Store result
    c_ptr = C + pid_mat * stride_cba \
            + offs_m[:, None] * stride_cm \
            + offs_n[None, :] * stride_cn
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input  – Tensor[B, N, N, dim]  (float32, on CUDA)
        mask   – Tensor[B, N, N] (bool/float)  (on CUDA)
        weights – dict with all linear / norm parameters
        config  – {"dim": int, "hidden_dim": int}
    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched matmul kernel:
    #   we need shape (B * hidden_dim, N, N) for each operand.
    #   Permute to bring hidden_dim next to batch so that each (b, h)
    #   pair becomes a separate matrix.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Right operand must be transposed for the dot‑product definition.
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N) with swapped axes

    # Cast to FP16 for the kernel (GPU tensor cores)
    A = left_perm.view(B * hidden_dim, N, N).to(torch.float16)
    B = right_T.view(B * hidden_dim, N, N).to(torch.float16)
    C = torch.empty_like(A)                               # (B*H, N, N), FP16

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_mats = B * hidden_dim
    grid = (
        num_mats,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (in elements) for each tensor
    stride_aba, stride_am, stride_ak = A.stride()
    stride_bba, stride_bk, stride_bn = B.stride()
    stride_cba, stride_cm, stride_cn = C.stride()

    batch_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        stride_aba, stride_am, stride_ak,
        stride_bba, stride_bk, stride_bn,
        stride_cba, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape kernel output back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # Output layer norm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.unsqueeze(-1)          # broadcast gating over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out
