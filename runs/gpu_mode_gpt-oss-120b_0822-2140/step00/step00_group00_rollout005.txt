"""
TriMul "outgoing" kernel (AlphaFold3).

The heavy part of the forward pass is

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the projected (and gated) tensors.
For every hidden channel `h` we have a matrix‑matrix multiplication
(L_i×K) · (K×J) → (I×J).  The kernel below computes a batched
matmul over the combined dimension (batch * hidden) using Triton.
All other operations (layer‑norms, linear projections, gates, final
linear) stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matmul (M=N=seq_len, K=seq_len)
# --------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    left_ptr,               # const *float32
    right_ptr,              # const *float32
    out_ptr,                # *float32
    B, H, N,                # sizes
    stride_b_left, stride_h_left, stride_i_left, stride_k_left,
    stride_b_right, stride_h_right, stride_k_right, stride_j_right,
    stride_b_out, stride_h_out, stride_i_out, stride_j_out,
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in reduction dimension
    hidden_dim: tl.constexpr
):
    """Compute out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,k,j]"""

    pid = tl.program_id(0)          # combined (b, h) index
    pid_i = tl.program_id(1)        # block row
    pid_j = tl.program_id(2)        # block col

    # ----- decode (b, h) from linear pid -----------------------
    b = pid // hidden_dim
    h = pid %  hidden_dim

    # ----- tile offsets ----------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Masks for out‑of‑bounds (needed for non‑multiple tile sizes)
    i_mask = i_start + tl.arange(0, BLOCK_M) < N
    j_mask = j_start + tl.arange(0, BLOCK_N) < N

    # Pointers to the beginning of the tile for this (b,h)
    left_tile_ptr = left_ptr \
        + b * stride_b_left + h * stride_h_left \
        + i_start * stride_i_left
    right_tile_ptr = right_ptr \
        + b * stride_b_right + h * stride_h_right \
        + j_start * stride_j_right  # note: right is already transposed (k, j)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over k ------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)

        # Load A block: [M, K]
        a_ptr = left_tile_ptr + k_start * stride_k_left
        a = tl.load(
            a_ptr + tl.arange(0, BLOCK_M)[:, None] * stride_i_left
                 + tl.arange(0, cur_k)[None, :] * stride_k_left,
            mask=i_mask[:, None] & (tl.arange(0, cur_k)[None, :] < cur_k),
            other=0.0
        )

        # Load B block: [K, N]
        b_ptr = right_tile_ptr + k_start * stride_k_right
        b = tl.load(
            b_ptr + tl.arange(0, cur_k)[:, None] * stride_k_right
                 + tl.arange(0, BLOCK_N)[None, :] * stride_j_right,
            mask=(tl.arange(0, cur_k)[:, None] < cur_k) & j_mask[None, :],
            other=0.0
        )

        # dot product
        acc += tl.dot(a, b)

    # ----- write back ------------------------------------------
    out_tile_ptr = out_ptr \
        + b * stride_b_out + h * stride_h_out \
        + i_start * stride_i_out + j_start * stride_j_out

    tl.store(
        out_tile_ptr + tl.arange(0, BLOCK_M)[:, None] * stride_i_out
                     + tl.arange(0, BLOCK_N)[None, :] * stride_j_out,
        acc,
        mask=i_mask[:, None] & j_mask[None, :]
    )


# --------------------------------------------------------------
# Host‑side entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input, mask, weight_dict, config)
            input  – Tensor [B, N, N, dim]   (float32, on CUDA)
            mask   – Tensor [B, N, N]        (float32 or bool, on CUDA)
            weight_dict – mapping of all linear / layer‑norm parameters
            config – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]          # H
    B, N, _, _ = inp.shape

    # ------------------------------------------------------------------
    # 1) Input layer‑norm  (LN over last dim)
    # ------------------------------------------------------------------
    inp = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(inp, w["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(inp, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask (if any).  mask shape = [B,N,N]; broadcast to hidden dim.
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(inp, w["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(inp, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(inp, w["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the Triton matmul:
    #    left  : (B, N, N, H)   -> (B, H, N, N)
    #    right : (B, N, N, H)   -> transpose k/j -> (B, H, N, N)
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B,H,N,N)
    # right_T has dimensions (k, j) -> we need (k, j) order
    right_t = right.permute(0, 3, 2, 1).contiguous()        # (B,H,N,N)

    # Flatten (B, H) into a single dimension for the kernel launch
    left_flat = left_t.view(B * hidden, N, N)
    right_flat = right_t.view(B * hidden, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)                 # (B*H,N,N)

    # ------------------------------------------------------------------
    # 6) Triton batched matmul launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # program_id 0  -> (b, h) combined
        triton.cdiv(N, BLOCK_M),        # program_id 1  -> rows
        triton.cdiv(N, BLOCK_N),        # program_id 2  -> cols
    )

    # strides for the flattened view (after view, strides are the same as before view)
    s_b_l, s_h_l, s_i_l, s_k_l = left_t.stride()      # (B, H, N, N)
    s_b_r, s_h_r, s_k_r, s_j_r = right_t.stride()

    s_b_o, s_h_o, s_i_o, s_j_o = out_flat.stride()

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B, hidden, N,
        s_b_l, s_h_l, s_i_l, s_k_l,
        s_b_r, s_h_r, s_k_r, s_j_r,
        s_b_o, s_h_o, s_i_o, s_j_o,
        BLOCK_M, BLOCK_N, BLOCK_K,
        hidden_dim=hidden,
        num_warps=4,
    )

    # Reshape back to (B,N,N,H)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output layer‑norm, gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, w["to_out.weight"])   # [B,N,N,dim]

    return out
