"""
TriMul “outgoing” implementation with a custom Triton kernel for the
k‑reduction (the `einsum` part).  

The kernel performs, for each batch and hidden channel *d*  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply `A @ B.T` for every
hidden dimension.  The surrounding LayerNorm / linear / gating operations
stay in PyTorch for simplicity.  Only the O(N³) reduction is off‑loaded to
Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton GEMM‑style kernel used for the core TriMul reduction
# --------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    N, H,                                   # problem size
    stride_li, stride_lk, stride_ld,        # left strides (i,k,d)
    stride_rj, stride_rk, stride_rd,        # right strides (j,k,d)
    stride_oi, stride_oj, stride_od,        # out strides (i,j,d)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """ Compute out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] """

    pid_i = tl.program_id(0)          # i‑block index
    pid_j = tl.program_id(1)          # j‑block index

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_off = i_base + tl.arange(0, BLOCK_I)
    j_off = j_base + tl.arange(0, BLOCK_J)

    mask_i = i_off < N
    mask_j = j_off < N

    # loop over the hidden dimension in chunks of BLOCK_D
    for d0 in range(0, H, BLOCK_D):
        # accumulator for this d‑chunk, shape (BLOCK_I, BLOCK_J, BLOCK_D)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # k‑tile loop (the reduction dimension)
        for k0 in range(0, N, BLOCK_K):
            k_off = k0 + tl.arange(0, BLOCK_K)
            mask_k = k_off < N

            # pointer offsets for the d‑chunk
            d_range = d0 + tl.arange(0, BLOCK_D)

            # load left[i,k,d]  -> (BLOCK_I, BLOCK_K, BLOCK_D)
            left = tl.load(
                left_ptr
                + i_off[:, None, None] * stride_li
                + k_off[None, :, None] * stride_lk
                + d_range[None, None, :] * stride_ld,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & (d_range[None, None, :] < H),
                other=0.0,
            )

            # load right[j,k,d] -> (BLOCK_J, BLOCK_K, BLOCK_D)
            right = tl.load(
                right_ptr
                + j_off[:, None, None] * stride_rj
                + k_off[None, :, None] * stride_rk
                + d_range[None, None, :] * stride_rd,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & (d_range[None, None, :] < H),
                other=0.0,
            )

            # inner‑product for each d in the block
            for db in range(0, BLOCK_D):
                # guard against the last incomplete d‑block
                if (d0 + db) >= H:
                    break
                a = left[:, :, db]            # (BLOCK_I, BLOCK_K)
                b = right[:, :, db]           # (BLOCK_J, BLOCK_K)
                # Σ_k a[i,k] * b[j,k] = (a @ b.T)[i,j]
                acc[:, :, db] += tl.dot(a, b, trans_b=True)

        # store the completed d‑chunk
        for db in range(0, BLOCK_D):
            d = d0 + db
            if d >= H:
                break
            out_ptrs = (
                out_ptr
                + i_off[:, None] * stride_oi
                + j_off[None, :] * stride_oj
                + d * stride_od
            )
            tl.store(
                out_ptrs,
                acc[:, :, db],
                mask=mask_i[:, None] & mask_j[None, :],
            )

def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Args:
        data: Tuple (input, mask, weights, config)
            input  – Tensor[B, N, N, C]   (float32)
            mask   – Tensor[B, N, N]      (bool/float)
            weights – dict of model parameters
            config  – dict with keys:
                dim, hidden_dim, nomask (bool)

    Returns:
        Tensor[B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]

    # -----------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim = C)
    # -----------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * ln_w + ln_b                       # [B,N,N,C]

    # -----------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # -> [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden dim)
    # -----------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.unsqueeze(-1).to(x.dtype)         # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------
    # 5) Core TriMul reduction via Triton
    # -----------------------------------------------------------
    out = torch.empty((B, N, N, hidden), device=device, dtype=torch.float32)

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 8

        # make sure the slices are contiguous (required for correct strides)
        left_b = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b = out[b].contiguous()

        # strides (in elements, not bytes)
        stride_li, stride_lk, stride_ld = left_b.stride()
        stride_rj, stride_rk, stride_rd = right_b.stride()
        stride_oi, stride_oj, stride_od = out_b.stride()

        grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))
        _trimul_gemm_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            hidden,
            stride_li,
            stride_lk,
            stride_ld,
            stride_rj,
            stride_rk,
            stride_rd,
            stride_oi,
            stride_oj,
            stride_od,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # -----------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```
