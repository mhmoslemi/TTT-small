"""
TriMul “outgoing” implementation with a custom Triton kernel.

The heavy 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a tiled reduction kernel (i‑j‑d tiles, k‑loop).
All other operations (LayerNorms, Linear projections, gating, mask handling)
are performed with PyTorch which already uses highly‑optimized cuBLAS/aten kernels.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimension up to 384.  It respects optional masking and produces a
float‑32 output identical to the reference PyTorch implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:   out = Σ_k left[...,k,:] * right[...,k,:]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1️⃣  Compute the (b,i_tile,j_tile) we are responsible for
    # --------------------------------------------------------------
    total_i = tl.cdiv(N, BLOCK_I)
    total_j = tl.cdiv(N, BLOCK_J)

    b = pid // (total_i * total_j)
    pid_remaining = pid % (total_i * total_j)
    i_tile = pid_remaining // total_j
    j_tile = pid_remaining % total_j

    i0 = i_tile * BLOCK_I
    j0 = j_tile * BLOCK_J

    # --------------------------------------------------------------
    # 2️⃣  Offsets for the tile
    # --------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3️⃣  Loop over k in blocks
    # --------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k0)
        offs_k = k0 + tl.arange(0, BLOCK_K)

        # pointers with full 4‑D addressing
        left_tile_ptr = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        right_tile_ptr = (
            right_ptr
            + b * stride_r_b
            + offs_j[:, None, None] * stride_r_j
            + offs_k[None, :, None] * stride_r_k
            + offs_h[None, None, :] * stride_r_h
        )

        # load (i,k,h) and (j,k,h) – out‑of‑bounds entries are zero‑filled
        left_tile = tl.load(
            left_tile_ptr,
            mask=mask_i[:, None, None] & (offs_k[None, :, None] < N) & (offs_h[None, None, :] < H),
            other=0.0,
        )
        right_tile = tl.load(
            right_tile_ptr,
            mask=mask_j[:, None, None] & (offs_k[None, :, None] < N) & (offs_h[None, None, :] < H),
            other=0.0,
        )

        # reduction Σ_k   left[i,k,h] * right[j,k,h]
        #   we iterate over the k‑dimension explicitly (BLOCK_K is tiny)
        for kk in range(0, BLOCK_K):
            # (i,h) * (j,h)  →  (i,j,h) via broadcasting
            l = left_tile[:, kk, :]            # (i, h)
            r = right_tile[:, kk, :]           # (j, h)
            acc += l[:, None, :] * r[None, :, :]   # (i, j, h)

    # --------------------------------------------------------------
    # 4️⃣  Store the tile
    # --------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_h[None, None, :] * stride_o_h
    )
    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & (offs_h[None, None, :] < H),
    )

# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor[B, N, N, dim]   (float32)
        mask    – torch.Tensor[B, N, N] (optional, may be all‑ones)
        weights – dict with all linear / layer‑norm parameters
        config  – dictionary with keys:
                  * dim          – input/​output channel dimension
                  * hidden_dim   – internal hidden dimension
                  * nomask (bool) – True ⇒ ignore the mask

    Returns
    -------
    torch.Tensor[B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣  Apply optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣  Gating (sigmoid activations)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Triton kernel:  Σ_k left[...,k,:] * right[...,k,:]
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=input_tensor.device)

    # strides in element units (not bytes)
    stride_l = left.stride()   # (B, N, N, H)
    stride_r = right.stride()
    stride_o = out.stride()

    # Tile sizes – chosen to fit H200 SM 90 shared memory comfortably
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32           # hidden_dim is a multiple of 32 in all test configs
    BLOCK_K = 64           # reasonable K‑tile for N up to 1024

    grid = (
        triton.cdiv(B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J), 1),
    )
    trimul_kernel[grid](
        left, right, out,
        B, N, hidden,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6️⃣  post‑processing: LayerNorm (hidden), gating, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
