"""
TriMul (outgoing) – fast Triton implementation.

The heavy part of the TriMul forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for each hidden channel *d* :

    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

We:
1. Perform the LayerNorms and linear projections / gates in PyTorch (they are
   cheap compared to the O(N³) work).
2. Reshape the tensors so that each (batch, hidden) pair becomes a separate
   matrix of shape (N, N).  The multiplication is then a batched GEMM of shape
   (B·hidden, N, N) × (B·hidden, N, N)ᵀ.
3. Launch a custom Triton kernel (`batched_matmul_kernel`) that computes the
   GEMM for many blocks in parallel.  The kernel works in FP16 internally and
   writes the result back in FP16; the host code converts to FP32 afterwards.
4. Apply the final LayerNorm, out‑gate and output linear projection.

The implementation supports the full range of test configurations (different
batch sizes, sequence lengths, dim / hidden_dim, with or without a mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched matrix‑multiply kernel
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,                # pointers
    M, N, K,                # matrix sizes (M×K) @ (K×N) = (M×N)
    stride_Am, stride_Ak,   # A stride over M and K
    stride_Bk, stride_Bn,   # B stride over K and N
    stride_Cm, stride_Cn,   # C stride over M and N
    batch_stride_A, batch_stride_B, batch_stride_C,   # stride to next batch
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes many independent GEMMs:
        C[b] = A[b] @ B[b]^T          (A: M×K, B: N×K, B is stored as K×N)
    The batch dimension is folded into the program id.
    """
    pid = tl.program_id(0)

    # ---------- batch / block decomposition ----------
    blocks_m = tl.cdiv(M, BLOCK_M)
    blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (blocks_m * blocks_n)
    pid_in_batch = pid % (blocks_m * blocks_n)
    pid_m = pid_in_batch // blocks_n
    pid_n = pid_in_batch % blocks_n

    # ---------- coordinate offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for this batch
    a_ptr = A + batch_id * batch_stride_A
    b_ptr = B + batch_id * batch_stride_B
    c_ptr = C + batch_id * batch_stride_C

    # ---------- load A block ----------
    a_offs = offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    a = tl.load(a_ptr + a_offs,
                mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
                other=0.0)

    # ---------- accumulator ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K dimension ----------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load B block (note that B is stored as K×N, we need the transpose)
        b_offs = (tl.arange(0, BLOCK_K) + k * BLOCK_K)[:, None] * stride_Bk + \
                 offs_n[None, :] * stride_Bn
        b = tl.load(b_ptr + b_offs,
                    mask=(tl.arange(0, BLOCK_K) + k * BLOCK_K < K)[:, None] &
                         (offs_n < N)[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

        # Advance A & B pointers for the next K‑tile
        if k + 1 < tl.cdiv(K, BLOCK_K):
            a_offs += BLOCK_K * stride_Ak
            a = tl.load(a_ptr + a_offs,
                        mask=(offs_m < M)[:, None] &
                             (tl.arange(0, BLOCK_K) + (k + 1) * BLOCK_K < K)[None, :],
                        other=0.0)

    # ---------- write back ----------
    c = acc.to(tl.float16)
    c_offs = offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptr + c_offs, c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# ----------------------------------------------------------------------
# The public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]  (may be all‑ones)
        - weights      : dict of torch.Tensor with the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (no bias in the reference is ok)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(dtype), bias=None
    )  # [B,N,N,hidden]
    right = F.linear(
        x, weights["right_proj.weight"].to(dtype), bias=None
    )

    # ------------------------------------------------------------------
    # 3. Mask (if supplied) – the config ``nomask`` tells us whether the mask
    #    is all‑ones; we still respect the tensor that is passed.
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones_like(input_tensor[..., 0], dtype=dtype, device=device)
    mask = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype), bias=None)
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype), bias=None)
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype), bias=None)
    )

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5. Batched GEMM:   out = left @ right⁻ᵀ   for each hidden channel
    #    Reshape to (B*hidden, N, N) and run the Triton kernel.
    # ------------------------------------------------------------------
    # (B,N,N,hidden) -> (B, hidden, N, N) -> (B*hidden, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Cast to FP16 for the kernel (precision loss is acceptable; we convert back)
    A = left_.to(torch.float16)
    B_ = right_.to(torch.float16)
    C = torch.empty_like(A, dtype=torch.float16)

    # stride information (in elements, not bytes)
    stride_Am = A.stride(1)           # stride over M (row)
    stride_Ak = A.stride(2)           # stride over K (col)
    stride_Bk = B_.stride(1)          # B is stored as K×N
    stride_Bn = B_.stride(2)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    batch_stride_A = A.stride(0)
    batch_stride_B = B_.stride(0)
    batch_stride_C = C.stride(0)

    # Block sizes – chosen to fit H200 L2/shared memory well.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_blocks_m = triton.cdiv(N, BLOCK_M)
    total_blocks_n = triton.cdiv(N, BLOCK_N)
    total_programs = (B * hidden_dim) * total_blocks_m * total_blocks_n

    grid = (total_programs,)

    batched_matmul_kernel[grid](
        A, B_, C,
        N, N, N,
        stride_Am, stride_Ak,
        stride_Bk, stride_Bn,
        stride_Cm, stride_Cn,
        batch_stride_A, batch_stride_B, batch_stride_C,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 and reshape to original layout
    out = C.to(dtype).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype), bias=None)

    return out
