"""
TriMul “outgoing” implementation.

The heavy N³ reduction is performed as a batched GEMM:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a per‑hidden‑dim matrix multiplication
    (left_d) @ (right_d)^T .
We use a small Triton kernel for the element‑wise mask‑gate fusion
(and for the final gated output) – the O(N³) work is done with
torch.bmm on the reshaped tensors (highly tuned cuBLAS on H200).
All tensors are kept in the user‑provided dtype (fp16‑friendly)
and the final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernels (element‑wise product + optional mask)
# --------------------------------------------------------------

@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, m_ptr, out_ptr,                     # pointers
    N_ELEM: tl.constexpr, BLOCK: tl.constexpr         # meta‑params
):
    """out = a * b * m   (all tensors are 1‑D flat views)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    m = tl.load(m_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * b * m, mask=mask)


@triton.jit
def elemwise_gate_kernel(
    a_ptr, g_ptr, out_ptr, N_ELEM: tl.constexpr, BLOCK: tl.constexpr
):
    """out = a * g   (used for the final out‑gate multiplication)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    g = tl.load(g_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * g, mask=mask)


# --------------------------------------------------------------
# Helper – launch a 1‑D Triton kernel
# --------------------------------------------------------------
def launch_elemwise_mul(a: torch.Tensor, b: torch.Tensor,
                        m: torch.Tensor, out: torch.Tensor):
    """Launch elemwise_mul_kernel on flat contiguous tensors."""
    assert a.is_contiguous() and b.is_contiguous() and m.is_contiguous() and out.is_contiguous()
    n_elem = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n_elem, BLOCK),)
    elemwise_mul_kernel[grid](
        a.data_ptr(),
        b.data_ptr(),
        m.data_ptr(),
        out.data_ptr(),
        N_ELEM=n_elem,
        BLOCK=BLOCK,
    )

def launch_elemwise_gate(a: torch.Tensor, g: torch.Tensor, out: torch.Tensor):
    """Launch elemwise_gate_kernel on flat contiguous tensors."""
    assert a.is_contiguous() and g.is_contiguous() and out.is_contiguous()
    n_elem = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n_elem, BLOCK),)
    elemwise_gate_kernel[grid](
        a.data_ptr(),
        g.data_ptr(),
        out.data_ptr(),
        N_ELEM=n_elem,
        BLOCK=BLOCK,
    )


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton
    element‑wise kernel and a batched GEMM for the N³ reduction.

    Args:
        data: (input, mask, weights, config)
            input   – Tensor[bs, N, N, dim] (float16/float32 on CUDA)
            mask    – Tensor[bs, N, N]   (float16/float32 on CUDA)
            weights – dict with all linear / norm parameters
            config  – dict containing "dim" and "hidden_dim"

    Returns:
        Tensor[bs, N, N, dim] (float32, on the same device as input)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # LayerNorm (no bias for the first norm, bias is provided separately)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # torch.layer_norm expects normalized_shape = dim
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gates (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Apply mask (if supplied) and the per‑side gates with a Triton kernel
    # ------------------------------------------------------------------
    # mask shape [bs, N, N] → [bs, N, N, 1] (broadcastable)
    mask_unsq = mask.unsqueeze(-1).to(dtype)

    # make everything contiguous for the kernel
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_unsq = mask_unsq.contiguous()

    left_masked = torch.empty_like(left)
    right_masked = torch.empty_like(right)

    # left_masked = left * left_gate * mask
    launch_elemwise_mul(left, left_gate, mask_unsq, left_masked)
    # right_masked = right * right_gate * mask
    launch_elemwise_mul(right, right_gate, mask_unsq, right_masked)

    # ------------------------------------------------------------------
    # Batched GEMM over the hidden dimension
    #   out_d = left_d @ right_d^T   for each d in [hidden_dim]
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # reshape to (B*hidden_dim, N, N)
    left_flat  = left_masked.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_masked.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # matrix multiply (right needs to be transposed on the second axis)
    # use fp16 internally if possible for speed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim) and multiply by out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # apply final out‑gate with Triton
    out_gate = out_gate.contiguous()
    out = out.contiguous()
    gated_out = torch.empty_like(out)
    launch_elemwise_gate(out, out_gate, gated_out)

    # ------------------------------------------------------------------
    # Final linear projection back to original dim
    # ------------------------------------------------------------------
    final = F.linear(gated_out, weights["to_out.weight"])

    # ensure float32 output as required
    return final.to(torch.float32)
