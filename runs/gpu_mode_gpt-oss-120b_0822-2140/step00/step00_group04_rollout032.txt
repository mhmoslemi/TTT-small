def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid0 // H
    h = pid0 % H

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_start in range(0, N, BLOCK_K):
        k_len = tl.minimum(BLOCK_K, N - k_start)
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # load left: [i,k]
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:,None]*stride_left_i + cur_k[None,:]*stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

        # load right transposed: need [k,j]
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + cur_k[:,None]*stride_right_k + offs_j[None,:]*stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)

        # dot
        acc += tl.dot(left, right)  # left (I,K) right (K,J) -> (I,J)

    # store output
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_i[:,None]*stride_out_i + offs_j[None,:]*stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:,None] & mask_j[None,:])

"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden dimension *d*:
    out[b,:, :,d] = left[b,:, :,d] @ right[b,:, :,d].transpose(1,2)

We keep the cheap operations (LayerNorm, Linear, gating, final
normalisation) in PyTorch and off‑load the N³ reduction to a single
Triton kernel.  The kernel works on a block of (i,j) pairs and a
single hidden dimension *d* (combined with the batch index into the
first program‑id dimension).  It tiles the reduction dimension *k*
to reuse registers and uses `tl.dot` for the inner product.

The kernel launches with a 3‑D grid:
    (batch * hidden_dim, ceil(N/BI), ceil(N/BJ))

and produces a tensor of shape `[B, N, N, hidden_dim]` in FP32.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left   [B, I, K, H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right  [B, J, K, H]   (stored as (b, j, k, h))
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out    [B, I, J, H]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tile sizes (compile‑time constants)
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where the right tensor is accessed with its second and third axes
    swapped (i.e. we use right[b,k,j,h] so that a dot product can be used).
    """
    pid0 = tl.program_id(0)          # (b * H) dimension
    pid_i = tl.program_id(1)          # i‑block
    pid_j = tl.program_id(2)          # j‑block

    # decode batch and hidden dimension from the first id
    b = pid0 // H
    h = pid0 % H

    # block start indices
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # linear offsets inside a block
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the (i,j) tile
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    #   reduction over k
    # ------------------------------------------------------------------- #
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k                # absolute k indices
        mask_k = cur_k < N

        # left tile  : shape (BLOCK_I, cur_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + cur_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # (I, K)

        # right tile (transposed) : shape (cur_K, BLOCK_J)
        # we need right[b, k, j, h] → swap j and k axes.
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + cur_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # (K, J)

        # dot product over the reduction axis K
        acc += tl.dot(left, right)   # (I, J)

    # ------------------------------------------------------------------- #
    #   store result
    # ------------------------------------------------------------------- #
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )

# --------------------------------------------------------------------------- #
# Python entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel for the N³ reduction ----------
    out = torch.empty((bs, N, N, hidden_dim), dtype=dtype, device=device)

    # block sizes – trade‑off between register pressure and occupancy
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    # strides (contiguous layout: [B, N, N, H])
    stride_left = left.stride()    # (N*N*H, N*H, H, 1)
    stride_right = right.stride()  # (N*N*H, N*H, H, 1)
    stride_out = out.stride()      # (N*N*H, N*H, H, 1)

    grid = (
        bs * hidden_dim,                     # combine batch and hidden dim
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )
    _trimul_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        hidden_dim,
        # left strides
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        # right strides
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        # out strides
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Final normalisation, gating and projection ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
