"""
TriMul “outgoing” kernel (AlphaFold‑3)

The heavy part of the TriMul forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is mathematically a batched matrix multiplication per hidden channel
`d` ( left @ rightᵀ ).  The kernel below computes this operation completely
in Triton.  All surrounding LayerNorms, linear projections and gating are
performed with native PyTorch for simplicity; they are cheap compared to the
O(N³·H) contraction.

The kernel works for arbitrary batch size `B`, sequence length `N`,
and hidden dimension `H` (the “hidden_dim” argument).  It is written for
SM‑90 GPUs (H200/H100) and uses float16 arithmetic with a float32
accumulator, returning a float32 result.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel performing: out = left @ rightᵀ   (per batch & per hidden)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,                # *const float16
    right_ptr,               # *const float16
    out_ptr,                 # *float32
    B, N, H,                 # int32
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,   # rows of output tile
    BLOCK_N: tl.constexpr,   # cols of output tile
    BLOCK_K: tl.constexpr,   # reduction tile
):
    """Triton implementation of the TriMul einsum.

    Grid layout (1‑D):
        pid = b * H * (n_m * n_n) + h * (n_m * n_n) + m * n_n + n
        where
            n_m = ceil(N / BLOCK_M)
            n_n = ceil(N / BLOCK_N)
    """
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # decode 1‑D program id → (b, h, tile_m, tile_n)
    # ---------------------------------------------------------------
    n_m = tl.cdiv(N, BLOCK_M)
    n_n = tl.cdiv(N, BLOCK_N)
    pid_bh = pid // (n_m * n_n)                # index over (B * H)
    pid_rem = pid % (n_m * n_n)
    pid_m = pid_rem // n_n
    pid_n = pid_rem % n_n

    b = pid_bh // H
    h = pid_bh % H

    # ---------------------------------------------------------------
    # offsets within the current tile
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # main reduction over k
    # ---------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            offs_m[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k + \
            h * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: we need rightᵀ → shape (BLOCK_K, BLOCK_N)
        #   right[j,k] -> after transposition we load as (k, j)
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            offs_k[:, None] * stride_r_k + \
            offs_n[None, :] * stride_r_j + \
            h * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(left, right)

    # ---------------------------------------------------------------
    # write back
    # ---------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        offs_m[:, None] * stride_o_i + \
        offs_n[None, :] * stride_o_j + \
        h * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a Triton kernel for the
    N³ contraction.

    Args:
        data: (input, mask, weights, config)
            input  – torch.Tensor, shape [B, N, N, dim] (float32)
            mask   – torch.Tensor, shape [B, N, N] (bool/float) or None
            weights – dict of torch.Tensors containing all model parameters
            config – dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # helper: LayerNorm without creating nn.Module
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        # x shape [..., C]; weight/bias shape [C]
        return F.layer_norm(x, (x.shape[-1],), weight, bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 1) Input norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask has shape [B,N,N] – broadcast over hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for einsum (left @ rightᵀ per hidden channel)
    # ------------------------------------------------------------------
    # cast to half to reduce bandwidth; accumulator stays fp32
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides (in element units, not bytes) – needed by the kernel
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tiling parameters – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_programs = B * hidden_dim * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (total_programs,)

    _trimul_einsum_kernel[grid](
        left_h,
        right_h,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: to_out_norm, gating and final linear
    # ------------------------------------------------------------------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate        # element‑wise broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
