"""
TriMul “outgoing” implementation with a fused Triton kernel for the
core N×N × N×N (over the K dimension) multiplication.

Algorithm
---------
1. Layer‑norm the input (dim = C)                     → x
2. Linear projections (no bias)                       → left, right  (shape B,N,N,H)
3. Optional mask (broadcasted on the hidden dim)
4. Gating: sigmoid(linear(x))                         → left_gate, right_gate, out_gate
5. Apply gates to left/right
6. Triton kernel:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (equivalent to a batch of H matrix‑multiplications per batch element)
7. Layer‑norm over the hidden dimension, multiply by out_gate
8. Final linear projection back to dim C
All computations are FP32, the kernel works on arbitrary B,N,H values
that satisfy the test constraints.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ------------------------------------------------------------------
# Triton kernel: batched matmul for each (batch, hidden) pair.
# ------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                      # tensor sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute for a single (batch, hidden) sub‑matrix:
    out[i, j] = Σ_k left[i, k] * right[j, k]   (right is NOT transposed in memory)
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Tile indices
    # --------------------------------------------------------------
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    # (batch * hidden) index
    bh = pid // (num_tiles_i * num_tiles_j)
    tile_i = (pid // num_tiles_j) % num_tiles_i
    tile_j = pid % num_tiles_j

    # split bh into batch b and hidden h
    b = bh // H
    h = bh % H

    # Offsets within the tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the beginning of the (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K loop
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_off   = k_start + offs_k
        mask_k  = k_off < N

        # left[i, k]
        a_ptr = left_base + offs_i[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[j, k]   (note the j index first – no transpose needed)
        b_ptr = right_base + offs_j[:, None] * stride_right_i + k_off[None, :] * stride_right_k
        b = tl.load(b_ptr, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        acc += tl.dot(a, b)            # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ------------------------------------------------------------------
# Host‑side entry point
# ------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    TriMul outgoing forward pass with a fused Triton kernel for the
    N³ multiplication.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, C]  (C = dim)
        mask    – Tensor[B, N, N] (may be all‑ones)
        weights – dict of all linear / layer‑norm parameters
        config  – {'dim': C, 'hidden_dim': H}

    Returns
    -------
    Tensor[B, N, N, C]   (float32, same device as input)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = torch.float32

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm over C
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections (bias=False)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if supplied)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)            # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton kernel: N³ multiplication
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                     # shape = [B, N, N, H]
    left  = left.contiguous()
    right = right.contiguous()

    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides (in elements, not bytes)
    s_left  = left.stride()
    s_right = right.stride()
    s_out   = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # total number of program instances
    tiles_i = triton.cdiv(N, BLOCK_M)
    tiles_j = triton.cdiv(N, BLOCK_N)
    total   = B * hidden_dim * tiles_i * tiles_j
    grid    = (total,)

    _trimul_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate                               # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])      # -> [B, N, N, dim]

    return out
