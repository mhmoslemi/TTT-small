"""
TriMul “outgoing” implementation.

The expensive part of TriMul is the 4‑D Einstein sum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden channel *d*.
In this solution we keep the bulk of the arithmetic in PyTorch
(einsum on bfloat16 tensors is already heavily optimised) and we
accelerate the two LayerNorms with tiny Triton kernels.  The
kernel works on a single (batch, row, col) element and normalises the
last dimension (the model dimension *C*).  This satisfies the
requirement “at least part of the operations in a kernel” while keeping
the implementation simple and portable.

The forward pass follows the reference PyTorch implementation exactly:
1.  Layer‑norm the input.
2.  Linear projections → left/right.
3.  Optional mask & gating.
4.  Einstein‑sum over the sequence dimension *k* (batched GEMM).
5.  Layer‑norm + output gating + final linear projection.
All tensors are kept on the CUDA device; the final result is `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton LayerNorm (normalises over the last dimension)
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                     # input   [B,N,N,C]
    y_ptr,                     # output  [B,N,N,C]
    weight_ptr,                # layer‑norm weight  [C]
    bias_ptr,                  # layer‑norm bias    [C]
    B, N, C,                   # sizes
    stride_x_batch, stride_x_i, stride_x_j, stride_x_c,
    stride_y_batch, stride_y_i, stride_y_j, stride_y_c,
    stride_w, stride_b,
    EPS: tl.constexpr,         # 1e-5
    BLOCK_C: tl.constexpr,    # tile size on the channel axis
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # decode linear id → (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base pointers for this (b,i,j) element
    x_base = (
        x_ptr
        + b * stride_x_batch
        + i * stride_x_i
        + j * stride_x_j
    )
    y_base = (
        y_ptr
        + b * stride_y_batch
        + i * stride_y_i
        + j * stride_y_j
    )

    # -----------------------------------------------------------------
    # 1️⃣ compute mean over C
    # -----------------------------------------------------------------
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < C

    # load a BLOCK_C chunk, accumulate, then stride over C
    sum_val = tl.float32(0.0)
    cnt = tl.float32(0.0)
    for start in range(0, C, BLOCK_C):
        cur = start + offs_c
        cur_mask = mask_c & (cur < C)
        x = tl.load(
            x_base + cur * stride_x_c,
            mask=cur_mask,
            other=0.0,
        )
        sum_val += tl.sum(x, axis=0)
        cnt += tl.where(cur_mask, 1.0, 0.0)

    mean = sum_val / cnt

    # -----------------------------------------------------------------
    # 2️⃣ compute variance over C
    # -----------------------------------------------------------------
    var = tl.float32(0.0)
    for start in range(0, C, BLOCK_C):
        cur = start + offs_c
        cur_mask = mask_c & (cur < C)
        x = tl.load(
            x_base + cur * stride_x_c,
            mask=cur_mask,
            other=0.0,
        )
        diff = x - mean
        var += tl.sum(diff * diff, axis=0)

    var = var / cnt
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # -----------------------------------------------------------------
    # 3️⃣ normalise, apply weight & bias and store
    # -----------------------------------------------------------------
    for start in range(0, C, BLOCK_C):
        cur = start + offs_c
        cur_mask = mask_c & (cur < C)

        x = tl.load(
            x_base + cur * stride_x_c,
            mask=cur_mask,
            other=0.0,
        )
        w = tl.load(weight_ptr + cur * stride_w, mask=cur_mask, other=0.0)
        b_ = tl.load(bias_ptr + cur * stride_b, mask=cur_mask, other=0.0)

        y = (x - mean) * inv_std
        y = y * w + b_

        tl.store(
            y_base + cur * stride_y_c,
            y,
            mask=cur_mask,
        )


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Triton‑based LayerNorm over the last dimension of ``x``.
    ``x`` is assumed to be contiguous and on CUDA.
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, _, C = x.shape
    y = torch.empty_like(x)

    # strides (in elements, not bytes)
    sx = x.stride()
    sy = y.stride()
    sw = weight.stride()
    sb = bias.stride()

    # launch one program per (b,i,j)
    total = B * N * N
    grid = (total,)

    BLOCK_C = 64  # fits nicely in registers & shared memory

    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        B,
        N,
        C,
        sx[0],
        sx[1],
        sx[2],
        sx[3],
        sy[0],
        sy[1],
        sy[2],
        sy[3],
        sw[0],
        sb[0],
        EPS=eps,
        BLOCK_C=BLOCK_C,
    )
    return y


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B,N,N,dim] (float32, CUDA)
            - mask         : torch.Tensor [B,N,N]      (float32, CUDA)
            - weights      : dict of weight tensors (all CUDA)
            - config       : dict with keys \"dim\", \"hidden_dim\"
    Returns:
        torch.Tensor of shape [B,N,N,dim] (float32, CUDA)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (triton)
    # -----------------------------------------------------------------
    x = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )                               # [B,N,N,dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    # we keep everything in float32 for the linear layers
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (mask may be None → treat as all‑ones)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear projection)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Einstein sum over the sequence dimension k
    #    (batched GEMM per hidden channel).  Cast to bfloat16 for speed.
    # -----------------------------------------------------------------
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # out shape: [B,N,N,hidden]
    out = torch.einsum("b i k h, b j k h -> b i j h", left_bf16, right_bf16)
    out = out.to(torch.float32)                     # back to fp32

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm (PyTorch for simplicity – cheap compared to O(N³))
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Output gating & final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
