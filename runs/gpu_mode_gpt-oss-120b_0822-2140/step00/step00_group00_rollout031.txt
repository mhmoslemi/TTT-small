"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part is the N‑by‑N matrix product that is summed over the
sequence dimension k:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension *d* this is a standard GEMM:
    out_d = left_d @ right_dᵀ   (shape N×N)

We fuse this GEMM across the hidden dimension (H) and batch (B) with a
single Triton kernel.  The surrounding LayerNorms, linear projections,
gates and the final projection are cheap enough to stay in PyTorch.

Input:
    x  – [B, N, N, C]   (float32)
    mask – [B, N, N]    (float32, may be all‑ones)
    weights – dict of raw weight tensors
    config  – {"dim":C, "hidden_dim":H}

Output:
    y – [B, N, N, C] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton GEMM kernel:  C = A @ Bᵀ   where A,B : (batch_h, N, N)
# batch_h = B * hidden_dim
# --------------------------------------------------------------------
@triton.jit
def triton_gemm_out_kernel(
    a_ptr,                # *float16
    b_ptr,                # *float16 (will be read as transposed)
    c_ptr,                # *float16
    batch_h, M, N, K,     # int32
    stride_ab, stride_am, stride_ak,    # strides for A
    stride_bb, stride_bk, stride_bn,    # strides for B (note: B is stored normally, we treat it as transposed)
    stride_cb, stride_cm, stride_cn,    # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)                     # batch element
    pid_m = tl.program_id(0)                     # block row
    pid_n = tl.program_id(1)                     # block col

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # pointers for the current batch
    a_batch_ptr = a_ptr + pid_z * stride_ab
    b_batch_ptr = b_ptr + pid_z * stride_bb
    c_batch_ptr = c_ptr + pid_z * stride_cb

    # pointers to the start of the block
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    # B is read as transposed: original B shape (N, N) with strides (stride_bb1, stride_bb2)
    # we need Bᵀ → row stride = original col stride (stride_bk), col stride = original row stride (stride_bn)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=m_mask[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & n_mask[None, :], other=0.0)
        acc += tl.dot(a, b)                 # (M,K)·(K,N) → (M,N)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=m_mask[:, None] & n_mask[None, :])

# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with a fused GEMM kernel.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              input_tensor : torch.Tensor [B, N, N, C]  (float32)
              mask_tensor  : torch.Tensor [B, N, N]   (float32, may be None)
              weights_dict : mapping from weight name to torch.Tensor
              config_dict  : {"dim": C, "hidden_dim": H}
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ----------------------------------------------------------------
    # LayerNorm on the input (dim = C)
    # ----------------------------------------------------------------
    eps = 1e-5
    x = input_tensor
    x = F.layer_norm(
        x,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------
    # shape -> [B, N, N, H]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------
    # Apply mask (if any)
    # ----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)                # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ----------------------------------------------------------------
    # Gates (sigmoid)
    # ----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # Prepare tensors for batched GEMM:
    #   left_perm/right_perm : [B, H, N, N]
    #   flatten batch*H -> batch_h
    # ----------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left_perm.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(-1, N, N)  # [B*H, N, N]

    batch_h = left_flat.shape[0]

    # cast to float16 for the GEMM kernel
    A = left_flat.to(torch.float16)
    B_mat = right_flat.to(torch.float16)          # will be used as transposed inside kernel
    C_out = torch.empty_like(A, dtype=torch.float16)

    # strides (in elements)
    stride_ab, stride_am, stride_ak = A.stride()
    stride_bb, stride_bc, stride_bd = B_mat.stride()   # (batch, N, N)

    # For transposed read we need:
    #   row stride   = original column stride  = stride_bd (==1 for contiguous)
    #   col stride   = original row stride     = stride_bc (==N)
    stride_bk = stride_bd
    stride_bn = stride_bc

    stride_cb, stride_cm, stride_cn = C_out.stride()

    # ----------------------------------------------------------------
    # Launch Triton GEMM kernel
    # ----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
        batch_h,                        # batch*hidden
    )
    triton_gemm_out_kernel[grid](
        A,
        B_mat,
        C_out,
        batch_h,
        N,
        N,
        N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    # ----------------------------------------------------------------
    out_hidden = C_out.to(dtype).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ----------------------------------------------------------------
    # Final LayerNorm (hidden_dim) + gating + linear projection back to C
    # ----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_hidden = out_hidden * out_gate   # broadcasting over last dim H

    # Linear projection to original dim C
    out = F.linear(out_hidden, weights["to_out.weight"])   # [B,N,N,C]

    return out
