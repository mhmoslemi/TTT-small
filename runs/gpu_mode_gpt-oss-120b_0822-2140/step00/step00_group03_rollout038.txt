"""
TriMul “outgoing” implementation.

The forward pass consists of:
1. LayerNorm over the channel dimension (dim).
2. Two linear projections → left/right (dim → hidden_dim).
3. Optional pair‑wise mask.
4. Gated element‑wise multiplication (sigmoid gates).
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of hidden_dim independent matmuls:
       left[b,:,:,d] @ right[b,:,:,d].T
   This N×N × N×N multiplication (∝ N³) is performed with a
   custom Triton kernel for speed.
6. LayerNorm over hidden_dim, gating with out_gate and final linear
   back to dim.

All heavy arithmetic (step 5) runs on the GPU via Triton; the
remaining lightweight ops stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: batched matmul  (left @ rightᵀ) per hidden channel
# ------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides of the 4‑D tensors [B,N,N,H]
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute out = left @ rightᵀ for every (b, h) pair.

    left, right, out shapes: [B, N, N, H]
    The kernel is launched with a 1‑D grid where each program instance
    processes a tile (i_tile, j_tile) for a given (b, h) pair.
    """

    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_i * num_j

    # decode batch (b, h) and tile indices
    batch_id = pid // tiles_per_batch                # 0 … B*H-1
    tile_id  = pid % tiles_per_batch
    pid_i = tile_id // num_j                        # i‑tile index
    pid_j = tile_id %  num_j                        # j‑tile index

    # map batch_id → original (b, h)
    b = batch_id // H
    h = batch_id %  H

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for the tail of the N dimension
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for the current (b,h) slice
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)   → indices (i, k)
        a_ptrs = (left_base
                  + offs_i[:, None] * stride_li
                  + offs_k[None, :] * stride_lk)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right block: (K, N) → indices (k, j)
        b_ptrs = (right_base
                  + offs_j[None, :] * stride_rj
                  + offs_k[:, None] * stride_rk)
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0)

        # dot : (M,K)·(K,N) → (M,N)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = (out_base
                + offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Helper that launches the kernel
# ------------------------------------------------------------
def _trimul_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul:  out = left @ rightᵀ   per hidden channel.
    Input shapes:  [B, N, N, H] (contiguous, float32)
    Returns: same shape, float32.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    device = left.device

    # make sure tensors are contiguous for simple stride extraction
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – tuned for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # number of program instances
    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_tiles = B * H * num_i * num_j

    grid = (total_tiles,)

    _trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb, stride_li=stride_li, stride_lk=stride_lk, stride_lh=stride_lh,
        stride_rb=stride_rb, stride_rj=stride_rj, stride_rk=stride_rk, stride_rh=stride_rh,
        stride_ob=stride_ob, stride_oi=stride_oi, stride_oj=stride_oj, stride_oh=stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ------------------------------------------------------------
# Main entry point required by the benchmark harness
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- 1) LayerNorm over `dim` ----------
    # x_norm = (x - μ) / sqrt(σ² + eps) * w + b
    mu = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mu) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mu) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ---------- 2) Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- 3) Optional pair‑wise mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ---------- 4) Gated element‑wise multiplication ----------
    left_gate  = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5) Core contraction via Triton ----------
    # left/right shapes: [B, N, N, hidden_dim]
    out = _trimul_matmul_triton(left, right)          # [B, N, N, hidden_dim]

    # ---------- 6) LayerNorm over hidden_dim ----------
    mu2 = out.mean(dim=-1, keepdim=True)
    var2 = ((out - mu2) ** 2).mean(dim=-1, keepdim=True)
    out = (out - mu2) * torch.rsqrt(var2 + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ---------- 7) Final gating and linear ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
