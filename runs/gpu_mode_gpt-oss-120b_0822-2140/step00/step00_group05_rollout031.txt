    right = F.linear(x, weights["right_proj.weight"])

"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of the TriMul operation is the einsum

    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]

where left/right are projected, gated and masked tensors of shape
[ B, N, N, H ] (H = hidden_dim).  This can be rewritten as a batched
matrix‑multiply per hidden dimension:

    left_perm  : [B, H, N, N]   (i, k)
    right_perm : [B, H, N, N]   (k, j)   ← transpose the last two axes

    out_perm   = left_perm @ right_perm   (per batch‑hidden slice)

Thus we need a fast GEMM for many (B·H) independent N×N × N×N
multiplications.  The kernel below implements a standard tiled
matrix‑multiply in Triton, works for any batch dimension, and is
invoked from the Python entry‑point `custom_kernel`.  All other
operations (LayerNorm, Linear, gating, mask handling, final
normalisation and projection) are performed with regular PyTorch
kernels for simplicity and numerical stability.

The final output is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B  (A: M×K, B: K×N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_aa, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cc, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col
    pid_b = tl.program_id(2)               # batch / hidden slice

    # ------------------------------------------------------------------
    # Offsets within the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # batch offsets (one‑dimensional batch stride)
    a_batch = pid_b * stride_aa
    b_batch = pid_b * stride_bb
    c_batch = pid_b * stride_cc

    # Pointer arithmetic
    a_ptrs = a_ptr + a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    c_ptrs = c_ptr + c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    # Masks for out‑of‑bounds loads/stores
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Accumulator in fp32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        cur_k = offs_k + k * BLOCK_K
        cur_mask_k = cur_k < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & cur_mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=cur_mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot does a matmul of the small tiles
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result (back to fp16 / bf16)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator implemented with a
    Triton‑accelerated batched matrix‑multiply for the N³ term.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim]
        * mask_tensor  : torch.Tensor [B, N, N]   (may be all‑ones)
        * weights_dict : dict of torch.Tensor with the same keys as in
                         the reference PyTorch implementation
        * config_dict  : dict containing at least ``dim`` and
                         ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask_f = torch.ones_like(x[..., 0], dtype=x.dtype, device=device)
    else:
        mask_f = mask.to(x.dtype)
    mask_f = mask_f.unsqueeze(-1)                   # [B, N, N, 1]

    left  = left  * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the N³ einsum -> batched GEMM
    #    Use fp16 for speed; the kernel accumulates in fp32.
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden dim before the sequence axes
    # left_perm : [B, H, N, N]  (i,k)
    # right_perm: [B, H, N, N]  (k,j) – note the transpose on the last two axes
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # Merge batch & hidden dim → (B*H, N, N) for a single GEMM batch
    left_flat  = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)   # fp16

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H,                            # batch * hidden
    )

    # Compute strides (torch stride is in elements, not bytes)
    stride_aa = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cc = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                         # M, N, K
        stride_aa, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cc, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # B, N, N, H

    return out.to(torch.float32)
