"""
TriMul "outgoing" implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (dim‑last).
2. Linear projections → left, right (hidden_dim) and three sigmoidal gates.
3. Optional element‑wise mask applied to left/right.
4. Multiply left/right with their respective gates.
5. Fused Triton kernel computes, for every hidden dimension h,
   out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h],
   i.e. a batch of H matrix‑multiplications of size N×N.
6. Layer‑norm over the hidden dimension, apply the output gate and final
   linear projection back to dim.

The kernel is a classic tiled GEMM where the K‑dimension (sequence length)
is the reduction axis.  The batch‑hidden index is encoded in the 3rd
program‑id dimension, so a 3‑D grid launch covers (B·H) independent
matrix‑multiplies.  The result stays in float32 (the usual precision on H200).

Only the cubic‑complex part (step 5) is in Triton; the surrounding Linear /
LayerNorm / sigmoid logic stays in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N, H,                                 # sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for a given hidden slice `h` and batch `b`:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    The kernel is launched with a 3‑D grid:
        (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    """
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_z = tl.program_id(2)          # combined batch*hidden index

    batch_idx = pid_z // H            # b
    hid_idx   = pid_z % H             # h

    # offsets within the N×N matrix
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (i, k) -----
        left_ptrs = (left_ptr +
                     batch_idx * stride_l_b +
                     hid_idx   * stride_l_h +
                     offs_i[:, None] * stride_l_i +
                     offs_k[None, :] * stride_l_k)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # ----- load right tile (j, k) transposed to (k, j) -----
        # we need right[batch, j, k, h] → treat k as row, j as col.
        right_ptrs = (right_ptr +
                      batch_idx * stride_r_b +
                      hid_idx   * stride_r_h +
                      offs_k[:, None] * stride_r_k +
                      offs_j[None, :] * stride_r_i)   # i‑stride == j‑stride in layout
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # accumulate dot product over k‑tile
        acc += tl.dot(left_tile, right_tile)

    # store the result
    out_ptrs = (out_ptr +
                batch_idx * stride_o_b +
                hid_idx   * stride_o_h +
                offs_i[:, None] * stride_o_i +
                offs_j[None, :] * stride_o_k)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])



def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input, mask, weights, config)
            input  – torch.Tensor            [B, N, N, dim]
            mask   – torch.Tensor            [B, N, N]  (may be all‑ones)
            weights – dict of torch.Tensors   (model parameters)
            config – dict with keys:
                     dim, hidden_dim, nomask (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (dim‑last)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → left / right
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and apply them
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden_dim
    out = torch.empty((B, N, N, H), device=device, dtype=left.dtype)

    # strides in element units (not bytes)
    l_str = left.stride()   # (B, i, k, h)
    r_str = right.stride()
    o_str = out.stride()

    # Tiling parameters (tuned for Hopper – can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch × hidden
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for Hopper
    )

    # ------------------------------------------------------------------
    # 6. Output processing (LayerNorm → gate → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape → [B, N, N, dim]

    # ensure float32 output (model works in fp32)
    return out.to(torch.float32)
