"""
TriMul “outgoing” forward pass with a custom Triton batched‑GEMM kernel.

The heavy O(N³) part:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is realised as a batched matrix multiplication:
    left   → (B, H, N, N)   (permute to B,H,N,N)
    right  → (B, H, N, N)   (permute + transpose on the last two axes)

    out    =   left @ rightᵀ   performed by a Triton kernel that
    computes many independent GEMMs (batch = B·H) in one launch.

All other operations (LayerNorm, Linear, Sigmoid, mask) stay in PyTorch.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix sizes (all equal to seq_len)
    stride_az, stride_am, stride_ak,   # a: batch, row, col strides
    stride_bz, stride_bn, stride_bk,   # b: batch, row, col strides (B is already transposed)
    stride_cz, stride_cm, stride_cn,   # c: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   where A,B,C each have a leading batch dimension.
    A : (batch, M, K)
    B : (batch, N, K)   (note the K‑dim is the inner dimension, so B is effectively transposed)
    C : (batch, M, N)
    All tensors are stored in row‑major order.
    """
    pid_m = tl.program_id(0)                     # block row
    pid_n = tl.program_id(1)                     # block col
    pid_z = tl.program_id(2)                     # batch index

    # ------------------------------------------------------------------
    # Offsets for the block of A and B that this program works on
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch
    a_batch_ptr = a_ptr + pid_z * stride_az
    b_batch_ptr = b_ptr + pid_z * stride_bz
    c_batch_ptr = c_ptr + pid_z * stride_cz

    # base pointers for the block
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:        [B, N, N]      (bool / float)
            - weights: dict of all learnable parameters (float32)
            - config:   dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input channel dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])
    right = F.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM with Triton (left @ rightᵀ) over the sequence dimension
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                         # B = batch size, N = sequence length

    # reshape to (B*hidden, N, N) and transpose right for the matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_t = right_perm.transpose(-2, -1).contiguous()     # (B, H, N, N) with K as inner dim

    total_batches = B * hidden
    A = left_perm.view(total_batches, N, N)                  # (batch, M, K)
    Bmat = right_t.view(total_batches, N, N)                 # (batch, N, K)  <-- K same as N
    C = torch.empty_like(A)                                 # (batch, M, N)

    # launch parameters (tuned for Hopper; can be adjusted)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        total_batches,
    )

    _batched_gemm_kernel[grid](
        A,
        Bmat,
        C,
        N,
        N,
        N,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        Bmat.stride(0),
        Bmat.stride(1),
        Bmat.stride(2),
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: (B, N, N, hidden)
    out = C.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, w["to_out.weight"])

    return out.to(torch.float32)
