"""
TriMul (outgoing) kernel – fast Triton implementation of the core N³ pairwise
product.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) with the provided LN parameters.
2. Linear projections (left/right) + gates (left/right/out) – all done in PyTorch
   (these are cheap compared with the N³ term).
3. Mask the projected tensors if a mask is supplied.
4. Reshape the tensors so that the hidden dimension H (= hidden_dim) is fused
   with the batch dimension:
        left  : (B, N, N, H) → (B·H, N, N)   (i‑k matrix)
        right : (B, N, N, H) → (B·H, N, N)   (j‑k matrix)
   Transpose the right matrix to obtain the required “k‑j” ordering.
5. Compute for every fused batch the matrix product
        C = left @ rightᵀ                     # shape (B·H, N, N)
   using a custom Triton kernel (blocked GEMM).  The kernel works on
   three‑dimensional tensors (batch, M, N) and is fully fused – no temporary
   materialisation of the K‑loop.
6. Reshape the result back to (B, N, N, H).
7. Apply the output LayerNorm, multiply with the out‑gate, and a final linear
   projection back to the original channel dimension C.
8. Return a float‑32 tensor of shape (B, N, N, C).

The most expensive part – the N³ contraction – is executed on the GPU by the
Triton kernel, giving a large speed‑up over the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton GEMM kernel (batched, block‑wise)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,          # (batch, M, K)  – left matrix
    B_ptr,          # (batch, K, N)  – right matrix (already transposed)
    C_ptr,          # (batch, M, N)  – output
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------ block indices (batch, row, col) --------------------
    batch_grid = tl.cdiv(BATCH, 1)                # just BATCH
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_m * num_n)
    rem = pid % (num_m * num_n)
    pid_m = rem // num_n
    pid_n = rem % num_n

    # ------------ compute offsets ------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = A_ptr + batch_idx * stride_Ab \
                   + offs_m[:, None] * stride_Am \
                   + offs_k[None, :] * stride_Ak

    b_ptrs = B_ptr + batch_idx * stride_Bb \
                   + offs_k[:, None] * stride_Bk \
                   + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------ main K‑loop ----------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # a: (BM, BK), b: (BK, BN)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------ write back -----------------------------------------
    c_ptrs = C_ptr + batch_idx * stride_Cb \
                    + offs_m[:, None] * stride_Cm \
                    + offs_n[None, :] * stride_Cn

    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input:  torch.Tensor, shape [B, N, N, C] (float32)
            - mask:   torch.Tensor, shape [B, N, N] (bool / float)
            - weights: dict with all linear / LN parameters (float32 tensors)
            - config:  dict containing at least ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]          # H

    device = inp.device
    B, N, _, _ = inp.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (LN over channel dim)
    # ------------------------------------------------------------------
    inp = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shapes: (B, N, N, C) @ (C, H).t  -> (B, N, N, H)
    left = F.linear(inp, weights["left_proj.weight"])
    right = F.linear(inp, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if present (mask may be all‑ones when nomask=True)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(inp)        # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(inp, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(inp, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(inp, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Reshape for the N³ contraction
    #   left  : (B, N, N, H) -> (B*H, N, N)   (i,k)
    #   right : (B, N, N, H) -> (B*H, N, N)   (j,k) then transpose to (k,j)
    # ------------------------------------------------------------------
    # bring hidden dim forward and fuse with batch
    left_f = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_f = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where dim2= k, dim3=j

    left_mat = left_f.view(B * hidden, N, N)         # (B*H, M=N, K=N)
    right_mat = right_f.view(B * hidden, N, N)       # (B*H, K=N, N)

    # ------------------------------------------------------------------
    # Allocate output buffer (float16 inside the kernel, converted later)
    # ------------------------------------------------------------------
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    batch = B * hidden
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_blocks = batch * grid_m * grid_n
    grid = (total_blocks,)

    # strides
    a_stride_b, a_stride_m, a_stride_k = left_mat.stride()
    b_stride_b, b_stride_k, b_stride_n = right_mat.stride()
    c_stride_b, c_stride_m, c_stride_n = out_mat.stride()

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,
        N,
        N,
        a_stride_b,
        a_stride_m,
        a_stride_k,
        b_stride_b,
        b_stride_k,
        b_stride_n,
        c_stride_b,
        c_stride_m,
        c_stride_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Convert back to float32 and reshape to original layout
    # ------------------------------------------------------------------
    out_fp32 = out_mat.to(torch.float32)
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
