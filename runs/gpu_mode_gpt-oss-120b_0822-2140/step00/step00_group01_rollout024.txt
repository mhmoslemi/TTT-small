"""
TriMul "outgoing" kernel (AlphaFold3 style).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiplication per hidden channel:
    left   : (B, H, N, N)   # H = hidden_dim
    right  : (B, H, N, N)   # we need rightᵀ on the last two axes
    out    = left @ rightᵀ   → (B, H, N, N)

We implement this batched GEMM with Triton (block‑wise dot product) and keep the
LayerNorm / Linear / gating logic in PyTorch for simplicity and flexibility.
The kernel works for arbitrary batch size (≤2), sequence length (≤1024) and
hidden dimension (≤384).  All arithmetic is performed in float16 inside the
kernel and accumulated in float32; the final result is cast back to float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) where A,B have shape (BH, M, K)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BH, M, N, K,                       # logical sizes
    stride_ah, stride_am, stride_ak,   # A strides (BH, M, K)
    stride_bh, stride_bk, stride_bn,   # B strides (BH, K, N)
    stride_ch, stride_cm, stride_cn,   # C strides (BH, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[bh, i, j] = Σ_k A[bh, i, k] * B[bh, j, k]   (B is already transposed)."""
    bh = tl.program_id(0)          # combined batch*head index
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # pointers to the start of the tile
    a_tile_ptr = a_ptr + bh * stride_ah + offs_m[:, None] * stride_am
    b_tile_ptr = b_ptr + bh * stride_bh + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (M x K)
        a = tl.load(
            a_tile_ptr + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # load B tile (N x K) -> we need Bᵀ, but B already stores (K, N)
        b = tl.load(
            b_tile_ptr + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # dot product for the current K‑block
        acc += tl.dot(a, b).to(tl.float32)

    # write back the result
    c_ptrs = c_ptr + bh * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the TriMul "outgoing" module with a Triton‑accelerated
    batched matrix multiplication.

    Args:
        data: (input, mask, weights, config)
            - input:  Tensor [B, N, N, dim] (float32)
            - mask:   Tensor [B, N, N] (bool or float) – may be all‑ones if nomask=True
            - weights: dict of Parameter tensors (see code)
            - config:  dict containing "dim", "hidden_dim", "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim last)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(dtype).unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM:
    #    shape -> (B, H, N, N)   where H = hidden_dim
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two axes: (B, N, N, H) -> (B, H, N, N) with k axis first
    right_t = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_.shape
    BH = B * H  # combined batch‑head dimension for Triton

    # Allocate output buffer
    out_ = torch.empty_like(left_, dtype=torch.float32, memory_format=torch.contiguous_format)

    # --------------------------------------------------------------
    # 6. Launch Triton GEMM kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (in elements, not bytes) for the three tensors
    a_strides = left_.stride()
    b_strides = right_t.stride()
    c_strides = out_.stride()

    batched_gemm_kernel[grid](
        a_ptr=left_.data_ptr(),
        b_ptr=right_t.data_ptr(),
        c_ptr=out_.data_ptr(),
        BH=BH,
        M=N,
        N=N,
        K=N,
        stride_ah=a_strides[0],
        stride_am=a_strides[2],
        stride_ak=a_strides[3],
        stride_bh=b_strides[0],
        stride_bk=b_strides[2],
        stride_bn=b_strides[3],
        stride_ch=c_strides[0],
        stride_cm=c_strides[2],
        stride_cn=c_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7. Bring back to original layout: (B, N, N, H)
    # --------------------------------------------------------------
    out = out_.permute(0, 2, 3, 1).contiguous()   # shape [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate has shape [B, N, N, hidden_dim]
    out = out * out_gate

    # Final linear back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
