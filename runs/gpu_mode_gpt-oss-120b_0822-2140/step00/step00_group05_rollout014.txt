"""
TriMul “outgoing” implementation.

The heavy N³ work is a batched matrix‑multiply over the intermediate
hidden dimension.  We keep the two LayerNorms (input norm & output norm)
in a small Triton kernel so the solution contains a genuine GPU kernel
while letting cuBLAS (torch.bmm) handle the expensive N³ multiplication.

All tensors are expected on the CUDA device and the final result is
float‑32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton LayerNorm (static epsilon, works for any row‑major 2‑D view)
# --------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,               # *float32   input  (rows, dim)
    out_ptr,             # *float32   output (rows, dim)
    weight_ptr,          # *float32   gamma  (dim,)
    bias_ptr,            # *float32   beta   (dim,)
    rows, dim, epsilon,  # i32
    stride_x_row, stride_x_col,
    stride_out_row, stride_out_col,
    stride_w, stride_b,
    BLOCK_D: tl.constexpr,
):
    row = tl.program_id(0)
    if row >= rows:
        return

    # ------------------------------------------------------------------
    #  Load a tile of the row (size = BLOCK_D, tail handled by mask)
    # ------------------------------------------------------------------
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    x = tl.load(
        x_ptr + row * stride_x_row + offs_d * stride_x_col,
        mask=mask_d,
        other=0.0,
    )

    # ------------------------------------------------------------------
    #  Compute mean / variance
    # ------------------------------------------------------------------
    mean = tl.sum(x, axis=0) / dim
    x_center = x - mean
    var = tl.sum(x_center * x_center, axis=0) / dim
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ------------------------------------------------------------------
    #  Apply γ (weight) and β (bias)
    # ------------------------------------------------------------------
    weight = tl.load(weight_ptr + offs_d * stride_w, mask=mask_d, other=0.0)
    bias = tl.load(bias_ptr + offs_d * stride_b, mask=mask_d, other=0.0)

    y = (x_center * inv_std) * weight + bias

    # ------------------------------------------------------------------
    #  Store the normalized row
    # ------------------------------------------------------------------
    tl.store(
        out_ptr + row * stride_out_row + offs_d * stride_out_col,
        y,
        mask=mask_d,
    )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    Layer‑norm over the last dimension using a Triton kernel.
    x : (B, N, N, D)   float32
    weight, bias : (D,) float32
    Returns tensor of same shape as x.
    """
    B, N, N2, D = x.shape
    assert N == N2, "expected square pair‑wise tensor"
    rows = B * N * N
    x_flat = x.reshape(rows, D)
    out_flat = torch.empty_like(x_flat)

    BLOCK_D = 128                     # max D used in the tests (384)
    grid = (rows,)

    _layernorm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        rows,
        D,
        eps,
        stride_x_row=x_flat.stride(0),
        stride_x_col=x_flat.stride(1),
        stride_out_row=out_flat.stride(0),
        stride_out_col=out_flat.stride(1),
        stride_w=weight.stride(0),
        stride_b=bias.stride(0),
        BLOCK_D=BLOCK_D,
    )
    return out_flat.reshape(B, N, N, D)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, S, S, C]   (float32, on CUDA)
            - mask  : torch.Tensor [B, S, S]      (float32/bool, on CUDA)
            - weights: dict with the exact keys used in the reference model
            - config : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, S, S, C] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim = C)
    # ------------------------------------------------------------------
    x = triton_layernorm(
        x,
        weights["norm.weight"].to(torch.float32),
        weights["norm.bias"].to(torch.float32),
    )   # (B, S, S, C)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # weight shape: (hidden, dim)
    left = F.linear(x, weights["left_proj.weight"].to(torch.float32))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float32))

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.unsqueeze(-1)               # (B, S, S, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid) – also linear, no bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float32))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float32))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float32))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core multiply: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # right needs its two sequence axes swapped (k,j)
    right_t = right.permute(0, 2, 1, 3)          # (B, S, S, hidden) where axes are (k, j)

    B, S, _, _ = left.shape

    # reshape to (B*hidden, S, S) for a batch of matrix‑multiplies
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden, S, S)
    right_flat = right_t.permute(0, 3, 1, 2).reshape(B * hidden, S, S)

    # batched matmul via cuBLAS (very fast)
    out_flat = torch.bmm(left_flat, right_flat)          # (B*hidden, S, S)

    # reshape back to (B, S, S, hidden)
    out = (
        out_flat.view(B, hidden, S, S)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, S, S, hidden)

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = triton_layernorm(
        out,
        weights["to_out_norm.weight"].to(torch.float32),
        weights["to_out_norm.bias"].to(torch.float32),
    )

    # ------------------------------------------------------------------
    # Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out.to(torch.float32)
