"""
TriMul “outgoing” implementation optimized with Triton.

Algorithm
---------
1.  Layer‑norm the 4‑D input tensor (B, N, N, D) using the provided
    `norm.{weight,bias}`.
2.  Linear projections → `left`, `right`  (B,N,N,H) where H = hidden_dim.
3.  Apply the optional pair mask (unsqueezed on the channel axis) and the
    three sigmoid‑gates (left_gate, right_gate, out_gate).
4.  Fuse the heavy O(N³) contraction:
        out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
    This is a batched matrix‑multiply per hidden channel:
        out[b,h] = left[b,h] @ right[b,h]ᵀ
    We flatten the (B, H) dimensions and run a custom Triton kernel that
    performs a packed GEMM for each (B·H) matrix pair.
5.  Reshape the result back to (B,N,N,H) and apply a second layer‑norm,
    the output‑gate, and the final linear projection to recover the original
    channel dimension D.

The kernel uses float‑32 arithmetic (the tensors are already in fp32) and
stores the result in fp32.  All non‑triton work (norms, linear layers,
sigmoids, masking) is performed with regular torch ops – the only
performance‑critical part (the N³ contraction) is executed on the GPU by
Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication (A @ Bᵀ) for many small matrices
# ----------------------------------------------------------------------------- #
@triton.jit
def _batched_matmul_kernel(
    a_ptr,                # *float32   A matrix batch pointer
    b_ptr,                # *float32   B matrix batch pointer (already transposed)
    c_ptr,                # *float32   output batch pointer
    M, N, K,              # matrix dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ B   where
        A : (batch, M, K)
        B : (batch, K, N)   (already the transpose of the original right tensor)
        C : (batch, M, N)
    """
    pid_b = tl.program_id(2)          # batch id
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # -------------------------------------------------------------
    # compute block offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask to avoid out‑of‑bounds loads
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers for the first tile of A and B
    a_ptrs = (
        a_ptr
        + pid_b * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_ptrs = (
        b_ptr
        + pid_b * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                     # (BM, BK) @ (BK, BN) → (BM, BN)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # -------------------------------------------------------------
    # Write back the result
    # -------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------------- #
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, D]
        - mask         : torch.Tensor   [B, N, N]   (may be all‑ones)
        - weights      : dict of torch.Tensor
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor  [B, N, N, D]   (float32)
    """
    # -------------------------------------------------
    # unpack inputs
    # -------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (c_z)

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm  (torch implementation – cheap compared to O(N³))
    # --------------------------------------------------------------------- #
    eps = 1e-5
    x = (input_tensor - input_tensor.mean(-1, keepdim=True))
    x = x / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True) + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # --------------------------------------------------------------------- #
    # 2. Linear projections (biasless) → (B,N,N,H)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # shape (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                 # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 4. Gates
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Heavy contraction – batched matmul per hidden channel
    # --------------------------------------------------------------------- #
    # reshape to (B, H, N, N) → flatten batch dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    # transpose right on the last two axes so that we can compute A @ Bᵀ
    right_t = right.transpose(-2, -1).contiguous()   # (B, H, N, N)

    batch = B * H
    left_flat = left.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)

    out_flat = torch.empty_like(left_flat)

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch                      # batch dimension
    )
    _batched_matmul_kernel[grid](
        left_flat,                # a_ptr
        right_flat,               # b_ptr (already transposed)
        out_flat,                 # c_ptr
        N, N, N,                  # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6. Output layer‑norm, out‑gate and final linear
    # --------------------------------------------------------------------- #
    out = (out - out.mean(-1, keepdim=True))
    out = out / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate                               # broadcast on H

    # final linear projection back to D
    out = F.linear(out, weights["to_out.weight"])

    return out
