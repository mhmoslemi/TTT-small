"""
TriMul “outgoing” operator (AlphaFold‑3) – Triton‑accelerated contraction.

The heavy part of the module is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We implement it as a Triton kernel that computes a tile of (i,j,d)
elements for a single batch.  Each thread block produces
BLOCK_M × BLOCK_N × BLOCK_D scalars and loops over the K dimension in
small BLOCK_K chunks, loading the corresponding slices of ``left`` and
``right`` and accumulating the element‑wise products.
All other operations (LayerNorm, linear projections, gates, final
LayerNorm/linear) are kept in PyTorch.

The kernel works for any batch size, sequence length N (up to 1024 in
the tests) and hidden dimension (typically 128).  It returns a
float‑32 tensor of shape [B,N,N,hidden_dim] which is then post‑processed
by the remaining PyTorch layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------------- #
# Triton kernel
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const *float  [N, N, D] (i,k,d)
    right_ptr,         # const *float  [N, N, D] (j,k,d)
    out_ptr,           # *float        [N, N, D] (i,j,d)
    N, D,              # int32, int32
    # strides (batch dimension is handled on the host)
    stride_li, stride_lk, stride_ld,
    stride_rj, stride_rk, stride_rd,
    stride_oi, stride_oj, stride_od,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
):
    """ Compute   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]  for a single batch. """
    # --------------------------------------------------------------------- #
    # thread‑block coordinates
    # --------------------------------------------------------------------- #
    i = tl.program_id(0) * BLOCK_M + tl.arange(0, BLOCK_M)
    j = tl.program_id(1) * BLOCK_N + tl.arange(0, BLOCK_N)
    d = tl.program_id(2) * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < D

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # loop over K dimension in blocks
    # --------------------------------------------------------------------- #
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---- left: shape (M, K, D) ------------------------------------- #
        left_ptrs = left_ptr \
            + i[:, None, None] * stride_li \
            + k[None, :, None] * stride_lk \
            + d[None, None, :] * stride_ld
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0
        )

        # ---- right: shape (N, K, D) ------------------------------------ #
        right_ptrs = right_ptr \
            + j[:, None, None] * stride_rj \
            + k[None, :, None] * stride_rk \
            + d[None, None, :] * stride_rd
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0
        )

        # ---- accumulate the outer product over the K dimension ------------- #
        # left  : (M, D)   right : (N, D)
        # broadcasting gives (M, N, D)
        for kk in range(0, BLOCK_K):
            l = left[:, kk, :]          # (M, D)
            r = right[:, kk, :]         # (N, D)
            acc += l[:, None, :] * r[None, :, :]

    # --------------------------------------------------------------------- #
    # write back
    # --------------------------------------------------------------------- #
    out_ptrs = out_ptr \
        + i[:, None, None] * stride_oi \
        + j[None, :, None] * stride_oj \
        + d[None, None, :] * stride_od
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    )

    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Triton kernel – batch‑wise contraction
    # --------------------------------------------------------------------- #
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # tile sizes – tuned for H200 (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_D = 32          # hidden_dim is 128 ⇒ 4 tiles
    BLOCK_K = 32

    # stride information (batch dimension handled on host)
    stride_li, stride_lk, stride_ld = left.stride()[1:]   # (i,k,d)
    stride_rj, stride_rk, stride_rd = right.stride()[1:] # (j,k,d)
    stride_oi, stride_oj, stride_od = out.stride()[1:]   # (i,j,d)

    for b in range(B):
        # pointers to the b‑th batch slice
        left_ptr  = left[b].data_ptr()
        right_ptr = right[b].data_ptr()
        out_ptr   = out[b].data_ptr()

        grid = (
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            triton.cdiv(hidden_dim, BLOCK_D),
        )
        trimul_outgoing_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_li, stride_lk, stride_ld,
            stride_rj, stride_rk, stride_rd,
            stride_oi, stride_oj, stride_od,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_D=BLOCK_D,
            BLOCK_K=BLOCK_K,
        )

    # --------------------------------------------------------------------- #
    # 6. Post‑processing (LayerNorm, gating, final linear)
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate          # broadcast over (i,j) already matches shape
    out = F.linear(out, weights["to_out.weight"])

    return out
```
