"""
TriMul “outgoing” implementation.
--------------------------------
* LayerNorm + Linear + gating are done in PyTorch (they are cheap compared
  to the N³ part).
* The heavy N³ contraction

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

  is rewritten as a batched matrix multiplication   (left @ rightᵀ)
  for every hidden dimension *d*.  We fuse the batch and hidden
  dimension (M = B·D) and launch a Triton kernel that computes many
  (M, N, N) matmuls with tiling in the M, N and K dimensions.
* Loads are performed in fp16 (the model weights are fp32 → fp16) and
  the accumulation is in fp32 for numeric stability.  The final result
  is cast back to fp32 before the remaining LayerNorm/Linear steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul with the right‑hand side transposed
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    a_ptr,                     # left  [B,N,N,D]  (i,k) ordering
    b_ptr,                     # right [B,N,N,D]  (j,k) ordering (will be read transposed)
    c_ptr,                     # out   [B,N,N,D]
    B, N, D,                   # dimensions
    stride_am, stride_ak, stride_ad,                     # left strides
    stride_bm, stride_bi, stride_bj, stride_bd,          # right strides
    stride_cm, stride_ci, stride_cj, stride_cd,          # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       for a tile (BLOCK_M, BLOCK_N) of the (i,j) matrix and a batch‑hidden
       index m = b*D + d.
    """
    pid_m = tl.program_id(0)          # index over (B*D) dimension
    pid_n = tl.program_id(1)          # index over output column j

    total_m = B * D
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_m
    mask_n = offs_n < N

    # decode (batch, hidden) from the flattened m index
    b_idx = offs_m // D
    d_idx = offs_m % D

    # ------------------------------------------------------------------
    # base pointers for the current (b,d) slice
    # ------------------------------------------------------------------
    a_base = a_ptr + b_idx[:, None] * stride_am + d_idx[:, None] * stride_ad
    b_base = b_ptr + b_idx[:, None] * stride_bm + d_idx[:, None] * stride_bd
    c_base = c_ptr + b_idx[:, None] * stride_cm + d_idx[:, None] * stride_cd

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # tiled reduction over k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,d]  -> (i,k)
        a_ptrs = a_base + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)    # (BLOCK_M, BLOCK_K)

        # right[b,j,k,d] needs the (j,k) order → we read with j as column,
        # k as row (i.e. the matrix is already transposed for the dot)
        b_ptrs = b_base + offs_n[None, :] * stride_bj + offs_k[:, None] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_m[:, None] & mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)    # (BLOCK_K, BLOCK_N)

        # dot: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back – cast to fp16 inside the store to reduce bandwidth,
    # the caller will convert back to fp32.
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: layer‑norm that works with pre‑packed weight & bias tensors
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    """Manual LayerNorm to avoid creating a nn.Module."""
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    inv = torch.rsqrt(var + eps)
    return (x - mean) * inv * weight + bias


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   (float32)
            mask         : [B, N, N]        (float32, 0/1)
            weights      : dict of tensors (already on CUDA)
            config       : dict with "dim" and "hidden_dim"
    Returns:
        out: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim = model dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – cast to fp16 for the kernel
    # ------------------------------------------------------------------
    # shape: (..., hidden_dim)
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any).  mask is float32 0/1, we broadcast on the
    #    hidden dimension.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating – sigmoid after linear projection (still fp32 for stability)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # bring gates to fp16 for the element‑wise multiplication
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy N³ contraction via Triton
    # ------------------------------------------------------------------
    # flatten (batch, hidden) → M dimension
    M = B * hidden_dim

    # ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty_like(left, dtype=torch.float16)   # will hold [B,N,N,hidden_dim]

    # strides (in elements, not bytes) for the 4‑D tensors
    s_left = left.stride()   # (N*N*hidden_dim, N*hidden_dim, hidden_dim, 1)
    s_right = right.stride()
    s_out = out.stride()

    # tile sizes – chosen to fit into shared memory on Hopper
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),   # program ids over (batch * hidden)
        triton.cdiv(N, BLOCK_N),   # program ids over output columns j
    )

    _trimul_kernel[
        grid
    ](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        # left strides
        s_left[0], s_left[2], s_left[3],
        # right strides (note: we need i‑stride = stride for dim1, j‑stride = stride for dim2)
        s_right[0], s_right[2], s_right[1], s_right[3],
        # output strides
        s_out[0], s_out[2], s_out[1], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # convert back to fp32 for the remaining layers
    out = out.to(torch.float32)                # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm on hidden dimension)
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
