"""
TriMul “outgoing” implementation (forward only).

The costly part of TriMul is the einsum  
    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]      (d = hidden_dim)

For each hidden channel this is a matrix multiplication
    (N×N) @ (N×N)ᵀ  →  (N×N)

We fuse this N×N‑by‑N×N multiplication across all batches and hidden
channels with a single Triton kernel. All other ops (LayerNorm,
Linear, gating, final projection) stay in PyTorch because they are
memory‑bound and already highly optimized.

The kernel computes a tiled MatMul for a given (batch, hidden) pair:
    acc[i_tile, j_tile] += Σ_k left[i_tile, k] * right[j_tile, k]

The kernel is launched with a 3‑D grid:
    (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B * hidden_dim)

The final output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul over the “k” dimension for each hidden slice
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # static sizes
    stride_l_b, stride_l_i, stride_l_k,          # left strides
    stride_r_b, stride_r_j, stride_r_k,          # right strides
    stride_o_b, stride_o_i, stride_o_j,          # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    # program ids -------------------------------------------------------
    pid_m = tl.program_id(0)      # tile over output rows   (i)
    pid_n = tl.program_id(1)      # tile over output cols   (j)
    pid_z = tl.program_id(2)      # combined (batch, hidden)

    b = pid_z // H                # batch index
    h = pid_z % H                 # hidden channel index

    # tile offsets ------------------------------------------------------
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # pointers with correct strides
        # left:   (b, i, k, h)
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            i_offset[:, None] * stride_l_i + \
            k_offset[None, :] * stride_l_k + \
            h * stride_l_b // H   # stride over hidden is stride_l_b // H (contiguous)

        # right:  (b, j, k, h) -> we need (k, j) for dot
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            k_offset[:, None] * stride_r_k + \
            j_offset[None, :] * stride_r_j + \
            h * stride_r_b // H

        # load tiles (zero‑pad out‑of‑bounds)
        a = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        b_tile = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b_tile)

    # store result
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        i_offset[:, None] * stride_o_i + \
        j_offset[None, :] * stride_o_j + \
        h * stride_o_b // H

    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Host helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, H)   float32/float16
    returns      : (B, N, N, H)   float32
    """
    B, N, _, H = left.shape
    left_flat = left.reshape(B * H, N, N).contiguous()
    right_flat = right.reshape(B * H, N, N).contiguous()
    out = torch.empty_like(left_flat)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_b, stride_r_j, stride_r_k = right_flat.stride()
    stride_o_b, stride_o_i, stride_o_j = out.stride()

    # launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out.reshape(B, N, N, H).to(torch.float32)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        (batch, seq_len, seq_len, dim)  float32
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Helper: linear with weight only (no bias)
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: (..., dim_in)   w: (dim_out, dim_in)
        return F.linear(x, w)

    # ------------------------------------------------------------------
    # LayerNorm (no bias for dim, bias provided separately)
    # ------------------------------------------------------------------
    def layer_norm(x, w, b):
        # w,b are 1‑D (dim,)
        return F.layer_norm(x, x.shape[-1:], weight=w, bias=b, eps=1e-5)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor,
                   weights["norm.weight"],
                   weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present). mask shape (B,N,N) -> broadcast
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        m = mask.unsqueeze(-1)                # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: sum over k -> batched matmul per hidden channel
    # ------------------------------------------------------------------
    # Use float16 inside the kernel for speed, cast back to float32
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out = tri_mul_triton(left_h, right_h)   # (B,N,N,hidden) float32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden -> hidden)
    # ------------------------------------------------------------------
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final projection back to original dim
    # ------------------------------------------------------------------
    out = linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
