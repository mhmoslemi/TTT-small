"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input `x`  (dim = C).
2. Linear projections → left / right (C → H) and three sigmoid‑gates.
3. Apply mask (if present) and gates.
4. Compute the core 4‑D einsum
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication over the sequence dimension:
        left  : (B, H, N, N)   (i‑k)
        right : (B, H, N, N)   (j‑k) → transpose k‑j → (B, H, N, N) (k‑j)
        out   = left @ rightᵀ   → shape (B, H, N, N)
   The multiplication is performed by a custom Triton GEMM kernel
   (block‑wise K‑reduction) on the flattened batch dimension B*H.
5. Reshape back to (B, N, N, H) and apply a second LayerNorm + out‑gate.
6. Final linear projection back to C.

Only the heavy N³ reduction (step 4) lives in Triton; everything else is
handled by PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton GEMM kernel (float16 → float16 accumulation in fp32)
# --------------------------------------------------------------
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    batch, M, N, K,               # problem size
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                # flattened batch (B*H)
    pid_m = tl.program_id(0)                # tile over M (i)
    pid_n = tl.program_id(1)                # tile over N (j)

    # -------------------- offsets ------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----------- pointers for the current tile ----------------
    a_ptrs = a_ptr + pid_b * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- reduction over K ----------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        # load a and b (mask out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)                      # (M,K)·(K,N) → (M,N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ----------- write back ----------------------------------
    c_ptrs = c_ptr + pid_b * stride_c_batch \
                   + offs_m[:, None] * stride_c_m \
                   + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton GEMM kernel.

    Args:
        data: tuple (x, mask, weights, config)
            x       – torch.Tensor, shape [B, N, N, C] (float32)
            mask    – torch.Tensor, shape [B, N, N] (bool/float) or None
            weights – dict of parameter tensors
            config  – dict with keys:
                       "dim" (C), "hidden_dim" (H), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core einsum via batched GEMM (Triton)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) → flatten batch*H
    B, N, _, _ = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous()            # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()          # (B, H, N, N)  (k‑j)

    batch_flat = B * hidden_dim
    a = left_mat.view(batch_flat, N, N).to(torch.float16)
    b = right_mat.view(batch_flat, N, N).to(torch.float16)

    c = torch.empty_like(a)                                    # fp16 output

    # launch Triton GEMM
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_flat,
    )
    _matmul_kernel[grid](
        a, b, c,
        batch_flat, N, N, N,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = c.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm, out‑gate and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])             # back to C

    return out
