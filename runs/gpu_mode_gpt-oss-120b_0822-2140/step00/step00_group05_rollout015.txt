"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------

1. **Layer‑norm** of the input `x` using the provided LN weight/bias.
2. **Linear projections** → left / right (shape B,N,N,H) and three gates
   (left‑gate, right‑gate, out‑gate) with a sigmoid.
3. **Fused element‑wise kernel** (Triton)  
   Performs `proj = proj * mask * gate` for the left and right tensors.
   This demonstrates a non‑trivial use of Triton while keeping the
   heavy contraction on the GPU‑accelerated matrix‑multiply path.
4. **Contraction** over the third dimension `k`  
   The einsum
   ```
   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   ```
   is equivalent to a batched matrix‑multiply for every hidden
   dimension `d`.  After permuting the tensors we call
   `torch.matmul`, which uses cuBLAS (very fast on Hopper).
5. **Output processing** – Layer‑norm over the hidden dimension,
   gating with `out_gate`, and the final linear projection back to
   `dim`.

The kernel works on contiguous tensors and is launched with a 1‑D grid.
All tensors are kept in `float32`; the kernel itself also uses `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------------- #
# Triton kernel: fused element‑wise multiplication (proj * mask * gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _fuse_proj_kernel(
    proj_ptr,      # [size]   – left or right projection
    mask_ptr,      # [size]   – mask (already broadcasted)
    gate_ptr,      # [size]   – gate (sigmoid output)
    out_ptr,       # [size]   – destination
    size,          # total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    m = offs < size                                   # guard

    proj = tl.load(proj_ptr + offs, mask=m)           # load projection
    mask = tl.load(mask_ptr + offs, mask=m)           # load mask (broadcasted)
    gate = tl.load(gate_ptr + offs, mask=m)           # load gate

    tl.store(out_ptr + offs, proj * mask * gate, mask=m)


# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul forward pass – “outgoing” variant.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor [B, N, N, D]
        - mask  : torch.Tensor [B, N, N]  (may be all‑ones if `nomask=True`)
        - weights: dict of pre‑loaded parameters
        - config : dict with keys
            * dim         – input channel dimension D
            * hidden_dim  – hidden projection dimension H
            * nomask      – bool, whether mask is present

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32, on the same device).
    """
    # --------------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device
    dtype = torch.float32

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Gates (sigmoid) – also linear without bias
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------------- #
    # 4. Mask handling – broadcast to hidden dim
    # --------------------------------------------------------------------- #
    if nomask:
        # an all‑ones mask saves a memory copy
        mask_fused = torch.ones_like(left, dtype=dtype, device=device)
    else:
        # mask: [B,N,N] → [B,N,N,1] → broadcast
        mask_fused = mask.unsqueeze(-1).to(dtype).expand_as(left)

    # --------------------------------------------------------------------- #
    # 5. Triton‑based fused multiplication (proj * mask * gate)
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    total_elems = left.numel()  # B * N * N * H
    BLOCK = 131072  # ~128 Ki elements per program; fits comfortably in shared memory

    # allocate fused tensors
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # launch kernel for left
    grid = (triton.cdiv(total_elems, BLOCK),)
    _fuse_proj_kernel[grid](
        left.contiguous().reshape(-1),
        mask_fused.contiguous().reshape(-1),
        left_gate.contiguous().reshape(-1),
        left_fused.reshape(-1),
        total_elems,
        BLOCK=BLOCK,
    )
    # launch kernel for right
    _fuse_proj_kernel[grid](
        right.contiguous().reshape(-1),
        mask_fused.contiguous().reshape(-1),
        right_gate.contiguous().reshape(-1),
        right_fused.reshape(-1),
        total_elems,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------------- #
    # 6. Contraction over k  (batched matmul per hidden dimension)
    #    left_fused [B,N,N,H] → (B,H,N,N)
    #    right_fused transposed on the last two dims
    # --------------------------------------------------------------------- #
    left_perm  = left_fused.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right_fused.permute(0, 3, 2, 1)  # (B, H, N, N)

    # batched matrix multiplication: (B,H,N,N) @ (B,H,N,N)ᵀ → (B,H,N,N)
    out_hidden = torch.matmul(left_perm, right_perm)  # (B, H, N, N)

    # back to [B,N,N,H]
    out = out_hidden.permute(0, 2, 3, 1)

    # --------------------------------------------------------------------- #
    # 7. Output layer‑norm (over hidden dim), gating and final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate          # gate after LN

    # final linear: hidden -> dim
    out = F.linear(out, weights["to_out.weight"])   # shape [B,N,N,dim]

    return out.to(dtype)
```
