"""
TriMul “outgoing” forward pass (AlphaFold3) with a custom Triton kernel.

The heavy part is the 4‑D contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix‑multiply per hidden channel:
    out[b,d] = left[b,d] @ right[b,d].transpose(-2,-1)

We keep the LayerNorms, linear layers and gating in PyTorch (they are cheap) and
offload the large N×N×hidden matrix multiplication to a fused Triton kernel.
The kernel works on FP16 data for speed and the final result is cast back to
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ B^T   (A,B : [batch, M, K] , [batch, N, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,          # *float16   [batch, M, K]
    b_ptr,          # *float16   [batch, N, K]
    c_ptr,          # *float16   [batch, M, N]
    M, N, K,
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bn, stride_bk,   # strides for B
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch dimension (B*hidden)
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A tile: shape [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K),
                    other=0.0)

        # load B tile and transpose it on‑the‑fly: shape [BLOCK_N, BLOCK_K]
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bn) + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_bk
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K),
                    other=0.0)
        # b is currently [BLOCK_N, BLOCK_K]; we need [BLOCK_K, BLOCK_N] for dot
        b = tl.trans(b)

        # dot: (M x K) @ (K x N) → (M x N)
        acc += tl.dot(a, b)

    # write back result in FP16
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a fused Triton kernel
    for the N³ contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------- LayerNorm on the input -------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------- Linear projections ----------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------- Mask (optional) --------------------------------
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(dtype)                 # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------- Gates -----------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # Prepare tensors for the Triton batched matmul
    #   left, right : [B,N,N,hidden]  →  [B*hidden, N, N]  (contiguous)
    # --------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B,hidden,N,N]
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    # cast to FP16 for the kernel
    left_mat = left_mat.to(torch.float16)
    right_mat = right_mat.to(torch.float16)

    batch = B * hidden
    M = N
    K = N
    N_ = N

    # strides in elements (not bytes)
    a_stride = left_mat.stride()
    b_stride = right_mat.stride()
    # a: [batch, M, K]
    stride_a_batch = a_stride[0]
    stride_am = a_stride[1]
    stride_ak = a_stride[2]
    # b: [batch, N, K]  (we will treat as [batch, K, N] after transpose in kernel)
    stride_b_batch = b_stride[0]
    stride_bn = b_stride[1]          # stride over N (row of B after transpose)
    stride_bk = b_stride[2]          # stride over K

    # output tensor
    out_mat = torch.empty((batch, M, N_), dtype=torch.float16, device=device)

    # output strides
    c_stride = out_mat.stride()
    stride_c_batch = c_stride[0]
    stride_cm = c_stride[1]
    stride_cn = c_stride[2]

    # --------------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # fits well in shared memory on H200

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a_ptr=left_mat.data_ptr(),
        b_ptr=right_mat.data_ptr(),
        c_ptr=out_mat.data_ptr(),
        M=M,
        N=N_,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_b_batch=stride_b_batch,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_c_batch=stride_c_batch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------------
    # Reshape back to [B,N,N,hidden] and continue in FP32
    # --------------------------------------------------------------------
    out = out_mat.to(dtype).view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------- Output LayerNorm -------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------- Multiply by out gate ----------------------------
    out = out * out_gate

    # ------------------- Final linear projection -------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
