"""
TriMul “outgoing” kernel (AlphaFold3).

*   LayerNorm and Linear layers are executed in PyTorch (they are cheap).
*   The cubic‑cost part  
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]    
    is implemented as a batched matrix‑multiply per hidden dimension `d`
    using a custom Triton kernel.
*   The kernel works on FP16 for the heavy inner‑product and writes back
    FP32 (the final `to_out` expects float‑32).
*   Supports optional masking, arbitrary batch size (1‑2) and sequence
    lengths up to 1024.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B,N,N,H] (i,k,d)
    right_ptr,         # const float*  [B,N,N,H] (j,k,d)
    out_ptr,           #       float*  [B,N,N,H] (i,j,d)
    B, N, H,           # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched mat‑mul over the “k” dimension, one hidden dimension per
    program (program_id(3))."""
    pid_b = tl.program_id(0)               # batch index
    pid_i = tl.program_id(1)               # block row (i)
    pid_j = tl.program_id(2)               # block col (j)
    pid_d = tl.program_id(3)               # hidden dimension

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulate in FP32, work in FP16 for bandwidth
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for this batch & hidden dimension
    left_base  = left_ptr \
               + pid_b * stride_left_b \
               + pid_d * stride_left_d
    right_base = right_ptr \
               + pid_b * stride_right_b \
               + pid_d * stride_right_d
    out_base   = out_ptr \
               + pid_b * stride_out_b \
               + pid_d * stride_out_d

    # ------------------------------------------------------------------
    # Loop over k
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + k_offset
        mask_k = cur_k < N

        # left[i,k]  (shape BLOCK_M x BLOCK_K)
        left_ptrs = left_base \
                  + (i_offset[:, None] * stride_left_i) \
                  + (cur_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float16)

        # right_T[k,j]  (shape BLOCK_K x BLOCK_N)
        # we load the transposed view directly
        right_ptrs = right_base \
                   + (cur_k[:, None] * stride_right_k) \
                   + (j_offset[None, :] * stride_right_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float16)

        # matmul (BLOCK_M x BLOCK_K) * (BLOCK_K x BLOCK_N) -> (BLOCK_M x BLOCK_N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    out_ptrs = out_base \
            + (i_offset[:, None] * stride_out_i) \
            + (j_offset[None, :] * stride_out_j)

    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Fast forward pass of the “outgoing” TriMul module.

    Args:
        data: (input, mask, weights, config)
            input  – tensor (B, N, N, dim)  (float32, on CUDA)
            mask   – tensor (B, N, N)       (bool/float, on CUDA)
            weights – dict of parameters (see above)
            config  – dict with keys: dim, hidden_dim, nomask (bool)

    Returns:
        Tensor of shape (B, N, N, dim), dtype float32, on the same device.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the 4‑D tensor (per channel)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # F.layer_norm expects normalized_shape = last dim
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]         # (hidden_dim, dim)

    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask   (mask shape: B,N,N)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(dtype).unsqueeze(-1)            # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy cubic term via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) – tensors are contiguous in the last dim
    stride_l = left.stride()   # (N*N*H, N*H, H, 1)
    stride_r = right.stride()
    stride_o = out.stride()

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        hidden_dim,               # hidden dimension as a 4th grid axis
    )
    trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 6) to_out_norm (LayerNorm on hidden_dim) and gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)         # (B,N,N,dim)

    return out
