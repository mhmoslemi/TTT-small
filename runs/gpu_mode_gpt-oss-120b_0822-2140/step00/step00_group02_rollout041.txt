"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden channel *d* the operation is a matrix multiplication
    (N×N) • (N×N)ᵀ .
We implement this batched GEMM in Triton (fp16 compute, fp32 accumulation)
and keep the remaining LayerNorm / Linear / gating work in PyTorch.
The final result is cast back to fp32, passed through the output LayerNorm,
gated and projected to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: one hidden channel per program‑id (fast enough for H≤768)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    L_ptr, R_ptr, O_ptr,                 # pointers
    B, N, H,                              # problem sizes
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for a single hidden channel `h` and a single batch `b`:
        O[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
    """
    pid0 = tl.program_id(0)          # encodes (batch, hidden)
    pid_m = tl.program_id(1)          # block row   (i)
    pid_n = tl.program_id(2)          # block col   (j)

    # decode batch and hidden index
    num_h = H
    batch = pid0 // num_h
    h     = pid0 %  num_h

    # --------------------------------------------------------------
    # block offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_block * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current tile
        L_ptrs = (L_ptr
                  + batch * stride_lb
                  + offs_i[:, None] * stride_li
                  + offs_k[None, :] * stride_lk
                  + h * stride_ld)
        R_ptrs = (R_ptr
                  + batch * stride_rb
                  + offs_j[:, None] * stride_rj
                  + offs_k[None, :] * stride_rk
                  + h * stride_rd)

        # load tiles (fp16)
        L = tl.load(L_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)
        R = tl.load(R_ptrs,
                    mask=(mask_j[:, None] & mask_k[None, :]),
                    other=0.0)

        # GEMM: (M,K) • (N,K)ᵀ   →   (M,N)
        acc += tl.dot(L, R, trans_b=True)   # acc in fp32

    # store result (fp16)
    O_ptrs = (O_ptr
              + batch * stride_ob
              + offs_i[:, None] * stride_oi
              + offs_j[None, :] * stride_oj
              + h * stride_od)

    tl.store(O_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim] (float32, on CUDA)
        mask    – Tensor[B, N, N] (bool/float, on CUDA)
        weights – dict with all linear / layer‑norm parameters (FP32)
        config  – dict with `dim` and `hidden_dim`

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                                 # (B,N,N,dim)  FP32

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (mask is broadcast on hidden dim)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.unsqueeze(-1)                # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) – also linear w/o bias
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core einsum: move to Triton (fp16 compute, fp32 acc)
    # --------------------------------------------------------------
    # cast to half for the kernel
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h   = torch.empty_like(left_h, dtype=torch.float16)

    # stride information (int64)
    stride_lb, stride_li, stride_lk, stride_ld = left_h.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_h.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                         # batch * hidden channel
        triton.cdiv(N, BLOCK_M),            # i‑blocks
        triton.cdiv(N, BLOCK_N)             # j‑blocks
    )

    _trimul_einsum_kernel[grid](
        left_h, right_h, out_h,
        B, N, hidden,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # back to fp32 for the remaining ops
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 6) Output LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
