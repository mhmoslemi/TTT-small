"""
TriMul “outgoing” implementation (forward only).

The heavy N³ part (einsum `left[i,k,d] * right[j,k,d] -> out[i,j,d]`) is
computed with a custom Triton kernel that performs a batched matrix‑multiply
for each hidden dimension:

    out_d[b, i, j] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorm, Linear projections, sigmoids, final
LayerNorm/Linear) are done with PyTorch – this satisfies the requirement
that *some* work is performed inside a Triton kernel while keeping the
implementation simple and correct.

The kernel works on FP16 data (the original AlphaFold3 kernels use bfloat16;
FP16 is also fast on Hopper).  The final output is converted back to
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: one hidden dimension at a time
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float16   (i, k) for a fixed batch & hidden dim
    right_ptr,         # *float16   (j, k) for the same batch & hidden dim
    out_ptr,           # *float16   (i, j) accumulation result
    N,                 # int        sequence length
    stride_left_i, stride_left_k,
    stride_right_j, stride_right_k,
    stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[i, j] = Σ_k left[i, k] * right[j, k]   (dot‑product over k)
    for a single hidden channel and a single batch element.
    """
    pid_m = tl.program_id(0)          # block index in i dimension
    pid_n = tl.program_id(1)          # block index in j dimension

    # ---------------------------------------------------------------
    # block coordinates
    # ---------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # reduction over k
    # ---------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: (M, K)
        left_ptrs = left_ptr + (offs_i[:, None] * stride_left_i
                                + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right: we need shape (K, N) to use tl.dot(A, B)
        right_ptrs = right_ptr + (offs_k[:, None] * stride_right_k
                                  + offs_j[None, :] * stride_right_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # dot: (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(left, right)

    # ---------------------------------------------------------------
    # write result
    # ---------------------------------------------------------------
    out_ptrs = out_ptr + (offs_i[:, None] * stride_out_i
                          + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Python entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – mixed PyTorch + Triton implementation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]       (bool/int, may be all‑ones)
        - weights      : dict of pre‑loaded tensors
        - config       : dict with "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape  # input shape = [B, N, N, dim]

    # -----------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    eps = 1e-5
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Apply mask (if present)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # same shape as left/right

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel for the N³ reduction (per hidden channel)
    # -----------------------------------------------------------------
    # Work in FP16 inside the kernel – cast once.
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    s_b_l, s_i_l, s_k_l, s_h_l = left.stride()
    s_b_r, s_j_r, s_k_r, s_h_r = right.stride()
    s_b_o, s_i_o, s_j_o, s_h_o = out.stride()

    # Triton block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    for b in range(B):
        # base pointers for the current batch
        left_batch_ptr = left.data_ptr() + b * s_b_l * left.element_size()
        right_batch_ptr = right.data_ptr() + b * s_b_r * right.element_size()
        out_batch_ptr = out.data_ptr() + b * s_b_o * out.element_size()

        for h in range(hidden):
            # offset to the required hidden channel
            left_ptr = left_batch_ptr + h * s_h_l * left.element_size()
            right_ptr = right_batch_ptr + h * s_h_r * right.element_size()
            out_ptr = out_batch_ptr + h * s_h_o * out.element_size()

            grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
            tri_mul_kernel[grid](
                left_ptr,
                right_ptr,
                out_ptr,
                N,
                stride_left_i=s_i_l,
                stride_left_k=s_k_l,
                stride_right_j=s_j_r,
                stride_right_k=s_k_r,
                stride_out_i=s_i_o,
                stride_out_j=s_j_o,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
            )

    # back to float32 for the rest of the pipeline
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm (over hidden) + gating
    # -----------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # out_gate has shape [B,N,N,hidden]
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7) Final linear projection back to dim
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
