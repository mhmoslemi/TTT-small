"""
TriMul “outgoing” implementation (forward only).

The expensive part of the forward pass is the einsum:
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

For a fixed batch‐and‑head index (b,d) this is exactly a matrix
multiplication   L(b,d) @ R(b,d).T   where L,R ∈ ℝ^{N×N}.
We fuse this N×N × N×N batched GEMM with a custom Triton kernel.
All other operations (LayerNorms, linear projections, sigmoids, masks)
are performed with native PyTorch – they are cheap compared with the
N³ reduction.

The kernel works on 2‑D blocks (BLOCK_M × BLOCK_N) with a K‑tiling.
Inputs are cast to fp16, the accumulator is fp32 (via tl.dot) and the
result is written back as fp16; the host code converts it to fp32
before the final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ Bᵀ)  –  A,B shape (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,               # *float16   (B*H, M, K)
    b_ptr,               # *float16   (B*H, N, K)   (note: B is already transposed)
    c_ptr,               # *float16   (B*H, M, N)
    M, N, K,             # matrix sizes
    stride_az, stride_am, stride_ak,   # strides of A
    stride_bz, stride_bk, stride_bn,   # strides of B (already transposed)
    stride_cz, stride_cm, stride_cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    # offsets inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers for the current batch/head
    a_batch_ptr = a_ptr + pid_z * stride_az
    b_batch_ptr = b_ptr + pid_z * stride_bz
    c_batch_ptr = c_ptr + pid_z * stride_cz

    # pointer arithmetic for the block
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)               # fp32 accumulation
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back (store as fp16, host will cast to fp32)
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper: layer‑norm without running‑mean (statistics computed per tensor)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv = torch.rsqrt(var + eps)
    y = (x - mean) * inv
    if weight is not None:
        y = y * weight
    if bias is not None:
        y = y + bias
    return y

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a fused
    batched GEMM (implemented in Triton).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask         : torch.Tensor [B, N, N]     (float32 or bool)
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor  [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape
    assert C == dim, "input last dim must match config dim"

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim], fp32

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # shape: (hidden_dim, dim)
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ   (per hidden dimension)
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (accumulator stays fp32)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # transpose the second matrix (swap last two axes)
    # after view, shape will be (B*H, N, N)
    left_flat  = left_f16.contiguous().view(B * hidden_dim, N, N)
    right_t    = right_f16.transpose(-2, -1).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16, shape (B*H,N,N)

    # launch parameters (tuned for Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch*head dimension
        triton.cdiv(N, BLOCK_M),           # M blocks
        triton.cdiv(N, BLOCK_N),           # N blocks
    )

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_t,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_az=left_flat.stride(0),
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bz=right_t.stride(0),
        stride_bk=right_t.stride(1),
        stride_bn=right_t.stride(2),
        stride_cz=out_flat.stride(0),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden_dim) and cast to fp32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection back to dim

    return out
