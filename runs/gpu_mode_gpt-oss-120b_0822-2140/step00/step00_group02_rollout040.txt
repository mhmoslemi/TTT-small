"""
TriMul "outgoing" implementation with a fused Triton kernel for the
N‑dimensional contraction:

    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

The heavy N³ work is performed as a batched GEMM over the hidden
dimension (d).  All other ops (LayerNorms, linear projections, gating,
and final linear) stay in PyTorch for clarity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ) over the
# hidden dimension.  Input tensors are expected as
#   left  : [B, H, N, K]   (K == N)
#   right : [B, H, N, K]
# Output:
#   out   : [B, H, N, N]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh_id = tl.program_id(0)                # flattened batch * hidden
    row_id = tl.program_id(1)               # block row index
    col_id = tl.program_id(2)               # block col index

    # offsets within the N‑dimensional matrix
    offs_i = row_id * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = col_id * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer to the start of this (b,h) slice
    left_base  = left_ptr  + bh_id * stride_left_bh
    right_base = right_ptr + bh_id * stride_right_bh
    out_base   = out_ptr   + bh_id * stride_out_bh

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:   [M, K]
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block (transposed): [K, N]
        right_ptrs = right_base \
                     + offs_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_j
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate a @ b  ->  [M, N]
        acc += tl.dot(a, b)

        # advance pointers for next K‑tile
        # (pointer arithmetic handled by recomputing a/b each iteration)

    # write back
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input   : [B, N, N, dim]   (float32)
        mask    : [B, N, N]        (float32 or bool)
        weights : dict of pretrained parameters
        config  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    out : [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hd]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) + apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Rearrange to [B, H, N, K] where K == N
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, K]
    right = right.permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 6. Triton GEMM over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.empty((B, hidden_dim, N, N), dtype=torch.float32,
                      device=device)

    # strides for pointer arithmetic (in elements, not bytes)
    stride_left_bh, stride_left_i, stride_left_k, _ = left.stride()
    stride_right_bh, stride_right_j, stride_right_k, _ = right.stride()
    stride_out_bh, stride_out_i, stride_out_j, _ = out.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_bh=stride_left_bh,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_bh=stride_right_bh,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_out_bh=stride_out_bh,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # bring back to original layout [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7. Final LayerNorm over hidden_dim, gating and linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
