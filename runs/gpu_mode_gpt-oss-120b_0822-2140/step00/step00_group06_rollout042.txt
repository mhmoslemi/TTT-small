"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the TriMul forward is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    out_d = left_d @ right_d.T   for every hidden dimension d.

The code below:
* performs LayerNorm, linear projections and gating in regular PyTorch (these
  are cheap compared to the N³ term);
* launches a Triton kernel that carries out the batched GEMM for all
  (batch, hidden) pairs in FP16 and writes the result in FP16;
* finishes the pipeline with a second LayerNorm, an output gate and a final
  linear projection back to `dim`.

The kernel works for any batch size (1‑2) and sequence length up to 1024,
with hidden dimension up to 384.  All tensors are assumed to reside on the
CUDA device already.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# -------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# -------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    out_ptr,            # *[B, N, N, H]  FP16 output
    left_ptr,           # *[B, N, N, H]  FP16 left tensor
    right_ptr,          # *[B, N, N, H]  FP16 right tensor
    B, N, H,            # int scalars
    # strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    # block sizes (constexpr)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM over the sequence dimension (i×j) for a single hidden
    slice `d`.  Tile‑wise computation with a K‑loop.
    """
    # -----------------------------------------------------------------
    # program IDs
    #   pid0 : combined (batch, hidden) dimension   → 0 .. B*H-1
    #   pid1 : tile over i (rows)
    #   pid2 : tile over j (cols)
    # -----------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # decode batch & hidden index
    b = pid0 // H                 # batch index
    h = pid0 % H                  # hidden index

    # offsets for the i‑ and j‑tiles
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the current batch/hidden slice
    out_base = out_ptr + b * stride_o_b + h * stride_o_d
    left_base = left_ptr + b * stride_l_b + h * stride_l_d
    right_base = right_ptr + b * stride_r_b + h * stride_r_d

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # K‑loop
    # -----------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kp in range(0, num_k):
        offs_k = kp * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (i, k)
        left_ptrs = (
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right: (j, k) – we will need its transpose (k, j)
        right_ptrs = (
            right_base
            + offs_j[:, None] * stride_r_j
            + offs_k[None, :] * stride_r_k
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # transpose right to shape (k, j)
        right_t = tl.trans(right, (1, 0))

        # matrix‑multiply the tiles (fp16 × fp16 → fp32 accumulation)
        acc += tl.dot(left, right_t)

    # -----------------------------------------------------------------
    # write result (cast back to fp16)
    # -----------------------------------------------------------------
    out_ptrs = (
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Helper functions (layer‑norm without torch.nn)
# -------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    """Manual LayerNorm over the last dimension."""
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# -------------------------------------------------------------
# Entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation using a fused Triton
    kernel for the N³ multiplication.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask:         [B, N, N]      (float32 or bool)
            - weights: dict of the model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    #  (B,N,N,dim) @ (dim, hidden) → (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        # mask shape → [B,N,N,1] to broadcast over hidden
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Core TriMul – launch the Triton kernel in FP16
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_o = out_fp16.stride()
    stride_l = left_fp16.stride()
    stride_r = right_fp16.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                # batch × hidden dimension
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
    )

    _trimul_kernel[grid](
        out_fp16,
        left_fp16,
        right_fp16,
        B, N, hidden_dim,
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_fp16.to(torch.float32)   # back to FP32 for the remaining ops

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
