@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute global batch*D id
    batch_d = pid
    total = B * D
    if batch_d >= total:
        return
    b = batch_d // D
    d = batch_d % D

    # compute block coordinates
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # offsets for M,N
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask within bounds
    mask_i = i < N
    mask_j = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left block (i,k)
        left_off = (b * stride_left_b + d * stride_left_d + i[:,None] * stride_left_i + k_offset[None,:] * stride_left_k)
        left = tl.load(left_ptr + left_off, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

        # load right block (j,k) need transpose: we need right[j,k] but we need right_T[k,j] for matmul.
        # So we load right at (b,d,j,k) then use tl.dot which expects (M,K) and (K,N) -> (M,N)
        right_off = (b * stride_right_b + d * stride_right_d + j[:,None] * stride_right_j + k_offset[None,:] * stride_right_k)
        right = tl.load(right_ptr + right_off, mask=mask_j[:,None] & mask_k[None,:], other=0.0)

        # dot
        acc += tl.dot(left, right)

        # advance pointers implicitly by varying offsets next iteration

    # store result
    out_off = (b * stride_out_b + d * stride_out_d + i[:,None] * stride_out_i + j[None,:] * stride_out_j)
    tl.store(out_ptr + out_off, acc.to(tl.float16) if tl.float16 else acc, mask=mask_i[:,None] & mask_j[None,:])

"""
TriMul "outgoing" implementation (AlphaFold3).

The expensive part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication:
    out_b,d = left_b,d @ right_b,d.T   (both are N×N matrices)

We keep LayerNorms and Linear projections in PyTorch (they are cheap) and
off‑load the N³ reduction to a custom Triton kernel that processes one
(batch, hidden) pair per program instance.  The kernel works on float32
(and keeps the accumulator in fp32) and writes the result back in the same
layout as the input tensors.

The final output is brought back to float32, goes through a second
LayerNorm, an output gate and a linear projection – exactly as the reference
PyTorch module.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matmul for the triple‑mul reduction
# --------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    # strides of left   (B,N,N,D)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides of right  (B,N,N,D)
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides of out    (B,N,N,D)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid0 = tl.program_id(0)                     # batch * hidden index
    total = B * D
    if pid0 >= total:
        return

    b = pid0 // D
    d = pid0 % D

    pid_i = tl.program_id(1)                    # block row (i)
    pid_j = tl.program_id(2)                    # block col (j)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,d]
        left_off = (
            b * stride_left_b + d * stride_left_d
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b,j,k,d]  (we need the k‑dimension aligned with left)
        right_off = (
            b * stride_right_b + d * stride_right_d
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        right = tl.load(
            right_ptr + right_off,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # write the result
    out_off = (
        b * stride_out_b + d * stride_out_d
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptr + out_off,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with all linear / norm parameters (float32)
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    left = left.contiguous()
    right = right.contiguous()

    # ---------- Triton reduction (einsum) ----------
    out = torch.empty_like(left)   # shape (B,N,N,hidden)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # program_id(0) iterates over (batch, hidden)
        triton.cdiv(N, BLOCK_M),       # program_id(1) rows
        triton.cdiv(N, BLOCK_N),       # program_id(2) cols
    )

    _trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_d=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_j=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_d=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_d=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------- Second LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ---------- Final linear ----------
    output = F.linear(out, weights["to_out.weight"])

    return output
