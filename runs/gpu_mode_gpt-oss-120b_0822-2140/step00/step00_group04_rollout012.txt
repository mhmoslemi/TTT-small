"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input `x`  (last dimension = dim).
2. Linear projections (no bias) → left/right tensors of shape [B,N,N,H].
3. Optional mask (broadcasted on the hidden dimension) is applied.
4. Gating networks: sigmoid(linear(x)) and element‑wise multiplication.
5. Core O(N³) operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per head:
        out = left @ right.transpose(-1,-2)
   Implemented with a custom Triton kernel that processes many
   (batch,head) pairs in parallel and uses fp16 arithmetic with
   fp32 accumulation.
6. Hidden‑dim layer‑norm, gating, and final linear projection back to `dim`.
   The final result is cast to float32 as required.

The heavy O(N³) step is fully fused in Triton; the surrounding
operations stay in PyTorch for flexibility.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel for the 3‑D einsum (left @ rightᵀ) per batch‑head.
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const * fp16  [B,N,N,H]
    right_ptr,         # const * fp16  [B,N,N,H]
    out_ptr,           #       * fp16  [B,N,N,H]

    B, N, H,           # int64 scalar sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled matrix‑multiply where the right operand is read
    transposed (k, j) to avoid an explicit transpose.
    """
    pid_m = tl.program_id(0)   # tile row   (i)
    pid_n = tl.program_id(1)   # tile col   (j)
    pid_b = tl.program_id(2)   # batch index
    pid_h = tl.program_id(3)   # head   index (hidden dimension)

    # ---- offsets within M/N dimensions ------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---- base pointers for this (batch,head) -------------------------------
    left_base  = left_ptr  + pid_b * stride_left_b  + pid_h * stride_left_h
    right_base = right_ptr + pid_b * stride_right_b + pid_h * stride_right_h
    out_base   = out_ptr   + pid_b * stride_out_b   + pid_h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- tiled K loop -------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M, K) – layout (i,k)
        left_ptrs = (left_base
                     + offs_m[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (K, N) – we read the transposed view (k,j)
        right_ptrs = (right_base
                      + offs_k[:, None] * stride_right_k
                      + offs_n[None, :] * stride_right_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # dot‑product accumulation
        acc += tl.dot(left, right)

    # ---- store result (cast back to fp16) ----------------------------------
    out_ptrs = (out_base
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Helper: LayerNorm without creating a nn.Module (bias is optional)
# ---------------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """x: [..., C]; weight/bias: (C,)"""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias

# ---------------------------------------------------------------------------
# Entry point required by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]
            - mask  : torch.Tensor [B, N, N]  (may be all‑ones)
            - weights: dict of torch.Tensor containing all linear / norm params
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called H in the paper

    # -----------------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"]
    )

    # -----------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep FP32 for the gating later
    # -----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # 3. Apply mask (if any) – mask is broadcast on hidden dimension
    # -----------------------------------------------------------------------
    if mask is not None:
        mask_u = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_u
        right = right * mask_u

    # -----------------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of left/right
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5. Core O(N³) multiplication – Triton kernel
    # -----------------------------------------------------------------------
    # Cast to fp16 for bandwidth‑efficient matmul; keep a FP32 copy for later
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty_like(left_fp16)   # same shape [B,N,N,H]

    B, N, _, H = left_fp16.shape

    # stride information (int64)
    s_lb, s_li, s_lk, s_lh = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rh = right_fp16.stride()
    s_ob, s_oi, s_oj, s_oh = out_fp16.stride()

    # Tile sizes – tuned for Hopper SM90, fp16 compute
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows i
        triton.cdiv(N, BLOCK_N),   # cols j
        B,                         # batch
        H                          # hidden dimension
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4   # enough to keep SM busy
    )

    # Back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
