"""
TriMul “outgoing” implementation.

The heavy part of the module is the three‑tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication for every hidden
dimension d:

    out_d = left_d @ right_d.transpose(-2,-1)

The kernel below computes this contraction in FP16 using Triton.
All cheap operations (LayerNorm, Linear, gating, final LayerNorm and
projection) are performed with PyTorch in FP32/F16.  The final result is
returned in FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: one hidden channel (d) per program instance
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                # const float16*  [B,N,N,H]
    right_ptr,               # const float16*  [B,N,N,H]
    out_ptr,                 #       float16*  [B,N,N,H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides (j,k order)
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,   # i‑tile size
    BLOCK_N: tl.constexpr,   # j‑tile size
    BLOCK_K: tl.constexpr,   # k‑tile size
):
    # program ids
    pid_b = tl.program_id(0)                      # batch
    pid_i = tl.program_id(1)                      # i‑tile
    pid_j = tl.program_id(2)                      # j‑tile
    d     = tl.program_id(3)                      # hidden channel

    # tile offsets
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds handling
    mask_i = i_off < N
    mask_j = j_off < N

    # pointers for the current hidden channel
    left_base  = left_ptr  + pid_b * stride_lb + d * stride_ld
    right_base = right_ptr + pid_b * stride_rb + d * stride_rd
    out_base   = out_ptr   + pid_b * stride_ob + d * stride_od

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + k_off
        mask_k = cur_k < N

        # -------- load left : (M, K) ----------
        left_ptrs = left_base \
            + i_off[:, None] * stride_li \
            + cur_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # -------- load right transposed : (K, N) ----------
        # right is stored as (b, j, k, d); we want (k, j) -> transpose on‑the‑fly
        right_ptrs = right_base \
            + cur_k[:, None] * stride_rk \
            + j_off[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # -------- accumulate ----------
        acc += tl.dot(left, right)       # (M,K) @ (K,N) -> (M,N)

    # store result in FP16
    out_ptrs = out_base \
        + i_off[:, None] * stride_oi \
        + j_off[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Args:
        data: tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim]   (float32)
            mask   : torch.Tensor [B, N, N]       (bool/float)
            weights: dict of all linear / layer‑norm parameters
            config : dict with keys dim, hidden_dim, nomask (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm + Linear + gating (PyTorch)
    # ------------------------------------------------------------------
    eps = 1e-5
    # LayerNorm over the last dim (dim)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + eps
    )
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections (no bias)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left  = left * mask_
        right = right * mask_

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # cast to FP16 for the contraction
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 2. Triton contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        hidden_dim,          # one program per hidden channel
    )
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # back to FP32 for the remaining ops
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 3. Final LayerNorm, gating and projection (PyTorch)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
