"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the forward pass is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication per hidden channel:
    out_d = left_d @ right_d.T   with left_d,right_d shaped (B, N, N).

We fuse this multiplication into a custom Triton kernel that processes
all hidden channels (H) together (batch dimension = B*H).  The rest of the
operations (layer‑norms, linear projections, gating) are kept in PyTorch for
brevity.  The kernel works in FP16 (fast arithmetic) while accumulating in
FP32 to keep numerical stability; the final result is cast back to FP32.

Supported shapes:
    B ∈ {1,2}, N ∈ {128,…,1024}, dim ∈ {128,384}, hidden_dim = 128
Both masked and un‑masked inputs are handled.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched matmul C = A @ B^T
#   A : (batch, M, K)   – left tensor   (M = N, K = N)
#   B : (batch, N, K)   – right tensor  (we treat its K‑dim as the inner dim,
#                                          and transpose it inside the kernel)
#   C : (batch, M, N)   – output
# ----------------------------------------------------------------------------- #
@triton.jit
def _triton_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_ab,          # A : (batch, M, K)
    stride_bn, stride_bk, stride_bb,          # B : (batch, N, K)
    stride_cm, stride_cn, stride_cb,          # C : (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)      # block-row
    pid_n = tl.program_id(1)      # block-col
    pid_b = tl.program_id(2)      # batch (B * hidden)

    # --------------------------------------------------------------------- #
    # Offsets within the current block
    # --------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # --------------------------------------------------------------------- #
    # Pointers + loads
    # --------------------------------------------------------------------- #
    a_ptrs = A + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)   # (BM, BK)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)   # (BK, BN)
        acc += tl.dot(a, b)                                                    # (BM, BN)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------------- #
    # Store result (FP16 for bandwidth, accumulated in FP32)
    # --------------------------------------------------------------------- #
    c_ptrs = C + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------------- #
# Public entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel
    for the expensive batched matrix multiplication.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask:        torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights:     dict of torch.Tensors (model parameters)
            - config:      dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------- #
    # 1) Layer‑norm over the last dimension (dim)
    # ----------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)

    # ----------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden)
    right = F.linear(x, weights["right_proj.weight"])  # (B, N, N, hidden)

    # ----------------------------------------------------------------- #
    # 3) Optional mask (broadcast on hidden dimension)
    # ----------------------------------------------------------------- #
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B, N, N, 1)
        left  = left  * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------- #
    # 4) Gating streams (sigmoid)
    # ----------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # (B,N,N,hidden)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # (B,N,N,hidden)

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 5) Batched matrix multiplication via Triton
    #    out_d = left_d @ right_d.T   for each hidden channel d
    # ----------------------------------------------------------------- #
    B, N, _, _ = left.shape

    # fp16 for bandwidth, accumulate in fp32 inside the kernel
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # fuse batch (B) and hidden (H) dimensions → (B*H, N, N)
    left_flat  = left_f16.reshape(B * hidden, N, N).contiguous()
    right_flat = right_f16.reshape(B * hidden, N, N).contiguous()
    out_flat   = torch.empty_like(left_flat, dtype=torch.float16)

    # strides (in elements) for the three dimensions
    stride_ab, stride_am, stride_ak = left_flat.stride()   # (batch, M, K)
    stride_bb, stride_bn, stride_bk = right_flat.stride()  # (batch, N, K)
    stride_cb, stride_cm, stride_cn = out_flat.stride()   # (batch, M, N)

    # ----------------------------------------------------------------- #
    # Launch parameters (tuned for Hopper – can be adjusted)
    # ----------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over rows (i)
        triton.cdiv(N, BLOCK_N),   # blocks over cols (j)
        B * hidden,                # batch * hidden
    )

    _triton_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_am, stride_ak, stride_ab,
        stride_bn, stride_bk, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden) and cast to FP32 for the remaining ops
    out = out_flat.to(torch.float32).view(B, N, N, hidden)

    # ----------------------------------------------------------------- #
    # 6) Final LayerNorm, gating, and projection back to dim
    # ----------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
