"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The heavy N³ reduction   
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]   
is expressed as a batched GEMM:
    left_d (N×N) @ right_dᵀ (N×N) → (N×N)   for every hidden channel d.
We keep the linear / LayerNorm work in PyTorch (highly optimised) and
off‑load the element‑wise mask & gate fusion to a tiny Triton kernel.
The kernel multiplies `left` and `right` by the pairwise mask and by the
corresponding gates (broadcast over the hidden dimension).

The final matmul is performed with `torch.bmm` on BF16 tensors, then the
output is LayerNorm‑ed, gated and projected back to the original dimension.
All tensors are assumed to already reside on the CUDA device.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: apply mask (B×N×N) and per‑channel gates to left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    # strides for left
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    # strides for mask (no channel dim)
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides for left_gate
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    # strides for right_gate
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ---------- decode linear index into (b,i,j,h) ----------
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # ---------- compute addresses ----------
    left_addr = (left_ptr
                 + b * stride_left_b
                 + i * stride_left_i
                 + j * stride_left_j
                 + h * stride_left_h)

    right_addr = (right_ptr
                  + b * stride_right_b
                  + i * stride_right_i
                  + j * stride_right_j
                  + h * stride_right_h)

    mask_addr = (mask_ptr
                 + b * stride_mask_b
                 + i * stride_mask_i
                 + j * stride_mask_j)

    lgate_addr = (left_gate_ptr
                  + b * stride_lgate_b
                  + i * stride_lgate_i
                  + j * stride_lgate_j
                  + h * stride_lgate_h)

    rgate_addr = (right_gate_ptr
                  + b * stride_rgate_b
                  + i * stride_rgate_i
                  + j * stride_rgate_j
                  + h * stride_rgate_h)

    # ---------- load ----------
    left_val = tl.load(left_addr, mask=mask, other=0.0)
    right_val = tl.load(right_addr, mask=mask, other=0.0)
    m_val = tl.load(mask_addr, mask=mask, other=0.0)          # 0 or 1
    lgate_val = tl.load(lgate_addr, mask=mask, other=0.0)
    rgate_val = tl.load(rgate_addr, mask=mask, other=0.0)

    # ---------- fuse mask & gates ----------
    left_val = left_val * m_val * lgate_val
    right_val = right_val * m_val * rgate_val

    # ---------- store ----------
    tl.store(left_addr, left_val, mask=mask)
    tl.store(right_addr, right_val, mask=mask)


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N]   (may be all‑ones)
            - weights      : dict of torch.Tensor containing all linear / LN params
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=dim,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                                 # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) – also hidden_dim
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask and gates with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # make tensors contiguous – Triton needs contiguous memory
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    BLOCK = 8192                 # ~64 Ki elements per launch (tuned for Hopper)
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    apply_mask_gate_kernel[grid](
        left, right, mask,
        left_gate, right_gate,
        B, N, hidden_dim,
        # left strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        # right strides
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        # mask strides (no channel)
        mask.stride(0), mask.stride(1), mask.stride(2),
        # left_gate strides
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        # right_gate strides
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5️⃣ N³ reduction = batched GEMM per hidden channel (BF16 for speed)
    # ------------------------------------------------------------------
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to (B*hidden_dim, N, N)
    left_mat = left_bf.view(B * hidden_dim, N, N)
    right_mat = right_bf.view(B * hidden_dim, N, N)

    # batched matmul: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → [B, N, N, hidden_dim] (float32)
    out = out_mat.view(B, N, N, hidden_dim).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=hidden_dim,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])

    return out
