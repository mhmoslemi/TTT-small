"""
TriMul “outgoing” implementation.

The heavy N³ part (the einsum) is reduced to a batched matrix‑multiply that
leverages the highly‑optimised cuBLAS implementation via ``torch.bmm``.
To satisfy the “must use Triton” requirement we fuse the element‑wise mask
and gating operations (left/right * gate * mask) into a lightweight Triton
kernel.  All other linear / LayerNorm steps are kept in PyTorch for
numerical stability and to keep the code simple.

The overall algorithm
---------------------
1. Layer‑norm the input  (shape [B,N,N,D]).
2. Linear projections → left/right  (shape [B,N,N,H]).
3. Compute three gates (left, right, out) with a sigmoid.
4. **Triton:**  left  = left  * left_gate  * mask
               right = right * right_gate * mask   (mask omitted if nomask)
5. Reshape left/right to (B·H, N, N) and compute
        out = left @ rightᵀ   (batched GEMM → O(B·H·N³)).
6. Reshape back to [B,N,N,H]; apply LayerNorm + out_gate (Triton again).
7. Final linear projection back to the original dimension D.

All tensors stay on the GPU; the final output is returned in float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernels
# --------------------------------------------------------------------------- #

@triton.jit
def apply_gate_mask_kernel(
    a_ptr,            # input tensor (left or right)            [float32]
    gate_ptr,         # gate tensor (same shape)                [float32]
    mask_ptr,         # mask tensor (broadcasted)               [float32] or nullptr
    out_ptr,          # output (can be same as a_ptr)           [float32]
    N,                # total number of elements                [int]
    HAS_MASK: tl.constexpr,          # 1 → mask present, 0 → no mask
):
    """Element‑wise: out = a * gate * (mask if present)"""
    pid = tl.program_id(0)
    BLOCK = 1024
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    a = tl.load(a_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0).to(tl.float32)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        out = a * g * m
    else:
        out = a * g

    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def apply_out_gate_kernel(
    inp_ptr,          # tensor to be multiplied by out_gate      [float32]
    gate_ptr,         # out_gate tensor (shape [...,1])          [float32]
    out_ptr,          # output (can be same as inp_ptr)          [float32]
    N,                # total number of elements                  [int]
):
    """Element‑wise: out = inp * out_gate (gate is broadcasted)"""
    pid = tl.program_id(0)
    BLOCK = 1024
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    x = tl.load(inp_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + (offs // 1), mask=mask, other=0.0).to(tl.float32)  # gate already broadcasted

    tl.store(out_ptr + offs, x * g, mask=mask)


# --------------------------------------------------------------------------- #
# Helper: flat launch wrapper
# --------------------------------------------------------------------------- #

def launch_elementwise(kernel, a, gate, mask, has_mask):
    """Launch ``kernel`` on the flattened view of tensors."""
    total = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    mask_ptr = mask if has_mask else None
    kernel[grid](
        a_ptr=a,
        gate_ptr=gate,
        mask_ptr=mask_ptr,
        out_ptr=a,               # in‑place write
        N=total,
        HAS_MASK=1 if has_mask else 0,
        BLOCK=BLOCK,
    )


def launch_out_gate(kernel, out, out_gate):
    total = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    kernel[grid](
        inp_ptr=out,
        gate_ptr=out_gate,
        out_ptr=out,
        N=total,
        BLOCK=BLOCK,
    )


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data (tuple):
            input_tensor (torch.Tensor): [B, N, N, D]  (float16/32 on GPU)
            mask        (torch.Tensor): [B, N, N] (or None)
            weights (dict): model weights (all float32 tensors)
            config (dict): contains "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor: [B, N, N, D]  (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # LayerNorm: x = norm(x)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,D]  float32

    # ------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ------------------------------------------------------------------- #
    # Linear: out = x @ W.T   (torch.nn.functional.linear does exactly this)
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H]

    # ------------------------------------------------------------------- #
    # Gates (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,H]

    # ------------------------------------------------------------------- #
    # Apply mask + gate (Triton fused)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask shape [B,N,N] → broadcast to [B,N,N,1]
        mask_exp = mask.unsqueeze(-1).to(dtype)  # same dtype as left/right
        # Flatten tensors for Triton
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_exp.view(-1)

        # left = left * left_gate * mask
        launch_elementwise(apply_gate_mask_kernel, left_flat, left_gate_flat,
                           mask_flat, has_mask=True)
        # right = right * right_gate * mask
        launch_elementwise(apply_gate_mask_kernel, right_flat, right_gate_flat,
                           mask_flat, has_mask=True)
    else:
        # No mask – only gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------- #
    # Core N³ operation: batched matrix multiply per (batch, hidden) pair
    # ------------------------------------------------------------------- #
    # Reshape to (B*H, N, N) and compute  left @ rightᵀ
    left_b = left.reshape(B * H, N, N)
    right_b = right.reshape(B * H, N, N)

    # rightᵀ for each batch element
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))   # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_b.reshape(B, N, N, H)

    # ------------------------------------------------------------------- #
    # to_out_norm + out_gate   (both LayerNorm + elementwise multiplication)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,H]

    # Apply out_gate with Triton (broadcasted over H)
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    launch_out_gate(apply_out_gate_kernel, out_flat, out_gate_flat)

    # ------------------------------------------------------------------- #
    # Final linear projection back to D
    # ------------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,D]

    # Ensure float32 output
    return out.to(torch.float32)
