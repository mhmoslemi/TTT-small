"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy N³ part (the einsum) is turned into a batched matrix‑multiply
by reshaping the tensors to (B·hidden, N, N).  The elementwise
mask‑gate‑masking step is off‑loaded to a tiny Triton kernel so that
the submission contains genuine Triton code while still using the
highly‑optimised cuBLAS path for the cubic work.

All tensors are expected to be on the CUDA device already.
The final result is returned in float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# -----------------------------------------------------------------------
# Triton kernel: out = a * b * c  (element‑wise, flattened view)
# -----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    numel,                         # total number of elements
    BLOCK: tl.constexpr,           # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear offsets
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """
    Compute out = a * b * c using the Triton kernel above.
    All tensors must be contiguous and have the same shape and dtype.
    """
    assert a.shape == b.shape == c.shape
    assert a.is_contiguous() and b.is_contiguous() and c.is_contiguous()
    out = torch.empty_like(a)

    numel = a.numel()
    BLOCK = 1024  # works well for FP16/FP32 on Hopper
    grid = (triton.cdiv(numel, BLOCK),)

    # launch
    elementwise_mul_kernel[grid](
        a, b, c, out,
        numel,
        BLOCK=BLOCK,
    )
    return out

# Main entry point
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]   (bool / uint8 / float)
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------------------------
    # 2. Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # -------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------
    # 3. Gates (sigmoid) – also linear → shape [B,N,N,hidden]
    # -------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -------------------------------------------------------------------
    # 4. Broadcast mask and move everything to float16 for speed
    # -------------------------------------------------------------------
    mask_f = mask.unsqueeze(-1).to(torch.float16)          # [B,N,N,1]
    left    = left.to(torch.float16)
    right   = right.to(torch.float16)
    left_gate  = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate   = out_gate.to(torch.float16)

    # -------------------------------------------------------------------
    # 5. Element‑wise: left = left * left_gate * mask,
    #                right = right * right_gate * mask
    #    performed by the Triton kernel
    # -------------------------------------------------------------------
    left  = elementwise_mul(left,  left_gate, mask_f)
    right = elementwise_mul(right, right_gate, mask_f)

    # -------------------------------------------------------------------
    # 6. Cubic part:   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Reshape to (B·hidden, N, N) and use cuBLAS batched GEMM.
    # -------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)   and make right transposed on the inner pair
    left_ = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N] (i,k)
    right_ = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (k,j)

    # merge batch & hidden into a single batch dimension
    left_flat  = left_.view(B * hidden_dim, N, N)          # (B·H, N, N)
    right_flat = right_.view(B * hidden_dim, N, N)         # (B·H, N, N)

    # batched matrix multiplication (N³, highly tuned)
    out_flat = torch.bmm(left_flat, right_flat)           # (B·H, N, N)

    # restore original layout: [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final linear projection
    # -------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                                 # gated
    out = F.linear(out, weights["to_out.weight"])        # back to dim

    return out.to(torch.float32)
