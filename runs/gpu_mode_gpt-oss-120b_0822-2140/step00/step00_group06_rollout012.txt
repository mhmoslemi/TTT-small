"""
TriMul “outgoing” implementation.

The expensive part of the module is the three‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication of shape
    (B·H, N, N) × (B·H, N, N)ᵀ   with H = hidden_dim.

We fuse this multiplication in a single Triton kernel that:
    * works on the packed view (B·H, N, N)
    * processes a block of rows (M) × columns (N) at a time
    * iterates over the K‑dimension in blocks.
All surrounding operations (LayerNorm, linear projections, gating, the final
linear and LayerNorm) are kept in PyTorch – they are cheap compared with the
N³ · H work of the einsum.

The kernel is launched with a three‑dimensional grid
    (grid_M, grid_N, B·H)   where program_id(2) encodes both the batch
    index and the hidden‑dimension slice.
The result is reshaped back to [B, N, N, H] and the remaining
PyTorch post‑processing yields the final [B, N, N, dim] tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape = (batch*H, N, N))
# ------------------------------------------------------------
@triton.jit
def triton_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    N,                                 # sequence length (square side)
    stride_ah, stride_ai, stride_ak,   # A strides: (batch_h, row, col)
    stride_bh, stride_bi, stride_bk,   # B strides
    stride_ch, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[bh, i, j] = Σ_k A[bh, i, k] * B[bh, j, k]
    where bh = batch * hidden_dim index.
    """

    # ------------------------------------------------------------------
    # 1) program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_z = tl.program_id(2)          # batch*hidden index

    # ------------------------------------------------------------------
    # 2) offsets within the block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 3) accumulator (use float32 for better accuracy)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # ----------- load a tile (M x K) --------------------------------
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptrs = (a_ptr +                                             #
                  pid_z * stride_ah +                                 # batch*hidden stride
                  offs_m[:, None] * stride_ai +                       # row stride
                  offs_k[None, :] * stride_ak)                        # col stride
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ----------- load b tile (N x K)  (note: we will trans‑pose it) --
        # we need B[bh, j, k] i.e. row = j, col = k.
        b_ptrs = (b_ptr +                                             #
                  pid_z * stride_bh +                                 # batch*hidden
                  offs_n[:, None] * stride_bi +                       # row (j) stride
                  offs_k[None, :] * stride_bk)                        # col (k) stride
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # ----------- compute block dot product ---------------------------
        # a : (M, K)   b : (N, K)   we need (M,N) += a @ bᵀ
        # tl.dot expects (M,K) @ (K,N), so we transpose b on‑the‑fly.
        acc += tl.dot(a, b, trans_b=True)

    # ------------------------------------------------------------------
    # 5) write back
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr +                                                 #
              pid_z * stride_ch +                                     # batch*hidden
              offs_m[:, None] * stride_ci +                           # row stride
              offs_n[None, :] * stride_cj)                            # col stride
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ------------------------------------------------------------
# Helper: packed view utilities
# ------------------------------------------------------------
def _pack(t: torch.Tensor) -> Tuple[torch.Tensor, int]:
    """Reshape (B,N,N,H) -> (B*H,N,N) and return the new tensor + H."""
    B, N, _, H = t.shape
    t = t.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    t = t.view(B * H, N, N)
    return t, H


def _unpack(t: torch.Tensor, B: int, N: int, H: int) -> torch.Tensor:
    """Inverse of _pack: (B*H,N,N) -> (B,N,N,H)."""
    t = t.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return t


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel for
    the N³ × hidden_dim reduction.

    Args:
        data: (input, mask, weights, config)
            - input  : torch.Tensor of shape [B, N, N, dim]
            - mask   : torch.Tensor of shape [B, N, N]  (may be None)
            - weights: dict with all linear / norm parameters
            - config : dict containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dtype = torch.float32

    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(inp,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N,1) broadcast over hidden_dim
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton‑accelerated 3‑D einsum
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (memory bandwidth/throughput win)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # pack to (B*H, N, N)
    left_packed, H = _pack(left)      # H == hidden_dim
    right_packed, _ = _pack(right)

    # allocate output
    out_packed = torch.empty_like(left_packed, dtype=torch.float16, device=device)

    # launch configuration (tuned for Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim,                   # batch * hidden slice
    )

    triton_matmul_transpose_kernel[grid](
        a_ptr=left_packed,
        b_ptr=right_packed,
        c_ptr=out_packed,
        N=N,
        stride_ah=left_packed.stride(0),
        stride_ai=left_packed.stride(1),
        stride_ak=left_packed.stride(2),
        stride_bh=right_packed.stride(0),
        stride_bi=right_packed.stride(1),
        stride_bk=right_packed.stride(2),
        stride_ch=out_packed.stride(0),
        stride_ci=out_packed.stride(1),
        stride_cj=out_packed.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # unpack back to (B,N,N,hidden_dim) and cast to fp32
    out = _unpack(out_packed.to(dtype), B, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6) post‑einsum LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # -> (B,N,N,dim)

    return out.to(dtype)
