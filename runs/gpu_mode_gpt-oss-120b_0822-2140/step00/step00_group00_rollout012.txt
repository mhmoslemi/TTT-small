"""
TriMul “outgoing” implementation.
The heavy O(N³·C) part (the einsum) is performed by a custom Triton kernel.
All element‑wise ops (LayerNorm, linear projections, gates) are kept in
torch for simplicity.  The kernel computes

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with left/right already masked and gated.
The kernel works for any batch size ≤2, sequence length up to 1024 and
hidden dimension that is a multiple of 8 (the tests use 128 or 384).
Only the forward pass is required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: reduction over the “k” dimension of left/right
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const * left  [B,N,N,H]
    right_ptr,              # const * right [B,N,N,H]
    out_ptr,                # * out          [B,N,N,H]

    B, N, H,                # shape parameters

    # strides for left  (batch, i, k, d)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right (batch, j, k, d)
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for out   (batch, i, j, d)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # ----------------------------------------------------------
    # 1) program IDs  (grid is (B, ceil(N/BLOCK_I), ceil(N/BLOCK_J)))
    # ----------------------------------------------------------
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # base offsets for this block
    off_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    off_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    off_d = tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds i/j/d
    mask_i = off_i < N
    mask_j = off_j < N
    mask_d = off_d < H

    # combine masks for the final store
    # shape (BLOCK_I, BLOCK_J, BLOCK_D)
    mask_ij = mask_i[:, None] & mask_j[None, :]
    mask_ijd = mask_ij[:, :, None] & mask_d[None, None, :]

    # pointers to the start of the current batch
    left_batch_ptr  = left_ptr  + pid_b * stride_left_b
    right_batch_ptr = right_ptr + pid_b * stride_right_b
    out_batch_ptr   = out_ptr   + pid_b * stride_out_b

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ----------------------------------------------------------
    # 2) reduction over k dimension (blocked)
    # ----------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # -------- load left[i,k,d] --------
        # shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_off = (
            off_i[:, None, None] * stride_left_i
            + cur_k[None, :, None] * stride_left_k
            + off_d[None, None, :] * stride_left_d
        )
        left = tl.load(left_batch_ptr + left_off, mask=mask_i[:, None, None] & mask_k[None, :] & mask_d[None, None, :], other=0.0)

        # -------- load right[j,k,d] --------
        # shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_off = (
            off_j[:, None, None] * stride_right_j
            + cur_k[None, :, None] * stride_right_k
            + off_d[None, None, :] * stride_right_d
        )
        right = tl.load(right_batch_ptr + right_off,
                        mask=mask_j[:, None, None] & mask_k[None, :] & mask_d[None, None, :],
                        other=0.0)

        # -------- broadcast and multiply --------
        # left : (I, K, D) -> (I, 1, K, D)
        # right: (J, K, D) -> (1, J, K, D)
        # product -> (I, J, K, D)
        prod = left[:, None, :, :] * right[None, :, :, :]

        # sum over K dimension -> (I, J, D)
        acc += tl.sum(prod, axis=2).to(tl.float32)

    # ----------------------------------------------------------
    # 3) write out
    # ----------------------------------------------------------
    out_off = (
        off_i[:, None, None] * stride_out_i
        + off_j[None, :, None] * stride_out_j
        + off_d[None, None, :] * stride_out_d
    )
    tl.store(out_batch_ptr + out_off, acc.to(tl.float16), mask=mask_ijd)


# --------------------------------------------------------------
# Helper: simple LayerNorm (weight & bias are 1‑D tensors)
# --------------------------------------------------------------
def fused_layernorm(x, weight, bias, eps=1e-5):
    # x: [..., C]  float32
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------
# Entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, on CUDA)
        - mask         : [B, N, N]        (float32/bool, on CUDA)
        - weights      : dict of tensors (all float32, on CUDA)
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------------
    x = fused_layernorm(input_tensor, weights["norm.weight"], weights["norm.bias"])

    # -----------------------------------------------------------------
    # 2) Linear projections (bias‑free) – we stay in fp32 for accuracy
    # -----------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Apply the optional mask (mask is broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1)        # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid) and gating of left/right
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel: compute the “tri‑mul” reduction
    # -----------------------------------------------------------------
    # Convert to float16 for bandwidth‑limited kernel, keep a FP32 copy for safety
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h = torch.empty_like(left_h)          # [B,N,N,hidden]

    # strides (must be in elements, not bytes)
    s_lb, s_li, s_lk, s_ld = left_h.stride()
    s_rb, s_rj, s_rk, s_rd = right_h.stride()
    s_ob, s_oi, s_oj, s_od = out_h.stride()

    # tile sizes – chosen to fit comfortably into H200 shared memory
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden_dim is a multiple of 8 (128, 384)

    grid = (B,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J))

    trimul_outgoing_kernel[grid](
        left_h, right_h, out_h,
        B, N, hidden_dim,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    # Back to fp32 for the remaining layers
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Final LayerNorm, gating, and output projection
    # -----------------------------------------------------------------
    out = fused_layernorm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape [B,N,N,dim]

    return out
