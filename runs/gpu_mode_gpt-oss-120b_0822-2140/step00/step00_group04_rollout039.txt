"""
TriMul forward (outgoing) kernel.

The heavy‑weight part of the module is the pairwise multiplication
summation over the intermediate sequence dimension `k`

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]      (1)

For every batch `b` and hidden channel `d` this is exactly a
matrix‑multiply of `left[b, :, :, d]` (N×N) with the transpose of
`right[b, :, :, d]` (N×N).  The kernel below fuses the three
operations (load‑mask‑gate, GEMM and store) for all `(b,d)` pairs.
All other Linear/LayerNorm ops stay in PyTorch; the final result
remains float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton GEMM kernel (left @ right.T) for many (batch, channel) pairs
# ----------------------------------------------------------------------
@triton.jit
def _triton_tri_mul_gemm(
    left_ptr,               # const * left  [B, N, N, H]  (float16)
    right_ptr,              # const * right [B, N, N, H]  (float16)
    out_ptr,                # * out          [B, N, N, H]  (float16)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr,  # rows of the output tile
    BLOCK_N: tl.constexpr,  # cols of the output tile
    BLOCK_K: tl.constexpr,  # reduction tile
):
    """ Compute out = left @ rightᵀ for every (b, h) pair.

    left  shape: (B, N, N, H) indexed as [b, i, k, h]
    right shape: (B, N, N, H) indexed as [b, j, k, h]
    out   shape: (B, N, N, H) indexed as [b, i, j, h]
    """
    pid = tl.program_id(0)

    # total number of tiles = B * H * ceil(N/BM) * ceil(N/BN)
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_pair = num_m * num_n
    b = pid // (H * tiles_per_pair)
    tmp = pid % (H * tiles_per_pair)
    h = tmp // tiles_per_pair
    tile_id = tmp % tiles_per_pair
    tile_i = tile_id // num_n
    tile_j = tile_id % num_n

    # tile coordinates in the output matrix
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h] -> (M, K)
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + h * stride_left_h
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left_val = tl.load(left_ptrs,
                           mask=(mask_i[:, None] & mask_k[None, :]),
                           other=0.0).to(tl.float32)

        # right[b, j, k, h] -> (N, K)
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + h * stride_right_h
                      + offs_j[:, None] * stride_right_i
                      + offs_k[None, :] * stride_right_k)
        right_val = tl.load(right_ptrs,
                            mask=(mask_j[:, None] & mask_k[None, :]),
                            other=0.0).to(tl.float32)

        # dot over K dimension
        acc += tl.dot(left_val, right_val)

    # store fp16
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_k)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton kernel
    for the N³ einsum part.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor [B, N, N, dim] (float32)
        mask    – Tensor [B, N, N] (float32 or bool)
        weights – dict with all linear / layer‑norm parameters
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor – [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # LayerNorm on the input (dim)
    # --------------------------------------------------------------
    # compute mean & var over the last dimension
    x_flat = input_tensor.view(-1, dim)
    mean = x_flat.mean(-1, keepdim=True)
    var = x_flat.var(-1, unbiased=False, keepdim=True)
    mean = mean.view(B, N, N, 1)
    var = var.view(B, N, N, 1)

    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b          # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Cast to float16 for the heavy N³ kernel
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output in fp16
    out_fp16 = torch.empty((B, N, N, hidden),
                           dtype=torch.float16,
                           device=device)

    # --------------------------------------------------------------
    # Triton launch parameters
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # strides for contiguous tensors (B, N, N, H)
    stride_left = left_fp16.stride()   # (B*N*N*H, N*H, H, 1)
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()

    grid = (
        triton.cdiv(B * hidden * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), 1),
    )

    _triton_tri_mul_gemm[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # back to fp32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)            # [B,N,N,hidden]

    # --------------------------------------------------------------
    # LayerNorm on the hidden dimension
    # --------------------------------------------------------------
    out_flat = out.view(-1, hidden)
    mean_h = out_flat.mean(-1, keepdim=True)
    var_h = out_flat.var(-1, unbiased=False, keepdim=True)
    mean_h = mean_h.view(B, N, N, 1)
    var_h = var_h.view(B, N, N, 1)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out - mean_h) * torch.rsqrt(var_h + eps)
    out = out * to_out_norm_w + to_out_norm_b   # [B,N,N,hidden]

    # --------------------------------------------------------------
    # Apply output gate and final projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
