def custom_kernel(data):
   input_tensor, mask, weights, config = data
   dim, hidden_dim = config['dim'], config['hidden_dim']
   device = input_tensor.device
   # layernorm and linear using torch ops
   # Use torch.nn.functional.linear (no bias)
   # Use torch.nn.functional.layer_norm (need weight,bias)
   # We'll apply weights manually.

   # Norm
   x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5)
   x = x * weights['norm.weight'] + weights['norm.bias']

   # Linear projections
   # shape: [B,N,N,dim] -> [B,N,N,hidden]
   x_flat = x.view(-1, dim)  # (B*N*N, dim)
   left = torch.nn.functional.linear(x_flat, weights['left_proj.weight']).view_as(x).type(torch.float16)
   right = torch.nn.functional.linear(x_flat, weights['right_proj.weight']).view_as(x).type(torch.float16)

   # mask
   if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

   # gates
   left_gate = torch.nn.functional.linear(x_flat, weights['left_gate.weight']).view_as(x).sigmoid().type(torch.float16)
   right_gate = torch.nn.functional.linear(x_flat, weights['right_gate.weight']).view_as(x).sigmoid().type(torch.float16)
   out_gate = torch.nn.functional.linear(x_flat, weights['out_gate.weight']).view_as(x).sigmoid().type(torch.float16)

   left = left * left_gate
   right = right * right_gate

   # cast to float16 for kernel acceleration, compute out in float32 accumulation
   left = left.to(torch.float16)
   right = right.to(torch.float16)

   B, N, _, H = left.shape

   # allocate output
   out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

   # compute strides
   stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
   stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
   stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

   # block sizes
   BLOCK_M = 64
   BLOCK_N = 64
   BLOCK_K = 32

   grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B*H)

   tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
   )

   # out_norm: LayerNorm over hidden_dim

   out = torch.nn.functional.linear(out_flat, weights['to_out.weight']).view(B, N, N, dim)

"""
TriMul “outgoing” implementation.

*   LayerNorm + linear projections + sigmoid gates are done in PyTorch.
*   The heavy 4‑D einsum  (... i k d, ... j k d -> ... i j d)  is rewritten as a
    batched matrix multiplication per hidden channel:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    which is equivalent to  left @ right.transpose(2,1)  for each d.
*   The N³·hidden work is performed by a custom Triton kernel that tiles the
    k‑dimension and accumulates in fp32 while the inputs are fp16.
*   The final LayerNorm, output gate and projection are done again in PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_bh = tl.program_id(2)                 # combined batch * hidden index
    pid_i  = tl.program_id(0)                 # block over i
    pid_j  = tl.program_id(1)                 # block over j

    # decode batch and hidden from flattened index
    b = pid_bh // H
    h = pid_bh % H

    # block coordinates
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ranges within the block
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds threads
    mask_i = i_range < N
    mask_j = j_range < N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k_base in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_base, BLOCK_K)

        k_range = k_base + tl.arange(0, BLOCK_K)

        # load left: shape (BLOCK_M, cur_k)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i_range[:, None] * stride_l_i
            + k_range[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (k_range[None, :] < N), other=0.0)

        # load right (transposed): shape (cur_k, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + k_range[:, None] * stride_r_k
            + j_range[None, :] * stride_r_j
        )
        right = tl.load(
            right_ptrs,
            mask=(k_range[:, None] < N) & mask_j[None, :],
            other=0.0,
        )

        # dot: (BLOCK_M, cur_k) @ (cur_k, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # store result
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i_range[:, None] * stride_o_i
        + j_range[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
            mask         : torch.Tensor  [B, N, N]        (bool / float)
            weights      : dict of torch.Tensors (float32)
            config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        out : torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1️⃣  LayerNorm on the last dimension (dim)
    # ----------------------------------------------------------------------
    eps = 1e-5
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]          # [B,N,N,dim]

    # ----------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)  →  hidden dimension
    # ----------------------------------------------------------------------
    # flatten the first three axes for a single matmul
    x_flat = x.view(-1, dim)                                      # [(B·N·N), dim]

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    # optional mask (broadcasted on hidden dim)
    if not nomask:
        m = mask.unsqueeze(-1).to(left.dtype)                     # [B,N,N,1]
        left = left * m
        right = right * m

    # ----------------------------------------------------------------------
    # 3️⃣  Gating (sigmoid) – keep in fp16 for speed
    # ----------------------------------------------------------------------
    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden).sigmoid()

    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ----------------------------------------------------------------------
    # 4️⃣  Core N³·hidden multiplication via Triton
    # ----------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        B * hidden,                       # combined batch‑*‑hidden dimension
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_j,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 5️⃣  Output LayerNorm + output gate
    # ----------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate.to(out.dtype)                # broadcast gate

    # ----------------------------------------------------------------------
    # 6️⃣  Final linear projection back to dim
    # ----------------------------------------------------------------------
    out_flat = out.view(-1, hidden)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
