"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each (batch, head) pair this is a matrix multiplication
    out_{d} = left_{d} @ right_{d}.T          (N×N)·(N×N) → (N×N)

The kernel computes many independent (B·hidden_dim) matrix multiplies on
the GPU using Triton.  All cheap operations (LayerNorm, Linear, gates, mask)
are kept in PyTorch.  The kernel works with float16 data (the model is
trained in float32) and the final result is cast back to float32.

The launch grid is (ceil(N/M), ceil(N/N), B*H) where each program instance
computes a TILE_M×TILE_N tile of the output for a fixed (batch, head) pair.
The inner‑k loop tiles over K with BLOCK_K.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# -------------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # dimensions
    stride_lb, stride_li, stride_lk, stride_ld, # left strides (b,i,k,d)
    stride_rb, stride_rj, stride_rk, stride_rd, # right strides (b,j,k,d)
    stride_ob, stride_oi, stride_oj, stride_od, # out strides (b,i,j,d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)                     # tile row (i)
    pid_n = tl.program_id(1)                     # tile col (j)
    pid_h = tl.program_id(2)                     # combined batch*head index

    # compute (batch, head) from pid_h
    b = pid_h // H
    d = pid_h % H

    # offsets for the tile
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundary conditions
    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- load left tile (M x K) -----
        # left[b, i, k, d]
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_offset[:, None] * stride_li
            + k_offset[None, :] * stride_lk
            + d * stride_ld
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)               # promote to fp32 for accumulation

        # ----- load right tile (K x N) -----
        # we need right transposed: right[b, k, j, d]
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + k_offset[:, None] * stride_rk
            + j_offset[None, :] * stride_rj
            + d * stride_rd
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float32)

        # matrix multiplication for this K‑tile
        acc += tl.dot(left, right)

    # ----- store output tile (M x N) -----
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_offset[:, None] * stride_oi
        + j_offset[None, :] * stride_oj
        + d * stride_od
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : Tensor[B, N, N, dim]  (float32)
            - mask  : Tensor[B, N, N]      (bool / float32) – may be omitted
            - weights: dict of all linear / norm parameters (float32)
            - config : {'dim': int, 'hidden_dim': int, 'nomask': bool}
    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (mask shape B,N,N)
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        m = mask.unsqueeze(-1).to(left.dtype)          # (B,N,N,1)
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton kernel: einsum over k
    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)            # same shape [B,N,N,H]

    # strides in element units
    s_lb, s_li, s_lk, s_ld = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rd = right_fp16.stride()
    s_ob, s_oi, s_oj, s_od = out_fp16.stride()

    # tile sizes – chosen to fit H200 SMs well
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * H,                            # (batch, head) pairs
    )

    _trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_rj,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_fp16.to(torch.float32)   # back to fp32

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
