"""
TriMul “outgoing” implementation (forward only).

The heavy part of the operator is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T .

We keep the LayerNorms/linear projections in PyTorch (they are cheap compared
to the O(N³) contraction) and fuse the element‑wise mask‑and‑gate multiplication
with a Triton kernel.  The kernel works on an arbitrary batch size, sequence
length and hidden dimension; it writes both masked‑left and masked‑right
tensors in a single pass to minimise memory traffic.

The final contraction is carried out with ``torch.einsum`` which internally
uses cuBLAS‑style kernels and is sufficiently fast for the test sizes.
All tensors are kept on the GPU and the final result is returned as float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: apply mask and per‑feature gate to both left and right tensors
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_in_ptr,   # [B,N,N,D]
    right_in_ptr,  # [B,N,N,D]
    mask_ptr,      # [B,N,N,1]  (broadcasted)
    left_gate_ptr, # [B,N,N,D]
    right_gate_ptr,# [B,N,N,D]
    left_out_ptr,  # [B,N,N,D]
    right_out_ptr, # [B,N,N,D]
    B, N, D,
    # strides (in elements) for each tensor
    stride_li0, stride_li1, stride_li2, stride_li3,
    stride_ri0, stride_ri1, stride_ri2, stride_ri3,
    stride_m0, stride_m1, stride_m2,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    # compile‑time block size
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    total = B * N * N * D
    mask = offs < total

    # ----- compute multi‑dimensional indices from linear offset -----
    b = offs // (N * N * D)
    rem = offs % (N * N * D)
    i = rem // (N * D)
    rem2 = rem % (N * D)
    j = rem2 // D
    d = rem2 % D

    # ----- load values ------------------------------------------------
    left  = tl.load(
        left_in_ptr
        + b * stride_li0 + i * stride_li1 + j * stride_li2 + d * stride_li3,
        mask=mask,
        other=0.0,
    )
    right = tl.load(
        right_in_ptr
        + b * stride_ri0 + i * stride_ri1 + j * stride_ri2 + d * stride_ri3,
        mask=mask,
        other=0.0,
    )
    # mask is scalar per (b,i,j), broadcasted over d
    m = tl.load(
        mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2,
        mask=mask,
        other=1.0,
    )
    left_gate = tl.load(
        left_gate_ptr
        + b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + d * stride_lg3,
        mask=mask,
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr
        + b * stride_rg0 + i * stride_rg1 + j * stride_rg2 + d * stride_rg3,
        mask=mask,
        other=0.0,
    )

    # ----- apply mask and gate -----------------------------------------
    left_out = left * m * left_gate
    right_out = right * m * right_gate

    # ----- store results ------------------------------------------------
    tl.store(
        left_out_ptr
        + b * stride_li0 + i * stride_li1 + j * stride_li2 + d * stride_li3,
        left_out,
        mask=mask,
    )
    tl.store(
        right_out_ptr
        + b * stride_ri0 + i * stride_ri1 + j * stride_ri2 + d * stride_ri3,
        right_out,
        mask=mask,
    )


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a fused Triton kernel
    for the mask‑gate multiplication.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]   (may be all‑ones)
        - weights      : dict of parameter tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor : [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4) Mask handling + fuse mask & gate with Triton
    # --------------------------------------------------------------
    # mask is either provided or, if nomask, we treat it as all‑ones.
    # In the nomask case we can skip the kernel entirely.
    if mask is None:
        left_masked  = left * left_gate
        right_masked = right * right_gate
    else:
        # expand mask to shape [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype)

        # allocate output tensors
        left_masked  = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # launch kernel
        total_elems = B * N * N * hidden_dim
        BLOCK = 65536  # 64 K threads per launch (tunable)
        grid = (triton.cdiv(total_elems, BLOCK),)

        mask_gate_kernel[grid](
            left.in_ptr(),
            right.in_ptr(),
            mask_exp.in_ptr(),
            left_gate.in_ptr(),
            right_gate.in_ptr(),
            left_masked.in_ptr(),
            right_masked.in_ptr(),
            B,
            N,
            hidden_dim,
            # strides (in elements)
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            BLOCK=BLOCK,
        )
        # sync is implicit at kernel return

    # --------------------------------------------------------------
    # 5) Core contraction: Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Equivalent to batched matmul per hidden dimension:
    #        out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T
    # --------------------------------------------------------------
    # We use torch.einsum which internally maps to efficient GEMM kernels.
    out = torch.einsum("b i k d, b j k d -> b i j d", left_masked, right_masked)

    # --------------------------------------------------------------
    # 6) Output LayerNorm (hidden_dim) + gating + final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(dtype)
