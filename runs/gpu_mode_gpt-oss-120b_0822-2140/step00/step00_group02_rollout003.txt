"""
TriMul “outgoing” forward pass using a small Triton kernel.

The heavy part of the operator – the element‑wise masking and gating of the
projected tensors – is fused into a Triton kernel.  The remaining
operations (LayerNorm, linear projections, the N³‑sum over the third index,
and the final linear) are performed with PyTorch (the einsum is executed in
bfloat16 for speed and cast back to float32).

All tensors are expected to reside on the CUDA device already.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel:  left = left * left_gate * mask
#                right = right * right_gate * mask
# All tensors are flattened to 1‑D, the kernel works on a single stride.
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_mask_gate_kernel(
    left_ptr, right_ptr,          # [total]  float32
    mask_ptr,                     # [total]  float32  (0.0 or 1.0)
    lgate_ptr, rgate_ptr,         # [total]  float32  (sigmoid outputs)
    total_elems,                  # scalar   int32
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load
    left  = tl.load(left_ptr  + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    msk   = tl.load(mask_ptr  + offs, mask=mask, other=1.0)
    lgate = tl.load(lgate_ptr + offs, mask=mask, other=1.0)
    rgate = tl.load(rgate_ptr + offs, mask=mask, other=1.0)

    # fuse
    left  = left  * lgate * msk
    right = right * rgate * msk

    # store back
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# --------------------------------------------------------------------------- #
# Entry point called by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask         : torch.Tensor  [B, N, N]       (float32, 0/1)
        weights      : dict of torch.Tensor
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)          # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Gates (linear + sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------- #
    # 4. Fuse mask and gates with a Triton kernel
    # ------------------------------------------------------------------- #
    # mask: [B,N,N] -> broadcast to hidden dim
    mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden)

    # flatten everything to 1‑D for the kernel
    total = left.numel()
    left_f  = left.contiguous().view(-1)
    right_f = right.contiguous().view(-1)
    mask_f  = mask_exp.contiguous().view(-1)
    lgate_f = left_gate.contiguous().view(-1)
    rgate_f = right_gate.contiguous().view(-1)

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    trimul_mask_gate_kernel[grid](
        left_f.data_ptr(),
        right_f.data_ptr(),
        mask_f.data_ptr(),
        lgate_f.data_ptr(),
        rgate_f.data_ptr(),
        total,
        BLOCK=BLOCK,
    )
    # reshape back
    left  = left_f.view(B, N, N, hidden)
    right = right_f.view(B, N, N, hidden)

    # ------------------------------------------------------------------- #
    # 5. The N³ reduction  out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    #    performed with an einsum in bfloat16 for speed.
    # ------------------------------------------------------------------- #
    left_b  = left.to(torch.bfloat16)
    right_b = right.to(torch.bfloat16)
    out = torch.einsum('b i k h, b j k h -> b i j h', left_b, right_b)
    out = out.to(torch.float32)                      # back to FP32

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate                              # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])    # [B,N,N,dim]

    return out
