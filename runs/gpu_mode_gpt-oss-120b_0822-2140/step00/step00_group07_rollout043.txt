"""
TriMul “outgoing” implementation.

The heavy N³ work is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension d this is exactly a matrix multiplication:
    L_d (N×N) @ R_dᵀ (N×N) → (N×N)

The kernel below performs a batched GEMM for every hidden channel.
All other operations (LayerNorm, linear projections, sigmoids, masks) are
handled in native PyTorch for simplicity.  The kernel is written for
Hopper (SM‑90) and uses FP32 accumulation; inputs are cast to FP16 for
higher throughput while the final result is kept in FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: batched matmul for each hidden channel
# ------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,           # [B, N, N, H]  (i,k,d) layout
    right_ptr,          # [B, N, N, H]  (j,k,d) layout – we read it transposed as (k,j)
    out_ptr,            # [B, N, N, H]  (i,j,d)

    B, N, H,            # problem sizes

    # strides of the input / output tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ rightᵀ for every hidden channel."""
    # ------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------
    bh_id   = tl.program_id(0)                # combined batch * hidden
    pid_m   = tl.program_id(1)                # block row   (i)
    pid_n   = tl.program_id(2)                # block col   (j)

    batch_id = bh_id // H
    hid_id   = bh_id % H

    # ------------------------------------------------------------
    # coordinate ranges for this block
    # ------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers that will be advanced in the K‑loop
    left_base  = left_ptr \
               + batch_id * stride_left_b \
               + hid_id   * stride_left_d \
               + offs_i[:, None] * stride_left_i

    right_base = right_ptr \
               + batch_id * stride_left_b \
               + hid_id   * stride_left_d \
               + offs_j[None, :] * stride_left_i   # j uses the original “i” stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left block   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # rightᵀ block shape (BLOCK_K, BLOCK_N)
        # we read right as (k, j) so that a dot product gives (i, j)
        right_ptrs = right_base + k_offset[:, None] * stride_left_k
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # GEMM for this K‑tile
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------
    # write back
    # ------------------------------------------------------------
    out_ptrs = out_ptr \
             + batch_id * stride_out_b \
             + hid_id   * stride_out_d \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Helper: Python‑side launch of the kernel
# ------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right:  [B, N, N, H]  (float16 or float32)
    returns out:   [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (in elements)
    s_b, s_i, s_k, s_h = left.stride()
    # output strides are the same layout
    s_ob, s_oi, s_oj, s_oh = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # fits well in registers on Hopper

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_b, s_i, s_k, s_h,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ------------------------------------------------------------
# Main entry point required by the judge
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a custom Triton kernel
    for the N³ reduction.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim] (float32)
            mask   – torch.Tensor [B, N, N] (bool/float) – may be all‑ones
            weights – dict of pre‑loaded model parameters (float32)
            config  – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = x.device

    # -----------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left  = F.linear(x, weights["left_proj.weight"])          # (hidden_dim, dim)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Optional mask (mask may be missing, the caller passes a tensor of ones)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating values (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (still accumulated in fp32)
    # -----------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # -----------------------------------------------------------------
    # N³ reduction via Triton
    # -----------------------------------------------------------------
    out = trimul_gemm(left_fp16, right_fp16)   # [B, N, N, hidden_dim] float32

    # -----------------------------------------------------------------
    # Final LayerNorm → out_gate → final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # broadcast out_gate over hidden_dim
    out = out * out_gate.unsqueeze(-1)

    # Linear back to original dimension
    output = F.linear(out, weights["to_out.weight"])   # (dim, hidden_dim)

    return output
