"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension *d*.
We keep the cheap LayerNorm / Linear / gating logic in PyTorch and
accelerate the N×N matmul using a Triton kernel that works on one
hidden channel at a time (hidden_dim ≤ 128, so the extra Python loop is cheap).

The kernel computes a tiled GEMM for each (batch, channel) pair:
    left  : (B, N, K)   –   (i,k)
    right : (B, N, K)   –   (j,k)   (we load the *k* dimension first so that we can dot‑product)
    out   : (B, N, N)   –   (i,j)

The result is finally passed through a LayerNorm, a gating scalar and a
output linear layer – all in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for a single hidden channel
# ----------------------------------------------------------------------
@triton.jit
def _matmul_slice_kernel(
    left_ptr,                     # *B*N*K  (i,k)
    right_ptr,                    # *B*N*K  (j,k)   we will read it transposed
    out_ptr,                      # *B*N*N  (i,j)
    B, N,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_j, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute  out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    for a **single** hidden channel (the caller advances the hidden‑dim pointer).
    """
    pid = tl.program_id(0)

    # number of tiles per matrix
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # decode batch, tile_i, tile_j from linear pid
    b   = pid // (num_i * num_j)
    tid = pid % (num_i * num_j)
    tile_i = tid // num_j
    tile_j = tid %  num_j

    # offsets for the current tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k]   (M,K)
        ptrs_left = left_ptr + \
                    b * stride_left_b + \
                    offs_i[:, None] * stride_left_i + \
                    offs_k[None, :] * stride_left_k
        a = tl.load(ptrs_left,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[b,j,k]  we load it transposed -> (K,N)
        ptrs_right = right_ptr + \
                     b * stride_right_b + \
                     offs_k[:, None] * stride_right_k + \
                     offs_j[None, :] * stride_right_j
        b_mat = tl.load(ptrs_right,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # a (M,K) @ b_mat (K,N) -> (M,N)
        acc += tl.dot(a, b_mat)

    # write result
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper for a single hidden channel
# ----------------------------------------------------------------------
def _matmul_slice(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor,
                  BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    """
    left/right/out : contiguous torch tensors of shape (B,N,N) on CUDA.
    The kernel is launched with one program per (batch, tile_i, tile_j).
    """
    B, N, _ = left.shape
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (B *
            triton.cdiv(N, BLOCK_M) *
            triton.cdiv(N, BLOCK_N),)

    _matmul_slice_kernel[grid](
        left,
        right,
        out,
        B, N,
        stride_left[0], stride_left[1], stride_left[2],
        stride_right[0], stride_right[1], stride_right[2],
        stride_out[0], stride_out[1], stride_out[2],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32 or bf16)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float)
        - weights_dict : mapping of weight/bias tensors
        - config_dict  : {"dim":int, "hidden_dim":int}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (bias & weight provided)
    # ------------------------------------------------------------------
    # input shape: (B,N,N,dim)
    x = input_tensor
    # flatten the last two dims for torch.layer_norm API
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    # Shape after linear: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)                     # (B,N,N,1)
        left = left * mask_unsq
        right = right * mask_unsq

    # gates (sigmoid)
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Batched N×N matmul per hidden channel using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)          # (B,N,N,hidden_dim), float32

    # make sure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # stride of the hidden‑dim dimension (last dim)
    stride_h = left.stride()[-1]

    # iterate over hidden channels (max 128 → cheap Python loop)
    for d in range(hidden_dim):
        # views of shape (B,N,N)
        left_d = left[..., d].contiguous()
        right_d = right[..., d].contiguous()
        out_d = out[..., d].contiguous()

        _matmul_slice(left_d, right_d, out_d,
                      BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # ------------------------------------------------------------------
    # 4. Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                       # gating (broadcastable)

    # linear back to original dim
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
