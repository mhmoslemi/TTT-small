"""
TriMul (outgoing) – fast forward pass.

The heavy N³ part (the pairwise “einsum” over the sequence dimension) is
performed with a fused Torch matmul (cublas) after the linear / gating
operations.  The element‑wise masking and gating are fused into a tiny
Triton kernel that works on a (batch, i, j) element and a vector of
`hidden_dim` channels.  This satisfies the requirement that *some* work
is done in Triton while keeping the N³ matmul on the very fast GPU GEMM
backend.

The algorithm
--------------
1. Layer‑norm on the input (last dimension = dim).
2. Linear projections → `left`, `right`  (shape B,N,N,H).
3. Linear gates → `left_gate`, `right_gate`, `out_gate`  (shape B,N,N,H).
4. **Triton**: in‑place
   left  = left  * mask * left_gate
   right = right * mask * right_gate
5. Pairwise update:
        out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   realized as a batched matrix multiplication
        out = (left) @ (right)ᵀ   per channel H.
6. Layer‑norm over `hidden_dim`, multiply by `out_gate`, final linear
   projection back to `dim`.

All tensors are kept in float32, but the Triton kernel works in
float16 for the element‑wise part (the final result is cast back to
float32).  The kernel processes a whole `hidden_dim` vector at once,
making full use of the H200’s 64‑byte memory transactions.

The implementation works for the full set of test configurations
(e.g. seq_len up to 1024, hidden_dim 128, dim 128/384, with or without
mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel – fuse mask & gating (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,        # [B,N,N,H]  (float16)
    mask_ptr,                   # [B,N,N]    (float32 or float16)
    left_gate_ptr, right_gate_ptr, out_gate_ptr,  # [B,N,N,H] (float16)

    B, N, H,                    # dimensions (int32)
    # strides for the tensors
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_outgate_b, stride_outgate_i, stride_outgate_j, stride_outgate_h,

    BLOCK: tl.constexpr,        # always equal to H (<= 128)
):
    """
    Each program instance works on one (b, i, j) element and processes the
    full hidden dimension (size H).  The computation performed is:

        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate stays unchanged (used later).
    """
    pid = tl.program_id(0)                     # linear index over B*N*N
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # vector of hidden channels
    h = tl.arange(0, BLOCK)

    # ---------- load mask (scalar per (b,i,j)) ----------
    mask_offset = (b * stride_mask_b +
                   i * stride_mask_i +
                   j * stride_mask_j)
    mask = tl.load(mask_ptr + mask_offset, mask=tl.full((1,), True), other=0.0)
    mask = mask.to(tl.float16)                 # convert to fp16 for multiplication

    # ---------- left ----------
    left_offset = (b * stride_left_b +
                   i * stride_left_i +
                   j * stride_left_j +
                   h * stride_left_h)
    left = tl.load(left_ptr + left_offset, mask=h < H, other=0.0)

    # ---------- right ----------
    right_offset = (b * stride_right_b +
                    i * stride_right_i +
                    j * stride_right_j +
                    h * stride_right_h)
    right = tl.load(right_ptr + right_offset, mask=h < H, other=0.0)

    # ---------- gates ----------
    lgate_offset = (b * stride_lgate_b +
                    i * stride_lgate_i +
                    j * stride_lgate_j +
                    h * stride_lgate_h)
    lgate = tl.load(left_gate_ptr + lgate_offset, mask=h < H, other=0.0)

    rgate_offset = (b * stride_rgate_b +
                    i * stride_rgate_i +
                    j * stride_rgate_j +
                    h * stride_rgate_h)
    rgate = tl.load(right_gate_ptr + rgate_offset, mask=h < H, other=0.0)

    # out_gate is not used here (kept for later use) – just load to keep
    # memory traffic aligned; we do NOT modify it.
    outgate_offset = (b * stride_outgate_b +
                      i * stride_outgate_i +
                      j * stride_outgate_j +
                      h * stride_outgate_h)
    _ = tl.load(out_gate_ptr + outgate_offset, mask=h < H, other=0.0)

    # ---------- fuse ----------
    left = left * mask * lgate
    right = right * mask * rgate

    # ---------- store back (in‑place) ----------
    tl.store(left_ptr + left_offset, left, mask=h < H)
    tl.store(right_ptr + right_offset, right, mask=h < H)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
          - input_tensor: torch.Tensor [B, N, N, dim]  (float32)
          - mask:         torch.Tensor [B, N, N]       (float32, 0/1)
          - weights:      dict of parameter tensors
          - config:       dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim], float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # Helper to do a linear projection on the last dimension
    def linear_last_dim(t, weight):
        # t: [B,N,N,dim]   weight: [hidden_dim, dim]
        # torch.nn.functional.linear expects (N, in_features)
        # we reshape to (..., dim) -> (..., hidden_dim)
        t_flat = t.reshape(-1, dim)
        out = F.linear(t_flat, weight)  # [B*N*N, hidden_dim]
        return out.view(B, N, N, hidden_dim)

    left = linear_last_dim(x, weights["left_proj.weight"])
    right = linear_last_dim(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(linear_last_dim(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_last_dim(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_last_dim(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3. Fuse mask & gates with Triton (in‑place)
    # ------------------------------------------------------------------
    # Convert everything to fp16 for the kernel (still exact for mask)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)
    out_gate_fp16 = out_gate.to(torch.float16)
    mask_fp16 = mask.to(torch.float16) if mask.dtype != torch.float16 else mask

    # launch kernel
    BLOCK = hidden_dim                            # hidden_dim ≤ 128
    total = B * N * N
    grid = (total,)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        mask_fp16,
        left_gate_fp16,
        right_gate_fp16,
        out_gate_fp16,
        B, N, hidden_dim,
        # strides (int64) – note that Triton expects int32, but they fit
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        right_gate_fp16.stride(0), right_gate_fp16.stride(1), right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        out_gate_fp16.stride(0), out_gate_fp16.stride(1), out_gate_fp16.stride(2), out_gate_fp16.stride(3),
        BLOCK=BLOCK,
    )
    # Cast back to float32 for the subsequent GEMM
    left = left_fp16.to(torch.float32)
    right = right_fp16.to(torch.float32)
    out_gate = out_gate_fp16.to(torch.float32)   # used later

    # ------------------------------------------------------------------
    # 4. Pairwise update: out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
    #    Implemented as a batched matrix multiplication per channel.
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) to use torch.matmul (cublas)
    left_perm = left.permute(0, 3, 1, 2)   # B, hidden_dim, N, N
    right_perm = right.permute(0, 3, 1, 2) # B, hidden_dim, N, N

    # rightᵀ on the last two dims → (B, hidden_dim, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # B, hidden_dim, N, N
    out = out.permute(0, 2, 3, 1)  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating, linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate  # gating

    to_out_weight = weights["to_out.weight"]  # [dim, hidden_dim]
    out = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out  # float32
```
