"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched matrix‑multiply
kernel written in Triton.

The heavy part of the module is the einsum:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden dimension *d* this is exactly a matrix multiplication
    (left_d) @ (right_d)ᵀ   with shapes (N, N) @ (N, N).

We therefore reshape the tensors to a batch of size (B·H) and launch a
single Triton kernel that computes many independent (M×K)·(K×N) matmuls.
All cheap operations (LayerNorm, Linear, Sigmoid) stay in PyTorch.

The kernel works in fp16 for speed and the result is cast back to fp32
before the final LayerNorm, gating and output projection.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched matmul kernel
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ B   (A,B,C are fp16, accumulation in fp32)"""
    pid_batch = tl.program_id(0)   # which (B·H) matrix
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]

    # compute base pointers for this batch
    a_ptr = A_ptr + pid_batch * stride_Ab
    b_ptr = B_ptr + pid_batch * stride_Bb
    c_ptr = C_ptr + pid_batch * stride_Cb

    # pointers to the start of the block
    a_tile_ptr = a_ptr + (offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak)
    b_tile_ptr = b_ptr + (offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_tile_ptr, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_tile_ptr += BLOCK_K * stride_Ak
        b_tile_ptr += BLOCK_K * stride_Bk

    # write back (store as fp16 to save bandwidth)
    c = acc.to(tl.float16)
    c_tile_ptr = c_ptr + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_tile_ptr, c, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑fused
    batched matmul for the N³ interaction term.

    Args
    ----
    data: (input, mask, weights, config)
        input  – Tensor[B, N, N, dim] (float32)
        mask   – Tensor[B, N, N] (float32, 0 or 1) – may be all‑ones when nomask=True
        weights – dict with the LayerNorm / Linear parameters (float32)
        config  – dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5                     # LayerNorm epsilon – matches torch default

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matmul over the hidden dimension (the N³ part)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B*H, N, N)   and transpose right for the required order
    left_ = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) -> (k, j)

    batch_h = B * hidden_dim
    left_flat = left_.view(batch_h, N, N)
    right_flat = right_.view(batch_h, N, N)

    # cast to fp16 for the kernel (still keep a float32 copy for safety)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # allocate output
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_M = triton.cdiv(N, BLOCK_M)
    num_blocks_N = triton.cdiv(N, BLOCK_N)

    grid = (batch_h, num_blocks_M, num_blocks_N)

    # strides (in elements) for the three‑dim tensors
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Bk, stride_Bn = right_fp16.stride()
    stride_Cb, stride_Cm, stride_Cn = out_fp16.stride()

    batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim) and up‑cast to fp32
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
