"""
TriMul “outgoing” implementation.
- LayerNorms and linear projections are done with torch (FP16 for speed).
- The heavy 4‑D einsum
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is reduced to a batched GEMM over the sequence dimension.
  For each channel *d* we compute a matrix product
      (N×N) @ (N×N)^T
  using a custom Triton kernel that streams tiles of size 128×128×64.
- The kernel works on FP16 inputs, accumulates in FP32 and writes FP16.
  The final output is cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM A @ B^T  (A,B : [batch, M, K])
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent batches."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers with batch offset
    a_batch = A_ptr + pid_b * stride_batch_a
    b_batch = B_ptr + pid_b * stride_batch_b
    c_batch = C_ptr + pid_b * stride_batch_c

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile  (M x K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B tile  (N x K)  – note we need Bᵀ, so we load (K x N) and treat it as transposed
        b = tl.load(
            b_batch + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # (M,K)·(K,N) → (M,N)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back FP16 (store as FP16 to save bandwidth, later cast to FP32)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_batch + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]        (float32)
            - mask:        [B, N, N]               (float32 or bool)
            - weights: dict with all linear / LN parameters (float32)
            - config:  dict with keys
                * dim, hidden_dim, nomask (bool)
    Returns:
        out: [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)                     # work in FP16 from here

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) -------------------------------------------------
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) ---------------------------------------------------------
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) -------------------------------------------------------------
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM (A @ Bᵀ)
    #    left:  [B, N, N, H] -> (B*H, N, N)
    #    right: [B, N, N, H] -> transpose last two dims then (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_tile = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (batch, M, K)
    right_tile = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)       # (batch, K, N) already transposed

    # Output placeholder
    out_tile = torch.empty_like(left_tile)   # FP16

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H,                             # batch
    )

    batched_gemm_kernel[grid](
        left_tile,
        right_tile,
        out_tile,
        N, N, N,
        left_tile.stride(0), left_tile.stride(1), left_tile.stride(2),
        right_tile.stride(0), right_tile.stride(1), right_tile.stride(2),
        out_tile.stride(0), out_tile.stride(1), out_tile.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # ensure fp16 kernel (default based on tensor dtype)
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, H] (still FP16)
    # ------------------------------------------------------------------
    out = out_tile.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Final LayerNorm, gating and projection back to dim -----------------------
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gate (FP16)

    # Project back to original dimension
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.to(torch.float32)             # required output dtype
