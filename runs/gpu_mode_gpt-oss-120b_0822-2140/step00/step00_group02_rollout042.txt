"""
tri_mul_triton.py

Highly‑optimized “outgoing” TriMul (Triangle Multiplicative Update) forward
pass for AlphaFold‑3 style models.

The kernel implements the cubic part

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication of `left` with the *transpose* of
`right` for every hidden channel `d`.  All other operations
(LayerNorms, linear projections, sigmoids and mask handling) are performed in
regular PyTorch – they are cheap compared to the N³ work and keep the code
readable.  The kernel processes a single hidden channel per program‑id on the
second axis, so the compute is a classic GEMM (M × K) @ (K × N) with
accumulation in FP32 and stores the result back in FP16 before the final
linear layer.

The launch grid is:
    grid = (B * ceil(N/BLOCK_M) * ceil(N/BLOCK_N), hidden_dim)

where `B` is the batch size (≤2), `N` the sequence length (≤1024) and
`hidden_dim` (≤384).  The kernel is written for Triton 3.3.1 and
targets NVIDIA H200 (SM‑90) – it uses 128‑bit loads/store and 32‑bit
accumulators for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   for a *single* hidden channel `d`
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float16* left  [B, N, N, H]
    right_ptr,              # *float16* right [B, N, N, H]
    out_ptr,                # *float16* out   [B, N, N, H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile size for output i dimension
    BLOCK_N: tl.constexpr,  # tile size for output j dimension
    BLOCK_K: tl.constexpr   # tile size for reduction dimension k
):
    # --------------------------------------------------------------
    # 1) decode program ids
    # --------------------------------------------------------------
    pid = tl.program_id(0)            # combines batch and (i,j) tiles
    d   = tl.program_id(1)            # hidden channel index

    # number of tiles along i and j
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # decode batch, tile_i, tile_j
    batch = pid // (num_i * num_j)
    tid   = pid % (num_i * num_j)
    tile_i = tid // num_j
    tile_j = tid % num_j

    # --------------------------------------------------------------
    # 2) pointers + bounds
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the output tile (M × N) – FP32 for safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) main k‑loop: accumulate left[i,k] * right[j,k]  (right transposed)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile : shape (M, K) -----------------------
        # address: batch * stride_l_b + i*stride_l_i + k*stride_l_k + d*stride_l_h
        left_ptrs = (
            left_ptr
            + batch * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + d * stride_l_h
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ----- load right tile transposed : shape (K, N) ------------
        # we want rightᵀ, i.e. access element (k, j) of original `right`.
        # address: batch * stride_r_b + j*stride_r_j + k*stride_r_k + d*stride_r_h
        right_ptrs = (
            right_ptr
            + batch * stride_r_b
            + offs_j[None, :] * stride_r_j   # j dimension becomes column
            + offs_k[:, None] * stride_r_k   # k dimension is row
            + d * stride_r_h
        )
        right_T = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)

        # ----- dot‑product over the K dimension --------------------
        # tl.dot(A, B) where A: (M, K), B: (K, N) → (M, N)
        acc += tl.dot(left, right_T).to(tl.float32)

    # --------------------------------------------------------------
    # 4) write back – cast to FP16 (the rest of the model expects this)
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + d * stride_o_h
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator, using a Triton kernel for
    the O(N³) matrix‑multiply‑like step.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool / float)
        - weights      : dict of torch.Tensor parameters
        - config       : dict containing `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1) unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 2) LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    input_ln = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 3) Linear projections (no bias) – we keep them in fp16
    # --------------------------------------------------------------
    # helper: linear without bias
    def linear_no_bias(x, w):
        # x : [B,N,N,dim] , w : [hidden_dim, dim] (as in nn.Linear)
        # torch linear expects (..., in_features) @ weight.T
        return torch.einsum('... d, hd -> ... h', x, w)

    left  = linear_no_bias(input_ln, weights["left_proj.weight"]).to(torch.float16)
    right = linear_no_bias(input_ln, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------
    # 4) Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5) Gates (sigmoid) – stay in fp16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(input_ln, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(linear_no_bias(input_ln, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(linear_no_bias(input_ln, weights["out_gate.weight"])).to(torch.float16)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Triton kernel: out = left @ rightᵀ   (shape B,N,N,hidden_dim)
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements, not bytes) for the 4‑D tensors
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    # launch parameters – choose tile sizes that fit H200 SM‐90 well
    BLOCK_M = 64   # tile rows
    BLOCK_N = 64   # tile cols
    BLOCK_K = 32   # reduction tile

    grid = (
        triton.cdiv(B * N, BLOCK_M) * triton.cdiv(N, BLOCK_N),  # program_id(0)
        hidden_dim,                                            # program_id(1)
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # H200 has many cores – 4 warps works well for these shapes
    )
    # --------------------------------------------------------------
    # 7) to_out_norm -> out_gate -> final linear
    # --------------------------------------------------------------
    # to_out_norm operates on hidden_dim
    out = out.to(torch.float32)                     # back to fp32 for LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)

    # final linear: hidden_dim -> dim  (no bias)
    out = torch.einsum('... h, dh -> ... d', out, weights["to_out.weight"])
    return out
