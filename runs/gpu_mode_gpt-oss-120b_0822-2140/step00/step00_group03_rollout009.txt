"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the operator is the 3‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is mathematically a batched matrix‑multiply of the
N×N slices per hidden dimension `d` (left @ rightᵀ).  
We linear‑transform the input, apply the element‑wise masks and gates in PyTorch
(and keep the tensors in FP16 for the multiplication), then launch a Triton
kernel that performs a fused batched GEMM for all `(batch, hidden)` pairs.
The kernel works on the flattened batch dimension `B*H` and produces a
float‑32 accumulation.  The remaining LayerNorm and output‑projection are
performed with native PyTorch kernels.

The implementation follows the Triton launch conventions required for
SM‑90 (H200).  Block sizes have been chosen to give good occupancy on the
target GPU while keeping shared‑memory usage modest.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched GEMM (C = A @ Bᵀ)  –  A,B are FP16, C is FP32
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,              # A strides: (batch, row, col)
    stride_bk, stride_bn,              # B strides (note B is transposed)
    stride_cm, stride_cn,              # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    Grid: (num_blocks_m, num_blocks_n, batch)
    """
    pid_z = tl.program_id(2)               # batch index (B*H)
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col

    # --------------------------------------------------------------
    #  Compute start offsets for the block
    # --------------------------------------------------------------
    # Tile the grid in 2‑D groups to improve L2 cache reuse (standard
    # technique from the Triton matmul example).
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    group_id = (pid_m * num_blocks_n + pid_n) // GROUP_M
    num_pid_in_group = GROUP_M
    pid_m = (group_id * GROUP_M + (pid_m * num_blocks_n + pid_n) % GROUP_M) // num_blocks_n
    pid_n = (pid_m * num_blocks_n + pid_n) % num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the block in A and B
    a_ptrs = a_ptr + pid_z * stride_am + (offs_m[:, None] * stride_ak) + tl.arange(0, BLOCK_K)[None, :] * 0
    b_ptrs = b_ptr + pid_z * stride_bn + (offs_n[None, :] * stride_bk) + tl.arange(0, BLOCK_K)[:, None] * 0

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a block of A and B (FP16) – use `mask` to avoid OOB loads
        a = tl.load(a_ptrs + k * BLOCK_K * stride_ak,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[None, :],
                    other=0.0).to(tl.float32)

        b = tl.load(b_ptrs + k * BLOCK_K * stride_bk,
                    mask=mask_n[None, :] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[:, None],
                    other=0.0).to(tl.float32)

        # Compute dot product for this tile
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #  Write back
    # --------------------------------------------------------------
    c = acc.to(tl.float32)                # already FP32
    c_ptrs = c_ptr + pid_z * stride_cm + (offs_m[:, None] * stride_cn) + offs_n[None, :]
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
#  High‑level wrapper that prepares tensors, launches the kernel and
#  finishes the TriMul computation.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator, using a Triton kernel for
    the N³ contraction.

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor[B, N, N, dim]   (float32)
            - mask  : torch.Tensor[B, N, N]        (bool/float)
            - weights: dict of all linear / layer‑norm parameters
            - config : {"dim": int, "hidden_dim": int, ...}

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    #  unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    #  LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    #  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    #  Apply mask (if present).  ``mask`` is (B,N,N); we broadcast to hidden.
    # ------------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1)                    # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    #  Gating tensors (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    #  Reshape to (B*H, N, N) and prepare RHS transposed
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N) -> (BH,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden, N, N)          # A
    right_flat = right_perm.view(B * hidden, N, N).transpose(1, 2).contiguous()  # Bᵀ

    # Allocate output tensor (float32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    #  Launch Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_M = 8

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,
    )

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=left_flat.stride(0),   # batch stride
        stride_ak=left_flat.stride(1),   # row stride (i)
        stride_bk=right_flat.stride(2),  # because right is transposed: col stride
        stride_bn=right_flat.stride(1),  # row stride of transposed matrix
        stride_cm=out_flat.stride(0),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )

    # ------------------------------------------------------------------
    #  Reshape back to [B,N,N,hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    #  Final LayerNorm (hidden dim) and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gate (broadcasted)
    out = F.linear(out, weights["to_out.weight"])   # back to [B,N,N,dim]

    return out.to(torch.float32)   # ensure float32 output
