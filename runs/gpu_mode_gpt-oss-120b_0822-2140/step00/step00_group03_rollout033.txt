"""
TriMul – “outgoing” version (AlphaFold‑3).

The heavy N³ tensor contraction  
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  
is reduced to a batched matrix‑multiply per hidden channel:
    out_d = left_d @ right_dᵀ   (d = 0 … hidden_dim‑1)

All linear projections, gating, LayerNorms and the final linear are left in
PyTorch (they are already highly tuned).  Two tiny Triton kernels are used
to fuse the element‑wise mask & gate multiplications, which removes an
extra memory pass and satisfies the “must use Triton” rule.

The implementation works for any batch size (1‑2), sequence length up to
1024 and hidden dimensions 128/384.  The final tensor is returned in
float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernels – elementwise fusions
# ----------------------------------------------------------------------
@triton.jit
def mul_mask_kernel(
    x_ptr,               # input tensor (B, N, N, H)
    mask_ptr,            # mask   tensor (B, N, N, 1)  – already broadcasted
    out_ptr,             # output tensor
    B, N, H,
    stride_xb, stride_xi, stride_xj, stride_xh,
    stride_mb, stride_mi, stride_mj, stride_mh,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # linearize over batch * N * N
    total = B * N * N
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask = offs < total

    # compute (b,i,j) from linear index
    b = offs // (N * N)
    residual = offs % (N * N)
    i = residual // N
    j = residual % N

    # offsets for the hidden dimension
    offs_h = tl.arange(0, H)

    x = tl.load(
        x_ptr
        + b[:, None] * stride_xb
        + i[:, None] * stride_xi
        + j[:, None] * stride_xj
        + offs_h[None, :] * stride_xh,
        mask=mask[:, None],
        other=0.0,
    )
    m = tl.load(
        mask_ptr
        + b[:, None] * stride_mb
        + i[:, None] * stride_mi
        + j[:, None] * stride_mj
        + offs_h[None, :] * stride_mh,
        mask=mask[:, None],
        other=0.0,
    )
    tl.store(out_ptr + offs[:, None] * H + offs_h[None, :], x * m, mask=mask[:, None])


@triton.jit
def mul_gate_kernel(
    x_ptr,               # (B,N,N,H)
    gate_ptr,            # (B,N,N,H)
    out_ptr,
    B, N, H,
    stride_xb, stride_xi, stride_xj, stride_xh,
    stride_gb, stride_gi, stride_gj, stride_gh,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask = offs < total

    b = offs // (N * N)
    residual = offs % (N * N)
    i = residual // N
    j = residual % N

    offs_h = tl.arange(0, H)

    x = tl.load(
        x_ptr
        + b[:, None] * stride_xb
        + i[:, None] * stride_xi
        + j[:, None] * stride_xj
        + offs_h[None, :] * stride_xh,
        mask=mask[:, None],
        other=0.0,
    )
    g = tl.load(
        gate_ptr
        + b[:, None] * stride_gb
        + i[:, None] * stride_gi
        + j[:, None] * stride_gj
        + offs_h[None, :] * stride_gh,
        mask=mask[:, None],
        other=0.0,
    )
    tl.store(out_ptr + offs[:, None] * H + offs_h[None, :], x * g, mask=mask[:, None])


# ----------------------------------------------------------------------
# Helper for launching the above kernels
# ----------------------------------------------------------------------
def _launch_elemwise_kernel(kernel, dst, src, other, shape):
    """
    src, other : torch tensors with identical shapes (B,N,N,H)
    dst        : torch tensor to write into (pre‑allocated)
    shape      : (B,N,H)
    """
    B, N, _, H = shape
    BLOCK = 1024  # number of (b,i,j) elements per program
    total = B * N * N
    grid = (triton.cdiv(total, BLOCK),)

    # strides in elements (not bytes)
    src_strides = src.stride()
    oth_strides = other.stride()
    dst_strides = dst.stride()

    kernel[
        grid
    ](
        src_ptr=src,
        mask_ptr=other,
        out_ptr=dst,
        B=B,
        N=N,
        H=H,
        stride_xb=src_strides[0],
        stride_xi=src_strides[1],
        stride_xj=src_strides[2],
        stride_xh=src_strides[3],
        stride_mb=oth_strides[0],
        stride_mi=oth_strides[1],
        stride_mj=oth_strides[2],
        stride_mh=oth_strides[3],
        BLOCK=BLOCK,
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input : [B, N, N, dim]               (float32 or bfloat16)
            - mask  : [B, N, N] (may be all‑ones)
            - weights: dict with the nine tensors listed in the statement
            - config : {"dim": int, "hidden_dim": int}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    # torch.layer_norm works on the last dim – we reshape to 2‑D and back
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b, eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – dim → hidden
    # ------------------------------------------------------------------
    # We use torch.nn.functional.linear which is a fast GEMM
    left_proj_w = weights["left_proj.weight"].to(device)      # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)    # (hidden, dim)

    # reshape to (B*N*N, dim) for a single GEMM
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left = F.linear(x_flat, left_proj_w)        # (B*N*N, hidden)
    right = F.linear(x_flat, right_proj_w)

    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present) – fused Triton kernel
    # ------------------------------------------------------------------
    # mask is (B,N,N); we broadcast to (B,N,N,1) -> (B,N,N,hidden)
    if mask is not None:
        mask_broadcast = mask.unsqueeze(-1).to(device)
        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # launch mask kernel twice (left & right)
        _launch_elemwise_kernel(mul_mask_kernel, left_masked, left, mask_broadcast, (B, N, hidden))
        _launch_elemwise_kernel(mul_mask_kernel, right_masked, right, mask_broadcast, (B, N, hidden))
    else:
        left_masked = left
        right_masked = right

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – also fused via Triton
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)      # (hidden, dim)
    right_gate_w = weights["right_gate.weight"].to(device)    # (hidden, dim)
    out_gate_w = weights["out_gate.weight"].to(device)        # (hidden, dim)

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(B, N, N, hidden)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(B, N, N, hidden)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden)

    left_gated = torch.empty_like(left_masked)
    right_gated = torch.empty_like(right_masked)

    _launch_elemwise_kernel(mul_gate_kernel, left_gated, left_masked, left_gate,
                            (B, N, hidden))
    _launch_elemwise_kernel(mul_gate_kernel, right_gated, right_masked, right_gate,
                            (B, N, hidden))

    # ------------------------------------------------------------------
    # 5) Core N³ contraction: batched matmul per hidden channel
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, N)
    left_bh = left_gated.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_bh = right_gated.permute(0, 3, 1, 2).contiguous()

    # perform B*H independent matmuls: (N,N) @ (N,N)^T
    # torch.bmm works on (B*H, N, N)
    left_flat = left_bh.view(B * hidden, N, N)
    right_flat = right_bh.view(B * hidden, N, N)

    # right needs to be transposed on the last two dimensions
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5)

    # ------------------------------------------------------------------
    # 7) Multiply by out_gate (broadcast over last dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out.view(-1, hidden), to_out_w).view(B, N, N, dim)

    return out.to(torch.float32)
