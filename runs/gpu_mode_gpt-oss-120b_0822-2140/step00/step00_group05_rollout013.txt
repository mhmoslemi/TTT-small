"""
TriMul “outgoing” kernel (AlphaFold3) – fast N^3 reduction.

The heavy part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is exactly a batched matrix multiplication
    left[b,:, :, d] @ right[b,:, :, d].T .
We implement this reduction in Triton (FP16 inputs, FP32 accumulation) and keep
the surrounding LayerNorm / Linear / gating logic in PyTorch for clarity.
The kernel works for any batch size (1‑2) and any hidden dimension (`hidden_dim`).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel:  out = left @ right.T  (per batch * hidden_dim slice)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16  [B,N,N,H]
    right_ptr,         # *fp16  [B,N,N,H]
    out_ptr,           # *fp32  [B,N,N,H]
    B, N, H,
    stride_left_B, stride_left_i, stride_left_k, stride_left_d,
    stride_right_B, stride_right_i, stride_right_k, stride_right_d,
    stride_out_B, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    ########################################
    #  program ids   (i block, j block, batch*head)
    ########################################
    pid_m = tl.program_id(0)          # i‑block
    pid_n = tl.program_id(1)          # j‑block
    pid_h = tl.program_id(2)          # combined (batch, hidden)

    # decode combined id → (batch, hidden)
    b = pid_h // H
    d = pid_h % H

    # base pointer for this (b,d) slice
    left_base  = left_ptr  + b * stride_left_B  + d * stride_left_d
    right_base = right_ptr + b * stride_right_B + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_B   + d * stride_out_d

    # offset of the first element of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds for i / j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]   (M x K)
        left_ptrs = left_base + \
            (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right_T[k, j] (K x N) – we load right as (k, j) so that the
        # second dimension lines up with j and we can use tl.dot directly.
        right_ptrs = right_base + \
            (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # matrix‑multiply‑accumulate
        acc += tl.dot(left_tile, right_tile)

    # write back the result (FP32)
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Heavy N‑dimensional reduction is performed by `tri_mul_kernel`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input / output channel size
    hidden_dim = config["hidden_dim"] # inner channel size (c_z)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    # left / right projections
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # optional mask handling
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp
    else:
        # mask may be all‑ones anyway; no work needed
        pass

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and element‑wise modulation
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core N^3 reduction via Triton (FP16 inputs, FP32 out)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to fp16 for the kernel (better bandwidth)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32).contiguous()

    # strides (in elements)
    stride_left = left_fp16.stride()   # (B, N, N, H)
    stride_right = right_fp16.stride()
    stride_out = out_fp32.stride()

    # tile sizes – tuned for Hopper (SM90) with 128‑bit loads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        hidden_dim,
        stride_left[0],  # batch stride
        stride_left[1],  # i‑stride (second dimension)
        stride_left[2],  # k‑stride (third dimension)
        stride_left[3],  # d‑stride (last)
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # out_fp32 now holds [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5. Post‑processing: LayerNorm → gating → final linear
    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate (broadcast over the last dim)
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
