    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

                                       weight=weights["norm.weight"],

                                       eps=eps)

    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    right = right * right_gate

    out = torch.empty_like(left, dtype=torch.float32, device=device)

    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right.stride()

"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused Triton kernel.

The heavy part of the module is the 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B,N,N,H] (H = hidden_dim).
The kernel computes a tiled matrix‑multiply for each channel block,
reducing over the k‑dimension.  All other operations (LayerNorm,
Linear, gates, masking) are performed with regular PyTorch kernels,
which keeps the implementation simple while still off‑loading the
N³ work to Triton.

The kernel works for any batch size (1‑2), any sequence length up to
1024 and hidden dimensions that are multiples of BLOCK_H (we use 32
which divides 128/384).  The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,:] += Σ_k left[b,i,k,:] * right[b,j,k,:]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const *float32
    right_ptr,         # const *float32
    out_ptr,           # *float32
    B, N, H,           # int32 scalars
    # strides for left  [b,i,k,d]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right [b,j,k,d]
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out   [b,i,j,d]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Tile‑wise matrix multiplication over the k dimension for a block
    of (i,j) positions and a block of hidden channels.
    """
    pid_i = tl.program_id(0)          # tile along i
    pid_j = tl.program_id(1)          # tile along j
    pid_b = tl.program_id(2)          # batch index

    # ----- offsets for the current tile ---------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    b = pid_b

    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # loop over hidden‑channel blocks
    for h_start in range(0, H, BLOCK_H):
        # accumulator for this channel block
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # iterate over k in tiles
        for k_base in range(0, N, BLOCK_K):
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ----- pointers ------------------------------------------------
            # left:   [b, i, k, h]
            left_ptrs = left_ptr + \
                b * stride_left_b + \
                offs_i[:, None] * stride_left_i + \
                offs_k[None, :] * stride_left_k + \
                h_start * stride_left_h

            # right:  [b, j, k, h]
            right_ptrs = right_ptr + \
                b * stride_right_b + \
                offs_j[:, None] * stride_right_j + \
                offs_k[None, :] * stride_right_k + \
                h_start * stride_right_h

            # load tiles (mask out‑of‑bounds with zero)
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_j[:, None] & mask_k[None, :],
                other=0.0
            )
            # matmul over k (right is transposed)
            acc += tl.dot(left_tile, right_tile, trans_b=True)

        # ----- store the result --------------------------------------------
        out_ptrs = out_ptr + \
            b * stride_out_b + \
            offs_i[:, None] * stride_out_i + \
            offs_j[None, :] * stride_out_j + \
            h_start * stride_out_h

        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None] & mask_j[None, :]
        )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple (input, mask, weights, config)
            - input  : torch.Tensor [B, N, N, dim]   (float32)
            - mask   : torch.Tensor [B, N, N] (bool/int) – may be omitted when config["nomask"] is True
            - weights: dict of torch.Tensor containing all linear / LN parameters
            - config : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) LayerNorm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (mask shape [B,N,N] → [B,N,N,1])
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)       # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------------
    # 4) Gating vectors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triton fused contraction
    # --------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # tensor strides (in elements, not bytes)
    s_left = left.stride()   # (b,i,k,d)
    s_right = right.stride()
    s_out = out.stride()

    # tile configuration – chosen to fit H200 SM resources
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32                     # must divide hidden_dim (128/384)

    grid = (
        triton.cdiv(N, BLOCK_I),     # i‑tiles
        triton.cdiv(N, BLOCK_J),     # j‑tiles
        B,                           # batch dimension
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 6) Final LayerNorm, gating, and projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
