"""
tri_mul_outgoing.py
-------------------

Implementation of the *outgoing* TriMul operator (AlphaFold‑3) with the
computationally intensive 4‑D einsum off‑loaded to a custom Triton kernel.
The kernel computes, for each batch **b**, head **h** and pair of residues
(i, j):

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

i.e. a batched matrix‑multiply over the “k” axis for every hidden dimension.
All other operations (LayerNorms, linear projections, sigmoids, masking) are
performed in PyTorch – this keeps the host code simple while the O(N³·H)
core is executed efficiently on the GPU.

The kernel tiles the (i, j) plane (BLOCK_M×BLOCK_N) and the head dimension
(BLOCK_H).  The reduction over **k** is tiled with BLOCK_K and performed with
`tl.dot`.  The implementation works for the full range of test configurations
(N = 128‑1024, B ∈ {1,2}, hidden_dim ∈ {128,384}) on Hopper‑class GPUs.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul over the reduction axis k for every head
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,          # const float*  [B, N, N, H]
    right_ptr,         # const float*  [B, N, N, H] (will be read transposed on k‑j)
    out_ptr,           #       float*  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # ----------------------------------------------
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Tiled implementation of
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid_m = tl.program_id(0)          # tile over i
    pid_n = tl.program_id(1)          # tile over j
    pid_z = tl.program_id(2)          # combined batch & head tile

    num_h_tiles = tl.cdiv(H, BLOCK_H)
    batch_idx = pid_z // num_h_tiles
    h_tile = pid_z % num_h_tiles

    # ------------------------------------------------------------------
    # Offsets for the three tiled dimensions
    # ------------------------------------------------------------------
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    h_off = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)

    i_mask = i_off < N
    j_mask = j_off < N
    h_mask = h_off < H

    # ------------------------------------------------------------------
    # Accumulator: (BLOCK_M, BLOCK_N, BLOCK_H)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # --------------------------------------------------------------
        # Load a tile of left:  (i, k, h)
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch_idx * stride_left_b
            + i_off[:, None, None] * stride_left_i
            + k_off[None, :, None] * stride_left_k
            + h_off[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )  # shape (BM, BK, BH)

        # --------------------------------------------------------------
        # Load a tile of right but **transpose** the (j,k) axes so we get
        # (k, j, h). This enables a direct dot-product.
        # --------------------------------------------------------------
        right_ptrs_T = (
            right_ptr
            + batch_idx * stride_right_b
            + k_off[:, None, None] * stride_right_k
            + j_off[None, :, None] * stride_right_j
            + h_off[None, None, :] * stride_right_h
        )
        right_T = tl.load(
            right_ptrs_T,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )  # shape (BK, BN, BH)

        # --------------------------------------------------------------
        # Compute the dot‑product for each head dimension independently.
        # --------------------------------------------------------------
        for h in range(0, BLOCK_H):
            # shape (BM, BK) @ (BK, BN) -> (BM, BN)
            acc[:, :, h] += tl.dot(left[:, :, h], right_T[:, :, h])

    # ------------------------------------------------------------------
    # Write the result back
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_idx * stride_out_b
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + h_off[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton‑accelerated
    einsum.  The function receives a tuple:

        (input, mask, weights, config)

    *input* : torch.Tensor [B, N, N, dim] (float32, already on CUDA)
    *mask*  : torch.Tensor [B, N, N]   (float32 or bool, already on CUDA)
    *weights*: dict with the nine parameters required by the reference
               implementation.
    *config* : dict containing ``dim`` and ``hidden_dim`` (``c`` and ``c_z``).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper lambdas for layer‑norm & linear using supplied weights
    # ------------------------------------------------------------------
    eps = 1e-5

    def layer_norm(x, w, b):
        # x: [..., C]; w, b: [C]
        return F.layer_norm(x, (x.shape[-1],), weight=w, bias=b, eps=eps)

    def linear(x, w):
        # x: [..., C_in]; w: [C_out, C_in]
        return F.linear(x, w)

    # ------------------------------------------------------------------
    # 1. Input normalisation
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask is broadcastable on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating scalars
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the 4‑D einsum
    # ------------------------------------------------------------------
    # Allocate the output tensor for the einsum (still in float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # Strides (in elements) for each tensor – needed by the kernel
    stride_left = left.stride()   # (b,i,k,h)
    stride_right = right.stride() # (b,j,k,h) – note j and k order
    stride_out = out.stride()     # (b,i,j,h)

    # Choose tile sizes.  Empirically good for Hopper/H200.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * triton.cdiv(hidden_dim, BLOCK_H),
    )

    tri_mul_einsum_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],   # j‑stride
        stride_right[2],   # k‑stride
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing (layer‑norm, gate, final linear)
    # ------------------------------------------------------------------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])

    return out
