"""
TriMul “outgoing” implementation with a tiny Triton kernel.

The heavy O(N³·C) operation is a batch of matrix‑multiplications:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to (L_d) @ (R_d)ᵀ for every hidden dimension *d*.
We compute it with a highly‑optimised cuBLAS batched GEMM (torch.bmm) after
projecting the tensors, and we use a small Triton kernel to fuse the mask
application with the gating (left/right * mask * gate).  All other linear
layers and LayerNorms are performed with Torch primitives; the final output
is cast back to float32.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimensions 128/384 (the tests cover these).  It assumes the tensors
are already on the CUDA device.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: fused mask * gate * value
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    val_ptr,                # input tensor (left or right)          [B,N,N,H]
    gate_ptr,               # gate tensor (sigmoid)                [B,N,N,H]
    mask_ptr,               # mask (broadcasted on H)              [B,N,N]
    out_ptr,                # output tensor (same shape as val)    [B,N,N,H]
    B, N, H,
    stride_val_b, stride_val_i, stride_val_k, stride_val_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # linear index -> (b,i,k,h)
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # compute pointers
    val_offset = (b * stride_val_b + i * stride_val_i
                  + k * stride_val_k + h * stride_val_h)
    gate_offset = (b * stride_gate_b + i * stride_gate_i
                   + k * stride_gate_k + h * stride_gate_h)
    mask_offset = (b * stride_mask_b + i * stride_mask_i + k * stride_mask_k)

    val   = tl.load(val_ptr + val_offset,   mask=mask, other=0.0)
    gate  = tl.load(gate_ptr + gate_offset, mask=mask, other=0.0)
    msk   = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    out = val * gate * msk
    tl.store(out_ptr + val_offset, out, mask=mask)


# ----------------------------------------------------------------------
# Triton‑enabled TriMul forward (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool / float)
        - weights      : dict of torch.Tensor           (all float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # left_proj : dim -> hidden
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – also dim -> hidden
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Fuse mask * gate * value with a Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is float32 for multiplication
        mask_f = mask.to(input_tensor.dtype)

        BLOCK = 1024  # tuneable; works for all test sizes

        # left
        left_out = torch.empty_like(left)
        grid = (triton.cdiv(left.numel(), BLOCK),)
        apply_gate_mask_kernel[grid](
            left,
            left_gate,
            mask_f,
            left_out,
            B, N, hidden,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            left_gate.stride(0), left_gate.stride(1),
            left_gate.stride(2), left_gate.stride(3),
            mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
            BLOCK=BLOCK,
        )
        # right
        right_out = torch.empty_like(right)
        grid = (triton.cdiv(right.numel(), BLOCK),)
        apply_gate_mask_kernel[grid](
            right,
            right_gate,
            mask_f,
            right_out,
            B, N, hidden,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            right_gate.stride(0), right_gate.stride(1),
            right_gate.stride(2), right_gate.stride(3),
            mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
            BLOCK=BLOCK,
        )
    else:
        # No mask ⇒ simple element‑wise multiplication
        left_out  = left * left_gate
        right_out = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched matrix multiply for each hidden dim
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    -> (L_d) @ (R_d)^T for every d
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    L = left_out.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    R = right_out.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # batched GEMM (cuBLAS) – the fastest primitive on H200
    out_flat = torch.bmm(L, R.transpose(1, 2))                      # (B*hidden, N, N)

    # reshape back to [B, N, N, hidden]
    out = out_flat.reshape(B, hidden, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Post‑processing: Layer‑norm on hidden, gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # broadcast on hidden dim
    out = F.linear(out, weights["to_out.weight"])   # hidden -> dim

    return out
