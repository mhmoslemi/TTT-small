"""
TriMul “outgoing” implementation.

The heavy part of the module is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batch of H independent matrix‑multiplications
    out_d = left_d @ right_d.T   (left/right are N×N, H = hidden_dim).

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch (fast
CUDA kernels) and implement the N³ contraction in Triton.  The kernel works
on a tile (BLOCK_M × BLOCK_N) of the output for a fixed (batch, head)
pair, looping over K‑tiles.  All arithmetic is accumulated in fp32 for
numerical stability, while the inputs/outputs are stored in fp16 to
exploit the H200 tensor‑cores.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output   (float32)
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: tiled batched GEMM for each hidden dimension (head)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # fp16, [B, N, N, H]
    right_ptr,         # fp16, [B, N, N, H]
    out_ptr,           # fp16, [B, N, N, H]
    B, N, H,           # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid0 = tl.program_id(0)          # batch * head dimension
    pid1 = tl.program_id(1)          # tile row (i)
    pid2 = tl.program_id(2)          # tile col (j)

    # decode batch and hidden (head) index
    b = pid0 // H
    h = pid0 % H

    # tile offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # ------------------------------------------------------------------
    # pointers for the current tile
    # left  : shape (M, K)  -> rows = i, cols = k
    # right : shape (K, N)  -> rows = k, cols = j   (note the transpose)
    # ------------------------------------------------------------------
    left_tile_ptr = (
        left_ptr
        + b * stride_left_b
        + h * stride_left_h
        + offs_i[:, None] * stride_left_i
        + offs_k[None, :] * stride_left_k
    )

    right_tile_ptr = (
        right_ptr
        + b * stride_right_b
        + h * stride_right_h
        + offs_k[:, None] * stride_right_k
        + offs_j[None, :] * stride_right_j
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = cur_k * BLOCK_K
        cur_k_valid = (k_offset + offs_k) < N

        # load tiles; out‑of‑bounds elements are set to 0.0
        a = tl.load(
            left_tile_ptr + k_offset * stride_left_k,
            mask=mask_i[:, None] & cur_k_valid[None, :],
            other=0.0,
        )
        b_tile = tl.load(
            right_tile_ptr + k_offset * stride_right_k,
            mask=cur_k_valid[:, None] & mask_j[None, :],
            other=0.0,
        )
        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b_tile)

    # store result tile back to global memory
    out_tile_ptr = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_tile_ptr, acc.to(tl.float16), mask=mask_ij)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 / bool)
        - weights_dict : mapping of the eight weight tensors
        - config_dict  : { "dim": int, "hidden_dim": int, "nomask": bool }

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and mask application
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_unsq = mask.unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and gating multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Main contraction – Triton kernel
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (GPU tensor‑cores)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # tile sizes – tuned for Hopper tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # (batch * head) dimension
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
    )

    # strides for contiguous tensors [B, N, N, H]
    s_left = left_fp16.stride()   # (N*N*H, N*H, H, 1)
    s_right = right_fp16.stride()
    s_out = out_fp16.stride()

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],          # stride over j (second N)
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
