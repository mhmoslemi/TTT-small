"""
TriMul “outgoing” implementation.

The heavy O(N³) part (the einsum over the shared sequence dimension) is
implemented as a batched matrix‑multiply in Triton.  The surrounding
LayerNorm / Linear / gating operations are kept in PyTorch because they are
trivial compared with the N³ work.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Linear projections → left / right (shape B,N,N,H) and three gates.
3. Optional mask (broadcasted on the hidden dimension) is applied.
4. Gating: left *= sigmoid(left_gate), right *= sigmoid(right_gate).
5. Reshape left/right to (B*H, N, N) and transpose right so that the
   multiplication becomes a standard batched GEMM:
        out_flat[bh, i, j] = Σ_k left_flat[bh, i, k] * right_flat_T[bh, k, j]
6. The GEMM is executed by a Triton kernel (`triton_matmul_kernel`).
   The kernel works in fp16 with fp32 accumulation.
7. Reshape the result back to (B,N,N,H) and apply the output
   LayerNorm, out‑gate and final linear projection.
8. Return a float‑32 tensor of shape (B,N,N,dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (fp16 input / fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def triton_matmul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # matrix dimensions
    M, N, K,
    # strides (batch, row, col)
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch*head index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # --------------------------------------------------------------
    # Compute offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----------------------------------------------------------------
    # Pointers for the current block
    # ----------------------------------------------------------------
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------
    # Main K‑loop
    # ----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)  # fp16
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)   # fp16
        acc += tl.dot(a, b)               # fp32 accumulation
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write back (convert to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host side helper that launches the kernel
# ----------------------------------------------------------------------
def triton_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: a,b  ->  c
    a,b : (B*H, N, N)   fp16
    returns c : (B*H, N, N)  fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert b.shape == (B_H, K, N)

    c = torch.empty_like(a)

    # block size choice – tuned for Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B_H,
    )

    triton_matmul_kernel[grid](
        a,
        b,
        c,
        M=N,
        N=N,
        K=K,
        # strides for a (batch, row, col)
        stride_am=a.stride(1),
        stride_ak=a.stride(2),
        stride_ab=a.stride(0),
        # strides for b
        stride_bk=b.stride(1),
        stride_bn=b.stride(2),
        stride_bb=b.stride(0),
        # strides for c
        stride_cm=c.stride(1),
        stride_cn=c.stride(2),
        stride_cb=c.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            input  : torch.Tensor [B, N, N, dim]   (float32)
            mask   : torch.Tensor [B, N, N]       (float32, 0/1) – may be None
            weights: dict of torch.Tensors with the same names as in the
                     reference implementation.
            config : dict containing at least:
                     - dim (int)
                     - hidden_dim (int)
                     - nomask (bool) – if True the mask can be ignored.
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over last dim (dim = D)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # (B,N,N,D)  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.unsqueeze(-1)                # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the N³ multiplication
    #    left  : (B,N,N,H) -> (B,H,N,N)   (i,k)
    #    right : (B,N,N,H) -> (B,H,N,N)   (k,j)  (transpose first two seq dims)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()          # (B,H,N,N)
    right = right.permute(0, 3, 2, 1).contiguous()        # (B,H,N,N)  (k,j)

    # flatten batch*head dimension
    left_flat = left.view(B * H, N, N).to(torch.float16)
    right_flat = right.view(B * H, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Batched GEMM via Triton
    # ------------------------------------------------------------------
    out_flat = triton_matmul(left_flat, right_flat)      # (B*H,N,N) fp16

    # back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # (B,N,N,H) float32

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out.to(torch.float32)
