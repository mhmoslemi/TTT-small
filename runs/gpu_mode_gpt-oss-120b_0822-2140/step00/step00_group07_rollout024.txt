"""
TriMul “outgoing” implementation (forward only).

The heavy N³ operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per hidden dimension:
    out_d = left_d @ right_dᵀ .
We compute the linear / layer‑norm parts in PyTorch (highly‑optimised cuBLAS)
and fuse the element‑wise mask‑and‑gate multiplications with a tiny Triton
kernel.  The final result is float‑32 as required.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    out_left_ptr, out_right_ptr,
    left_ptr, right_ptr,
    mask_ptr, left_gate_ptr, right_gate_ptr,
    N_ELEMS,                     # total number of elements = B*N*N*H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < N_ELEMS

    # load
    l = tl.load(left_ptr + offs, mask=m)
    r = tl.load(right_ptr + offs, mask=m)
    mk = tl.load(mask_ptr + offs, mask=m)          # already broadcasted to H
    lg = tl.load(left_gate_ptr + offs, mask=m)
    rg = tl.load(right_gate_ptr + offs, mask=m)

    # fuse:   out = tensor * mask * gate
    out_l = l * mk * lg
    out_r = r * mk * rg

    # store
    tl.store(out_left_ptr + offs, out_l, mask=m)
    tl.store(out_right_ptr + offs, out_r, mask=m)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape   # x is [B, N, N, dim]

    # --------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]

    # --------------------------------------------------------------
    # Linear projections + gates (bias‑free)
    # --------------------------------------------------------------
    left   = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right  = F.linear(x, weights["right_proj.weight"])

    left_gate   = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate  = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate    = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # Fuse mask, left‑gate, right‑gate with a Triton kernel
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, device=device, dtype=x.dtype)
    mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    total_elems = left.numel()
    BLOCK = 1024                          # modest block size, works for all test shapes
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fused, right_fused,
        left, right,
        mask_exp, left_gate, right_gate,
        total_elems,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # Batched matrix multiplication per hidden dimension:
    #    out_d = left_d @ right_dᵀ   (d = 0 … hidden_dim‑1)
    # --------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # rightᵀ  → (B*H, N, N)
    right_t = right_mat.transpose(1, 2)

    # torch.bmm uses cuBLAS → very fast on H200
    out_mat = torch.bmm(left_mat, right_t)               # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
