"""
TriMul “outgoing” implementation (forward only).

The expensive part is the contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply per hidden channel:

    out_{b,d} = left_{b,d} @ (right_{b,d})ᵀ      (N×N) @ (N×N)

The kernel below performs this batched mat‑mul in half‑precision (fp16)
with fp32 accumulation and stores the result as fp32.  All other
operations (LayerNorms, linear projections, gating, final linear)
are kept in PyTorch for simplicity.

The launch layout is
    pid0 : batch * hidden_dim
    pid1 : block‑row of N
    pid2 : block‑col of N
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: (B * H) independent matmuls of shape (N,N) @ (N,N)^T
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    left_ptr,               # fp16  [B,N,N,H]
    right_ptr,              # fp16  [B,N,N,H]
    out_ptr,                # fp32  [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs
    # pid0 = batch * hidden_dim index
    # pid1 = block row
    # pid2 = block col
    # --------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch = pid0 // H          # 0 .. B-1
    channel = pid0 % H         # 0 .. H-1

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (batch, channel)
    a_base = left_ptr + batch * stride_left_b + channel * stride_left_h
    b_base = right_ptr + batch * stride_right_b + channel * stride_right_h
    c_base = out_ptr + batch * stride_out_b + channel * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=(offs_m < N)[:, None] & (offs_k < N)[None, :],
            other=0.0,
        ).to(tl.float16)

        # Load B tile (row‑major, need the transpose on‑the‑fly)
        #   B is indexed as (j, k); we want (k, j) for dot product.
        b = tl.load(
            b_base
            + offs_n[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k,
            mask=(offs_n < N)[:, None] & (offs_k < N)[None, :],
            other=0.0,
        ).to(tl.float16)

        # acc += A @ Bᵀ   (Bᵀ is (k, n) because b is (n, k))
        acc += tl.dot(a, b, trans_b=True)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    tl.store(
        c_base
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j,
        acc,
        mask=(offs_m < N)[:, None] & (offs_n < N)[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of TriMul (outgoing) with a Triton‑accelerated
    contraction.  All tensors are assumed to be on CUDA already.

    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device

    # ------------------------------------------------------------------
    # Helper: LayerNorm without bias/weight handling
    # ------------------------------------------------------------------
    def layer_norm(inp, weight, bias):
        # normalized over the last dimension
        return F.layer_norm(
            inp,
            normalized_shape=weight.shape,
            weight=weight,
            bias=bias,
            eps=1e-5,
        )

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – keep fp32 for now, convert later
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask (optional)
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1)                       # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton batched mat‑mul: left @ rightᵀ  (per hidden channel)
    #    Use fp16 for the two inputs, fp32 accumulation.
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # strides for the original contiguous layout [B,N,N,H]
    stride_b, stride_i, stride_k, stride_h = left_fp16.stride()
    # right strides are identical (same layout)
    stride_rb, stride_rj, stride_rk, stride_rh = right_fp16.stride()
    # output strides (float32)
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                 # pid0 : batch * channel
        triton.cdiv(N, BLOCK_M),       # pid1 : rows
        triton.cdiv(N, BLOCK_N),       # pid2 : cols
    )

    _batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Second LayerNorm (hidden_dim) + out gating
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
