"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

We implement the heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
as a batched matrix multiply:   out_d = left_d @ right_dᵀ
for every hidden dimension *d*.  The per‑channel GEMM is executed in
Triton, while the cheap LayerNorm / Linear / gating steps stay in PyTorch.
All arithmetic is carried out in float32 and the final tensor has shape
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with B transposed (C = A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_transpose_kernel(
    A_ptr, B_ptr, C_ptr,                         # pointers
    BATCH, M, N, K,                               # sizes
    stride_a_batch, stride_a_m, stride_a_k,       # A strides
    stride_b_batch, stride_b_n, stride_b_k,       # B strides (original order)
    stride_c_batch, stride_c_m, stride_c_n,       # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch index `b` in [0, BATCH):
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (B is used transposed)
    A : (BATCH, M, K)
    B : (BATCH, N, K)          -- original layout, we read as (k, j)
    C : (BATCH, M, N)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k‑range for this iteration
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # --------------------------------------------------------------
        # Load a tile of A : shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # --------------------------------------------------------------
        # Load a tile of Bᵀ.
        # B is stored as (batch, N, K).  For the transpose we need B[j, k]
        # → original address uses row=j (stride_b_n) and col=k (stride_b_k).
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_n[None, :] * stride_b_n   # j dimension (original row)
            + offs_k[:, None] * stride_b_k   # k dimension (original col)
        )
        b = tl.load(b_ptrs, mask=mask_n[None, :] & mask_k[:, None], other=0.0)

        # --------------------------------------------------------------
        # Accumulate
        # --------------------------------------------------------------
        acc += tl.dot(a, b)   # (M×K)·(K×N) → (M×N)

    # --------------------------------------------------------------
    # Write out
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   float32
            mask         : [B, N, N]        (0/1)  (may be all‑ones)
            weights      : dict of all linear / layer‑norm parameters
            config       : {"dim": int, "hidden_dim": int}
    Returns:
        out : [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) + mask
    # ------------------------------------------------------------------
    # x used for all linear layers: shape [B,N,N,dim]
    # flatten last two dims to perform a single matmul per projection
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    left_proj_w = weights["left_proj.weight"]      # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]    # (hidden, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden)

    # optional mask (broadcast to hidden)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) + elementwise gating
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy reduction: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    -> batched GEMM per hidden dimension using Triton
    # ------------------------------------------------------------------
    # Rearrange to (B*hidden, N, N) for the kernel
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # strides (in elements) for each tensor
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Triton block sizes – tuned for Hopper SM90 (shared memory 64 KiB)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # small enough to keep registers low

    grid = (
        B * hidden,                     # program_id(0) – batch * hidden
        triton.cdiv(N, BLOCK_M),        # program_id(1) – rows
        triton.cdiv(N, BLOCK_N),        # program_id(2) – cols
    )

    batched_gemm_transpose_kernel[grid](
        left_flat, right_flat, out_flat,
        B * hidden, N, N, N,                 # BATCH, M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply final gate (broadcast over hidden)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]
    out_flat = out.reshape(-1, hidden)                     # (B*N*N, hidden)
    out = torch.nn.functional.linear(out_flat, to_out_w)  # (B*N*N, dim)
    out = out.view(B, N, N, dim)

    return out
