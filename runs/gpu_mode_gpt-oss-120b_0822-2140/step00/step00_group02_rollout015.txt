"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of TriMul is the three‑dimensional contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where `left` and `right` have shape `[B, N, N, H]` (H = hidden_dim).
The implementation below

* performs the LayerNorm / Linear / gating parts in PyTorch (they are cheap)
* launches a Triton kernel that evaluates (1) for **all** (b,i,j) in parallel.
  Each program instance computes a whole slice of the hidden dimension
  (BLOCK_D values) for a single `(b,i,j)` coordinate, reducing over `k`
  in chunks of `BLOCK_K`.  The kernel works for any batch size, sequence
  length and hidden dimension that fit in memory and is written for the
  Hopper H200 (SM‑90) but does not contain any device‑specific intrinsics.

The kernel returns a tensor of dtype `float32` as required by the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # dimensions
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_D: tl.constexpr,                # hidden‑dim tile size
    BLOCK_K: tl.constexpr,                # reduction tile size over k
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    One program per (b,i,j).  The hidden dimension is processed in tiles
    of size BLOCK_D; the reduction over k is tiled with BLOCK_K.
    """
    # ------------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------------
    pid_b = tl.program_id(0)               # batch index
    pid_i = tl.program_id(1)               # i index
    pid_j = tl.program_id(2)               # j index

    # ------------------------------------------------------------------
    # offsets for the fixed i and j (scalar per program)
    # ------------------------------------------------------------------
    i = pid_i
    j = pid_j

    # ------------------------------------------------------------------
    # hidden‑dim loop
    # ------------------------------------------------------------------
    d_base = 0
    while d_base < H:
        # d offsets inside the current tile
        d_offsets = d_base + tl.arange(0, BLOCK_D)
        mask_d = d_offsets < H

        # accumulator for the current d‑tile
        acc = tl.zeros([BLOCK_D], dtype=tl.float32)

        # ------------------------------------------------------------------
        # reduction over k
        # ------------------------------------------------------------------
        k_base = 0
        while k_base < N:
            k_offsets = k_base + tl.arange(0, BLOCK_K)
            mask_k = k_offsets < N

            # ------------------------------------------------------------------
            # load a (BLOCK_K, BLOCK_D) tile from left and right
            # ------------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_lb
                + i * stride_li
                + k_offsets[:, None] * stride_lk
                + d_offsets[None, :] * stride_ld
            )
            right_ptrs = (
                right_ptr
                + pid_b * stride_rb
                + j * stride_rj
                + k_offsets[:, None] * stride_rk
                + d_offsets[None, :] * stride_rd
            )

            # mask combines k‑tile and d‑tile masks
            tile_mask = mask_k[:, None] & mask_d[None, :]

            left_tile  = tl.load(left_ptrs,  mask=tile_mask, other=0.0)   # (K, D)
            right_tile = tl.load(right_ptrs, mask=tile_mask, other=0.0)   # (K, D)

            # element‑wise product and reduction over k
            prod = left_tile * right_tile          # (K, D)
            acc += tl.sum(prod, axis=0)            # (D,)

            k_base += BLOCK_K

        # ------------------------------------------------------------------
        # write the accumulated tile to output
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_ob
            + i * stride_oi
            + j * stride_oj
            + d_offsets * stride_od
        )
        tl.store(out_ptrs, acc, mask=mask_d)

        d_base += BLOCK_D


# ----------------------------------------------------------------------
# Helper: LayerNorm in pure PyTorch (fast enough for the small‑dim case)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    """Layer‑norm over the last dimension."""
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) * torch.rsqrt(var + eps)
    return x * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input, mask, weights, config)
            input  : [B, N, N, dim]   torch.float32 on CUDA
            mask   : [B, N, N]        torch.float32 (0/1) on CUDA
            weights: dict of torch tensors (see description)
            config : dict with keys "dim", "hidden_dim"
    Returns:
        out : torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = layer_norm(input_tensor, norm_weight, norm_bias)          # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    # shape: (..., dim) @ (dim, hidden)ᵀ  -> (..., hidden)
    # we use torch.einsum for simplicity – this part is cheap.
    left = torch.einsum("bnkd,hd->bnkh", x, left_proj_w)          # [B,N,N,hidden]
    right = torch.einsum("bnkd,hd->bnkh", x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ext = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_ext
        right = right * mask_ext

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also linear w/out bias
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(torch.einsum("bnkd,hd->bnkh", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("bnkd,hd->bnkh", x, right_gate_w))
    out_gate   = torch.sigmoid(torch.einsum("bnkd,hd->bnkh", x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for contiguous tensors
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # launch parameters – tuned for Hopper (SM‑90)
    BLOCK_D = 32            # hidden‑dim tile (must divide hidden or be padded)
    BLOCK_K = 32            # reduction tile over k

    grid = (B, N, N)        # one program per (b,i,j)
    trimul_contract_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # good default for H200
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w      = weights["to_out.weight"].to(device)

    out = layer_norm(out, to_out_norm_w, to_out_norm_b)          # [B,N,N,hidden]
    out = out * out_gate                                        # broadcast over hidden
    out = torch.einsum("bnkh,hd->bndk", out, to_out_w)          # final proj → [B,N,N,dim]

    return out
```
