"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the N‑by‑N matrix multiplication that sums over the
third sequence dimension (the einsum
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]).
For each batch and each hidden channel d this is exactly a matrix product
    left_i,k (N×N) @ right_j,kᵀ (N×N) → out_i,j (N×N).

The kernel below performs a batched matmul in Triton (one batch per
(B·hidden_dim) pair).  All cheap tensor ops (layer‑norm, linear projections,
gates, final norm + linear) are kept in PyTorch – they are memory‑bound and
the cuBLAS‑like matmul dominates the runtime.  The kernel uses a classic
tiling strategy (BLOCK_M×BLOCK_N tiles, iterating over K) and loads the
right‑hand matrix transposed by swapping its K/N strides, thus avoiding an
explicit transpose.

The host‑side `custom_kernel` assembles the full forward pass, loads the
provided weights, and launches the Triton kernel.  The final output is
returned as a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# --------------------------------------------------------------------------- #
# Triton batched matmul kernel (left @ rightᵀ)                               #
# --------------------------------------------------------------------------- #
@triton.jit
def _triton_matmul_kernel(
    a_ptr,                # left  matrix,   (batch, M, K)
    b_ptr,                # right matrix,  (batch, K, N)  (but we pass transposed strides)
    c_ptr,                # output,        (batch, M, N)
    M, N, K,
    stride_am, stride_ak,          # a: batch‑stride, M‑stride, K‑stride
    stride_bk, stride_bn,          # b: batch‑stride, K‑stride, N‑stride  (already transposed)
    stride_cm, stride_cn,          # c: batch‑stride, M‑stride, N‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                      # tile row
    pid_n = tl.program_id(1)                      # tile col
    pid_b = tl.program_id(2)                      # batch

    #---------------------------------
    # Pointers for the current batch
    #---------------------------------
    a_batch = a_ptr + pid_b * stride_am
    b_batch = b_ptr + pid_b * stride_bk
    c_batch = c_ptr + pid_b * stride_cm

    #---------------------------------
    # Offsets within the tile
    #---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for OOB rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #---------------------------------
    # Loop over K dimension
    #---------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a: (M, K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_ak
                           + offs_k[None, :] * stride_am)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # b (transposed): original b is (K, N), we want (N, K) for dot
        # using swapped strides we read element (k, n) as if it were (n, k)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bn
                           + offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # Advance pointers for next k‑block
        a_batch += BLOCK_K * stride_am
        b_batch += BLOCK_K * stride_bk

    # Store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cn
                        + offs_n[None, :] * stride_cm)
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Helper: fused LayerNorm (no bias handling in kernel – keep in PyTorch)
# --------------------------------------------------------------------------- #
def _layer_norm(x, weight, bias, eps=1e-5):
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    N³ multiplication.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32/float16)
            - mask  : torch.Tensor [B, N, N]       (bool/float)
            - weights: dict of torch.Tensors containing the model parameters
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------- #
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )                                 # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------- #
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4) Gates (sigmoid) and multiplication
    # ------------------------------------------------------------------- #
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Batched matrix multiplication: left @ rightᵀ
    #    Work with shape (B*hidden, N, N) for efficient launch
    # ------------------------------------------------------------------- #
    # reshape to (batch_h, N, N) where batch_h = B * hidden
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    out_mat = torch.empty_like(left_mat)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * hidden,                   # batch dimension
    )

    # Strides (in elements, not bytes)
    # left (batch, M, K)
    stride_am = left_mat.stride(0)      # batch stride
    stride_ak = left_mat.stride(1)      # M stride
    stride_ak2 = left_mat.stride(2)     # K stride  (named stride_ak for consistency)

    # right (batch, K, N) – we will pass transposed strides
    stride_b_batch = right_mat.stride(0)
    stride_bk = right_mat.stride(2)     # original N stride  (becomes K stride after transpose)
    stride_bn = right_mat.stride(1)     # original K stride  (becomes N stride after transpose)

    # out (batch, M, N)
    stride_cm = out_mat.stride(0)
    stride_cn = out_mat.stride(2)       # N stride
    stride_cm2 = out_mat.stride(1)      # M stride

    _triton_matmul_kernel[grid](
        left_mat,                # a_ptr
        right_mat,               # b_ptr (will be read transposed)
        out_mat,                 # c_ptr
        N, N, N,
        # a strides
        stride_am, stride_ak, stride_ak2,
        # b (transposed) strides
        stride_b_batch, stride_bk, stride_bn,
        # c strides
        stride_cm, stride_cn, stride_cm2,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original tensor shape: (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6) Post‑processing: norm, gating, final linear
    # ------------------------------------------------------------------- #
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
