"""TriMul “outgoing” implementation with a fused batched matmul kernel (Triton).

The heavy part of the forward pass is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].transpose(-2,-1)

The kernel computes this batched matmul for all batch‐ and channel‑indices
(B * hidden_dim) in one launch, using the classic tiling strategy
(BLOCK_M × BLOCK_N tiles, loop over K).  All other operations (LayerNorm,
linear projections, gating, final LayerNorm + linear) are performed with
torch‑ops for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel : batched matmul (C = A @ Bᵀ) for A,B shape (BC, M, K) and (BC, N, K)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    A_ptr,                # *float32   (BC, M, K)
    B_ptr,                # *float32   (BC, N, K)   (will be read as K x N after pointer arithmetic)
    C_ptr,                # *float32   (BC, M, N)   result
    M, N, K,              # matrix dimensions
    stride_A_bc, stride_A_m, stride_A_k,
    stride_B_bc, stride_B_n, stride_B_k,
    stride_C_bc, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bc = tl.program_id(2)               # outer batch‑channel dimension
    pid_m  = tl.program_id(0)               # tile row
    pid_n  = tl.program_id(1)               # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for the starting position of this (bc, m, n) tile
    a_ptrs = A_ptr + pid_bc * stride_A_bc + offs_m[:, None] * stride_A_m \
             + tl.arange(0, BLOCK_K)[None, :] * stride_A_k
    b_ptrs = B_ptr + pid_bc * stride_B_bc + tl.arange(0, BLOCK_K)[:, None] * stride_B_k \
             + offs_n[None, :] * stride_B_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile (M × K) and B tile (K × N)
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] &
                         (tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[:, None] &
                         (offs_n < N)[None, :],
                    other=0.0)

        # Accumulate product
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_A_k
        b_ptrs += BLOCK_K * stride_B_k

    # Write back the result tile
    c_ptrs = C_ptr + pid_bc * stride_C_bc + offs_m[:, None] * stride_C_m \
             + offs_n[None, :] * stride_C_n
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32 on CUDA)
        - mask_tensor  : [B, N, N]        (float32/bool on CUDA)
        - weights_dict : contains all linear / LN parameters
        - config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (over last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    # Linear: y = x @ W.T   (torch.nn.functional.linear expects weight shape (out, in))
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Apply optional mask (mask is broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections of the *original* x)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core einsum -> fused batched matmul via Triton
    # --------------------------------------------------------------
    # transpose right on the last two axes so that we multiply left @ rightᵀ
    right_t = right.permute(0, 2, 1, 3).contiguous()           # [B,N,N,hidden_dim] -> [B,N,N,hidden_dim] with (j,k) -> (k,j)

    B, N, _, _ = left.shape
    bc = B * hidden_dim                                     # combined outer dimension

    # reshape to (BC, M, K) / (BC, N, K)
    left_flat  = left.permute(0, 3, 1, 2).reshape(bc, N, N).contiguous()
    right_flat = right_t.permute(0, 3, 1, 2).reshape(bc, N, N).contiguous()
    out_flat   = torch.empty_like(left_flat)                # (BC, N, N)

    # strides (in elements)
    stride_A_bc, stride_A_m, stride_A_k = left_flat.stride()    # (N*N, N, 1)
    stride_B_bc, stride_B_n, stride_B_k = right_flat.stride()  # (N*N, N, 1)
    stride_C_bc, stride_C_m, stride_C_n = out_flat.stride()    # (N*N, N, 1)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        bc,                        # batch‑channel dimension
    )
    batch_matmul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N, N, N,
        stride_A_bc, stride_A_m, stride_A_k,
        stride_B_bc, stride_B_n, stride_B_k,
        stride_C_bc, stride_C_m, stride_C_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # no bias as in the reference

    return out.float()      # guarantee float32 output
