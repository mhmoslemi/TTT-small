"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated inner product.

The heavy work is the 3‑dimensional einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication per hidden
dimension d:
    out[b,…,d] = left[b,…,d] @ right[b,…,d].T

We fuse this with Triton (fp16 work, fp32 accumulation) while keeping
the surrounding LayerNorm / Linear / gating logic in PyTorch.
The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------------------
# Triton kernel: batched matmul (batch, M, K) @ (batch, K, N) → (batch, M, N)
# All tensors are contiguous, fp16 work, fp32 accumulation.
# -------------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                            # matrix sizes
    stride_batch_A, stride_batch_B, stride_batch_C,  # batch strides (in elements)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch index

    # ------------------- offsets within the tile -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ----------------------------------- pointers -------------------
    # A : (batch, M, K)   row‑major => stride_batch_A, stride_M = K, stride_K = 1
    a_ptrs = A_ptr + pid_b * stride_batch_A \
                     + offs_m[:, None] * K \
                     + tl.arange(0, BLOCK_K)[None, :]
    # B : (batch, K, N)
    b_ptrs = B_ptr + pid_b * stride_batch_B \
                     + tl.arange(0, BLOCK_K)[:, None] * N \
                     + offs_n[None, :]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------- loop over K dimension --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, mask out‑of‑bounds with zeros
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # advance pointers to the next K‑block
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K * N

    # store result (convert back to fp16 for memory efficiency)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + pid_b * stride_batch_C \
                     + offs_m[:, None] * N \
                     + offs_n[None, :]
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, c, mask=mask)


# -------------------------------------------------------------------------
# Helper: layer‑norm without torch.nn.Module (works in fp32)
# -------------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    return (x - mean) * torch.rsqrt(var + eps) * weight + bias

# Main entry point
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.
    Only the inner product (the einsum) is executed on Triton.
    All other linear / gating / layer‑norm steps stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------- LayerNorm on the input --------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
        eps=eps,
    )

    # --------------------- Linear projections (fp16) -----------------
    # Linear without bias: out = x @ weight.T
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))      # [B,N,N, H]
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

        right = right * mask_

    # -------------------------- Gates (sigmoid) ---------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    left = left * left_gate
    right = right * right_gate

    # -------------------- Triton‑accelerated inner product -----------
    # left:  [B, N, N, H]   -> (B, H, N, N) for convenient batch flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)

    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)

    M = N
    K = N

    B_ = right_perm.view(batch, K, N)
    C = torch.empty_like(A, dtype=torch.float16)               # (batch, M, N)

    # contiguous guarantees strides: (batch_stride, K, 1) etc.
    stride_batch_A = A.stride(0)
    stride_batch_B = B_.stride(0)
    stride_batch_C = C.stride(0)

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_matmul_kernel[grid](
        A.data_ptr(),
        B_.data_ptr(),
        C.data_ptr(),
        M,
        N,
        K,
        stride_batch_A,
        stride_batch_B,
        stride_batch_C,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ----------------------- to_out_norm (LayerNorm) ---------------
    out = out.to(dtype)                                          # fp32 for stability
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
        eps=eps,
    )

    out = out * out_gate.to(dtype)                               # gating (fp32)

    # --------------------------- final linear ------------------------
    out = F.linear(out, weights["to_out.weight"].to(dtype))      # [B,N,N,dim]

    return out
```
