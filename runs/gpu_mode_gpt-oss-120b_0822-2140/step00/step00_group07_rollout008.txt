"""
TriMul “outgoing” implementation with a custom Triton kernel for the
core N×N‑by‑N matrix‑multiplication per hidden channel.
The kernel computes, for every batch b and hidden dimension d,

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly the `einsum('... i k d, ... j k d -> ... i j d', …)` in the
reference PyTorch code.
All other operations (LayerNorm, linear projections, gates, final
normalisation/linear) are kept in PyTorch for simplicity.  The kernel uses
float16 for the inner GEMM and accumulates in float32, then the result is
cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (M=N, K=N, N=N) per hidden channel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    A_ptr,                     # left  : (batch, M, K)
    B_ptr,                     # right : (batch, K, N)
    C_ptr,                     # out   : (batch, M, N)
    M, N, K,                   # problem size (all equal to seq_len)
    stride_am, stride_ak,      # strides for A
    stride_bk, stride_bn,      # strides for B
    stride_cm, stride_cn,      # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication with a 2‑D grid:
        pid_m = program_id(0)  -> block row (i)
        pid_b = program_id(1)  -> batch index (b * hidden_dim)
    """
    pid_m = tl.program_id(0)          # block row
    pid_b = tl.program_id(1)          # batch * hidden_dim

    # ------------------------------------------------------------------
    # Compute offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)               # same for all blocks
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary conditions
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for the current batch
    a_ptrs = A_ptr + pid_b * stride_am * M + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bk * K + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K dimension
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (convert to fp16 for compute)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # FP16 dot, accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # Increment pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cm * M + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton GEMM kernel for the TriMul einsum
# ----------------------------------------------------------------------
def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, D)  float16
    returns      : (B, N, N, D)  float16   (still in fp16, cast later)
    """
    B, N, _, D = left.shape
    # Reshape to (B*D, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # Strides (row‑major)
    stride_am, stride_ak = left_mat.stride()[1], left_mat.stride()[2]   # (N, 1)
    stride_bk, stride_bn = right_mat.stride()[1], right_mat.stride()[2] # (N, 1)
    stride_cm, stride_cn = out_mat.stride()[1], out_mat.stride()[2]     # (N, 1)

    # Block sizes – tuned for Hopper GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), B * D)   # (row blocks, batch*D)
    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Reshape back to (B, N, N, D)
    out = out_mat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a custom Triton kernel
    for the N³ reduction.

    Args:
        data: (input, mask, weights, config)
            input   – Tensor[B, N, N, dim] (float32)
            mask    – Tensor[B, N, N]      (float32 or bool)
            weights – dict of model parameters
            config  – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in FP32 for accuracy
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]     # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)           # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also FP32
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction via Triton (compute in fp16)
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    out_f16 = tri_mul_gemm(left_f16, right_f16)          # (B,N,N,hidden_dim) fp16
    out = out_f16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
