"""
TriMul (outgoing) – Triton‑accelerated implementation.

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply after transposing the
second “N” axis of *right*.  We fuse this matmul with a custom Triton
kernel (float16 accumulation → float32 result) while keeping the
surrounding layer‑norms, gates and linear layers in PyTorch for
readability.

The algorithm
--------------
1. Layer‑norm `x` (dim = config["dim"]).
2. Linear projections → `left`, `right`   (shape [B,N,N,hidden]).
3. (Optional) mask multiplication.
4. Gate values = sigmoid(linear(x)).  Apply gates to `left`/`right`.
5. Transpose `right` on the two sequence axes and reshape both tensors
   to [B*hidden, N, N].  Call the Triton kernel to compute the batched
   matmul (float16 → float32).
6. Reshape back to [B,N,N,hidden] → `out`.
7. Layer‑norm on hidden dim, multiply by `out_gate` and final linear
   projection back to `dim`.

Only the N³ matmul is performed on the GPU with Triton – this gives a
large speed‑up while keeping the implementation short and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) for tensors of shape
#   A : [batch, M, K]
#   B : [batch, K, N]
#   C : [batch, M, N]
# Accumulate in fp32, store in fp32 (input is cast to fp16 for speed).
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_am, stride_ak,          # strides of A  (batch, M, K)
    stride_bz, stride_bk, stride_bn,          # strides of B  (batch, K, N)
    stride_cz, stride_cm, stride_cn,          # strides of C  (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 1e-6
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # --------------------------------------------------------------
    # compute offsets for the 2‑D tile this program works on
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile in A and B
    a_ptrs = a_ptr + batch_id * stride_az + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a tile of A and B (fp16)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float16)

        # dot‑product accumulation (fp32)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # write the result back
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (no bias -> bias optional)
# ----------------------------------------------------------------------
def fused_layer_norm(x, weight, bias, eps=1e-5):
    # x : [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    inv = torch.rsqrt(var + eps)
    out = (x - mean) * inv
    if weight is not None:
        out = out * weight
    if bias is not None:
        out = out + bias
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]      (float32 or bool)
            - weights: dict of torch.Tensor containing every Linear/LayerNorm weight
            - config : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = fused_layer_norm(
        x,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Apply mask if it exists (mask is broadcast over hidden)
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid of linear + optional mask)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Prepare tensors for the batched GEMM
    #     left  : [B, hidden, N, N]
    #     right : [B, hidden, N, N]  (transpose the two N axes)
    #     → reshape to [B*hidden, N, N] for the kernel
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape

    left_t = left.permute(0, 3, 1, 2).contiguous()          # [B, hidden, N, N]
    right_t = right.permute(0, 3, 2, 1).contiguous()        # [B, hidden, N, N] (swap i↔j)

    left_flat = left_t.view(B * hidden, N, N)
    right_flat = right_t.view(B * hidden, N, N)

    # allocate output buffer (float32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6️⃣  Launch Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (in elements, not bytes)
    stride_az = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bz = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cz = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # cast inputs to fp16 for the kernel (kernel will cast back internally)
    a_ptr = left_flat.to(torch.float16).contiguous()
    b_ptr = right_flat.to(torch.float16).contiguous()

    batched_gemm_kernel[grid](
        a_ptr, b_ptr, out_flat,
        N, N, N,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣  Reshape back to original 4‑D layout
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 8️⃣  Final LayerNorm, out‑gate and output linear
    # ------------------------------------------------------------------
    out = fused_layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
