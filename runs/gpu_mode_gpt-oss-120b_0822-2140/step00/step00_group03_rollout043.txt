"""
TriMul “outgoing” implementation.

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that does a blocked
matrix‑multiply for every (batch, hidden) head.
All other operations (LayerNorms, linear projections, sigmoids, masking)
are done in PyTorch for clarity.

The kernel works on tensors with layout (B, N, N, D) and uses the
following stride convention (as produced by ``torch.contiguous``):
    stride_b = N * N * D
    stride_i = N * D          # first N axis
    stride_j = D              # second N axis
    stride_d = 1
The kernel receives these strides as arguments, builds 2‑D tiles
(BLOCK_M×BLOCK_K) × (BLOCK_K×BLOCK_N) and accumulates with
FP32 precision.  The result is written back in FP32.

The overall algorithm:
1. Layer‑norm over the last dimension.
2. Linear projections (left/right) and gated linear layers.
3. Optional mask multiplication.
4. Triton‑accelerated N³ reduction.
5. Output LayerNorm, out‑gate multiplication and final linear.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for the core einsum
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,            # *float32
    right_ptr,           # *float32
    out_ptr,             # *float32
    B, N, D,             # int32  (batch, seq_len, hidden_dim)
    # strides for left  (B,N,N,D)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    # strides for right (B,N,N,D) – we will read it as (k,j) after transpose
    stride_r_b, stride_r_k, stride_r_j, stride_r_d,
    # strides for out   (B,N,N,D)
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a blocked matrix multiply for each (batch, hidden) head:

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The loop iterates over k in tiles of size BLOCK_K.
    """
    pid_head = tl.program_id(0)                     # combined (batch, hidden) index
    pid_m    = tl.program_id(1)                     # block‑row index (i)
    pid_n    = tl.program_id(2)                     # block‑col index (j)

    # --------------------------------------------------------------
    # decode batch and hidden dim from the linear head id
    # --------------------------------------------------------------
    b = pid_head // D
    d = pid_head % D

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator for the output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # reduction over k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)
        left_offset = (
            b * stride_l_b
            + d * stride_l_d
            + offs_m[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile, transposed so that we get (BLOCK_K, BLOCK_N)
        # original right indexing: right[b, j, k, d]
        # after transpose (k, j) we load with stride_k for rows and stride_j for columns
        right_offset = (
            b * stride_r_b
            + d * stride_r_d
            + offs_k[:, None] * stride_r_k   # k -> rows
            + offs_n[None, :] * stride_r_j   # j -> cols
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    out_offset = (
        b * stride_o_b
        + d * stride_o_d
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: contiguous view + stride extraction
# ----------------------------------------------------------------------
def _as_contiguous(t: torch.Tensor) -> Tuple[torch.Tensor, Tuple[int, int, int, int]]:
    """
    Returns a contiguous copy of *t* (if needed) and its 4‑dim stride tuple:
    (stride_b, stride_i, stride_j, stride_d) for layout (B, N, N, D).
    """
    if not t.is_contiguous():
        t = t.contiguous()
    strides = t.stride()               # (stride_b, stride_i, stride_j, stride_d)
    return t, strides


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑based N³ kernel.

    Args:
        data: (input, mask, weights, config)
            - input:   Tensor[B, N, N, dim]      (float32)
            - mask:    Tensor[B, N, N] or None  (float32/bool)
            - weights: dict with the exact keys used in the reference model
            - config:  dict containing at least 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – shape (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask is [B,N,N]; expand to D dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gated linear layers
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate          # (B,N,N,hidden_dim)
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton N³ reduction: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    left, left_str = _as_contiguous(left)
    right, right_str = _as_contiguous(right)

    # allocate output tensor
    out = torch.empty_like(left)     # (B,N,N,hidden_dim), float32
    out, out_str = _as_contiguous(out)

    # unpack strides (they are int64, cast to int32 for Triton)
    stride_l_b, stride_l_i, stride_l_j, stride_l_d = map(int, left_str)
    stride_r_b, stride_r_i, stride_r_j, stride_r_d = map(int, right_str)
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = map(int, out_str)

    # For the kernel we need the right tensor interpreted as (k,j) after transpose:
    #   stride_r_k = stride_r_i   (original i‑axis becomes k after transposition)
    #   stride_r_j = stride_r_j   (original j‑axis stays as j)
    stride_r_k = stride_r_i

    # Block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                 # one program per (batch, hidden) head
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trimul_outgoing_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        D=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_j,   # left's second N‑axis corresponds to k
        stride_l_d=stride_l_d,
        stride_r_b=stride_r_b,
        stride_r_k=stride_r_k,   # row stride for k
        stride_r_j=stride_r_j,   # column stride for j
        stride_r_d=stride_r_d,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_d=stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim), out‑gate multiplication
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
