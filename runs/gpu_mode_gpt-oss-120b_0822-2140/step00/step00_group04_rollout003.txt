"""
TriMul (outgoing) – fast Triton implementation

The heavy part of the TriMul forward pass is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B, N, N, H) (H = hidden_dim).  
For each hidden dimension `d` this is exactly a batched matrix‑multiply
`left[..., :, :, d] @ right[..., :, :, d].T`.  The Triton kernel below
computes this contraction tile‑wise over the (i, j) plane while keeping
the full hidden‑dimension vector inside a tile (BLOCK_H).  All other
operations (LayerNorms, Linear projections, gates, masking and the final
output Linear) are performed with regular PyTorch kernels – they are
memory‑bound and cheap compared with the O(N³·H) contraction.

The kernel works for any batch size (B≤2), sequence length N up to 1024,
and hidden dimension H that is a multiple of BLOCK_H (the remainder is
handled by masking).  The final tensor is returned in `torch.float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # tensor sizes
    stride_left_pb, stride_left_pi, stride_left_pk, stride_left_ph,
    stride_right_pb, stride_right_pj, stride_right_pk, stride_right_ph,
    stride_out_pb, stride_out_pi, stride_out_pj, stride_out_ph,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # program IDs – three‑dimensional grid: (batch, i‑tile, j‑tile)
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # ----------------------------------------------------------
    # Offsets for the current tile
    # ----------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator: (BLOCK_M, BLOCK_N, BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over the reduction dimension k
    # ----------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load a tile of left  : shape (M, K, H)
        # left layout = (b, i, k, h)  (contiguous)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_pb
            + (offs_i[:, None, None] * stride_left_pi)
            + (offs_k[None, :, None] * stride_left_pk)
            + (tl.arange(0, BLOCK_H)[None, None, :] * stride_left_ph)
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (M, K, H)

        # ------------------------------------------------------------------
        # Load a tile of right : shape (N, K, H)
        # right layout = (b, j, k, h)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_pb
            + (offs_j[:, None, None] * stride_right_pj)
            + (offs_k[None, :, None] * stride_right_pk)
            + (tl.arange(0, BLOCK_H)[None, None, :] * stride_right_ph)
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (N, K, H)

        # ------------------------------------------------------------------
        # Accumulate: for each hidden channel h,
        #   acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]
        # We do this by broadcasting the (M,K,H) and (N,K,H) tensors
        # over the K dimension and performing an element‑wise product.
        # ------------------------------------------------------------------
        # Expand dimensions to (M,1,K,H) and (1,N,K,H) then multiply
        # and sum over K.
        prod = left[:, None, :, :] * right[None, :, :, :]          # (M,N,K,H)
        acc += tl.sum(prod, axis=2)                               # (M,N,H)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_pb
        + (offs_i[:, None, None] * stride_out_pi)
        + (offs_j[None, :, None] * stride_out_pj)
        + (tl.arange(0, BLOCK_H)[None, None, :] * stride_out_ph)
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# --------------------------------------------------------------
# Host‑side helper that launches the kernel
# --------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right:  (B, N, N, H)  float32
    returns out : (B, N, N, H)  float32
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # ------------------------------------------------------
    # Strides (in elements, not bytes)
    # ------------------------------------------------------
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # left: (b,i,k,h)
    stride_left_pb, stride_left_pi, stride_left_pk, stride_left_ph = s_left
    # right: (b,j,k,h) – same physical layout, just rename axes
    stride_right_pb, stride_right_pj, stride_right_pk, stride_right_ph = s_right
    # out: (b,i,j,h)
    stride_out_pb, stride_out_pi, stride_out_pj, stride_out_ph = s_out

    # ------------------------------------------------------
    # Tile sizes – tuned for Hopper
    # ------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32  # hidden dimension tile

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_pb,
        stride_left_pi,
        stride_left_pk,
        stride_left_ph,
        stride_right_pb,
        stride_right_pj,
        stride_right_pk,
        stride_right_ph,
        stride_out_pb,
        stride_out_pi,
        stride_out_pj,
        stride_out_ph,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    return out


# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Fast Triton‑based forward pass for the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        - mask_tensor : torch.Tensor   [B, N, N]        (float32 or bool)
        - weights_dict : dict of parameter tensors (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------
    x = input_tensor
    # mean / var over last dimension
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------
    # 3. Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------
    # 4. Gates (sigmoid) and gating
    # ------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------
    # 5. Core contraction – Triton kernel
    # ------------------------------------------------------
    out_hidden = tri_mul_triton(left, right)          # (B, N, N, H)

    # ------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim)
    # ------------------------------------------------------
    mean_o = out_hidden.mean(dim=-1, keepdim=True)
    var_o = out_hidden.var(dim=-1, unbiased=False, keepdim=True)
    out_hidden_norm = (out_hidden - mean_o) * torch.rsqrt(var_o + eps)
    out_hidden_norm = out_hidden_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------
    # 7. Apply out_gate and final linear projection back to dim
    # ------------------------------------------------------
    out_hidden_norm = out_hidden_norm * out_gate           # gating after norm
    out = F.linear(out_hidden_norm, weights["to_out.weight"])   # (B,N,N,dim)

    return out.to(torch.float32)
