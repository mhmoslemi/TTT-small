"""
TriMul “outgoing” implementation optimized with a Triton batched‑matrix‑multiply
kernel.

The heavy part of the module is the N‑wise contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every hidden dimension d (called hidden_dim in the config).  By reshaping
the tensors to (B*hidden, N, N) we turn this into a standard batch‑matmul
between a matrix A and the transpose of B.  The kernel below computes that
matmul in FP32 using a tiling strategy (BLOCK_M × BLOCK_N tiles, inner
reduction over BLOCK_K).  All other operations (layer‑norms, linear layers,
sigmoids, masking) are performed with regular PyTorch which is cheap compared
to the O(N³·hidden) work.

The kernel runs on a 3‑D grid:
    dim0 – batch × hidden (treated as independent matmul problems)
    dim1 – tiles over the output rows (M)
    dim2 – tiles over the output columns (N)

The final output is reshaped back to [B, N, N, hidden] and the remaining
LayerNorm / gating / linear projection are applied in PyTorch.  The function
`custom_kernel` follows the required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where
#   A : (B*H, M, K)
#   B : (B*H, K, N)
#   C : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # problem size
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """FP32 batched matmul.  Each program instance processes one (batch,head)
    pair and a BLOCK_M × BLOCK_N tile of the output."""
    batch_id = tl.program_id(0)          # which (batch * hidden) slice
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds loads / stores
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ------------------------------------------------------------------
    # Pointers to the start of the tile
    # ------------------------------------------------------------------
    a_tile_ptr = (
        a_ptr
        + batch_id * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_tile_ptr = (
        b_ptr
        + batch_id * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_tile_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_tile_ptr += BLOCK_K * stride_a_k
        b_tile_ptr += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_tile_ptr = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_tile_ptr, acc, mask=mask)


# ----------------------------------------------------------------------
# Entry point that follows the required signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (torch.Tensor, CUDA)
            - mask         : [B, N, N]        (torch.Tensor, CUDA) – may be all‑ones
            - weights      : dict of parameter tensors
            - config       : dict with keys
                * dim          : int  (input feature dimension)
                * hidden_dim   : int  (inner hidden dimension)
                * nomask       : bool – True ⇒ ignore mask (mask may be None)
    Returns:
        output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton
    #    reshape to (B*hidden, N, N) and compute A @ B.T
    # ------------------------------------------------------------------
    #   left : (B,N,N,hidden)          -> (B*hidden, N, N)
    #   right: (B,N,N,hidden) -> transpose K and J -> (B,N,hidden,N)
    #          -> (B*hidden, N, N)
    lhs = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)         # (BH, M, K)
    rhs = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)        # (BH, K, N)

    # Output buffer
    out_buf = torch.empty_like(lhs)

    # Strides (in elements, not bytes)
    a_strides = lhs.stride()   # (batch, M, K)
    b_strides = rhs.stride()   # (batch, K, N)
    c_strides = out_buf.stride()

    # Tiling parameters – tuned for Hopper SM90, FP32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        lhs,
        rhs,
        out_buf,
        N,
        N,
        N,
        a_strides[0],
        a_strides[1],
        a_strides[2],
        b_strides[0],
        b_strides[1],
        b_strides[2],
        c_strides[0],
        c_strides[1],
        c_strides[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original 4‑D layout
    out = out_buf.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
