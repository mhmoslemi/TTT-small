"""
TriMul (outgoing) – fast Triton implementation
================================================
The heavy part of the TriMul forward pass is the N‑by‑N “einsum”

    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

which is equivalent to a batched matrix multiplication per hidden channel *d* :

    out_b,d = left_b,d @ right_b,d.T          (N × N) @ (N × N)ᵀ

We fuse this operation in a custom Triton kernel.  The surrounding
layer‑norms, linear projections and gating are inexpensive and stay in
PyTorch.  The kernel processes a tile (BLOCK_I × BLOCK_J) of the output
matrix for a single (batch, hidden‑channel) pair, accumulating over the
*k* dimension in blocks of size BLOCK_K.  The kernel runs in FP16 for the
loads, accumulates in FP32 and writes FP32, matching the reference
implementation.

The final output is post‑processed with a second LayerNorm, an output gate
and a linear projection back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           #       float32 *
    B, N, H,           # int32 scalars
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # 1) identify batch (b) and hidden channel (h) for this program
    # ---------------------------------------------------------------
    pid0 = tl.program_id(0)          # (b * H) * ceil(N / BLOCK_I) + tile_i
    pid1 = tl.program_id(1)          # tile_j

    num_i_tiles = tl.cdiv(N, BLOCK_I)
    bh = pid0 // num_i_tiles          # flat index over (batch, hidden)
    tile_i = pid0 % num_i_tiles

    b = bh // H                       # batch index
    h = bh % H                        # hidden‑channel index

    # ---------------------------------------------------------------
    # 2) tile offsets
    # ---------------------------------------------------------------
    offs_i = tile_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid1 * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the (BLOCK_I × BLOCK_J) tile
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ---------------------------------------------------------------
    # 3) loop over the reduction dimension k in blocks
    # ---------------------------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]   → shape (BLOCK_I, BLOCK_K)
        left = tl.load(
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right[b, j, k, h]  → shape (BLOCK_J, BLOCK_K)
        right = tl.load(
            right_ptr
            + b * stride_rb
            + offs_j[:, None] * stride_ri
            + offs_k[None, :] * stride_rk
            + h * stride_rh,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # matrix‑multiply: (BLOCK_I × BLOCK_K) @ (BLOCK_J × BLOCK_K)ᵀ
        # tl.dot returns FP32 even when inputs are FP16
        acc += tl.dot(left, right, trans_b=True)

    # ---------------------------------------------------------------
    # 4) write the tile back
    # ---------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    einsum step.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]               # called H in the paper

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension ``dim``
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → shape [B, N, N, H]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # (H, dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask handling (optional)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                         # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton accelerated einsum (N³ reduction)
    #    We cast to fp16 for the kernel loads, keep accumulation in fp32.
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements) for pointer arithmetic inside the kernel
    s_lb, s_li, s_lk, s_lh = left_h.stride()
    s_rb, s_ri, s_rk, s_rh = right_h.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tile sizes – tuned for Hopper SM90, keep them powers of two
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I) * B * hidden_dim,   # program_id(0)
        triton.cdiv(N, BLOCK_J),                    # program_id(1)
    )
    trimul_out_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension, output‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
