"""
TriMul “outgoing” kernel (AlphaFold‑3) – Triton + PyTorch hybrid.

The expensive part of TriMul is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiply per hidden dimension:
    (left)   : (B, H, N, N)         ← (B,N,N,H) view
    (rightᵀ) : (B, H, N, N) (right transposed on the last two axes)

We compute the rest (LayerNorm, linear projections, gating) with torch
because they are cheap, then launch a custom Triton GEMM that
processes **B × hidden_dim** independent N×N matrix products.
The kernel works in FP32 (the model runs in float32) and returns a
contiguous tensor ready for the final LayerNorm + linear.

The kernel is launch‑parameterised by
    BLOCK_M, BLOCK_N, BLOCK_K – chosen for H200 (128/128/32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: [batch, M, K] , [batch, K, N])
# -------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem size
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bk, stride_bn,   # B strides: batch, K, N
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # -------------------------------------------------------------
    # 1) map linear pid → (batch, block_m, block_n)
    # -------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # -------------------------------------------------------------
    # 2) tile offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for edge handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # 3) pointer arithmetic (add batch offset first)
    # -------------------------------------------------------------
    a_ptrs = a_ptr + batch * stride_ah \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak

    b_ptrs = b_ptr + batch * stride_bh \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # 4) main K‑loop
    # -------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] + _k * BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] + _k * BLOCK_K < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # 5) write‑back
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch * stride_ch \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------
# Helper for LayerNorm without bias (bias handled separately)
# -------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    # x: [..., C]  weight/bias: [C]
    return F.layer_norm(x, normalized_shape=x.shape[-1:], weight=weight, bias=bias, eps=1e-5)


# -------------------------------------------------------------------
# Entry point required by the evaluator
# -------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a custom Triton GEMM.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim]  (float32, on CUDA)
        mask    – Tensor[B, N, N]       (float32 or bool, on CUDA)
        weights – dict with the nine weight / bias tensors
        config  – dict with keys ``dim`` and ``hidden_dim`` (ints)
    Returns
    -------
    Tensor[B, N, N, dim]  (float32, on CUDA)
    """
    # -------------------------------------------------------------------
    # unpack
    # -------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = inp.shape

    # -------------------------------------------------------------------
    # 1) Input LayerNorm
    # -------------------------------------------------------------------
    x = _layer_norm(inp,
                    weight=weights["norm.weight"].to(torch.float32),
                    bias=weights["norm.bias"].to(torch.float32))

    # -------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"].to(torch.float32))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float32))

    # -------------------------------------------------------------------
    # 3) Optional mask (mask is broadcast on the hidden dimension)
    # -------------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_unsq
        right = right * mask_unsq

    # -------------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # -------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float32)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float32)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float32)))

    # -------------------------------------------------------------------
    # 5) Apply gates
    # -------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM
    #    left  : (B, N, N, H) → (B*H, N, N)
    #    rightᵀ: (B, N, N, H) → (B*H, N, N) after transposition on the last two axes
    # -------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()

    # transpose the last two axes of right to get the correct contraction pattern
    right_t = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    # reshape to a batch of independent matrices
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, M, K)
    right_mat = right_t.view(B * hidden_dim, N, N)                               # (B*H, K, N)

    # -------------------------------------------------------------------
    # 7) Launch Triton GEMM
    # -------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    # strides (in elements) for the three‑dimensional tensors
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    # output tensor
    out_mat = torch.empty_like(left_mat)  # (B*H, N, N)
    stride_ch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    total_blocks = (triton.cdiv(N, BLOCK_M) *
                    triton.cdiv(N, BLOCK_N) *
                    (B * hidden_dim))

    grid = (total_blocks,)

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N, N=N, K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # reasonable default for H200
    )

    # -------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, hidden_dim)
    # -------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------
    # 9) Output LayerNorm, gating and final linear projection
    # -------------------------------------------------------------------
    out = _layer_norm(out,
                      weight=weights["to_out_norm.weight"].to(torch.float32),
                      bias=weights["to_out_norm.bias"].to(torch.float32))

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(torch.float32))   # (B,N,N,dim)

    return out
