"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix multiplication
(left @ rightᵀ) for every hidden dimension * d* and every batch.
We fuse the batch*hidden loop into a single Triton kernel that
computes C = A @ Bᵀ where A and B have shape (M,N) with M = N = seq_len.
All other operations (LayerNorms, Linear + gating, masking and the final
Linear) stay in PyTorch – they are cheap compared with the O(N³) work.
The kernel works with fp16 inputs and accumulates in fp32, finally
storing fp16 results that are later converted back to fp32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batch‑hidden fused)
# ---------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (M=N=seq_len, K=seq_len)
    stride_am, stride_ak,               # strides for A (M‑dim, K‑dim)
    stride_bk, stride_bn,               # strides for B (K‑dim, N‑dim) – note B is *not* transposed
    stride_cm, stride_cn,               # strides for C (M‑dim, N‑dim)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ Bᵀ for one (batch * hidden) slice.
    A : (M, K)    – stored row‑major
    B : (N, K)    – stored row‑major (we read it transposed)
    C : (M, N)
    """
    pid = tl.program_id(0)          # 2‑D grid of (M‑blocks, N‑blocks)
    pid_z = tl.program_id(1)        # fused batch*hidden index

    # -----------------------------------------------------------------
    # Block coordinates
    # -----------------------------------------------------------------
    num_block_m = tl.cdiv(M, BLOCK_M)
    num_block_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_block_n           # block row index
    pid_n = pid %  num_block_n           # block col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # Pointers for the current slice (pid_z)
    # -----------------------------------------------------------------
    # A is (M, K)
    a_base = a_ptr + pid_z * stride_am * M
    # B is (N, K) but we need Bᵀ → we will load with swapped strides
    b_base = b_ptr + pid_z * stride_bn * N

    # -----------------------------------------------------------------
    # Accumulator
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Main K‑loop
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # ----- load A block (M×K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am
                           + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & ((cur_k + tl.arange(0, BLOCK_K)) < K),
                    other=0.0)

        # ----- load Bᵀ block (K×N) == reading B as (N, K) transposed
        b_ptrs = b_base + (offs_n[None, :] * stride_bn
                           + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk)
        b = tl.load(b_ptrs,
                    mask=mask_n[None, :] & ((cur_k + tl.arange(0, BLOCK_K)) < K),
                    other=0.0)

        # ----- dot‑product
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # -----------------------------------------------------------------
    # Store result
    # -----------------------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_cm * M \
                   + offs_m[:, None] * stride_cm \
                   + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : Tuple[Tensor, Tensor, Dict[str, Tensor], Dict]
        input   – (B, N, N, dim)   float32
        mask    – (B, N, N)        (may be all‑ones if config[\"nomask\"])
        weights – mapping from the module’s parameter names to tensors
        config  – dict containing at least
                  dim, hidden_dim, nomask (bool)

    Returns
    -------
    Tensor of shape (B, N, N, dim)  float32
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    device = inp.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(inp,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)          # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dim)
    # -----------------------------------------------------------------
    if not cfg.get("nomask", True):
        mask_ = mask.unsqueeze(-1)                # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) and apply gates
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Batched GEMM left @ rightᵀ  (the N³ part)
    # -----------------------------------------------------------------
    # fp16 for speed, fp32 accumulation inside the kernel
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape                 # B = batch size

    # reshape to (B*hidden, N, N) and make the hidden dim contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left  = left.view(-1, N, N)                     # (B*hidden, N, N)
    right = right.view(-1, N, N)

    # output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # strides for Triton (int64 -> Python int)
    stride_am, stride_ak = left.stride()    # (M‑stride, K‑stride) -> (N, 1)
    stride_bk, stride_bn = right.stride()   # (K‑stride, N‑stride) -> (1, N)
    stride_cm, stride_cn = out.stride()

    # block size – 128 works well on Hopper for fp16 GEMM
    BLOCK = 128
    grid_m = triton.cdiv(N, BLOCK)
    grid_n = triton.cdiv(N, BLOCK)
    grid_z = left.shape[0]                  # B * hidden
    grid = (grid_m * grid_n, grid_z)

    tri_mul_kernel[grid](
        left, right, out,
        N, N, N,                         # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK, BLOCK_N=BLOCK, BLOCK_K=BLOCK,
    )

    # reshape back to (B, N, N, hidden)
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden dimension)
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    # -----------------------------------------------------------------
    # 7) multiply by out_gate and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate                     # (B,N,N,hidden) broadcast
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    # ensure final dtype = float32 (as required by the spec)
    return out.to(torch.float32)
