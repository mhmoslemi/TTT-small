"""
TriMul “outgoing” implementation with a custom Triton kernel.

The costly part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply L_d @ R_d.T for every hidden dimension d.
The kernel below computes C = A @ Bᵀ for many independent matrices
(group = batch * hidden_dim).  All other operations (LayerNorms, Linear
projections, gates, etc.) are done in PyTorch – they are cheap compared to the
N³ work.

The kernel works on FP16 inside the matmul (the inputs are cast to half) and
writes back FP16; the wrapper converts the result back to FP32 before the
remaining FP32 post‑processing.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ  (A,B shape: [G, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_transpose_kernel(
    a_ptr,          # *float16   A [G, M, K]
    b_ptr,          # *float16   B [G, N, K] (will be used as Bᵀ)
    c_ptr,          # *float16   C [G, M, N]
    G, M, N, K,
    stride_ag, stride_am, stride_ak,      # strides for A
    stride_bg, stride_bn, stride_bk,      # strides for B (note: B is stored row‑major)
    stride_cg, stride_cm, stride_cn,      # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a tile of C = A @ Bᵀ for a single group."""
    gid = tl.program_id(2)                     # which group (batch * hidden)
    pid_m = tl.program_id(0)                   # block row
    pid_n = tl.program_id(1)                   # block col

    # ----- offsets -------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this group
    a_base = a_ptr + gid * stride_ag
    b_base = b_ptr + gid * stride_bg
    c_base = c_ptr + gid * stride_cg

    # pointers for this tile
    a_ptrs = a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_base + (offs_n[:, None] * stride_bn + offs_k[None, :] * stride_bk)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- k‑loop ---------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        # B is used transposed: we need B[j,k] -> we loaded rows j, cols k directly
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k[None, :] < K), other=0.0)

        # dot: (M x K) @ (N x K)^T  ==  Σ_k a[i,k] * b[j,k]
        acc += tl.dot(a, b)                    # both are FP16, accumulation in FP32

        # move to next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----- store ----------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton matmul
# ----------------------------------------------------------------------
def triton_batch_matmul_transpose(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b: [G, N, N]   (float16)
    returns c: [G, N, N]  (float16)
    Computes c = a @ bᵀ per group.
    """
    assert a.dtype == b.dtype == torch.float16
    G, M, K = a.shape
    _, N, _ = b.shape
    assert M == N == K

    # output
    c = torch.empty_like(a)

    # strides (in elements)
    stride_ag = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_bg = b.stride(0)
    stride_bn = b.stride(1)
    stride_bk = b.stride(2)

    stride_cg = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    # tile sizes – tuned for Hopper (SM90)  (fits in 64KB shared mem)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        G,
    )

    batched_matmul_transpose_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        G=G,
        M=M,
        N=N,
        K=K,
        stride_ag=stride_ag,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bg=stride_bg,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_cg=stride_cg,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# The public entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the "outgoing" TriMul operator.
    Only the N³ einsum is executed on GPU via Triton; everything else
    uses regular PyTorch (FP32) for numerical stability.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + eps
    )
    x = x * norm_weight + norm_bias

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)  -> shape [..., hidden_dim]
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Apply mask if present (mask is [bs, N, N])
    # --------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)               # [bs, N, N, 1]
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------------
    # 4) Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Heavy N³ part – batched matmul L @ Rᵀ per hidden dimension
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                     # B == batch size
    # reshape to [B*hidden_dim, N, N] and cast to fp16 for the kernel
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)

    # Triton kernel
    out_ = triton_batch_matmul_transpose(left_, right_)   # fp16 result

    # reshape back to [B, N, N, hidden_dim] and cast to fp32
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gate, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
