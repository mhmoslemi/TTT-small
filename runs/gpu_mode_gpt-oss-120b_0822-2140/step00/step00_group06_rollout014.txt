"""
TriMul “outgoing” implementation.
The heavy part is the N‑dimensional 3‑tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply
    out_d = left_d @ right_dᵀ   (left_d,right_d : [B,N,N])

The kernel computes this GEMM for each hidden channel `d` in parallel.
All other operations (LayerNorm, linear projections, gating, final
normalisation) are kept in PyTorch because they are cheap compared to the
O(N³) contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # const float*  left  [B,N,N,H]
    right_ptr,         # const float*  right [B,N,N,H]
    out_ptr,           #       float*  out   [B,N,N,H]

    B, N, H,           # sizes (int32)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_N: tl.constexpr,   # tile size on i/j dimensions
    BLOCK_K: tl.constexpr,   # tile size on reduction dimension
):
    """Compute out = left @ rightᵀ for each hidden slice."""
    pid_i = tl.program_id(0)          # i‑block
    pid_j = tl.program_id(1)          # j‑block
    pid_b = tl.program_id(2)          # batch
    pid_h = tl.program_id(3)          # hidden channel

    # ------------------------------------------------------------------- #
    # Offsets for the current tile
    # ------------------------------------------------------------------- #
    i_off = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for OOB handling
    mask_i = i_off < N
    mask_j = j_off < N

    # pointers to the first element of this (b,h) slice
    left_base  = left_ptr  + pid_b * stride_left_b  + pid_h * stride_left_h
    right_base = right_ptr + pid_b * stride_right_b + pid_h * stride_right_h
    out_base   = out_ptr   + pid_b * stride_out_b   + pid_h * stride_out_h

    acc = tl.zeros((BLOCK_N, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Reduction over k
    # ------------------------------------------------------------------- #
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left  : (i,k)   shape (BLOCK_N, BLOCK_K)
        left_ptrs = left_base + \
                    i_off[:, None] * stride_left_i + \
                    cur_k[None, :] * stride_left_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # rightᵀ : (k,j)   shape (BLOCK_K, BLOCK_N)
        # note the swapped strides: we stride first over k, then over j.
        right_ptrs = right_base + \
                     cur_k[:, None] * stride_right_k + \
                     j_off[None, :] * stride_right_j
        b_t = tl.load(right_ptrs,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0)

        # GEMM on the tile
        acc += tl.dot(a, b_t)          # (BLOCK_N, BLOCK_N)

    # ------------------------------------------------------------------- #
    # Write back
    # ------------------------------------------------------------------- #
    out_ptrs = out_base + \
               i_off[:, None] * stride_out_i + \
               j_off[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input  : [B, N, N, dim]   (float32, on CUDA)
        - mask   : [B, N, N]        (float32 or bool)
        - weights: dict of weight tensors (all float32)
        - config : dict containing
            * dim        – input feature dimension
            * hidden_dim – hidden dimension (projection size)
            * eps        – LayerNorm epsilon (optional, default 1e-5)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------- #
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(x,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # ------------------------------------------------------------------- #
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------- #
    # shape: [B,N,N,hidden] = x @ Wᵀ
    left = torch.matmul(x, weights["left_proj.weight"].T)          # (B,N,N,hidden)
    right = torch.matmul(x, weights["right_proj.weight"].T)

    # ------------------------------------------------------------------- #
    # Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(torch.matmul(x, weights["left_gate.weight"].T))
    right_gate = torch.sigmoid(torch.matmul(x, weights["right_gate.weight"].T))
    out_gate = torch.sigmoid(torch.matmul(x, weights["out_gate.weight"].T))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Allocate output tensor for the contraction
    # ------------------------------------------------------------------- #
    out = torch.empty((B, N, N, hidden), dtype=torch.float32,
                      device=x.device)

    # ------------------------------------------------------------------- #
    # Prepare strides for Triton (in elements)
    # ------------------------------------------------------------------- #
    stride_left = left.stride()      # (B*N*N*H, N*N*H, N*H, H) -> (Lb, Li, Lk, Lh)
    stride_right = right.stride()
    stride_out = out.stride()

    # launch parameters – tile sizes tuned for Hopper SM90
    BLOCK_N = 64      # along i and j
    BLOCK_K = 64      # reduction tile

    grid = (triton.cdiv(N, BLOCK_N),
            triton.cdiv(N, BLOCK_N),
            B,
            hidden)

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],   # note: j is the second spatial axis
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                      # enough for fp32 compute
    )

    # ------------------------------------------------------------------- #
    # Hidden‑dim LayerNorm + gating
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate                     # broadcast over hidden

    # ------------------------------------------------------------------- #
    # Final linear back to dim
    # ------------------------------------------------------------------- #
    out = torch.matmul(out, weights["to_out.weight"].T)   # (B,N,N,dim)

    return out
