"""
TriMul “outgoing” implementation with a custom Triton kernel.

The heavy part of the TriMul forward pass is the contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which can be expressed as a batched matrix‑multiply per hidden channel:
    out_h = left_h @ right_h.T     (left_h, right_h ∈ ℝ^{N×N})

The kernel below computes this batched GEMM for all (batch, hidden) pairs.
All other ops (LayerNorm, Linear, gating, final Linear) are done in PyTorch
so we keep the code simple while still moving the N³ work to Triton.

The kernel works for any batch ≤2, sequence length up to 1024 and hidden
dimension ≤128.  It uses float32 accumulation (inputs may be cast to float16
for higher throughput on Hopper, but the final result is stored in float32).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul left @ right.T  (both have shape B×N×N×H)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # ------------------------------------------------------------------
    # Decode batch (b) and hidden channel (h) from pid_bh
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # Compute offsets for the current tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers for the left (i,k) and right (j,k) blocks.
    #   left  : shape (B, N, N, H)  -> indices (b, i, k, h)
    #   right : shape (B, N, N, H)  -> indices (b, j, k, h)
    # ------------------------------------------------------------------
    # Linear offset helpers (k will be added inside the loop)
    left_base = left_ptr + b * stride_lb + h * stride_lh + offs_i[:, None] * stride_li
    right_base = right_ptr + b * stride_rb + h * stride_rh + offs_j[None, :] * stride_rj

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles of size BLOCK_K
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left[i,k] and right[j,k] blocks (shape BLOCK_M×BLOCK_K and BLOCK_N×BLOCK_K)
        left = tl.load(
            left_base + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        right = tl.load(
            right_base + offs_k[:, None] * stride_rk,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0,
        )
        # dot: (M×K)·(N×K)ᵀ  →  (M×N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Write out the result tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        input_tensor : [B, N, N, dim]    (float32)
        mask         : [B, N, N]          (float32 or bool)
        weights      : dict with all Linear/LayerNorm parameters
        config       : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton kernel
    # ------------------------------------------------------------------
    # Allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (convert to int64 for Triton)
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb=l_str[0],
        stride_li=l_str[1],
        stride_lk=l_str[2],
        stride_lh=l_str[3],
        stride_rb=r_str[0],
        stride_rj=r_str[1],
        stride_rk=r_str[2],
        stride_rh=r_str[3],
        stride_ob=o_str[0],
        stride_oi=o_str[1],
        stride_oj=o_str[2],
        stride_oh=o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
