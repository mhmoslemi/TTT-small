"""
TriMul “outgoing” kernel (AlphaFold3).

Algorithm
---------

1.  Layer‑norm the 4‑D input `x`  →  (B, N, N, C).
2.  Linear projections `left_proj`, `right_proj` (C → H) produce
    `left` and `right` of shape (B, N, N, H).
3.  Optional mask (B, N, N, 1) is broadcast‑multiplied into left/right.
4.  Compute three gating vectors (C → H) → sigmoid:
       `left_gate`, `right_gate`, `out_gate`.
5.  Fuse the element‑wise multiplication
        left  = left  * left_gate
        right = right * right_gate
    with a tiny Triton kernel (fully vectorised, no Python loops).
6.  The core N³ operation:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is equivalent to a batched matrix‑multiply for each hidden
    channel:
        reshape left/right to (B·H, N, N) and perform
        out = L @ Rᵀ   (using Torch’s highly‑optimised cuBLAS GEMM).
7.  Reshape back → (B, N, N, H) and apply a second LayerNorm.
8.  Multiply by `out_gate` (broadcast) and a final linear
    (H → C) to obtain the result (B, N, N, C).

Only steps 5 (gate‑multiplication) are performed in Triton; the rest
use PyTorch primitives which already achieve near‑optimal performance
on Hopper GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of a tensor with its gate.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mul_kernel(
    tensor_ptr,          # float*  (contiguous)
    gate_ptr,            # float*  (contiguous, same shape)
    n_elements,          # int64   total number of elements
    BLOCK: tl.constexpr  # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements

    t = tl.load(tensor_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)
    tl.store(tensor_ptr + offs, t * g, mask=mask)


def _apply_gate(tensor: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """In‑place multiply ``tensor`` by ``gate`` using Triton."""
    assert tensor.is_contiguous() and gate.is_contiguous()
    total = tensor.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    _gate_mul_kernel[grid](
        tensor_ptr=tensor,
        gate_ptr=gate,
        n_elements=total,
        BLOCK=BLOCK,
    )
    return tensor


# ----------------------------------------------------------------------
# Main entry‑point required by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor   [B, N, N, C]   (float32, CUDA)
            - mask  : torch.Tensor   [B, N, N]     (float32, CUDA) or None
            - weights: dict of torch.Tensor with the exact keys used in the
                       reference PyTorch implementation.
            - config : dict with keys ``dim`` (C) and ``hidden_dim`` (H).

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack ------------------------------------------------------------
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # aliases for weight tensors (all are float32, already on device)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (C → H) – bias‑free, keep fp32
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = F.linear(x, left_proj_w)          # [B, N, N, H]
    right = F.linear(x, right_proj_w)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (C → H) + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Fuse gate multiplication with Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _apply_gate(left, left_gate)
    _apply_gate(right, right_gate)

    # ------------------------------------------------------------------
    # 6. Core N³ contraction via batched GEMM (cuBLAS)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).reshape(B * H, N, N)       # (B·H, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * H, N, N)     # (B·H, N, N)

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))

    # Restore original layout (B, N, N, H)
    out = out_perm.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm (over hidden dimension H)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Multiply by out_gate (broadcast) and final linear (H → C)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)      # [B, N, N, C]

    return out.float()
