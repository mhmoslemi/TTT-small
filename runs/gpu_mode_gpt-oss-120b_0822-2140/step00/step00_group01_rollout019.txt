"""
TriMul “outgoing” implementation.

The heavy O(N³) part is the einsum
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is exactly a batched matrix multiplication
    out = left @ rightᵀ   (per batch * hidden dimension).

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
but fuse the large matrix‑multiply into a single Triton kernel.
The kernel works for any batch size B∈{1,2}, sequence length N up to 1024
and hidden dimension H (=config["hidden_dim"], typically 128). It
produces a float‑32 output as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   where left/right have shape
#   [B, N, N, H]  (last dimension = hidden)
# The kernel is launched with a 3‑D grid:
#   dim0 : (B * H)   – each program handles one (batch, hidden) slice
#   dim1 : ceil(N / BLOCK_M) – tiles over the i‑dimension
#   dim2 : ceil(N / BLOCK_N) – tiles over the j‑dimension
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # const *float32
    right_ptr,              # const *float32
    out_ptr,                # *float32
    B, N, H,                # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size in i direction
    BLOCK_N: tl.constexpr,  # tile size in j direction
    BLOCK_K: tl.constexpr,  # tile size in k (reduction) direction
):
    # --------------------------------------------------------------
    # 1) identify the (batch, hidden) slice this program works on
    # --------------------------------------------------------------
    pid = tl.program_id(0)               # linear index over B*H
    b = pid // H                         # batch index
    h = pid % H                          # hidden‑channel index

    # base pointers for the selected (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --------------------------------------------------------------
    # 2) tile indices for the output matrix (i, j)
    # --------------------------------------------------------------
    pid_m = tl.program_id(1)                         # tile row
    pid_n = tl.program_id(2)                         # tile col
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M) # i positions
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N) # j positions
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator for a BLOCK_M × BLOCK_N tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left block: shape (BLOCK_M, BLOCK_K) -----------------
        # left layout: (i, k) with strides (stride_left_i, stride_left_k)
        left_ptrs = left_base + offs_m[:, None] * stride_left_i \
                               + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- right block (transposed) : shape (BLOCK_K, BLOCK_N) -
        # we need rightᵀ, i.e. access right[j, k] as (k, j)
        # right layout: (i, k) with strides (stride_right_i, stride_right_k)
        # after transposition the stride for the first axis becomes stride_right_i
        right_ptrs = right_base + offs_k[:, None] * stride_right_i \
                                 + offs_n[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # ---- dot product over the K dimension --------------------
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # --------------------------------------------------------------
    # 4) write the tile back
    # --------------------------------------------------------------
    out_ptrs = out_base + offs_m[:, None] * stride_out_i \
                         + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: layer‑norm without bias‑gradient (pure forward)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass.

    Arguments
    ----------
    data :
        (input, mask, weights, config)
        - input : torch.Tensor  [B, N, N, D]   (float32 on CUDA)
        - mask  : torch.Tensor  [B, N, N]      (bool / float, may be all‑ones)
        - weights : dict of pre‑loaded parameters
        - config  : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden size (often 128)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim = D)
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor,
                    weights["norm.weight"],
                    weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, H]
    left = F.linear(x, weights["left_proj.weight"])   # (out_features=H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy part – compute out = left @ rightᵀ per hidden channel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)      # [B,N,N,H] float32

    # strides for the 4‑D tensors (contiguous)
    stride_left = left.stride()       # (sB, sI, sK, sH)
    stride_right = right.stride()
    stride_out = out.stride()

    # launch parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_i=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm over hidden, gating, final Linear
    # ------------------------------------------------------------------
    out = _layer_norm(out,
                      weights["to_out_norm.weight"],
                      weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,D]

    return out
