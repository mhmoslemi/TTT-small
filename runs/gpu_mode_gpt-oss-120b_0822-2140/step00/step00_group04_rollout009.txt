"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batch of matrix multiplications
left[b, :, :, d] @ right[b, :, :, d].T  for each hidden channel `d`.
The implementation below

* applies LayerNorm / Linear / sigmoid gates in PyTorch (light‑weight)
* launches a Triton kernel that computes the N³ contraction in FP16
  (blocked over the sequence dimension and a small slice of the hidden
   channel dimension to stay inside register limits)
* finishes the forward pass in PyTorch (LayerNorm + final Linear).

The kernel is written for Hopper (sm_90) but works on any recent GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ---------------------------------------------------------------
# Triton kernel: block‑wise outer‑product reduction over the k‑axis.
# ---------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute a tiled outer‑product sum:
       out[b,i,j,h] += Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid = tl.program_id(0)

    # -----------------------------------------------------------
    # decode flat program id -> (b, block_i, block_j, block_h)
    # -----------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(H, BLOCK_H)

    b = pid // (num_i * num_j * num_h)
    tmp = pid % (num_i * num_j * num_h)

    block_i = tmp // (num_j * num_h)
    tmp = tmp % (num_j * num_h)

    block_j = tmp // num_h
    block_h = tmp % num_h

    # -----------------------------------------------------------
    # coordinate ranges for this tile
    # -----------------------------------------------------------
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h = block_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator in FP32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------
    # loop over the reduction axis k
    # -----------------------------------------------------------
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_outer * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- left : (M, K, H) -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + k[None, :, None] * stride_left_k
            + h[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- right : (N, K, H) -----
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j[:, None, None] * stride_right_j
            + k[None, :, None] * stride_right_k
            + h[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # outer‑product over i and j, keep (M,N,K,H)
        # left  : (M, K, H) -> (M, 1, K, H)
        # right : (N, K, H) -> (1, N, K, H)
        left_exp  = left[:, None, :, :]          # (M,1,K,H)
        right_exp = right[None, :, :, :]         # (1,N,K,H)

        prod = left_exp * right_exp               # (M,N,K,H)
        acc += tl.sum(prod, axis=2)                # sum over K → (M,N,H)

    # -----------------------------------------------------------
    # write results (cast back to FP16)
    # -----------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i[:, None, None] * stride_out_i
        + j[None, :, None] * stride_out_j
        + h[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ---------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton kernel
    for the N³ contraction.  All other linear / norm / gate operations are
    performed with PyTorch which is cheap compared to the O(N³) work.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # == H

    # ------------------------------------------------------------------
    # 1. LayerNorm over the `dim` channel (same as nn.LayerNorm)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float32)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float32)

    # use torch.einsum for a dense matmul on the last dim (fast enough)
    left = torch.einsum("bnkc,hd->bnkh", x, left_proj_w)   # [B,N,N,H]
    right = torch.einsum("bnkc,hd->bnkh", x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gate values (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float32)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float32)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float32)

    left_gate = torch.sigmoid(torch.einsum("bnkc,hd->bnkh", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("bnkc,hd->bnkh", x, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum("bnkc,hd->bnkh", x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the contraction
    # ------------------------------------------------------------------
    # Convert to FP16 for the heavy compute
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – chosen to respect register pressure on H200
    BLOCK_M = 32          # i‑axis
    BLOCK_N = 32          # j‑axis
    BLOCK_K = 32          # reduction axis
    BLOCK_H = 8           # slice of hidden dim per kernel launch

    # launch one grid per hidden‑dim slice
    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_h = min(BLOCK_H, hidden_dim - h_start)

        # offsets for the current H slice
        left_ptr  = left.data_ptr() + h_start * stride_left_h * left.element_size()
        right_ptr = right.data_ptr() + h_start * stride_right_h * right.element_size()
        out_ptr   = out.data_ptr() + h_start * stride_out_h * out.element_size()

        # total number of program instances
        num_i = triton.cdiv(N, BLOCK_M)
        num_j = triton.cdiv(N, BLOCK_N)
        num_h = triton.cdiv(cur_h, BLOCK_H)          # always 1 here
        total_blocks = B * num_i * num_j * num_h

        grid = (total_blocks,)

        trimul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            B, N, cur_h,
            stride_left_b, stride_left_i, stride_left_k, stride_left_h,
            stride_right_b, stride_right_j, stride_right_k, stride_right_h,
            stride_out_b, stride_out_i, stride_out_j, stride_out_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 6. Remaining post‑processing in PyTorch (LayerNorm, gate, final Linear)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                         # [B,N,N,H]

    # to_out_norm: LayerNorm over hidden_dim
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # multiply by out_gate (broadcast over i,j)
    out = out * out_gate

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(device).to(torch.float32)   # (dim, hidden_dim)
    out = torch.einsum("bnkh,dh->bnkd", out, to_out_w)                # [B,N,N,dim]

    return out
