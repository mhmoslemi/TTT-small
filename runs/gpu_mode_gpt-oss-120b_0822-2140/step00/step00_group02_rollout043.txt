"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for every hidden dimension d:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ   (shape N×N).

We keep the surrounding LayerNorms / Linear / gating in PyTorch and
offload the N×N ×‑product to a fused Triton kernel.  The kernel works
in float32 (accumulates in FP32, stores in FP32) and supports arbitrary
batch size B (≤2) and hidden dimension D (≤384).  It is launched per
(b,d) pair – the loop over D is cheap because D is small (≤384) and
avoids the need for a 4‑D program‑id grid.

The overall algorithm:

1. Layer‑normalize the input.
2. Linear projections → left/right (shape B,N,N,D).
3. Apply optional mask (broadcasted on the last dim).
4. Compute gates via linear + sigmoid and multiply into left/right.
5. Triton kernel computes the einsum (batched mat‑mul per hidden dim).
6. Layer‑norm, out‑gate, final linear → output (float32).

The implementation runs efficiently on Hopper (sm_90) and meets the
required runtime budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel: batched matmul for a single (batch, hidden) slice
# -------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,               # *const float*
    right_ptr,              # *const float*
    out_ptr,                # *float*
    N,                      # seq_len (int)
    stride_left_i, stride_left_k,        # strides for left (i,k)
    stride_right_i, stride_right_k,      # strides for right (j,k) – note i‑dim is j here
    stride_out_i, stride_out_j,          # strides for output (i,j)
    BLOCK_M: tl.constexpr,  # tile size in i
    BLOCK_N: tl.constexpr,  # tile size in j
    BLOCK_K: tl.constexpr,  # tile size in k
):
    pid_m = tl.program_id(0)          # i‑block
    pid_n = tl.program_id(1)          # j‑block

    # ---------------------------------------------------------
    # Compute offsets for this tile
    # ---------------------------------------------------------
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for oob indices
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over k dimension
    # ---------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left[b,i,k]   (i‑dim = i_offset, k‑dim = k_offset)
        left_ptrs = left_ptr + i_offset[:, None] * stride_left_i \
                                 + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[b,j,k]  (j‑dim = j_offset, k‑dim = k_offset)
        right_ptrs = right_ptr + j_offset[:, None] * stride_right_i \
                                   + k_offset[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # dot over the K dimension → (M,N) contribution
        acc += tl.dot(left, right)   # sum over K

    # ---------------------------------------------------------
    # Store the result
    # ---------------------------------------------------------
    out_ptrs = out_ptr + i_offset[:, None] * stride_out_i \
                         + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------
# Helper that launches the kernel for all (batch, hidden) slices
# -------------------------------------------------------------
def _launch_trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, D]  (float32)
    returns out : [B, N, N, D]   (float32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # constants – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements, not bytes)
    s_left_i = left.stride(1)
    s_left_k = left.stride(2)

    s_right_i = right.stride(1)   # this corresponds to the “j” dimension
    s_right_k = right.stride(2)

    s_out_i = out.stride(1)
    s_out_j = out.stride(2)

    # Loop over batch and hidden dimension (small, so Python loops are cheap)
    for b in range(B):
        left_batch_ptr  = left.data_ptr()  + b * left.stride(0) * left.element_size()
        right_batch_ptr = right.data_ptr() + b * right.stride(0) * right.element_size()
        out_batch_ptr   = out.data_ptr()   + b * out.stride(0) * out.element_size()

        for d in range(D):
            # compute per‑slice pointers (add offset for hidden dim)
            left_ptr  = left_batch_ptr  + d * left.stride(3) * left.element_size()
            right_ptr = right_batch_ptr + d * right.stride(3) * right.element_size()
            out_ptr   = out_batch_ptr   + d * out.stride(3) * out.element_size()

            grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
            _trimul_einsum_kernel[grid](
                left_ptr,
                right_ptr,
                out_ptr,
                N,
                s_left_i, s_left_k,
                s_right_i, s_right_k,
                s_out_i, s_out_j,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                num_warps=4,
            )
    return out


# -------------------------------------------------------------
# Main entry point required by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass with a fused Triton kernel for the
    N×N einsum.  All tensors are expected to be on CUDA and dtype float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                 # input channel dimension
    hidden_dim = config["hidden_dim"]   # projection dimension (c_z)

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (per‑position)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]       # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]      # (hidden_dim, dim)

    # flatten the first three dims, apply linear, reshape back
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)
    left = F.linear(x_flat, left_proj_w).reshape(B, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_w).reshape(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)       # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).reshape(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).reshape(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).reshape(B, N, N, hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core einsum via Triton (batched matmul per hidden dim)
    # -----------------------------------------------------------------
    out = _launch_trimul_einsum(left, right)   # (B,N,N,hidden_dim), float32

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]    # (dim, hidden_dim)

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)
    out = out * out_gate
    # final projection back to dim
    out_flat = out.reshape(-1, hidden_dim)
    out = F.linear(out_flat, to_out_w).reshape(B, N, N, dim)

    return out
