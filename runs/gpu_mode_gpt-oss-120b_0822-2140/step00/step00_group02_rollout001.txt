"""
TriMul (outgoing) – fast Triton implementation of the heavy N³ einsum.

Algorithm
---------
1.  Layer‑Norm + Linear projections (left, right) + gating – done in PyTorch
    (these are cheap compared to the O(N³) contraction).
2.  Compute the contracted tensor
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    for every batch * hidden channel.  This is a batched matrix‑multiply
    for each hidden dimension:
        out_h = left_h @ right_hᵀ .
    The contraction is performed by a custom Triton kernel that tiles the
    i‑ and j‑dimensions (M × N) and tiles the reduction dimension k.
    All arithmetic is carried out in fp32 (inputs are fp16 for bandwidth).
3.  Layer‑Norm on the hidden dimension, gating, final Linear projection –
    again in PyTorch.

The kernel processes a 3‑D grid:
    (block_i, block_j, batch*hidden)
so each (b, h) pair launches an independent mat‑mul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul for each hidden channel (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *[B, N, N, H] left input   (fp16)
    right_ptr,         # *[B, N, N, H] right input  (fp16)
    out_ptr,           # *[B, N, N, H] output       (fp32)
    B, N, H,           # problem sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    The right matrix is accessed as (j,k) and used directly (no explicit transpose)
    because the reduction index k aligns with the third axis of ``right``.
    """

    pid_m = tl.program_id(0)               # block row   (i)
    pid_n = tl.program_id(1)               # block col   (j)
    pid_z = tl.program_id(2)               # combined batch‑hidden index

    b = pid_z // H                         # batch index
    h = pid_z % H                          # hidden‑channel index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (i, k) slice
        l_ptr = (left_ptr
                 + b * stride_l_b
                 + h * stride_l_h
                 + offs_m[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k)
        l = tl.load(l_ptr,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # right: (j, k) slice – note the order of strides (j then k)
        r_ptr = (right_ptr
                 + b * stride_r_b
                 + h * stride_r_h
                 + offs_n[None, :] * stride_r_j
                 + offs_k[:, None] * stride_r_k)
        r = tl.load(r_ptr,
                    mask=(mask_n[None, :] & mask_k[:, None]),
                    other=0.0).to(tl.float32)

        # inner product over k
        acc += tl.dot(l, r)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    o_ptr = (out_ptr
             + b * stride_o_b
             + h * stride_o_h
             + offs_m[:, None] * stride_o_i
             + offs_n[None, :] * stride_o_j)
    tl.store(o_ptr,
             acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton‑accelerated
    N³ contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]       (float32 or bool)
            - weights      : dict of torch.Tensor parameters
            - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm (no bias handling by torch.nn.LayerNorm to avoid extra
    #               memory allocations – we implement the affine part manually)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        # x: [..., C]
        mean = x.mean(-1, keepdim=True)
        var = x.var(-1, unbiased=False, keepdim=True)
        x_norm = (x - mean) / torch.sqrt(var + eps)
        return x_norm * weight + bias

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                 # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) and gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton contraction
    # --------------------------------------------------------------
    # Cast to fp16 for bandwidth, keep accumulation in fp32 inside kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides for Triton (in element units, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_f16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_f16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – chosen to fit SM shared memory on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        B * hidden,                       # batch × hidden
    )

    _tri_mul_kernel[grid](
        left_f16, right_f16, out,
        B, N, hidden,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden dim)
    # --------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7. Apply output gate and final linear projection
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
