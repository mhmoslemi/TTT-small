"""
TriMul “outgoing” implementation (forward only)

The heavy part of the TriMul forward pass is the 3‑way reduction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where
    left = (x·W_left)  ⊙ sigmoid(x·W_left_gate)
    right = (x·W_right) ⊙ sigmoid(x·W_right_gate)

We keep the LayerNorm / Linear / Sigmoid work in PyTorch (they are cheap)
and accelerate the N³⋅H reduction with a custom Triton kernel.
The kernel tiles the (i, j, d) output space (M‑tile × N‑tile × D‑tile)
and iterates over k‑tiles, accumulating the outer‑product
`left[i,k,:] * right[j,k,:]` for each hidden channel.
All arithmetic is performed in float32 and the final result is returned
as a float32 tensor.

The kernel is deliberately written to handle any batch size B∈{1,2},
any sequence length N (up to 1024 in the tests) and any hidden
dimension H (128 in the reference).  The block sizes have been chosen
to give good occupancy on Hopper (SM‑90) while fitting comfortably in
registers/shared memory.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel
# -------------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,                     # [B, N, N, H]   left tensor
    right_ptr,                    # [B, N, N, H]   right tensor
    out_ptr,                      # [B, N, N, H]   output tensor
    B, N, H,                      # static dimensions
    # strides (in elements) for each tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,        # i‑tile
    BLOCK_N: tl.constexpr,        # j‑tile
    BLOCK_D: tl.constexpr,        # hidden‑channel tile
    BLOCK_K: tl.constexpr,        # reduction‑tile (k)
):
    """
    Compute:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using tiling over (i,j,d) and looping over k.
    """
    pid_i = tl.program_id(0)                # i‑tile index
    pid_j = tl.program_id(1)                # j‑tile index
    pid_d = tl.program_id(2)                # d‑tile index (covers all batches)

    # -----------------------------------------------------------------
    # Tile offsets
    # -----------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    d_base = pid_d * BLOCK_D

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # -----------------------------------------------------------------
    # Loop over batch (B ≤ 2, cheap to serialize inside the kernel)
    # -----------------------------------------------------------------
    for b in range(0, B):
        # accumulator for the current tile (float32)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # -----------------------------------------------------------------
        # Reduction over k
        # -----------------------------------------------------------------
        for k_base in range(0, N, BLOCK_K):
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # -------------------------------------------------------------
            # Load left[tile]  shape (M, K, D)
            # -------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_d[None, None, :] * stride_left_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )

            # -------------------------------------------------------------
            # Load right[tile] shape (N, K, D)
            # -------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + b * stride_right_b
                + offs_j[:, None, None] * stride_right_j
                + offs_k[None, :, None] * stride_right_k
                + offs_d[None, None, :] * stride_right_h
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )

            # -------------------------------------------------------------
            # Outer‑product across the hidden dimension for each k
            #   left[:, k, d] (M, D)  *  right[:, k, d] (N, D)
            #   => contribution (M, N, D) = left[:,k,:] (M,1,D) * right[:,k,:] (1,N,D)
            # -------------------------------------------------------------
            # Broadcasting does the outer‑product automatically
            acc += left[:, None, :, :] * right[None, :, :, :]   # (M,N,D)

        # -----------------------------------------------------------------
        # Write the tile back to memory
        # -----------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_d[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
        )


# -------------------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple
            - input_tensor : torch.Tensor      [B, N, N, dim]
            - mask_tensor  : torch.Tensor      [B, N, N]   (bool / float)
            - weights      : dict of torch.Tensor (model parameters)
            - config       : dict with keys
                * dim          : input feature dimension
                * hidden_dim   : hidden dimension used inside TriMul

    Returns:
        torch.Tensor   [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    assert hidden_dim == weights["left_proj.weight"].shape[0]

    # -------------------------------------------------------------
    # Helper to fetch weight/bias tensors (already on device)
    # -------------------------------------------------------------
    def get(name):
        return weights[name].to(device)

    # -------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=get("norm.weight"),
        bias=get("norm.bias"),
    )

    # -------------------------------------------------------------
    # Linear projections (no bias)
    # -------------------------------------------------------------
    left = F.linear(x, get("left_proj.weight"))      # [B,N,N,hidden]
    right = F.linear(x, get("right_proj.weight"))    # [B,N,N,hidden]

    # -------------------------------------------------------------
    # Apply mask (if any).  mask shape: [B,N,N] → [B,N,N,1]
    # -------------------------------------------------------------
    mask = mask_tensor.unsqueeze(-1).to(dtype)       # broadcastable
    left = left * mask
    right = right * mask

    # -------------------------------------------------------------
    # Gating sigmoids
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, get("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, get("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x, get("out_gate.weight")))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # Triton kernel: core N³⋅H multiplication
    # -------------------------------------------------------------
    # Ensure contiguous layout (needed for stride calculations)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides in *elements* (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Tile sizes – tuned for Hopper, adjust if needed
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_D = 32
    BLOCK_K = 64

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_d = triton.cdiv(hidden_dim, BLOCK_D)   # one grid dimension for d‑tiles

    trimul_einsum_kernel[
        (grid_i, grid_j, grid_d)
    ](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_h=s_left[3],
        stride_right_b=s_right[0],
        stride_right_j=s_right[1],
        stride_right_k=s_right[2],
        stride_right_h=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_h=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------
    # Post‑processing (LayerNorm, gating, final linear)
    # -------------------------------------------------------------
    # to_out_norm works on the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=get("to_out_norm.weight"),
        bias=get("to_out_norm.bias"),
    )
    out = out * out_gate
    out = F.linear(out, get("to_out.weight"))   # final projection back to dim

    return out.to(dtype)   # ensure float32 output
