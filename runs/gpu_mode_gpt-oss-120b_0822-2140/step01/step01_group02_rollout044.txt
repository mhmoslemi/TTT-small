The “outgoing” TriMul operator can be broken down into three cheap PyTorch parts  
(LayerNorm → Linear → gates) and one *very* expensive contraction  

\[
\text{out}[b,i,j,d] = \sum_{k} \text{left}[b,i,k,d]\;\text{right}[b,j,k,d] .
\]

For a fixed head `d` this is exactly a matrix multiplication  
`left[b,:, :, d] @ right[b,:, :, d].T`.  
All heads are independent, therefore we can treat the pair **(batch, head)**
as a single batch dimension and run a *batched GEMM* in one Triton kernel.

The kernel works in **fp16** (the model is trained in fp32, but this contraction
is tolerant to reduced precision) and accumulates in **fp32** for numerical
stability.  After the GEMM we reshape the result back to `[B, N, N, hidden_dim]`,
apply the final LayerNorm + gate and the output linear layer – all performed
in PyTorch because they are cheap compared with the O(N³·H) work.

The implementation follows the two lessons that actually helped:

| Lesson | What we do |
|--------|-------------|
| **Permute to (B, H, N, N) before kernel** | `left/right` are reshaped to `(B*H, N, N)` (contiguous) before launching the kernel. |
| **Pass correct strides** | Strides are taken from the contiguous tensors and passed to the kernel; no implicit flattening is used. |

The kernel itself is a straightforward tiled GEMM (block‑wise loading, `tl.dot`,
and accumulation).  Block sizes `128×128×32` work well on Hopper/H200 and give
good occupancy; the kernel is launched with three‑dimensional grid
`(ceil(N/128), ceil(N/128), B*H)`.

"""
TriMul “outgoing” forward pass (no backward).

The heavy O(N³·H) part is a batched GEMM:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is realized as
    C = A @ Bᵀ          with A,B shape (B*H, N, N)

A custom Triton kernel computes the batched GEMM in fp16 (fp32 accumulation).
All other operations (LayerNorm, Linear, gates) are cheap and stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : [batch, M, K] , [batch, N, K])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col)
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where B is logically transposed."""
    pid_batch = tl.program_id(2)      # batch dimension (B*H)
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # ------------------------------------------------------------------
    # Offsets of the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tiles in A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_az \
                  + offs_m[:, None] * stride_am \
                  + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                  + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator (fp32 for better numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load Bᵀ tile
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as fp16 to save bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for fp16 tensors.
    a, b : (batch, N, N), dtype=torch.float16, contiguous.
    Returns c with the same shape / dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, N, K = a.shape
    assert K == N and b.shape == (batch, N, N)

    c = torch.empty_like(a)

    # Strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – good defaults for Hopper (sm_90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch,                     # batch*head
    )
    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # 4 warps gives good occupancy for 128×128 tiles
    )
    return c


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of parameter tensors (float32)
        - config       : dict with keys
              "dim" (int), "hidden_dim" (int), "nomask" (bool, default True)

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim], dtype float32
    """
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Rearrange to (B, hidden_dim, N, N) -> (B*hidden_dim, N, N)
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM: C = left @ rightᵀ  (still fp16)
    # --------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(left, right)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden_dim) and cast to fp32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Post‑projection LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate shape (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
