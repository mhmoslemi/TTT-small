## Triton‑accelerated “outgoing” TriMul implementation

```python
"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B×N×N×C).
2. Linear projections → left / right tensors (B×N×N×H)  (no bias).
3. (Optional) mask the projected tensors.
4. Gated‑linear transforms (sigmoid gates) applied element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM A @ Bᵀ for every (b,h) pair.
   The GEMM is computed by a Triton kernel that
   * loads A and B in fp16,
   * accumulates in fp32,
   * writes fp32 output.
6. Layer‑norm + out‑gate on the result, followed by the final linear
   projection back to the original channel dimension C.

Only the heavy matrix‑multiplication is executed in Triton; all auxiliary
operations stay in PyTorch for simplicity and because they are not the
performance bottleneck.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _gemm_T_kernel(
    A_ptr,                      # *float16   (batch, M, K)
    B_ptr,                      # *float16   (batch, N, K)  (will be read as K×N)
    C_ptr,                      # *float32   (batch, M, N)
    M, N, K,                    # matrix dimensions (int32)
    stride_Ab, stride_Am, stride_Ak,   # strides of A (int64)
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (int64)  (Bk = original col stride, Bn = original row stride)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C (int64)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                 # batch*head index
    pid_m = tl.program_id(0)                 # block row
    pid_n = tl.program_id(1)                 # block col

    # ------------------------------------------------------------------
    # 1) Compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A (row‑major).  A shape: (batch, M, K)
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )

    # Pointers for B, but we need Bᵀ → we read B as (K, N)
    # Original B shape is (batch, N, K).  Strides: (stride_Bb, stride_Bn, stride_Bk)
    # For the transposed view we treat K‑stride = stride_Bk (original column stride)
    # and N‑stride = stride_Bn (original row stride).
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk   # K dimension (original column)
        + offs_n[None, :] * stride_Bn   # N dimension (original row)
    )

    # ------------------------------------------------------------------
    # 2) Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)  # fp16 -> fp32

        # Load Bᵀ tile
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Matrix multiply for the tile (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # 4) Write result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator with a Triton‑accelerated
    batched matrix multiplication.

    Parameters
    ----------
    data : (input, mask, weights, config)
        input   – torch.Tensor, shape [B, N, N, C]  (float32)
        mask    – torch.Tensor, shape [B, N, N]    (bool/float or None)
        weights – dict with the same keys as the reference PyTorch module
        config  – dict with at least ``dim`` and ``hidden_dim`` keys,
                  optional ``nomask`` (True → ignore mask).

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # Convenience views of weight tensors
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (broadcast over hidden dim)
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM (A @ Bᵀ) per head
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Convert to fp16 for bandwidth‑limited loads
    left_fp16 = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # Output buffer (fp32 accumulator)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # ------------------------------------------------------------------
    # Strides (in element units, not bytes)
    # ------------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (B*H, N, N) → (N*N, N, 1)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()  # (B*H, N, N) → (N*N, N, 1)

    # For the transposed view of B: K‑stride = original column stride (=1)
    # N‑stride = original row stride (=N)
    stride_Bk = stride_Bc        # = 1
    stride_Bn = stride_Br        # = N

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # Tiling parameters – tuned for Hopper (SM90)
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per thread block
    BLOCK_N = 64   # cols per thread block
    BLOCK_K = 32   # reduction tile size

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (rows)
        triton.cdiv(N, BLOCK_N),   # blocks along N (cols)
        B * H                      # batch * hidden
    )

    _gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N,                # M
        N,                # N
        N,                # K (the summed dimension)
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim C

    return out
```
