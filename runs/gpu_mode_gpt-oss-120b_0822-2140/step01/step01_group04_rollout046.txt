"""
TriMul “outgoing” kernel (AlphaFold3).

The expensive step of the module is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply per hidden dimension `d`:

    (B, H, N, N) @ (B, H, N, N)^T   →   (B, H, N, N)

where `H = hidden_dim`.  The implementation below:
* performs all cheap ops (LayerNorm, Linear, gating, final LayerNorm/Linear) in
  vanilla PyTorch,
* executes the O(N³) contraction with a Triton kernel that works on tensors
  shaped `(B, H, N, N)`.  The kernel uses FP16 loads, FP32 accumulation and
  stores the result in FP32, matching the reference behavior.
* supports optional masking, arbitrary batch size (1‑2), sequence length up to
  1024 and hidden dimensions 128/384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)^T
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    N, H,                               # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_i,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Compute, for each (batch, head) pair:
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ   (both A and B have shape (N, N)).
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb % H

    # ------------------------------------------------------------------
    # Tile offsets & masks
    # ------------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A[b,h,i,k]   shape (TILE_M, TILE_K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ -> we need B[b,h,j,k]; load as (TILE_K, TILE_N)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dim
                  + offs_n[None, :] * stride_b_i)  # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K (FP16 * FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 / bf16)
        - mask         : torch.Tensor of shape [B, N, N]  (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors containing the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (True → no mask).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32          # final output dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=w("norm.weight"),
                     bias=w("norm.bias"),
                     eps=eps)                       # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))

    if not nomask:
        # mask shape -> [B,N,N,1] for broadcasting
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction using Triton
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) – make contiguous for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # Strides expressed in element counts (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, j, i)   (we will index transposed)
    c_s = out_t.stride()    # (batch, head, i, j)

    # Tile sizes – empirically good for Hopper GPUs
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,                     # 8 warps fits H200 well
    )

    # Back to original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm, gating and linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=w("to_out_norm.weight"),
                       bias=w("to_out_norm.bias"),
                       eps=eps)
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))  # [B,N,N,dim]

    return out.to(dtype)
