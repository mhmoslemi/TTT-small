"""
TriMul (outgoing) – fast forward pass.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication of the form
    (B·H, N, N) @ (B·H, N, N)^T .
We implement this mat‑mul in a single Triton kernel that works on
half‑precision inputs and accumulates in fp32 for accuracy.
All other ingredients (LayerNorm, Linear, sigmoids, final projection)
are kept in PyTorch, which is cheap compared to the O(N³·H) work.
The kernel is launched with a 3‑D grid:
    (batch_head, n_blocks_M, n_blocks_N)
and uses tiling (BLOCK_M × BLOCK_N × BLOCK_K) that fits the
Hopper SM‑90 architecture (shared‑memory‑free WMMA‑like dot).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A,B are (BH,N,N) in fp16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,                # fp16 left  (BH,N,N)
    B_ptr,                # fp16 right (BH,N,N)
    C_ptr,                # fp16 output (BH,N,N)
    BH, N,                # int64
    # strides (in elements)
    stride_ah, stride_am, stride_an,   # A stride batch, row, col
    stride_bh, stride_bm, stride_bn,   # B stride batch, row, col
    stride_ch, stride_cm, stride_cn,   # C stride batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch‑head index
    pid_m = tl.program_id(1)        # block‑row
    pid_n = tl.program_id(2)        # block‑col

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for the current batch‑head
    a_batch_ptr = A_ptr + pid * stride_ah
    b_batch_ptr = B_ptr + pid * stride_bh
    c_batch_ptr = C_ptr + pid * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A block (M x K) ----
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                 + offs_k[None, :] * stride_an)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)

        # ---- load B block (N x K)  (we need Bᵀ, i.e. K x N) ----
        # load as (K x N) so dot works with a (M x K) @ b (K x N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bm
                                 + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float16)

        # ---- dot product (M×K)·(K×N) -> (M×N) ----
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # write result back (down‑cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point expected by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32/bool or None, CUDA)
        * weights_dict : all learned parameters (float32, CUDA)
        * config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, w["right_proj.weight"])  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True) and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM
    #    shape -> (B, hidden_dim, N, N) -> (BH, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left = left.view(BH, N, N).to(torch.float16)
    right = right.view(BH, N, N).to(torch.float16)

    # output buffer
    out_buf = torch.empty_like(left)   # fp16, (BH,N,N)

    # ------------------------------------------------------------------
    # 6️⃣ Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # compute strides (in elements, not bytes)
    stride_ah, stride_am, stride_an = left.stride()
    stride_bh, stride_bm, stride_bn = right.stride()
    stride_ch, stride_cm, stride_cn = out_buf.stride()

    batched_gemm_kernel[grid](
        left,
        right,
        out_buf,
        BH,
        N,
        stride_ah,
        stride_am,
        stride_an,
        stride_bh,
        stride_bm,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to (B,N,N,hidden_dim) and cast to fp32
    # ------------------------------------------------------------------
    out = out_buf.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8️⃣ Post‑norm on hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 9️⃣ Element‑wise multiplication with the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B,N,N,dim]

    return out
