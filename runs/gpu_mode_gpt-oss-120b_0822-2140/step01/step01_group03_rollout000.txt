```python
"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm on the input tensor    [B, N, N, C]               (float32)
2. Linear projections → left / right  [B, N, N, H]               (float32)
3. (Optional) mask the two projected tensors.
4. Gated‑linear transforms (sigmoid) → element‑wise multiplication.
5. Core contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `A @ Bᵀ` for every (b,h) pair.
   We reshape to (B·H, N, N) and launch a Triton kernel that
   * loads A and Bᵀ in float16,
   * accumulates in float32,
   * writes the result in float32.
6. Layer‑norm on the hidden‑dim result, multiply by out‑gate, final linear
   projection back to the original channel dimension C.

The kernel is tuned for Hopper (SM‑90): 128 × 128 tiles, K‑tile = 32,
fp16 I/O, fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, M, K])   →   C : [batch, M, N]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_T_kernel(
    A_ptr,                     # *fp16   (batch, M, K)
    B_ptr,                     # *fp16   (batch, N, K)  – read as K×N (transpose)
    C_ptr,                     # *fp32   (batch, M, N)
    M, N, K,                   # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,      # strides of A
    stride_Bb, stride_Bk, stride_Bn,      # strides of B (note: Bk = col‑stride, Bn = row‑stride)
    stride_Cb, stride_Cm, stride_Cn,      # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch index = B * H
    pid_m = tl.program_id(0)          # block‑row
    pid_n = tl.program_id(1)          # block‑col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile
    a_ptrs = (A_ptr + pid_b * stride_Ab
              + offs_m[:, None] * stride_Am
              + offs_k[None, :] * stride_Ak)

    # B is accessed transposed: we need (K, N) layout.
    b_ptrs = (B_ptr + pid_b * stride_Bb
              + offs_k[:, None] * stride_Bk   # step along original column (K dimension)
              + offs_n[None, :] * stride_Bn)  # step along original row   (N dimension)

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K dimension (tiled)
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)        # fp16 → fp32
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)
        acc += tl.dot(a, b)                       # (M×K)·(K×N) → (M×N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Write out the tile
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + pid_b * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C],  float32
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys
              * "dim"         – input channel dimension C
              * "hidden_dim"  – hidden dimension H
              * "nomask"      – True → ignore mask, False → use mask

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C], float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (C‑dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask ∈ {0,1} → float32 broadcast
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM with Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # (B, N, N, H) → (B·H, N, N)   – convenient for the kernel
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Use fp16 inside the kernel (half‑precision bandwidth)
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # Output buffer (fp32 accumulator)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # Strides in **elements** (not bytes) – required by the kernel
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()      # (batch, M, K)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()    # (batch, N, K)

    # For the transposed view of B we swap the last two strides:
    stride_Bk = stride_Bc   # step along original column (K)
    stride_Bn = stride_Br   # step along original row    (N)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # Tiling parameters – tuned for SM90 (Hopper)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),     # rows
        triton.cdiv(N, BLOCK_N),     # cols
        B * H                        # batch * head
    )

    _batched_matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to C

    return out
```
