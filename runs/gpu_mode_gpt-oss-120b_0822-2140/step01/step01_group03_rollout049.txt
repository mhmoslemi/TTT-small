## Triton‑accelerated “outgoing” TriMul

```python
"""
custom_kernel – Triton‑accelerated forward pass of the TriMul “outgoing” operator
(AlphaFold‑3).  The heavy part (the N³ contraction) is rewritten as a batched
matrix‑multiply  out = left @ rightᵀ   performed by a single Triton kernel.
The surrounding layer‑norms, linear layers and gating are kept in PyTorch for
clarity.  Mixed‑precision (fp16 loads / fp32 accumulation) is used inside the
kernel; the returned tensor is fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A and B have shape [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_T_kernel(
    A_ptr,                     # *fp16   (batch, M, K)  (= left)
    B_ptr,                     # *fp16   (batch, N, K)  (= right, read as K×N)
    C_ptr,                     # *fp32   (batch, M, N)  (= out)
    M, N, K,                   # matrix dimensions (all == seq_len)
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note B is accessed transposed)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                 # batch * head index
    pid_m = tl.program_id(0)                 # block row
    pid_n = tl.program_id(1)                 # block column

    # ------------------------------------------------------------------
    # offsets of the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # pointer arithmetic
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # ------------------------------------------------------------------
    # accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # tiled K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)   # (M×K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)   # (K×N)
        acc += tl.dot(a, b)                     # (M×N)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of TriMul “outgoing”.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool / float) or None
        - weights      : dict of model parameters (all float32)
        - config       : dict containing                dim, hidden_dim,
                        and optionally nomask (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b                 = weights["norm.weight"], weights["norm.bias"]
    left_proj_w                    = weights["left_proj.weight"]
    right_proj_w                   = weights["right_proj.weight"]
    left_gate_w                    = weights["left_gate.weight"]
    right_gate_w                   = weights["right_gate.weight"]
    out_gate_w                     = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b   = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w                       = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left  = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N] → [B, N, N, 1]
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM left @ rightᵀ per head
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # reshape to (B*H, N, N) so each head is an independent batch element
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside kernel for bandwidth, keep fp32 result
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)   # (B*H, N, N)

    # --------------------------------------------------------------
    # stride extraction (in elements, not bytes)
    # --------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # right is read transposed: K dimension = original column stride
    stride_Bk = stride_Bc               # step along K (original col)
    stride_Bn = stride_Br               # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # --------------------------------------------------------------
    # launch parameters (tuned for Hopper, SM90)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * heads
    )

    _trimul_matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original channel dimension C

    return out
```

The implementation respects all constraints:

* Works for any `B∈{1,2}`, `N∈{128,…,1024}`, `dim∈{128,384}`, `hidden_dim=128`.
* Handles both masked and un‑masked inputs.
* Only the N³‑dominant contraction is executed in Triton; everything else stays in high‑level PyTorch for simplicity and numerical stability.
* Uses mixed‑precision inside the kernel and returns a `float32` tensor, matching the reference PyTorch module.
