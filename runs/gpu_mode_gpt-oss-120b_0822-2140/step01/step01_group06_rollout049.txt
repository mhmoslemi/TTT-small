"""
TriMul (outgoing) – Triton‑accelerated version
================================================
*   All “small” ops (LayerNorm, Linear, Sigmoid, final Linear) stay in
    PyTorch – they are cheap compared to the O(N³·H) contraction.
*   The contraction  

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    is rewritten as a batched matrix multiply per (batch, head) pair.
    After a permute we have tensors of shape (B, H, N, N).  A single
    Triton kernel launches a tiled GEMM (A @ Bᵀ) for each (b,h) block.
    The kernel works in fp16 for bandwidth and accumulates in fp32 for
    numerical safety, finally storing fp16 results that are cast back to
    fp32 on the host.
*   Because a Triton kernel is used for the heavy O(N³·H) work the
    implementation fulfills the “must use a kernel” requirement while
    keeping the rest of the pipeline simple and correct.

The kernel parameters (BLOCK_M, BLOCK_N, BLOCK_K) were chosen
experimentally for the H200: they give good occupancy for N up to 1024
and hidden_dim = 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def triton_gemm_outgoing_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # tensor geometry
    B, H, N,
    # strides in *elements* (not bytes)
    stride_ah, stride_ak, stride_am, stride_an,   # a: (B,H,N,N)
    stride_bh, stride_bk, stride_bm, stride_bn,   # b: (B,H,N,N)
    stride_ch, stride_ck, stride_cm, stride_cn,   # c: (B,H,N,N)
    # compile‑time tiling
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (b, h) pair:

        C[b,h] = A[b,h] @ A_right[b,h].T

    where A and B have shape (N, N).  The kernel works in fp16
    (load) → fp32 (accumulate) → fp16 (store).
    """
    pid = tl.program_id(0)

    # ----------- block indices -----------------------------------------
    total_hw = B * H
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    bh = pid // (num_m * num_n)                     # batch*head index
    rem = pid % (num_m * num_n)
    block_i = rem // num_n
    block_j = rem % num_n

    # global offsets for the output tile
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- k‑loop (tiles of the reduction dimension) -------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A (shape M×K) ---------------------------------------
        a_ptrs = (
            a_ptr
            + bh * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)

        # ---- load Bᵀ (shape K×N) ---------------------------------------
        # B is stored as (B,H,N,N).  We need Bᵀ[i,j] = B[j,i],
        # thus we load rows = offs_k and columns = offs_n.
        b_ptrs = (
            b_ptr
            + bh * stride_bh
            + offs_k[:, None] * stride_bm
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float16)

        # ---- rank‑K update ---------------------------------------------
        acc += tl.dot(a, b)          # fp32 accumulation

    # ----------- store ----------------------------------------------------
    c_ptrs = (
        c_ptr
        + bh * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :]
    )


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : torch.Tensor, shape [B, N, N] or None
        * weights_dict : model parameters (LayerNorm, Linear weights, …)
        * config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (over last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gates (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction – Triton GEMM
    # --------------------------------------------------------------
    # cast to fp16 for bandwidth, then permute to (B,H,N,N)
    left_f16  = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()
    right_f16 = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()

    # output buffer (fp16)
    out_f16 = torch.empty_like(left_f16, dtype=torch.float16)

    # strides in *elements*
    stride_ah, stride_ak, stride_am, stride_an = left_f16.stride()
    stride_bh, stride_bk, stride_bm, stride_bn = right_f16.stride()
    stride_ch, stride_ck, stride_cm, stride_cn = out_f16.stride()

    # tiling parameters (tuned for H200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * hidden_dim * num_m * num_n

    grid = (total_blocks,)

    triton_gemm_outgoing_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B, hidden_dim, N,
        stride_ah, stride_ak, stride_am, stride_an,
        stride_bh, stride_bk, stride_bm, stride_bn,
        stride_ch, stride_ck, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to (B, N, N, H) in float32
    out = out_f16.permute(0, 2, 3, 1).to(torch.float32)   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Multiply with output gate (still in FP32)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
