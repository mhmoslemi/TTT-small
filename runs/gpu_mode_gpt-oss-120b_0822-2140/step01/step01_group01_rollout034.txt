"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The computational bottleneck of the outgoing TriMul operator is the
3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]            (N×N)
    R = right[b,:, :, d]           (N×N)
    out[b,:, :, d] = L @ Rᵀ        (N×N)

We therefore launch a *batched* GEMM kernel that computes
`M = B * hidden_dim` independent products `A @ Bᵀ` where each operand has
shape `(N, N)`.  The kernel is tiled in three dimensions (M‑tile, N‑tile,
K‑tile) and works entirely in FP16 for the loads while accumulating in
FP32 for numerical stability.  All surrounding operations
(LayerNorm, linear projections, sigmoids, masking) are cheap and stay in
PyTorch.

The kernel returns a tensor of shape `(B, hidden_dim, N, N)` which is
permuted back to `(B, N, N, hidden_dim)` before the final LayerNorm and
linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: many independent GEMMs (A @ Bᵀ) of shape (N,N)·(N,N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # [M, N, N] pointers
    N,                                  # sequence length
    stride_ah, stride_ai, stride_ak,    # A strides (M, N, N)
    stride_bh, stride_bj, stride_bk,    # B strides (M, N, N)
    stride_ch, stride_ci, stride_cj,    # C strides (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)      # block row (i)
    pid_n = tl.program_id(1)      # block col (j)
    pid_z = tl.program_id(2)      # flattened batch*hidden

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the (pid_z)-th matrix pair
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop (tiles)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  (BLOCK_M, BLOCK_K) – load in FP16
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B_T[k, j] : we need B[j, k] transposed => (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k dimension (row of B_T)
            + offs_n[None, :] * stride_bj          # j dimension (col of B_T)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 dot, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  FP32
    returns       : (B, H, N, N) FP32 where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                                            # flattened batch*head

    # flatten to (M, N, N) – contiguous memory layout is required
    left_f = left.reshape(M, N, N).contiguous()
    right_f = right.reshape(M, N, N).contiguous()
    out = torch.empty_like(left_f)

    # strides as int64 (torch uses signed int64)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # Tile sizes – chosen to give good occupancy on SM90 (H200/H100)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # pid_m
        triton.cdiv(N, BLOCK_N),          # pid_n
        M,                                # pid_z
    )
    _trimul_gemm_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]
        mask         : torch.Tensor [B, N, N] (may be None)
        weights      : dict of torch.Tensors containing all model parameters
        config       : dict with keys
                       - "dim"
                       - "hidden_dim"
                       - "nomask" (bool, True if mask should be ignored)

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # --------------------------------------------------------------
    # 3. Apply mask if present
    # --------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N) → (B,N,N,1) for broadcasting
        m = mask.to(device).unsqueeze(-1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device))
    )

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ for each hidden slice
    #    Convert layout to (B, H, N, N) expected by the kernel.
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    out_perm = _batched_gemm(left_perm, right_perm)    # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(device))

    # ensure float32 output irrespective of internal precision
    return out.to(torch.float32)
