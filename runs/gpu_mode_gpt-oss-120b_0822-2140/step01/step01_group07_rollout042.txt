"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The only O(N³) operation is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply of shape
    (B·H, N, N)  @  (B·H, N, N).T
with H = hidden_dim.

We keep all LayerNorm / Linear / gating work in PyTorch (highly tuned in
torch.nn.functional) and fuse the GEMM with a single Triton kernel.
The kernel works on fp16 (half) inputs – the model weights are already
fp16‑compatible on H200 – but accumulates in fp32 and writes fp32 so the
final tensor has the required dtype.

Key ideas
---------
*   Collapse the batch and hidden dimensions: (B,N,N,H) → (B·H,N,N).
*   Tile the GEMM with BLOCK_M = BLOCK_N = 64, BLOCK_K = 32 – this fits
    nicely into one warp on Hopper and lets us use 8 warps per CTA.
*   Load the two operands as half, compute with tl.dot (fp32 accumulator),
    store the result as fp32.
*   Optional mask is applied *outside* the kernel (a cheap element‑wise
    multiplication) – this avoids extra branching inside the hot loop.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ   where left/right have shape (M,N,N)
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,          # *fp16   [M, N, K]  (K == N)
    right_ptr,         # *fp16   [M, N, K]
    out_ptr,           # *fp32   [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides (elements)
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,   # how many tiles of M per CTA group (for load balancing)
):
    # ------------------------------------------------------------------- #
    # 1️⃣  Program ID & tile offsets
    # ------------------------------------------------------------------- #
    pid = tl.program_id(0)          # linear index over the 2‑D grid of tiles
    num_pid_m = tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N)   # total tiles per matrix
    num_pid_z = tl.cdiv(M, GROUP_M)                         # number of matrix groups

    # matrix (z) we are working on and the tile within that matrix
    pid_z = pid // num_pid_m
    pid_tile = pid % num_pid_m

    pid_m = pid_tile // tl.cdiv(N, BLOCK_N)   # row‑tile index inside the matrix
    pid_n = pid_tile % tl.cdiv(N, BLOCK_N)   # col‑tile index inside the matrix

    # Global offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------- #
    # 2️⃣  Pointer arithmetic (note the transpose on the right operand)
    # ------------------------------------------------------------------- #
    # left  : (M,N,K)   index (z, i, k)
    a_ptrs = left_ptr + pid_z * GROUP_M * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_lk

    # right : we need Bᵀ → read as (z, k, j)
    b_ptrs = right_ptr + pid_z * GROUP_M * stride_rm \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_rk \
                     + offs_n[None, :] * stride_rn

    # ------------------------------------------------------------------- #
    # 3️⃣  Accumulator (fp32) and main K‑loop
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # masks for the edges of the tensors
    mask_m = offs_m < N
    mask_n = offs_n < N

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a block of A and B – they are stored as fp16
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K),
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 dot → fp32 accumulator
        acc += tl.dot(a, b, out=acc)

        # advance K dimension
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # 4️⃣  Write result (fp32)
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * GROUP_M * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor : [B, N, N, dim]   (torch.float32)
        * mask         : [B, N, N]        (torch.bool or 0/1 float)
        * weights      : dict of torch tensors (see source code)
        * config       : {"dim": int, "hidden_dim": int}

    Returns
    -------
    out : torch.Tensor
        [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # H in the paper

    # --------------------------------------------------------------- #
    # 1️⃣  LayerNorm on the input (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # --------------------------------------------------------------- #
    # 2️⃣  Linear projections (no bias)
    # --------------------------------------------------------------- #
    # Keep data in FP16 for the GEMM – the weight matrices are stored as fp16
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------- #
    # 3️⃣  Optional mask (element‑wise)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4️⃣  Gating (sigmoid) – keep these in FP32 (cheap)
    # --------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5️⃣  Reshape to (M, N, N) where M = B * hidden_dim
    # --------------------------------------------------------------- #
    # (B,N,N,H) → (B,H,N,N) → (M,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim
    left = left.view(M, N, N)          # fp16
    right = right.view(M, N, N)

    # output buffer (fp32)
    out = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------- #
    # 6️⃣  Launch Triton GEMM
    # --------------------------------------------------------------- #
    # Tiles – chosen to give good occupancy on Hopper (8 warps, 64² threads)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # Group size for better load‑balance across matrices
    GROUP_M = 8   # i.e. each CTA processes 8 consecutive matrices

    grid = (
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N) * triton.cdiv(M, GROUP_M),
    )

    # Strides in *elements* (not bytes)
    stride_lm, stride_ln, stride_lk = left.stride()
    stride_rm, stride_rn, stride_rk = right.stride()
    stride_om, stride_on, stride_ok = out.stride()

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        M,
        N,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
        num_warps=8,          # Hopper likes 8 warps for fp16 dot
        num_stages=3,
    )

    # --------------------------------------------------------------- #
    # 7️⃣  Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------- #
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # --------------------------------------------------------------- #
    # 8️⃣  Output LayerNorm + gating + final linear
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim, fp32

    return out
