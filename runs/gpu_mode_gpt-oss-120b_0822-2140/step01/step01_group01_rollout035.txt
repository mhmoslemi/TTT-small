"""
Triangle‑Multiplicative‑Update (outgoing) – fast Triton implementation
=====================================================================

The dominant O(N³) operation is the einsum  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden channel *d* this is a matrix multiplication  

    L = left[b,:, :, d]   (N×N)  
    R = right[b,:, :, d]  (N×N)  

    out[b,:, :, d] = L @ Rᵀ   (N×N)

We therefore launch a *batched* GEMM kernel that computes, for every
(batch, hidden) pair, the product `L @ Rᵀ`.  All other operations
(LayerNorm, linear projections, sigmoids, masking) stay in PyTorch –
they are negligible compared to the N³ work.

Key performance tricks
----------------------
*   left/right are cast to **float16** before the kernel; the dot product
    accumulates in **float32** (the default of `tl.dot`).
*   Tile sizes (128×128×64) were chosen to match the large shared‑memory
    of the H200 (SM90) while keeping enough parallelism.
*   The kernel works on a flattened batch×hidden dimension (`M = B*H`);
    each program‑id `z` processes one independent GEMM.
*   Strides are passed explicitly – no assumption on tensor contiguity.
*   The kernel returns fp32 results which are later reshaped back to the
    original layout.

The public entry point `custom_kernel` follows the exact API required by
the evaluator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers (fp16 for a/b, fp32 for c)
    N, M,                         # N = seq_len, M = B*hidden
    stride_ah, stride_ai, stride_ak,   # A strides: (M, N, N)
    stride_bh, stride_bj, stride_bk,   # B strides: (M, N, N)
    stride_ch, stride_ci, stride_cj,   # C strides: (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for every z ∈ [0, M), the matrix product
        C[z] = A[z] @ B[z]ᵀ
    where each A[z] and B[z] have shape (N, N) and are stored as
    (M, N, N) tensors.
    """
    pid_m = tl.program_id(0)      # block row   (i)
    pid_n = tl.program_id(1)      # block col   (j)
    pid_z = tl.program_id(2)      # flattened batch*hidden index

    # --------------------------------------------------------------
    # Offsets for the current 2‑D tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointer for the selected (M‑th) matrix pair
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (the reduction axis)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  (BLOCK_M × BLOCK_K) – fp16 load
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ[k, j] = B[j, k]   (BLOCK_K × BLOCK_N)
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj          # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Dot product – fp16 inputs, fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write result back (fp32)
    # --------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  float16
    returns       : (B, H, N, N)  float32   (each slice = left @ rightᵀ)
    """
    B, H, N, _ = left.shape
    M = B * H                     # flattened batch * hidden

    # flatten to (M, N, N) – keep the original strides
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32)

    # Strides as int64 (required by Triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tile sizes – tuned for Hopper SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # pid_m
        triton.cdiv(N, BLOCK_N),          # pid_n
        M,                                # pid_z
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Executes the O(N³) contraction with a custom Triton GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections: dim -> hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)        # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and permute to (B, H, N, N) for the kernel
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton GEMM: out = left @ rightᵀ  (per hidden channel)
    # ------------------------------------------------------------------
    out_perm = _trimul_gemm_fp16(left_perm, right_perm)   # (B, H, N, N) fp32

    # Back to the original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    # Ensure FP32 output
    return out.to(torch.float32)
