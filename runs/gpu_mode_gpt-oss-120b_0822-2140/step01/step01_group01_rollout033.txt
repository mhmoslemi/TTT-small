"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched GEMM L @ Rᵀ per (batch, hidden‑dim) slice.
We fuse the whole O(N³) work in a single Triton kernel:

* left / right are first produced in FP32, masked and gated in PyTorch.
* They are then cast to FP16 (half‑precision) and reshaped to
      (B, H, N, N) – the layout expected by the kernel.
* The kernel loads FP16 tiles, does `tl.dot` (FP16 × FP16 → FP32)
  and accumulates in FP32, which gives the best trade‑off between
  bandwidth (half‑precision) and numerical safety (FP32 accumulation).
* After the GEMM the result is kept in FP32, passed through the output
  LayerNorm, out‑gate and final linear projection (still in PyTorch).

Only the O(N³) part is in Triton; the rest stays in PyTorch because it
is cheap compared with the matrix‑multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to FP16 tensors
    N, M,                              # N = seq_len, M = B * hidden_dim
    stride_ah, stride_ai, stride_ak,   # strides of A   (M,N,N)
    stride_bh, stride_bj, stride_bk,   # strides of B   (M,N,N)
    stride_ch, stride_ci, stride_cj,   # strides of C   (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for M independent (N×N) matrices.
    A, B, C are stored as (M, N, N).  Accumulation is FP32, loads are FP16.
    """
    pid_m = tl.program_id(0)   # block row   (i)
    pid_n = tl.program_id(1)   # block col   (j)
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # ------------------------------------------------------------------
    #  Tile coordinates
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    #  Pointers to the current (M‑th) matrix pair
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over the K‑dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  shape (BLOCK_M, BLOCK_K) – FP16 load
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # B_T[k, j] i.e. B[j, k] transposed to match dot
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk   # k‑dim (row of Bᵀ)
            + offs_n[None, :] * stride_bj   # j‑dim (col of Bᵀ)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # Multiply‑accumulate (FP16 × FP16 → FP32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Store the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None] & mask_n[None, :])


def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    Arguments:
        left, right : (B, H, N, N) torch.float16 tensors
    Returns:
        out : (B, H, N, N) torch.float32 tensor   (FP16 inputs → FP32 output)
    """
    B, H, N, _ = left.shape
    M = B * H                                   # flattened batch*head

    # flatten to (M, N, N) – keep the same memory layout
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32)

    # strides as int64 (they are already in element units)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------------------
    # Tune these block sizes for Hopper (SM90).  128×128 tiles give good occupancy,
    # while K‑tiles of 64 keep shared‑memory usage ~ 2 * 64 * 64 * 2bytes ≈ 16KB.
    # -------------------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )

    trimul_gemm_fp16_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        M,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    All cheap ops (LayerNorm, Linear, Sigmoid, masking) stay in PyTorch.
    The O(N³) contraction is performed by a fused FP16‑FP32 Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections: dim -> hidden_dim (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Mask (if present)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, H, N, N) FP16 layout expected by the kernel.
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to half‑precision before the kernel – saves bandwidth.
    left_fp16  = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Batched GEMM in Triton (outputs FP32)
    out_fp32_perm = trimul_gemm_fp16(left_fp16, right_fp16)   # (B, H, N, N)

    # Back to original layout (B, N, N, H)
    out = out_fp32_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
