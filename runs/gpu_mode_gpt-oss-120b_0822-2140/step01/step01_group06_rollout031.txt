"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The cubic contraction  

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is expressed as a batched GEMM per (batch, hidden_dim) head:

    out_head = left_head @ right_headᵀ          (N × N) @ (N × N)ᵀ

All LayerNorms, Linear projections and sigmoid‑gates are performed in
PyTorch.  The heavy O(N³·H) matrix multiplication is carried out by a
single Triton kernel that tiles the three dimensions (M, N, K).  The
kernel works on fp16 data (max throughput on Hopper) while accumulating
in fp32 to retain numeric stability.  The produced tensor is cast back
to fp32 for the remaining operations.

The kernel is launched with a three‑dimensional grid:
    dim0 – concatenated (batch, hidden_dim) index,
    dim1 – block row index,
    dim2 – block column index.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers (fp16)
    batch_head, M, N, K,                # matrix dimensions (int)
    stride_ah, stride_am, stride_ak,    # strides for A (elements)
    stride_bh, stride_bk, stride_bn,    # strides for B (elements)
    stride_ch, stride_cm, stride_cn,    # strides for C (elements)
    BLOCK_M: tl.constexpr,              # tile size in M dimension
    BLOCK_N: tl.constexpr,              # tile size in N dimension
    BLOCK_K: tl.constexpr,              # tile size in K dimension
):
    # ------------------------------------------------------------------
    # 0. Program IDs
    # ------------------------------------------------------------------
    bh_id = tl.program_id(0)                 # which (batch,head) matrix
    pid_m = tl.program_id(1)                 # block row index
    pid_n = tl.program_id(2)                 # block col index

    # ------------------------------------------------------------------
    # 1. Offsets within this matrix
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current matrix
    a_base = a_ptr + bh_id * stride_ah
    b_base = b_ptr + bh_id * stride_bh
    c_base = c_ptr + bh_id * stride_ch

    # ------------------------------------------------------------------
    # 2. Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3. Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : (M, K)   -> rows = offs_m, cols = offs_k
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B : (N, K)^{T}  we need Bᵀ ⇒ rows = offs_k, cols = offs_n
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # cast to fp32, dot, accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4. Write result (cast back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_base
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point used by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32/bool or None, CUDA)
        * weights_dict : model parameters
        * config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape
    nomask = cfg.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Triton GEMM: left @ rightᵀ  (batched over B*H)
    # --------------------------------------------------------------
    # Convert to (B, H, N, N) order and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Flatten batch and head dimension
    left_f16 = left.view(-1, N, N).to(torch.float16)   # (B*H, N, N)
    right_f16 = right.view(-1, N, N).to(torch.float16)
    out_f16 = torch.empty_like(left_f16, dtype=torch.float16)

    # Strides (in elements) for the contiguous (B*H, N, N) tensors
    stride_batch = N * N
    stride_row = N
    stride_col = 1

    # Tile sizes – tuned for Hopper/SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_mats = B * hidden_dim
    grid = (
        total_mats,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_gemm_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        total_mats,
        N, N, N,
        stride_batch, stride_row, stride_col,   # A strides
        stride_batch, stride_row, stride_col,   # B strides (same layout)
        stride_batch, stride_row, stride_col,   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough to keep the GPU busy
    )

    # Reshape back to (B, N, N, H) and cast to fp32 for the rest
    out = out_f16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Multiply by output gate (element‑wise, cheap)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
