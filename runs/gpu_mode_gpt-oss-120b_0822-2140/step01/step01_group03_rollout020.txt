assistantfinal**TriMul “outgoing” – Triton‑accelerated forward pass**

The module consists of a few cheap element‑wise operations (LayerNorm, linear
projections, sigmoid gates) that we keep in PyTorch, and a *single* expensive
contraction  

\[
\text{out}[b,i,j,h]=\sum_{k=0}^{N-1}
\text{left}[b,i,k,h]\;\text{right}[b,j,k,h]\;,
\]

which is a batched matrix‑multiply **A @ Bᵀ** for every hidden‑channel *h*.
The kernel launches a 2‑D grid over the output tiles (M‑dim, N‑dim) and a
3‑rd dimension over the fused batch × head dimension (≤ 2 × 128).  
We load the two operands in **fp16**, accumulate in **fp32**, and store the
result in **fp32**.  The surrounding code follows the reference PyTorch
implementation exactly, only replacing the \(N^{3}\) contraction with the
Triton kernel.

"""
TriMul “outgoing” forward pass.

- LayerNorm + linear projections + sigmoid gates are done in PyTorch.
- Core contraction (batched GEMM A @ Bᵀ) is performed by a Triton kernel.
- Mixed‑precision: fp16 loads, fp32 accumulation, fp32 output.
- Supports batch size 1‑2, sequence length up to 1024, hidden_dim = 128,
  arbitrary dim (e.g. 128 or 384) and optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # *fp16   (batch, M, K)   -> M = N, K = N
    B_ptr,                     # *fp16   (batch, N, K)   read as K×N (transpose)
    C_ptr,                     # *fp32   (batch, M, N)
    M, N, K,                   # matrix dims
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(2)          # batch * hidden
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current tile
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)               # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)               # (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)                               # (BLOCK_M, BLOCK_N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write back
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float) or None
        - weights      : dict of parameter tensors (float32)
        - config       : dict containing at least {"dim", "hidden_dim", "nomask"}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    w_norm, b_norm = weights["norm.weight"], weights["norm.bias"]
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    w_to_out_norm, b_to_out_norm = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    w_to_out = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last axis = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=w_norm, bias=b_norm)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, w_left_proj)    # [B, N, N, hidden_dim]
    right = F.linear(x, w_right_proj)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM (A @ Bᵀ) per hidden channel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim

    # reshape to (B*H, N, N) and make contiguous for correct strides
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (bandwidth‑friendlier)
    left_fp16 = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    # output buffer (fp32 accumulator)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # B is stored as (batch*H, N, N) → (b, row, col)
    # We need B as (K, N) where K is the original column index
    stride_Bk = stride_Bc            # step along K (original column)
    stride_Bn = stride_Br            # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # tile sizes – good for Hopper/SM90 up to N=1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * hidden
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=w_to_out_norm, bias=b_to_out_norm)
    out = out * out_gate
    out = F.linear(out, w_to_out)          # back to original dim C

    return out
