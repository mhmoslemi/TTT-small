```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused
batched matrix‑multiply kernel written in Triton.

Key ideas
---------
*   The heavy part is   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
    For each hidden channel d this is a matrix multiplication
        (N × N) @ (N × N)^T .
*   We reshape the tensors to (B*H, N, N) and launch a single Triton
    kernel that computes a batched GEMM for all (B*H) matrices.
*   Operands are loaded as float16 (memory bound) and accumulated in
    float32 (accuracy).  The final result is cast back to float32.
*   Tile sizes: BLOCK_M = BLOCK_N = 64, BLOCK_K = 32 – a good fit for
    Hopper SMs (SM‑90) and the tested sequence lengths.
*   All other operations (LayerNorm, linear projections, sigmoids,
    gating, final linear) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  (A @ Bᵀ)   where
#   A : (batch, M, K)     (float16)
#   B : (batch, N, K)     (float16) – we treat it as transposed inside the kernel
#   C : (batch, M, N)     (float32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_outgoing_kernel(
    A_ptr,                # const float16 *
    B_ptr,                # const float16 *
    C_ptr,                # float32 *
    M, N, K,              # matrix dimensions (int64)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_n, stride_b_k,   # strides for B
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    #  Program IDs
    # ---------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile row   index
    pid_n = tl.program_id(1)          # tile col   index
    pid_b = tl.program_id(2)          # batch*head index

    # ---------------------------------------------------------------
    #  Offsets within the tile
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K                # same for every iteration

    # ---------------------------------------------------------------
    #  Pointers to the beginning of the current tile
    # ---------------------------------------------------------------
    a_ptrs = (A_ptr + pid_b * stride_a_batch
              + offs_m[:, None] * stride_a_m
              + offs_k[None, :] * stride_a_k)

    b_ptrs = (B_ptr + pid_b * stride_b_batch
              + offs_n[None, :] * stride_b_n      # note: we treat B as (batch,N,K)
              + offs_k[:, None] * stride_b_k)    # and later dot with its K‑dim

    # ---------------------------------------------------------------
    #  Accumulator (float32)
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    #  Main K‑loop
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A and B (both in fp16)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ---------------------------------------------------------------
    #  Write back the result (float32)
    # ---------------------------------------------------------------
    c_ptrs = (C_ptr + pid_b * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (bool/float32 or None)
        - weights      : dict of parameter tensors (all float32, CUDA)
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # called hidden_dim in the paper
    B, N, _, _ = input_tensor.shape        # input shape is [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) → shape [B, N, N, hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Batched GEMM  (left @ rightᵀ)  – the expensive part
    # ------------------------------------------------------------------
    #   left/right : [B, N, N, hidden] → [B*hidden, N, N]
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_t.view(B * hidden, N, N)       # (B*H, M, K)
    right_flat = right_t.view(B * hidden, N, N)      # (B*H, N, K)

    # output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ---------------------------------------------------------------
    # launch configuration – tuned for Hopper
    # ---------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),          # blocks along N (cols)
        triton.cdiv(N, BLOCK_M),          # blocks along M (rows)
        B * hidden,                       # batch*head dimension
    )

    # strides (int64) – note the order: (batch, row, col)
    a_batch, a_m, a_k = left_flat.stride()
    b_batch, b_n, b_k = right_flat.stride()   # right is (batch, N, K)
    c_batch, c_m, c_n = out_flat.stride()

    _batched_gemm_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_m, a_k,
        b_batch, b_n, b_k,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout  [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣  Output LayerNorm (hidden) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

The kernel above follows the proven “permute‑to‑(B,H,N,N)” pattern, passes exact strides, uses FP16 loads with FP32 accumulation, and employs a 64 × 64 × 32 tiling that matches the memory bandwidth and shared‑memory limits of the NVIDIA H200 (SM‑90).  All remaining bookkeeping (LayerNorm, gating, final projection) stays in efficient PyTorch, giving a correct yet fast implementation of the outgoing TriMul operator.
