"""
TriMul “outgoing” Triton implementation (AlphaFold‑3 style)

The dominant work of the module is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We treat every (batch, hidden) pair as an independent GEMM and launch a
single Triton kernel that performs a tiled A @ Bᵀ for *M = B·hidden* matrices.
All cheap operations (LayerNorm, linear projections, sigmoids, masking) stay
in PyTorch.  The kernel works in FP16 for the loads (the model weights are
FP32) and accumulates in FP32 – the final result is cast back to FP32 to
match the reference implementation.

The kernel parameters (BLOCK_M, BLOCK_N, BLOCK_K) have been chosen for the
H200 (SM90) – 128×128 tiles for the output and a 64‑element K‑tile give good
SM occupancy while keeping shared‑memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, out
    N,                                 # sequence length
    stride_ah, stride_ai, stride_ak,   # left  (M, N, N) strides
    stride_bh, stride_bj, stride_bk,   # right (M, N, N) strides
    stride_ch, stride_ci, stride_cj,   # out   (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A @ Bᵀ where A,B have shape (M, N, N) and M = batch * hidden_dim.
    The kernel computes a separate (N,N) matrix product for each M‑slice.
    """
    pid_z = tl.program_id(2)          # flattened (batch * hidden)
    pid_m = tl.program_id(0)          # output‑row block
    pid_n = tl.program_id(1)          # output‑col block

    # ------------------------------------------------------------------
    # 1️⃣  Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2️⃣  Pointers to the current (M‑th) matrix pair
    # ------------------------------------------------------------------
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # accumulator in FP32 (more stable than pure FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3️⃣  Loop over K dimension (tile‑wise)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  → (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # right_T[k,j]  → (BLOCK_K, BLOCK_N)
        # original right is indexed as right[j,k]; we load it transposed.
        b_ptrs = (
            b_batch_ptr
            + offs_k[:, None] * stride_bk          # k‑dim (row of Bᵀ)
            + offs_n[None, :] * stride_bj          # j‑dim (col of Bᵀ)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # dot product (FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4️⃣  Store the result (cast back to FP32)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch_ptr
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.

    Args:
        left  : (B, H, N, N)   FP32 tensor
        right : (B, H, N, N)   FP32 tensor

    Returns:
        out   : (B, H, N, N)   FP32 tensor where out[b,h] = left[b,h] @ right[b,h]ᵀ
    """
    assert left.shape == right.shape, "left/right must share shape"
    B, H, N, _ = left.shape
    M = B * H                                         # flatten batch * hidden

    # reshape to (M, N, N) – contiguous memory layout is required
    left_f = left.reshape(M, N, N).contiguous()
    right_f = right.reshape(M, N, N).contiguous()
    out_f = torch.empty_like(left_f)

    # strides as int64 (needed by Triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out_f.stride()

    # Tunable tile sizes – empirically good for Hopper/SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch*hidden)
    )
    trimul_gemm_kernel[grid](
        left_f,
        right_f,
        out_f,
        N,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out_f.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point – the function required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    The heavy contraction is executed with a fused Triton GEMM kernel;
    all other ops (LayerNorm, Linear, Sigmoid, Masking) stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections: dim → hidden_dim  (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)        # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)           # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) followed by element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Triton GEMM: out = left @ rightᵀ   per hidden channel
    #    Permute to (B, H, N, N) layout expected by the kernel.
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm = trimul_gemm(left_perm, right_perm)      # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)     # (dim, hidden)
    out = F.linear(out, to_out_w)                      # (B, N, N, dim)

    # The reference implementation returns float32, keep that dtype.
    return out.to(torch.float32)
