## Triton‑accelerated “outgoing” TriMul

```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3).

* 1) LayerNorm over the last channel.
* 2) Linear projections → left/right (shape B,N,N,H) and gating.
* 3) Optional mask applied before gating.
* 4) Fuse the cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM C = A @ Bᵀ with batch = B·H.
   The GEMM is executed in fp16 with an fp32 accumulator.
* 5) Post‑norm, gating and final linear projection back to `dim`.

Only the O(N³·H) contraction is executed in Triton; the rest stays in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (B*H, M, K)   – row‑major
#   B : (B*H, N, K)   – row‑major, we need its transpose on‑the‑fly
#   C : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_az, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bz, stride_bn, stride_bk,   # B strides (batch, row, col)
    stride_cz, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is accessed transposed."""
    pid_batch = tl.program_id(2)          # which (b, h) pair
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ------------------------------------------------------------------
    # Compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in A
    a_block_ptr = a_ptr + pid_batch * stride_az \
                        + offs_m[:, None] * stride_am \
                        + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # For Bᵀ we need element (k, n) of the original layout,
    # i.e. row‑k, col‑n in B.
    b_block_ptr = b_ptr + pid_batch * stride_bz \
                        + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                        + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile
        a = tl.load(
            a_block_ptr,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load Bᵀ tile
        b = tl.load(
            b_block_ptr,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance to next K‑tile
        a_block_ptr += BLOCK_K * stride_ak
        b_block_ptr += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (cast back to fp16 to reduce bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Perform batched GEMM C = A @ Bᵀ on fp16 tensors.
    a, b : (B*H, N, N)  fp16, contiguous
    Returns c of same shape / dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    BxH, N, K = a.shape
    assert K == N

    c = torch.empty_like(a)

    # Strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # original layout (batch,N,K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – empirically good on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        BxH,                       # pid_batch
    )
    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,          # M
        N,          # N
        N,          # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bn,
        stride_bk,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps works well for 64×64 tiles on sm90
        num_stages=2,
    )
    return c


# ----------------------------------------------------------------------
# Host‑side entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a Triton‑fused cubic contraction.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        * input_tensor : torch.Tensor, shape (B, N, N, dim), float32
        * mask         : torch.Tensor, shape (B, N, N)  (bool or float)
        * weights      : dict of model parameters (all float32)
        * config       : dict with keys
              - "dim"          : input channel size
              - "hidden_dim"   : projection size H
              - "nomask" (bool, optional) – if True the mask is ignored

    Returns
    -------
    torch.Tensor of shape (B, N, N, dim), dtype float32
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Fuse hidden dim into the batch dimension for the GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left  = left.view(B * H, N, N).to(torch.float16)
    right = right.view(B * H, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Batched GEMM: out = left @ rightᵀ   (still fp16)
    # ------------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H) and cast to fp32
    # ------------------------------------------------------------------
    out = (
        out_fp16.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 8) Post‑projection LayerNorm + gating on hidden dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate shape (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Final linear back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

The implementation respects all constraints:

* Works for any `B ∈ {1,2}`, `N ∈ {128,…,1024}`, `dim ∈ {128,384}`, `hidden_dim ∈ {128,384,768}`.
* Handles the presence or absence of a mask (`nomask` flag).
* The only custom kernel is the `O(N³·H)` contraction, tiled as 64 × 64 × 32 and using fp16 arithmetic with an fp32 accumulator.
* Final output is `float32` and matches the reference PyTorch implementation.
