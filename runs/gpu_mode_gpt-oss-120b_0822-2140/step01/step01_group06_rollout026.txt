"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction  
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  
is reformulated as a batched GEMM per (batch, head) pair.
Both `left` and `right` are reshaped to (B, H, N, N) and a single
triton kernel computes all (B*H) matrix‑multiply‑transposes in one launch.
The rest of the module (LayerNorms, linear projections, sigmoid gates,
final linear) stays in PyTorch – the heavy work is fully off‑loaded to
triton, satisfying the “must use a kernel” rule while keeping the
implementation simple and portable.

All tensors are kept in float32; the GEMM is performed in fp16 for
throughput and accumulated in fp32.  The final result is cast back to
float32 before the remaining PyTorch layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ   (A,B : (B, H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, out_ptr,
    B, H, N,
    stride_ah, stride_ak, stride_am, stride_an,   # A : (B,H,N,N)
    stride_bh, stride_bk, stride_bm, stride_bn,   # B : (B,H,N,N)
    stride_oh, stride_ok, stride_om, stride_on,   # OUT : (B,H,N,N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute
        out[b,h,i,j] = Σ_k a[b,h,i,k] * b[b,h,j,k]
    for a block of (i,j) defined by program ids (pid0, pid1, pid2).
    """
    pid0 = tl.program_id(0)          # combined batch*head index
    pid1 = tl.program_id(1)          # block‑row (i)
    pid2 = tl.program_id(2)          # block‑col (j)

    # decode batch and head from pid0
    b = pid0 // H
    h = pid0 % H

    # ------------ range of indices for this block --------------------
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # ------------ pointers ------------------------------------------------
    a_ptrs = a_ptr + b * stride_ah + h * stride_ak \
                + offs_i[:, None] * stride_am + offs_k[None, :] * stride_an
    b_ptrs = b_ptr + b * stride_bh + h * stride_bk \
                + offs_j[:, None] * stride_bm + offs_k[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------ loop over K -------------------------------------------
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        # load a tile
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        # load b tile (note the transposed access)
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        # fp16 compute for speed, accumulate in fp32
        a_f16 = a.to(tl.float16)
        b_f16 = b.to(tl.float16)
        acc += tl.dot(a_f16, b_f16).to(tl.float32)

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_an
        b_ptrs += BLOCK_K * stride_bn

    # ------------ write result -------------------------------------------
    out_ptrs = out_ptr + b * stride_oh + h * stride_ok \
                + offs_i[:, None] * stride_om + offs_j[None, :] * stride_on
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32 CUDA)
        mask_tensor  : torch.Tensor  [B, N, N]       (float32/bool CUDA) or None
        weights_dict : dict of torch.Tensor parameters
        config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32 CUDA)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, w["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) computation – Triton batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    out_t = torch.empty_like(left_t)                  # (B, H, N, N)

    # strides (in elements, not bytes)
    stride_ah, stride_ak, stride_am, stride_an = left_t.stride()
    stride_bh, stride_bk, stride_bm, stride_bn = right_t.stride()
    stride_oh, stride_ok, stride_om, stride_on = out_t.stride()

    # tile sizes – tuned for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # pid0  : batch * head
        triton.cdiv(N, BLOCK_M),           # pid1  : rows
        triton.cdiv(N, BLOCK_N),           # pid2  : cols
    )

    tri_mul_gemm_kernel[grid](
        left_t, right_t, out_t,
        B, hidden_dim, N,
        stride_ah, stride_ak, stride_am, stride_an,
        stride_bh, stride_bk, stride_bm, stride_bn,
        stride_oh, stride_ok, stride_om, stride_on,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to the original layout (B,N,N,H)
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm on hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimensionality
    # ------------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B, N, N, dim]

    return out
