"""
TriMul “outgoing” implementation with a fused batched GEMM kernel.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply of shape (B·H, N, N) with the transpose of the
second operand.  We rewrite the four‑dimensional tensors to (B, H, N, N),
flatten the first two dimensions (M = B*H) and run a single Triton kernel that
computes many independent GEMMs in parallel:

    out[m, i, j] = Σ_k left[m, i, k] * right[m, j, k]   (right accessed as k‑j)

The kernel works in float16 for the loads and accumulates in float32 for
numerical stability.  All surrounding LayerNorm, Linear and gating logic stays
in PyTorch – this gives a large speed‑up while keeping the implementation
easy to verify.

The kernel is launched with three‑dimensional grid:
    dim‑0 : tiles over rows   (i)
    dim‑1 : tiles over cols   (j)
    dim‑2 : different matrices (M = B*hidden_dim)

The launch parameters (BLOCK_M, BLOCK_N, BLOCK_K) were chosen to give high
occupancy on Hopper (SM‑90) while keeping shared‑memory pressure low.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ   where left/right have shape (M, N, N)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16  [M, N, N]
    right_ptr,         # *fp16  [M, N, N]
    out_ptr,           # *fp32  [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides (elements)
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # which matrix (m = batch*head)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers (row‑major layout)
    # --------------------------------------------------------------
    # left  : (M, N, K)   index (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                         + offs_m[:, None] * stride_ln \
                         + offs_k[None, :] * stride_lk

    # right : we need Bᵀ on the last two dims => access as (z, k, j)
    b_ptrs = right_ptr + pid_z * stride_rm \
                          + offs_k[:, None] * stride_rk \
                          + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot works on fp16 inputs, result is fp32 (as we accumulate)
        acc += tl.dot(a, b)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # --------------------------------------------------------------
    # Store the result (float32)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask         : torch.Tensor[B, N, N]       (bool/float, may be None)
        - weights      : dict of torch.Tensor (all float32)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor[B, N, N, dim]  (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections: left, right  (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Fuse the N³ contraction with Triton.
    #     Permute to (B, hidden, N, N) → view as (M, N, N) where M = B*hidden
    # ------------------------------------------------------------------- #
    left_f = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_f = right.permute(0, 3, 1, 2).contiguous()

    # cast to fp16 for the kernel (still accurate enough for protein models)
    left_f16 = left_f.to(torch.float16)
    right_f16 = right_f.to(torch.float16)

    # flatten the first two dims
    M = B * hidden
    left_view = left_f16.view(M, N, N)
    right_view = right_f16.view(M, N, N)

    out_view = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Strides in **elements** (not bytes) – required by the kernel
    # --------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_view.stride()
    stride_rm, stride_rn, stride_rk = right_view.stride()
    stride_om, stride_on, stride_ok = out_view.stride()

    # --------------------------------------------------------------
    # Kernel launch configuration
    # --------------------------------------------------------------
    # These tiles were found to give high occupancy on Hopper.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        M,                         # independent GEMMs
    )

    tri_mul_gemm_kernel[grid](
        left_view,
        right_view,
        out_view,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on SM90
    )

    # --------------------------------------------------------------
    # Reshape back to [B, N, N, hidden]
    # --------------------------------------------------------------
    out = out_view.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm, gating and final linear (dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
