"""
TriMul “outgoing” (AlphaFold‑3) – fast Triton implementation
================================================================
The expensive part of the forward pass is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel *d* this is a matrix‑multiply
(N × N) @ (N × N)ᵀ.  We treat each hidden channel as an
independent batch element, i.e. we compute a batched
matrix‑multiply of shape (B·H, N, N) × (B·H, N, N)ᵀ.
The kernel below uses fp16 arithmetic (tensor‑core friendly)
with an fp32 accumulator, a 2‑D tiling (BLOCK_M × BLOCK_N)
and a 1‑D K‑tile (BLOCK_K).  The host code does the
layer‑norms, linear projections, gating and the final
layer‑norm+linear in PyTorch – only the cubic work is
off‑loaded to Triton.

The kernel works for any batch size ≤ 2, any sequence
length up to 1024 and arbitrary hidden dimension (the
hidden dimension is folded into the batch dimension).
It also supports an optional mask; when *nomask* is true
the mask tensor is ``None``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B*H, N, N) @ (B*H, N, N)ᵀ  →  (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_outgoing_kernel(
    A_ptr,                     # [B*H, N, K]   (K == N)
    B_ptr,                     # [B*H, N, K]   (K == N)
    C_ptr,                     # [B*H, N, N]   result
    M, N, K,                   # matrix sizes (M==N==K==seq_len)
    stride_ah, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bm, stride_bk,   # strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A @ Bᵀ   where A and B have the same shape.
    The kernel is launched with a 3‑D grid:
        pid_m – tile index in the output‑row dimension
        pid_n – tile index in the output‑col dimension
        pid_h – batch*hidden index
    """

    pid_m = tl.program_id(0)          # tile along M (output rows)
    pid_n = tl.program_id(1)          # tile along N (output cols)
    pid_h = tl.program_id(2)          # batch*hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the tile
    a_ptrs = A_ptr + pid_h * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_h * stride_bh + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # ------------------------------------------------------------------
    # Accumulator in fp32 (tensor‑core friendly)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles – cast to fp16 for the matmul, store as fp32 for accumulation
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float16)

        # Tensor‑core dot (fp16 * fp16 → fp32)
        acc += tl.dot(a, b)

        # Advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (fp32)
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_h * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data : (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]  (float32, CUDA)
            mask         : torch.Tensor [B, N, N] or None
            weights      : dict containing all linear / LN parameters
            config       : {'dim': int, 'hidden_dim': int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # hidden_dim == c_z in the paper
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B, N, N, 1]
        left  = left  * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – same shape as left/right
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Fuse hidden channel into batch dimension and launch Triton
    # --------------------------------------------------------------
    #   left  : [B, N, N, H]  ->  [B*H, N, N]
    #   right : [B, N, N, H]  ->  [B*H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left_perm.view(B * hidden, N, N)
    right_flat = right_perm.view(B * hidden, N, N)

    # Cast to fp16 for tensor‑core speed – the kernel will accumulate in fp32
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    out_flat_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)   # result in fp32

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for H200)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64                         # fits 128‑bit (8 fp16) registers nicely

    grid = (
        triton.cdiv(N, BLOCK_N),          # tiles over output‑col (N)
        triton.cdiv(N, BLOCK_M),          # tiles over output‑row (N)
        B * hidden,                       # batch*hidden dimension
    )

    # Strides (batch, row, col) for the three tensors
    a_batch, a_row, a_col = left_fp16.stride()
    b_batch, b_row, b_col = right_fp16.stride()
    c_batch, c_row, c_col = out_flat_fp32.stride()

    batched_matmul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat_fp32,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out = out_flat_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm (over hidden dim) + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

### How the lessons were applied
* **Lesson 1 (permute to (B, H, N, N))** – implemented before the kernel; this gives
  contiguous memory for the two matrix‑multiply operands.
* **Lesson 2 (correct strides)** – the exact strides of the three tensors are
  computed on the host (`a_batch, a_row, a_col …`) and passed to the kernel;
  the kernel uses them for every load/store.
* **New ideas** – use **fp16 arithmetic with an fp32 accumulator** (tensor‑core
  friendly) and a **64 × 64 × 64 tiling** that matches the Hopper
  architecture, providing a large speed‑up over the previous 128‑wide
  implementation.
* Only the cubic work (`left @ rightᵀ`) is in Triton; the cheap
  LayerNorms, Linear layers, and sigmoids stay in highly‑optimized PyTorch,
  keeping the overall implementation simple yet performant.
