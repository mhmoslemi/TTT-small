"""
TriMul (outgoing) – fast forward pass.

The cubic contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched GEMM with a custom Triton kernel.
All other operations (LayerNorm, Linear, sigmoids, final projection)
stay in PyTorch for simplicity.  The kernel works on tensors laid out as

    (B, H, M, K)   where M = N = K = seq_len and H = hidden_dim

and writes the result into (B, H, M, N).  The kernel accumulates in
float32 (inputs may be cast to float16 for higher throughput) and
uses three‑dimensional grid indexing:
    pid0 – batch * head
    pid1 – block row
    pid2 – block col
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³·H) part.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                # left   (B,H,M,K)  fp16 (or fp32)
    b_ptr,                # right  (B,H,N,K)  fp16 (or fp32)
    c_ptr,                # out    (B,H,M,N)  fp32
    B, H, M, N, K,        # static sizes
    stride_abh, stride_ah, stride_am, stride_ak,   # a strides
    stride_bbh, stride_bh, stride_bn, stride_bk,   # b strides
    stride_cbh, stride_ch, stride_cm, stride_cn,   # c strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Identify the (batch, head) pair this program works on
    # ------------------------------------------------------------------
    pid_bh = tl.program_id(0)                # 0 … B*H-1
    batch_idx = pid_bh // H
    head_idx  = pid_bh %  H

    pid_m = tl.program_id(1)                 # block row index
    pid_n = tl.program_id(2)                 # block col index

    # ------------------------------------------------------------------
    # Compute absolute offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------------------- load A tile (M x K) -----------------------
        a_ptrs = a_ptr \
            + batch_idx * stride_abh \
            + head_idx  * stride_ah \
            + offs_m[:, None] * stride_am \
            + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ------------------- load B tile (N x K) -----------------------
        # we need B transposed (K x N) for the dot
        b_ptrs = b_ptr \
            + batch_idx * stride_bbh \
            + head_idx  * stride_bh \
            + offs_k[:, None] * stride_bk \
            + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # ------------------- compute tile -------------------------------
        # a : (BLOCK_M, BLOCK_K), b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)          # fp32 accumulation

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = c_ptr \
        + batch_idx * stride_cbh \
        + head_idx  * stride_ch \
        + offs_m[:, None] * stride_cm \
        + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point – called by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] on the same device as the input
        (float32).
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    device     = input_tensor.device
    B, N, _, _ = input_tensor.shape          # input shape: [B, N, N, dim]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (over last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                         # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if supplied)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction via a Triton batched GEMM
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for deterministic strides
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  <-- (M=i, K=k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  <-- (N=j, K=k)

    # optional mixed‑precision: cast to fp16 for the kernel work
    left_fp16  = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # output buffer (FP32)
    out_buf = torch.empty((B, hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                # batch * head dimension
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )

    # extract strides (in elements, not bytes) for the three tensors
    s_ab, s_ah, s_am, s_ak = left_fp16.stride()   # (B, H, M, K)
    s_bb, s_bh, s_bn, s_bk = right_fp16.stride()  # (B, H, N, K)
    s_cb, s_ch, s_cm, s_cn = out_buf.stride()    # (B, H, M, N)

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        B, hidden_dim, N, N, N,           # M=N, N=N, K=N
        s_ab, s_ah, s_am, s_ak,
        s_bb, s_bh, s_bn, s_bk,
        s_cb, s_ch, s_cm, s_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=True,                        # just to keep the constexpr
    )

    # bring back to original layout: (B, N, N, H)
    out = out_buf.permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                         # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise product with the output gate
    # --------------------------------------------------------------
    out = out * out_gate                     # still [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimensionality
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
