"""
TriMul “outgoing” forward pass.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply of shape (B·H, N, N) with the transpose of the
second operand.  We fuse this N³ term in a single Triton kernel that

* works on fp16 inputs (the linear projections are cast to half‑precision)
* accumulates in fp32 via `tl.dot`
* uses a 64×64×64 tiling that matches the Hopper SM‑90 tensor‑core shape
* launches one program‑id per (row‑tile, col‑tile, matrix‑id) triple.

All other operations (LayerNorm, Linear projections, gating, final
LayerNorm/Linear) stay in PyTorch – they are cheap compared to the O(N³)
work.  The kernel therefore gives a large speed‑up while keeping the
implementation simple and numerically stable (final result stays in fp32).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ   where
#   left  : (M, N, K)   fp16
#   right : (M, N, K)   fp16
#   out   : (M, N, N)   fp32
#   M = B * hidden_dim   ,  K == N
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_fp16_kernel(
    left_ptr,          # *fp16   [M, N, K]
    right_ptr,         # *fp16   [M, N, K]
    out_ptr,           # *fp32   [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over output rows   (i)
    pid_n = tl.program_id(1)          # tile over output cols   (j)
    pid_z = tl.program_id(2)          # which matrix (batch * head)

    # ------------------------------------------------------------------- #
    # Offsets for this tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers to the start of the tiles
    # ------------------------------------------------------------------- #
    # left : (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm + \
             offs_m[:, None] * stride_ln + \
             offs_k[None, :] * stride_lk

    # right transposed on the last two dims: we need Bᵀ  -> (z, k, j)
    b_ptrs = right_ptr + pid_z * stride_rm + \
             offs_k[:, None] * stride_rk + \
             offs_n[None, :] * stride_rn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main GEMM loop over K dimension
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

        # advance K‑dim pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Store the result (fp32)
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om + \
               offs_m[:, None] * stride_on + \
               offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data : (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            mask         : torch.Tensor [B, N, N]      (bool/float) or None
            weights      : dict of torch.Tensor parameters
            config       : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]          # called hidden_dim in the paper

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections  (no bias) → fp16 for speed
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)       # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) – stay in fp32 for numerical stability
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate.to(left.dtype)
    right = right * right_gate.to(right.dtype)

    # ------------------------------------------------------------------- #
    # 5. Reshape to (M, N, N) where M = B * hidden
    # ------------------------------------------------------------------- #
    # (B,N,N,hidden) -> (B, hidden, N, N) -> (M, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # output buffer (fp32)
    out_reshape = torch.empty((B * hidden, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 6. Gather strides (in elements, not bytes)
    # ------------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # ------------------------------------------------------------------- #
    # 7. Triton launch configuration
    #    Tile size 64×64×64 matches SM‑90 tensor‑core shape and gives good
    #    occupancy for the (B·hidden) dimension up to 384.
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows  (i)
        triton.cdiv(N, BLOCK_N),          # cols  (j)
        B * hidden                        # independent matrices
    )

    tri_mul_fp16_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps ≈ 2 tensor‑core warps per SM on Hopper
    )

    # ------------------------------------------------------------------- #
    # 8. Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------- #
    out = out_reshape.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 9. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim, fp32

    return out
