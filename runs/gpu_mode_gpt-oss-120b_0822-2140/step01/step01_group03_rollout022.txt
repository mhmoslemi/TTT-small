"""
TriMul “outgoing” Triton implementation
=====================================

The forward pass of AlphaFold‑3’s TriangleMultiplicativeUpdate (outgoing)
can be written as a batched GEMM:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

For every head *h* we have a matrix multiplication between
`left_h  ∈ ℝ^{N×N}` and the transpose of `right_h ∈ ℝ^{N×N}`.
All other operations (layer‑norms, linear projections, sigmoid gates and the
final projection) are cheap compared to the N³ contraction, so we keep them
in PyTorch and only off‑load the batched GEMM to a custom Triton kernel.

Key ideas
---------
*   The kernel works on fp16 data (half‑precision loads) but accumulates in
    fp32 to retain numeric quality.
*   The kernel processes **B·H** independent GEMMs in parallel; each GEMM is
    tiled with three compile‑time constants (BLOCK_M, BLOCK_N, BLOCK_K).
*   We avoid an explicit mask inside the kernel – the mask is applied before
    the GEMM, which is cheaper because it is a simple broadcasted
    element‑wise multiplication on contiguous tensors.
*   The kernel uses the same layout as the reference implementation
    (`[B, H, N, N]` → view `(B·H, N, N)`) and therefore needs only the
    usual three‑dimensional strides.  The “transpose” of the right operand is
    achieved by swapping the K‑ and N‑strides when loading.

The implementation below works for all benchmark configurations
(N up to 1024, batch ≤ 2, hidden_dim = 128, and arbitrary dim).  It
returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr,                     # *fp16   (batch, M, K)   – left
    B_ptr,                     # *fp16   (batch, N, K)   – right (read as K×N)
    C_ptr,                     # *fp32   (batch, M, N)   – output
    M, N, K,                   # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note the swapped K/N)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of GEMMs in float16 -> float32:
        C[b, m, n] = Σ_k A[b, m, k] * B[b, n, k]   (i.e. Bᵀ)
    """
    pid_b = tl.program_id(2)            # batch index (B*H)
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block column

    # ------------------------------------------------------------------
    # Block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the current tile
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    # B is logically stored as (batch, N, K) but we want Bᵀ → (K, N)
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk   # step along K (original column)
        + offs_n[None, :] * stride_Bn   # step along N (original row)
    )

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)            # fp16 → fp32
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # tl.dot does a small‑matrix multiply‑accumulate in fp32
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float)
        - weights      : dict of learned parameters (all float32)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = (
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ Pre‑linear + layer‑norm (still in PyTorch – cheap)
    # ------------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # Linear projections (bias‑free)
    left = F.linear(x, left_proj_w)          # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)        # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in PyTorch
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction – batched GEMM in Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                 # H = hidden_dim

    # (B, H, N, N) → (B*H, N, N) contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_view = left_perm.view(B * H, N, N)
    right_view = right_perm.view(B * H, N, N)

    # fp16 inside the kernel (half the memory traffic)
    left_fp16 = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # output buffer – fp32 accumulator → final result
    out_buf = torch.empty_like(left_view, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 6.1 Strides (in elements, not bytes) – required by the kernel
    # ------------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # We will read B as (K,N) → swap the last two strides
    stride_Bk = stride_Bc                     # step along original column (K)
    stride_Bn = stride_Br                     # step along original row    (N)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # 6.2 Tile size – tuned for Hopper (SM90).  Small N (256) still works fine.
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch × head
    )

    _tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, hidden] and post‑processing
    # ------------------------------------------------------------------
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)                # back to original dim

    return out
