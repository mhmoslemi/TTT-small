"""
TriMul “outgoing” (AlphaFold‑3) – forward only.
The only expensive part is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply (N×N) @ (N×N)ᵀ for every hidden channel *d*.
We fuse that multiplication in a single Triton kernel that works on
(B·hidden) independent matrices of size N×N.
All other ops (LayerNorm, Linear, gating) stay in PyTorch – they are cheap
compared to the O(N³) work.

Implementation details
----------------------
*   Input tensors are float32 on the device.
*   After LayerNorm, Linear and gating we cast *left* and *right* to FP16 –
    the H200 tensor cores compute the dot in FP16 while the accumulator is
    FP32 (tl.dot does this automatically).  This gives a ~2× speed‑up.
*   We permute the tensors from (B,N,N,H) → (B,H,N,N) and flatten the first
    two dimensions so the kernel sees a 3‑D tensor (batch, M, K) where
    batch = B·H, M = N, K = N.
*   The kernel tiles the output with BLOCK_M × BLOCK_N tiles and walks the
    reduction dimension K in BLOCK_K steps.  Tile sizes (128,128,32) are a
    good trade‑off on Hopper (SM‑90) – they keep each SM busy and fit
    comfortably in shared memory.
*   Strides are passed explicitly; no assumption about contiguity is made.
*   The kernel returns FP32 results, which are reshaped back to the original
    (B,N,N,H) layout and processed by the final LayerNorm + linear projection.

The host‑side `custom_kernel` follows the exact API required by the evaluator.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: (B·H) matrix‑multiply C = A @ Bᵀ   (A,B : N×N)
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes (M=N=N, K=N)
    stride_ab, stride_am, stride_ak,   # strides for A: (batch, row, col)
    stride_bb, stride_bm, stride_bk,   # strides for B: (batch, row, col)
    stride_cb, stride_cm, stride_cn,   # strides for C: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over output rows   (M)
    pid_n = tl.program_id(1)          # tile over output cols   (N)
    pid_b = tl.program_id(2)          # batch * hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile (A row‑major, B will be read as row‑major
    # but interpreted as transposed because we stride with b_m on the "n" axis)
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles – masks prevent out‑of‑bounds reads
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Public entry point – follows the required signature
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]       (bool/float, CUDA) or None
        - weights      : dict of torch.Tensors (all float32, CUDA)
        - config       : dict with keys "dim" (input dim) and "hidden_dim"

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    B, N, _, _   = input_tensor.shape      # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → hidden dim
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also applied before the heavy multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 for the GEMM (Tensor‑core friendly)
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Rearrange to (B, hidden, N, N) → flatten first two dims.
    #     The kernel sees (batch = B*hidden, M = N, K = N).
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    left  = left.view(B * hidden_dim, N, N)          # [B*H, N, N]
    right = right.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 7️⃣ Allocate output buffer (float32 – accumulator type)
    # ------------------------------------------------------------------
    out_fp32 = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=left.device)

    # ------------------------------------------------------------------
    # 8️⃣ Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # program_id(0) – rows
        triton.cdiv(N, BLOCK_N),          # program_id(1) – cols
        B * hidden_dim,                   # program_id(2) – batch*hidden
    )

    # Strides (batch, row, col) in elements, not bytes
    a_batch, a_row, a_col = left.stride()
    b_batch, b_row, b_col = right.stride()
    c_batch, c_row, c_col = out_fp32.stride()

    _batched_gemm_outgoing_kernel[grid](
        left,
        right,
        out_fp32,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 10️⃣ Output LayerNorm (hidden) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
