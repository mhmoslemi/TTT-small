"""
TriMul “outgoing” forward pass (no gradients).

The O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication:
    left_slice  : (N, N)   = left[b, :, :, d]
    right_T     : (N, N)   = right[b, :, :, d].transpose(-2, -1)
    out_slice   = left_slice @ right_T

We fuse the B×H independent GEMMs in a single Triton kernel.
All other operations (LayerNorm, Linear projections, sigmoid gates) are
performed with PyTorch – they are cheap compared with the N³ work.
The kernel works in FP16 internally, accumulates in FP32 and stores the
result back as FP16; the host then casts to FP32 and continues.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape [B, M, K], [B, N, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_az, stride_am, stride_ak,  # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,  # strides for B (batch, row, col)
    stride_cz, stride_cm, stride_cn,  # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM where B is logically transposed.
    Each program ID corresponds to a (batch, tile_row, tile_col).
    """
    pid_batch = tl.program_id(2)   # batch (B*H)
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current tile
    a_ptrs = a_ptr + pid_batch * stride_az \
                   + offs_m[:, None] * stride_am \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- K loop --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # load Bᵀ tile
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot product
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------- write back --------------------
    c = acc.to(tl.float16)                     # store as fp16
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches `batched_gemm_kernel` on two fp16 tensors
    of shape (B*H, N, N).  Returns the fp16 result tensor of the same shape.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "left/right tensors must be square"

    c = torch.empty_like(a)

    # strides in **elements**
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – good defaults for Hopper/SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles in M dimension
        triton.cdiv(N, BLOCK_N),   # tiles in N dimension
        B_H,                       # batch dimension (B * hidden_dim)
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Host entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool / float)
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys
            * "dim"          – input channel dimension
            * "hidden_dim"  – projection dimension
            * "nomask" (bool) – True if mask should be ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])       # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)        # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Reshape for the GEMM: (B, H, N, N) -> (B*H, N, N) and cast to fp16
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*head dimension
    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Batched GEMM C = left @ rightᵀ   (still fp16)
    # ------------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # shape (B*H, N, N)

    # ------------------------------------------------------------------
    # 7. Restore original shape and dtype (float32)
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N)          # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).to(torch.float32)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Post‑projection LayerNorm (hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # broadcasted over (B, N, N, H)

    # ------------------------------------------------------------------
    # 9. Final linear back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
