"""
TriMul “outgoing” kernel – fast Triton reduction.

The expensive part of the original PyTorch implementation is the
pair‑wise contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is mathematically a batched matrix‑multiply
    out[b, h] = left[b, h] @ right[b, h].T
for every (batch, head) pair.

We move this O(B·H·N³) work into a single Triton GEMM kernel.
All other operations (LayerNorms, linear projections, gating, optional mask)
remain in PyTorch – they are negligible compared with the N³ reduction.

The kernel works on tensors in layout (B, H, N, N) (contiguous) and
stores the result in the same layout; a cheap permute brings the tensor
back to the original (B, N, N, H) shape.
Mixed‑precision (float16) is used inside the kernel while the final output
remains float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction
# --------------------------------------------------------------------------- #
@triton.jit
def trmul_gemm_kernel(
    # pointers
    a_ptr,          # left  : (B, H, M, K)
    b_ptr,          # right : (B, H, N, K)   (we will read it column‑wise → transposed)
    c_ptr,          # out   : (B, H, M, N)
    # sizes
    B, H, M, N, K,
    # strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak,   # a: (B, H, M, K)
    stride_bh, stride_bn, stride_bk,   # b: (B, H, N, K)
    stride_ch, stride_cm, stride_cn,   # c: (B, H, M, N)
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (b, h):
        C = A @ A_right.T
    where
        A    = left [M, K]   (M = N, K = N)
        A_right = right [N, K]
    The kernel accumulates in fp32 and writes fp32.
    """

    pid = tl.program_id(0)

    # ------------------------------------------------------------------- #
    # 1) map linear pid -> (b, h, block_m, block_n)
    # ------------------------------------------------------------------- #
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    total_blocks_per_batch_head = num_blocks_m * num_blocks_n
    bh = pid // total_blocks_per_batch_head                     # 0 .. B*H-1
    rem = pid % total_blocks_per_batch_head
    block_m = rem // num_blocks_n
    block_n = rem % num_blocks_n

    # ------------------------------------------------------------------- #
    # 2) coordinate offsets for this tile
    # ------------------------------------------------------------------- #
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for edge tiles
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # 3) accumulate
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- load A tile (M x K) -----
        a = tl.load(
            a_ptr
            + bh * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=(mask_m[:, None] & mask_k[None, :]),
            other=0.0,
        )  # (BLOCK_M, BLOCK_K)

        # ----- load B tile (K x N) -----
        # B is stored as (B, H, N, K).  We need the transposed view (K, N).
        b = tl.load(
            b_ptr
            + bh * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=(mask_k[:, None] & mask_n[None, :]),
            other=0.0,
        )  # (BLOCK_K, BLOCK_N)

        # dot‑product accumulation
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------- #
    # 4) write result
    # ------------------------------------------------------------------- #
    tl.store(
        c_ptr
        + bh * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        acc,
        mask=(mask_m[:, None] & mask_n[None, :]),
    )


# --------------------------------------------------------------------------- #
# Helper – element‑wise mask (optional, stays in PyTorch)
# --------------------------------------------------------------------------- #
def apply_mask(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """x: [B,N,N,H], mask: [B,N,N] -> broadcasted multiplication."""
    if mask is None:
        return x
    # mask has shape (B,N,N); unsqueeze last dim and rely on broadcasting
    return x * mask.unsqueeze(-1)


# --------------------------------------------------------------------------- #
# Main entry point required by the problem statement
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy pair‑wise reduction is performed by a Triton GEMM kernel.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim] (float32, CUDA)
        - mask         : [B, N, N]     (float32 0/1)  or None
        - weights      : dict of tensors (LayerNorm & Linear params)
        - config       : dict with keys
              dim        – input channel size
              hidden_dim – projection size (the “head” dimension H)
              nomask     – bool, True ⇒ mask is all‑ones

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # 0️⃣ unpack
    # ------------------------------------------------------------------- #
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------- #
    def layer_norm(x, w, b):
        return F.layer_norm(
            x,
            normalized_shape=(w.shape[0],),
            weight=w,
            bias=b,
            eps=1e-5,
        )

    x = layer_norm(inp, weights["norm.weight"], weights["norm.bias"])  # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------- #
    def linear(x, w):
        # w: [out_dim, in_dim] , bias is omitted by definition
        return F.linear(x, w)                         # keeps last dim

    left = linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcasted over H)
    # ------------------------------------------------------------------- #
    if not nomask:
        left = apply_mask(left, mask)
        right = apply_mask(right, mask)

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projection)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Triton GEMM – core TriMul reduction
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape               # H == hidden_dim
    # cast to fp16 for bandwidth; accumulation stays in fp32 inside kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape to (B, H, N, N) – the kernel expects this layout
    left_t = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right_fp16.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # output placeholder
    out_t = torch.empty_like(left_t, dtype=torch.float32)  # (B, H, N, N)

    # strides (in elements)
    stride_ah, stride_am, stride_ak = left_t.stride()  # B, H, N, N order -> (B*H*N, H*N, N, 1)
    stride_bh, stride_bn, stride_bk = right_t.stride()
    stride_ch, stride_cm, stride_cn = out_t.stride()

    # ------------------------------------------------------------------- #
    # 5.1 launch Triton kernel
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_blocks = B * H * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    trmul_gemm_kernel[grid](
        a_ptr=left_t,
        b_ptr=right_t,
        c_ptr=out_t,
        B=B,
        H=H,
        M=N,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # out_t now holds [B, H, N, N] fp32

    # bring back to original layout [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1).contiguous()      # (B,N,N,H)

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection (hidden_dim → dim)
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
