assistantfinal**Solution Overview**

We implement the *outgoing* TriMul operator (`TriMul.forward` in the reference PyTorch code) with a **single Triton kernel** that performs the heavy
`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]` contraction.
All cheap operations (LayerNorm, linear projections, gates, final linear) stay in PyTorch – they are negligible compared with the `O(N³·H)` work.
The kernel fuses the **batched GEMM**  

```
C = A @ Bᵀ            # A,B : (B·H, N, N)
```

where each head `h` of each batch `b` is an independent matrix‑multiply.
We work in **float16** inside the kernel (the model tolerates this) and accumulate
in **float32** for numerical stability.  
After the kernel the result is reshaped back to `[B, N, N, H]`, Layer‑Normed,
gated and projected to the original dimension.

The implementation follows the Triton launch rules and is tuned for the
Hopper SM₉₀ architecture (block size 128 × 128, K‑tile 32, 4 warps).

```python
"""
TriMul “outgoing” forward pass (no gradients).

The dominant work is the O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiplication C = A @ Bᵀ with shape
    A,B : (B·H, N, N)   (float16)
    C   : (B·H, N, N)   (float16)

We implement this contraction in a single Triton kernel; the surrounding
LayerNorms, linear projections, gates and final projection are kept in PyTorch
for readability and negligible cost.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape [B·H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (int)
    stride_az, stride_am, stride_ak,         # stride of A: (batch, row, col)
    stride_bz, stride_br, stride_bc,         # stride of B: (batch, row, col)
    stride_cz, stride_cm, stride_cn,         # stride of C: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is transposed (Bᵀ)."""
    pid_batch = tl.program_id(2)   # batch × head dimension (B·H)
    pid_m = tl.program_id(0)       # block row
    pid_n = tl.program_id(1)       # block col

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile.  For Bᵀ we need to read B[k, n]
    a_ptrs = a_ptr + pid_batch * stride_az \
                + offs_m[:, None] * stride_am \
                + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    b_ptrs = b_ptr + pid_batch * stride_bz \
                + tl.arange(0, BLOCK_K)[:, None] * stride_br \
                + offs_n[None, :] * stride_bc

    # Accumulator in FP32 (more accurate than pure FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A (M×K) and a tile of Bᵀ (K×N)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # TL dot does FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        # Advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_br

    # --------------------------------------------------------------
    # Write‑back (store as FP16 to reduce bandwidth)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel (FP16 tensors)
# ----------------------------------------------------------------------
def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : (B·H, N, N) float16, contiguous, row‑major.
    Returns c of the same shape / dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "square matrices are required"

    c = torch.empty_like(a)

    # Strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()   # (batch, row, col)
    stride_bz, stride_br, stride_bc = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32          # fits nicely in registers on Hopper

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B_H,                       # batch * head
    )
    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_br,
        stride_bc,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,      # tuned for 128×128 tiles on sm90
        num_stages=2,
    )
    return c


# ----------------------------------------------------------------------
# The public entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul “outgoing” implementation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N] (bool / float)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
            * "dim"          – input/output channel dimension
            * "hidden_dim"   – inner dimension H
            * "nomask"       – bool, True when mask is not provided

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)       # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Rearrange to (B, H, N, N) and flatten batch×head → (B·H, N, N)
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM C = left @ rightᵀ  (still fp16)
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)      # (B·H, N, N)

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, H) and cast to fp32
    # --------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )

    # --------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm (hidden_dim) + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # out_gate shape (B,N,N,H)

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
