"""
TriMul “outgoing” implementation (forward only).

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply of shape (B·H, N, N) with the transpose of the
second operand.  We fuse this GEMM into a single Triton kernel that
* loads the two operands as fp16 (half‑precision) to halve memory traffic,
* accumulates in fp32 for numeric stability,
* stores the result in fp32.

All other ops (LayerNorm, Linear, gating) stay in PyTorch – they are cheap
relative to the N³ work and keep the code simple.  The kernel works for any
batch size B∈{1,2}, sequence length N up to 1024 and hidden dimension H
(=hidden_dim) up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ   (batched GEMM)
# left/right : (M, N, K)  with K == N, dtype fp16
# out        : (M, N, N)  dtype fp32
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16   [M, N, K]
    right_ptr,         # *fp16   [M, N, K]
    out_ptr,           # *fp32   [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides (elements)
    stride_rm, stride_rn, stride_rk,   # right strides (elements)
    stride_om, stride_on, stride_ok,   # out strides (elements)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM   out[z] = left[z] @ right[z].T
    where each matrix has shape (N, K) with K == N.
    """
    pid_m = tl.program_id(0)   # tile over rows   (i)
    pid_n = tl.program_id(1)   # tile over cols   (j)
    pid_z = tl.program_id(2)   # independent matrix (B * H)

    # ------------------------------------------------------------------- #
    # Offsets for the current tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers for the current tile
    # ------------------------------------------------------------------- #
    # left : (M, N, K)  -> index (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # right is read transposed on the last two dimensions:
    # we need right[z, k, j]
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    # ------------------------------------------------------------------- #
    # Accumulator (fp32)
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main K‑loop
    # ------------------------------------------------------------------- #
    num_iters = tl.cdiv(K, BLOCK_K)
    for _ in range(num_iters):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0,
                    eviction_policy='evict_last')
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    eviction_policy='evict_last')
        # a,b are fp16, dot returns fp32
        acc += tl.dot(a, b)

        # advance K‑dimension
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Write result (fp32)
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” block.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool/float, or None)
        - weights      : dict of torch.Tensor (all float32)
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)  -> [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid) – all performed in FP32
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Cast to fp16 for the GEMM (half‑precision reduces traffic)
    # ------------------------------------------------------------------- #
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, H, N, N) and collapse batch & head → M
    left_fp16  = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_fp16 = right_fp16.permute(0, 3, 1, 2).contiguous()

    M = B * hidden_dim
    left_fp16  = left_fp16.view(M, N, N)   # (M, N, N)
    right_fp16 = right_fp16.view(M, N, N)

    # Output buffer (fp32)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 6) Launch Triton GEMM
    # ------------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_fp16.stride()
    stride_rm, stride_rn, stride_rk = right_fp16.stride()
    stride_om, stride_on, stride_ok = out_fp32.stride()

    # Tile sizes – empirically good for Hopper (SM90) with fp16 loads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        M                          # different matrices
    )

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps works well on H200 for fp16
        num_stages=3               # double‑buffering
    )

    # ------------------------------------------------------------------- #
    # 7) Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------- #
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 8) Output LayerNorm + gating + final Linear (dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
