"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core contraction.

Algorithm
---------
1. Layer‑norm the input tensor (B×N×N×C).
2. Linear projections → left / right tensors (B×N×N×H).
3. Optional element‑wise mask (broadcast on the hidden dimension).
4. Gated linear units (sigmoid gates) applied to left, right and the final output.
5. Core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   For each head h this is a matrix multiplication
       out_slice = left_slice @ right_sliceᵀ
   where left_slice/right_slice are (N×N).  A single Triton kernel performs a
   batched GEMM for all (batch × head) pairs, using fp16 loads and fp32
   accumulation.
6. Post‑processing: Layer‑norm → out‑gate → final linear projection back to C.

The implementation keeps the expensive reduction (step 5) on the GPU while
leveraging PyTorch for the cheap element‑wise parts.  It works for
B∈{1,2}, N up to 1024 and hidden_dim = 128 as required by the benchmark.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
#   A, B : (batch, M, K)   (here M = N, K = N)
#   C    : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # *float16  (batch, M, K)
    B_ptr,                     # *float16  (batch, N, K)  (read as K×N)
    C_ptr,                     # *float32  (batch, M, N)
    M, N, K,                   # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (B is read transposed)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                # batch index (B*H)
    pid_m = tl.program_id(0)                # block row
    pid_n = tl.program_id(1)                # block col

    # ---- block offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- masks for out‑of‑bounds ----
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- pointer arithmetic ----
    a_ptrs = (A_ptr + pid_b * stride_Ab
              + offs_m[:, None] * stride_Am
              + offs_k[None, :] * stride_Ak)

    # B is accessed as (K,N) → stride_Bk points along K, stride_Bn along N
    b_ptrs = (B_ptr + pid_b * stride_Bb
              + offs_k[:, None] * stride_Bk
              + offs_n[None, :] * stride_Bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- main k‑loop ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)            # (M,BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)            # (BLOCK_K,N)

        acc += tl.dot(a, b)                         # (M,N)

        # advance pointers to next k‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---- store result ----
    c_ptrs = (C_ptr + pid_b * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool or float)
        - weights      : dict of parameter tensors (all float32)
        - config       : dict containing {"dim", "hidden_dim", "nomask"}.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # --------------------------------------------------------------
    # unpack & config
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # view weight tensors
    # --------------------------------------------------------------
    norm_w, norm_b               = weights["norm.weight"], weights["norm.bias"]
    left_proj_w                  = weights["left_proj.weight"]
    right_proj_w                 = weights["right_proj.weight"]
    left_gate_w                  = weights["left_gate.weight"]
    right_gate_w                 = weights["right_gate.weight"]
    out_gate_w                   = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w                     = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)  # [B,N,N,H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via Triton GEMM
    #    out = left @ rightᵀ   (per head)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                      # H == hidden_dim

    # reshape to (B*H, N, N) and make contiguous for simple stride handling
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (bandwidth) – keep a copy to avoid destroying grads
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # output buffer (fp32 accumulator)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # ------------------------------------------------------------------
    # stride extraction (elements, not bytes)
    # ------------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (B*H, N, N) → (N*N, N, 1)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()  # same layout
    # For the transposed read we need:
    stride_Bk = stride_Bc          # step along K (original column)
    stride_Bn = stride_Br          # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # tuning parameters – good defaults for Hopper (SM90)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * heads
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to dim C

    return out
