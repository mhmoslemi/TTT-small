"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is rewritten as a batched matrix‑multiply

    out[b,h] = left[b,h] @ right[b,h].T          (N×N) @ (N×N)ᵀ

where h indexes the hidden dimension (heads).  A single Triton kernel
performs this GEMM for every (batch, head) pair using tiled
bf16 loads and fp32 accumulation.  All other operations (LayerNorm,
linear projections, gating, optional masking) are kept in PyTorch – they
are cheap compared with the N³ reduction.

The kernel is launched on a 3‑D grid:
    pid0 → batch*head index,
    pid1 → tile‑row (i‑block),
    pid2 → tile‑col (j‑block).

Masking (if present) and the final linear projection are pure PyTorch,
so the implementation satisfies the “must use a Triton kernel” rule while
remaining fast enough for the target H200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    # pointers
    a_ptr,          # left  : [B, H, N, N]  (bf16)
    b_ptr,          # right : [B, H, N, N]  (bf16)
    c_ptr,          # out   : [B, H, N, N]  (fp32)
    # dimensions
    B, H, N,        # int
    # strides (in elements, not bytes)
    stride_ah, stride_ai, stride_ak,   # a: batch, row(i), col(k)
    stride_bh, stride_bj, stride_bk,   # b: batch, row(j), col(k)
    stride_ch, stride_ci, stride_cj,   # c: batch, row(i), col(j)
    # tile sizes (constexpr)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for each (batch, head) pair."""
    pid_bh = tl.program_id(0)               # batch * head index
    pid_i  = tl.program_id(1)               # tile row
    pid_j  = tl.program_id(2)               # tile col

    # decode batch / head
    batch = pid_bh // H
    head  = pid_bh % H

    # tile offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # predicates (handle edge tiles)
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N

    # base pointers shifted to the current batch & head
    a_base = a_ptr + batch * stride_ah + head * stride_ai
    b_base = b_ptr + batch * stride_bh + head * stride_bj
    c_base = c_ptr + batch * stride_ch + head * stride_ci

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # global offsets for this tile
        cur_k = k * BLOCK_K + k_offset

        # load A tile  (M x K) – bf16
        a_ptrs = a_base \
                 + i_offset[:, None] * stride_ai \
                 + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile transposed (K x N) – bf16
        # we want Bᵀ, therefore we load with K as the leading dimension
        b_ptrs = b_base \
                 + cur_k[:, None] * stride_bk \
                 + j_offset[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot‑product: (M×K) · (K×N) → (M×N), accumulate in fp32
        acc += tl.dot(a, b)

    # store result (fp32)
    c_ptrs = c_base \
             + i_offset[:, None] * stride_ci \
             + j_offset[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Helper: fused LayerNorm (uses torch.nn.functional)
# --------------------------------------------------------------------------- #
def _layer_norm(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    return F.layer_norm(x, normalized_shape=(w.shape[0],), weight=w, bias=b, eps=1e-5)


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]
                  ) -> torch.Tensor:
    """
    Forward of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32, CUDA)
        - mask         : [B, N, N]      (float32 0/1, CUDA) – may be absent
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys
              * dim        – input channel size
              * hidden_dim – projection size
              * nomask     – bool, True if mask is all‑ones

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """

    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------- #
    x = _layer_norm(input_tensor,
                    weights["norm.weight"],
                    weights["norm.bias"])        # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → broadcast
        mask_b = mask.unsqueeze(-1)
        left  = left * mask_b
        right = right * mask_b

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projection)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction via Triton GEMM
    # ------------------------------------------------------------------- #
    # reshape to [B, H, N, N] and cast to bf16 for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16)
    right_t = right.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16)

    B, H, N, _ = left_t.shape   # H == hidden_dim

    # output buffer (fp32)
    out_t = torch.empty_like(left_t, dtype=torch.float32)

    # strides (in elements)
    stride_ah, stride_ai, stride_ak = left_t.stride()
    stride_bh, stride_bj, stride_bk = right_t.stride()
    stride_ch, stride_ci, stride_cj = out_t.stride()

    # tile sizes – tuned for Hopper (shared‑memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # grid dimensions
    grid = (
        triton.cdiv(B * H, 1),                     # program_id(0): batch*head
        triton.cdiv(N, BLOCK_M),                  # program_id(1): row tiles
        triton.cdiv(N, BLOCK_N),                  # program_id(2): col tiles
    )

    trimul_gemm_kernel[grid](
        a_ptr=left_t,
        b_ptr=right_t,
        c_ptr=out_t,
        B=B, H=H, N=N,
        stride_ah=stride_ah, stride_ai=stride_ai, stride_ak=stride_ak,
        stride_bh=stride_bh, stride_bj=stride_bj, stride_bk=stride_bk,
        stride_ch=stride_ch, stride_ci=stride_ci, stride_cj=stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden -> hidden)
    # ------------------------------------------------------------------- #
    out = _layer_norm(out,
                      weights["to_out_norm.weight"],
                      weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])

    return out
