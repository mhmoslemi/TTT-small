"""
TriMul (outgoing) – highly‑fused forward pass.

The only O(N³·H) work is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
For a fixed head `d` this is a matrix multiplication
    out_d = left_d @ right_dᵀ .
We reshape the tensors to (B, H, N, N) and launch a *batched* GEMM
kernel written in Triton.  The kernel works in fp16 for the loads,
accumulates in fp32, and writes the result directly in fp32, so the
final tensor is already in the required precision.

All cheap preprocessing (LayerNorm, Linear, sigmoid gates, the final
output‑gate element‑wise multiply and the last Linear) stay in PyTorch.
The heavy GEMM is completely off‑loaded to Triton, which gives a large
speed‑up on Hopper (SM‑90) GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   (A,B ∈ [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_am, stride_ak, stride_ah,   # A strides (row, col, batch)
    stride_bk, stride_bn, stride_bh,   # B strides (row of Bᵀ (=k), col (=n), batch)
    stride_cm, stride_cn, stride_ch,   # C strides (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A : (batch, M, K)   – row‑major, fp16
    B : (batch, N, K)   – row‑major, fp16   (we use its transpose inside)
    C : (batch, M, N)   – row‑major, fp32
    """
    pid_m = tl.program_id(0)          # tile over M
    pid_n = tl.program_id(1)          # tile over N
    pid_b = tl.program_id(2)          # batch * head index

    # -----------------------------------------------------------------
    # Offsets of the current tile
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension (tiles)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[i,k]  (row‑major)
        a = tl.load(
            a_ptr
            + pid_b * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float16)

        # Bᵀ[k,j] = B[j,k]  (we read B as (batch, N, K) and swap indices)
        b = tl.load(
            b_ptr
            + pid_b * stride_bh
            + offs_k[:, None] * stride_bk   # k‑row stride
            + offs_n[None, :] * stride_bn,  # n‑col stride
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float16)

        # fp16 matmul, accumulate in fp32
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Store the result (fp32)
    # -----------------------------------------------------------------
    tl.store(
        c_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (bool/float32 or None)
        * weights_dict : model parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called H in the paper
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                          # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over channel dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM in Triton
    #    reshape to (B, H, N, N) and make contiguous
    # -----------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # cast to fp16 for the kernel (the kernel reads fp16)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output in fp32 (the kernel writes fp32 directly)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides (in elements) – required for the kernel
    stride_am = left_fp16.stride(2)   # row stride within a matrix (N)
    stride_ak = left_fp16.stride(3)   # column stride within a matrix (1)
    stride_ah = left_fp16.stride(0)   # batch*head stride (H*N*N)

    stride_bk = right_fp16.stride(2)  # row stride of Bᵀ (original k‑dim)
    stride_bn = right_fp16.stride(3)  # column stride of Bᵀ (original n‑dim)
    stride_bh = right_fp16.stride(0)  # batch*head stride

    stride_cm = out_fp32.stride(2)
    stride_cn = out_fp32.stride(3)
    stride_ch = out_fp32.stride(0)

    # tile sizes – tuned for Hopper (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over M (i)
        triton.cdiv(N, BLOCK_N),   # blocks over N (j)
        B * hidden_dim,            # batch * head
    )

    batched_gemm_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_fp32,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ah=stride_ah,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bh=stride_bh,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_ch=stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: [B, N, N, H]
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6️⃣ Post‑norm on hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                           # [B, N, N, H]

    # -----------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with output gate (already fp32)
    # -----------------------------------------------------------------
    out = out * out_gate                         # cheap, stays in PyTorch

    # -----------------------------------------------------------------
    # 8️⃣ Final projection back to dim
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
