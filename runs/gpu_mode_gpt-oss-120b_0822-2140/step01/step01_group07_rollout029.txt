"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm on the input  (..., dim)
2. Two linear projections → left/right  (dim → hidden_dim, no bias)
3. Optional mask (broadcasted on the hidden dim)
4. Gating sigmoids (left_gate, right_gate, out_gate)
5. Fuse the cubic contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   into a batched GEMM:
       * reshape left/right to (M, N, N) where M = B * hidden_dim
       * load as fp16, accumulate in fp32 (tl.dot)
   Tile size = 64×64×64, 8 warps → optimal for SM‑90.
6. Output Layer‑norm + out_gate + final linear (hidden_dim → dim)

All tensors are kept on the device; only the expensive N³ part is
implemented in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  out = left @ rightᵀ
#   left  : (M, N, K)  where K == N
#   right : (M, N, K)  (will be read transposed, i.e. (M, K, N) inside)
#   out   : (M, N, N)
#   loads are fp16, accumulation is fp32, result stored fp32.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16  [M, N, K]
    right_ptr,         # *fp16  [M, N, K]
    out_ptr,           # *fp32  [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # which (batch * head) matrix
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)

    # ------------------------------------------------------------------- #
    # Offsets for this tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers (M‑dimension is flattened, so we just add pid_z * stride_m)
    # ------------------------------------------------------------------- #
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk        # (M,N,K)

    # right is accessed as transposed on the last two dims:
    # we need right[z, k, n]  →  (k, n) order instead of (n, k)
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn        # (M,K,N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main K‑loop
    # ------------------------------------------------------------------- #
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                     # fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                     # fp16

        # tl.dot works in fp32 when inputs are fp16
        acc += tl.dot(a, b)                    # (BM, BK) @ (BK, BN) → (BM, BN)

        # advance K‑dimension
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Store result
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of model parameters (all torch.float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype torch.float32
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) → (B,N,N,hidden)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Fuse the cubic contraction with Triton
    # ------------------------------------------------------------------- #
    #   reshape to (B, hidden, N, N) → flatten first two dims → (M,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden

    left_reshape = left.view(M, N, N).to(torch.float16)   # fp16 for bandwidth
    right_reshape = right.view(M, N, N).to(torch.float16)

    out_reshape = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # strides in *elements* (not bytes)
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # ------------------------------------------------------------------- #
    # kernel launch configuration
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows (i)
        triton.cdiv(N, BLOCK_N),      # cols (j)
        M,                            # independent matrices (batch*head)
    )

    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                  # optimal for fp16×fp16→fp32 on SM‑90
    )

    # reshape back to (B,N,N,hidden)
    out = out_reshape.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm, gating, final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
