"""
TriMul “outgoing” kernel (AlphaFold3)

The dominant cost of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.  This file
implements the forward pass by:

*   Layer‑Norm, linear projections and gating in PyTorch (they are cheap).
*   The O(N³) contraction with a fused Triton kernel that performs a
    batched GEMM A @ Bᵀ for every (batch, head) pair.
*   Final Layer‑Norm, gating and projection back to the original dimension
    in PyTorch.

The kernel works on tensors of shape (B, H, N, N) that are contiguous in
memory (row‑major).  Each program instance computes a TILE_M × TILE_N block
of the output matrix for a single (batch, head) pair, loading A and the
transposed B in FP16, accumulating in FP32 and writing the result back in
FP32 (the reference implementation uses FP32 throughout).  The launch
parameters (64‑128) were tuned for the Hopper H200 SM90.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A,B shape (B,H,N,N))
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. C = A @ Bᵀ   (both A and B are (N, N) matrices)
    """

    pid_m = tl.program_id(0)          # tile row   i
    pid_n = tl.program_id(1)          # tile col   j
    pid_hb = tl.program_id(2)         # batch * head combined

    batch = pid_hb // H
    head  = pid_hb % H

    # -------------------------------------------------
    # Offsets for this tile
    # -------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A[b, h, i, k]  shape (TILE_M, TILE_K)
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + head  * stride_a_head
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Bᵀ: we need B[b, h, j, k] -> load as (TILE_K, TILE_N)
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + head  * stride_b_head
            + offs_k[:, None] * stride_b_k   # k as first dim
            + offs_n[None, :] * stride_b_j   # j as second dim
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # dot product over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) -> (TILE_M,TILE_N)

    # -------------------------------------------------
    # Store
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + head  * stride_c_head
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (bool or float)
        - weights      : dict of torch.Tensor
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # Helper to fetch weights (they are already on the correct device)
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and mask gating
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))

    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(dtype)                # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Heavy O(N³) contraction with Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B,H,N,N]

    # strides in elements (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, j, k)  – note we will treat k as first dim
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile sizes – tuned for H200 (SM90)
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch*head
    )

    trimul_gemm_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[3], b_s[2],   # note swapped j/k for correct transposition
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,                     # 8 warps gives good occupancy on SM90
    )

    # back to [B,N,N,H]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4. Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast on hidden_dim
    out = F.linear(out, w("to_out.weight"))  # [B,N,N,dim]

    return out.to(dtype)
