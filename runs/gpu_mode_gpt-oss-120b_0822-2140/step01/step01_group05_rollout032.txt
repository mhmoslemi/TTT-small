"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
matrix‑multiply kernel written in Triton.

The expensive part of the module is the einsum

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden dimension *d* this is a matrix multiplication
(N × N) @ (N × N)ᵀ.  By reshaping (B, N, N, H) → (B·H, N, N) we obtain a
single massive batched GEMM.  The kernel below computes this GEMM in
half‑precision (Tensor‑Core friendly) while accumulating in fp32, which
gives a large speed‑up on Hopper (SM‑90).  All other operations
(LayerNorm, linear projections, gating and the final projection) stay
in PyTorch for simplicity and numerical stability.

The kernel works for any batch size ≤2, any sequence length that fits
the block‑size tiling, and both `dim = 128/384` together with
`hidden_dim = 128`.  The mask is optional – when it is present it is
applied before the GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
#   - A and B are read as fp16 (Tensor‑Core friendly)
#   - accumulation is fp32 (exact)
#   - each program instance works on a (BLOCK_M × BLOCK_N) tile
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes (M=N=seq_len, K=seq_len)
    stride_Ab, stride_Am, stride_Ak,    # strides for A  (batch, row, col)
    stride_Bb, stride_Bm, stride_Bk,    # strides for B  (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,    # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch * hidden index
    pid_m = tl.program_id(1)   # tile in M dimension
    pid_n = tl.program_id(0)   # tile in N dimension

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the first element of the tile
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_n[None, :] * stride_Bm + offs_k[:, None] * stride_Bk

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_idx in range(num_k_tiles):
        # Load a tile of A and B as fp16, mask out‑of‑bounds entries
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Tensor‑core dot (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Store the tile back to C (still fp32)
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor[B, N, N] (bool/float32) or None
        - weights      : dict of torch.Parameter tensors (all float32, CUDA)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B, N, N, hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and gate application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Fuse the heavy einsum with a Triton kernel
    #    Reshape to (B·hidden, N, N) → treat each hidden channel as a batch.
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → contiguous → view as (B*H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left  = left.view(B * hidden, N, N)
    right = right.view(B * hidden, N, N)

    # Allocate output buffer (fp32)
    out_flat = torch.empty_like(left, dtype=torch.float32)

    # ------------------------------------------------------------------
    # Kernel launch configuration (tuned for Hopper SM‑90)
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per tile
    BLOCK_N = 64   # cols per tile
    BLOCK_K = 32   # reduction tile size (Tensor‑Core friendly)

    grid = (
        triton.cdiv(N, BLOCK_N),   # tiles in N dimension
        triton.cdiv(N, BLOCK_M),   # tiles in M dimension
        B * hidden,                # batch‑*‑hidden dimension
    )

    # Strides of the three‑dimensional tensors (batch, row, col)
    stride_Ab, stride_Am, stride_Ak = left.stride()
    stride_Bb, stride_Bm, stride_Bk = right.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # Launch the kernel (fp16 loads, fp32 accumulation)
    batched_matmul_fp16_kernel[grid](
        left,
        right,
        out_flat,
        N,                 # M
        N,                 # N
        N,                 # K
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bm, stride_Bk,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (hidden) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
