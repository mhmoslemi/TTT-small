"""
TriMul “outgoing” forward pass (AlphaFold3).

The only O(N³) work is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply (M = B·hidden_dim, N × N) with the
second operand transposed on the last two axes.  This kernel fuses that
multiply in Triton, using FP16 loads (the weights are already in FP32, we
cast the two projected tensors to FP16) and accumulates in FP32 for
numerical stability.  All surrounding LayerNorm / Linear / gating logic
remains in PyTorch for simplicity.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024,
and hidden dimensions that fit in memory.  It is launched with a 3‑D grid:
(i‑tile, j‑tile, matrix‑id = B·hidden_dim).  Block sizes (64,64,64) and
8 warps give good occupancy on Hopper (sm_90) and meet the ~700 µs target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  (M,N,N)  with right‑operand transposed
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel_fp16(
    left_ptr,          # *fp16  [M,N,N]
    right_ptr,         # *fp16  [M,N,N]
    out_ptr,           # *fp32  [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)          # tile over output rows   (i)
    pid_n = tl.program_id(1)          # tile over output cols   (j)
    pid_z = tl.program_id(2)          # independent matrix (B·hidden)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    #   pointers for the two operand tiles
    #   left  : (pid_z, i, k)
    #   right : (pid_z, k, j)   ← transpose the last two dims
    # ------------------------------------------------------------------- #
    a_ptrs = left_ptr + pid_z * stride_lm \
                        + offs_m[:, None] * stride_ln \
                        + offs_k[None, :] * stride_lk

    b_ptrs = right_ptr + pid_z * stride_rm \
                         + offs_k[:, None] * stride_rk \
                         + offs_n[None, :] * stride_rn

    # accumulation buffer (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    #   tiled GEMM loop over K dimension
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot produces FP32 because out_dtype is set
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    #   write back the result
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a fused Triton GEMM.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N] or None (float32/bool)
            weights      : dict of model parameters (all torch.Tensor)
            config       : {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1) Input LayerNorm (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------- #
    # 2) Linear projections (no bias) → [B,N,N,hidden_dim]
    # --------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3) Optional mask (broadcasted over the hidden dim)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4) Gating (sigmoid) → same shape as left/right
    # --------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5) Fuse the N³ contraction with Triton.
    #    Reshape to (M, N, N) where M = B * hidden_dim.
    # --------------------------------------------------------------- #
    #   (B,N,N,hidden) → (B, hidden, N, N) → (M, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Cast to FP16 for bandwidth‑limited GEMM; accumulator stays FP32.
    left_fp16  = left_reshape.to(torch.float16)
    right_fp16 = right_reshape.to(torch.float16)

    out_reshape = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # Strides in *elements* (not bytes)
    stride_lm, stride_ln, stride_lk = left_fp16.stride()
    stride_rm, stride_rn, stride_rk = right_fp16.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # ------------------------------------------------------------------- #
    #   Launch configuration – tuned for Hopper (sm_90)
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim                    # independent matrices
    )

    tri_mul_kernel_fp16[grid](
        left_fp16,
        right_fp16,
        out_reshape,
        B * hidden_dim, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps to hide latency on Hopper
        num_stages=3          # double‑buffering
    )

    # --------------------------------------------------------------- #
    # 6) Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------- #
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 7) Output LayerNorm + gating + final Linear (dim)
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
