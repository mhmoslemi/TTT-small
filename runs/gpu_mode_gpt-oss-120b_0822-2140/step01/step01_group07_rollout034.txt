"""
TriMul (outgoing) – fast forward pass
-------------------------------------

*   LayerNorm + Linear + gating stay in PyTorch (highly‑optimized ATen).
*   The N³ contraction

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    is rewritten as a batched GEMM  C = A @ Bᵀ  with

        A = left.permute(0,3,1,2).view(M,N,N)   (M = B*hidden_dim)
        B = right.permute(0,3,1,2).view(M,N,N)

    The GEMM is performed by a single Triton kernel.
    *   Inputs are cast to **float16** (half‑precision) to halve memory traffic.
    *   Accumulation is done in **float32** for numerical stability.
    *   Tiling:  BLOCK_M = BLOCK_N = 64, BLOCK_K = 64, GROUP_SIZE_M = 8.
    *   8 warps (num_warps=8) gives full utilisation on Hopper (SM‑90).

The kernel returns a fp32 tensor `out_reshape` which is reshaped back to
`[B, N, N, hidden_dim]`.  The remaining LayerNorm‑gate‑Linear pipeline
produces the final `[B, N, N, dim]` output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton GEMM kernel (C = A @ Bᵀ)                                            #
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # logical dimensions
    stride_am, stride_an, stride_ak,   # A strides (M,N,K)
    stride_bm, stride_bn, stride_bk,   # B strides (M,N,K)
    stride_cm, stride_cn, stride_ck,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
    NUM_STAGES: tl.constexpr,
):
    """Batched GEMM:   C[m, i, j] = Σₖ A[m, i, k] * B[m, j, k]"""

    pid = tl.program_id(0)                     # linear id over (M-tile, N-tile)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # --------------------------------------------------------------
    # 2‑D tiling of the output matrix (i, j) + a third dim for batch‑head
    # --------------------------------------------------------------
    pid_m = pid // num_pid_n                    # tile row index
    pid_n = pid %  num_pid_n                    # tile col index
    pid_z = tl.program_id(1)                    # batch*hidden index

    # --------------------------------------------------------------
    # Offsets inside the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers for the current tile (no need for extra stage buffers)
    # --------------------------------------------------------------
    a_ptrs = A_ptr + pid_z * stride_am \
                 + offs_m[:, None] * stride_an \
                 + offs_k[None, :] * stride_ak

    # B is accessed transposed on its last two dims: we need B[m, k, j]
    b_ptrs = B_ptr + pid_z * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop (blocked)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (fp16) – out‑of‑bounds values are zeroed
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 dot, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance K dimension
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store the result (fp32)
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_z * stride_cm \
                 + offs_m[:, None] * stride_cn \
                 + offs_n[None, :] * stride_ck

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a fused Triton GEMM.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        - mask         : torch.Tensor  [B, N, N]      (bool/float) or None
        - weights      : dict of torch.Tensor with the model parameters
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) Norm + Linear projections (all in FP32, PyTorch)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 2) optional mask (applied before gating)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)       # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 3) gating (sigmoid) – keep in FP32
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 4) Prepare tensors for the GEMM
    #    Cast to FP16 to lower memory traffic.
    # ------------------------------------------------------------------- #
    # (B,N,N,hidden) -> (B,hidden,N,N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # fuse batch & head dimension
    M = B * hidden
    left  = left.view(M, N, N).to(torch.float16)
    right = right.view(M, N, N).to(torch.float16)

    # output buffer (FP32 because we accumulate in FP32)
    out_fp32 = torch.empty((M, N, N), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------- #
    # 5) launch Triton kernel
    # ------------------------------------------------------------------- #
    stride_am, stride_an, stride_ak = left.stride()
    stride_bm, stride_bn, stride_bk = right.stride()
    stride_cm, stride_cn, stride_ck = out_fp32.stride()

    # tuning parameters – chosen for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_SIZE_M = 8          # groups rows to improve L2 cache reuse
    NUM_STAGES = 3            # prefetch stages (default in many tutorials)

    # one program per output tile (M‑tile × N‑tile) * batch*head
    grid_rows = triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (grid_rows, M)

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out_fp32,
        M, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        NUM_STAGES=NUM_STAGES,
        num_warps=8,
    )

    # ------------------------------------------------------------------- #
    # 6) reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------- #
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 7) final LayerNorm + gating + Linear (still in FP32)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to [B,N,N,dim]

    return out
