"""
TriMul “outgoing” – fast forward pass.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply where each (b,d) pair multiplies an
(N × N) matrix with the transpose of another (N × N) matrix.
We reshape the tensors to (B·H , N , N) (H = hidden_dim) and fuse the
GEMM into a single Triton kernel.  The kernel tiles the output with
BLOCK_M × BLOCK_N work‑items and walks the K‑dimension in blocks of
BLOCK_K.  All arithmetic is performed in float32 (the inputs are
cast to fp16 on the host to cut memory traffic, then cast back
inside the kernel).  The surrounding LayerNorm / Linear / gating logic
remains in PyTorch – this gives a good trade‑off between implementation
complexity and performance on Hopper (sm_90).

The kernel works for any batch size (1‑2), any sequence length up to 1024,
and any pair (dim, hidden_dim) used in the test matrix (e.g. 128/384 ↔ 128).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ   where left/right have shape
# (M, N, K)   with K == N.  M = B * hidden_dim.
# The kernel expects fp16 inputs but accumulates in fp32 and writes fp32.
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,          # *fp16   [M,N,K]
    right_ptr,         # *fp16   [M,N,K]
    out_ptr,           # *fp32   [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides (elements)
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # Program identifiers
    # ---------------------------------------------------------------
    pid_m = tl.program_id(0)          # row tile   (i)
    pid_n = tl.program_id(1)          # col tile   (j)
    pid_z = tl.program_id(2)          # independent matrix (batch*head)

    # ---------------------------------------------------------------
    # Offsets within the tile
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # Pointers to the start of the current tile
    # ---------------------------------------------------------------
    # left is accessed as (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # right is accessed as (z, j, k) but we need its transpose (k, j)
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Main K‑loop
    # ---------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # fp16 → fp32
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # fp16 → fp32

        # (BLOCK_M x BLOCK_K) @ (BLOCK_K x BLOCK_N)
        acc += tl.dot(a, b)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ---------------------------------------------------------------
    # Write out the result
    # ---------------------------------------------------------------
    c_ptrs = out_ptr + pid_z * stride_om \
                     + offs_m[:, None] * stride_on \
                     + offs_n[None, :] * stride_ok

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    The computationally heavy N³ contraction is executed by a fused
    Triton GEMM kernel; all other operations (LayerNorm, Linear, gating)
    stay in PyTorch.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of learned parameters (all torch.Tensor)
        - config       : dict with keys “dim” and “hidden_dim”

    Returns
    -------
    out : torch.Tensor, shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # often 128

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Optional mask – broadcast over the hidden dimension
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid) – also applied after the GEMM
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Fuse the N³ contraction with Triton.
    #    Reshape to (B*H, N, N) where H = hidden_dim.
    # ------------------------------------------------------------------- #
    #   (B,N,N,H) -> (B,H,N,N) -> (M,N,N)   with M = B*H
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 to halve memory traffic (the kernel converts back to fp32)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    M = B * hidden_dim
    left_fp16  = left_fp16.view(M, N, N)
    right_fp16 = right_fp16.view(M, N, N)

    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # ---------------------------------------------------------------
    # Strides (in elements) – required by the kernel
    # ---------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_fp16.stride()
    stride_rm, stride_rn, stride_rk = right_fp16.stride()
    stride_om, stride_on, stride_ok = out_fp32.stride()

    # ---------------------------------------------------------------
    # Kernel launch configuration (tuned for Hopper)
    # ---------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32                         # fits nicely in shared memory

    grid = (
        triton.cdiv(N, BLOCK_M),         # rows
        triton.cdiv(N, BLOCK_N),         # cols
        M                                 # independent matrices
    )

    _tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # optimum for fp32 on sm_90
    )

    # ---------------------------------------------------------------
    # Reshape back to the original layout: [B, N, N, hidden_dim]
    # ---------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6) Output LayerNorm + gating + final linear (dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
