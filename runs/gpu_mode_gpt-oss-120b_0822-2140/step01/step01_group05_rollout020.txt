"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
GEMM kernel written in Triton.

The expensive part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
For each hidden dimension `d` this is a matrix multiplication
    (N×N) @ (N×N)ᵀ .
We treat every (batch, hidden) pair as an independent batch and compute
all of them together in a single Triton kernel.  The kernel works
in fp16 (fast on Hopper) while accumulating in fp32 to keep numerical
stability.  All the surrounding LayerNorms, linear projections and
gating are kept in PyTorch – this gives a huge speed‑up over a pure‑PyTorch
implementation while still satisfying the correctness requirements.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel :  (B*H, N, N)  @  (B*H, N, N).T   ->  (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                 # pointers
    M, N, K,                             # matrix sizes (all = seq_len)
    stride_Ab, stride_Am, stride_Ak,     # strides for A  (batch, M, K)
    stride_Bb, stride_Bm, stride_Bk,     # strides for B  (batch, M, K)
    stride_Cb, stride_Cm, stride_Cn,     # strides for C  (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (all dims are square)."""
    pid_b = tl.program_id(0)          # batch*hidden index
    pid_m = tl.program_id(1)          # M‑tile index
    pid_n = tl.program_id(2)          # N‑tile index

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_n[None, :] * stride_Bm
        + offs_k[:, None] * stride_Bk
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension (blocked)
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        ).to(tl.float16)          # cast to fp16 for the dot
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # fp16 dot, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (bool/float or None)
        - weights      : dict of model parameters (all CUDA float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension ``dim``
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right  :  [B,N,N,hidden]
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM:   out = left @ rightᵀ   (per hidden channel)
    # --------------------------------------------------------------
    #   left / right : [B, N, N, hidden]  →  [B, hidden, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 for the kernel (H200 has very fast fp16 matmul)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Collapse batch & hidden into a single batch dimension
    left_flat = left_fp16.view(B * hidden, N, N)
    right_flat = right_fp16.view(B * hidden, N, N)

    # Output tensor (float32 to keep final precision)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # --------------------------------------------------------------
    # Triton launch parameters
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # pid_b
        triton.cdiv(N, BLOCK_M),        # pid_m
        triton.cdiv(N, BLOCK_N),        # pid_n
    )

    # strides (batch, row, col)
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bm, stride_Bk = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bm,
        stride_Bk,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to the original layout  [B, N, N, hidden]
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (hidden) + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to ``dim``
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
