"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a custom Triton kernel.

The costly part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiply (N×N) @ (N×N)ᵀ for every hidden channel d.
We fuse all hidden channels of the batch into one huge batch
    (B·hidden, N, N) and compute the product with a single
    Triton kernel that works in FP16 (accumulate in FP32) and uses
    tiling + double‑buffering.  All other operations (LayerNorm, linear
    projections, gating, final LayerNorm/linear) stay in PyTorch.

The kernel is written for Hopper (sm_90) and targets the
sequence lengths used in the benchmark (256‑1024).  BLOCK sizes were
chosen to give high occupancy while keeping shared‑memory usage modest:
    BLOCK_M = BLOCK_N = 64   (tile of the output matrix)
    BLOCK_K = 64             (tile of the reduction dimension)

The implementation supports both masked and un‑masked inputs and works for
any batch size B∈{1,2}, hidden dimension hidden_dim, and model dimension dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B*H, N, N) @ (B*H, N, N)ᵀ  ->  (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (M=N=seq_len, K=seq_len)
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col) – note row/col swapped for transpose
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # program ids
    # ---------------------------------------------------------------
    pid_z = tl.program_id(0)        # batch*hidden dimension
    pid_m = tl.program_id(1)        # row tile
    pid_n = tl.program_id(2)        # col tile

    # ---------------------------------------------------------------
    # tile offsets
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---------------------------------------------------------------
    # boundary masks
    # ---------------------------------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ---------------------------------------------------------------
    # pointers for the current tile (A is row‑major, B is accessed transposed)
    # ---------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_z * stride_az
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_z * stride_bz
        + offs_k[:, None] * stride_bk          # k dimension is first (row of Bᵀ)
        + offs_n[None, :] * stride_bn          # n dimension is second (col of Bᵀ)
    )

    # ---------------------------------------------------------------
    # accumulator in FP32
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # main reduction loop over K
    # ---------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B tiles (FP16); mask out‑of‑bounds elements
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---------------------------------------------------------------
    # write back (convert to FP16)
    # ---------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point exposed to the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor[B, N, N, dim]   (float32, CUDA)
        mask    – torch.Tensor[B, N, N] (bool/float) or None
        weights – dict of torch.Tensors (all float32, CUDA)
        config  – dict with keys:
            "dim"         – input feature dimension
            "hidden_dim"  – hidden dimension used inside the module
            "nomask"      – bool, True if mask is guaranteed to be all‑ones
    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32, CUDA)
    """
    # ---------------------------------------------------------------
    # unpack
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # ---------------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ---------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 5. Batched GEMM in FP16 via Triton
    #    reshape to (B, hidden, N, N) → (B*hidden, N, N)
    # ---------------------------------------------------------------
    # (B, N, N, hidden) -> (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # cast to half for the kernel
    left_h = left.view(B * hidden, N, N).to(torch.float16)
    right_h = right.view(B * hidden, N, N).to(torch.float16)

    out_h = torch.empty_like(left_h)

    # launch parameters – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
    )

    # strides for the 3‑D tensors (batch, row, col)
    a_batch, a_row, a_col = left_h.stride()
    b_batch, b_row, b_col = right_h.stride()
    c_batch, c_row, c_col = out_h.stride()

    batched_gemm_kernel[grid](
        left_h,                         # A_ptr
        right_h,                        # B_ptr
        out_h,                          # C_ptr
        N, N, N,                        # M, N, K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,          # note: we treat B as transposed inside the kernel
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------------
    # 6. Restore original layout (B, N, N, hidden) and FP32
    # ---------------------------------------------------------------
    out = out_h.to(torch.float32).view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------
    # 7. Output LayerNorm + out‑gate
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ---------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ---------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
