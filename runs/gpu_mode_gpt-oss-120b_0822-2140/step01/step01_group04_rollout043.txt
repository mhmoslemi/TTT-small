"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

The dominant O(N³) work is the contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We reform the tensors to shape (B, H, N, N) and compute
    out[b,h] = left[b,h] @ right[b,h].T
with a custom Triton kernel that:
* works on fp16 inputs (to halve bandwidth) and accumulates in fp32
* tiles the output with 128×128 blocks and a 64‑element K‑tile
* uses 8 warps per block (good for Hopper’s 128‑thread blocks)
All cheap ops (LayerNorm, Linear, gating, final projection) stay in PyTorch.

The host function `custom_kernel` follows the signature required by the
testing harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ   (A,B shape = (B,H,N,N))
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem size (seq_len, hidden_dim)
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]   (i.e. A @ Bᵀ)."""
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ----------- tile offsets ----------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ---- load A (M × K)  ----
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + head  * stride_a_head
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # ---- load Bᵀ (K × N) ----   note: B is stored as (i,j) but we need (j,k)
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + head  * stride_b_head
            + offs_k[:, None] * stride_b_k   # k dimension first
            + offs_n[None, :] * stride_b_j   # j dimension second
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # dot‑product over K   (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # ----------- store ----------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + head  * stride_c_head
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors        model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor
        Output tensor  [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    use_mask = not cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # shortcuts for weight tensors
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden]

    if use_mask:
        # mask shape -> [B,N,N,1] for broadcasting
        m = mask.to(dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction via Triton
    #    reshape to (B, hidden, N, N) – contiguous layout required
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in element units (not bytes)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j)   we will read as (k,j)
    c_s = out_t.stride()     # (batch, head, i, j)

    # tiling parameters – tuned for Hopper (128‑thread blocks)
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),        # i‑tiles
        triton.cdiv(N, TILE_N),        # j‑tiles
        B * hidden,                    # batch*head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,          # 8 warps → 128 threads per block
    )

    # back to original layout (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate          # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
