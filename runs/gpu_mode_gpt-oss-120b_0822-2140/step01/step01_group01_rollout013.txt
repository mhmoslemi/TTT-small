"""
TriMul “outgoing” implementation (alpha‑Fold‑3 style)

The expensive operation is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension `d` this is a dense matrix product
    L_d = left[b,:, :, d]          (N×N)
    R_d = right[b,:, :, d]         (N×N)
    out_d = L_d @ R_dᵀ             (N×N)

We therefore treat the batch and the hidden dimension as one large batch:
    B·H independent GEMMs of shape (N,N)·(N,N)ᵀ.
The kernel below computes each GEMM in fp16 (to maximise bandwidth) while
accumulating in fp32 for numerical stability.  All remaining
normalisation / gating / linear layers are done in PyTorch – they are
trivial compared with the O(N³) work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    N, M,                                    # N = seq_len, M = B*H
    stride_ah, stride_ai, stride_ak,         # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,         # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,         # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a BLOCK_M×BLOCK_N tile of one GEMM
    (identified by program_id(2) == pid_z).  The GEMM is A @ Bᵀ, therefore we
    load B transposed (k‑j order) to be able to use a single tl.dot.
    """
    pid_m = tl.program_id(0)      # tile row   (i)
    pid_n = tl.program_id(1)      # tile column(j)
    pid_z = tl.program_id(2)      # flattened (batch * hidden)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for tail handling
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the selected GEMM
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in fp32 (more accurate than fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the reduction axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of A   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load a tile of Bᵀ   shape (BLOCK_K, BLOCK_N)
        # original B is indexed B[j, k]; we load it transposed
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj          # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 dot, accumulate into fp32
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    left/right : (B, H, N, N)  (fp16)
    returns    : (B, H, N, N)  (fp16, fp32 accumulation)
    """
    B, H, N, _ = left.shape
    M = B * H                                 # flattened batch*head

    # reshape to (M, N, N) – contiguous memory layout is required
    left_f  = left.reshape(M, N, N).contiguous()
    right_f = right.reshape(M, N, N).contiguous()

    out = torch.empty_like(left_f, dtype=torch.float16)

    # strides as int64 (required by Triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tile sizes – tuned for Hopper SMs (large shared‑memory, 128 B vectors)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )
    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy O(N³) contraction is executed by the Triton GEMM kernel above.
    All other operations (LayerNorm, Linear, Sigmoid, masking) are performed
    in PyTorch – they are inexpensive compared with the contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H in the paper

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections: dim → hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional masking (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.to(device).unsqueeze(-1)          # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (B, H, N, N) layout
    # ------------------------------------------------------------------
    # Convert to the layout expected by the kernel and cast to fp16
    left_perm  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm = trimul_gemm(left_perm, right_perm)        # (B, H, N, N) fp16

    # Back to the original layout (B, N, N, H) – keep fp16 for the next norm
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b)                # fp16 -> fp32 inside
    out = out * out_gate                                 # broadcast gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim (no bias)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)       # (dim, hidden)
    out = F.linear(out, to_out_w)                        # (B,N,N,dim)

    # Ensure final dtype is float32 (as required by the reference)
    return out.to(torch.float32)
