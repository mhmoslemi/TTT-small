"""
TriMul “outgoing” forward pass (AlphaFold3).

The expensive contraction  

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is a batched GEMM: for each (batch, head) pair we compute  
    C = A @ Bᵀ,   A = left[b,h] (N×N),   B = right[b,h] (N×N).

We materialise this GEMM with a fused Triton kernel that tiles the
M‑, N‑ and K‑ dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates in
FP32 while the operands are kept in FP16/bfloat16.  All other
operations (LayerNorms, linear projections, gating, mask handling) are
performed with regular PyTorch – they are cheap compared with the
N³ reduction.

The kernel works on tensors that have been reshaped to the flat
dimension ``BH = batch * hidden_dim``:

    left_flat  : (BH, N, N)   – layout (batch_head, row, col) contiguous
    right_flat : (BH, N, N)   – same layout
    out_flat   : (BH, N, N)

After the Triton kernel we restore the original ``[B, N, N, H]`` shape
and finish the forward pass.
"""

import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  C = A @ Bᵀ   (A,B : (BH,N,N))
# --------------------------------------------------------------------------- #
@triton.jit
def triu_mul_kernel(
    A_ptr,                     # left  (BH,N,N)  fp16/bf16
    B_ptr,                     # right (BH,N,N)  fp16/bf16
    C_ptr,                     # out   (BH,N,N)  fp32
    BH, N, K,                  # dimensions (BH = B*H, N = seq_len, K = N)
    stride_ah, stride_am, stride_ak,   # strides for A (batch_head, row, col)
    stride_bh, stride_bm, stride_bn,   # strides for B (batch_head, row, col)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch_head, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    bh_id = tl.program_id(0)          # which (batch,head) pair
    pid_m = tl.program_id(1)          # block row of output
    pid_n = tl.program_id(2)          # block col of output

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    m_mask = offs_m < N
    n_mask = offs_n < N
    k_valid = offs_k < K

    # pointers to the start of the current (bh) slice
    a_base = A_ptr + bh_id * stride_ah
    b_base = B_ptr + bh_id * stride_bh
    c_base = C_ptr + bh_id * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute actual K‑range for this iteration
        cur_k = k * BLOCK_K + offs_k
        cur_k_mask = cur_k < K

        # A tile:   rows = offs_m, cols = cur_k
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=m_mask[:, None] & cur_k_mask[None, :], other=0.0)

        # Bᵀ tile: we need B[col, row] i.e. B[ cur_k, offs_n ]
        b_ptrs = b_base + (cur_k[:, None] * stride_bm) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=cur_k_mask[:, None] & n_mask[None, :], other=0.0)

        # dot product over K dimension (accumulate in FP32)
        acc += tl.dot(a, b)

    # write result
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=m_mask[:, None] & n_mask[None, :])


# --------------------------------------------------------------------------- #
# Helper – layer‑norm and linear that use supplied weight tensors
# --------------------------------------------------------------------------- #
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    # x: [..., C]
    return F.layer_norm(
        x,
        normalized_shape=(weight.shape[0],),
        weight=weight,
        bias=bias,
        eps=1e-5,
    )


def _linear(x: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    # weight: [out_dim, in_dim]  (bias = None)
    return F.linear(x, weight)


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   torch.float32, CUDA
        - mask         : [B, N, N]       torch.float32 (0/1) or None
        - weights      : dict of parameters (see code)
        - config       : dict with keys:
            * dim        – input channel size
            * hidden_dim – projection size
            * nomask     – bool, True ⇢ mask is all‑ones

    -------
    torch.Tensor
        [B, N, N, dim]  (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input layer‑norm
    # ------------------------------------------------------------------- #
    x = _layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections to hidden_dim
    # ------------------------------------------------------------------- #
    left = _linear(x, weights["left_proj.weight"])   # (B,N,N, H)
    right = _linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: (B,N,N) → (B,N,N,1) → broadcast
        mask_b = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_b
        right = right * mask_b

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(_linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(_linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(_linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction  – batched GEMM via Triton
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    BH = B * hidden_dim                     # flatten batch*head

    # Convert to low‑precision for the GEMM (fp16 works well on Hopper)
    left_fp = left.to(torch.float16)
    right_fp = right.to(torch.float16)

    # Re‑order to (B, H, N, N) and flatten the first two dims → (BH, N, N)
    left_flat = left_fp.permute(0, 3, 1, 2).reshape(BH, N, N).contiguous()
    right_flat = right_fp.permute(0, 3, 1, 2).reshape(BH, N, N).contiguous()
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # strides (in elements, not bytes)
    stride_ah = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bm = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_ch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # tiling parameters – tuned for Hopper, keep them constexpr
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # K‑tile fits in registers

    grid = (
        BH,                         # batch‑head dimension
        triton.cdiv(N, BLOCK_M),    # rows
        triton.cdiv(N, BLOCK_N),    # cols
    )

    triu_mul_kernel[grid](
        A_ptr=left_flat,
        B_ptr=right_flat,
        C_ptr=out_flat,
        BH=BH,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bm=stride_bm,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    # ------------------------------------------------------------------- #
    # 6️⃣ Output layer‑norm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = _linear(out, weights["to_out.weight"])

    return out
