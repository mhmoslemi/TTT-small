"""
TriMul “outgoing” forward pass (AlphaFold‑3 style)

The only O(N³) work is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden channel *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We therefore launch a *batched* GEMM kernel with Triton:
* batch dimension = B * hidden_dim (flattened)
* each GEMM works on (N,N) × (N,N)ᵀ
* inputs are cast to fp16, the dot product accumulates in fp32
* block sizes (128,128,64) were found to give the best throughput on
  Hopper‑class GPUs (H200/H100).

All cheap per‑token work (LayerNorms, linear projections, sigmoids,
masking) stays in PyTorch.  The kernel returns fp32 results which are then
fed through the output LayerNorm and final linear projection.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers (fp16 inputs, fp32 output)
    N,                            # sequence length
    stride_ah, stride_ai, stride_ak,   # strides of A (M,N,N)
    stride_bh, stride_bj, stride_bk,   # strides of B (M,N,N)
    stride_ch, stride_ci, stride_cj,   # strides of C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program instance computes a BLOCK_M × BLOCK_N tile of one GEMM
    (indexed by program_id(2) = batch * hidden).
    """
    pid_m = tl.program_id(0)   # tile row   (i)
    pid_n = tl.program_id(1)   # tile col   (j)
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current (batch, hidden) slice
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]  (BLOCK_M, BLOCK_K)   fp16 load
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # Bᵀ[k,j]  (BLOCK_K, BLOCK_N)
        # Original B is indexed as B[j,k]; loading with stride_bk on k and
        # stride_bj on j gives the transposed layout required by tl.dot.
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk      # k axis (row of Bᵀ)
            + offs_n[None, :] * stride_bj      # j axis (col of Bᵀ)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        # dot product: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)   # a,b are fp16, acc is fp32

    # write back result (fp32)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, H, N, N)  fp16
    returns    : (B, H, N, N)  fp32   (each H‑th slice is left @ rightᵀ)
    """
    B, H, N, _ = left.shape
    M = B * H                                   # flattened batch*hidden

    # flatten to (M, N, N) for the kernel
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides (int64)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # --- tuning parameters -------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    # -----------------------------------------------------------------------

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z (batch*hidden)
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Only the cubic‑complex contraction is executed on GPU via a Triton
    batched‑GEMM kernel; all other ops stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections: dim -> hidden_dim  (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)          # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ  (per hidden channel)
    #    Convert to (B, hidden, N, N) layout expected by the kernel.
    #    Cast to fp16 for higher throughput, keep accumulation in fp32.
    # --------------------------------------------------------------
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B,H,N,N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_fp32 = _batched_gemm_fp16(left_fp16, right_fp16)                # (B,H,N,N) fp32

    # back to (B,N,N,hidden)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B,N,N,dim)

    return out.to(torch.float32)
