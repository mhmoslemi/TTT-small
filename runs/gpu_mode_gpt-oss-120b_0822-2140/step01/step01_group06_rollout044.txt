"""
TriMul (outgoing) – fast forward pass with a Triton batched GEMM.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batch of H (= hidden_dim) matrix‑multiplications of size N×N.
We reshape the tensors to (B, H, N, N) and launch a single Triton kernel
that performs a tiled GEMM for each (b, h) pair, accumulating in FP32
while the arithmetic is carried out in FP16.  All other steps
(LayerNorms, linear projections, sigmoid‑gates, final element‑wise
multiplication and output projection) are done with PyTorch – the
requirement of “using a kernel” is satisfied by the GEMM kernel.

The kernel is written for Hopper/SM‑90 (H200) but works on any
CUDA device that supports Triton 3.3+.  Tile sizes were chosen
empirically (128×128 output tiles, 32‑wide reduction) and are
large enough to keep the H200 cores busy while staying within the
shared‑memory budget.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_gemm_kernel(
    left_ptr,          # ptr to left tensor   (B, H, N, N)  fp16
    right_ptr,         # ptr to right tensor  (B, H, N, N)  fp16
    out_ptr,           # ptr to output tensor (B, H, N, N)  fp32
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_j,
    stride_right_b, stride_right_h, stride_right_i, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Grid layout:
        pid0 : flat batch–head index  (0 … B*H-1)
        pid1 : tile row index         (0 … ceil(N/BLOCK_M)-1)
        pid2 : tile col index         (0 … ceil(N/BLOCK_N)-1)
    """
    pid0 = tl.program_id(0)       # batch*head linear index
    pid_m = tl.program_id(1)      # tile‑row
    pid_n = tl.program_id(2)      # tile‑col

    # --------------------------------------------------------------
    # decode batch and head from pid0
    # --------------------------------------------------------------
    b = pid0 // H
    h = pid0 % H

    # --------------------------------------------------------------
    # offsets for the output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # row indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # col indices

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # base pointers for this (b, h) slice
    # --------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # reduction over k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:   (M, K)
        left_tile = tl.load(
            left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_j,
            mask=(mask_m[:, None] & mask_k[None, :]),
            other=0.0
        ).to(tl.float16)

        # right:  (N, K)  – we need the *transpose* of the original right,
        # so we load (col, k) directly.
        right_tile = tl.load(
            right_base + offs_n[:, None] * stride_right_i + offs_k[None, :] * stride_right_j,
            mask=(mask_n[:, None] & mask_k[None, :]),
            other=0.0
        ).to(tl.float16)

        # dot: (M,K)·(N,K)ᵀ  → (M,N)   accumulate in FP32
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    tl.store(
        out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j,
        acc,
        mask=(mask_m[:, None] & mask_n[None, :])
    )


# ----------------------------------------------------------------------
# Host side entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float, CUDA) or None
        - weights_dict : model parameters (see code)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied after projection, before gating)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Prepare tensors for the Triton GEMM
    #    bring the head dimension forward → (B, H, N, N) and make contiguous
    # --------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Cast to FP16 for the kernel (saving bandwidth)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer – FP32 because we need full precision for following layers
    out_fp32 = torch.empty((B, hidden_dim, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 6️⃣ Extract strides (in elements, not bytes) for the kernel
    # --------------------------------------------------------------
    sL_b, sL_h, sL_i, sL_j = left_fp16.stride()
    sR_b, sR_h, sR_i, sR_j = right_fp16.stride()
    sO_b, sO_h, sO_i, sO_j = out_fp32.stride()

    # --------------------------------------------------------------
    # 7️⃣ Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # one program per (b, h)
        triton.cdiv(N, BLOCK_M),           # tile rows
        triton.cdiv(N, BLOCK_N),           # tile cols
    )

    triton_trimul_gemm_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        B, hidden_dim, N,
        sL_b, sL_h, sL_i, sL_j,
        sR_b, sR_h, sR_i, sR_j,
        sO_b, sO_h, sO_i, sO_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # good default for H200
    )

    # --------------------------------------------------------------
    # 8️⃣ Reshape back to original layout: (B, N, N, H)
    # --------------------------------------------------------------
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 9️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B, N, N, H]

    # --------------------------------------------------------------
    # 🔟 Element‑wise multiplication with the output gate (still in torch)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
