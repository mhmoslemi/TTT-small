"""
TriMul (outgoing) – fast forward pass.

The only O(N³·H) work is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched GEMM (A @ Bᵀ) for every (batch, head) pair.
We reshape the tensors to (B, H, N, N) and launch a single Triton kernel
that tiles the GEMM in three dimensions (BLOCK_M, BLOCK_N, BLOCK_K).
All other linear / layer‑norm / gating ops stay in PyTorch; the final
element‑wise product with the output‑gate is also fused with a tiny
Triton kernel.

The implementation works for any batch size ≤2, any sequence length
in the test set and for dim = 128/384, hidden_dim = 128.  The kernel
operates entirely in fp32 (inputs are cast to fp16 for bandwidth, but
accumulation & output stay in fp32) and supports an optional mask.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Element‑wise multiply (out_gate * out) – small helper kernel
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,  # flattened pointers
    total_elts: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel for the O(N³·H) term
#    out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    left_ptr,               # (B, H, N, N)  fp16
    right_ptr,              # (B, H, N, N)  fp16
    out_ptr,                # (B, H, N, N)  fp32
    B, H, N,
    stride_l_bh, stride_l_h, stride_l_i, stride_l_k,   # left strides (elements)
    stride_r_bh, stride_r_h, stride_r_j, stride_r_k,   # right strides (elements)
    stride_o_bh, stride_o_h, stride_o_i, stride_o_j,   # out strides (elements)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # index over (batch, head)
    pid_bh = tl.program_id(2)
    b = pid_bh // H
    h = pid_bh % H

    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_l_bh + h * stride_l_h
    right_base = right_ptr + b * stride_r_bh + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_bh + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:    (M, K)
        a = tl.load(
            left_base + offs_m[:, None] * stride_l_i
                       + offs_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right block transposed: we need (K, N) = right[j,k]ᵀ
        b = tl.load(
            right_base + offs_k[:, None] * stride_r_k
                        + offs_n[None, :] * stride_r_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # a @ bᵀ  -> (M,N)
        acc += tl.dot(a, b)

    # write back
    tl.store(
        out_base + offs_m[:, None] * stride_o_i
                 + offs_n[None, :] * stride_o_j,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# 3. Main entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]  (float32, CUDA)
        * mask_tensor  : [B, N, N]      (float32/bool or None)
        * weights_dict : all learned parameters (torch tensors)
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # called H in the paper
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating: linear + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – tiled GEMM in Triton
    #    We work in fp16 for bandwidth, accumulate in fp32.
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # cast to fp16 for the kernel (inputs are fp32)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # output buffer (fp32)
    out_fp32 = torch.empty_like(left_f16, dtype=torch.float32)  # (B, H, N, N)

    # strides in *elements* (not bytes)
    s_l = left_f16.stride()   # (s_bh, s_h, s_i, s_k)
    s_r = right_f16.stride()
    s_o = out_fp32.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # dim 0 – rows
        triton.cdiv(N, BLOCK_N),          # dim 1 – cols
        B * hidden_dim,                   # dim 2 – batch*head
    )
    batched_gemm_kernel[grid](
        left_f16,
        right_f16,
        out_fp32,
        B, hidden_dim, N,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # permute back to (B, N, N, H)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with output gate (fused Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total_elts = out_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elts, BLOCK),)
    elemwise_mul_kernel[grid](
        out_flat, out_gate_flat, out_flat,
        total_elts,
        BLOCK=BLOCK,
    )
    # out now holds out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
