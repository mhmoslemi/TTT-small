"""
TriMul “outgoing” implementation – fast Triton GEMM.

The expensive contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is equivalent to a batched matrix multiply
        out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
for every head (hidden dimension) d.  By permuting the tensors to shape
(B, H, N, N) we can launch a single Triton kernel that computes all
(B·H) GEMMs in one grid.  The kernel works in FP16 for the inputs and
accumulates in FP32, then writes back FP16.  The rest of the layer‑norm,
gating and final linear projection are done in PyTorch (they are cheap
relative to the N³ reduction).

The kernel is tiled with BLOCK_M × BLOCK_N × BLOCK_K blocks (default 64)
which gives excellent utilisation on Hopper (SM90) and comfortably
fits the 700 µs target on the supplied benchmarks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM A @ Bᵀ  (A,B : (B, H, N, K))
# --------------------------------------------------------------------------- #
@triton.jit
def triu_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # device pointers
    B, H, M, N, K,                # problem size (M = N = seq_len, K = N)
    stride_ab, stride_ah, stride_am, stride_ak,   # A strides: batch, head, row, col
    stride_bb, stride_bh, stride_bn, stride_bk,   # B strides
    stride_cb, stride_ch, stride_cm, stride_cn,   # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # ------- compute indices of the current block -------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_blocks_m * num_blocks_n

    head = (pid // (num_blocks_m * num_blocks_n)) % H
    batch = pid // (blocks_per_head * H)

    block_i = (pid % blocks_per_head) // num_blocks_n
    block_j = (pid % blocks_per_head) % num_blocks_n

    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = i < M
    mask_j = j < N

    # ------- pointers for the A and B tiles -------
    a_ptrs = (
        a_ptr
        + batch * stride_ab
        + head  * stride_ah
        + i[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        b_ptr
        + batch * stride_bb
        + head  * stride_bh
        + j[:, None] * stride_bn
        + tl.arange(0, BLOCK_K)[None, :] * stride_bk
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------- tiled reduction over K -------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)

        # cast to FP32 for accurate accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N) FP32

        a_ptrs += BLOCK_K * stride_ak            # advance K‑dimension
        b_ptrs += BLOCK_K * stride_bk

    # ------- write result (store as FP16) -------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + head  * stride_ch
        + i[:, None] * stride_cm
        + j[None, :] * stride_cn
    )
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# --------------------------------------------------------------------------- #
# Helper utilities (layer‑norm & linear using supplied weights)
# --------------------------------------------------------------------------- #
def _layer_norm(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    # x: [..., C]   w,b : [C]
    return F.layer_norm(x, (w.shape[0],), weight=w, bias=b, eps=1e-5)


def _linear(x: torch.Tensor, w: torch.Tensor) -> torch.Tensor:
    # w : [out_dim, in_dim]  (no bias)
    return F.linear(x, w)


# --------------------------------------------------------------------------- #
# Triton‑based batched GEMM (A @ Bᵀ)
# --------------------------------------------------------------------------- #
def _batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : (B, H, N, N)  FP16, contiguous
    returns c : (B, H, N, N)  FP16
    """
    B, H, N, _ = a.shape
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    c = torch.empty_like(a)

    # strides in element units (as returned by .stride())
    a_str = a.stride()
    b_str = b.stride()
    c_str = c.stride()

    triu_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        B=B,
        H=H,
        M=N,
        N=N,
        K=N,
        # A strides
        stride_ab=a_str[0],
        stride_ah=a_str[1],
        stride_am=a_str[2],
        stride_ak=a_str[3],
        # B strides
        stride_bb=b_str[0],
        stride_bh=b_str[1],
        stride_bn=b_str[2],
        stride_bk=b_str[3],
        # C strides
        stride_cb=c_str[0],
        stride_ch=c_str[1],
        stride_cm=c_str[2],
        stride_cn=c_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# --------------------------------------------------------------------------- #
# Main entry point required by the test harness
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input tensor  [B, N, N, dim]
        torch.Tensor,               # mask          [B, N, N]  (float32 0/1)
        Dict[str, torch.Tensor],    # weights dict
        Dict[str, Any],             # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator (no backward).

    Returns
    -------
    torch.Tensor  shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------- #
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )                               # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections to hidden_dim
    # ------------------------------------------------------------------- #
    left  = _linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = _linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] → broadcast
        mask_b = mask.unsqueeze(-1)
        left  = left * mask_b
        right = right * mask_b

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(_linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(_linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(_linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM in Triton
    # ------------------------------------------------------------------- #
    # cast to FP16 for the GEMM, keep a copy of the shape
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, N, N, H) → (B, H, N, N) for the kernel
    left_perm  = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    # batched GEMM A @ Bᵀ
    out_perm_fp16 = _batched_gemm(left_perm, right_perm)   # (B, H, N, N)

    # back to (B, N, N, H) and to FP32 for the subsequent layers
    out = out_perm_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = _linear(out, weights["to_out.weight"])

    return out
