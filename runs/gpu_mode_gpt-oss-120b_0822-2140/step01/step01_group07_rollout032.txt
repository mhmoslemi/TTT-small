"""
TriMul “outgoing” implementation with a fused Triton GEMM.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply (B·H, N, N) @ (B·H, N, N)ᵀ where H = hidden_dim.
We reorder tensors to (B, H, N, N) → (M, N, N) (M = B·H) and compute the
product with a tiled fp16 GEMM kernel (accumulating in fp32).  All LayerNorm,
Linear and gating logic stays in PyTorch for simplicity.

The kernel works on half‑precision inputs, writes fp32 results and is launched
with a 3‑D grid: (tiles‑i, tiles‑j, matrix‑id).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  out = left @ rightᵀ
# left, right : (M, N, N)  (half)          – M = B * hidden_dim
# out         : (M, N, N)  (float32)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16  [M, N, N]
    right_ptr,         # *fp16  [M, N, N]
    out_ptr,           # *fp32  [M, N, N]
    M, N, K,           # K == N  (the reduction dim)
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_cm, stride_cn, stride_ck,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # which matrix (batch * hidden)

    # ------------------------------------------------------------------- #
    # offsets within the tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # pointers to the current tile (note: we access rightᵀ, i.e. (z, k, j))
    # ------------------------------------------------------------------- #
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn   # transposed view

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # main GEMM loop over K
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)          # left is fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)          # right is fp16 (transposed)

        # fp16 dot, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # store result
    # ------------------------------------------------------------------- #
    c_ptrs = out_ptr + pid_z * stride_cm \
                     + offs_m[:, None] * stride_cn \
                     + offs_n[None, :] * stride_ck
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of torch.Tensors (the model parameters)
            - config       : dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias) → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid) → still float32
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Cast to fp16 (bandwidth reduction) and reorder to (B, H, N, N)
    # ------------------------------------------------------------------- #
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B,N,N,H) → (B,H,N,N) → (M,N,N) where M = B*H
    left_h  = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer in fp32
    out_fp32 = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 6) Launch Triton GEMM kernel
    # ------------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_h.stride()
    stride_rm, stride_rn, stride_rk = right_h.stride()
    stride_cm, stride_cn, stride_ck = out_fp32.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim,                   # independent matrices
    )

    tri_mul_gemm_kernel[grid](
        left_h,
        right_h,
        out_fp32,
        B * hidden_dim,
        N,
        N,                                 # K == N
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                      # good for fp16 on Hopper
    )

    # ------------------------------------------------------------------- #
    # 7) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 8) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B, N, N, dim]

    return out
