"""
TriMul “outgoing” kernel (AlphaFold‑3)

The only O(N³) work in the forward pass is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every hidden channel *d*.  This is a batched matrix‑multiply
(left @ rightᵀ) with shape (B·H, N, N) × (B·H, N, N) → (B·H, N, N).
The kernel below computes that multiplication in FP16 (to halve
memory traffic) while accumulating in FP32 for numerical stability.
All other ops (LayerNorm, linear/gates, final projection) stay in
PyTorch – they are cheap compared with the N³ term.

The kernel works for any batch size B∈{1,2}, sequence length N up to
1024 and hidden dimension H (called ``hidden_dim`` in the config).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B*H, N, N) @ (B*H, N, N)ᵀ   →   (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_outgoing_kernel(
    A_ptr,                     # fp16 left   [B*H, N, K]
    B_ptr,                     # fp16 right  [B*H, N, K]
    C_ptr,                     # fp32 out    [B*H, N, N]
    M, N, K,                   # matrix sizes (M=N, K=N)
    stride_Ab, stride_Ar, stride_Ac,   # strides for A (batch, row, col)
    stride_Bb, stride_Br, stride_Bc,   # strides for B (batch, row, col)
    stride_Cb, stride_Cr, stride_Cc,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched (A @ Bᵀ) where A,B are fp16, accumulator is fp32."""
    pid_b = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for a tile of A (row‑major) and B (row‑major, later used as Bᵀ)
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Ar
        + offs_k[None, :] * stride_Ac
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_n[None, :] * stride_Br
        + offs_k[:, None] * stride_Bc
    )

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles.  Use FP16 load, mask out‑of‑bounds elements.
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Bc

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cr
        + offs_n[None, :] * stride_Cc
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (bool/float32) or None
        - weights      : dict of parameter tensors (all float32, CUDA)
        - config       : {'dim': int, 'hidden_dim': int, ...}

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]          # H
    B, N, _, _ = input_tensor.shape        # [B, N, N, dim]

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → [B, N, N, H]
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if present) – broadcast on hidden dim
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Batched matmul: (B, N, N, H) @ (B, N, N, H)ᵀ  →  (B, N, N, H)
    #    We treat each hidden head as an extra batch dimension:
    #       (B*H, N, N) @ (B*H, N, N)ᵀ
    # --------------------------------------------------------------
    #   (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * hidden, N, N).to(torch.float16)   # fp16 for bandwidth
    right = right.view(B * hidden, N, N).to(torch.float16)

    out_fp32 = torch.empty_like(left, dtype=torch.float32)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # Triton launch configuration – chosen to fit H200 shared‑memory well.
    #   BLOCK_M/N = 64 gives 64×64 tiles (16 KB) per block;
    #   BLOCK_K = 64 tiles the K‑dimension.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
    )

    # Strides for the three‑dimensional tensors (batch, row, col)
    stride_Ab, stride_Ar, stride_Ac = left.stride()
    stride_Bb, stride_Br, stride_Bc = right.stride()
    stride_Cb, stride_Cr, stride_Cc = out_fp32.stride()

    _batched_matmul_outgoing_kernel[grid](
        left,
        right,
        out_fp32,
        N,                # M
        N,                # N
        N,                # K
        stride_Ab, stride_Ar, stride_Ac,
        stride_Bb, stride_Br, stride_Bc,
        stride_Cb, stride_Cr, stride_Cc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden dim) + out‑gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
