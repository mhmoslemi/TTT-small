"""
TriMul “outgoing” forward pass.

The only O(N³) work is the contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply

    (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)

where H = hidden_dim.  We fuse this GEMM with a Triton kernel.
All surrounding LayerNorm / Linear / gating logic stays in PyTorch.
The kernel works on fp16 inputs (better memory bandwidth) and accumulates
in fp32, producing a fp32 output that matches the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  A @ Bᵀ  (A,B shape = (M,N,N))
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *float16  [M,N,N]
    right_ptr,         # *float16  [M,N,N]
    out_ptr,           # *float32  [M,N,N]
    M, N, K,           # dimensions (K == N)
    stride_lm, stride_ln, stride_lk,   # strides of left
    stride_rm, stride_rn, stride_rk,   # strides of right
    stride_om, stride_on, stride_ok,   # strides of out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # which matrix (B*H)
    pid_m = tl.program_id(1)          # row tile   (i)
    pid_n = tl.program_id(2)          # col tile   (j)

    # ----------------------------------- offsets & masks ----------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ----------------------------------- pointers ----------------------------------- #
    # left : (M,N,N) → index (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # rightᵀ : we need (z, k, j)
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------- GEMM loop ----------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot returns fp32 because acc is fp32
        acc += tl.dot(a, b)

        # advance in K‑direction
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ----------------------------------- store ----------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
            - mask         : torch.Tensor [B, N, N] or None   (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------- 1. Input LayerNorm
    # --------------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------- 2. Linear projections (no bias)
    # --------------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------- 3. Optional mask
    # --------------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------- 4. Gating (sigmoid)
    # --------------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------- 5. Fuse the N³ contraction with Triton
    # --------------------------------------------------------------------------- #
    # shape to (B, H, N, N) → (M, N, N) where M = B * H
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim

    left_reshape  = left.view(M, N, N).to(torch.float16)   # fp16 for bandwidth
    right_reshape = right.view(M, N, N).to(torch.float16)

    out_reshape = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # strides in element units
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # ------------------------------------------------------------------- #
    # launch configuration (tuned for Hopper)
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,                              # pid_z
        triton.cdiv(N, BLOCK_M),        # pid_m (row tiles)
        triton.cdiv(N, BLOCK_N),        # pid_n (col tiles)
    )

    tri_mul_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        M, N, N,                         # M, N, K (K == N)
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                     # good for fp16 on Hopper
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------- 6. Output LayerNorm, gating, final linear
    # --------------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
