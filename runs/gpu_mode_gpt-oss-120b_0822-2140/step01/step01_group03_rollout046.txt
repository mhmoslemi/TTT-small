"""
TriMul “outgoing” (AlphaFold‑3) – fast Triton implementation.

Algorithm
---------
1️⃣  Layer‑norm on the 4‑D input tensor (B×N×N×C).
2️⃣  Linear projections → left / right tensors (B×N×N×H)   (H = hidden_dim).
3️⃣  (Optional) mask the projected tensors.
4️⃣  Gated linear transforms (sigmoid) applied element‑wise.
5️⃣  Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched matrix multiplication `left @ rightᵀ` for every
    (batch, head) pair.  The heavy O(N³) work is done in a single Triton
    kernel that tiles the GEMM (BLOCK_M×BLOCK_N tiles, BLOCK_K reduction).
    Loads are fp16 for bandwidth, accumulation is fp32 for accuracy.
6️⃣  Post‑processing – layer‑norm + out‑gate + final linear projection back to C.

The kernel works for batch ≤2, N up to 1024 and hidden_dim = 128 (the
benchmark requirements).  Mixed‑precision is used only inside the kernel;
the function returns a float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _trmul_outgoing_kernel(
    A_ptr,                     # *float16  (batch, M, K)
    B_ptr,                     # *float16  (batch, N, K)  (will be read as K×N)
    C_ptr,                     # *float32  (batch, M, N)
    M, N, K,                   # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note Bk = stride of original col)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is transposed."""
    pid_b = tl.program_id(2)                 # batch * hidden_dim index
    pid_m = tl.program_id(0)                 # tile row
    pid_n = tl.program_id(1)                 # tile col

    # ----- tile offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ----- masks for out‑of‑bounds -----
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointers for the current tile -----
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    # B is read as transposed: we need element (k, j) → stride_Bk for k, stride_Bn for j
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main reduction over K -----
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float32)   # (M, K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)   # (K, N)
        acc += tl.dot(a, b)                     # (M, N)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ----- write result -----
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of weight tensors (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ Layer‑norm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # [B, N, N, H]
    right = F.linear(x, right_proj_w)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – also bias‑free linear layers
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core contraction (batched GEMM left @ rightᵀ) using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # reshape to (B*H, N, N) – convenient for the kernel
    left_t   = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t  = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel, fp32 accumulator / output
    left_fp16  = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)   # (B*H, N, N)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For transposed read we swap row/col strides
    stride_Bk = stride_Bc   # step along K (original column)
    stride_Bn = stride_Br   # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # Tuning parameters – empirically good for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch*head
    )

    _trmul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)            # back to original dim C

    return out
```

**Explanation of design choices**

* **Tile sizes** – `BLOCK_M = BLOCK_N = 64` gives good occupancy on the H200 while keeping shared‑memory pressure low. `BLOCK_K = 32` matches the native tensor‑core tile and enables fp16‑fp32 accumulation without many spill/reloads.

* **Strides for transposed access** – the kernel reads the right tensor as `Bᵀ` by swapping the row/column strides (`stride_Bk` / `stride_Bn`). This avoids an explicit transpose and saves memory bandwidth.

* **Mixed‑precision** – inputs to the kernel are cast to `float16` (the model weights are `float32`). Loads are `float16`, the dot product is accumulated in `float32`, and the final result is stored as `float32`. This matches the reference implementation’s numerical precision while halving memory traffic.

* **Mask handling** – the optional mask is applied before gating, exactly as in the reference code. When `nomask=True` the extra multiplication is omitted, avoiding unnecessary work.

* **Layer‑norm / gating / final projection** – kept in PyTorch (`torch.nn.functional`) because they are cheap compared to the O(N³) contraction and benefit from highly‑optimized cuBLAS / CUDA kernels.

The function `custom_kernel` therefore satisfies the specification: it implements the outgoing TriMul forward pass, fuses the expensive contraction with a Triton kernel, supports all listed configurations, and returns a float‑32 tensor.
