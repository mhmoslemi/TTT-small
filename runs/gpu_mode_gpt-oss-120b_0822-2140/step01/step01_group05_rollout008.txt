"""
TriMul “outgoing” implementation (AlphaFold‑3) with an
accelerated batched matrix‑multiply kernel written in Triton.

The heavy part of the forward pass is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which for each hidden channel *d* is a matrix multiplication
    (N × N) @ (N × N)ᵀ .
We fuse all hidden channels of a batch together and compute
B·hidden independent matrix products in a single Triton kernel.
The kernel works in FP16 (much higher throughput on H200) while the
accumulator is FP32 for numerical stability.  All other
operations (LayerNorm, linear projections, gating, final projection)
stay in PyTorch – they are cheap compared with the N³ work.

The kernel expects tensors in layout (B, H, N, N) with contiguous memory,
so we permute before the launch and permute back afterwards.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (A @ Bᵀ) for many (B*H) matrices of size N×N
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers  (fp16)
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A: (batch, M, K)
    stride_bb, stride_bm, stride_bk,   # strides for B: (batch, M, K)
    stride_cb, stride_cm, stride_cn,   # strides for C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each batch element.
    A, B, C are 3‑D tensors stored as (batch, M, K) , (batch, N, K) , (batch, M, N)
    The kernel works on FP16 inputs but accumulates in FP32.
    """
    pid_b = tl.program_id(0)               # batch*hidden index
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block col

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tile‑wise)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for t in range(num_k_tiles):
        # load tiles (convert fp16 -> fp32 on the fly)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # dot over the K‑tile
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store result (cast back to FP16)
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input  : torch.Tensor [B, N, N, dim]   (float32, CUDA)
            - mask   : torch.Tensor [B, N, N]       (bool/float, CUDA) or None
            - weights: dict of torch.Parameter tensors (float32, CUDA)
            - config : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]            # c_z in the paper
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (optional) – broadcast on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare for the batched matrix multiply:
    #    (B, N, N, hidden) -> (B, hidden, N, N) -> contiguous -> view as (B*hidden, N, N)
    #    Cast to fp16 for the kernel.
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    left_flat = left_fp16.view(B * hidden, N, N)   # (B*H, N, N)
    right_flat = right_fp16.view(B * hidden, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden,                     # program_id(0) – batch*hidden
        triton.cdiv(N, BLOCK_M),       # program_id(1) – rows
        triton.cdiv(N, BLOCK_N),       # program_id(2) – cols
    )

    # Strides of the three‑dimensional tensors
    a_batch, a_m, a_k = left_flat.stride()
    b_batch, b_m, b_k = right_flat.stride()
    c_batch, c_m, c_n = out_flat.stride()

    batched_matmul_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_m, a_k,
        b_batch, b_m, b_k,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Cast back to FP32 and restore original layout (B, N, N, hidden)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (hidden) + gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
