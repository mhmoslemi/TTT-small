The “outgoing” `TriMul` module can be written as a *batched* matrix‑multiply:

\[
\text{out}_{b,i,j,h}= \sum_{k=0}^{N-1}
      \text{left}_{b,i,k,h}\;
      \text{right}_{b,j,k,h},
\qquad
b\in[0,B),\;h\in[0,H)
\]

If we reshape the tensors to  

* `L`  – `(B·H, N, N)`   (row‑major, dtype `float16`)  
* `R`  – `(B·H, N, N)`   (row‑major, dtype `float16`)

the operation becomes a **batched GEMM** `L @ Rᵀ`.  
The kernel below tiles the multiplication in three dimensions
(`BLOCK_M`, `BLOCK_N`, `BLOCK_K`) and accumulates in FP32 for numerical
stability.  All surrounding linear layers, LayerNorms and gating are kept
in PyTorch – they are cheap compared with the \(N^3\) core contraction.

The implementation follows the “lessons” from the previous attempts:

* **Lesson 1 – permute before the kernel** – we move the hidden dimension to
  the second axis (`(B, H, N, N)`) and `contiguous()` the view.  
* **Lesson 2 – batched GEMM** – the kernel receives the full batch×head
  dimension as the third program‑id.
* **Strides are passed explicitly** – no reliance on flat pointers.
* **Mask handling** – optional, broadcasted on the hidden dimension.
* **Mixed‑precision** – inputs are cast to `float16` for the kernel,
  accumulation is performed in `float32`, final result is `float32`.

The kernel works for any batch size ≤ 2, sequence length up to 1024,
`dim` ∈ {128, 384} and `hidden_dim` = 128 (as required by the benchmark).

```python
"""
custom_kernel – Triton‑accelerated forward pass of the TriMul “outgoing” operator.

The heavy part (∑_k left[...,k,:] * right[...,k,:]) is expressed as a
batched GEMM L @ Rᵀ where L and R have shape (B*H, N, N).  A Triton kernel
tiles the multiplication (BLOCK_M × BLOCK_N × BLOCK_K) and accumulates in
FP32 while loading in FP16 for bandwidth.  All other operations (LayerNorm,
linear projections, gating, final norm + linear) are performed with PyTorch
functions.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, M, K] with K == N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_T_kernel(
    A_ptr,                     # *fp16  (batch, M, K)
    B_ptr,                     # *fp16  (batch, N, K) – will be read as K×N
    C_ptr,                     # *fp32  (batch, M, N)
    M, N, K,                   # int32 matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # strides of A  (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides of B  (batch, col, row) – transposed view
    stride_Cb, stride_Cm, stride_Cn,   # strides of C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)          # batch*head index
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # -------------------- block offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current block
    a_ptrs = A_ptr + pid_batch * stride_Ab \
             + offs_m[:, None] * stride_Am \
             + offs_k[None, :] * stride_Ak

    b_ptrs = B_ptr + pid_batch * stride_Bb \
             + offs_k[:, None] * stride_Bk \
             + offs_n[None, :] * stride_Bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- main loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # (M, K')
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (K', N)

        acc += tl.dot(a, b)                         # (M, N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # -------------------- write back --------------------
    c_ptrs = C_ptr + pid_batch * stride_Cb \
             + offs_m[:, None] * stride_Cm \
             + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float)
        - weights      : dict of parameters (all float32)
        - config       : dict with keys
                         * "dim"          – input channel size C
                         * "hidden_dim"   – projection size H
                         * "nomask" (opt) – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # weight shortcuts
    # --------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dimension = dim)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    # F.linear expects (..., in_features)
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w) # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM L @ Rᵀ (per hidden head)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # reshape to (B*H, N, N) and make them contiguous
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (half the bandwidth), fp32 accumulation
    left_fp16 = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # --------------------------------------------------------------
    # stride extraction (in elements, not bytes)
    # --------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()          # (batch, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()        # (batch, row, col)

    # For the transposed view of B we need
    #   stride over K (original column)  -> stride_Bc
    #   stride over N (original row)     -> stride_Br
    stride_Bk = stride_Bc
    stride_Bn = stride_Br

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # --------------------------------------------------------------
    # launch parameters (tuned for Hopper / SM90)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),     # rows
        triton.cdiv(N, BLOCK_N),     # cols
        B * H                        # batch * head
    )

    _trimul_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to dim C

    return out
```
