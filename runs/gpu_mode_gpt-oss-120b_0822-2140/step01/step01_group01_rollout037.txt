"""
TriMul “outgoing” forward pass (AlphaFold‑3 style)

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched GEMM: for each (b,d) slice we multiply
    L = left[b,:,:,d]          (N×N)
    R = right[b,:,:,d]         (N×N)
    out[b,:,:,d] = L @ Rᵀ      (N×N)

We flatten the batch and hidden dimensions (M = B*D) and launch a single
Triton kernel that computes M independent GEMMs.  The kernel works on
fp16 tensors (to exploit H200 tensor‑cores) while accumulating in fp32,
so the final result has the same precision as the reference implementation.
All other cheap operations (LayerNorm, linear projections, sigmoids,
masking, final norm / linear) stay in PyTorch.

The kernel tiles the multiplication with BLOCK_M × BLOCK_N × BLOCK_K blocks.
Typical block sizes (64,64,64) give good occupancy on Hopper SM‑90.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    N, M,                           # N = seq_len, M = B*hidden_dim
    stride_ah, stride_ai, stride_ak,  # A: (M,N,N) strides
    stride_bh, stride_bj, stride_bk,  # B: (M,N,N) strides
    stride_ch, stride_ci, stride_cj,  # C: (M,N,N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each (m) in [0, M).
    Input/Output tensors are expected to be in fp16, accumulation in fp32.
    """
    pid_m = tl.program_id(0)          # block-row (i)
    pid_n = tl.program_id(1)          # block-col (j)
    pid_z = tl.program_id(2)          # flattened (batch * hidden)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers for the selected (m = pid_z) matrix pair
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K in BLOCK_K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i,k]  (shape BLOCK_M × BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai          # i dimension
            + offs_k[None, :] * stride_ak          # k dimension
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load B_T[k,j] (i.e. B[j,k]) => shape BLOCK_K × BLOCK_N
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k dimension (row of B_T)
            + offs_n[None, :] * stride_bj          # j dimension (col of B_T)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 dot -> fp32 accumulation
        acc += tl.dot(a, b)

    # Write back C[i,j] (fp32)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul: left @ rightᵀ per hidden channel.
    left/right : (B, H, N, N)  fp16 or fp32
    returns    : (B, H, N, N)  fp32
    """
    B, H, N, _ = left.shape
    M = B * H                               # flattened batch*hidden
    # reshape to (M,N,N) – Triton works with contiguous tensors
    left_f = left.reshape(M, N, N).contiguous()
    right_f = right.reshape(M, N, N).contiguous()
    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64 (required by Triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Block size – chosen for Hopper (SM‑90) tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        M,                         # independent GEMMs
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The O(N³) contraction is executed by a fused Triton GEMM kernel;
    the remaining LayerNorm / Linear / Sigmoid / Mask operations stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1. Input LayerNorm ----------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ---------- 2. Linear projections (dim -> hidden_dim) ----------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)        # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # ---------- 3. Optional masking ----------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)           # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- 4. Gating ----------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Triton GEMM (left @ rightᵀ) ----------
    # Re‑order to (B, H, N, N) layout expected by the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 before the kernel (tensor‑core friendly)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    out_perm = _trimul_gemm(left_fp16, right_fp16)      # (B, H, N, N) fp32

    # Back to the original layout (B, N, N, hidden_dim)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Output LayerNorm + out‑gate ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ---------- 7. Final linear projection back to dim ----------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
