"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (B,N,N,C)            → (B,N,N,C)
2️⃣ Linear projections `left` / `right`            → (B,N,N,H)
3️⃣ Optional binary mask (broadcast over H)      → (B,N,N,H)
4️⃣ Gating (sigmoid of linear projections)       → (B,N,N,H)
5️⃣ Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (A @ Bᵀ) for every head h.
   • `left/right` are permuted to (B,H,N,N) and made contiguous.
   • A custom Triton kernel performs tiled FP16 × FP16 → FP32 accumulation.
   • Result is returned as FP16 and reshaped back to (B,N,N,H).
6️⃣ Layer‑norm over the hidden dimension H.
7️⃣ Multiply by the output gate.
8️⃣ Final linear projection back to the original channel size C.

Only the large N³ reduction is performed in Triton; everything else stays in
PyTorch (LayerNorm, Linear, sigmoid, etc.).  The kernel works for any batch
size ≤ 2, any sequence length up to 1024 and hidden dimensions that are a
multiple of the block size (the kernel will also handle non‑multiple sizes via
masking).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A,B : (BH, N, N))
# --------------------------------------------------------------
@triton.jit
def triton_trimul_gemm_kernel(
    A_ptr,                     # float16*   (BH, M, K)  – left
    B_ptr,                     # float16*   (BH, N, K)  – right (will be accessed as Bᵀ)
    C_ptr,                     # float16*   (BH, M, N)  – output
    M, N, K,                   # int32      matrix dimensions (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A
    stride_bh, stride_bk, stride_bn,   # strides for B (Bᵀ access)
    stride_ch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block col
    pid_bh = tl.program_id(2)           # batch‑head index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the tile in A and Bᵀ
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + pid_bh * stride_ah + offs_m[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_bh * stride_bh + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a tile of A (float16)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[None, :],
                    other=0.0)
        # load a tile of Bᵀ (float16)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[:, None] & mask_n[None, :],
                    other=0.0)

        # multiply‑accumulate (produces fp32)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write result (store as fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + pid_bh * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# Helper: element‑wise mask multiplication (optional, kept for completeness)
# --------------------------------------------------------------
@triton.jit
def mask_mul_kernel(src_ptr, mask_ptr, dst_ptr, N_ELEM, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < N_ELEM
    src = tl.load(src_ptr + offs, mask=m, other=0.0)
    mk  = tl.load(mask_ptr + offs, mask=m, other=0.0)
    tl.store(dst_ptr + offs, src * mk, mask=m)


def _apply_mask_triton(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """src and mask have identical shape (B,N,N,H).  Returns src*mask."""
    assert src.shape == mask.shape
    numel = src.numel()
    BLOCK = 131072                     # 128 K is safe for all test shapes
    grid = (triton.cdiv(numel, BLOCK),)
    out = torch.empty_like(src)
    mask_mul_kernel[grid](
        src_ptr=src,
        mask_ptr=mask,
        dst_ptr=out,
        N_ELEM=numel,
        BLOCK=BLOCK,
    )
    return out


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]
                  ) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, C], float32, CUDA
            - mask         : torch.Tensor, shape [B, N, N]  (0/1), float32, CUDA
            - weights      : dict of torch.Tensor containing all linear / LN params
            - config       : dict with keys
                * dim        – input channel size C
                * hidden_dim – projection size H
                * nomask     – bool, whether mask is all‑ones (default True)

    Returns:
        torch.Tensor of shape [B, N, N, C], float32, on CUDA.
    """
    # ----------------------------------------------------------
    # unpack
    # ----------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------
    # convenience helpers (torch implementations)
    # ----------------------------------------------------------
    def layer_norm(tensor, w, b):
        return F.layer_norm(tensor, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    def linear(tensor, w):
        # w : (out_dim, in_dim) – bias disabled in the reference
        return F.linear(tensor, w)

    # ----------------------------------------------------------
    # 1️⃣ Input layer‑norm
    # ----------------------------------------------------------
    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])          # (B,N,N,C)

    # ----------------------------------------------------------
    # 2️⃣ Project to hidden dimension
    # ----------------------------------------------------------
    left  = linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------
    # 3️⃣ Optional mask (binary, broadcast over H)
    # ----------------------------------------------------------
    if not nomask:
        # broadcast mask to the hidden dimension
        mask_b = mask.unsqueeze(-1).expand_as(left)
        left  = _apply_mask_triton(left,  mask_b)
        right = _apply_mask_triton(right, mask_b)

    # ----------------------------------------------------------
    # 4️⃣ Gating
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # (B,N,N,H)
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5️⃣ Core TriMul reduction – batched GEMM in Triton
    # ----------------------------------------------------------
    # Cast to fp16 for the kernel (kernel accumulates in fp32)
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # Permute to (B, H, N, N) and make contiguous → shape (BH, N, N)
    left_perm  = left_f16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_f16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    BH = B * H

    # Output buffer (BH, N, N), fp16
    out_f16 = torch.empty((BH, N, N), dtype=torch.float16, device=x.device)

    # Strides (as integers)
    stride_ah = left_perm.stride(0)   # BH stride
    stride_ai = left_perm.stride(2)   # row i stride (N)
    stride_ak = left_perm.stride(3)   # col k stride (1)

    stride_bh = right_perm.stride(0)  # BH stride
    stride_bk = right_perm.stride(3)  # k   stride (1)
    stride_bn = right_perm.stride(2)  # j   stride (N)

    stride_ch = out_f16.stride(0)     # BH stride
    stride_cm = out_f16.stride(1)     # i   stride (N)
    stride_cn = out_f16.stride(2)     # j   stride (1)

    # Tile sizes – chosen to give good occupancy on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH,                         # batch‑head dimension
    )

    triton_trimul_gemm_kernel[grid](
        A_ptr=left_perm,
        B_ptr=right_perm,
        C_ptr=out_f16,
        M=N, N=N, K=N,
        stride_ah=stride_ah, stride_ai=stride_ai, stride_ak=stride_ak,
        stride_bh=stride_bh, stride_bk=stride_bk, stride_bn=stride_bn,
        stride_ch=stride_ch, stride_cm=stride_cm, stride_cn=stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H) and cast to float32 for the rest
    out = out_f16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ----------------------------------------------------------
    # 6️⃣ Output layer‑norm (over hidden dim H)
    # ----------------------------------------------------------
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ----------------------------------------------------------
    # 7️⃣ Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 8️⃣ Final linear projection back to C
    # ----------------------------------------------------------
    out = linear(out, weights["to_out.weight"])

    return out
