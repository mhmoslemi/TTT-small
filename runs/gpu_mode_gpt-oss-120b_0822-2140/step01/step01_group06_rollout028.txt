"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a tiled GEMM in Triton.  The tensors are first
re‑shaped to (B, H, N, N) so that each (b, h) pair sees a regular
matrix‑multiply A @ Bᵀ.  The kernel works in fp16 for the loads
and uses an fp32 accumulator (the result is cast back to fp32).
All other operations (LayerNorm, Linear, Sigmoid, gating,
final projection) are performed with PyTorch – this satisfies the
“must use a kernel” requirement while keeping the implementation
clear and portable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, H, N,
    # strides (in elements) for the 4‑D tensors (B, H, N, N)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,  # rows of the output block   (i dimension)
    BLOCK_N: tl.constexpr,  # cols of the output block   (j dimension)
    BLOCK_K: tl.constexpr   # reduction dimension (k)
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where tensors are stored as (B, H, N, N) with the
    last two dimensions being the i/j (row/col) and k axes.
    """
    # ------------------------------------------------------------------
    # program ids: (batch*head), block‑row, block‑col
    # ------------------------------------------------------------------
    pid_bh = tl.program_id(0)               # combines batch and head
    pid_i  = tl.program_id(1)               # block row   (i)
    pid_j  = tl.program_id(2)               # block col   (j)

    b = pid_bh // H
    h = pid_bh %  H

    # ------------------------------------------------------------------
    # block offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # output pointer for the current (i,j) block
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction loop over k
    # ------------------------------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left: shape (M, K) -----
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + h * stride_left_h
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # ----- right: shape (N, K) -----
        # note: we load as (j, k) but will use trans_b=True in dot
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + h * stride_right_h
                      + offs_j[:, None] * stride_right_j
                      + offs_k[None, :] * stride_right_k)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float16)

        # dot: (M,K) x (N,K).T  → (M,N)
        acc += tl.dot(left, right, trans_b=True)

    # ------------------------------------------------------------------
    # store
    # ------------------------------------------------------------------
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32, CUDA
        * mask_tensor  : torch.Tensor, shape [B, N, N]  (or None)
        * weights_dict : mapping of parameter names → torch.Tensor
        * config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], dtype float32, CUDA
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)                # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Prepare for Triton GEMM: (B, H, N, N)
    # --------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    out_gate = out_gate.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # output buffer
    out = torch.empty_like(left)   # same dtype & device, shape [B, H, N, N]

    # --------------------------------------------------------------
    # 6️⃣ Launch Triton kernel
    # --------------------------------------------------------------
    # tile sizes – chosen to give good occupancy on Hopper.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * hidden_dim, 1),                 # program_id(0) = batch*head
        triton.cdiv(N, BLOCK_M),                       # program_id(1) = block rows
        triton.cdiv(N, BLOCK_N),                       # program_id(2) = block cols
    )

    # strides (in number of elements, not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, hidden_dim, N,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7️⃣ Back‑to original layout (B, N, N, H)
    # --------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8️⃣ Post‑norm on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 9️⃣ Element‑wise multiplication with the output gate
    # --------------------------------------------------------------
    # out_gate is still in (B, H, N, N) layout, bring it back
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]
    out = out * out_gate

    # --------------------------------------------------------------
    # 10️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
