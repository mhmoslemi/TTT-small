"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

The only O(N³) operation is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with left/right shape (B, N, N, H) where H = hidden_dim.
We rewrite it as a batched mat‑mul per hidden dimension:

    out_t[b, h, i, j] = left_t[b, h, i, :] @ right_t[b, h, j, :].T

i.e. for every (b, h) pair we multiply an (N×H) matrix with the transpose of an
(N×H) matrix, producing an (N×N) result.  This is exactly a GEMM and can be
handled efficiently with a custom Triton kernel that works on tiles
(TILE_M × TILE_N) and accumulates in FP32 while loading in FP16.

All other cheap operations (LayerNorm, Linear, sigmoids, gating) stay in
PyTorch.  The kernel works for any batch size ≤ 2, any sequence length up to
1024 and hidden dimensions that are a multiple of 8 (the configs we test).

The host‑side `custom_kernel` follows the same API as demanded by the
evaluation harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, H) @ (B, H, N, H)ᵀ  → (B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    A_ptr,                     # left  tensor pointer   (B,H,N,H)
    B_ptr,                     # right tensor pointer   (B,H,N,H)
    C_ptr,                     # output pointer         (B,H,N,N)
    N,                         # seq_len
    H,                         # hidden_dim
    stride_a_batch, stride_a_head, stride_a_i, stride_a_h,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_h,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,   # how many TILE_M rows per program_id(0) block
):
    """
    Compute for each (batch, head):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    (note that B is accessed with its last dimension as the reduction axis,
    i.e. we multiply A with the transpose of B along the hidden dimension.)
    """
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------------------------------------------------------
    # Compute row / column offsets for this tile
    # ------------------------------------------------------------------
    # Grouping tiles in the M dimension reduces the number of program ids
    # needed for large N.
    pid_m = pid_m * GROUP_SIZE_M + tl.program_id(3)

    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K = hidden_dim
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(H, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < H

        # --------------------------------------------------------------
        # Load A[b, h, i, k]  → shape (TILE_M, TILE_K)
        # --------------------------------------------------------------
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_h)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # --------------------------------------------------------------
        # Load B[b, h, j, k]  → we need Bᵀ, i.e. (k, j)
        # --------------------------------------------------------------
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_h   # k as first dim
                  + offs_n[None, :] * stride_b_i)   # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # --------------------------------------------------------------
        # Dot product over K
        # --------------------------------------------------------------
        acc += tl.dot(a, b)            # (TILE_M,TILE_K) @ (TILE_K,TILE_N)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32 or bfloat16)
        - mask         : torch.Tensor [B, N, N] (bool/float). May be all‑ones.
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helpers to fetch parameters (they are already on the correct device)
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )                                   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → (B,N,N,H)
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,H]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.to(x.dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy contraction via Triton GEMM kernel
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # allocate output in the same layout
    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in element counts (required by the kernel)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will index transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # ------------------------------------------------------------------
    # Tile sizes – chosen to fill H200 SMs efficiently.
    # ------------------------------------------------------------------
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64          # hidden_dim is a multiple of 64 in all test configs
    GROUP_SIZE_M = 1    # can be tuned; 1 works for all N up to 1024

    # grid dimensions
    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch*head
        GROUP_SIZE_M,                    # extra dim for grouping
    )

    _trimul_gemm_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=8,
        num_stages=3,
    )

    # back to original layout (B, N, N, H)
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))  # [B, N, N, dim]

    return out.to(dtype)
