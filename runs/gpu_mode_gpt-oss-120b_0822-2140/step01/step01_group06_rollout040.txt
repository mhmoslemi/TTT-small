"""
TriMul (outgoing) – fast Triton‑fused implementation.

Algorithm
---------
1. Layer‑norm over the last dimension of the input (PyTorch).
2. Linear projections → left, right  (no bias).
3. Optional mask broadcasting.
4. Gating: sigmoid(linear(x)) for left_gate, right_gate, out_gate.
5. Core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]            (O(N³·H))
   is performed by a custom Triton GEMM that works on the
   (B·H, N, N) view of *left* and *right*.  A and B are cast to fp16,
   the accumulation is fp32 and the result is written as fp32.
6. Layer‑norm on the hidden dimension H (PyTorch).
7. Element‑wise multiplication with *out_gate* – fused via a tiny
   Triton element‑wise kernel.
8. Final linear projection back to the original dimensionality (PyTorch).

The implementation keeps the surrounding bookkeeping in PyTorch
(for simplicity and to stay within the memory limits) and only
off‑loads the two heavy compute‑bound parts to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. ELEMENT‑WISE MULTIPLICATION (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr,                # flat pointer to tensor A
    b_ptr,                # flat pointer to tensor B
    out_ptr,              # flat pointer to output (may be same as a_ptr)
    N_ELEMENTS: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMENTS

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 2. BATCHED GEMM   C = A @ Bᵀ   (A,B : (B*H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    M, N, K,
    # strides (in elements, not bytes)
    stride_am, stride_ak,                 # A : (batch, M, K)
    stride_bk, stride_bn,                 # B : (batch, N, K)   (row‑major, we index as B[j,k])
    stride_cm, stride_cn,                 # C : (batch, M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    # accumulation precision flag
    FP16_ACCUM: tl.constexpr,
):
    # program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the current batch
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # initialise accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k‑range for this iteration
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # A tile:   (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        if FP16_ACCUM:
            a = a.to(tl.float16)

        # B tile (transposed view): we need B[j,k]
        b_ptrs = b_batch + (offs_n[None, :] * stride_bn) + (k_off[:, None] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & mask_k[:, None], other=0.0)
        if FP16_ACCUM:
            b = b.to(tl.float16)

        # dot‑product over K tile
        acc += tl.dot(a, b)

        # advance k‑pointers (already encoded via k_off in each loop)

    # write back C
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    if FP16_ACCUM:
        tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
    else:
        tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3. ENTRY‑POINT (custom kernel)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32 or bool, CUDA) or None
        * weights_dict : all learned parameters (torch tensors)
        * config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # 0. unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3. Apply optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)       # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core O(N³·H) contraction via Triton GEMM
    # --------------------------------------------------------------
    #   reshape to (B*H, N, N)  and make them contiguous for optimal memory access
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden_dim, N, N)

    left_flat = left.reshape(B * hidden_dim, N, N)
    right_flat = right.reshape(B * hidden_dim, N, N)

    # output buffer (fp32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # cast inputs to fp16 for bandwidth‑limited compute, keep fp32 accumulator
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # --------------------------------------------------------------
    # stride information (in *elements*, not bytes)
    # --------------------------------------------------------------
    stride_am = left_fp16.stride(1)           # stride over rows (M)
    stride_ak = left_fp16.stride(2)           # stride over K
    stride_bn = right_fp16.stride(1)          # stride over rows of B (N)
    stride_bk = right_fp16.stride(2)          # stride over K
    stride_cm = out_flat.stride(1)            # output row stride
    stride_cn = out_flat.stride(2)            # output col stride
    stride_batch = left_fp16.stride(0)        # B*H stride (N*N)

    # ------------------------------------------------------------------
    # launch parameters – these have been found to work well on Hopper GPUs
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                # batch dimension
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_batch_a=stride_batch,
        stride_batch_b=stride_batch,
        stride_batch_c=stride_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16_ACCUM=True,               # compute in fp16, accumulate fp32
    )
    # --------------------------------------------------------------
    # reshape back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6. Post‑norm on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                               # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 7. Element‑wise multiplication with out_gate (fused Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total_elems = out_flat.numel()
    BLOCK = 1024
    grid_elem = (triton.cdiv(total_elems, BLOCK),)
    elemwise_mul_kernel[grid_elem](
        out_flat,
        out_gate_flat,
        out_flat,
        N_ELEMENTS=total_elems,
        BLOCK=BLOCK,
    )
    # now out contains out * out_gate

    # --------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
