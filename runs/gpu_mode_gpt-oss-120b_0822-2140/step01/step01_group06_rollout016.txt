"""
TriMul (outgoing) – fast Triton‑fused implementation.

Core idea
---------
The expensive contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched matrix‑multiply for each head *h* (hidden_dim) and batch *b*:
    (N×N) @ (N×N)ᵀ  →  (N×N)
We reshape the tensors to (B, H, N, N) and launch a dedicated Triton GEMM
kernel that tiles the M, N and K dimensions (BLOCK_M×BLOCK_N×BLOCK_K).
All arithmetic is performed in fp16 (the two inputs) with an fp32 accumulator
to retain numeric stability.  After the kernel we cast back to fp32 and
continue the remaining lightweight operations (LayerNorm, gating, final
projection) in PyTorch.

Only the O(N³·H) part is in Triton – this satisfies the “must use a kernel”
requirement while keeping the implementation simple and portable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction.
# --------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,                # fp16  (B, H, N, N)
    right_ptr,               # fp16  (B, H, N, N)
    out_ptr,                 # fp16  (B, H, N, N)

    B, H, N,                 # problem sizes
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,

    BLOCK_M: tl.constexpr,   # tile size in output rows   (i)
    BLOCK_N: tl.constexpr,   # tile size in output cols   (j)
    BLOCK_K: tl.constexpr,   # tile size in reduction dim (k)
):
    # ---------- program IDs ----------
    pid0 = tl.program_id(0)          # combined batch * head
    pid1 = tl.program_id(1)          # block row   (i)
    pid2 = tl.program_id(2)          # block col   (j)

    # decode batch & head
    b = pid0 // H
    h = pid0 % H

    # ---------- offsets ----------
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A (left) ----
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # ---- load B (right) ----
        # right is accessed as (j, k) because we need the k‑dimension as the inner one.
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        b_mat = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # ---- matrix multiply tile ----
        acc += tl.dot(a, b_mat)          # fp16 * fp16 -> fp32 accumulation

    # ---------- write result ----------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]  (float32, CUDA)
        * mask_tensor  : torch.Tensor, shape [B, N, N] or None
        * weights_dict : model parameters (LayerNorm, Linear, etc.)
        * config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – Triton GEMM
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for simple stride handling
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # cast to fp16 for the kernel (memory‑friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer (fp16)
    out_fp16 = torch.empty_like(left_fp16)

    # strides in *elements* (not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_fp16.stride()
    stride_r_b, stride_r_h, stride_r_j, stride_r_k = right_fp16.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_fp16.stride()

    # tiling parameters – chosen for Hopper SM90 (128‑thread warps)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # batch * head
        triton.cdiv(N, BLOCK_M),                # row tiles
        triton.cdiv(N, BLOCK_N),                # col tiles
    )

    tri_mul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        hidden_dim,
        N,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_j,
        stride_r_k,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # convert back to fp32 and reorder to original layout [B, N, N, H]
    out = out_fp16.to(torch.float32).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Multiply by the output gate (element‑wise, still in PyTorch)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
