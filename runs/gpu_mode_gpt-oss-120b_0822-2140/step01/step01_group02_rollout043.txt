"""
TriMul “outgoing” forward pass (no gradients).

The expensive O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication
    left_slice   : (N, N)   = left[b, :, :, d]
    right_T_slice: (N, N)   = right[b, :, :, d].transpose(-2,-1)
    out_slice    = left_slice @ right_T_slice

We fuse **all** B·H independent GEMMs (one per batch * hidden_dim) in a
single Triton kernel.  The rest of the TriMul pipeline (LayerNorms,
Linear projections, sigmoidal gates and the final projection) stays in
PyTorch – they are cheap compared with the N³ work.

The kernel works in FP16 internally with FP32 accumulation (the
contraction tolerates reduced precision).  After the GEMM we cast back
to FP32 and finish the remaining PyTorch steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (B*H, N, N)   (fp16)
#   B : (B*H, N, N)   (fp16)   – we need Bᵀ inside the kernel
#   C : (B*H, N, N)   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims (int)
    stride_az, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # B strides (batch, row, col)
    stride_cz, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed."""
    pid_batch = tl.program_id(2)          # batch × hidden_dim index
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ------------------------------------------------------------------
    # Offsets for the current block (row i, col j)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the A‑tile (no transpose)
    a_ptrs = a_ptr + pid_batch * stride_az \
                  + offs_m[:, None] * stride_am \
                  + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K); for Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                  + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16 × fp16 → fp32 dot product
        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (cast back to fp16 to keep bandwidth low)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (B*H, N, N) in fp16.
    Returns a tensor of the same shape and dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N and b.shape == (B_H, N, N)

    c = torch.empty_like(a)

    # Strides are in element units
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # original layout (batch,N,K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Block sizes – good defaults for Hopper (sm90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # program_id(0)
        triton.cdiv(N, BLOCK_N),   # program_id(1)
        B_H,                       # program_id(2)
    )

    _batched_gemm_outgoing_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # tuned for 128×128 tiles on Hopper
        num_stages=2,
    )
    return c


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator with a fused Triton GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                  "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (shape broadcasted to [...,1])
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM:
    #    (B, N, N, hidden) -> (B*hidden, N, N)  (fp16, contiguous)
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) then collapse batch & head.
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Batched GEMM: C = left @ rightᵀ   (still fp16)
    # ------------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(left, right)   # (B*hidden, N, N)

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden) and cast to fp32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Post‑projection LayerNorm (hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate shape: (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
