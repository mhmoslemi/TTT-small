"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The dominant work of TriMul is the 4‑D contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel *d* (named “head” in the code) this is a dense
matrix‑matrix multiply L @ Rᵀ where  

    L = left[b,:, :, d]   (N×N)
    R = right[b,:, :, d]  (N×N)

We therefore launch a Triton kernel that computes **M = B·H** independent
GEMMs of shape (N×N)·(N×N)ᵀ.  The kernel works in FP16 for the loads and the dot,
accumulates in FP32, and writes the result back in FP32.  All other
operations (LayerNorm, linear projections, sigmoid gating, masking) are cheap
and stay in PyTorch.

The implementation follows the “batched GEMM” lessons:
*   permute tensors to (B, H, N, N) before the kernel – this makes the memory
    layout contiguous for each GEMM,
*   pass correct strides to the kernel,
*   tile the i‑, j‑ and k‑dimensions (BLOCK_M, BLOCK_N, BLOCK_K) and use
    `tl.dot` for the inner product,
*   use mixed‑precision (FP16 loads, FP32 accumulation) which is optimal on
    Hopper/SM90.

The host side `custom_kernel` performs the surrounding Linear / LayerNorm /
Sigmoid work, launches the Triton kernel, and finally projects back to the
original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, out
    N, M,                              # N = seq_len, M = B*H (flattened)
    stride_ah, stride_ai, stride_ak,   # strides of A (M, N, N)
    stride_bh, stride_bj, stride_bk,   # strides of B (M, N, N)
    stride_ch, stride_ci, stride_cj,   # strides of C (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for M independent (N×N) matrices.
    A, B, C are stored as (M, N, N) in row‑major order.
    """
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_z = tl.program_id(2)   # flattened batch * head index

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the beginning of the current (M‑th) matrices
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)               # cast to fp16 for dot

        # Bᵀ[k, j]  (original B is indexed [j, k]; we load transposed)
        # shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk        # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj        # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float16)

        # Dot product, accumulation in fp32
        acc += tl.dot(a, b)

    # Store result (FP32)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.

    Arguments
    ---------
    left, right : torch.Tensor of shape (B, H, N, N)  (float32)

    Returns
    -------
    out : torch.Tensor of shape (B, H, N, N) (float32)
    """
    B, H, N, _ = left.shape
    M = B * H                                 # flattened batch*head

    # reshape to (M, N, N) – this is what the kernel expects
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f)

    # strides as 64‑bit integers (torch returns them as python ints already)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – good coverage for N up to 1024 on Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # program_id 0  (i‑tiles)
        triton.cdiv(N, BLOCK_N),   # program_id 1  (j‑tiles)
        M,                         # program_id 2  (batch*head)
    )

    trimul_gemm_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        M,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # heuristic for Hopper SMs
    )

    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        * input_tensor : (B, N, N, dim)   float32
        * mask         : (B, N, N)       (bool/int) – may be omitted if config["nomask"]
        * weights      : dict of torch.Tensor – all linear / norm parameters
        * config       : dict with keys
              dim, hidden_dim, nomask (bool)

    Returns
    -------
    torch.Tensor of shape (B, N, N, dim) (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (dim -> dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)      # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also in FP32
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, H, N, N) layout required by the kernel.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    out_perm = trimul_gemm(left_perm, right_perm)       # (B, H, N, N)

    # Back to the original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
