## Final implementation

```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated GEMM.

The expensive operation is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden dimension `d` this is a matrix multiplication
    (N × N) @ (N × N)ᵀ
We fuse all hidden channels and the batch dimension, treat them as a
single batch of size `B*hidden`, and compute the GEMM with a custom
Triton kernel that:

* works on fp16 inputs,
* accumulates in fp32,
* uses a 64×64×64 tile (good for Hopper SM‑90),
* respects arbitrary strides (the tensors are made contiguous
  after a `(B, H, N, N)` permute).

All other operations (LayerNorm, linear projections, gating, final
linear) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (A @ Bᵀ) where A and B have shape (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    A_ptr,               # const float16*  (B*H, M, K)
    B_ptr,               # const float16*  (B*H, N, K) – we read transposed
    C_ptr,               #       float32*  (B*H, M, N)
    M, N, K,             # matrix sizes (all int32)
    stride_Am, stride_Ak,    # strides for A (row, col)  – batch stride is passed separately
    stride_Bm, stride_Bk,    # strides for B (row, col)
    stride_Cm, stride_Cn,    # strides for C (row, col)
    batch_stride_A, batch_stride_B, batch_stride_C,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch*hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the tile for this batch element
    a_ptrs = A_ptr + pid_b * batch_stride_A + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * batch_stride_B + offs_n[None, :] * stride_Bm + offs_k[:, None] * stride_Bk

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)                     # fp16 -> fp32 when used in dot
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # tl.dot automatically promotes to fp32 accumulator
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write result
    c_ptrs = C_ptr + pid_b * batch_stride_C + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32, CUDA)
        - mask          : torch.Tensor[B, N, N]      (bool or float, may be None)
        - weights       : dict of parameter tensors (all float32, CUDA)
        - config        : {'dim': int, 'hidden_dim': int}

    Returns
    -------
    torch.Tensor[B, N, N, dim]  (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # called hidden_dim in the reference code
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) – broadcast on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        # mask can be bool or float; cast to same dtype as left/right
        mask_ = mask.unsqueeze(-1).to(left.dtype)       # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the GEMM kernel
    #    (B, H, N, N) → (B*H, N, N) and cast to fp16
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * hidden, N, N).to(torch.float16)
    right_flat = right.view(B * hidden, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float32)   # accumulator is fp32

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # blocks in N (cols)
        triton.cdiv(N, BLOCK_M),   # blocks in M (rows)
        B * hidden,                # batch*hidden
    )

    # Strides of the flattened 3‑D tensors (batch, row, col)
    s_batch_a, s_row_a, s_col_a = left_flat.stride()    # (N*N, N, 1)
    s_batch_b, s_row_b, s_col_b = right_flat.stride()
    s_batch_c, s_row_c, s_col_c = out_flat.stride()

    trmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        s_row_a, s_col_a,
        s_row_b, s_col_b,
        s_row_c, s_col_c,
        s_batch_a,
        s_batch_b,
        s_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (hidden) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
