### Triton kernel + `custom_kernel`

```python
"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input (last dim = dim).
2. Linear projections → left / right (shape B,N,N,hidden_dim).
3. Optional mask (broadcast on the hidden dimension).
4. Gating: sigmoid(linear(x)) applied element‑wise.
5. Core operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication
        (B*H, N, N) @ (B*H, N, N)ᵀ .
   Implemented in Triton with fp16 loads, fp32 accumulation.
6. Layer‑norm + out‑gate + final linear back to the original dim.
The function returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch_head, M, K]),  C : [batch_head, M, N]
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,          # *fp16   (batch_head, M, K)
    B_ptr,          # *fp16   (batch_head, N, K)  (read as K×N)
    C_ptr,          # *fp32   (batch_head, M, N)
    M, N, K,        # matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (note Bk = stride of original column)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program id's
    pid_bh = tl.program_id(2)        # batch * hidden_dim
    pid_m  = tl.program_id(0)        # block row
    pid_n  = tl.program_id(1)        # block col

    # -------------------------------------------------------------
    # block start offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                    # [BLOCK_K]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this block
    a_ptrs = (
        A_ptr
        + pid_bh * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    # B is stored as (batch_head, N, K); we need it transposed → (K,N)
    b_ptrs = (
        B_ptr
        + pid_bh * stride_Bb
        + offs_k[:, None] * stride_Bk   # step in K (original column)
        + offs_n[None, :] * stride_Bn   # step in N (original row)
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # main K‑loop
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                          # (M,N) += (M,K)*(K,N)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # -------------------------------------------------------------
    # write result
    # -------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_bh * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float)  
        - weights      : dict of parameter tensors (all float32)
        - config       : dict containing {"dim", "hidden_dim", "nomask"}.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B,N,N] -> [B,N,N,1]
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction with Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape        # H == hidden_dim

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (bandwidth) – keep original dtype for safety
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # output buffer (fp32 accumulator)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # ------------------------------------------------------------------
    # strides (in element units, not bytes) – required by the kernel
    # ------------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (batch_head, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # we read B as transposed: K‑step = original column stride, N‑step = original row stride
    stride_Bk = stride_Bc
    stride_Bn = stride_Br

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # launch parameters – tuned for Hopper/SM90
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * H                         # batch * head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: layer‑norm → out‑gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)       # back to original dim C

    return out
```
