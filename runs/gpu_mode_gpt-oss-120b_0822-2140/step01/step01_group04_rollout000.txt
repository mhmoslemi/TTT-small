"""
TriMul “outgoing” kernel (AlphaFold3) – Triton + PyTorch.

The only O(N³) work is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension *d* (here called
`hidden_dim`).  This kernel computes, for every (batch, head) pair,

    C[b, h, i, j] = A[b, h, i, k] @ B[b, h, j, k]^T      (k summed)

i.e. `A @ Bᵀ`.  All surrounding LayerNorm / Linear / gating ops stay in
PyTorch because they are cheap compared with the N³ contraction.

*   Inputs are cast to FP16 for bandwidth.
*   Accumulation is performed in FP32 for numeric stability.
*   Tile sizes (M,N,K) are chosen to fit nicely into a single thread‑block
    on Hopper (SM90) and to expose enough parallelism for the test
    configurations (N up to 1024, hidden_dim up to 384).

The host entry‑point `custom_kernel` follows the signature required by the
grader.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ  (shape (B, H, N, N) @ (B, H, N, N)^T)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers (FP16)
    N, H,                         # problem dimensions
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. C = A @ Bᵀ  (both A and B are N×N matrices per head).
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ---------------------- tile offsets ----------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ---- loop over K dimension (tiles) ----
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # Load A tile: shape (TILE_M, TILE_K)
        a_ptrs = (A_ptr + batch * stride_a_batch
                           + head   * stride_a_head
                           + offs_i[:, None] * stride_a_i
                           + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load B tile transposed: we need B[j, k] -> access as (k, j)
        # B layout is (batch, head, i, j). To get Bᵀ we treat j as the row
        # dimension and k as the column dimension.
        b_ptrs = (B_ptr + batch * stride_b_batch
                           + head   * stride_b_head
                           + offs_k[:, None] * stride_b_k   # k dimension first
                           + offs_j[None, :] * stride_b_j)  # j dimension second
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # Dot product over K
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # ---------------------- store result ----------------------
    c_ptrs = (C_ptr + batch * stride_c_batch
                       + head   * stride_c_head
                       + offs_i[:, None] * stride_c_i
                       + offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32 or bfloat16)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with at least ``dim`` and ``hidden_dim`` keys.
          May also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # keep original dtype for later casts

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # shortcuts for weight tensors
    # ------------------------------------------------------------------
    def w(name):
        return weights[name].to(device)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (the last one)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → shape (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,H]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; convert to same dtype as left/right
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear projections, bias‑free)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – Triton kernel
    #    Rearrange to (B, H, N, N) and make contiguous for coalesced loads.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=torch.float32)   # result in FP32

    # strides in *elements* (not bytes) – required by Triton
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)  (we will load transposed)
    c_s = out_t.stride()    # (batch, head, i, j)

    # Tile sizes – tuned for Hopper SM90
    TILE_M = 128
    TILE_N = 128
    TILE_K = 32   # small K tile gives good register reuse

    grid = (
        triton.cdiv(N, TILE_M),        # rows
        triton.cdiv(N, TILE_N),        # cols
        B * hidden_dim,                # batch * head
    )

    trimul_outgoing_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,            # 8 warps fit well on SM90
        num_stages=3,
    )

    # Back to original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + gating + linear back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    # out_gate has shape [B, N, N, H]; broadcast over the same hidden dim.
    out = out * out_gate
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    # Return in the same dtype as the input (most callers expect fp32)
    return out.to(dtype)
