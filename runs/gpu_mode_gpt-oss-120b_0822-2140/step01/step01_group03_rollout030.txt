"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy part of the module is the 3‑D contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

for every batch `b` and hidden head `h`.  This is exactly a batched
matrix‑multiply `left_h @ right_h.T` where each matrix has shape (N,N).
We fuse all `B * hidden_dim` GEMMs into a single Triton kernel,
using fp16 loads (to cut bandwidth) and fp32 accumulation (to keep
numerical quality).  The rest of the pipeline (layer‑norms, linear
projections, sigmoid gates, final projection) stays in PyTorch.

The kernel works for batch ≤ 2, sequence length up to 1024 and hidden
dimension 128 (the values used in the benchmark).  Block sizes are
chosen for Hopper (SM 90) but work on any recent NVIDIA GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton GEMM kernel: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,          # *fp16  (batch, M, K)
    B_ptr,          # *fp16  (batch, N, K)   (will be read as K×N)
    C_ptr,          # *fp32  (batch, M, N)
    M, N, K,        # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (after transposition)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                # batch*head index
    pid_m = tl.program_id(0)                # block row
    pid_n = tl.program_id(1)                # block col

    # ---- block start coordinates ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- boundary masks ----
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- pointers for this block ----
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # ---- accumulator in fp32 ----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- main K‑loop ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)       # (M,blkK)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)       # (blkK,N)

        acc += tl.dot(a, b)                     # (M,N)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---- write result ----
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model weights (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optionally ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # weight shortcuts
    # --------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_w = weights["left_proj.weight"]
    right_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_w)          # [B,N,N,hidden]
    right = F.linear(x, right_w)        # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction → batched GEMM left @ rightᵀ per head
    # --------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden
    # reshape to (B*H, N, N) and make contiguous for nice strides
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside kernel (inputs), fp32 output
    left_fp16 = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)   # (B*H, N, N)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # after transposition we read B as (K,N) where K is original column
    stride_Bk = stride_Bc               # step along K (original col)
    stride_Bn = stride_Br               # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # --------------------------------------------------------------
    # Block sizes – tuned for Hopper (SM90)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),    # rows
        triton.cdiv(N, BLOCK_N),    # cols
        B * H                       # batch*head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [B, N, N, hidden]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)                      # back to dim C

    return out
