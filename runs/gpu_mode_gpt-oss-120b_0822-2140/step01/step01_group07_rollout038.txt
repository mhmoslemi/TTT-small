"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The only O(N³) operation is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply (B·D hidden, N, N) with the second
operand transposed on its last two dimensions.
We fuse this contraction into a single Triton GEMM that works on fp16
inputs (to exploit Tensor Cores on Hopper) while accumulating in fp32.
All surrounding LayerNorm / Linear / gating logic stays in PyTorch for
clarity.

Input
    x    : [B, N, N, dim]           (float32)
    mask : [B, N, N]  or None       (bool/float, 0/1)
    weights : dict of torch tensors (all float32)
    config  : {"dim":int, "hidden_dim":int}
Output
    out : [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM   out = left @ rightᵀ
# left/right : (M, N, K)   with K == N
# out        : (M, N, N)
# All tensors are stored in row‑major (C‑style) order.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16   [M, N, K]
    right_ptr,         # *fp16   [M, N, K]
    out_ptr,           # *fp32   [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows (i)
    pid_n = tl.program_id(1)          # tile over cols (j)
    pid_z = tl.program_id(2)          # independent matrix (batch * hidden)

    # ------------------------------------------------------------------- #
    # Offsets & masks
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers – note that we need rightᵀ, i.e. (z, k, j)
    # ------------------------------------------------------------------- #
    # left[z, i, k]
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # rightᵀ : we read right[z, k, j]
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main GEMM loop
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    eviction_policy="evict_last")
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    eviction_policy="evict_last")

        # a,b are fp16; tl.dot promotes to fp32 internally
        acc += tl.dot(a, b)

        # advance along K
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Store result (fp32)
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator with a fused Triton GEMM.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------- #
    if mask is not None:
        m = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------- #
    # 4. Gating (sigmoid) – also in float32
    # --------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Cast to fp16 for Tensor‑Core GEMM
    # --------------------------------------------------------------- #
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B,N,N,hidden) -> (B, hidden, N, N) -> (M, N, N) where M = B*hidden
    left_reshape  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer in fp32 (final precision)
    out_reshape = torch.empty_like(left_reshape, dtype=torch.float32, device=device)

    # --------------------------------------------------------------- #
    # 6. Strides (in elements) for Triton
    # --------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # --------------------------------------------------------------- #
    # 7. Triton launch configuration
    #    TILE sizes tuned for Hopper (Tensor‑Core friendly)
    # --------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows i
        triton.cdiv(N, BLOCK_N),          # cols j
        B * hidden_dim,                   # independent GEMMs
    )

    tri_mul_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden_dim, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # Hopper handles 8 warps well for fp16/fp32
        num_stages=3,         # double‑buffering
    )

    # --------------------------------------------------------------- #
    # 8. Reshape back to [B,N,N,hidden] (float32)
    # --------------------------------------------------------------- #
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 9. Output LayerNorm, gating, final linear
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])     # back to dim

    return out
