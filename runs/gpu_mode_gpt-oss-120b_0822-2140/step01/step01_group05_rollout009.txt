## Final implementation

```python
"""
TriMul “outgoing” (AlphaFold‑3) – forward only.

The expensive part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* this is a matrix multiplication
    (N×N) @ (N×N)ᵀ
We treat every (batch, hidden) pair as an independent batch and
perform a fused batched mat‑mul in Triton.  The surrounding
LayerNorm / Linear / gating steps stay in PyTorch (they are cheap
relative to the O(N³) work).  Inside the kernel we work in FP16
(half‑precision) and accumulate in FP32, which matches the
behaviour of `torch.einsum` with fp32 inputs on Hopper GPUs.

The kernel is autotuned for three sensible tile sizes; the launcher
picks the best one for the given sequence length.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched matmul A @ Bᵀ
#   A : (B*H, M, K)       B : (B*H, N, K)
#   C : (B*H, M, N)  where M = N = seq_len, K = seq_len
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 32}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64,  'BLOCK_K': 32}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 128, 'BLOCK_K': 32}, num_warps=4, num_stages=2),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes (all = seq_len)
    stride_Ab, stride_Ar, stride_Ac,   # strides for A : (batch, row, col)
    stride_Bb, stride_Br, stride_Bc,   # strides for B : (batch, row, col)
    stride_Cb, stride_Cr, stride_Cc,   # strides for C : (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many batches in one launch.
    A, B, C are stored in row‑major order.
    """
    pid_b = tl.program_id(0)            # batch*hidden index
    pid_m = tl.program_id(1)            # tile in M dimension
    pid_n = tl.program_id(2)            # tile in N dimension

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the first element of the tile
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_n[None, :] * stride_Br + offs_k[:, None] * stride_Bc

    # Accumulator in FP32 (tl.dot does FP16×FP16 → FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (offs_k[None, :] < K)),
                    other=0.0)                     # FP16
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < K) & mask_n[None, :]),
                    other=0.0)                     # FP16

        acc += tl.dot(a, b)                # FP16 dot → FP32 accumulation

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Bc

    # ------------------------------------------------------------------
    # Write out
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(c_ptrs,
             acc.to(tl.float16),            # store as FP16 (later cast back to FP32)
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[batch, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor[batch, N, N] (bool/float32) or None
        - weights      : dict of torch.Tensor (all on CUDA)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[batch, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also in FP32
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to half for the heavy mat‑mul
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, N, N, H) --> (B, H, N, N) --> (B*H, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    left_h = left_h.view(B * hidden, N, N)
    right_h = right_h.view(B * hidden, N, N)

    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Triton launch
    # ------------------------------------------------------------------
    # Strides in elements (not bytes) – already correct for contiguous tensors
    stride_Ab, stride_Ar, stride_Ac = left_h.stride()   # batch, row, col
    stride_Bb, stride_Br, stride_Bc = right_h.stride()
    stride_Cb, stride_Cr, stride_Cc = out_h.stride()

    # Grid: (batch*hidden, M‑tiles, N‑tiles)
    grid = (
        B * hidden,
        triton.cdiv(N, 64),   # tile count in M (the autotune will adapt BLOCK_M)
        triton.cdiv(N, 64),   # tile count in N
    )

    batched_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        N,               # M
        N,               # N
        N,               # K
        stride_Ab, stride_Ar, stride_Ac,
        stride_Bb, stride_Br, stride_Bc,
        stride_Cb, stride_Cr, stride_Cc,
        # block sizes are picked by the autotuner; we still have to pass *some* constexprs
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=32,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 and reshape to original layout
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (hidden dim) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
