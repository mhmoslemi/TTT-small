```python
"""
TriMul “outgoing” forward – Triton‑accelerated core contraction.

Algorithm
---------
1. Layer‑norm the input tensor (B×N×N×C).
2. Linear projections → left / right tensors (B×N×N×H)   (no bias).
3. (Optional) apply the pair‑wise mask.
4. Compute sigmoid gates and apply them element‑wise.
5. Core operation (batched GEMM):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a set of B*H independent matrix multiplications
   A @ Bᵀ where A,B ∈ ℝ^{N×N}.  Implemented in Triton with
   fp16 loads, fp32 accumulation, and a 128×128×32 tile.
6. Layer‑norm + out‑gate on the result, then final linear back to C.
7. Return a float‑32 tensor of shape [B, N, N, C].

Only the heavy N³ contraction lives in Triton; everything else is
performed with regular PyTorch for simplicity and readability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # *float16   (batch, M, K)
    B_ptr,                     # *float16   (batch, N, K)   read as K×N
    C_ptr,                     # *float32   (batch, M, N)
    M, N, K,                   # matrix sizes (all int32)
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)            # batch*head index
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block column

    # -------------------------------------------------------------
    # tile start offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # pointers for this tile
    # -------------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # main K‑loop
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # (M, K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (K, N)
        acc += tl.dot(a, b)                           # (M, N)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # -------------------------------------------------------------
    # write result
    # -------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple (input, mask, weights, config)
        - input  : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask   : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights: dict with the same keys as the reference nn.Module
        - config : dict containing at least ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C], dtype float32.
    """
    # -------------------------------------------------------------
    # unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # weight shortcuts (all float32)
    # -------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # -------------------------------------------------------------
    # 1️⃣ Layer‑norm on the last dimension (C)
    # -------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # -------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)   # [B, N, N, H]

    # -------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # -------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # -------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # -------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM in Triton
    # -------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (half the traffic)
    left_fp16  = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # output buffer (fp32 accumulator)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # compute strides in **elements** (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()      # (batch, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # view right as transposed: we need stride over K (original col) and over N (original row)
    stride_Bk = stride_Bc        # step when moving along K
    stride_Bn = stride_Br        # step when moving along N (output column)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # tile sizes – good for Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * H                         # batch * head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # back to original layout [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 6️⃣ Post‑processing
    # -------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)               # back to dim C

    return out
```
