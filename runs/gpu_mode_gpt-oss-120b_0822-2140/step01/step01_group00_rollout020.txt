"""
TriMul “outgoing” forward pass – highly‑fused Triton core.

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched GEMM:
    left  : (B, H, N, N)   –  H = hidden_dim
    right : (B, H, N, N)   –  we need left @ rightᵀ
For each (b,h) we compute   C = A @ A_T   with A = left[b,h] ,
right is used transposed, so the kernel performs a tiled GEMM
(A : M×K, B : N×K, output C : M×N) with `trans_b=True`.
All other ops (LayerNorm, Linear, gating, masking) stay in PyTorch,
but the massive reduction is executed in fp32 inside the Triton kernel.
The final tensor is brought back to (B,N,N,dim) and passed through the
output LayerNorm / gate / projection as in the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction
# --------------------------------------------------------------------------- #
@triton.jit
def trmul_gemm_kernel(
    A_ptr,          # (B,H,N,N) left  – bf16
    B_ptr,          # (B,H,N,N) right – bf16 (will be used transposed)
    C_ptr,          # (B,H,N,N) out   – bf16
    BATCH, HEAD, M, N, K,
    stride_Ab, stride_Ah, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bh, stride_Bm, stride_Bk,   # strides of B
    stride_Cb, stride_Ch, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B ∈ bf16, C ∈ bf16, accumulate in fp32)."""
    pid_bh = tl.program_id(0)          # combined batch*head dimension
    pid_m  = tl.program_id(1)          # block row   (i)
    pid_n  = tl.program_id(2)          # block col   (j)

    # start offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block : (M, K)
        a_ptrs = (A_ptr + pid_bh * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B block : (N, K)  – we will use trans_b=True so it behaves as (K,N)
        b_ptrs = (B_ptr + pid_bh * stride_Bb
                  + offs_n[:, None] * stride_Bm
                  + offs_k[None, :] * stride_Bk)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b, trans_b=True)

    # write result (still fp32) → cast back to bf16
    c_ptrs = (C_ptr + pid_bh * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.bfloat16),
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Helper: launch the Triton kernel for a single (B,H,N,N) tensor pair
# --------------------------------------------------------------------------- #
def trmul_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B : torch tensors of shape (B, H, N, N), dtype=bfloat16, contiguous.
    Returns C of the same shape / dtype.
    """
    assert A.shape == B.shape
    BATCH, HEAD, N, _ = A.shape
    M = N
    K = N

    # block sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BATCH * HEAD,                     # program_id(0) – batch*head
        triton.cdiv(M, BLOCK_M),          # program_id(1) – rows
        triton.cdiv(N, BLOCK_N),          # program_id(2) – cols
    )

    # strides after the (B,H,N,N) layout (contiguous)
    s0, s1, s2, s3 = A.stride()  # (batch, head, row, col)
    C = torch.empty_like(A)

    trmul_gemm_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        BATCH=BATCH,
        HEAD=HEAD,
        M=M,
        N=N,
        K=K,
        stride_Ab=s0,
        stride_Ah=s1,
        stride_Am=s2,
        stride_Ak=s3,
        stride_Bb=s0,
        stride_Bh=s1,
        stride_Bm=s2,
        stride_Bk=s3,
        stride_Cb=s0,
        stride_Ch=s1,
        stride_Cm=s2,
        stride_Cn=s3,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, CUDA)
        - mask         : [B, N, N]       (float32, 0/1, CUDA) – may be None
        - weights      : dict of torch.Tensors (no bias on linear layers)
        - config       : dict with keys
              * dim        – input channel size
              * hidden_dim – projection size
              * nomask     – bool, True → mask is all‑ones

    Returns
    -------
    torch.Tensor
        [B, N, N, dim]  (float32, CUDA)
    """
    # ---------------------------- unpack ------------------------------------ #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ---------------------------- layer‑norm ------------------------------- #
    def layer_norm(t, w, b):
        return F.layer_norm(t, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    # input norm
    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])   # (B,N,N,dim)

    # ---------------------------- linear projections ------------------------ #
    def linear(t, w):
        # w : (out_dim, in_dim) – bias is omitted in the model
        return F.linear(t, w)

    left  = linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = linear(x, weights["right_proj.weight"])

    # ---------------------------- mask (optional) -------------------------- #
    if not nomask:
        # mask: (B,N,N) → broadcast over hidden dim
        mask_b = mask.unsqueeze(-1)                     # (B,N,N,1)
        left  = left * mask_b
        right = right * mask_b

    # ---------------------------- gating ----------------------------------- #
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ---------------------------- prepare for GEMM ------------------------ #
    # cast to bf16 to reduce traffic, kernel accumulates in fp32
    left  = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    # reshape to (B, H, N, N) – one independent GEMM per head
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    # ---------------------------- core TriMul (Triton) -------------------- #
    out = trmul_gemm(left, right)                  # (B, H, N, N), bf16

    # reshape back to (B, N, N, H) and to fp32 for the following layers
    out = out.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,hidden)

    # ---------------------------- output layer‑norm ------------------------ #
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ---------------------------- output gate ------------------------------ #
    out = out * out_gate

    # ---------------------------- final linear ----------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
