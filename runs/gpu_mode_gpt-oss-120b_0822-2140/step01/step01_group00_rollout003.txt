"""
TriMul “outgoing” kernel – fully‑fused batched GEMM in Triton.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a set of independent matrix multiplications
    out_perm[b,h,i,j] = left_perm[b,h,i,:] @ right_perm[b,h,:,j]ᵀ
where h = hidden_dim (the projection dimension).  After a permute,
left/right have shape (B, H, N, N).  For each (b, h) pair we launch a
Triton kernel that performs a tiled GEMM (BLOCK_I × BLOCK_J output,
BLOCK_K reduction).  The kernel accumulates in fp32, loads in fp16 for
bandwidth, and writes the result back in fp32.  All other operations
(layer‑norms, linear layers, gating, optional mask) are kept in PyTorch,
but the heavy reduction is now executed on the GPU without any
intermediate materialisation in Python.

The implementation supports:
    * batch size 1‑2,
    * sequence length up to 1024,
    * dim ∈ {128,384}, hidden_dim = 128,
    * optional binary mask (broadcast over the hidden dimension).

The final tensor is returned in float32 with shape [B, N, N, dim].
"""
import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for one (batch, head) pair
# --------------------------------------------------------------------------- #
@triton.jit
def trmul_gemm_kernel(
    # pointers
    left_ptr,          # fp16   (B, H, N, N)
    right_ptr,         # fp16   (B, H, N, N)
    out_ptr,           # fp32   (B, H, N, N)
    # problem size
    N,                 # sequence length (int)
    # strides (in elements, not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Perform one GEMM:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    where `right` is accessed with its last two dimensions swapped so that
    the reduction dimension `k` is the innermost one.
    """
    # -------------------------------------------------------------------- #
    # Identify the (batch, head) this program instance works on
    # -------------------------------------------------------------------- #
    pid = tl.program_id(0)                # 0‑th axis = batch*head
    total_h = tl.program_id_axis_stride(0)  # total number of (b,h) combos
    b = pid // total_h
    h = pid % total_h

    # -------------------------------------------------------------------- #
    # 2‑D grid over the output matrix (i, j)
    # -------------------------------------------------------------------- #
    i_start = tl.program_id(1) * BLOCK_I
    j_start = tl.program_id(2) * BLOCK_J

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # -------------------------------------------------------------------- #
    # Loop over reduction dimension k in BLOCK_K chunks
    # -------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # ---------------------------------------------------------------- #
        # Load a tile of left: shape (BLOCK_I, BLOCK_K)
        # ---------------------------------------------------------------- #
        i_idx = i_start + tl.arange(0, BLOCK_I)
        k_idx_l = k_start + tl.arange(0, BLOCK_K)

        mask_i = i_idx < N
        mask_k_l = k_idx_l < N
        mask_l = mask_i[:, None] & mask_k_l[None, :]

        left = tl.load(
            left_base
            + i_idx[:, None] * stride_l_i
            + k_idx_l[None, :] * stride_l_k,
            mask=mask_l,
            other=0.0,
        )   # fp16

        # ---------------------------------------------------------------- #
        # Load a tile of right (transposed on‑the‑fly): shape (BLOCK_K, BLOCK_J)
        # right is stored as (j, k) but we need (k, j) for GEMM.
        # ---------------------------------------------------------------- #
        j_idx = j_start + tl.arange(0, BLOCK_J)
        k_idx_r = k_start + tl.arange(0, BLOCK_K)

        mask_j = j_idx < N
        mask_k_r = k_idx_r < N
        mask_r = mask_k_r[:, None] & mask_j[None, :]

        right = tl.load(
            right_base
            + j_idx[None, :] * stride_r_j          # column index = j
            + k_idx_r[:, None] * stride_r_k,       # row index = k (because we want (k,j))
            mask=mask_r,
            other=0.0,
        )   # fp16

        # ---------------------------------------------------------------- #
        # Multiply‑accumulate
        # ---------------------------------------------------------------- #
        acc += tl.dot(left, right)               # (I,K)·(K,J) → (I,J) in fp32

    # -------------------------------------------------------------------- #
    # Write the result tile
    # -------------------------------------------------------------------- #
    i_idx = i_start + tl.arange(0, BLOCK_I)
    j_idx = j_start + tl.arange(0, BLOCK_J)
    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_o = mask_i[:, None] & mask_j[None, :]

    tl.store(
        out_base
        + i_idx[:, None] * stride_o_i
        + j_idx[None, :] * stride_o_j,
        acc,
        mask=mask_o,
    )


# --------------------------------------------------------------------------- #
# Helper: launch the GEMM kernel for the whole batch/head dimension
# --------------------------------------------------------------------------- #
def trmul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, H, N, N]  (fp16, contiguous)
    returns      : [B, H, N, N] (fp32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, N, _ = left.shape
    out = torch.empty((B, H, N, N), dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left.stride()
    stride_r_b, stride_r_h, stride_r_j, stride_r_k = right.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out.stride()

    # block sizes – tuned for Hopper (SM90) and up to N=1024
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # axis 0 : (batch, head)
        triton.cdiv(N, BLOCK_I),           # axis 1 : i‑tiles
        triton.cdiv(N, BLOCK_J),           # axis 2 : j‑tiles
    )

    trmul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        stride_l_b=stride_l_b,
        stride_l_h=stride_l_h,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_r_b=stride_r_b,
        stride_r_h=stride_r_h,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_o_b=stride_o_b,
        stride_o_h=stride_o_h,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point required by the autograder
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]
                  ) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   float32, CUDA
        - mask         : [B, N, N]       float32 (0/1) or None
        - weights      : dict of torch.Tensor (all linear / LN params)
        - config       : dict with keys
              * dim        – input channel size
              * hidden_dim – projection size
              * nomask     – bool, True if mask = all‑ones

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, CUDA)
    """

    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Helper: layer‑norm and linear using supplied weight tensors
    # ------------------------------------------------------------------- #
    def layer_norm(x, w, b):
        return F.layer_norm(
            x,
            normalized_shape=(w.shape[0],),
            weight=w,
            bias=b,
            eps=1e-5,
        )

    def linear(x, w):
        # w : [out_dim, in_dim] – bias is omitted (model uses bias‑less linear)
        return F.linear(x, w)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm (dim → dim)
    # ------------------------------------------------------------------- #
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Project to hidden_dim (dim → hidden_dim)
    # ------------------------------------------------------------------- #
    left = linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM in Triton
    #    Convert to (B, H, N, N) layout, cast to fp16 for higher bandwidth
    # ------------------------------------------------------------------- #
    #   left_perm[b, h, i, k] = left[b, i, k, h]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # GEMM kernel produces fp32 output in (B, H, N, N) layout
    out_fp32 = trmul_gemm(left_fp16, right_fp16)   # fp32

    # Back‑to original layout: (B, N, N, H)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate (element‑wise)
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection (hidden_dim → dim)
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
