"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

The only O(N³) work is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be seen as a batched mat‑mul per hidden dimension `d`:
    (B, H, N, N) @ (B, H, N, N)ᵀ   →   (B, H, N, N)

All cheap ops (LayerNorm, Linear, gating, final LayerNorm/Linear) stay in
PyTorch.  The heavy contraction is carried out by a Triton kernel that:
* works on FP16 tensors (inputs are cast, saving bandwidth)
* accumulates in FP32 (numerical safety)
* processes a TILE_M × TILE_N block per program instance
* loops over K in tiles of size TILE_K

The host‑side `custom_kernel` prepares the tensors, launches the kernel,
and performs the remaining PyTorch post‑processing.  The final result
has dtype ``torch.float32`` to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # tensor pointers
    N, H,                               # problem sizes
    stride_Ab, stride_Ah, stride_Ai, stride_Ak,
    stride_Bb, stride_Bh, stride_Bi, stride_Bj,
    stride_Cb, stride_Ch, stride_Ci, stride_Cj,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Compute, for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B have shape (N, N)).
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # -------------------------------------------------------------
    # 1) offsets inside the current tile
    # -------------------------------------------------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # 2) loop over the K dimension in tiles
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A[b, h, i, k]   shape (TILE_M, TILE_K)
        a_ptr = (A_ptr
                 + batch * stride_Ab
                 + head  * stride_Ah
                 + offs_i[:, None] * stride_Ai
                 + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ: we need B[b, h, j, k] → load as (TILE_K, TILE_N)
        b_ptr = (B_ptr
                 + batch * stride_Bb
                 + head  * stride_Bh
                 + offs_k[:, None] * stride_Bj   # k becomes first dim
                 + offs_j[None, :] * stride_Bi)  # j becomes second dim
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (M×K) @ (K×N) → (M×N)

    # -------------------------------------------------------------
    # 3) store result
    # -------------------------------------------------------------
    c_ptr = (C_ptr
             + batch * stride_Cb
             + head  * stride_Ch
             + offs_i[:, None] * stride_Ci
             + offs_j[None, :] * stride_Cj)
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, dim)  (float32)
        - mask         : torch.Tensor of shape (B, N, N)      (bool/float)
        - weights      : dict of torch.Tensors containing all model
                         parameters described in the problem statement.
        - config       : dict with keys ``dim`` and ``hidden_dim`` as
                         well as optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, dim) (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    use_mask = not cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # shortcuts for weight tensors (all already on the correct device)
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # shape (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim channels
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # (B,N,N,hidden_dim)
    right = F.linear(x, w("right_proj.weight"))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear layers, also bias‑free)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy contraction – Triton kernel
    #    Rearrange to (B, H, N, N) and make contiguous for coalesced loads.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    out_t = torch.empty_like(left_t, dtype=dtype)   # (B, H, N, N)

    # strides in **elements** (not bytes) – required by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j) (we will load transposed)
    c_s = out_t.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # launch configuration – tuned for Hopper (SM90)
    # ------------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # program id dim 0
        triton.cdiv(N, TILE_N),          # program id dim 1
        B * hidden_dim,                  # program id dim 2  (batch * head)
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,                     # good default for 64×64 tiles
    )

    # back to original layout: (B, N, N, hidden_dim)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))  # (B,N,N,dim)

    return out.to(dtype)
