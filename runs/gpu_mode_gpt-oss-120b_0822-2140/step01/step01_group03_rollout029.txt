## Triton‑accelerated *outgoing* TriMul

```python
"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core contraction.

Algorithm
---------
1. Layer‑norm the input (last dim = dim).
2. Linear projections → left / right tensors (shape B×N×N×H).
3. Optional mask (broadcast on H).
4. Gated‑linear transforms (sigmoid gates) applied element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM A @ Bᵀ for each (batch, head) pair.
   We launch a Triton kernel that tiles M,N,K (M,N≈64, K≈32), loads fp16,
   accumulates in fp32 and writes fp32.
6. Post‑processing: LayerNorm → out‑gate → final linear projection back to dim.

The kernel works for batch ≤2, sequence length up to 1024 and any hidden_dim
(typically 128).  Mixed‑precision is used only inside the kernel; the final
output is float‑32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # *float16   (batch, M, K)
    B_ptr,                     # *float16   (batch, N, K)   read as K×N
    C_ptr,                     # *float32   (batch, M, N)
    M, N, K,                   # matrix sizes (int32)
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)          # batch index = B*H
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ----- block offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointers for this block -----
    a_ptrs = (A_ptr + pid_batch * stride_Ab
              + offs_m[:, None] * stride_Am
              + offs_k[None, :] * stride_Ak)

    b_ptrs = (B_ptr + pid_batch * stride_Bb
              + offs_k[:, None] * stride_Bk   # K dimension
              + offs_n[None, :] * stride_Bn)  # N dimension (original rows)

    # ----- accumulator (fp32) -----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main K loop -----
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # (M, K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (K, N)

        acc += tl.dot(a, b)                                 # (M,N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ----- store result -----
    c_ptrs = (C_ptr + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float) or None
        - weights      : dict of parameter tensors (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ LayerNorm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1]
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – also bias‑free linear
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core contraction – batched GEMM A @ Bᵀ per (batch, head)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    # reshape to (B*H, N, N) and keep contiguity
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # fp16 inside the kernel for bandwidth, fp32 accumulation
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # ---- strides (in elements, not bytes) ----
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()          # (batch, M, K)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()         # (batch, N, K) original layout

    # For the transposed view Bᵀ we need:
    stride_Bk = stride_Bc            # step along K (original column)
    stride_Bn = stride_Br            # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ---- tiling parameters (tuned for Hopper) ----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch*head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, to_out_w)            # back to original dim C

    return out
```
