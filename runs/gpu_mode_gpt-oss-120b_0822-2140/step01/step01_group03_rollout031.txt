"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core contraction.

The heavy part of the module is the tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be expressed as a batched matrix‑multiply
(left_slice @ right_sliceᵀ) for each (batch, head) pair.
We fuse many of those GEMMs in a single Triton kernel that:
* loads the two 2‑D slices in fp16,
* accumulates in fp32 (to retain numeric stability),
* writes the result in fp32,
* handles arbitrary batch size (≤2) and hidden_dim (here 128).

All other ops (LayerNorm, Linear, gating, final projection) are cheap
and are performed with PyTorch.  The kernel therefore dominates runtime
while keeping the implementation simple and correct for all benchmark
shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # fp16   (batch*H, M, K)
    B_ptr,                     # fp16   (batch*H, N, K)  (will be read as K×N)
    C_ptr,                     # fp32   (batch*H, M, N)
    M, N, K,                   # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A  (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides of B  (batch, col, row) – note the transposed view
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids ---------------------------------------------------------
    pid_b = tl.program_id(2)                # index across (batch*H)
    pid_m = tl.program_id(0)                # block row index
    pid_n = tl.program_id(1)                # block col index

    # block offsets -------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds --------------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this block --------------------------------------------
    a_ptrs = (A_ptr +
              pid_b * stride_Ab +
              offs_m[:, None] * stride_Am +
              offs_k[None, :] * stride_Ak)

    b_ptrs = (B_ptr +
              pid_b * stride_Bb +
              offs_k[:, None] * stride_Bk +      # K dimension (original column)
              offs_n[None, :] * stride_Bn)      # N dimension (original row)

    # accumulator (fp32) --------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main K‑loop ---------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A & B tiles, apply masks, cast to fp32
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)   # (BLOCK_M, BLOCK_K)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                  # (BLOCK_M, BLOCK_N)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write result --------------------------------------------------------
    c_ptrs = (C_ptr +
              pid_b * stride_Cb +
              offs_m[:, None] * stride_Cm +
              offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
              mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of torch tensors (all float32)
        - config       : dict with keys:
                "dim"        – C
                "hidden_dim"– H
                "nomask"    – if True, ignore the mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    out_norm_w, out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on input (last dim = C)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1]
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # reshape to (B*H, N, N) and make contiguous for clean strides
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B*H, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [B*H, N, N]

    # fp16 inside kernel – reduces memory traffic
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # output buffer (fp32 accumulation)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # --------------------------------------------------------------
    # strides (in element units, not bytes) – required by Triton
    # --------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (batch*H, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()  # (batch*H, row, col)

    # For the transposed view of B we need:
    #   Bk – step along K (original column index)
    #   Bn – step along N (original row index)
    stride_Bk = stride_Bc   # column stride becomes K‑stride
    stride_Bn = stride_Br   # row stride becomes N‑stride

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # -----------------------------------------------------------------
    # Tile configuration – tuned for Hopper (SM90)
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H                             # batch * hidden
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=out_norm_w, bias=out_norm_b)
    out = out * out_gate
    out = F.linear(out, out_w)            # back to original dim C

    return out
