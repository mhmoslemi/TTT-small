## Triton‑accelerated **outgoing** TriMul

```python
"""
TriMul “outgoing” forward pass (no backward).

The only O(N³·H) work is the contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiply  C = A @ Bᵀ
with A,B ∈ (B·H, N, N).  The kernel computes this in FP16
(load·store) while accumulating in FP32, then the surrounding
PyTorch code handles LayerNorms, linear projections and gates.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  C = A @ Bᵀ   (batched GEMM)
#   A : (B·H, M, K)   – row‑major
#   B : (B·H, N, K)   – row‑major, we treat it as transposed inside the kernel
#   C : (B·H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (int)
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col) – note: row = k
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed."""
    pid_batch = tl.program_id(2)          # batch dimension (B·H)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current A‑tile (row i, col k)
    a_ptrs = a_ptr + pid_batch * stride_az \
                   + offs_m[:, None] * stride_am \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # For Bᵀ we need element (k, n) of the original B layout.
    b_ptrs = b_ptr + pid_batch * stride_bz \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A and Bᵀ
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # tl.dot does fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as FP16 to keep bandwidth low)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=store_mask)


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches ``batched_gemm_kernel``.
    Arguments:
        a, b : (B·H, N, N) tensors, dtype float16, contiguous.
    Returns:
        c : (B·H, N, N) tensor, dtype float16, contiguous.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "square matrices are required"

    c = torch.empty_like(a)

    # Strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # original layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – good default for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B_H,                       # batch (B·H)
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,          # M
        N,          # N
        N,          # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # works well for 128×128 tiles on H200
    )
    return c


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.
    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape (B, N, N, dim), dtype float32
        mask         : torch.Tensor, shape (B, N, N)  (bool or float)
        weights      : dict of torch.Tensor (float32) – same keys as the
                       reference PyTorch module
        config       : dict with at least:
                       * "dim"         – input channel dimension
                       * "hidden_dim" – hidden dimension (H)
                       * "nomask"      – bool, True if mask should be ignored
    Returns
    -------
    torch.Tensor of shape (B, N, N, dim), dtype float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim), float32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for batched GEMM
    #    shape -> (B, H, N, N) → (B*H, N, N)  in float16
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6. Core contraction via Triton batched GEMM
    #    out_fp16 = left @ rightᵀ   (shape B*H, N, N)
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left_fp16, right_fp16)

    # --------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H) and cast to FP32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Post‑projection LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # out_gate already shaped (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
