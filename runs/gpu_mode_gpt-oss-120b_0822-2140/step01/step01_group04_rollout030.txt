"""
TriMul “outgoing” kernel (AlphaFold3).

The cost‑dominating part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.  The kernel below
computes, for every (batch, hidden) pair, the product

    C = A @ Bᵀ            A,B : (N,N)

using Triton.  All cheap operations (layer‑norms, linear projections,
sigmoid gates, final projection) stay in PyTorch.  The kernel works in
fp16 for bandwidth and accumulates in fp32, writing the result back in fp32
to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem dimensions
    stride_a_batch, stride_a_head,
    stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head,
    stride_b_i, stride_b_j,            # note: B is stored as (batch,head,i,j)
    stride_c_batch, stride_c_head,
    stride_c_i, stride_c_j,            # C storage like A
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Computes for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ   (both A and B are N×N matrices)
    """
    pid_m = tl.program_id(0)   # tile row   (i)
    pid_n = tl.program_id(1)   # tile col   (j)
    pid_hb = tl.program_id(2)  # combined batch * head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------------------------------------------------------
    # offsets inside the tile
    # ------------------------------------------------------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ----- load A[i,k] ------------------------------------------------
        a_ptrs = (A_ptr + batch * stride_a_batch
                           + head  * stride_a_head
                           + offs_i[:, None] * stride_a_i
                           + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ----- load Bᵀ[k,j]  (B is stored as (i,j) so we swap axes) ---------
        b_ptrs = (B_ptr + batch * stride_b_batch
                           + head  * stride_b_head
                           + offs_k[:, None] * stride_b_i   # k as first dim
                           + offs_j[None, :] * stride_b_j)  # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) -> (TILE_M,TILE_N)

    # ------------------------------------------------------------------
    # store result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + batch * stride_c_batch
                       + head  * stride_c_head
                       + offs_i[:, None] * stride_c_i
                       + offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be all ones)
        - weights      : dict mapping weight names to tensors
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # convenience accessors for weight tensors
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)        # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight"))) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy O(N³) contraction via Triton
    #    Rearrange to (B, H, N, N) and make contiguous for nice strides.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B,H,N,N]

    # stride in elements (not bytes)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j)  (we will treat it as transposed)
    c_s = out_t.stride()     # (batch, head, i, j)

    # launch configuration
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),         # rows
        triton.cdiv(N, TILE_N),         # cols
        B * hidden_dim,                 # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,
    )

    # back to original layout [B,N,N,H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))    # [B,N,N,dim]

    return out.to(dtype)
