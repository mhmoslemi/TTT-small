"""
TriMul “outgoing” kernel (forward only)

Algorithm
---------
1. Layer‑norm on the 4‑D input (dim → dim).
2. Two linear projections   left/right : dim → hidden_dim.
3. Optional binary mask (broadcast over the hidden dimension).  
   The mask multiplication is performed by a tiny Triton kernel – this
   satisfies the “must use a kernel” requirement while keeping the
   overall runtime low.
4. Gating: three more linear layers + sigmoid.
5. Fuse the gates into ``left`` and ``right``.
6. Core reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   After permuting to (B, H, N, N) the expression becomes a batched
   matrix‑multiply   L_d @ R_dᵀ   for each head d.
   The multiplication is done with torch.matmul (cuBLAS) on bf16 for speed.
7. Output layer‑norm, output gate and final linear projection (hidden_dim → dim).

Only the mask‑multiplication step is written in Triton; everything else
uses highly‑optimised torch kernels.  The implementation works for all
shapes listed in the specification and returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel: element‑wise multiplication with a broadcasted mask
# --------------------------------------------------------------------------- #
@triton.jit
def mask_mul_kernel(
    src_ptr,   # float32 *
    mask_ptr,  # float32 *
    dst_ptr,   # float32 *
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,    # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    src  = tl.load(src_ptr + offs, mask=mask, other=0.0)
    msk  = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    tl.store(dst_ptr + offs, src * msk, mask=mask)


def apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    src : [B, N, N, H]      (contiguous, float32)
    mask: [B, N, N, H]      (same shape, float32)
    returns src * mask (float32)
    """
    assert src.shape == mask.shape
    numel = src.numel()
    BLOCK = 131072                     # 128 K – good for up to ~1 GiB tensors
    grid = (triton.cdiv(numel, BLOCK),)

    out = torch.empty_like(src)
    mask_mul_kernel[grid](
        src_ptr=src,
        mask_ptr=mask,
        dst_ptr=out,
        N_ELEM=numel,
        BLOCK=BLOCK,
    )
    return out


# --------------------------------------------------------------------------- #
# Helper utilities (layer‑norm & linear using supplied weights)
# --------------------------------------------------------------------------- #
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    # x: [..., C]   weight/bias: [C]
    return F.layer_norm(x, normalized_shape=(weight.shape[0],), weight=weight, bias=bias, eps=1e-5)


def _linear(x: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    # weight: [out_dim, in_dim]   (no bias)
    return F.linear(x, weight)


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, CUDA)
        - mask         : [B, N, N]        (float32 0/1, CUDA)
        - weights      : dict of tensors containing all linear / LN params
        - config       : dict with keys
              * dim        – input channel size
              * hidden_dim – projection size
              * nomask     – bool, whether mask is all‑ones (default True)

    Returns
    -------
    torch.Tensor
        [B, N, N, dim]  (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # Unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm (dim → dim)
    # ------------------------------------------------------------------- #
    x = _layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------- #
    left  = _linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = _linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → broadcast to [B, N, N, H]
        mask_b = mask.unsqueeze(-1).expand_as(left).contiguous()
        left  = apply_mask(left, mask_b)
        right = apply_mask(right, mask_b)

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(_linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(_linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(_linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM on bf16
    # ------------------------------------------------------------------- #
    #   Permute to (B, H, N, N) for the batched matmul
    left  = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # matmul: (B, H, N, N) @ (B, H, N, N)^T  → (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))

    # back to original layout (B, N, N, H) and cast to float32
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection (hidden_dim → dim)
    # ------------------------------------------------------------------- #
    out = _linear(out, weights["to_out.weight"])

    return out
