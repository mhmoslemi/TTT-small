"""
TriMul “outgoing” forward pass (AlphaFold3).

The costly part is the 4‑D einsum  
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  

which can be seen as a batched GEMM:
    (B·D, N, N) @ (B·D, N, N)ᵀ   →   (B·D, N, N)

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(they are cheap) and fuse the N³ contraction into a single Triton kernel.
The kernel works on *half* precision (fp16) for the two operands, accumulates
in fp32 (via `tl.dot`) and writes the result as fp32.  This reduces
memory traffic and improves arithmetic throughput on Hopper (sm_90).

The kernel is tiled with BLOCK_M = BLOCK_N = 64 and BLOCK_K = 32 and
uses 8 warps (good default for fp16 on H200).  Edge tiles are masked.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ   where left/right have shape (M,N,N)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16   [M,N,N]
    right_ptr,         # *fp16   [M,N,N]
    out_ptr,           # *fp32   [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows   (i)
    pid_n = tl.program_id(1)          # tile over cols   (j)
    pid_z = tl.program_id(2)          # matrix index (batch * hidden)

    # ------------------------------------------------------------------- #
    # Offsets for the current tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers for the two operands
    # left  : (M, N, K)   accessed as (z, i, k)
    # right : (M, N, K)   we need the transpose on the last two dims → (z, k, j)
    # ------------------------------------------------------------------- #
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main GEMM loop over K
    # ------------------------------------------------------------------- #
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → fp32

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Store the result
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                     + offs_m[:, None] * stride_on \
                     + offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) with a fused Triton GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]        (bool/float) or None
            - weights      : dict of torch.Tensor containing all linear /
                              layer‑norm parameters
            - config       : {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) → still in fp32
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Rearrange to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------- #
    # (B,N,N,hidden) → (B, hidden, N, N) → (M, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim

    left_reshape = left_perm.view(M, N, N).to(torch.float16)
    right_reshape = right_perm.view(M, N, N).to(torch.float16)

    # output tensor stays fp32 (as required by the rest of the model)
    out_reshape = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 6. Strides (in elements) for the kernel
    # ------------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # ------------------------------------------------------------------- #
    # 7. Launch the tiled GEMM kernel
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        M                         # separate matrix for each (b, head)
    )

    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # tuned for fp16 on Hopper/H200
    )

    # ------------------------------------------------------------------- #
    # 8. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 9. Output LayerNorm, gating and final linear (dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
