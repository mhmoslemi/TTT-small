"""
TriMul “outgoing” forward pass with a fused Triton GEMM.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply where each head (hidden dimension) is an
independent (N×N) @ (N×N)ᵀ.  The kernel operates on tensors of shape
(M, N, N) with M = B * hidden_dim and computes the product in FP32
(accumulation) while loading/storing in the original datatype (FP16/FP32).
All surrounding LayerNorm / Linear / gating work is kept in PyTorch for
clarity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ
# --------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr,                     # *DTYPE   [M, N, K]  (left)
    b_ptr,                     # *DTYPE   [M, N, K]  (right)
    c_ptr,                     # *DTYPE   [M, N, N]  (out)
    M, N, K,
    stride_am, stride_an, stride_ak,   # left strides
    stride_bm, stride_bn, stride_bk,   # right strides
    stride_cm, stride_cn, stride_ck,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    DTYPE: tl.constexpr,               # load/store dtype (fp16 or fp32)
    ACC_DTYPE: tl.constexpr,           # accumulation dtype (always fp32)
):
    pid_m = tl.program_id(0)            # tile over rows   (i)
    pid_n = tl.program_id(1)            # tile over cols   (j)
    pid_z = tl.program_id(2)            # batch * head index

    # ------------------------------------------------------------------
    # Offsets within the tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the current tile (no transpose on the first two dims)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_z * stride_am \
                    + offs_m[:, None] * stride_an \
                    + offs_k[None, :] * stride_ak

    # Right operand is used transposed on the last two dims:
    # we need B(z, k, j) where original layout is (z, j, k)
    b_ptrs = b_ptr + pid_z * stride_bm \
                    + offs_k[:, None] * stride_bk \
                    + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=ACC_DTYPE)

    # ------------------------------------------------------------------
    # Main GEMM loop over K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_idx in range(0, num_k_tiles):
        # load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
            dtype=DTYPE,
        )
        # load B tile (already the transposed view we need)
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
            dtype=DTYPE,
        )
        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b, out_dtype=ACC_DTYPE)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_cm \
                    + offs_m[:, None] * stride_cn \
                    + offs_n[None, :] * stride_ck
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing) with a fused Triton GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (fp32 or fp16)
            - mask         : torch.Tensor [B, N, N]  (bool/float) or None
            - weights      : dict of tensors (the same keys as the PyTorch model)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (same dtype as input)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – performed in PyTorch (cheap)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Fuse the N³ contraction with Triton
    #    reshape to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------
    # (B,N,N,hidden) -> (B, hidden, N, N) -> (M, N, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_reshape = torch.empty_like(left_reshape, device=device)

    # strides (in elements) needed by the kernel
    stride_am, stride_an, stride_ak = left_reshape.stride()
    stride_bm, stride_bn, stride_bk = right_reshape.stride()
    stride_cm, stride_cn, stride_ck = out_reshape.stride()

    # --------------------------------------------------------------
    # Kernel launch configuration
    # --------------------------------------------------------------
    # Block sizes tuned for Hopper (fp16 loads are 2× faster, fp32 fallback works)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),        # rows (i)
        triton.cdiv(N, BLOCK_N),        # cols (j)
        B * hidden_dim,                 # independent GEMMs
    )

    # Determine load/store dtype for the kernel
    if dtype == torch.float16 or dtype == torch.bfloat16:
        DTYPE = tl.float16 if dtype == torch.float16 else tl.bfloat16
    else:
        DTYPE = tl.float32
    ACC_DTYPE = tl.float32

    _tri_mul_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden_dim,
        N,
        N,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        DTYPE=DTYPE,
        ACC_DTYPE=ACC_DTYPE,
        num_warps=8,          # good for fp16 on Hopper, still fast for fp32
    )

    # reshape back to the original layout [B, N, N, hidden]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
