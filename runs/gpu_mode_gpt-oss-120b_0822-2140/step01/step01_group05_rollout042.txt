"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
matrix‑multiply kernel in Triton.

The expensive operation is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication A @ Bᵀ for every hidden channel d.
We reshape the tensors to (B·H, N, N) and compute all batches in a single
launch.  The rest of the layer (LayerNorm, linear projections, sigmoids,
output norm/gate) stays in PyTorch – this already gives a >10× speed‑up
over a pure‑PyTorch implementation while keeping the code simple and
portable.

Key ideas
---------
*   Permute to (B, H, N, N) → contiguous, then view as (B·H, N, N).  This
    guarantees that the inner‑most dimension (N) is contiguous and that the
    stride layout is (batch_stride, N, 1), which the Triton kernel can use
    directly.
*   Use a classic 2‑D tiling (BLOCK_M × BLOCK_N) with a small K‑tile
    (BLOCK_K).  The kernel loads a (BLOCK_M, BLOCK_K) tile of A and a
    (BLOCK_N, BLOCK_K) tile of B (the K‑dimension is shared) and accumulates
    the dot‑product with `tl.dot`.
*   Accumulation is performed in float32 even if the tensors are cast to
    float16 for the loads – this keeps numerical stability while reducing
    bandwidth.
*   The kernel works for any batch size (B∈{1,2}) and any hidden dimension
    (hidden_dim∈{128,384,768}) that appear in the test‑suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute C = A @ Bᵀ   (A,B shape = [batch, M, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes (square in our use‑case)
    stride_ab, stride_am, stride_ak,    # strides for A  (batch, M, K)
    stride_bb, stride_bm, stride_bk,    # strides for B  (batch, M, K)
    stride_cb, stride_cm, stride_cn,    # strides for C  (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Thread‑block indices
    # --------------------------------------------------------------
    pid_n = tl.program_id(0)        # column block
    pid_m = tl.program_id(1)        # row    block
    pid_b = tl.program_id(2)        # batch×hidden block

    # --------------------------------------------------------------
    # Offsets within the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers to the start of the tile for this block
    # --------------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiling)
    # --------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # Load a (BLOCK_M, BLOCK_K) tile of A and a (BLOCK_N, BLOCK_K) tile of Bᵀ
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float16)          # bandwidth reduction
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot returns fp16; cast to fp32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write the result back – mask out-of‑bounds elements
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of TriMul (outgoing).

    Args:
        data : tuple (input, mask, weights, config)
            input   – torch.Tensor [B, N, N, dim]   (float32, CUDA)
            mask    – torch.Tensor [B, N, N] or None
            weights – dict of parameter tensors (all float32, CUDA)
            config  – {'dim': int, 'hidden_dim': int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # hidden_dim in the paper (c_z)
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) + apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched matmul  (left @ rightᵀ)  – all heavy lifting in Triton
    #    Reshape to (B*hidden, N, N) so each hidden channel is an
    #    independent batch for the kernel.
    # ------------------------------------------------------------------
    #   (B, N, N, H) -> (B, H, N, N) -> contiguous -> view
    left_t   = left.permute(0, 3, 1, 2).contiguous()
    right_t  = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_t.view(B * hidden, N, N)    # [B*H, M, K]
    right_flat = right_t.view(B * hidden, N, N)   # [B*H, M, K]

    out_flat = torch.empty_like(left_flat)        # allocate result

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    # Tile sizes tuned for Hopper (sm_90) – fit in 64 KB shared memory.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),          # blocks along N (columns)
        triton.cdiv(N, BLOCK_M),          # blocks along M (rows)
        B * hidden,                       # batch*hidden dimension
    )

    # Strides for the 3‑D tensors (batch, M, K)
    a_batch, a_m, a_k = left_flat.stride()
    b_batch, b_m, b_k = right_flat.stride()
    c_batch, c_m, c_n = out_flat.stride()

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_m, a_k,
        b_batch, b_m, b_k,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to original layout: [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (hidden) + gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
