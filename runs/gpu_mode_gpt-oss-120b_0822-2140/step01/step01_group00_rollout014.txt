"""
TriMul “outgoing” implementation (AlphaFold3).

The heavy pair‑wise reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is rewritten as a batched matrix multiplication

    C[b,h,i,j] = A[b,h,i,:] @ B[b,h,j,:]ᵀ

with A,B ∈ ℝ^{N×N} for each (batch, head) pair.  A and B are first
produced by linear layers, masked and gated in PyTorch (fast element‑wise
operations).  The GEMM is executed by a custom Triton kernel that
tiles the three dimensions (M,N,K) and accumulates in fp32 while the
inputs are stored as fp16/bf16.  After the kernel the tensor is
permuted back to the original layout and the remaining LayerNorm,
gate and output linear are applied in PyTorch.

The kernel is deliberately simple:
* 3‑D grid – (block‑row, block‑col, batch*head)
* each program computes one (M×N) tile of C for a fixed (b,h)
* tiling parameters (BLOCK_M, BLOCK_N, BLOCK_K) are tuned for
  Hopper GPUs (64‑64‑32 works well for N up to 1024).
* No atomics, no synchronization – each tile is independent.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def trmul_gemm_kernel(
    A_ptr,            # [B, H, N, N]   fp16/bf16
    B_ptr,            # [B, H, N, N]   fp16/bf16 (will be accessed as Bᵀ)
    C_ptr,            # [B, H, N, N]   fp32 output
    BATCH: tl.constexpr,
    HEAD: tl.constexpr,
    N: tl.constexpr,
    stride_ah, stride_ai, stride_ak,  # A strides (b, h, i, k)
    stride_bh, stride_bi, stride_bk,  # B strides (b, h, j, k) – note k is last dim
    stride_ch, stride_ci, stride_ck,  # C strides (b, h, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------
    # program identifiers
    # -------------------------------------------------
    pid_m = tl.program_id(0)                 # block row   (i dimension)
    pid_n = tl.program_id(1)                 # block col   (j dimension)
    pid_bn = tl.program_id(2)                # batch * head index

    batch = pid_bn // HEAD
    head  = pid_bn % HEAD

    # -------------------------------------------------
    # tile offsets
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (when N is not a multiple of BLOCK_*)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------------------------------------------------
    # accumulator
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # loop over K dimension
    # -------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # -----------------------------------------------------------------
        # A tile: shape (BLOCK_M, BLOCK_K)  (i, k)
        # -----------------------------------------------------------------
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptrs = (
            A_ptr
            + batch * stride_ah
            + head  * stride_ai
            + offs_m[:, None] * stride_ak          # i‑stride
            + offs_k[None, :] * stride_ak          # k‑stride (last dim)
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # -----------------------------------------------------------------
        # B tile (transposed): shape (BLOCK_K, BLOCK_N)  (k, j)
        # -----------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + batch * stride_bh
            + head  * stride_bi
            + offs_k[:, None] * stride_bk          # k‑stride (last dim)
            + offs_n[None, :] * stride_bi          # j‑stride (second‑last dim)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # -----------------------------------------------------------------
        # matrix multiply‑accumulate
        # -----------------------------------------------------------------
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # write result
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_ch
        + head  * stride_ci
        + offs_m[:, None] * stride_ck          # i‑stride
        + offs_n[None, :] * stride_ck          # j‑stride (last dim)
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Helper: apply layer‑norm with supplied weight / bias
# --------------------------------------------------------------------------- #
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    return F.layer_norm(
        x,
        normalized_shape=(weight.shape[0],),
        weight=weight,
        bias=bias,
        eps=1e-5,
    )

# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   float32 CUDA tensor
        - mask         : [B, N, N]         float32 (0/1) or None
        - weights      : dict of weight tensors
        - config       : dict with keys
              * dim        – input channel size (C)
              * hidden_dim – projection dimension (H)
              * nomask     – bool, true when mask is all‑ones
    Returns
    -------
    torch.Tensor
        [B, N, N, dim]   float32 CUDA tensor
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1) input layer‑norm
    # ------------------------------------------------------------------- #
    x = _layer_norm(x, weights["norm.weight"], weights["norm.bias"])          # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2) linear projections to hidden_dim
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] → broadcast
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4) gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) core TriMul reduction via Triton GEMM
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape

    # cast to fp16 (or bf16) – the kernel works with either; we use fp16 here
    left_f16  = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_f16 = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    # output buffer (fp32) in the same layout
    out_buf = torch.empty_like(left_f16, dtype=torch.float32)

    # strides for the three tensors (batch, head, i/j, k)
    s_ah, s_ai, s_ak, s_al = left_f16.stride()   # (b, h, i, k)
    s_bh, s_bi, s_bk, s_bl = right_f16.stride()  # (b, h, j, k)
    s_ch, s_ci, s_ck, s_cl = out_buf.stride()    # (b, h, i, j)

    # tiling parameters – good for N up to 1024 on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks in i‑direction
        triton.cdiv(N, BLOCK_N),   # blocks in j‑direction
        B * hidden_dim,            # batch * head
    )

    trmul_gemm_kernel[grid](
        A_ptr=left_f16,
        B_ptr=right_f16,
        C_ptr=out_buf,
        BATCH=B,
        HEAD=hidden_dim,
        N=N,
        stride_ah=s_ah, stride_ai=s_ai, stride_ak=s_ak,
        stride_bh=s_bh, stride_bi=s_bi, stride_bk=s_bk,
        stride_ch=s_ch, stride_ci=s_ci, stride_ck=s_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to original layout [B,N,N,hidden]
    out = out_buf.permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------- #
    # 6) output layer‑norm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7) apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8) final linear projection (hidden_dim → dim)
    # ------------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])

    return out
