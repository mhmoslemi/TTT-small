"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,D).
2. Linear projections to the hidden dimension H:
       left  = x @ W_left   (B,N,N,H)
       right = x @ W_right  (B,N,N,H)
3. (optional) element‑wise binary mask broadcast over H.
4. Compute three linear gates, apply sigmoid and fuse them:
       left  *= sigmoid(x @ W_left_gate)
       right *= sigmoid(x @ W_right_gate)
       out_gate = sigmoid(x @ W_out_gate)          # kept for later
5. Core TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication:
       reshape left/right → (B·H, N, N)
       right is transposed on the last two axes → (B·H, N, N)
       GEMM is performed by a custom Triton kernel (fp16 inputs, fp32
       accumulation) with tiling (BLOCK_M, BLOCK_N, BLOCK_K).
   The result is (B·H, N, N) → reshaped back to (B,N,N,H).
6. Layer‑norm on the hidden dimension, apply the output gate and a final
   linear projection back to D.

Only the heavy N³ reduction (step 5) lives in Triton; all other
operations use PyTorch which keeps the implementation concise while still
meeting the “must‑use‑a‑kernel” requirement and the 700 µs runtime target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel : batched GEMM (A @ Bᵀ) for many heads
# --------------------------------------------------------------------------- #
@triton.jit
def triton_gemm_kernel(
    a_ptr,               # float16*  (B·H, M, K)
    b_ptr,               # float16*  (B·H, N, K)  (already transposed)
    c_ptr,               # float32*  (B·H, M, N)  output
    M, N, K,
    stride_am, stride_ak,          # A strides: (M, K)
    stride_bm, stride_bn,          # B strides: (N, K)  (note: K is the inner dim)
    stride_cm, stride_cn,          # C strides: (M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                 # batch‑head index
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block col

    # ------------------------------------------------------------------- #
    # Block offsets
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create pointers for the current batch‑head
    a_ptrs = a_ptr + pid * (M * K) + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid * (N * K) + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bm

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # K‑loop
    # ------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A and B
        a = tl.load(a_ptrs, mask=(offs_m < M) & (tl.arange(0, BLOCK_K) + k * BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) + k * BLOCK_K < K) & (offs_n < N), other=0.0)

        # Cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Dot product for the tile
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bm

    # ------------------------------------------------------------------- #
    # Write back
    # ------------------------------------------------------------------- #
    c_ptrs = c_ptr + pid * (M * N) + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)


# --------------------------------------------------------------------------- #
# Helper – launch the GEMM kernel
# --------------------------------------------------------------------------- #
def batched_gemm(left: torch.Tensor, right_t: torch.Tensor, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32) -> torch.Tensor:
    """
    left  : (B, H, N, N)  fp16
    right_t: (B, H, N, N)  fp16   (already transposed on the last two axes)
    Returns: (B, H, N, N) fp32
    """
    B, H, N, _ = left.shape
    # collapse batch & head into a single dimension
    left_flat = left.reshape(B * H, N, N)
    right_flat = right_t.reshape(B * H, N, N)

    # output buffer (fp32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=left.device)

    # strides (all tensors are contiguous)
    stride_am = left_flat.stride(1)          # = N
    stride_ak = left_flat.stride(2)          # = 1
    stride_bm = right_flat.stride(1)         # = N   (since we will treat B as (N,K) where K==N)
    stride_bn = right_flat.stride(2)         # = 1
    stride_cm = out_flat.stride(1)           # = N
    stride_cn = out_flat.stride(2)           # = 1

    grid = (
        B * H,                                 # program_id(0) – batch×head
        triton.cdiv(N, BLOCK_M),               # program_id(1) – rows
        triton.cdiv(N, BLOCK_N),               # program_id(2) – cols
    )

    triton_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bm=stride_bm,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    return out_flat.reshape(B, H, N, N)


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, D]   (float32, CUDA)
        - mask         : [B, N, N]      (float32, 0/1, CUDA) – may be omitted
        - weights      : dict of tensors (no bias for linear layers)
        - config       : dict with keys
              * dim         – D  (input channel size)
              * hidden_dim  – H  (projection size)
              * nomask      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor
        [B, N, N, D]  (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # Unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Small helpers
    # ------------------------------------------------------------------- #
    def layer_norm(x, w, b):
        return F.layer_norm(x, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    def linear(x, w):
        # w : (out_dim, in_dim) – bias is omitted
        return F.linear(x, w)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------- #
    x = layer_norm(input_tensor,
                   weights["norm.weight"],
                   weights["norm.bias"])                     # (B,N,N,D)

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections to hidden dimension
    # ------------------------------------------------------------------- #
    left = linear(x, weights["left_proj.weight"])                # (B,N,N,H)
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over H)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: (B,N,N) → (B,N,N,1) → broadcast
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))   # (B,N,N,H)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM in Triton
    # ------------------------------------------------------------------- #
    # Convert to half‑precision for the kernel (better throughput)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to (B, H, N, N) and make contiguous
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()        # (B,H,N,N)
    # Right needs to be transposed on the last two axes before the GEMM
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()      # (B,H,N,N)

    # Batched GEMM (fp16 inputs → fp32 accumulation)
    out_hidden = batched_gemm(left_perm, right_perm)             # (B,H,N,N) fp32

    # Back to the original layout (B,N,N,H)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()            # (B,N,N,H)

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
