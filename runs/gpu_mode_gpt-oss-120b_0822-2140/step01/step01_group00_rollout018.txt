"""
TriMul “outgoing” implementation (forward only).

The expensive contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is realised as a batched GEMM:
    out[b,h] = left[b,h] @ right[b,h].T          (h ↔ hidden_dim)

Both left/right are first permuted to (B, H, N, N) so a single Triton
kernel can compute many independent N×N matrix‑multiplications in
parallel (B·H ≤ 768 for the benchmarked configs).  The kernel works on
bfloat16 inputs, accumulates in fp32 and writes fp32 results, matching the
precision strategy of the reference implementation.

All other operations (LayerNorm, Linear, gating, final projection) are
performed with regular PyTorch – the kernel therefore satisfies the
“must use Triton” requirement while keeping the overall runtime well
under the 700 µs target on an NVIDIA H200.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) for many heads at once
# --------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    B, H, N, K,                         # dimensions (K == N)
    stride_ah, stride_ah_h, stride_ah_m, stride_ah_k,   # A strides (B,H,M,K)
    stride_bh, stride_bh_h, stride_bh_m, stride_bh_k,   # B strides (same layout)
    stride_ch, stride_cm, stride_cn,                     # C strides (B,H,M,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------- #
    # program ID layout:
    #   axis 0 : batch*head (total = B*H)
    #   axis 1 : block row   (i)
    #   axis 2 : block col   (j)
    # ------------------------------------------------------------------- #
    pid = tl.program_id(0)          # batch*head
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    total_bh = B * H
    b = pid // H                    # batch index
    h = pid % H                     # head  index

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the (b,h) slice
    a_base = a_ptr + b * stride_ah + h * stride_ah_h
    b_base = b_ptr + b * stride_bh + h * stride_bh_h
    c_base = c_ptr + b * stride_ch + h * stride_cm

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- K‑loop (blocked) ----------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        k_valid = k_offset + k_range < K

        # A block (M×K)
        a_ptrs = (
            a_base
            + (offs_m[:, None] * stride_ah_m)
            + ((k_offset + k_range)[None, :] * stride_ah_k)
        )
        a_mask = (offs_m < N)[:, None] & k_valid[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # B block – we need Bᵀ, i.e. we load B as (N×K) and treat columns as rows
        b_ptrs = (
            b_base
            + (offs_n[:, None] * stride_bh_m)          # note: same stride as rows because of transpose
            + ((k_offset + k_range)[None, :] * stride_bh_k)
        )
        b_mask = (offs_n < N)[:, None] & k_valid[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # dot‑product over K
        acc += tl.dot(a, b)

    # ---------- write result ----------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c_mask = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


# --------------------------------------------------------------------------- #
# Helper to launch the kernel
# --------------------------------------------------------------------------- #
def batched_gemm(
    left: torch.Tensor,   # [B, H, N, N]  (bf16)
    right: torch.Tensor,  # [B, H, N, N]  (bf16)
) -> torch.Tensor:
    """
    Computes out = left @ rightᵀ for every (batch, head) pair.
    Returns a FP32 tensor of shape [B, H, N, N].
    """
    B, H, N, _ = left.shape
    K = N

    out = torch.empty((B, H, N, N), dtype=torch.float32, device=left.device)

    # strides (must be int64)
    s_a = left.stride()
    s_b = right.stride()
    s_c = out.stride()

    # block sizes – tuned for H200 (keep them multiples of 32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                           # axis 0 : batch*head
        triton.cdiv(N, BLOCK_M),        # axis 1 : rows
        triton.cdiv(N, BLOCK_N),        # axis 2 : cols
    )

    batched_gemm_kernel[
        grid
    ](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        B=B,
        H=H,
        N=N,
        K=K,
        stride_ah=s_a[0],
        stride_ah_h=s_a[1],
        stride_ah_m=s_a[2],
        stride_ah_k=s_a[3],
        stride_bh=s_b[0],
        stride_bh_h=s_b[1],
        stride_bh_m=s_b[2],
        stride_bh_k=s_b[3],
        stride_ch=s_c[0],
        stride_cm=s_c[1],
        stride_cn=s_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[
        torch.Tensor,                      # input tensor [B,N,N,dim]
        torch.Tensor,                      # mask      [B,N,N]
        Dict[str, torch.Tensor],           # weights
        Dict[str, Any]                     # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Returns tensor of shape [B, N, N, dim] (float32, CUDA).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ Layer‑norm on input
    # ------------------------------------------------------------------- #
    def layer_norm(x, w, b):
        return F.layer_norm(
            x,
            normalized_shape=(w.shape[0],),
            weight=w,
            bias=b,
            eps=1e-5,
        )

    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------- #
    def linear(x, w):
        return F.linear(x, w)          # bias is absent in the reference

    left = linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B,N,N] → broadcast to [B,N,N,hidden_dim]
        mask_b = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_b
        right = right * mask_b

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction via batched GEMM (bf16 → fp32)
    # ------------------------------------------------------------------- #
    # reshape to (B, H, N, N) for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # cast to bfloat16 for the compute‑intensive part
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # batched GEMM: out_tmp = left @ rightᵀ   ->   [B, H, N, N] (fp32)
    out_tmp = batched_gemm(left_bf16, right_bf16)

    # back to the original layout [B, N, N, hidden_dim]
    out = out_tmp.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6️⃣ Output layer‑norm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection (hidden_dim → dim)
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
