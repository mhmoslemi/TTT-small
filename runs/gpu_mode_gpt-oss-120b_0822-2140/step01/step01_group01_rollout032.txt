"""
TriMul “outgoing” forward pass (AlphaFold‑3 style)

The expensive part is the 4‑D einsum  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* this is a matrix multiplication  
L = left[b,:, :, d]   (N×N)   and   R = right[b,:, :, d]   (N×N)  

→ out[b,:, :, d] = L @ Rᵀ.

We flatten (batch × hidden) → M and launch a single Triton kernel that
computes M independent GEMMs (N×N)·(N×N)ᵀ.  All other operations (LayerNorm,
linear projections, gates, masking) stay in PyTorch because they are
band‑width‑light compared with the O(N³) work.

The kernel:
* works on fp16 inputs to cut global‑memory traffic,
* accumulates in fp32 (numerically safe),
* uses three compile‑time tile sizes (BLOCK_M/N/K) that give good occupancy
  on Hopper SMs,
* is launched with a 3‑D grid (i‑tile, j‑tile, flattened batch·hidden).

The host‑side `custom_kernel` builds the tensors, applies the linear /
gate / mask logic, calls the Triton GEMM, and finishes the module with the
output LayerNorm and linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel: computes C = A @ Bᵀ for many independent (N,N) matrices
#   A : (M, N, N)   fp16
#   B : (M, N, N)   fp16   (the kernel reads B transposed)
#   C : (M, N, N)   fp32   (accumulation)
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    N, M,                         # N = seq_len, M = batch*hidden
    stride_ah, stride_ai, stride_ak,   # A strides  (M, N, N)
    stride_bh, stride_bj, stride_bk,   # B strides  (M, N, N)
    stride_ch, stride_ci, stride_cj,   # C strides  (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row  (i)
    pid_n = tl.program_id(1)          # block col  (j)
    pid_z = tl.program_id(2)          # flattened (batch * hidden)

    # ----- block offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) slice
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K loop (tile) -------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # B_T[k, j]  shape (BLOCK_K, BLOCK_N)  (original B indexed as B[j, k])
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk     # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj     # j dimension (col of Bᵀ)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # dot product over K dimension
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ----- write back ----------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)   fp16
    returns      : (B, H, N, N)   fp32   where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                                      # flatten batch*hidden

    # flatten to (M, N, N)
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty((M, N, N), dtype=torch.float32, device=left.device)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # tile sizes – empirically good for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch*hidden)
    )

    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy O(N³) contraction is performed by a fused Triton GEMM kernel.
    All other ops (LayerNorm, linear projections, sigmoids, masking) stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections  dim → hidden_dim  (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)        # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional masking
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.to(device).unsqueeze(-1)          # (B, N, N, 1)
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to fp16 and permute to (B, hidden, N, N) for the kernel
    # --------------------------------------------------------------
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # --------------------------------------------------------------
    # 6. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    # --------------------------------------------------------------
    out_fp32 = trimul_gemm(left_fp16, right_fp16)   # (B, hidden, N, N) fp32

    # back to original layout
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 7. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 8. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)       # (dim, hidden)
    out = F.linear(out, to_out_w)                        # (B, N, N, dim)

    return out.to(torch.float32)
