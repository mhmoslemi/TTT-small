```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused Triton GEMM.

The dominant O(N³·H) operation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched GEMM C = A @ Bᵀ where each hidden dimension `d` is a separate batch.
We:
   1. Apply LayerNorm and bias‑free Linear projections in PyTorch.
   2. Apply optional mask and sigmoid gates.
   3. Fuse the contraction into a single Triton kernel that works in FP16
      (FP32 accumulation) and returns FP16; the result is reshaped back,
      LayerNorm‑gated and projected to the original `dim`.

The kernel uses 128×128 tiles (M×N) with a K‑tile of 64, 8 warps and
FP32 accumulation – a good fit for NVIDIA Hopper (SM‑90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B shape [B, M, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (int)
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col)
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where B is logically transposed."""
    pid_batch = tl.program_id(2)               # batch index (B*H)
    pid_m = tl.program_id(0)                   # block row
    pid_n = tl.program_id(1)                   # block col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_az \
                 + offs_m[:, None] * stride_am \
                 + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  We need Bᵀ ⇒ element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                 + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A (M×K) and Bᵀ (K×N)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # tl.dot: FP16×FP16 -> FP32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as FP16 to save bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for FP16 tensors.
    * a, b – shape (B*H, N, N), dtype=torch.float16, contiguous.
    Returns c of identical shape & dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N

    c = torch.empty_like(a)

    # Strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # (batch, N, K) layout
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64          # larger K‑tile improves register reuse

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B_H,                       # batch * hidden_dim
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,          # M
        N,          # N
        N,          # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # optimal for 128×128 tiles on SM90
    )
    return c


# ----------------------------------------------------------------------
# Host‑side entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a fused Triton GEMM.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N] (bool or float)
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys
              "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask application
    # --------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N) -> (B,N,N,1) broadcast over hidden
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of bias‑free linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Rearrange to (B, H, N, N) → (B*H, N, N) and cast to FP16
    # --------------------------------------------------------------
    # (B,N,N,hidden) -> (B,hidden,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM: C = left @ rightᵀ   (still FP16)
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden) and cast to FP32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Post‑projection LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # out_gate shape (B,N,N,hidden)

    # --------------------------------------------------------------
    # 9. Final linear projection back to original `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
