"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We treat each (batch, hidden) pair as an independent GEMM and compute many
of them in one Triton launch.  The kernel works in FP16 (fast on H200) while
accumulating in FP32 to keep numerical quality.  All other operations
(LayerNorm, linear projections, sigmoids, masking) are performed in PyTorch –
they are cheap compared with the O(N³) work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for M = B*hidden independent matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # input / output pointers
    N, M,                               # N = seq_len, M = B*hidden_dim
    stride_ah, stride_ai, stride_ak,    # A strides  (M, N, N)
    stride_bh, stride_bj, stride_bk,    # B strides  (M, N, N)
    stride_ch, stride_ci, stride_cj,    # C strides  (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent (N×N) matrices."""
    pid_m = tl.program_id(0)      # block row   (i)
    pid_n = tl.program_id(1)      # block col   (j)
    pid_z = tl.program_id(2)      # flattened batch*hidden index

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the selected (batch, hidden) slice
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32 (higher precision than the loads)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Tile over the reduction dimension K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i,k]  (BLOCK_M × BLOCK_K)   – FP16
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load Bᵀ[k,j] = B[j,k]  (BLOCK_K × BLOCK_N) – FP16
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj          # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Compute dot product in FP16, cast to FP32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write the result back (FP16 to keep memory bandwidth low)
    # --------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    # store as FP16; the caller will cast to FP32 if needed
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    Input tensors are expected in shape (B, H, N, N) and dtype float16.
    Returns a tensor of the same shape and dtype.
    """
    B, H, N, _ = left.shape
    M = B * H                               # flattened batch*hidden

    # reshape to (M, N, N) – contiguous layout is required for correct strides
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides as int64 (must be passed to the kernel)
    # ------------------------------------------------------------------
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # Tile sizes – tuned for Hopper SMs (128‑way warp + 64KB shared memory)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch * hidden)
    )

    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Most work (LayerNorm, linear layers, gating, masking) stays in PyTorch.
    The O(N³) contraction is executed by a fused Triton kernel (trimul_gemm).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )

    # --------------------------------------------------------------
    # 2. Linear projections: dim → hidden_dim (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the Triton GEMM: (B, H, N, N) & FP16
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to FP16 for the kernel (memory‑bandwidth bound)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Triton GEMM → (B, H, N, N) in FP16
    out_fp16 = trimul_gemm(left_fp16, right_fp16)

    # Back to original layout and dtype
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim, apply out_gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(device))   # (B,N,N,dim)

    return out
