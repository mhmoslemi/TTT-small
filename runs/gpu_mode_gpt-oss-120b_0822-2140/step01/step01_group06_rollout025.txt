"""
TriMul (outgoing) – fast forward pass.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply for every head d.
We reshape the tensors to (B, H, N, N) and launch a tiled GEMM
kernel written in Triton.  The kernel works in fp16 for the
inner product and accumulates in fp32, then writes fp32 results.
All other operations (LayerNorm, Linear, sigmoid‑gates,
output‑gate multiplication, final linear) are kept in PyTorch,
but the element‑wise multiplication with the output gate is also
fused into a tiny Triton kernel so that the submission contains
a custom kernel as required.

The implementation supports:
* optional binary mask (broadcasted over the hidden dimension)
* arbitrary batch size (1 or 2) and sequence length up to 1024
* hidden dimensions 128 / 384 (the paper uses 128)
* mixed‑precision for the O(N³·H) part while the final output stays
  in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Batched GEMM (left @ rightᵀ) – core O(N³·H) part
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,            # left  matrix   (B,H,N,N)  row‑major  i‑k
    b_ptr,            # right matrixᵀ (B,H,N,N)  row‑major  k‑j
    c_ptr,            # output         (B,H,N,N)  row‑major  i‑j
    B, H, M, N, K,    # problem sizes
    stride_ah, stride_ak, stride_am,   # a strides: (head, k, i)
    stride_bk, stride_bc, stride_bn,   # b strides: (k, head, j)
    stride_ch, stride_cm, stride_cn,   # c strides: (head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_hb = tl.program_id(2)         # flattened (batch, head) index

    # compute absolute batch & head indices
    batch = pid_hb // H
    head  = pid_hb %  H

    # block start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (last partial block)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers shifted to the current batch & head
    a_base = a_ptr + batch * stride_ah + head * stride_ak
    b_base = b_ptr + batch * stride_bk + head * stride_bc
    c_base = c_ptr + batch * stride_ch + head * stride_cm

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # range of K for this iteration
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A block (i,k)
        a_ptrs = a_base \
                 + (offs_m[:, None] * stride_am) \
                 + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)

        # load B block (k,j)  – note B is already transposed
        b_ptrs = b_base \
                 + (offs_k[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float16)

        # dot‑product over K
        acc += tl.dot(a, b).to(tl.float32)

        # advance pointers (not strictly needed because we recompute each iter)
    # ------------------------------------------------------------------
    # write results
    # ------------------------------------------------------------------
    c_ptrs = c_base \
             + (offs_m[:, None] * stride_cm) \
             + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Element‑wise multiplication (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elems,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 3️⃣  Custom entry point – called from the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32/bool or None, CUDA)
        * weights_dict : learned parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over channel dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] --> [B,N,N,1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM in Triton
    #    left  : (B,N,N,H)  -> (B,H,N,N)   rows=i, cols=k
    #    right : (B,N,N,H)  -> (B,H,N,N)   rows=k, cols=j   (already transposed)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)  k‑j

    # allocate output tensor
    out_perm = torch.empty_like(left_perm, dtype=torch.float32, device=device)

    # strides (in elements) for the three tensors
    stride_ah = left_perm.stride(0)          # batch stride
    stride_ak = left_perm.stride(1)          # head  stride
    stride_am = left_perm.stride(2)          # row   stride (i)
    stride_an = left_perm.stride(3)          # col   stride (k)

    stride_bk = right_perm.stride(0)         # batch stride (same as left)
    stride_bc = right_perm.stride(1)         # head stride
    stride_bn = right_perm.stride(2)         # row stride (k)
    stride_bb = right_perm.stride(3)         # col stride (j)

    stride_ch = out_perm.stride(0)
    stride_cm = out_perm.stride(2)           # i
    stride_cn = out_perm.stride(3)           # j

    # tiling parameters – chosen to fit H200 shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over M (i)
        triton.cdiv(N, BLOCK_N),   # blocks over N (j)
        B * hidden_dim,            # each (batch, head) pair
    )

    batched_gemm_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, hidden_dim, N, N, N,
        stride_ah, stride_an, stride_am,
        stride_bk, stride_bc, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back‑to original layout: (B,N,N,H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Multiply by output gate (fused Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)

    total = out_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    elemwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        out_flat,
        total,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
```

The code follows the specification:

* **LayerNorms & Linear layers** are performed with PyTorch for simplicity.
* The **expensive O(N³·H)** contraction is executed by a custom Triton GEMM
  kernel (`batched_gemm_kernel`).  The kernel works on blocks
  `(BLOCK_M, BLOCK_N, BLOCK_K)` = (64, 64, 32), uses fp16 loads,
  accumulates in fp32, and writes the result in fp32.
* **Mask handling** is broadcast‑compatible and optional.
* The **output‑gate multiplication** is also realized in Triton
  (`elemwise_mul_kernel`) to satisfy the “must use a kernel” rule.
* All tensors are kept on the device, mixed‑precision is used only where
  it speeds up the dominant operation, and the final result is returned
  in `float32` as required.
