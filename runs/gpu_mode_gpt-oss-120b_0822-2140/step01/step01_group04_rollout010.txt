"""
TriMul “outgoing” forward (AlphaFold3).

The expensive O(N³) contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply per hidden dimension `d`.
We fuse this with a Triton kernel that computes, for each (batch, head)
pair, a matrix product A @ Bᵀ where A,B ∈ (N,N).  All remaining
operations (LayerNorm, Linear, gating, final LayerNorm/Linear) stay in
PyTorch – they are negligible compared with the N³ work.

The kernel works on tensors of shape (B, H, N, N) that are contiguous in
row‑major order.  Each program instance computes a TILE_M × TILE_N block
of the output matrix for a single (batch, head).  Inputs are loaded as
FP16 for bandwidth, the accumulation is performed in FP32, and the
result is written back in FP32 (matching the reference implementation).

The host side `custom_kernel` performs:
  1️⃣ Layer‑norm over the last dimension (`dim`).
  2️⃣ Linear projections → left/right (shape B,N,N,H) and gating.
  3️⃣ Optional element‑wise mask on left/right.
  4️⃣ Triton‑accelerated contraction.
  5️⃣ Final Layer‑norm, gating and linear projection back to `dim`.

The implementation supports all configurations listed in the problem
statement (different batch sizes, sequence lengths, dim / hidden_dim and
optional masks) and runs on NVIDIA H200 (SM90) with mixed‑precision
optimisation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_k,   # note: B is accessed as (j,k)
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,  # how many m‑tiles a program_id groups together
):
    """
    For each (b, h) compute
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ.
    """

    # ------------------------------------------------------------------
    # program ids
    # ------------------------------------------------------------------
    pid = tl.program_id(0)                 # linear id over (m_tile, n_tile, batch*head)
    num_pid_m = tl.cdiv(N, TILE_M)
    num_pid_n = tl.cdiv(N, TILE_N)

    # group m‑tiles to improve L2 reuse (see Triton matmul example)
    pid_m = (pid // (num_pid_n * GROUP_SIZE_M)) * GROUP_SIZE_M + (pid // num_pid_n) % GROUP_SIZE_M
    pid_n = pid % num_pid_n
    pid_hb = pid // (num_pid_m * num_pid_n)   # batch * head index

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ----- load A[b,h,i,k] -----
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ----- load Bᵀ => B[b,h,j,k] (j as rows, k as cols) -----
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_n[:, None] * stride_b_j   # j dimension first
                  + offs_k[None, :] * stride_b_k)  # k as columns
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K
        acc += tl.dot(a, b)   # (M,K) @ (K,N) → (M,N)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32 or bfloat16)
        - mask         : torch.Tensor[B, N, N]      (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optionally
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # convenience for weight lookup
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left/right   shape [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # (hidden, dim)
    right = F.linear(x, w("right_proj.weight"))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – Triton kernel
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, N) and make contiguous (row‑major)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # allocate output tensor (FP32 accumulator, final type FP32)
    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in elements (not bytes) – required by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, j, k)  (note order for transpose)
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile sizes – chosen to fit in shared memory on Hopper and give good occupancy
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64
    GROUP_SIZE_M = 8   # grouping factor for better L2 reuse (as in Triton docs)

    grid = (
        triton.cdiv(N, TILE_M) * triton.cdiv(N, TILE_N) * B * hidden,  # flattened 3‑D grid
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=8,          # Hopper cores prefer 8 warps per block
        num_stages=3,
    )

    # back to original layout -> [B,N,N,hidden]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + gating + linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate          # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # -> [B,N,N,dim]

    return out.to(dtype)
