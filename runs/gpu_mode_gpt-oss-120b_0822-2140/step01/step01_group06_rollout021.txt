"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The expensive contraction  

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply over the sequence dimension N.  
We reshape the tensors to (B, H, N, N) where H = hidden_dim and launch a
single Triton kernel that computes a tiled GEMM for every (batch, head)
pair, accumulating in FP32 and writing FP32 results.  All other
operations (LayerNorm, Linear, Sigmoid, final Linear) are performed with
PyTorch – the requirement “must use a kernel” is satisfied by the GEMM
kernel.

The kernel works for any B∈{1,2}, N∈{128…1024}, hidden_dim∈{128,384,768}
and respects optional masks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel for the TriMul contraction (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_gemm_kernel(
    left_ptr,                # [B, H, N, N]   (i,k)
    right_ptr,               # [B, H, N, N]   (j,k) – we will read it as (k,j)
    out_ptr,                 # [B, H, N, N]   (i,j)
    B, H, N,                 # problem sizes (constants for the kernel)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,   # tile size in i dimension
    BLOCK_N: tl.constexpr,   # tile size in j dimension
    BLOCK_K: tl.constexpr,   # tile size in k dimension
):
    """Compute out = left @ rightᵀ for each (batch, head) pair."""
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1️⃣ Map linear program id to (batch, head, tile_i, tile_j)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    programs_per_batch_head = num_blocks_m * num_blocks_n

    bh = pid // programs_per_batch_head               # batch*head id
    prog_in_bh = pid % programs_per_batch_head        # which tile in that (b,h)

    block_i = prog_in_bh // num_blocks_n
    block_j = prog_in_bh %  num_blocks_n

    batch = bh // H
    head  = bh % H

    # --------------------------------------------------------------
    # 2️⃣ Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the tile (k dimension will be advanced in the loop)
    a_base = left_ptr + batch * stride_l_b + head * stride_l_h \
                     + offs_i[:, None] * stride_l_i

    # right is accessed as (k, j) so that the inner‑product becomes a dot‑product.
    # We therefore compute a base pointer that steps first over k then over j.
    # The j‑stride is stride_r_j (second‑last dimension of right).
    b_base = right_ptr + batch * stride_r_b + head * stride_r_h

    # --------------------------------------------------------------
    # 3️⃣ Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4️⃣ Loop over K in tiles
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        # mask for the k dimension
        k_valid = (k_offset + tl.arange(0, BLOCK_K)) < N

        # ----------- load A (left) : shape (M, K) ----------
        a_ptrs = a_base + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & k_valid[None, :],
                    other=0.0,
                    dtype=tl.float16)
        a = a.to(tl.float32)

        # ----------- load B (rightᵀ) : shape (K, N) ----------
        # right is stored as (batch, head, j, k).  To obtain (k, j) we
        # stride first over k then over j.
        b_ptrs = b_base \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = b.to(tl.float32)

        # ----------- dot product accumulation ----------
        acc += tl.dot(a, b)          # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # 5️⃣ Store the result tile
    # --------------------------------------------------------------
    out_ptrs = out_ptr + batch * stride_o_b + head * stride_o_h \
                         + offs_i[:, None] * stride_o_i \
                         + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (AlphaFold‑3 version).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : torch.Tensor [B, N, N] or None
        * weights_dict : model parameters (LayerNorm, Linear, etc.)
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack arguments
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # called hidden_dim in the paper
    device = inp.device
    B, N, _, _ = inp.shape

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last channel (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(inp,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the channel dimension)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.unsqueeze(-1).to(x.dtype)            # [B, N, N, 1]
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Rearrange to (B, H, N, N) for the GEMM kernel
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # allocate output tensor
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 6️⃣ Launch Triton GEMM kernel
    # --------------------------------------------------------------
    # block sizes – tuned for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements) for each tensor
    s_l_b, s_l_h, s_l_i, s_l_k = left.stride()
    s_r_b, s_r_h, s_r_j, s_r_k = right.stride()
    s_o_b, s_o_h, s_o_i, s_o_j = out.stride()

    # number of program instances
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_progs = B * hidden * num_blocks_m * num_blocks_n
    grid = (total_progs,)

    triton_trimul_gemm_kernel[grid](
        left, right, out,
        B, hidden, N,
        s_l_b, s_l_h, s_l_i, s_l_k,
        s_r_b, s_r_h, s_r_j, s_r_k,
        s_o_b, s_o_h, s_o_i, s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7️⃣ Rearrange back to original layout [B, N, N, H]
    # --------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Post‑norm on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # --------------------------------------------------------------
    # 9️⃣ Multiply by output gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
