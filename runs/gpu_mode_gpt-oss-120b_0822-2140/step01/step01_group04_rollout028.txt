"""
TriMul “outgoing” kernel (AlphaFold3).

The expensive part of the module is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d` (treated as a
separate “head”).  This implementation moves only that contraction to a
custom Triton kernel while keeping the cheap LayerNorm/Linear/gating
operations in PyTorch.

Data layout:
    left  : (B, N, N, H)  -> permuted to (B, H, N, N)   # axes (batch, head, i, k)
    right : (B, N, N, H)  -> permuted to (B, H, N, N)   # axes (batch, head, i, k)

The kernel computes, for each (b, h),

    out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]

i.e. a regular matrix‑multiply of an (N×N) matrix with the transpose of another
(N×N) matrix.  FP16 is used for the loads (bandwidth) and FP32 for the
accumulator to preserve accuracy.  The result is written back in FP32 to match
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)^T
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    N, H,                               # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    where A and B have shape (N, N).  This is A @ Bᵀ.
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # tile offsets
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A: (TILE_M, TILE_K)   = left[b, h, i, k]
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ: we need B[j, k] -> load as (TILE_K, TILE_N)
        #   stride_b_j corresponds to the original k‑stride,
        #   stride_b_i corresponds to the original i‑stride (used for j here).
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k dimension
                  + offs_n[None, :] * stride_b_i)  # j dimension
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # store result
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32 or bf16)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be all‑ones
        - weights      : dict of parameter tensors
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # shortcut to fetch weights (they are already on the correct device)
    w = lambda name: weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the feature dimension ``dim``
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B, N, N, H]
    right = F.linear(x, w("right_proj.weight"))  # [B, N, N, H]

    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Heavy contraction using Triton
    # ------------------------------------------------------------------
    # bring the hidden dimension to the second axis for easy batching
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]  (i,k)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]  (i,k)

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in *elements* (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, k)
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile sizes – tuned for Hopper (SM90)
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,                     # 8 warps give good occupancy on SM90
    )

    # back to the original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 4. Final LayerNorm + gating + projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcasting over hidden dim
    out = F.linear(out, w("to_out.weight"))  # [B, N, N, dim]

    return out.to(dtype)
