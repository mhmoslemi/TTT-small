"""
TriMul “outgoing” – fully fused implementation.

The expensive reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is reformulated as a batched matrix‑multiply
    out[b,h] = left[b,h] @ right[b,h].T
with   left , right : (B, H, N, N)   (H = hidden_dim)

The kernel performs a tiled GEMM for every (batch, head) pair,
accumulating in fp32 while loading the inputs in bfloat16
(bandwidth‑saving on H200).  All remaining ops (LayerNorm, linear,
gating, final projection) stay in PyTorch – they are cheap compared
to the N³ reduction.

The kernel is launched with a 3‑D grid:
    pid_m : block‑row of the output matrix (i)
    pid_n : block‑col of the output matrix (j)
    pid_z : combined batch‑head index (b * H + h)

Both the forward pass and the kernel work for any
B ∈ {1,2}, N ∈ {128,…,1024}, dim ∈ {128,384},
hidden_dim ∈ {128,384,768}.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for TriMul
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    a_ptr,          # left  : (B, H, N, N)  bfloat16
    b_ptr,          # right : (B, H, N, N)  bfloat16   (will be used transposed)
    c_ptr,          # out   : (B, H, N, N)  float32
    # matrix sizes
    M, N, K,        # all equal to seq_len
    # strides (in elements, not bytes)
    stride_am, stride_ak, stride_ah,          # a : (batch*head, row, col)
    stride_bm, stride_bn, stride_bk, stride_bh,  # b : (batch*head, row, col) – row/col will be swapped
    stride_cm, stride_cn, stride_ch,          # c : (batch*head, row, col)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_z = tl.program_id(2)          # batch * head index

    # --------------------------------------------------------------- #
    #   Offsets for this block
    # --------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------------- #
    #   Pointers that already include the batch*head offset
    # ---------------------------------------------------------------- #
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------- #
    #   Loop over K dimension (the summed axis)
    # ---------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left : A[M, K]   (row i, col k)
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.bfloat16,
        ).to(tl.float32)

        # right : we need B[K, N] (col k, row j) because we will compute
        # A @ B where B is the transpose of the original right tensor.
        b = tl.load(
            b_batch_ptr
            + offs_k[:, None] * stride_bk   # K‑dimension is now the leading dim
            + offs_n[None, :] * stride_bn,  # N‑dimension is the trailing dim
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.bfloat16,
        ).to(tl.float32)

        # dot over K -> result (M, N)
        acc += tl.dot(a, b)

        # advance pointers for next K‑tile
        a_batch_ptr += BLOCK_K * stride_ak
        b_batch_ptr += BLOCK_K * stride_bk

    # ---------------------------------------------------------------- #
    #   Write out the block
    # ---------------------------------------------------------------- #
    c = acc.to(tl.float32)
    tl.store(
        c_batch_ptr
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Helper functions ----------------------------------------------------------- #
# --------------------------------------------------------------------------- #
def _layer_norm(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """LayerNorm with bias & weight supplied as tensors."""
    return F.layer_norm(
        x,
        normalized_shape=(w.shape[0],),
        weight=w,
        bias=b,
        eps=1e-5,
    )


def _linear(x: torch.Tensor, w: torch.Tensor) -> torch.Tensor:
    """Linear projection without bias."""
    return F.linear(x, w)


# --------------------------------------------------------------------------- #
# Main entry point ----------------------------------------------------------- #
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]    (float32, on CUDA)
        - mask         : [B, N, N]         (float32 0/1, may be omitted)
        - weights      : dict of tensors (see comment below)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------- #
    x = _layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # --------------------------------------------------------------- #
    # 2️⃣ Project to hidden dimension (no bias)
    # --------------------------------------------------------------- #
    left = _linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = _linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------- #
    if not nomask:
        # mask: (B,N,N) -> (B,N,N,1) -> broadcast
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of a linear projection)
    # --------------------------------------------------------------- #
    left_gate = torch.sigmoid(_linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(_linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(_linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – tiled GEMM in Triton
    # --------------------------------------------------------------- #
    # Cast to bfloat16 to reduce traffic; accumulation stays fp32
    left_bf16 = left.to(torch.bfloat16).contiguous()
    right_bf16 = right.to(torch.bfloat16).contiguous()

    # Permute to (B, H, N, N) for the GEMM formulation
    left_perm = left_bf16.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right_bf16.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    B, H, N_seq, _ = left_perm.shape
    M = N_seq
    N_dim = N_seq
    K = N_seq

    # strides (in elements)
    stride_al = left_perm.stride(0)   # batch*head stride
    stride_ah = left_perm.stride(1)   # head stride (should be N*N)
    stride_am = left_perm.stride(2)   # row stride (N)
    stride_ak = left_perm.stride(3)   # col stride (1)

    stride_bl = right_perm.stride(0)
    stride_bh = right_perm.stride(1)
    stride_bm = right_perm.stride(2)   # row (i) stride – will be treated as “col” after transpose
    stride_bn = right_perm.stride(3)   # col stride
    # after we load as (K,N) we need stride for K (col) and N (row) – we reuse
    stride_bk = stride_bm   # because K corresponds to original row dimension in right_perm
    # (see kernel loading pattern)

    # output tensor
    out_perm = torch.empty_like(left_perm, dtype=torch.float32)  # (B,H,N,N)

    stride_cl = out_perm.stride(0)
    stride_ch = out_perm.stride(1)
    stride_cm = out_perm.stride(2)
    stride_cn = out_perm.stride(3)

    # Tile sizes – tuned for Hopper (SM90) 128‑bit loads
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N_dim, BLOCK_N),
        B * H,
    )

    tri_mul_gemm_kernel[grid](
        a_ptr=left_perm,
        b_ptr=right_perm,
        c_ptr=out_perm,
        M=M,
        N=N_dim,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ah=stride_ah,
        stride_bm=stride_bm,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_bh=stride_bh,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_ch=stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Permute back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # --------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden → hidden)
    # --------------------------------------------------------------- #
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # --------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # --------------------------------------------------------------- #
    out = out * out_gate

    # --------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # --------------------------------------------------------------- #
    out = _linear(out, weights["to_out.weight"])

    return out
