```python
"""
TriMul “outgoing” forward pass (no gradients).

The dominant work is the triple‑product
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be rewritten as a batched matrix‑multiply
    C = A @ Bᵀ   with   A,B ∈ ℝ^{B·H × N × N}.

We:
  1. Apply the first LayerNorm and the four linear projections (PyTorch).
  2. Mask + sigmoid gates (PyTorch).
  3. Reshape to (B·H, N, N) and launch a Triton kernel that computes many
     independent GEMMs: each block computes a TILE_M×TILE_N tile of C.
     The kernel works in float16 for bandwidth, accumulates in float32,
     and stores back in float16.
  4. Reshape back, apply the second LayerNorm, the output gate and the final
     linear projection (PyTorch).

Only the O(N³·H) contraction is performed in Triton; the rest stays in
high‑level PyTorch for simplicity and because it is not a bottleneck.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_az, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # B strides (batch, row, col)
    stride_cz, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices."""
    pid_batch = tl.program_id(2)           # batch dimension (B·H)
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col

    # ----- block offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointers to the start of the current tile --------------------
    a_ptrs = a_ptr + pid_batch * stride_az \
                + offs_m[:, None] * stride_am \
                + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K). For Bᵀ we need element (k,n) → (row=k, col=n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                + offs_n[None, :] * stride_bn

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main K‑loop ---------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # load Bᵀ tile
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # dot does FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----- write‑back ----------------------------------------------------
    c = acc.to(tl.float16)                     # compress back to FP16
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for tensors of shape (B·H, N, N) in float16.
    Returns C in the same shape / dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N

    c = torch.empty_like(a)

    # strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # (batch, N, K) layout
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – 128×128 tiles fit nicely in shared memory on Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # keeps register pressure modest

    grid = (
        triton.cdiv(N, BLOCK_M),          # program_id(0) – rows
        triton.cdiv(N, BLOCK_N),          # program_id(1) – cols
        B_H,                              # program_id(2) – batch
    )
    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,      # empirically good for 128×128 on sm_90
    )
    return c


# ----------------------------------------------------------------------
# Host‑side entry point required by the problem
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   torch.float32
        mask         : [B, N, N]        torch.float32 / bool
        weights      : dict of torch.float32 tensors
        config       : dict with keys
                       - "dim" (int)
                       - "hidden_dim" (int)
                       - "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm over the channel dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)    # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gating sigmoid(linear(...))
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate          # (B,N,N,hidden_dim)
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ reshape for the batched GEMM: (B, H, N, N) → (B*H, N, N)
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6️⃣ Heavy contraction in Triton
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # (B*H, N, N) fp16

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, hidden_dim) and cast to fp32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # out_gate shape (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
