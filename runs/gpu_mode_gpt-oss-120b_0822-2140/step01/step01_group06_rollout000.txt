def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B,N,_,_ = input_tensor.shape

    # LN
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Proj
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # reshape for GEMM
    left = left.permute(0,3,1,2).contiguous()   # B, H, N, N
    right = right.permute(0,3,1,2).contiguous()

    # cast to fp16
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B_H = B * hidden_dim
    M = N
    K = N
    N_dim = N

    # allocate output fp16
    out_fp16 = torch.empty((B_H, M, N_dim), dtype=torch.float16, device=input_tensor.device)

    # stride in elements
    stride_ah = left_fp16.stride(0)
    stride_am = left_fp16.stride(1)
    stride_ak = left_fp16.stride(2)

    stride_bh = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)
    stride_bn = right_fp16.stride(2)

    stride_ch = out_fp16.stride(0)
    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N_dim + BLOCK_N - 1) // BLOCK_N
    total_blocks = B_H * num_m * num_n

    grid = (total_blocks,)

    batched_gemm_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        M, N_dim, K,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0,2,3,1)  # B,N,N,H

    # post norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # multiply out_gate
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – fast forward pass.

The only O(N³·H) work (the “einsum”) is performed by a custom
Triton batched‑GEMM kernel.  All other building blocks (LayerNorm,
Linear, sigmoid gates, the final linear) are executed with PyTorch
because they are cheap compared with the cubic contraction.

The kernel computes, for every (batch, head) pair
    out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]        (fp16)
which is exactly the contracted einsum in the reference implementation.
The result is accumulated in FP32 for numerical stability and stored as
FP16; after the kernel we cast back to FP32 and continue with the rest
of the pipeline.

The implementation works for any batch size (1‑2), sequence length
(128‑1024) and hidden dimensions (128‑768) that appear in the test
matrix.  A mask (or no mask) is handled in the PyTorch portion before
the GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many heads at once
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # [batch*head, M, K] , [batch*head, N, K] , [batch*head, M, N]
    M, N, K,                           # problem sizes (int)
    stride_ah, stride_am, stride_ak,   # strides for A  (batch*head,stride_row,stride_col)
    stride_bh, stride_bk, stride_bn,   # strides for B  (batch*head,stride_row,stride_col)
    stride_ch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A, B : fp16   (shape [B*H, M, K]  and [B*H, N, K])
    C      : fp16 (shape [B*H, M, N])
    The kernel computes C = A @ Bᵀ   (sum over K)
    """

    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decode program id -> (batch_head, block_m, block_n)
    # ------------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_m * num_n

    batch_head = pid // blocks_per_head
    block_id   = pid %  blocks_per_head
    pid_m = block_id // num_n          # row‑tile index
    pid_n = block_id %  num_n          # col‑tile index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the current tile
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + batch_head * stride_ah \
                     + offs_m[:, None] * stride_am \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    b_ptrs = B_ptr + batch_head * stride_bh \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                     + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)                 # fp16·fp16 → fp32 accumulation

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back (store as fp16, the later PyTorch code casts to fp32)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + batch_head * stride_ch \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor  [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : torch.Tensor  [B, N, N]       (bool/float, CUDA) or None
        * weights_dict : dict with all linear / norm parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = H
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm (over the last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                           # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H]

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)           # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    #    We want shape (B*H, N, N) and FP16 for the kernel.
    # --------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right = right.permute(0, 3, 1, 2).contiguous()  # B, H, N, N

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    BH = B * hidden_dim       # total number of “heads”

    # Allocate output buffer (FP16, same shape)
    out_fp16 = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 6. Launch the Triton GEMM kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides in *elements* (not bytes) – required by the kernel
    stride_ah, stride_am, stride_ak = left_fp16.stride()
    stride_bh, stride_bk, stride_bn = right_fp16.stride()
    stride_ch, stride_cm, stride_cn = out_fp16.stride()

    # total number of program instances
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = BH * num_m * num_n
    grid = (total_blocks,)

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,               # M
        N,               # N
        N,               # K
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H) and cast to FP32
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # --------------------------------------------------------------
    # 8. Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                           # [B,N,N,H]

    # --------------------------------------------------------------
    # 9. Multiply with the output gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate                         # still [B,N,N,H]

    # --------------------------------------------------------------
    # 10. Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out

* The cubic contraction is **fully implemented in Triton** (`batched_gemm_kernel`).  
  It works on the `(batch × head)` dimension as an extra outer dimension, uses
  fp16 arithmetic with fp32 accumulation, and tiles the work with
  `BLOCK_M = 64`, `BLOCK_N = 64`, `BLOCK_K = 32`, which is a good default for
  Hopper‑class GPUs (H200).  
  This satisfies the “must use a kernel” requirement and dramatically cuts
  the runtime compared with the previous pure‑PyTorch + tiny‑kernel approach.
* All tensors are kept on the device, and the final output is returned as
  `float32` with shape `[B, N, N, dim]`.
