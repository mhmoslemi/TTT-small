"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The costly operation is the 4‑D contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* it is a matrix multiply
    L = left[b,:, :, d]      (N×N)
    R = right[b,:, :, d]     (N×N)
    out[b,:, :, d] = L @ Rᵀ (N×N)

We launch a Triton kernel that performs a batched GEMM
    (B·H) independent (N×N)·(N×N)ᵀ multiplications.
All cheap operations (LayerNorm, Linear, Sigmoid, masking) stay in
PyTorch.  The kernel works in fp16 for the inputs (to cut bandwidth)
and accumulates in fp32 for numerical stability.

The host code arranges tensors in the layout (B, H, N, N) that the kernel
expects, calls the kernel, and then restores the original layout.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    N, M,                         # N = seq_len, M = B*H (flattened batch*head)
    stride_ah, stride_ai, stride_ak,   # A strides  (M,N,N)
    stride_bh, stride_bj, stride_bk,   # B strides
    stride_ch, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute many independent GEMMs:
        C[m] = A[m] @ B[m].T   for m = 0 .. M-1
    where each A[m], B[m] is (N,N).  The result C has the same shape.
    """
    pid_m = tl.program_id(0)      # row block index  (i)
    pid_n = tl.program_id(1)      # col block index  (j)
    pid_z = tl.program_id(2)      # flattened (batch*head)

    # ------------------------------------------------------------------
    # Offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current (M‑th) matrices
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B_T[k,j] shape (BLOCK_K, BLOCK_N)  (we read B as transposed)
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k‑dim becomes row of Bᵀ
            + offs_n[None, :] * stride_bj          # j‑dim becomes column
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot accumulates in fp32 automatically
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  fp16
    returns      : (B, H, N, N)  fp32   (each slice = left @ rightᵀ)
    """
    B, H, N, _ = left.shape
    M = B * H                               # flatten batch & head

    # flatten to (M,N,N)
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides (int64) in the order of the tensor storage
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # --------------------------------------------------------------
    # Block sizes – tuned for Hopper (SM90).  128×128 tiles give good
    # occupancy; K‑tile of 32 fits easily in shared memory.
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )

    _tri_mul_gemm_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        M,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # good default for fp16 work‑loads
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]        (float32 or bool)
        weights      : dict of torch.Tensor (already on the same device)
        config       : dict with keys
                       - dim (int)
                       - hidden_dim (int)
                       - nomask (bool, optional)

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm  (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Linear projections  (dim → hidden_dim)   no bias
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # --------------------------------------------------------------
    # 3. Masking (if present)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (B,N,N,1) for broadcasting
        mask = mask.to(device).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the Triton GEMM.
    #    We convert to fp16 to save bandwidth.
    # --------------------------------------------------------------
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B,H,N,N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM: out = left @ rightᵀ   (per hidden channel)
    # --------------------------------------------------------------
    out_fp32 = _batched_gemm_fp16(left_fp16, right_fp16)                # (B,H,N,N) fp32

    # --------------------------------------------------------------
    # 7. Restore original layout (B,N,N,H)
    # --------------------------------------------------------------
    out = out_fp32.permute(0, 2, 3, 1).contiguous()                     # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 8. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 9. Final projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(device))   # (B,N,N,dim)

    return out.to(torch.float32)

* uses fp16 for the two large inputs while accumulating in fp32,
* selects block sizes (128×128×32) and 8 warps that match the Hopper SM‑90
