## Triton‑accelerated **outgoing** TriMul

```python
"""
custom_kernel – Triton‑accelerated forward pass of the TriMul “outgoing”
operator (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor (shape B×N×N×C).
2. Linear projections → left / right tensors (B×N×N×H)  (no bias).
3. Optional mask broadcast on the hidden dimension.
4. Gated‑linear transform (sigmoid gates) applied element‑wise.
5. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM `A @ Bᵀ` for each (batch, head) pair.
   Implemented in Triton with fp16 loads and fp32 accumulation.
6. Post‑processing: layer‑norm on the hidden dimension, multiply by out‑gate,
   final linear projection back to the original channel dimension C.

The kernel works for batch ≤ 2, sequence length up to 1024 and hidden_dim = 128
(as required by the benchmark).  The final output is float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _matmul_T_kernel(
    A_ptr,                      # *float16   (batch, M, K)
    B_ptr,                      # *float16   (batch, N, K)   (read as K×N)
    C_ptr,                      # *float32   (batch, M, N)
    M, N, K,                    # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note Bk = stride of col, Bn = stride of row)
    stride_Cb, stride_Cm, stride_​Cn,  # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                 # batch * head index
    pid_m = tl.program_id(0)                 # block row
    pid_n = tl.program_id(1)                 # block column

    # ------------------------------------------------------------------
    # Block start offsets (in element units)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for this block
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk      # K dimension
        + offs_n[None, :] * stride_Bn      # N dimension
    )

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)

        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)               # (M×K)·(K×N) → (M×N)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights      : dict of parameter tensors (all float32)
        - config       : dict with keys
              * "dim"          – input channel dimension C
              * "hidden_dim"   – hidden dimension H
              * "nomask"       – True → ignore mask, False → apply mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = (
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction  (batched GEMM A @ Bᵀ per head)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)  (contiguous for Triton)
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel for bandwidth, fp32 accumulator → fp32 output
    left_fp16 = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For the transposed view Bᵀ we need:
    stride_Bk = stride_Bc                     # step along K (original column)
    stride_Bn = stride_Br                     # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # Tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H,                     # batch * head
    )

    _matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim C

    return out
```
