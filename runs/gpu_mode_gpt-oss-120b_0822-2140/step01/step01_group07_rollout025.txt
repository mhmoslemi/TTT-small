"""
TriMul “outgoing” – fast forward only.

The dominant work is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply
    (B·H, N, N)  @  (B·H, N, N).T
with H = hidden_dim.  We fuse this GEMM into a single Triton kernel.
All surrounding LayerNorm / Linear / gating logic stays in PyTorch (highly
optimised) while the N³ contraction runs in float16 memory bandwidth with
float32 accumulation.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024 and
hidden_dim ≤ 384.  It requires that the input tensors are contiguous and in
row‑major order (the host code performs the required permutes/contiguates).

The final output is float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  out = left @ rightᵀ
#   left , right : (M, N, K)   with K == N
#   out          : (M, N, N)
#   M = B * hidden_dim
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16  [M,N,K]
    right_ptr,         # *fp16  [M,N,K]
    out_ptr,           # *fp32  [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows   (i)
    pid_n = tl.program_id(1)          # tile over cols   (j)
    pid_z = tl.program_id(2)          # different matrix (batch*head)

    # ------------------------------------------------------------------- #
    # offsets for this tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # pointers – note that right is accessed transposed on the last two axes
    # ------------------------------------------------------------------- #
    # left(z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # rightᵀ(z, k, j)  -> we read (z, k, j) from the original layout
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # main loop over K dimension
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # load fp16, mask out‑of‑bounds elements with 0
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)        # (BM, BK)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)        # (BK, BN)

        acc += tl.dot(a, b)                       # (BM, BN)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # store result (still fp32)
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a fused Triton GEMM.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
            mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
            weights      : dict of torch.Tensor, exact keys used in the reference
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor: shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1️⃣ LayerNorm on the input (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # [B, N, N, dim]

    # --------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias) → [B, N, N, hidden_dim]
    # --------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hd)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid) – still in FP32
    # --------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5️⃣ Fuse the N³ contraction via Triton
    #    reshape to (M, N, N) with M = B * hidden_dim
    # --------------------------------------------------------------- #
    # (B,N,N,hd) -> (B,hd,N,N)
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 1, 2).contiguous()

    M = B * hidden_dim
    left_ = left_.view(M, N, N)      # (M,N,N)
    right_ = right_.view(M, N, N)

    # Cast to fp16 for bandwidth‑limited load, keep a fp32 buffer for output
    left_fp16 = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    out_fp32 = torch.empty_like(left_, dtype=torch.float32, device=device)  # (M,N,N)

    # --------------------------------------------------------------- #
    #  launch parameters – tuned for Hopper (sm_90)
    # --------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows (i)
        triton.cdiv(N, BLOCK_N),      # cols (j)
        M                             # number of independent GEMMs
    )

    # strides in *elements* (not bytes)
    stride_lm, stride_ln, stride_lk = left_fp16.stride()
    stride_rm, stride_rn, stride_rk = right_fp16.stride()
    stride_om, stride_on, stride_ok = out_fp32.stride()

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # more warps → better occupancy on H200
    )

    # --------------------------------------------------------------- #
    # 6️⃣ Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------- #
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 7️⃣ Output LayerNorm → gating → final linear
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
