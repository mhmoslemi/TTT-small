"""
TriMul “outgoing” implementation (forward only).

The expensive part is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply of shape
    (B·H, N, N)  @  (B·H, N, N)ᵀ   where H = hidden_dim.

We reshape the tensors to (M, N, N)   (M = B*H) and launch a
single Triton kernel that performs a tiled GEMM for every
matrix in the batch.  The kernel works in **float16** for the
loads (full‑precision weights are kept in fp32) and accumulates
in **float32**, therefore it can use Hopper tensor‑cores while
still producing a fp32 result.

All LayerNorm / Linear / gating logic stays in PyTorch – only the
N³ contraction is off‑loaded to Triton.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ
#   left  : (M, N, K)  (K == N)
#   right : (M, N, K)
#   out   : (M, N, N)
# --------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # *fp16  [M,N,K]
    right_ptr,         # *fp16  [M,N,K]
    out_ptr,           # *fp32  [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides (elements)
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows  (i)
    pid_n = tl.program_id(1)          # tile over cols  (j)
    pid_z = tl.program_id(2)          # which matrix in the batch (M)

    # -----------------------------------------------------------------
    # Offsets for this tile
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # Pointers (z dimension is folded into the leading stride)
    # -----------------------------------------------------------------
    # left  is (M,N,K)   we need  (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # right is accessed transposed on the last two dims:
    # we need Bᵀ(z, k, j)  ->  (z, k, j)
    b_ptrs = right_ptr + pid_z * stride_rm \
                       + offs_k[:, None] * stride_rk \
                       + offs_n[None, :] * stride_rn

    # -----------------------------------------------------------------
    # Accumulator in fp32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Main K‑loop
    # -----------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)          # fp16 load
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)          # fp16 load

        # dot works on fp16, we cast to fp32 after the dot
        acc += tl.dot(a, b).to(tl.float32)

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # -----------------------------------------------------------------
    # Write result (fp32)
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor : torch.Tensor, shape (B, N, N, C)  (float32)
        - mask         : torch.Tensor or None, shape (B, N, N) (bool/float)
        - weights      : dict of torch.nn.Parameter tensors (fp32)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape (B, N, N, C) (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, C = x.shape
    H = cfg["hidden_dim"]          # hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (C)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    # Keep fp32 for the projection, then cast to fp16 for the GEMM.
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Mask (if present) – broadcast over the hidden dimension
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Fuse the N³ contraction with Triton
    #     Reshape to (M, N, N) where M = B * H
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N) -> (M,N,N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    M = B * H

    left_reshape  = left_t.view(M, N, N).to(torch.float16)   # fp16 for GEMM
    right_reshape = right_t.view(M, N, N).to(torch.float16)

    out_reshape = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Strides (in elements) – required by Triton
    # --------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # --------------------------------------------------------------
    # Launch configuration – tuned for Hopper (fp16 tensor cores)
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64                     # K‑tiles that map to a WMMA tile

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows i
        triton.cdiv(N, BLOCK_N),      # cols j
        M                             # one independent matrix per (b,head)
    )

    tri_mul_outgoing_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps ≈ 256 threads → good occupancy on sm90
        num_ctas=1
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to (B,N,N,H)
    # --------------------------------------------------------------
    out = out_reshape.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Final LayerNorm, gating and output projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,C)

    return out
