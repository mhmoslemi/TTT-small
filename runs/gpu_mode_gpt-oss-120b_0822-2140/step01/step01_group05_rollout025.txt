## Triton‑accelerated “outgoing” TriMul

```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused
batched GEMM kernel written in Triton.

The heavy work is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

After permuting to (B, H, N, N) this becomes a batched matrix
multiply (B·H, N, N) @ (B·H, N, N)ᵀ.
The kernel runs in fp16 (Tensor‑cores) while accumulating in fp32
to retain the original fp32 output precision.
All other ops (LayerNorm, linear, gating) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B·H, N, N) @ (B·H, N, N)^T  → (B·H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,               # tensor pointers
    M, N, K,                           # matrix sizes (M=N=N, K=N)
    stride_ab, stride_am, stride_ak,   # strides of A [batch, row, col]
    stride_bb, stride_bm, stride_bk,   # strides of B [batch, row, col]
    stride_cb, stride_cm, stride_cn,   # strides of C [batch, row, col]
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)               # batch * head index
    pid_m = tl.program_id(1)               # tile row
    pid_n = tl.program_id(2)               # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the current tile
    # A : (batch, M, K)  – left[..., i, k, d]
    # B : (batch, N, K)  – right[..., j, k, d]   (note: we use the
    #                     second dimension as the output column index,
    #                     the third as the reduction dimension)
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_ab
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bb
        + offs_n[None, :] * stride_bm   # row = j (output column)
        + offs_k[:, None] * stride_bk   # col = k (reduction)
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        # a, b are fp16 – tl.dot promotes to fp32 internally
        acc += tl.dot(a, b)

        # move to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = (
        C_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32 or bool, may be None)
        - weights      : dict of parameter tensors (float32, CUDA)
        - config       : {'dim': int, 'hidden_dim': int}
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → [B, N, N, hidden]
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Mask (if present) – broadcast on hidden dim
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – broadcast on hidden dim
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ   (per hidden channel)
    #    We treat (B, hidden) as a single batch dimension.
    # --------------------------------------------------------------
    #   left  : [B, N, N, hidden] -> [B, hidden, N, N]
    #   right : [B, N, N, hidden] -> [B, hidden, N, N]
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Cast to fp16 for Tensor‑core acceleration (keep a fp32 copy for the
    # final result – the kernel accumulates in fp32).
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Flatten batch*head dimension
    lhs = left_fp16.view(B * hidden, N, N)
    rhs = right_fp16.view(B * hidden, N, N)

    out_fp32 = torch.empty_like(lhs, dtype=torch.float32)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # program id 0 = batch*head
        triton.cdiv(N, BLOCK_M),        # program id 1 = tile‑row
        triton.cdiv(N, BLOCK_N),        # program id 2 = tile‑col
    )

    # strides (batch, row, col) for the flattened tensors
    a_batch, a_row, a_col = lhs.stride()
    b_batch, b_row, b_col = rhs.stride()
    c_batch, c_row, c_col = out_fp32.stride()

    batched_gemm_outgoing_kernel[grid](
        lhs,
        rhs,
        out_fp32,
        N,                # M
        N,                # N
        N,                # K (reduction dim)
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden) + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
