"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]            (N×N)
    R = right[b,:, :, d]           (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We flatten the batch and hidden dimensions:
    M = B * hidden_dim
and compute *M* independent GEMMs of shape (N×N)·(N×N)ᵀ with a tiled
Triton kernel.  All tensors are kept in fp16 inside the kernel – the
accumulator is fp32 to preserve precision – and the final result is
converted back to fp32.  The remaining LayerNorm / Linear / Sigmoid
operations are cheap and stay in PyTorch.

The kernel is deliberately simple: it loads tiles of A and Bᵀ, performs
`tl.dot` (which internally uses tensor‑core instructions on Hopper) and
writes the result.  Block sizes (128,128,32) have been chosen to give
high occupancy on an H200 while keeping register pressure modest.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel_fp16(
    a_ptr, b_ptr, c_ptr,          # pointers
    N, M,                         # N = seq_len, M = B*hidden_dim
    stride_ah, stride_ai, stride_ak,   # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,   # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,   # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM (A @ Bᵀ) where A,B are stored as (M,N,N) in fp16."""
    pid_m = tl.program_id(0)          # block row   i
    pid_n = tl.program_id(1)          # block col   j
    pid_z = tl.program_id(2)          # flattened batch*head

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current (M‑th) matrix pair
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]  shape (BLOCK_M, BLOCK_K)  (fp16)
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Bᵀ[k,j]  shape (BLOCK_K, BLOCK_N)  (fp16)
        # Original B is indexed as B[j,k]; we load it transposed.
        b_ptrs = (
            b_batch_ptr
            + offs_k[:, None] * stride_bk      # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj      # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Dot product -> fp32 accumulation
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write back C[i,j]  (fp32)
    c_ptrs = (
        c_batch_ptr
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  (float16)
    returns       : (B, H, N, N)  (float32) where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                                     # flattened batch*head

    # reshape to (M, N, N)
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32)

    # strides (int64)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – chosen for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # pid_m
        triton.cdiv(N, BLOCK_N),      # pid_n
        M,                            # pid_z
    )
    trimul_gemm_kernel_fp16[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    All cheap ops (LayerNorm, Linear, Sigmoid, masking) are done in PyTorch.
    The O(N³) contraction is performed by a fused fp16 Triton GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over last dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections: dim -> hidden_dim (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)      # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Apply mask if present (mask shape: B,N,N)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Layout expected by kernel: (B, H, N, N) in fp16.
    # --------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp32 = trimul_gemm_fp16(left_fp16, right_fp16)  # (B, H, N, N) fp32

    # Back to original layout (B,N,N,hidden)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
