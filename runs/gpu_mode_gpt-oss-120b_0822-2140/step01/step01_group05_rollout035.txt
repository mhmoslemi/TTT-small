"""
TriMul “outgoing” (AlphaFold‑3) – forward only.

The expensive part is the 3‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* this is a matrix multiplication
    (N×N) @ (N×N)ᵀ .
We fuse *all* hidden channels of a batch into one large batched GEMM and
implement it with a tuned Triton kernel (fp16 inputs, fp32 accumulation).

All cheap ops (LayerNorm, Linear, gating, final Linear) stay in PyTorch so
that the kernel stays simple and portable.
The kernel works for any batch size ∈{1,2}, any sequence length up to 1024,
and any hidden dimension (the tests use 128).  The output is returned in
float32, exactly matching the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel: C = A @ Bᵀ   (A,B are fp16, acc is fp32)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,                 # fp16 pointer, shape (B*H, M, K)
    B_ptr,                 # fp16 pointer, shape (B*H, N, K)   (the K‑dim is last)
    C_ptr,                 # fp32 pointer, shape (B*H, M, N)
    M, N, K,               # matrix sizes (int)
    stride_Ab, stride_Am, stride_Ak,   # strides for A  (batch, row, col)
    stride_Bb, stride_Bn, stride_Bk,   # strides for B  (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,             # work‑grouping for better L2 reuse
):
    # ------------------------------------------------------------------
    # 1) program ids
    # ------------------------------------------------------------------
    pid = tl.program_id(0)                # linear id over the 2‑D tile grid
    batch_id = tl.program_id(1)           # (B*H) – each hidden channel is a batch

    # number of tiles
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # --------------------------------------------------------------
    # 2) 2‑D tile coordinates (pid_m, pid_n) with grouping
    # --------------------------------------------------------------
    group_id = pid // (GROUP_M * num_pid_n)
    first_pid_m = group_id * GROUP_M
    pid_m = first_pid_m + (pid % GROUP_M)
    pid_n = pid % num_pid_n

    # bounds
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 3) pointers to the start of the tile
    # ------------------------------------------------------------------
    a_ptrs = (A_ptr + batch_id * stride_Ab
              + offs_m[:, None] * stride_Am
              + tl.arange(0, BLOCK_K)[None, :] * stride_Ak)

    b_ptrs = (B_ptr + batch_id * stride_Bb
              + offs_n[None, :] * stride_Bn
              + tl.arange(0, BLOCK_K)[:, None] * stride_Bk)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) main loop over K tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a tile of A and B (fp16 → promote to fp32 automatically)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K),
                    other=0.0).to(tl.float32)

        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K - k * BLOCK_K)[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # dot‑product over the K dimension
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # 5) write back
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + batch_id * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor[bs, N, N] or None
        - weights      : dict of all linear / LN parameters (float32, CUDA)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # = hidden_dim in the paper
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → shape [B, N, N, hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if any).  Mask is broadcast on the hidden dimension.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – keep everything in float32 for stability
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM
    #
    #    left  : [B, N, N, H] → (B, H, N, N) → (B*H, N, N)
    #    right : [B, N, N, H] → (B, H, N, N) → transpose K↔N → (B*H, N, N)
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()         # (B, H, N, N)

    # right must be transposed on its last two axes (K ↔ N) for the
    # A @ Bᵀ formulation.
    right = right.transpose(-2, -1).contiguous()

    # flatten the (B, H) dimensions – each hidden channel is treated as an
    # independent batch for the GEMM kernel.
    left  = left.view(B * hidden, N, N)
    right = right.view(B * hidden, N, N)

    # ------------------------------------------------------------------
    # 6) Cast to fp16 for the GEMM (the kernel accumulates in fp32)
    # ------------------------------------------------------------------
    left_fp16  = left.half()
    right_fp16 = right.half()
    out_fp32   = torch.empty((B * hidden, N, N), dtype=torch.float32,
                            device=left.device)

    # ------------------------------------------------------------------
    # 7) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_M = 8        # typical value from Triton matmul examples

    grid = (
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),   # 1‑D grid for (m,n) tiles
        B * hidden                                            # batch*hidden dimension
    )

    # strides (batch, row, col)
    a_batch, a_row, a_col = left_fp16.stride()
    b_batch, b_row, b_col = right_fp16.stride()
    c_batch, c_row, c_col = out_fp32.stride()

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N,               # M
        N,               # N
        N,               # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output layer‑norm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
