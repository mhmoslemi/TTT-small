"""
TriMul “outgoing” (AlphaFold‑3) – fast Triton‑fused batched matrix multiply.

*   LayerNorm, linear projections and gating stay in PyTorch (they are cheap).
*   The expensive contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply (A @ Bᵀ) where each hidden channel d is
    treated as an independent batch element.  We launch a single Triton kernel
    that processes all (B·hidden) copies simultaneously.
*   Computation inside the kernel is performed in fp16 (Tensor‑Core friendly)
    with a fp32 accumulator for numerical stability; the result is written
    back as fp16 and immediately cast to fp32 for the remaining PyTorch steps.
*   Block sizes (64×64 tiles, K‑tile 32) were chosen to fit H200 shared‑memory
    limits while keeping high occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (A @ Bᵀ) with fp16 inputs, fp32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers (fp16)
    M, N, K,                      # matrix dimensions (all int64)
    stride_ab, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bb, stride_bm, stride_bk,   # strides for B (batch, row, col) – B is read transposed
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch*hidden index
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for A and B (B is accessed transposed)
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (with tiling)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles; mask out‑of‑bounds elements
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Tensor‑core dot (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

        # Advance K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write result (store as fp16, convert from fp32 accumulator)
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of parameter tensors (all on CUDA)
        - config       : dict with keys 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and reshape for batched matmul
    #    (B, N, N, hidden) -> (B*hidden, N, N)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, hidden, N, N) layout is convenient for stride calculation
    left_t = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_t = right_fp16.permute(0, 3, 1, 2).contiguous()

    left_flat  = left_t.view(B * hidden, N, N)   # [B*H, N, N]
    right_flat = right_t.view(B * hidden, N, N)

    out_flat_fp16 = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                     # batch*hidden dimension (program_id 0)
        triton.cdiv(N, BLOCK_M),       # rows (program_id 1)
        triton.cdiv(N, BLOCK_N),       # cols (program_id 2)
    )

    # Strides (batch, row, col) for each tensor
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat_fp16.stride()

    batched_matmul_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat_fp16,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden) and promote to fp32
    # ------------------------------------------------------------------
    out = out_flat_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
