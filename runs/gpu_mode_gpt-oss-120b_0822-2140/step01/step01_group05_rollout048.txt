"""
TriMul “outgoing” (AlphaFold‑3) forward pass with a fused
batched‑gemm kernel written in Triton.

*   A = left   = (B, H, N, N)   –‑ linear projection + mask + gate
*   B = right  = (B, H, N, N)   –‑ linear projection + mask + gate
*   out[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]          (A @ Bᵀ)

The expensive N³·H work is performed by the
`batched_gemm_kernel`.  All other ops (LayerNorm, Linear,
sigmoid‑gating, final LayerNorm + Linear) are kept in PyTorch
because they are already highly optimized by cuBLAS/cuDNN.
The kernel works with fp16 inputs (to exploit tensor‑cores) while
accumulating in fp32 for numerical stability; the final result is
cast back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  C = A @ Bᵀ   where A,B ∈ (B·H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,               # *fp16   (B*H, N, N)
    B_ptr,               # *fp16   (B*H, N, N)
    C_ptr,               # *fp16   (B*H, N, N)
    M, N, K,             # matrix dimensions (all = seq_len)
    stride_Ab, stride_Am, stride_Ak,   # strides for A  (batch, row, col)
    stride_Bb, stride_Bm, stride_Bk,   # strides for B  (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program works on a TILE_M × TILE_N output sub‑matrix for a
    particular batch‑*‑head (pid_b).  The K‑dimension is tiled with
    BLOCK_K and accumulated in fp32.
    """
    pid_b = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds loads
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    #  Pointers for the current tile (note the transpose for B)
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )                                    # (M_tile, K_tile)

    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_n[None, :] * stride_Bm      # <-- rows are the N‑dim (because we need Bᵀ)
        + offs_k[:, None] * stride_Bk
    )                                    # (K_tile, N_tile)

    # ------------------------------------------------------------------
    #  Accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over K‑tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K)[None, :],
                    other=0.0)                     # fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16

        # compute fp16 dot, cast to fp32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

        # advance K‑tile pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    #  Write result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input   : torch.Tensor   [B, N, N, dim]          (float32, CUDA)
        mask    : torch.Tensor   [B, N, N]  or None      (bool/float32)
        weights : dict of torch.Tensor parameters, all on CUDA
        config  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor   [B, N, N, dim]   (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # also called hidden_dim in the paper
    B, N, _, _ = input_tensor.shape        # input shape = [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) + apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    #    shape -> (B*hidden, N, N) and cast to fp16 for the kernel
    # ------------------------------------------------------------------
    #   left : [B, N, N, hidden] --> [B, hidden, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left.view(B * hidden, N, N)
    right_flat = right.view(B * hidden, N, N)

    # cast to fp16 (tensor‑cores)
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton launch configuration
    # ------------------------------------------------------------------
    # these block sizes were chosen to give good occupancy on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                 # program_id(0)  : batch*head
        triton.cdiv(N, BLOCK_M),    # program_id(1)  : rows
        triton.cdiv(N, BLOCK_N),    # program_id(2)  : cols
    )

    # strides (batch, row, col) for each tensor
    a_batch, a_row, a_col = left_fp16.stride()
    b_batch, b_row, b_col = right_fp16.stride()
    c_batch, c_row, c_col = out_fp16.stride()

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,               # M
        N,               # N
        N,               # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Convert back to fp32 and reshape to original layout
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (hidden) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
