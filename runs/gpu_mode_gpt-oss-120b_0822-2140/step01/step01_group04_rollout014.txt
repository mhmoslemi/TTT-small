"""
TriMul “outgoing” kernel (AlphaFold3) – Triton + PyTorch

The dominant cost is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply (per hidden dimension `d`) between a
(N,N) tensor and the transpose of another (N,N) tensor.  The kernel below
computes for every (batch, head) pair:

    C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]   (i.e. A @ Bᵀ)

Inputs `A` and `B` are cast to FP16 for bandwidth, accumulated in FP32,
and written back in FP32 (the reference implementation uses FP32 throughout).
All cheap operations (LayerNorm, linear projections, gating, final
LayerNorm/Linear) stay in PyTorch.

The kernel is tiled with (TILE_M, TILE_N, TILE_K) = (64,64,64) and
uses 8 warps – a good trade‑off for Hopper/RTX‑H200.  Masking is applied
outside the kernel (by zero‑ing the projected tensors).  The final
output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: (B, H, N, N) @ (B, H, N, N)ᵀ  →  (B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # raw pointers
    N, H,                                    # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    For each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ where both operands are (N, N) matrices.
    """
    pid_m = tl.program_id(0)          # tile row  (i)
    pid_n = tl.program_id(1)          # tile col  (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ----------- tile offsets ----------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ----- A : (TILE_M, TILE_K) -----
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_i[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ----- Bᵀ : we need B[j,k] → load as (TILE_K, TILE_N) -----
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k dimension first
                  + offs_j[None, :] * stride_b_i)  # j dimension second
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K
        acc += tl.dot(a, b)          # (M,K) @ (K,N) → (M,N)

    # ---------- store ----------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_i[:, None] * stride_c_i
              + offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (bool or float)
        - weights      : dict of torch.Tensor containing all parameters
        - config       : dict with at least ``dim`` and ``hidden_dim``.
          Optional key ``nomask`` (True → ignore mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype torch.float32.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helpers to fetch weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------
    def w(name): return weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) -> shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # (dim, hidden_dim) weight transposed internally
    right = F.linear(x, w("right_proj.weight"))

    # optional masking (mask is [B,N,N])
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction via Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for coalesced loads
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # allocate output tensor (FP32 accumulator)
    out_t = torch.empty_like(left_t, dtype=torch.float32, device=device)  # [B, H, N, N]

    # strides in element units (not bytes)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j)   we will index transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # launch configuration
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64
    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,                     # Hopper has many cores – 8 warps works well
    )

    # back to the original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Final LayerNorm + gating + linear back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                         # broadcast over hidden_dim
    out = F.linear(out, w("to_out.weight"))      # -> [B,N,N,dim]

    # ensure float32 output (the reference uses float32)
    return out.to(torch.float32)
