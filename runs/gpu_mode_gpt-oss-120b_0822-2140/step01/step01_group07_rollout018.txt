"""
TriMul “outgoing” (AlphaFold‑3) forward pass.

The costly contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply of (B·H, N, N) with the transpose of the second
operand.  We fuse this GEMM into a Triton kernel that works in fp16 (to reduce
memory traffic) while accumulating in fp32 for numeric stability.

All surrounding operations (LayerNorm, Linear, gating) stay in PyTorch – they
are cheap compared with the N³ term.  The kernel is launched with a 2‑D tiling
over the output matrix and a third dimension that indexes the independent
matrices (B·hidden_dim).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM   C = A @ Bᵀ
#   A, B : (M, N, K)   (K == N)
#   C    : (M, N, N)
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # matrix id  (batch * hidden)

    # --------------------------------------------------------------------- #
    # Offsets & masks for the tile
    # --------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------------- #
    # Pointers for the current tile
    #   A : (M,N,K)      index (z, i, k)
    #   B : (M,N,K)ᵀ    we need Bᵀ → (z, k, j)
    # --------------------------------------------------------------------- #
    a_ptrs = a_ptr + pid_z * stride_am \
                    + offs_m[:, None] * stride_an \
                    + offs_k[None, :] * stride_ak

    b_ptrs = b_ptr + pid_z * stride_bm \
                    + offs_k[:, None] * stride_bk \
                    + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # Main K‑loop
    # --------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot produces fp32 accumulation automatically
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------------- #
    # Store result
    # --------------------------------------------------------------------- #
    c_ptrs = c_ptr + pid_z * stride_cm \
                    + offs_m[:, None] * stride_cn \
                    + offs_n[None, :] * stride_ck

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) with a fused Triton GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   torch.float32
            - mask         : [B, N, N]        torch.bool / float (0/1) or None
            - weights      : dict of torch tensors (see code)
            - config       : {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) – also in float32
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Cast to fp16 for the GEMM, then reshape to (M, N, N)
    # ------------------------------------------------------------------- #
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B,N,N,hidden) -> (B,hidden,N,N) -> (M,N,N) where M = B*hidden
    left_tile  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_tile = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    out_tile = torch.empty((B * hidden, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 6. Launch Triton GEMM
    # ------------------------------------------------------------------- #
    stride_am, stride_an, stride_ak = left_tile.stride()
    stride_bm, stride_bn, stride_bk = right_tile.stride()
    stride_cm, stride_cn, stride_ck = out_tile.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden                        # independent matrices
    )

    _tri_mul_gemm_kernel[grid](
        left_tile,
        right_tile,
        out_tile,
        B * hidden, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                     # good for fp16 on Hopper
    )

    # ------------------------------------------------------------------- #
    # 7. Reshape back to (B,N,N,hidden)
    # ------------------------------------------------------------------- #
    out = out_tile.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 8. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B,N,N,dim]

    return out
