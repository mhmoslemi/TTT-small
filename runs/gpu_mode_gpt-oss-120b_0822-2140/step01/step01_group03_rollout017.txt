```python
"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input (C‑dim = dim) using the supplied weights.
2. Linear projections → left / right   (shape B×N×N×H, H = hidden_dim).
3. Optional mask (broadcast over H).
4. Compute three sigmoid gates and apply them element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM per hidden head:
       left_slice (N×N) @ right_sliceᵀ (N×N)
   The kernel loads fp16, accumulates in fp32, and writes fp32.
6. Post‑processing: LayerNorm + out‑gate + final linear back to dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel : C = A @ Bᵀ   (A,B : [batch*head, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                      # *float16   (batch*head, M, K)
    B_ptr,                      # *float16   (batch*head, N, K)   (read as K×N)
    C_ptr,                      # *float32   (batch*head, M, N)
    M, N, K,                    # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (B is stored as (batch, N, K))
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C[b,m,n] = Σ_k A[b,m,k] * B[b,n,k]   (i.e. B is transposed)
    for every batch*head index `b`.  Accumulates in fp32.
    """
    pid_b = tl.program_id(2)            # batch*head index
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block column

    # ---- tile coordinates ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- masks for out‑of‑bounds ----
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- pointers for the current tile ----
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- main K‑loop ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, apply masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # (M×K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (K×N)

        acc += tl.dot(a, b)                                # (M×N)

        # advance pointers for next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---- write result ----
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        - mask         : torch.Tensor, shape [B, N, N]       (bool / float)
        - weights      : dict of torch.Tensor (float32)
        - config       : dict with keys
            * dim          – input channel dimension C
            * hidden_dim   – projection dimension H
            * nomask       – True if mask should be ignored (default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ convenience aliases for weight tensors
    # --------------------------------------------------------------
    # LayerNorm on the input
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    # Gates (no bias)
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # Post‑processing
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]               # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections
    # --------------------------------------------------------------
    # `F.linear` interprets the last dimension as "in_features"
    left = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask_b = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_b
        right = right * mask_b

    # --------------------------------------------------------------
    # 6️⃣ Gates (sigmoid) – also linear without bias
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Core contraction – batched GEMM (left @ rightᵀ) per hidden head
    # --------------------------------------------------------------
    B, N, _, H = left.shape                # H == hidden_dim
    # reshape to (B*H, N, N)  – each GEMM works on a separate batch*head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # fp16 inside the kernel for bandwidth, keep fp32 accumulator
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides in *elements* (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For B we need stride along K (original column) and along N (original row)
    stride_Bk = stride_Bc    # step in K dimension (inner)
    stride_Bn = stride_Br    # step in N dimension (outer)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # Tiling parameters – tuned for Hopper (SM90)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * hidden
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ Post‑processing: LayerNorm → out‑gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim (C)

    return out
```
