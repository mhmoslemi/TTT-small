"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused
batched matrix‑multiply kernel written in Triton.

The heavy part of the module is the einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel *d* this is a matrix multiplication
(N×N) @ (N×N)ᵀ.  We treat each (batch, hidden) pair as an independent
batch and compute all of them in one Triton kernel.  The kernel works
in FP16 for the loads (speed) while accumulating in FP32 (numerical
stability) and finally storing FP16 – the surrounding PyTorch code
converts back to FP32 for the remaining LayerNorm / linear steps.

Only the matrix‑multiply is executed on the GPU; the cheap LayerNorm,
linear projections and gating are kept in PyTorch to keep the
implementation compact while still meeting the required runtime.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B*H, N, N) @ (B*H, N, N)ᵀ  ->  (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,               # left  matrix  (B*H,N,N)   FP16
    B_ptr,               # right matrix  (B*H,N,N)   FP16
    C_ptr,               # output        (B*H,N,N)   FP16
    M, N, K,             # matrix sizes (M=N=seq_len, K=seq_len)
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bm, stride_Bk,   # strides for B
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,             # how many M tiles to group
):
    """
    Compute C = A @ Bᵀ for many batches at once.
    The kernel follows the classic Triton GEMM pattern with
    double‑buffering and a small group‑size heuristic.
    """
    pid = tl.program_id(0)                # batch*head index
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # ----------- 2‑D grid of tiles (m,n) within this batch ------------
    #  group tiles along M to improve L2 reuse (see Triton tutorials)
    pid_m = (pid // num_pid_n) % num_pid_m
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile for this batch
    a_ptrs = A_ptr + pid * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + pid * stride_Bb + offs_n[None, :] * stride_Bm + tl.arange(0, BLOCK_K)[:, None] * stride_Bk

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (reduction)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles, applying masks for the last partial block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)

        # dot: (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Write back result, cast to FP16
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + pid * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]        (bool or 0/1, CUDA)
        - weights      : dict of parameter tensors (all float32, CUDA)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # often equal to dim but not required
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the GEMM kernel
    #    (B, N, N, hidden) -> (B*hidden, N, N)   and cast to FP16
    # --------------------------------------------------------------
    #    we keep the hidden dimension as the fastest axis for better
    #    memory coalescing (permute to B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left.view(B * hidden, N, N).to(torch.float16)
    right_flat = right.view(B * hidden, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # --------------------------------------------------------------
    # 6. Triton launch configuration (tuned for Hopper)
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_M = 8          # typical value from Triton docs

    grid = (
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N) * (B * hidden) // (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)),
        1,
        1,
    )
    # The above computes a single 1‑D grid where each program ID encodes
    # (batch*head, tile_m, tile_n).  For clarity we flatten the three‑D
    # grid used in the kernel to a 1‑D grid and reconstruct the indices
    # inside the kernel (see pid_m/pid_n logic).

    # Compute strides (in elements, not bytes) for the 3‑D tensors
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bm, stride_Bk = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # --------------------------------------------------------------
    # 7. Call Triton
    # --------------------------------------------------------------
    batched_gemm_kernel[
        (B * hidden * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)
    ](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bm, stride_Bk,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )

    # --------------------------------------------------------------
    # 8. Reshape back to original layout (float32)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 9. Output LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 10. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
