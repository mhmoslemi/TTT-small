"""
TriMul “outgoing” kernel (AlphaFold3)

The only cubic‑cost operation in TriMul is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`:
    (B, H, N, N) @ (B, H, N, N)ᵀ   →   (B, H, N, N)

The implementation below
* performs all cheap ops (LayerNorm, Linear, gating) in PyTorch,
* reshapes the tensors to (B, H, N, N) and launches a Triton kernel that
  computes the batched mat‑mul with FP16 loads and FP32 accumulation,
* keeps the final LayerNorm, gating and projection in PyTorch.

The Triton kernel tiles the output with (TILE_M, TILE_N) and iterates over
K in blocks of TILE_K.  FP32 accumulation is required for numerical
stability; the result is written back as FP32.  The kernel works for any
batch size ≤2, any sequence length up to 1024 and hidden dimensions that
fit in memory (the tests use 128‑ or 384‑dim `dim` and 128‑dim
`hidden_dim`).

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# --------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem dimensions
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B are (N, N) matrices)
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # batch * head index

    batch = pid_hb // H
    head  = pid_hb % H

    # ----------------------------------------------------------
    # tile offsets
    # ----------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # loop over K dimension
    # ----------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A : (TILE_M, TILE_K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[j, k] -> load as (TILE_K, TILE_N)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dim
                  + offs_n[None, :] * stride_b_i)  # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (M,K)@(K,N) -> (M,N)

    # ----------------------------------------------------------
    # write back
    # ----------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator using a Triton kernel for
    the cubic‑cost contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]  (float32 or bf16)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------
    # unpack arguments
    # ------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32          # final dtype required by the spec

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # helper to fetch weight tensors (they are already on the correct device)
    w = lambda name: weights[name]

    eps = 1e-6

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )                               # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) -> hidden dimension
    # --------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B, N, N, H]
    right = F.linear(x, w("right_proj.weight"))  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Mask (if present)
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcastable to [..., 1]
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gates (sigmoid of linear maps, no bias)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy cubic contraction → Triton kernel
    #    reshape to (B, H, N, N) and make contiguous
    # --------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # output buffer
    out_t = torch.empty_like(left_t, dtype=dtype)     # [B, H, N, N]

    # strides in elements (not bytes) – required by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)   (we will load transposed)
    c_s = out_t.stride()    # (batch, head, i, j)

    # --------------------------------------------------------------
    # launch parameters – tuned for Hopper (sm_90)
    # --------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),
        triton.cdiv(N, TILE_N),
        B * hidden_dim,
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,          # 4 warps → 128 threads per block, good for Hopper
    )

    # back to original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                   # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B, N, N, dim]

    return out.to(dtype)
