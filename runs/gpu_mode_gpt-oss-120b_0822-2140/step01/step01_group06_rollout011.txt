"""
TriMul (outgoing) – fast Triton implementation

The heavy O(N³·H) contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is expressed as a batched GEMM for each (batch, head) pair:
    left   : (B, H, N, N)     ← (M, K)
    rightᵀ : (B, H, N, N)     ← (N, K)   (the K‑dimension is shared)

A custom Triton kernel computes the GEMM in fp16 with fp32 accumulation.
All cheap operations (LayerNorms, Linear projections, sigmoids and the final
output‑gate multiplication) stay in PyTorch.  The kernel is launched with a
3‑D grid: (block_M, block_N, batch*head).  The implementation works for any
batch size ≤ 2, any sequence length up to 1024 and any hidden dimension.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    M, N, K,
    # strides for the matrix dimensions (per‑matrix)
    stride_am, stride_ak,            # left  (M‑dim, K‑dim)
    stride_bk, stride_bn,            # rightᵀ (K‑dim, N‑dim)  <-- note the swapped order
    stride_cm, stride_cn,            # output (M‑dim, N‑dim)
    # strides for the outer batch*head dimension
    stride_ab, stride_bb, stride_cb,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM:  C[b] = A[b] @ B[b].T   for all b = 0 … batch*head‑1"""

    pid_m = tl.program_id(0)          # block‑row
    pid_n = tl.program_id(1)          # block‑col
    pid_b = tl.program_id(2)          # batch*head index

    # ------------------------------------------------------------------
    # Offsets inside the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current (batch*head) matrix
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # ------------------------------------------------------------------
    # Pointers to the start of the tile (row‑major for A, column‑major for Bᵀ)
    # ------------------------------------------------------------------
    a_tile_ptr = a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_tile_ptr = b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile
        a = tl.load(a_tile_ptr, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K), other=0.0)
        # Load Bᵀ tile
        b = tl.load(b_tile_ptr, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k * BLOCK_K) & mask_n[None, :], other=0.0)

        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write result (cast back to fp16 for storage)
    # ------------------------------------------------------------------
    c_tile_ptr = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]            (float32, CUDA)
        * mask_tensor  : [B, N, N]   (may be None) (float32 or bool, CUDA)
        * weights_dict : model parameters (torch tensors on the same device)
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – Triton GEMM
    #    rearrange to (B, H, N, N) → (B*H, N, N)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_fp16  = left_perm.view(-1, N, N).to(torch.float16)   # (B*H, N, N)
    right_fp16 = right_perm.view(-1, N, N).to(torch.float16)  # same layout

    # allocate output in fp16
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------
    # Strides needed by the kernel (all in elements, not bytes)
    # ------------------------------------------------------------------
    batch_heads = B * hidden_dim
    stride_ab = left_fp16.stride(0)   # = N * N
    stride_bb = right_fp16.stride(0)  # = N * N
    stride_cb = out_fp16.stride(0)    # = N * N

    # matrix‑level strides (per‑matrix)
    stride_am = left_fp16.stride(1)   # row stride = N
    stride_ak = left_fp16.stride(2)   # col stride = 1
    # for the transposed right matrix we swap the last two strides
    stride_bk = right_fp16.stride(2)  # = 1   (step along K)
    stride_bn = right_fp16.stride(1)  # = N   (step along N after transpose)
    stride_cm = out_fp16.stride(1)    # = N
    stride_cn = out_fp16.stride(2)    # = 1

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # program id 0 → block‑row
        triton.cdiv(N, BLOCK_N),   # program id 1 → block‑col
        batch_heads,               # program id 2 → batch*head
    )

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_ab=stride_ab,
        stride_bb=stride_bb,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim) and cast to float32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # --------------------------------------------------------------
    out = out * out_gate   # shape matches exactly

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimensionality
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
