"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden channel *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We flatten the batch and hidden dimensions:
    M = B * hidden_dim
and compute M independent GEMMs (N×N)·(N×N)ᵀ with a single Triton kernel.
All cheap ops (layer‑norm, linear, sigmoids, masking) stay in PyTorch.
The kernel works in fp16 (Tensor‑Core friendly) while accumulating in fp32
and finally returns fp32 results.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,           # pointers
    N, M,                          # N = seq_len, M = B * hidden_dim
    stride_ah, stride_ai, stride_ak,   # strides of A (M,N,N)
    stride_bh, stride_bj, stride_bk,   # strides of B (M,N,N)
    stride_ch, stride_ci, stride_cj,   # strides of C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for M independent (N,N) matrices.

    A, B, C are stored as (M, N, N) in row‑major order.
    The kernel tiles the i‑j plane (BLOCK_M × BLOCK_N) and iterates over k
    in BLOCK_K sized chunks, using fp16 loads and fp32 accumulation.
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # which (batch,head) -> flattened M

    # ---- block offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---- base pointers for this (M‑th) matrix -------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over K dimension --------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B_T[k,j]  shape (BLOCK_K, BLOCK_N)
        # original B is indexed as B[j,k]; we load the transposed view.
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # ---- write result --------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.

    left / right : (B, H, N, N)   dtype = float16
    returns       : (B, H, N, N)   dtype = float32
    """
    B, H, N, _ = left.shape
    M = B * H                     # flattened batch*head

    # flatten to (M, N, N)
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – empirically good for Hopper (sm_90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # fits well in Tensor‑Core registers

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )
    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple(input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask         : torch.Tensor  [B, N, N]       (bool or float)
        weights      : dict of torch.Tensors (all parameters, no bias)
        config       : dict with keys:
            - dim (int)
            - hidden_dim (int)
            - nomask (bool)   (True → mask is ignored)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over feature dim
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)  dim -> hidden_dim
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)      # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcastable to (B,N,N,1)
        m = mask.to(device).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ  (per hidden channel)
    #    Convert to (B, hidden, N, N) layout expected by the kernel.
    #    Cast to fp16 for Tensor‑Core acceleration.
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm = trimul_gemm(left_perm, right_perm)   # (B,H,N,N), fp32

    # Back to the original layout (B,N,N,H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B,N,N,dim)

    return out.to(torch.float32)
