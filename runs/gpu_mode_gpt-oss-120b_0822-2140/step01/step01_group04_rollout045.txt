"""
custom_kernel: outgoing TriMul (AlphaFold3) forward pass.

The dominant O(N³) operation is the 3‑dimensional contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix‑multiply of the form
    (B, H, N, N)  @  (B, H, N, N)ᵀ   →   (B, H, N, N)

where H = hidden_dim.  
We off‑load this contraction to a Triton kernel that processes a
TILE_M×TILE_N block of the output for one (batch, head) pair.
All other layers (LayerNorm, Linear, gating, final LayerNorm/Linear)
remain in PyTorch – they are cheap compared to the O(N³) work.

The kernel:
* loads A and Bᵀ as FP16 (bandwidth‑saving),
* accumulates in FP32,
* writes the result back as FP32 (matching the reference model).

The host‑side `custom_kernel` prepares the tensors, launches the kernel,
and applies the remaining PyTorch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ   (A,B shape: (B, H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # const * fp32   (B,H,N,N)
    B_ptr,                     # const * fp32   (B,H,N,N)
    C_ptr,                     #       * fp32   (B,H,N,N) output
    N,                         # seq length
    H,                         # hidden_dim (treated as “heads”)
    stride_ah, stride_ah_h, stride_ah_i, stride_ah_k,
    stride_bh, stride_bh_h, stride_bh_i, stride_bh_j,
    stride_ch, stride_ch_h, stride_ch_i, stride_ch_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ.
    """

    pid_m = tl.program_id(0)          # tile row   i
    pid_n = tl.program_id(1)          # tile col   j
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # --------------------- tile offsets ---------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ---- load A[i,k] ------------------------------------------------
        a_ptrs = (A_ptr
                  + batch * stride_ah
                  + head  * stride_ah_h
                  + offs_i[:, None] * stride_ah_i
                  + offs_k[None, :] * stride_ah_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- load Bᵀ[k,j]  (B[j,k] in memory) ---------------------------
        b_ptrs = (B_ptr
                  + batch * stride_bh
                  + head  * stride_bh_h
                  + offs_k[:, None] * stride_bh_j   # k as first dim (j‑stride)
                  + offs_j[None, :] * stride_bh_i)  # j as second dim (i‑stride)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # --------------------- store result ---------------------
    c_ptrs = (C_ptr
              + batch * stride_ch
              + head  * stride_ch_h
              + offs_i[:, None] * stride_ch_i
              + offs_j[None, :] * stride_ch_j)

    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]  (float32 or bf16)
            - mask         : torch.Tensor [B, N, N]      (bool/float, may be all‑ones)
            - weights      : dict of torch.Parameter tensors
            - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # utility to fetch weights (biases are stored with the same key)
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden_dim]

    if not nomask:
        mask_f = mask.unsqueeze(-1).to(dtype)    # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) – also no bias in the original code
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction via Triton: (B, H, N, N) @ (B, H, N, N)ᵀ
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) – “heads” = hidden_dim
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B,H,N,N]

    # strides in element units (needed by Triton)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will read transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # tile sizes – tuned for Hopper (sm90); can be changed if needed
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,               # 4 warps works well on H200
    )

    # back to original layout [B,N,N,hidden_dim]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm + gating + linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, w("to_out.weight"))  # [B,N,N,dim]

    return out.to(dtype)
