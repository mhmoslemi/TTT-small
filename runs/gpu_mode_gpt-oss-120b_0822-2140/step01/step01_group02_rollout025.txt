```python
"""
custom_kernel – TriMul “outgoing” forward pass (no grad).

The O(N³·H) core is a batched matrix multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which we rewrite as a batch of GEMMs:
    (B·H, N, N) left   @   (B·H, N, N).transpose(-2,-1)  → (B·H, N, N)

A single Triton kernel performs all GEMMs in FP16 (accumulating in FP32).
The surrounding LayerNorm / Linear / sigmoid gates are kept in PyTorch
because they are negligible compared with the contraction.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : [batch, M, K])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # base pointers
    M, N, K,                           # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col)
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where B is read as transposed (i.e. we compute A @ Bᵀ)."""
    pid_batch = tl.program_id(2)          # batch index   (0 … B*H-1)
    pid_m = tl.program_id(0)              # block‑row index
    pid_n = tl.program_id(1)              # block‑col index

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tiles
    a_ptrs = a_ptr + pid_batch * stride_az \
                  + offs_m[:, None] * stride_am \
                  + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored in normal layout (batch, N, K).  For Bᵀ we need element
    # (k, n) → row‑k, col‑n in the original memory.
    b_ptrs = b_ptr + pid_batch * stride_bz \
                  + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # Load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load Bᵀ tile
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot product
        acc += tl.dot(a, b)

        # Advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store result as FP16 (bandwidth‑friendly)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                   + offs_m[:, None] * stride_cm \
                   + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for FP16 tensors.
    a, b : (B·H, N, N)  FP16, contiguous.
    returns c : same shape & dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, N, K = a.shape
    assert K == N and b.shape == (batch, N, N)

    c = torch.empty_like(a)

    # Strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout: (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # program_id(0)
        triton.cdiv(N, BLOCK_N),      # program_id(1)
        batch,                        # program_id(2)
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block (good for H200)
        num_stages=3,         # pipelining depth
    )
    return c


# ----------------------------------------------------------------------
# Host entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype=float32
            - mask         : torch.Tensor, shape [B, N, N] (bool or float)
            - weights      : dict of torch.Tensors (all float32)
            - config       : dict with keys
                * "dim"        – input feature dimension
                * "hidden_dim" – hidden dimension used inside TriMul
                * "nomask"     – bool, True → ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.to(left.dtype).unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Reshape for the batched GEMM  → (B*H, N, N)
    # ------------------------------------------------------------------
    # (B,N,N,hidden) → (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM C = left @ rightᵀ   (still fp16)
    # ------------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to (B,N,N,hidden) and cast to fp32
    # ------------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate computed earlier has shape (B,N,N,hidden)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

The kernel **only** handles the expensive `N³·hidden_dim` contraction; everything else stays in PyTorch.  
All tensor shapes, optional mask handling, and mixed‑precision casting follow the specification, and the code is ready to run on an NVIDIA H200 (sm_90).
