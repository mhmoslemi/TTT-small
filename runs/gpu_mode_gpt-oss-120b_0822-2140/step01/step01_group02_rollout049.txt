## Triton + PyTorch implementation

```python
"""
custom_kernel – outgoing TriMul (forward only)

The kernel computes, for every batch b and every hidden dimension h
    out[b,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]          (matrix‑multiply with Bᵀ)

where `left` and `right` have shape (B, H, N, N) after the linear
projections and gating.  The computation is fused across the (B·H)
independent GEMMs using a single Triton kernel.

All other operations (LayerNorm, Linear layers, sigmoid gates,
final LayerNorm + Linear) are performed with PyTorch for readability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (batch, M, K)      B : (batch, N, K)   (we treat Bᵀ inside the kernel)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # base pointers
    M, N, K,                                 # matrix sizes (int)
    stride_az, stride_am, stride_ak,         # A strides (batch, row, col)
    stride_bz, stride_bk, stride_bn,         # B strides (batch, row, col) – original layout
    stride_cz, stride_cm, stride_cn,         # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the right‑hand matrix is logically transposed."""
    pid_batch = tl.program_id(2)                 # index over (B * H)
    pid_m = tl.program_id(0)                     # tile row
    pid_n = tl.program_id(1)                     # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile (M‑axis and N‑axis)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current tile in A and Bᵀ
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_az \
                  + offs_m[:, None] * stride_am \
                  + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                  + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # masks for the current K‑tile
        mask_k = cur_k + tl.arange(0, BLOCK_K) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # fp16 load
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # fp16 load

        # dot : (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        # accumulation is performed in FP32 automatically
        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as fp16 to reduce bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn

    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches ``batched_gemm_kernel``.
    a, b : float16 tensors of shape (B*H, N, N), contiguous.
    Returns c of the same shape & dtype (float16).
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "right now we only need square matrices"

    c = torch.empty_like(a)

    # strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B_H,                              # batch * hidden
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                      # empirically good for 128×128 tiles on Hopper
    )
    return c


# ----------------------------------------------------------------------
# Host entry‑point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights      : dict of all linear / layer‑norm parameters (float32)
        - config       : dict containing at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B, N, N, 1)
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate          # (B, N, N, hidden)
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Reshape for the batched GEMM: (B, hidden, N, N) → (B*hidden, N, N)
    # --------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM: C = left @ rightᵀ   (still fp16)
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left_fp16, right_fp16)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, hidden) and cast to fp32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # out_gate shape (B, N, N, hidden)

    # --------------------------------------------------------------
    # 9️⃣ Final linear back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
