"""
TriMul “outgoing” forward pass (no gradients).

The heavy work is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiply C = A @ Bᵀ for every hidden head * batch.
All cheap ops (LayerNorm, Linear, sigmoids, mask, final projection) are kept in
PyTorch.  The contraction is performed in a custom Triton kernel that works in
FP16 internally but accumulates in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
#   A : (B·H, M, K)   – contiguous, fp16
#   B : (B·H, N, K)   – contiguous, fp16
#   C : (B·H, M, N)   – fp16 output
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # base pointers
    M, N, K,                        # matrix sizes (int)
    stride_az, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bz, stride_bn, stride_bk,   # B strides (batch, row, col) – original layout
    stride_cz, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where B is logically transposed (A @ Bᵀ).

    One program ID per (block_row, block_col, batch_head).
    """
    pid_batch = tl.program_id(2)          # index in the fused batch*head dim
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ------------------------------------------------------------------
    # Compute tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for A and Bᵀ
    a_tile_ptr = a_ptr + pid_batch * stride_az \
                       + offs_m[:, None] * stride_am \
                       + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  To get Bᵀ we need element (k, n)
    # → row k, col n in the original layout.
    b_tile_ptr = b_ptr + pid_batch * stride_bz \
                       + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                       + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a block of A (M×K_tile) and Bᵀ (K_tile×N)
        a = tl.load(
            a_tile_ptr,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )

        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (down‑cast to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_tile_ptr = c_ptr + pid_batch * stride_cz \
                       + offs_m[:, None] * stride_cm \
                       + offs_n[None, :] * stride_cn

    tl.store(c_tile_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Launch ``batched_gemm_kernel``.
    a, b : (B·H, N, N) fp16  (contiguous)
    Returns c of same shape / dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N

    c = torch.empty_like(a)

    # Strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Reasonable tiling for Hopper SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32          # fits well in registers, keeps shared memory low

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B_H,                          # batch * hidden
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,          # M
        N,          # N
        N,          # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bn,
        stride_bk,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps for 128×128 tiles on H200
        num_stages=3,
    )
    return c


# ----------------------------------------------------------------------
# Host‑side entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only “outgoing” TriMul.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        mask         : torch.Tensor  [B, N, N]      (bool / float)
        weights      : dict of torch.Tensors  (float32)
        config       : dict with keys
                       - "dim"        (int)
                       - "hidden_dim" (int)
                       - "nomask"     (bool, optional)

    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over the input channel dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Linear projections  (no bias in the reference model)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Rearrange to (B, hidden, N, N) and flatten batch*head
    # --------------------------------------------------------------
    # (B,N,N,hidden) → (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B = left.shape[0]
    left = left.view(B * hidden_dim, input_tensor.shape[1], input_tensor.shape[2])
    right = right.view(B * hidden_dim, input_tensor.shape[1], input_tensor.shape[2])

    # fp16 for the heavy GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM: C = left @ rightᵀ   (still fp16)
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # (B*hidden, N, N)

    # --------------------------------------------------------------
    # 7. Restore shape and convert back to fp32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, input_tensor.shape[1], input_tensor.shape[2])
    out = out.permute(0, 2, 3, 1).to(torch.float32)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 8. Post‑projection LayerNorm + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # shape broadcastable (B,N,N,hidden)

    # --------------------------------------------------------------
    # 9. Final linear back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
