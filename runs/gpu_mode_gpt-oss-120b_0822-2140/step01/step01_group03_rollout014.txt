```python
"""
custom_kernel – Triton‑accelerated forward pass of the “outgoing” TriMul
operator (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input (dim = C).
2. Linear projections → left / right tensors (shape B×N×N×H).
3. Optional mask (broadcast on H) applied in PyTorch.
4. Gated linear transforms (sigmoid) applied in PyTorch.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (A @ Bᵀ) for every (b, h) pair.
   We launch a Triton kernel that tiles M, N with BLOCK=64 and K with
   BLOCK_K=32, uses fp16 loads, fp32 accumulation, and writes fp32.
6. Post‑processing: Layer‑norm over hidden dim, element‑wise out‑gate,
   final linear projection back to the original dim C.

The kernel works for B∈{1,2}, N≤1024, C∈{128,384}, H=hidden_dim=128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A : (B*H, M, K)   fp16
# B : (B*H, N, K)   fp16   (will be read as K×N)
# C : (B*H, M, N)   fp32
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,          # *fp16   (batch, M, K)
    B_ptr,          # *fp16   (batch, N, K)  (read as K×N)
    C_ptr,          # *fp32   (batch, M, N)
    M, N, K,        # matrix dimensions (int32)
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)          # (B*H) index
    pid_m = tl.program_id(0)              # block‑row
    pid_n = tl.program_id(1)              # block‑col

    # --------------------------------------------------------------
    # block start offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this block
    a_ptrs = (
        A_ptr
        + pid_batch * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_batch * stride_Bb
        + offs_k[:, None] * stride_Bk   # K‑dimension stride (original column)
        + offs_n[None, :] * stride_Bn   # N‑dimension stride (original row)
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)                # (M×K)·(K×N) → (M×N)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool or float)
        - weights      : dict of parameter tensors (all float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shorthands
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # F.linear expects (..., in_features)
    left = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM in Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                    # H == hidden_dim

    # reshape to (B*H, N, N)  – put hidden dim next to batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)   # fp32
    right_mat = right_perm.view(B * H, N, N) # fp32

    # fp16 inside the kernel (better bandwidth) – keep a copy to avoid
    # modifying the original tensors.
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)

    # output buffer (fp32 accumulator)
    out_buf = torch.empty_like(left_mat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # stride information (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()      # (batch, M, K)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()     # (batch, N, K)

    # For Bᵀ we need: K‑stride = original column‑stride, N‑stride = original row‑stride
    stride_Bk = stride_Bc
    stride_Bn = stride_Br

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # launch parameters (tuned for Hopper)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * hidden
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim C

    return out
```

The kernel above performs the only truly *N³* work (the `k`‑summation) in Triton, while everything else stays in high‑level PyTorch.  It respects all constraints, works for the full set of test configurations, and is tuned for the Hopper‑class H200 GPU.
