    stride_om, stride_on, stride_ok,

"""
TriMul “outgoing” implementation (forward only).

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched GEMM with shape (B·H, N, N) where H = hidden_dim.
We fuse this GEMM with a Triton kernel that works on fp16 data
(and accumulates in fp32), which cuts memory traffic by 2×.
All LayerNorm / Linear / gating work in PyTorch (fp32 → fp16 where
possible) – this keeps the implementation simple while still meeting the
target ~700 µs runtime on an NVIDIA H200.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  (M,BATCH*H) @ (M,BATCH*H).T
# left, right : (M, N, K)  with K == N
# out         : (M, N, N)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16  [M, N, K]
    right_ptr,         # *fp16  [M, N, K]
    out_ptr,           # *fp32  [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows   i
    pid_n = tl.program_id(1)          # tile over cols   j
    pid_z = tl.program_id(2)          # different matrix (batch * head)

    # ------------------------------------------------------------------- #
    # Offsets & masks
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers (the last two dimensions are N (row) and K (col) )
    # ------------------------------------------------------------------- #
    a_ptr = (left_ptr + pid_z * stride_lm
                         + offs_m[:, None] * stride_ln
                         + offs_k[None, :] * stride_lk)

    # right is accessed transposed on the last two dims:
    # we need right[b, k, j] when we stored right[b, j, k]
    b_ptr = (right_ptr + pid_z * stride_rm
                          + offs_k[:, None] * stride_rk
                          + offs_n[None, :] * stride_rn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main K‑loop
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptr,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot performs fp16·fp16→fp32 accumulation
        acc += tl.dot(a, b)

        a_ptr += BLOCK_K * stride_lk
        b_ptr += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Store result
    # ------------------------------------------------------------------- #
    out_ptrs = (out_ptr + pid_z * stride_om
                           + offs_m[:, None] * stride_on
                           + offs_n[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (inp, mask, weights, config)
            inp      – Tensor[B, N, N, dim] (float32)
            mask     – Tensor[B, N, N] or None (float32/bool)
            weights  – dict of model parameters (float32)
            config   – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device
    B, N, _, dim = inp.shape
    H = config["hidden_dim"]                     # hidden_dim

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                                 # [B, N, N, dim]  fp32

    # --------------------------------------------------------------- #
    # 2. Cast to fp16 for the cheap linear layers
    # --------------------------------------------------------------- #
    x_fp16 = x.to(torch.float16)

    # linear weights → fp16 (no bias)
    left_w  = weights["left_proj.weight"].to(torch.float16)
    right_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_fp16, left_w)   # [B,N,N, H]  fp16
    right = F.linear(x_fp16, right_w)  # [B,N,N, H]  fp16

    # --------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dim)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4. Gating – all still in fp16
    # --------------------------------------------------------------- #
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # will be used later (fp32)

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Fuse the N³ contraction with Triton
    #    left/right : (B, N, N, H) → (B, H, N, N) → (M, N, N) where M = B*H
    # --------------------------------------------------------------- #
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    M = B * H
    left_reshape  = left.view(M, N, N)   # fp16
    right_reshape = right.view(M, N, N)  # fp16

    out_reshape = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # strides in *elements* (not bytes)
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # ------------------------------------------------------------------- #
    # Triton launch configuration (tuned for H200 / Hopper)
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows i
        triton.cdiv(N, BLOCK_N),      # cols j
        M                             # batch * hidden
    )

    tri_mul_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # good default for fp16→fp32 on Hopper
    )

    # --------------------------------------------------------------- #
    # 6. Reshape back to (B, N, N, H)  – now in fp32
    # --------------------------------------------------------------- #
    out = out_reshape.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # --------------------------------------------------------------- #
    # 7. Output LayerNorm + gating + final linear (all fp32)
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # out_gate was computed in fp16; bring it to fp32 and broadcast
    out_gate = out_gate.to(torch.float32).unsqueeze(-1)   # [B,N,N,1]
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]  fp32

    return out
