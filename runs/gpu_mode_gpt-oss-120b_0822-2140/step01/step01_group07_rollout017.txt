"""
TriMul “outgoing” forward pass.

The costly part is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply of shape (B·H, N, N) with the
transpose of the second operand.  We fuse this contraction with a
Triton GEMM kernel.  The rest of the module (LayerNorms, Linear
projections, and sigmoids) stays in PyTorch – they are cheap compared
to the N³ work.

Key optimisations
-----------------
*  Cast the two projected tensors to FP16 before the GEMM.  Memory
  traffic drops by 2× while the dot product is still carried out on
  Tensor‑cores.
*  Accumulate in FP32 to retain numerical stability (`acc` is fp32,
  each `tl.dot` is fp16 and is up‑cast before accumulation).
*  Tile the GEMM with 64×64 registers per thread‑block and a K‑tile
  of 64.  On Hopper a launch with 8 warps per block gives the best
  utilisation for the problem sizes in the test suite.
*  The mask (if present) is applied in the PyTorch part – this keeps the
  kernel simple and avoids an extra load.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ   where left/right have shape (M,N,N)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16  [M, N, N]
    right_ptr,         # *fp16  [M, N, N]
    out_ptr,           # *fp32  [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the TriMul contraction.
    Each program instance works on a (BLOCK_M, BLOCK_N) tile of the
    output matrix for a single (batch * head) entry.
    """
    pid_m = tl.program_id(0)          # tile over rows   i
    pid_n = tl.program_id(1)          # tile over cols   j
    pid_z = tl.program_id(2)          # matrix index     (batch * head)

    # ------------------------------------------------------------------- #
    # Offsets & masks
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointer arithmetic
    # ------------------------------------------------------------------- #
    # left:   (M, N, K) → we need A[z, i, k]
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # right transposed on the last two dimensions:
    # we need Bᵀ → B[z, k, j]
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    # ------------------------------------------------------------------- #
    # Accumulator (fp32)
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main K‑loop
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # fp16 dot → fp16 result, immediately up‑cast to fp32 for accumulation
        acc += tl.dot(a, b).to(tl.float32)

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Store result (fp32)
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a fused Triton GEMM.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N]      (bool / 0‑1 float)
        - weights      : dict of torch.Tensor with the same names as in the
                         reference PyTorch implementation
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm  (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # fp32
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the last dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) in fp32 – cheap
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Cast to fp16 and reshape to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------- #
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, N, N, H) → (B, H, N, N) → (M, N, N)
    left_reshape  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (fp32 because the rest of the model is in fp32)
    out_reshape = torch.empty_like(left_reshape, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows  (i)
        triton.cdiv(N, BLOCK_N),          # cols  (j)
        B * hidden_dim                    # independent matrices
    )

    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden_dim, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # Hopper works well with 8 warps for fp16 GEMM
    )

    # ------------------------------------------------------------------- #
    # 7. Back‑to [B, N, N, hidden_dim] (fp32)
    # ------------------------------------------------------------------- #
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 8. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out = out * out_gate               # broadcasted over the last dim
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
