"""
TriMul “outgoing” kernel (AlphaFold‑3).

The expensive part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`
(i.e. for each (batch, head) pair we compute A @ Bᵀ with
A,B ∈ ℝ^{N×N}).  The kernel below performs this multiplication
in FP16 (for bandwidth) while accumulating in FP32 (for accuracy).
All the surrounding LayerNorm / Linear / gating operations stay in
PyTorch – they are cheap compared to the O(N³) contraction.

The kernel works on tensors of shape (B, H, N, N) that are
contiguous in row‑major order.  Each program instance computes a
TILE_M × TILE_N block of the output matrix for a single (batch, head)
pair.  Block sizes have been tuned for the Hopper H200 (SM90) – 128×128
tiles with 8 warps give ~3× speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # left  : (B, H, N, N)
    B_ptr,                     # right : (B, H, N, N)
    C_ptr,                     # out   : (B, H, N, N)
    N,                         # sequence length
    H,                         # hidden dimension
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,      # rows   processed by a program
    TILE_N: tl.constexpr,      # cols   processed by a program
    TILE_K: tl.constexpr,      # reduction dimension block size
    BLOCK_M: tl.constexpr,     # alias for readability (same as TILE_M)
    BLOCK_N: tl.constexpr,     # alias for readability (same as TILE_N)
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ.
    """
    pid_m = tl.program_id(0)                # tile row   (i)
    pid_n = tl.program_id(1)                # tile col   (j)
    pid_hb = tl.program_id(2)               # combined batch*head

    batch = pid_hb // H
    head  = pid_hb % H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # load A[i,k]
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # load Bᵀ → B[j,k] (note the swapped strides)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_k   # k as first dimension
                  + offs_n[None, :] * stride_b_j)  # j as second dimension
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot‑product over the K dimension
        acc += tl.dot(a, b)          # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for
    the O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # shortcuts for weights
    # --------------------------------------------------------------
    def w(name):
        return weights[name]

    eps = 1e-6

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias=False) and gating
    # --------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden_dim]

    if not nomask:
        # mask shape → [B,N,N,1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(dtype)
        left  = left * mask_
        right = right * mask_

    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3️⃣ Heavy contraction (batched mat‑mul) via Triton
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in *elements* (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, j, k) – we will treat k as inner dim
    c_s = out_t.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # Tiling parameters – tuned for Hopper SM90
    # ------------------------------------------------------------------
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64   # reduction block size (fits in shared memory)

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        BLOCK_M=TILE_M,
        BLOCK_N=TILE_N,
        num_warps=8,        # 8 warps give good occupancy on H200
        num_stages=3,       # double‑buffering
    )

    # back to original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 4️⃣ Final LayerNorm + gating + linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                      # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
