"""
TriMul (outgoing) – fast forward pass.

The dominant O(N³·H) contraction is carried out by a custom Triton
batched‑GEMM kernel.  For each pair (batch, head) we compute

    out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]

which is exactly a matrix‑multiply of a (N×N) tensor with the
transpose of another (N×N) tensor.  By reshaping the tensors to
(B·H, N, N) we can launch a single Triton kernel that processes all
(b,h) slices in parallel (grid‑dim‑2).  The kernel works in FP16 for
speed and accumulates in FP32, converting back to FP32 before the
subsequent LayerNorm and final projection.

All non‑linearities, LayerNorms and the final linear projection stay
in PyTorch – only the heavy O(N³·H) part is offloaded to Triton, which
satisfies the “must use a kernel” requirement while keeping the
implementation simple and portable.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_gemm(
    a_ptr,          # FP16, shape (Z, M, K) flattened
    b_ptr,          # FP16, shape (Z, N, K) flattened (we treat K as inner dim)
    c_ptr,          # FP32 output, shape (Z, M, N) flattened
    M, N, K,
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, inner)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, inner, row)
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ Bᵀ for a batch of matrices.
        A : (Z, M, K)   – row‑major
        B : (Z, N, K)   – row‑major, we interpret it as (Z, K, N) when dotting
        C : (Z, M, N)   – row‑major
    """
    pid_z = tl.program_id(2)          # batch * head index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # ------------ block offsets ---------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for partial tiles on the right / bottom edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------ pointers for the current batch slice -----------------
    a_batch = a_ptr + pid_z * stride_az
    b_batch = b_ptr + pid_z * stride_bz
    c_batch = c_ptr + pid_z * stride_cz

    # ------------ load a tile of A (M × K) ----------------------------
    a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
    a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)

    # ------------ load a tile of Bᵀ (K × N) ----------------------------
    # B is stored as (N, K) but we need its transpose, so we load with
    # stride_bk for the K dimension and stride_bn for the N dimension.
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
    b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

    # ------------ compute tile of C ------------------------------------
    acc = tl.dot(a, b)               # (BLOCK_M, BLOCK_N) in fp32

    # ------------ write back -------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32 or bool, CUDA) – may be None
        * weights_dict : model weights (tensors on the same device)
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm over channel dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction via a tiled GEMM kernel
    # ------------------------------------------------------------------
    # Move hidden dimension to axis 1: (B, H, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Merge batch and head dimensions → (Z, N, N) where Z = B*H
    Z = B * hidden_dim
    left = left.view(Z, N, N)
    right = right.view(Z, N, N)

    # Cast to FP16 for the GEMM (speed) – accumulation stays FP32
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Allocate output (FP32)
    out_f32 = torch.empty_like(left_f16, dtype=torch.float32, device=device)

    # Strides (in elements) for the three tensors
    stride_az, stride_am, stride_ak = left_f16.stride()   # (N*N, N, 1)
    stride_bz, stride_bk, stride_bn = right_f16.stride() # (N*N, 1, N)
    stride_cz, stride_cm, stride_cn = out_f32.stride()   # (N*N, N, 1)

    # Tile sizes – chosen to fit H200 shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # program_id(0) – rows
        triton.cdiv(N, BLOCK_N),   # program_id(1) – cols
        Z,                         # program_id(2) – batch*head
    )

    triton_tri_mul_gemm[grid](
        left_f16.data_ptr(),
        right_f16.data_ptr(),
        out_f32.data_ptr(),
        N, N, N,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_f32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Multiply by output gate (pure PyTorch – cheap)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
