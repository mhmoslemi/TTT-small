"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We compute it with a highly‑tuned Triton kernel that performs, for every
(batch, head) pair, the multiplication   A @ Bᵀ   where

    A : (N, N)   = left_t[b, h, :, :]
    B : (N, N)   = right_t[b, h, :, :]   (we read it transposed)

All other inexpensive operations (layer‑norm, linear projections,
gating, final norm/linear) stay in PyTorch.

The kernel is autotuned for the H200 (SM90) – we try several
(tile_m, tile_n, tile_k, num_warps) configurations and pick the fastest
at launch time.  The inputs are cast to FP16 for bandwidth, accumulation
is done in FP32 (the fastest on Hopper), and the result is written back
in FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel (autotuned)
# --------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64, 'BLOCK_K': 32}, num_warps=8, num_stages=2),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 128, 'BLOCK_K': 32}, num_warps=8, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=8, num_stages=3),
        triton.Config({'BLOCK_M': 256, 'BLOCK_N': 64, 'BLOCK_K': 64}, num_warps=16, num_stages=3),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 256, 'BLOCK_K': 64}, num_warps=16, num_stages=3),
    ],
    key=['N', 'H']  # compile a separate binary for each (seq_len, hidden_dim) pair
)
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_i,   # note B is read transposed
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]  (i.e. A @ Bᵀ)."""
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_hb = tl.program_id(2)  # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A : (BLOCK_M, BLOCK_K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we want B[j,k] -> load as (BLOCK_K, BLOCK_N)
        #   stride_b_j points to the *k* dimension (because we transpose)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dim
                  + offs_n[None, :] * stride_b_i)  # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K (FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)          # (M,K) @ (K,N) → (M,N)

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32 or bfloat16)
        - mask         : torch.Tensor [B, N, N]      (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors (model params)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )                               # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden_dim]

    if not config.get("nomask", True):
        # mask shape: [B,N,N] → [B,N,N,1] for broadcasting
        mask_f = mask.to(dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) – also bias‑free linear
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction via Triton
    #    Rearrange to (B, H, N, N) → contiguous row‑major layout
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides expressed in number of elements (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j) – we read transposed
    c_s = out_t.stride()    # (batch, head, i, j)

    # Kernel launch grid
    grid = (
        triton.cdiv(N, 64),                     # program_id(0) – i tiles (BLOCK_M may vary per config)
        triton.cdiv(N, 64),                     # program_id(1) – j tiles (BLOCK_N may vary)
        B * hidden_dim,                         # program_id(2) – batch*head
    )

    # Launch the autotuned kernel
    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        num_warps=8,   # overridden by autotune configs when they match
    )

    # Back to original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating, and projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate               # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
