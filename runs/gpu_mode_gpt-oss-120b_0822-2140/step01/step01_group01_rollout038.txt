"""
TriMul “outgoing” forward pass (AlphaFold‑3 style)

The dominant O(N³) work is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* this is exactly a matrix‑matrix product  
L = left[b,:, :, d] (N×N)   and   R = right[b,:, :, d] (N×N)  

out[b,:, :, d] = L @ Rᵀ                     (N×N)

We therefore:
  1. Perform all cheap ops (LayerNorm, linear projections, masking,
     sigmoid gates) in PyTorch.
  2. Rearrange the tensors to (B, H, N, N) and launch a single Triton
     kernel that computes **M = B·H** independent GEMMs of shape
     (N×N)·(N×N)ᵀ.  The kernel tiles the i‑, j‑ and k‑axes (BLOCK_M,
     BLOCK_N, BLOCK_K = 128) and accumulates in fp32 while reading
     fp16 (half‑precision) – this gives a good speed/accuracy trade‑off
     on Hopper (SM‑90).
  3. Convert the result back to (B, N, N, H) and apply the output
     LayerNorm, out‑gate and final projection in PyTorch.

The implementation works for all test configurations (different batch
sizes, sequence lengths, hidden dimensions and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    N, M,                         # N = seq_len, M = B*H (flattened batch*head)
    stride_ah, stride_ai, stride_ak,   # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,   # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,   # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for each (M‑th) matrix pair.

    A and B are stored as (M, N, N) in row‑major order.  For a given
    program id (pid_m, pid_n, pid_z) we compute a BLOCK_M×BLOCK_N tile of
    the result matrix belonging to the pid_z‑th (batch,head) pair.
    """
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # flattened batch*head

    # ---- offsets within the N×N tile ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---- base pointers for the current (M‑th) matrix pair ------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension (tiles) -------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # fp16 → fp32

        # Bᵀ[k, j] = B[j, k]  shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k  (row of Bᵀ)
            + offs_n[None, :] * stride_bj          # j  (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)          # fp16 → fp32

        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  FP16
    returns       : (B, H, N, N)  FP32   (each slice = left @ rightᵀ)
    """
    B, H, N, _ = left.shape
    M = B * H                                 # flattened batch*head

    # flatten to (M, N, N) – stride information is kept
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # block sizes – tuned for Hopper (SM‑90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z (batch*head)
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   shape (B, N, N, dim)  (float32)
        - mask         : torch.Tensor   shape (B, N, N)      (bool or 0/1)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
            * dim, hidden_dim, nomask (bool)

    Returns
    -------
    torch.Tensor   shape (B, N, N, dim)    (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)   left/right
    # ------------------------------------------------------------------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(device),   # (hidden_dim, dim)
    )   # (B, N, N, hidden_dim)
    right = F.linear(
        x,
        weights["right_proj.weight"].to(device),
    )

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM: out = left @ rightᵀ  per hidden channel
    #    Convert to (B, H, N, N) layout and cast to fp16 for the kernel.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm = _batched_gemm(left_perm, right_perm)                       # (B, H, N, N) fp32

    # back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()                       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device),    # (dim, hidden_dim)
    )   # (B, N, N, dim)

    return out.to(torch.float32)
