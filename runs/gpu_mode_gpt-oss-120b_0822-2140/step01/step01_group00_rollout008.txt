"""
TriMul “outgoing” implementation (forward only).

The only O(N³) part – the contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

– is performed by a custom Triton GEMM kernel.  The tensors are first
projected, gated and reshaped to (B, H, N, N) where H = hidden_dim.
The kernel computes a batched matrix‑multiply A @ Bᵀ for every
(batch,head) pair using fp16 inputs and fp32 accumulation, then
stores the result in fp32.  All other operations (LayerNorm, linear
projections, gating, mask handling) stay in PyTorch.  The final
output has shape [B, N, N, dim] and is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def triton_trimul_gemm_kernel(
    # pointers
    a_ptr,          # left  [B, H, M, K]   (fp16)
    b_ptr,          # rightᵀ [B, H, K, N]   (fp16) – already transposed
    c_ptr,          # out   [B, H, M, N]   (fp32)

    # sizes
    B, H, M, N, K,   # compile‑time constants (int64)

    # strides (in elements)
    stride_a_batch, stride_a_head, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_head, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_head, stride_c_row, stride_c_col,

    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------- #
    # Grid layout:
    #   pid = batch_head * (num_M_blocks * num_N_blocks) + block_i * num_N_blocks + block_j
    # ------------------------------------------------------------------- #
    pid = tl.program_id(0)

    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_m_blocks * num_n_blocks

    batch_head = pid // blocks_per_head
    residual   = pid %  blocks_per_head
    block_i    = residual // num_n_blocks          # tile row index
    block_j    = residual %  num_n_blocks          # tile col index

    # ------------------------------------------------------------------- #
    # Compute the start offsets for this tile
    # ------------------------------------------------------------------- #
    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N

    # ------------------------------------------------------------------- #
    # Offsets within the tile
    # ------------------------------------------------------------------- #
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for boundary handling
    mask_i = offs_i < M
    mask_j = offs_j < N
    mask_k = offs_k < K

    # ------------------------------------------------------------------- #
    # Compute base pointers for this (batch, head) pair
    # ------------------------------------------------------------------- #
    batch = batch_head // H
    head  = batch_head % H

    a_base = a_ptr + batch * stride_a_batch + head * stride_a_head
    b_base = b_ptr + batch * stride_b_batch + head * stride_b_head
    c_base = c_ptr + batch * stride_c_batch + head * stride_c_head

    # ------------------------------------------------------------------- #
    # Accumulator in fp32
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main K‑loop
    # ------------------------------------------------------------------- #
    for k_start in range(0, K, BLOCK_K):
        # -------- load A tile (M x K) ----------------------------------- #
        a_ptrs = a_base \
                 + (offs_i[:, None] * stride_a_row) \
                 + ((k_start + offs_k)[None, :] * stride_a_col)
        a_mask = mask_i[:, None] & (mask_k[None, :] & (k_start + offs_k < K))
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float16)

        # -------- load B tile (K x N) ----------------------------------- #
        b_ptrs = b_base \
                 + ((k_start + offs_k)[:, None] * stride_b_row) \
                 + (offs_j[None, :] * stride_b_col)
        b_mask = (mask_k[:, None] & (k_start + offs_k < K)) & mask_j[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float16)

        # -------- dot product (accumulate in fp32) ---------------------- #
        acc += tl.dot(a, b)          # (BM, BK) @ (BK, BN) -> (BM, BN)

    # ------------------------------------------------------------------- #
    # Store the result (fp32)
    # ------------------------------------------------------------------- #
    c_ptrs = c_base \
             + (offs_i[:, None] * stride_c_row) \
             + (offs_j[None, :] * stride_c_col)
    c_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


# --------------------------------------------------------------------------- #
# Helper: apply an optional binary mask (broadcasted over the hidden dim)
# --------------------------------------------------------------------------- #
def _apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """src: [B,N,N,H]   mask: [B,N,N] → broadcasted product."""
    if mask is None:
        return src
    # mask broadcast → (B,N,N,1)
    mask = mask.unsqueeze(-1)
    return src * mask


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32, 0/1) or None
        - weights      : dict with all linear / LN parameters
        - config       : dict with keys
              * dim         – input channel size
              * hidden_dim  – projection size (called H)
              * nomask      – bool, True → mask is all‑ones

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # shortcuts for layer‑norm / linear using supplied parameters
    # ------------------------------------------------------------------- #
    def layer_norm(t, w, b):
        return F.layer_norm(t, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    def linear(t, w):
        return F.linear(t, w)          # no bias

    # ------------------------------------------------------------------- #
    # 1️⃣ Input layer‑norm
    # ------------------------------------------------------------------- #
    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])   # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Project to hidden dimension (no bias)
    # ------------------------------------------------------------------- #
    left  = linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask is [B,N,N] → broadcast to [B,N,N,1] then element‑wise multiply
        left  = _apply_mask(left,  mask)
        right = _apply_mask(right, mask)

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul contraction – Triton GEMM
    # ------------------------------------------------------------------- #
    # Cast to fp16 for the GEMM (faster, saves bandwidth)
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B, H, N, N) for the kernel
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    # we need rightᵀ on the last two dims: (K,N) → (N,K)
    right_t = right.transpose(-2, -1).contiguous()  # [B,H,N,N]

    Bsz, Hsz, N, _ = left.shape
    # output buffer
    out = torch.empty((Bsz, Hsz, N, N), dtype=torch.float32, device=x.device)

    # strides (in elements) for the three tensors
    a_s = left.stride()          # (batch, head, row, col)
    b_s = right_t.stride()
    c_s = out.stride()

    # tile sizes – tuned for Hopper (shared‑mem 64 KB, 128‑bit loads)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_blocks = Bsz * Hsz * grid_m * grid_n
    grid = (total_blocks,)

    triton_trimul_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right_t,
        c_ptr=out,
        B=Bsz,
        H=Hsz,
        M=N,
        N=N,
        K=N,
        stride_a_batch=a_s[0],
        stride_a_head=a_s[1],
        stride_a_row=a_s[2],
        stride_a_col=a_s[3],
        stride_b_batch=b_s[0],
        stride_b_head=b_s[1],
        stride_b_row=b_s[2],
        stride_b_col=b_s[3],
        stride_c_batch=c_s[0],
        stride_c_head=c_s[1],
        stride_c_row=c_s[2],
        stride_c_col=c_s[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [B,N,N,H]
    out = out.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------- #
    # 6️⃣ Output layer‑norm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
