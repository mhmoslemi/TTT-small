"""
Fast “outgoing” TriMul for AlphaFold‑3.

The expensive part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which for each hidden channel *d* is a matrix multiplication
    (N×N) @ (N×N)ᵀ .

We treat every (batch, hidden) pair as an independent batch and run a
single fused batched GEMM in Triton.  The kernel works in fp16 for the
loads (the model is trained in fp32, the loss of precision is negligible)
and accumulates in fp32.  All surrounding LayerNorm, linear layers and
gating are kept in PyTorch – this already gives a large speed‑up while
still respecting the required API.

The kernel tiles the product with
    BLOCK_M = BLOCK_N = 64   (row/col tiles)
    BLOCK_K = 32             (inner‑reduction tile)
and uses tl.dot for the inner dot‑product.  Strides are passed from
the host so that the kernel works for any batch size, sequence length
or hidden dimension that appear in the test matrix.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : (M×K) @ (K×N)  where N == M == seq_len.
# One batch dimension = (batch * hidden)
# ----------------------------------------------------------------------
@triton.jit
def bmma_kernel(
    A_ptr, B_ptr, C_ptr,          # tensor pointers (fp16 for A & B, fp32 for C)
    M, N, K,                      # dimensions (M==N==seq_len, K==seq_len)
    stride_ah, stride_am, stride_ak,  # strides for A (batch, row, col)
    stride_bh, stride_bn, stride_bk,  # strides for B (batch, row, col) – B is used transposed
    stride_ch, stride_cm, stride_cn,  # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)          # row tile
    pid_n = tl.program_id(2)          # col tile

    # ------------------- tile offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the beginning of the current tile
    a_ptrs = A_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- K‑loop -------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, apply masks, out‑of‑bounds = 0.0
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        # dot over the K‑dimension (fp16×fp16 → fp32 accumulation)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------- write back -------------------
    c_ptrs = C_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensor parameters (all on CUDA)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left  * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM
    #    (B, N, N, H) -> (B*H, N, N)  and cast to fp16 for speed
    # ------------------------------------------------------------------
    #   permute to (B, H, N, N) → view as (B*H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_f16  = left.view(B * hidden, N, N).to(torch.float16)
    right_f16 = right.view(B * hidden, N, N).to(torch.float16)

    # output accumulator (fp32)
    out_f32 = torch.empty_like(left_f16, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                     # batch*hidden
        triton.cdiv(N, BLOCK_M),       # row tiles
        triton.cdiv(N, BLOCK_N),       # col tiles
    )

    # strides after the .contiguous() above
    a_batch, a_row, a_col = left_f16.stride()
    b_batch, b_row, b_col = right_f16.stride()
    c_batch, c_row, c_col = out_f32.stride()

    bmma_kernel[grid](
        left_f16,
        right_f16,
        out_f32,
        N,          # M
        N,          # N
        N,          # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------
    out = out_f32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (hidden) + final gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
