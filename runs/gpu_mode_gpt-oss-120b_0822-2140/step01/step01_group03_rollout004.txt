```python
"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core.

Workflow
--------
1. Layer‑norm the input tensor   (B,N,N,C)
2. Linear projections → left / right tensors (B,N,N,H)   (no bias)
3. Optional masking (broadcast on H)
4. Gated‑linear transforms (sigmoid) applied element‑wise
5. Core contraction:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM `left @ rightᵀ` for every head `h`.
   Implemented as a tiled Triton kernel with FP16 loads, FP32 accumulation.
6. Post‑layer‑norm, out‑gate, final linear projection back to C.

The kernel works for:
    B ∈ {1,2},  N ≤ 1024,  C ∈ {128,384},  H = hidden_dim = 128
All tensors are expected to be on CUDA and use float32 for the outer
computation (the kernel uses float16 internally for bandwidth).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _matmul_T_kernel(
    A_ptr,                     # *float16  (batch, M, K)
    B_ptr,                     # *float16  (batch, N, K)   (will be read as K×N)
    C_ptr,                     # *float32  (batch, M, N)
    M, N, K,                   # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM with B transposed on‑the‑fly.
    Each program works on a tile (BLOCK_M × BLOCK_N) of the output matrix.
    """
    pid_b = tl.program_id(2)                 # batch * head index
    pid_m = tl.program_id(0)                 # tile‑row index
    pid_n = tl.program_id(1)                 # tile‑col index

    # ---- tile start offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- bounds masks ----
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- pointer arithmetic (row‑major) ----
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # ---- accumulator in FP32 ----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension ----
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)       # (M×K) tile
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)       # (K×N) tile (transposed)
        acc += tl.dot(a, b)                         # (M×N) tile

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---- write results ----
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float)
        - weights      : dict of parameter tensors (all float32)
        - config       : dict containing "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack everything
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the raw pair representation
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right  [B,N,N,H]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (B,N,N,1)
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also linear, no bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton GEMM
    #    out = left @ rightᵀ  (per head)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape            # H == hidden_dim

    # reshape to (B*H, N, N) and make contiguous for simple stride logic
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H,N,N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # use FP16 inside the kernel (bandwidth) – keep a FP32 output buffer
    left_fp16 = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)   # (B*H,N,N)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (batch, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For the transposed view we need:
    stride_Bk = stride_Bc          # step along K (original column)
    stride_Bn = stride_Br          # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # 5️⃣1️⃣ Tile sizes – tuned for Hopper (SM90)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H                             # batch * head dimension
    )

    _matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to C

    return out
```
