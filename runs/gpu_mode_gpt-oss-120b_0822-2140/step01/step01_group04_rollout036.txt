"""
TriMul “outgoing” kernel (AlphaFold‑3)

The only O(N³) part of the module is

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension d (the “outgoing”
contraction).  We implement this contraction with a single Triton kernel that
computes, for every (batch, head) pair, the product

    C = A @ Bᵀ     with   A,B ∈ (N,N)   and   head = hidden_dim

All remaining operations – LayerNorm, linear projections, gating – are cheap
and stay in PyTorch.  The kernel works on tensors of shape (B, H, N, N) that
are contiguous (row‑major).  FP16 is used for the loads to reduce memory
traffic, while accumulation is performed in FP32 to keep numerical quality.
The final result is returned as float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ  (A,B : B×H×N×N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    For every (b, h) compute
        C[b, h, i, j] = Σₖ A[b, h, i, k] * B[b, h, j, k]   (i.e. A @ Bᵀ)
    """
    pid_m = tl.program_id(0)          # tile row   i
    pid_n = tl.program_id(1)          # tile col   j
    pid_hb = tl.program_id(2)         # combined batch*head index

    batch = pid_hb // H
    head  = pid_hb %  H

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ---- A : (i,k) -------------------------------------------------
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_i[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- Bᵀ : we need B[j,k]  ->  load as (k,j) --------------------
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k dimension first
                  + offs_j[None, :] * stride_b_i)  # j dimension second
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # store
    # --------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_i[:, None] * stride_c_i
              + offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32 or bfloat16)
        - mask         : torch.Tensor[B, N, N]        (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensor, model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dtype = torch.float32
    eps = 1e-6

    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # helpers to fetch parameters (they are already on the correct device)
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )                                            # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False) & mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))

    if not nomask:
        # mask shape → [B,N,N,1] and broadcast
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction via Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # output buffer
    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in elements (not bytes) – required by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)   (we will read transposed)
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile sizes – chosen to give good occupancy on Hopper (SM90)
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # i tiles
        triton.cdiv(N, TILE_N),          # j tiles
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,          # 4 warps gives ~96 threads per block → good for H200
    )

    # back to the original layout: [B,N,N,hidden_dim]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm + gating + linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))  # [B,N,N,dim]

    return out.to(dtype)
