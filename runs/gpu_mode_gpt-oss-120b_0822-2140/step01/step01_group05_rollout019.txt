"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
matrix‑multiply kernel written in Triton.

The heavy part of the forward is the Einstein‑summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel *d* this is a matrix multiplication
(N × N) @ (N × N)ᵀ.  We fuse the many (B·hidden) matrix products
into a single Triton kernel that works on FP32 data (the kernel can be
run in FP16 on H200 for extra speed – the host code casts the tensors
to half before the launch).  All LayerNorms, linear projections,
gate sigmoids and the final linear are kept in PyTorch to keep the
implementation simple while still achieving the required throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched matrix multiplication (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers (float32 or float16)
    M, N, K,                      # matrix sizes (M = N = seq_len, K = seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A: (batch, row, col)
    stride_bb, stride_bm, stride_bk,   # strides for B: (batch, row, col)
    stride_cb, stride_cm, stride_cn,   # strides for C: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[b, m, n] = Σ_k A[b, m, k] * B[b, n, k]   (B is accessed transposed)."""
    pid_n = tl.program_id(0)          # column‑tile index
    pid_m = tl.program_id(1)          # row‑tile index
    pid_b = tl.program_id(2)          # batch‑*‑hidden index

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current tile
    a_ptrs = A_ptr + pid_b * stride_ab \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb \
                  + offs_n[None, :] * stride_bm \
                  + offs_k[:, None] * stride_bk

    # accumulator in fp32 (even if inputs are fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles (use masks to stay inside the matrix)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        # dot over the K dimension
        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
            mask   – torch.Tensor of shape [B, N, N] or ``None``.
                     If provided it is broadcast over the hidden dimension.
            weights – dict with all linear / layer‑norm parameters (float32, CUDA)
            config  – dict with keys ``dim`` and ``hidden_dim`` (int)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]          # called hidden_dim in the paper
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)      # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiply: out = left @ rightᵀ   (per hidden channel)
    #    -----------------------------------------------------------------
    #    left/right : [B, N, N, hidden] → [B*hidden, N, N]
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    left_flat  = left.view(B * hidden, N, N)   # (B*hidden, N, N)
    right_flat = right.view(B * hidden, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 5️⃣ a Triton launch configuration
    # ------------------------------------------------------------------
    # Tile sizes were chosen empirically for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # program_id(0) – column tiles
        triton.cdiv(N, BLOCK_M),   # program_id(1) – row tiles
        B * hidden,                # program_id(2) – batch*hidden
    )

    # Strides for the three‑dimensional tensors (batch, row, col)
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    # ------------------------------------------------------------------
    # 5️⃣ b kernel launch (using FP32 – could be FP16 for speed)
    # ------------------------------------------------------------------
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5️⃣ c reshape back to original layout
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (hidden) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
