"""
TriMul “outgoing” kernel (AlphaFold3).

The costly part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply per hidden dimension `d`:
    (B, H, N, N) @ (B, H, N, N)^T   →   (B, H, N, N)

The implementation:
* All cheap ops (LayerNorm, Linear, gating) stay in PyTorch.
* The O(N³) contraction is performed by a custom Triton kernel that
  processes a TILE_M × TILE_N block of the output for a single (batch, head)
  pair.  Inputs are cast to fp16 for bandwidth, accumulation is fp32, and the
  result is written back in fp32 (the reference uses fp32 throughout).

The kernel works for any batch size (1 or 2), any sequence length up to 1024,
and any hidden dimension (e.g. 128).  It also supports the optional mask
that zero‑out the projected tensors before the contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B, H, N, N) @ (B, H, N, N)^T  →  (B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers for left, right, output
    N, H,                         # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    For each (b, h) compute
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ where A and B have shape (N, N).
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head index

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------ offsets for the tile -----------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ---- load A (TILE_M × TILE_K) ----
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + head  * stride_a_head
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # ---- load Bᵀ (TILE_K × TILE_N) ----
        # we need B[j, k]  →  i‑index = offs_n,  j‑index = offs_k
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + head  * stride_b_head
            + offs_k[:, None] * stride_b_j   # k as first dimension
            + offs_n[None, :] * stride_b_i   # j as second dimension
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # dot product over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # ------------ store result -------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + head  * stride_c_head
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32 / bfloat16)
        - mask         : torch.Tensor [B, N, N]      (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32  # final output dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper for weights (biases are stored with a separate key)
    # ------------------------------------------------------------------
    def get(name):
        return weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=get("norm.weight"),
        bias=get("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (bias=False) and optional masking
    # ------------------------------------------------------------------
    left  = F.linear(x, get("left_proj.weight"))   # [B, N, N, H]
    right = F.linear(x, get("right_proj.weight"))  # [B, N, N, H]

    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)    # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3️⃣  Gating sigmoids (also bias‑free linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, get("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, get("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, get("out_gate.weight")))  # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣  Heavy contraction via Triton (batched matmul per hidden dim)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)   # [B, H, N, N]

    # strides in elements (not bytes)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will load transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # launch configuration – chosen to give good occupancy on Hopper
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),            # rows i
        triton.cdiv(N, TILE_N),            # cols j
        B * hidden_dim,                    # combined batch*head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,          # good for fp16 compute on H200
    )

    # back to original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5️⃣  Final LayerNorm, out‑gate, and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=get("to_out_norm.weight"),
        bias=get("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, get("to_out.weight"))  # [B, N, N, dim]

    return out.to(dtype)
