## Triton‑accelerated “outgoing” TriMul implementation

```python
"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm the input (B,N,N,C).
2️⃣ Linear projections → left / right tensors (B,N,N,H).
3️⃣ (Optional) mask the projected tensors.
4️⃣ Gated‑linear transforms (sigmoid gates) applied element‑wise.
5️⃣ Core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication per hidden slice:
       out_slice = left_slice @ right_sliceᵀ
   The kernel executes many independent GEMMs in parallel (batch = B*H)
   using fp16 loads, fp32 accumulation, and writes fp32 results.
6️⃣ Post‑process: Layer‑norm + out‑gate → final linear projection to C.

The kernel works for B∈{1,2}, N up to 1024, dim∈{128,384},
hidden_dim = 128 (as required by the benchmark).  Mixed‑precision is used
inside the kernel, but the function returns a float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                      # *float16   (batch, M, K)
    B_ptr,                      # *float16   (batch, N, K)  (read as K×N)
    C_ptr,                      # *float32   (batch, M, N)
    M, N, K,                    # matrix dimensions (all int32)
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (k‑ and n‑stride after transpose)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM where the second operand is accessed transposed.
    Each program instance works on one (batch, row‑block, col‑block) tile.
    """
    pid_b = tl.program_id(2)                 # batch*head index
    pid_m = tl.program_id(0)                 # block row
    pid_n = tl.program_id(1)                 # block col

    # -------- tile start indices ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------- pointers for this tile ----------
    a_ptrs = (A_ptr + pid_b * stride_Ab
              + offs_m[:, None] * stride_Am
              + offs_k[None, :] * stride_Ak)

    # B is read as transposed: we swap row/col strides
    b_ptrs = (B_ptr + pid_b * stride_Bb
              + offs_k[:, None] * stride_Bk   # k‑dimension uses original column‑stride
              + offs_n[None, :] * stride_Bn)  # n‑dimension uses original row‑stride

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------- main k‑loop ----------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # Promote to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)          # (M×K)·(K×N) → (M×N)

        # advance to next k‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # -------- write back ----------
    c_ptrs = (C_ptr + pid_b * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float) – may be omitted
        - weights      : dict of parameter tensors (all float32)
        - config       : dict containing {"dim", "hidden_dim", "nomask"}.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)               # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also linear + biasless
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction: batched GEMM left @ rightᵀ per hidden slice
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    # reshape to (B*H, N, N) and make them contiguous for stable strides
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (BH, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (BH, N, N)

    # fp16 inside the kernel (bandwidth) – keep fp32 accumulator and output
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    out_buf = torch.empty_like(left_, dtype=torch.float32)   # (BH, N, N)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # batch, row, col
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()

    # For transposed access: swap the inner two strides
    stride_Bk = stride_Bc   # step along K (original column)
    stride_Bn = stride_Br   # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # launch parameters – tuned for Hopper (SM90)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * hidden
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing : LayerNorm → out‑gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim C

    return out
```
