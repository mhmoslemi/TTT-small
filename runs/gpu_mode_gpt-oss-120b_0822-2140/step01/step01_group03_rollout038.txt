## Triton‑accelerated “outgoing” TriMul

```python
"""
custom_kernel – Triton‑accelerated forward pass of the TriMul “outgoing” operator
(AlphaFold‑3 style).

The heavy part is the contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched GEMM (left @ rightᵀ) for every hidden dimension `h`.
The kernel processes (B·H) independent matrix‑multiplications in parallel,
using fp16 loads, fp32 accumulation and a 128×128×32 tile schedule that
covers N up to 1024 on Hopper (SM‑90).
All other layers (LayerNorm, linear projections, gates) stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, M, K]), accumulate in fp32
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,               # *fp16   (batch, M, K)
    B_ptr,               # *fp16   (batch, N, K)  – read as K×N
    C_ptr,               # *fp32   (batch, M, N)
    M, N, K,             # matrix dims
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (k‑step, n‑step after transposition)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch * head index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this block
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # (M×K) fp16→fp32
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (K×N) fp16→fp32
        acc += tl.dot(a, b)                           # (M×N) fp32

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # store result
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of TriMul (outgoing) with a Triton‑accelerated core.
    Arguments
    ----------
    data : (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, C]   (float32)
        mask         : [B, N, N]      (bool / float) – may be omitted (nomask=True)
        weights      : dict of model parameters (all float32)
        config       : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # convenience aliases for weights
    # --------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present)
    # --------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via Triton (batched GEMM)
    # --------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    # reshape to (B*H, N, N) for the kernel
    left_  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (better bandwidth) – keep a copy to avoid mutating
    left_fp16  = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # output buffer (fp32 accumulator result)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()      # (batch, M, K)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()     # (batch, N, K)

    # For the transposed view we need:
    stride_Bk = stride_Bc                # step along K (original column)
    stride_Bn = stride_Br                # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # tile sizes – tuned for SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch*head dimension
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,                     # M, N, K
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out_gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)      # back to original dim C

    return out
```
