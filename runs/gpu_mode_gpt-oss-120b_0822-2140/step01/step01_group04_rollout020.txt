"""
TriMul “outgoing” kernel (AlphaFold3)

The only O(N³) work in the module is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d` (here called
`hidden_dim`).  This file implements that contraction with a custom Triton
kernel while the cheap LayerNorm / Linear / gating operations stay in
PyTorch.

The kernel computes, for each (batch, head) pair, the product

    C[b,h,i,j] = A[b,h,i,k] @ B[b,h,j,k].T          (A and B have shape N×N)

using FP16 loads, FP32 accumulation and FP32 stores.  The surrounding
PyTorch code follows the reference implementation exactly (including the
optional mask handling) and finally projects the hidden dimension back to
`dim`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # tensors (B,H,N,N) in row‑major order
    N, H,                         # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    For each (b, h) compute
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]   (i.e. A @ Bᵀ)
    """
    pid_i = tl.program_id(0)               # tile row   → i
    pid_j = tl.program_id(1)               # tile col   → j
    pid_hb = tl.program_id(2)              # combined batch * head

    batch = pid_hb // H
    head  = pid_hb %  H

    # tile offsets
    offs_i = pid_i * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_j * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ---- load A[i, k] -------------------------------------------------
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + head   * stride_a_head
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # ---- load B[j, k] (we need the transpose, i.e. Bᵀ[k, j]) ----------
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + head   * stride_b_head
            + offs_k[:, None] * stride_b_j   # k as first dim
            + offs_j[None, :] * stride_b_i   # j as second dim
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)        # (M,K) @ (K,N) → (M,N)

    # ---- store result ----------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + head   * stride_c_head
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (same dtype as ``input_tensor``)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helper to fetch weight tensors (biases are stored under separate keys)
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) + optional masking
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden_dim]

    if not nomask:
        # mask shape -> [B,N,N,1] and broadcast
        mask_ = mask.unsqueeze(-1).to(dtype)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and apply to projections
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy O(N³) contraction using Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=torch.float32)   # accumulation in FP32

    # strides in *elements* (not bytes) – needed by the kernel
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will load transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # tile sizes – tuned for Hopper (SM90) with fp16 loads
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),    # i‑tiles
        triton.cdiv(N, TILE_N),    # j‑tiles
        B * hidden_dim,            # one program per (batch, head)
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,               # Hopper prefers 4‑warp blocks for this size
    )

    # back to original layout: [B,N,N,hidden_dim]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm + gating + projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))    # [B,N,N,dim]

    # preserve original dtype (often float32)
    return out.to(dtype)
