"""
TriMul “outgoing” kernel (AlphaFold3) – Triton implementation.

The expensive part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply per hidden dimension `d`:

    out[b,d] = left[b,d] @ right[b,d].T          (left/right : N×N)

The kernel below computes this for all batches and hidden dimensions
simultaneously.  The surrounding operations (LayerNorm, Linear, gating,
final LayerNorm/Linear) are cheap and stay in PyTorch.  We use
FP16 loads for bandwidth, FP32 accumulation for accuracy and launch a
2‑D grid (tiles over i and j) together with a 1‑D grid over
(batch × hidden_dim).  Tile sizes (128×128) and 8 warps are tuned for
Hopper (sm_90) where each SM has 64 KB of shared memory and ample
register space.

The entry point `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ where A,B have shape (B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem dimensions
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Computes, for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B are N×N)
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head index

    batch = pid_hb // H
    head  = pid_hb % H

    # -------------------------------------------------
    # Offsets for this tile
    # -------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # -------------------------------------------------
    # Main reduction over K
    # -------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A : (TILE_M, TILE_K)  -> left[b, h, i, k]
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + head * stride_a_head
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Bᵀ : we need B[j, k] -> load as (TILE_K, TILE_N)
        # NOTE: stride_b_j is the stride of original “j” dimension,
        #       stride_b_i the stride of original “i”.  Swapping them
        #       yields the transpose.
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + head * stride_b_head
            + offs_k[:, None] * stride_b_j   # k as first dimension (original j)
            + offs_n[None, :] * stride_b_i   # n (j) as second dimension (original i)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # dot‑product over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) -> (TILE_M,TILE_N)

    # -------------------------------------------------
    # Write result
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + head * stride_c_head
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool / float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (biases are stored with the same key)
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False) and gating
    # ------------------------------------------------------------------
    # Linear: (B,N,N,dim) @ (hidden_dim, dim).T → (B,N,N,hidden_dim)
    left  = F.linear(x, w("left_proj.weight"))
    right = F.linear(x, w("right_proj.weight"))

    if not nomask:
        # mask shape -> [B, N, N, 1] for broadcasting
        mask_ = mask.to(dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Heavy contraction via Triton (batched mat‑mul)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # allocate output in FP32 (same dtype as final result)
    out_t = torch.empty_like(left_t, dtype=dtype)   # [B, H, N, N]

    # strides in elements (not bytes) – required by Triton
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j)
    c_s = out_t.stride()     # (batch, head, i, j)

    # Tile sizes – chosen to fit into H200 shared memory / registers
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows i
        triton.cdiv(N, TILE_N),          # cols j
        B * hidden_dim,                  # batch * head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0],
        a_s[1],
        a_s[2],
        a_s[3],
        b_s[0],
        b_s[1],
        b_s[2],
        b_s[3],
        c_s[0],
        c_s[1],
        c_s[2],
        c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,        # 8 warps = 256 threads, good utilization on Hopper
    )

    # back to original layout [B, N, N, hidden_dim]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4. Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate          # broadcasting over hidden_dim
    out = F.linear(out, w("to_out.weight"))   # → [B, N, N, dim]

    return out.to(dtype)
