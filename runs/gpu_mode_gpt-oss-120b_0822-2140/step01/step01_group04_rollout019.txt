"""
TriMul “outgoing” kernel (AlphaFold‑3).

The dominant cost is the 3‑dimensional contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is a batched matrix‑multiply per hidden dimension `d`.  We implement
this contraction with a Triton kernel that processes a (TILE_M × TILE_N)
block of the output for a single (batch, head) pair.  All other
operations (LayerNorm, Linear, gating, final LayerNorm/Linear) stay in
PyTorch because they are negligible compared with the O(N³) work.

The kernel works on tensors of shape (B, H, N, N) stored in row‑major
order (contiguous).  Inputs are cast to FP16 for bandwidth, the
accumulation is performed in FP32, and the result is written back as
FP32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ   (A,B : (B, H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
    BLOCK_K: tl.constexpr,             # size of the inner K loop per iteration
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ   (both A and B have shape (N, N))
    """
    pid_m = tl.program_id(0)           # row tile   (i)
    pid_n = tl.program_id(1)           # col tile   (j)
    pid_hb = tl.program_id(2)          # combined batch*head

    batch = pid_hb // H
    head  = pid_hb % H

    # ------------------- tile offsets -----------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------- K‑loop -----------------------
    # Loop in chunks of BLOCK_K to keep shared memory usage low.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load a block of A (shape TILE_M × BLOCK_K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load a block of Bᵀ.
        # B has shape (batch, head, i, j) where i == k coordinate,
        # we need B[j, k] → load as (BLOCK_K, TILE_N)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + cur_k[:, None] * stride_b_j   # k as first dim
                  + offs_n[None, :] * stride_b_i) # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Dot product over K dimension
        acc += tl.dot(a, b)   # (TILE_M,BLOCK_K) @ (BLOCK_K,TILE_N) → (TILE_M,TILE_N)

    # ------------------- store result -----------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N]     (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False) and optional mask
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,H]

    if not cfg.get("nomask", True):
        # mask shape -> [B,N,N,1] to broadcast over hidden dim
        mask_ = mask.to(inp.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy O(N³) contraction via Triton
    #    Rearrange to (B, H, N, N) contiguous layout
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B,H,N,N]

    out_t = torch.empty_like(left_t, dtype=torch.float32)   # [B,H,N,N]

    # strides in *elements* (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)  – we will load transposed
    c_s = out_t.stride()    # (batch, head, i, j)

    # Choose tile sizes. 128 works well for N up to 1024 on Hopper.
    # For very small N we fall back to the size itself to avoid over‑launch.
    TILE_M = min(128, N)
    TILE_N = min(128, N)
    TILE_K = min(128, N)
    # inner block for the K loop (helps register pressure)
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        BLOCK_K=BLOCK_K,
        num_warps=8,                     # Hopper is happy with 8 warps
        num_stages=3,                    # pipeline depth
    )

    # back to original layout: [B,N,N,H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out.to(torch.float32)
