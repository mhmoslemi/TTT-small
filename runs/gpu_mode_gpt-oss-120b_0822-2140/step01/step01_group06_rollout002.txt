"""
TriMul (outgoing) – fast forward pass.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply for every head d (hidden_dim) and batch b.
We fuse this O(N³·H) part into a single Triton GEMM kernel that works on
FP16 inputs and accumulates in FP32.  All other operations (LayerNorm,
Linear, sigmoids, the final gating and output projection) stay in PyTorch.
The kernel receives the exact strides of the (B·H, N, N) matrices, so no
flattening is required and memory access is fully coalesced.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) with FP16 loads / FP32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers for A, B, C
    M, N, K,                           # matrix dimensions (all equal to seq_len)
    stride_a_batch, stride_a_row, stride_a_col,   # strides for A
    stride_b_batch, stride_b_row, stride_b_col,   # strides for B (not transposed in memory)
    stride_c_batch, stride_c_row, stride_c_col,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[batch] = A[batch] @ B[batch].T   (A,B,C are (M,N) matrices)"""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # ---- offsets within the current block --------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension -------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block : (M, K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float16)

        # Bᵀ block : we need Bᵀ[k, n] = B[n, k]
        b = tl.load(
            b_batch + offs_n[None, :] * stride_b_row + offs_k[:, None] * stride_b_col,
            mask=mask_n[None, :] & mask_k[:, None],
            other=0.0,
        )
        b = b.to(tl.float16)

        # accumulate
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- write back -------------------------------------------------------
    c = acc.to(tl.float32)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]        (float32, CUDA)
        * mask_tensor  : [B, N, N] (may be None)
        * weights_dict : model parameters
        * config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, w["right_proj.weight"])  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))   # keep for later

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction: batched GEMM in Triton
    #    reshape to (B*H, N, N) → treat each (b,head) as a separate matmul
    # --------------------------------------------------------------
    # move hidden dim to second axis and make tensors contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    total_batch = B * hidden
    left  = left.view(total_batch, N, N)
    right = right.view(total_batch, N, N)

    # cast to fp16 for the kernel (the kernel reads fp16 and accumulates fp32)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # allocate output in fp32
    out = torch.empty_like(left, dtype=torch.float32)

    # strides (in element counts, not bytes)
    stride_a_batch = left_f16.stride(0)
    stride_a_row   = left_f16.stride(1)
    stride_a_col   = left_f16.stride(2)

    stride_b_batch = right_f16.stride(0)
    stride_b_row   = right_f16.stride(1)   # original row stride (j)
    stride_b_col   = right_f16.stride(2)   # original col stride (k)

    stride_c_batch = out.stride(0)
    stride_c_row   = out.stride(1)
    stride_c_col   = out.stride(2)

    # block sizes – chosen to fit H200 SMs and give good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_f16, right_f16, out,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, H, N, N) → (B, N, N, H)
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )   # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise gate (out_gate has shape [B,N,N,hidden])
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B, N, N, dim]

    return out
