"""
TriMul “outgoing” kernel (AlphaFold3) – Triton + PyTorch hybrid.

The expensive O(N³) contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed as a batched matrix‑multiply per hidden dimension `d`:

    out[b, d, i, j] = left_t[b, d, i, :] @ right_t[b, d, j, :].T

where left_t/right_t are (B, H, N, N) tensors (H = hidden_dim).  The
kernel `trimul_matmul_kernel` computes one TILE_M × TILE_N block of this
product using fp16 loads and fp32 accumulation.  All surrounding
operations (LayerNorm, Linear, gating, final LayerNorm/Linear) stay in
PyTorch because they are negligible compared with the N³ contraction.

The implementation works for any batch size (1‑2), sequence length N
(≤ 1024), model dimension `dim` (128 or 384) and hidden dimension
`hidden_dim` (128).  Masking is optional.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    N, H,                               # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    For each (b, h) compute
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B are (N, N) matrices).
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # offsets inside the tile
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A[b, h, i, k]  -> shape (TILE_M, TILE_K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[b, h, j, k]  (j is column, k is row)
        # load as (TILE_K, TILE_N)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dimension
                  + offs_n[None, :] * stride_b_i)  # j as second dimension
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # store result
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be all‑ones
        - weights      : dict mapping the parameter names used in the reference
                         implementation to torch.Tensor objects.
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (True → skip mask multiplication).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the result of TriMul.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helpers to fetch weight tensors
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the model dimension (last axis)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))       # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))      # [B,N,N,hidden_dim]

    if not nomask:
        # mask is [B,N,N]; broadcast on hidden dim
        mask_ = mask.unsqueeze(-1).to(dtype)        # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear layers, no bias)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction – Triton mat‑mul
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for the kernel; make contiguous for simple strides
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in elements (not bytes) – required by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j) – we will transpose inside kernel
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile sizes – tuned for Hopper (sm_90) – 64 works well for our N range
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # combined batch * head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,                     # 4 warps → good occupancy on H200
    )

    # back to original layout: [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating, and projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))  # [B, N, N, dim]

    return out.to(dtype)
