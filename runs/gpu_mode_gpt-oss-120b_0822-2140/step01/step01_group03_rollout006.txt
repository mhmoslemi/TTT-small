"""
TriMul “outgoing” kernel (AlphaFold3) – Triton‑accelerated forward pass.

The heavy part of the TriMul module is the contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be seen as a batched matrix multiplication
    C = A @ Bᵀ
with A,B ∈ ℝ^{(B·H) × N × N}.  This file:
  1. Performs the two LayerNorms and all linear / gating ops in PyTorch.
  2. Packs the hidden dimension as an extra batch dimension,
     converts to fp16 and calls a Triton kernel that computes the GEMM
     using fp16 loads, fp32 accumulation and a 128×128×32 tiling.
  3. Restores the original shape and applies the output LayerNorm,
     out‑gate and final linear projection.

The kernel works for:
  - batch size B ∈ {1,2}
  - sequence length N up to 1024
  - hidden_dim = 128 (required by the benchmark)
  - optional mask (broadcasted on the hidden dimension)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel : batched GEMM C = A @ Bᵀ
#   A : (batch, M, K)   fp16
#   B : (batch, N, K)   fp16   (we read it as K×N to get the transpose)
#   C : (batch, M, N)   fp32
# ----------------------------------------------------------------------
@triton.jit
def _trm_outgoing_gemm_T_kernel(
    A_ptr,                      # *fp16   (batch, M, K)
    B_ptr,                      # *fp16   (batch, N, K)   will be accessed as K×N
    C_ptr,                      # *fp32   (batch, M, N)
    M, N, K,                    # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (see docstring)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(2)          # batch * hidden_dim
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer arithmetic
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    # B is read as (K, N) – we therefore step with stride_Bk along K
    # and stride_Bn along N.
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk
        + offs_n[None, :] * stride_Bn
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # cast to fp32 for the dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)   # (M×K)·(K×N) → (M×N)

        # advance K‑dim pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write results
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float, may be None)
        - weights      : dict of torch.Tensor (all float32)
        - config       : dict with keys
            * dim          – input channel dimension C
            * hidden_dim   – hidden dimension H
            * nomask       – bool, True ⇒ mask is ignored

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) → (B, N, N, 1)
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    # bring hidden dim into the batch dimension
    # shape → (B, H, N, N) → (B*H, N, N) contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel for bandwidth
    left_fp16 = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides in *elements* (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (B*H*N*N, N, 1)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride() # (B*H*N*N, N, 1)

    # For Bᵀ we need:
    #   stride_Bk = stride along original column (fastest dim) = stride_Bc (=1)
    #   stride_Bn = stride along original row   = stride_Br (=N)
    stride_Bk = stride_Bc
    stride_Bn = stride_Br

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # -----------------------------------------------------------------------------
    # Tiling – chosen for Hopper (SM90). 128×128 tiles give good occupancy up to N=1024.
    # -----------------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),        # rows
        triton.cdiv(N, BLOCK_N),        # cols
        B * H                           # batch * hidden
    )

    _trm_outgoing_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)      # back to original dim C

    return out
