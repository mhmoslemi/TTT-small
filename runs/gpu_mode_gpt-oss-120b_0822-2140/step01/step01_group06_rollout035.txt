"""
TriMul (outgoing) – fast forward pass.

The only O(N³·H) work is a batched matrix‑multiply
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
We reshape the tensors so that each (batch, head) pair is a separate
(N,N)×(N,N)ᵀ GEMM.  A single Triton kernel performs a tiled fp16 GEMM
with fp32 accumulation for *all* heads in the batch.  All other
operations (layer‑norms, linear layers, sigmoid‑gates) are kept in
PyTorch.  The final element‑wise product with the output‑gate is also
fused with a tiny Triton kernel.

The kernel works for any B∈{1,2}, N∈{128,256,512,1024},
dim∈{128,384} and hidden_dim (called “H”) = 128.  It returns a float32
tensor of shape [B, N, N, dim].
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Batched GEMM kernel (fp16 inputs, fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def trimmul_gemm_kernel(
    a_ptr,                # left  (B*H, N, N)   fp16
    b_ptr,                # rightᵀ (B*H, N, N)  fp16  (already transposed)
    c_ptr,                # out   (B*H, N, N)  fp32
    BxH,                  # total number of (batch,head) pairs
    N,                    # sequence length
    stride_am, stride_ak, stride_ak2,   # a strides: (batch, row, col)
    stride_bm, stride_bk, stride_bk2,   # b strides (batch, row, col) – note rows/cols swapped
    stride_cm, stride_cn,               # c strides (batch, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)          # batch*head index
    pid_m  = tl.program_id(1)          # tile over rows   (i)
    pid_n  = tl.program_id(2)          # tile over cols   (j)

    # ------------------------------------------------------------------
    # offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for the output tile
    c_ptrs = c_ptr + pid_bh * stride_cm \
                   + offs_m[:, None] * stride_cn \
                   + offs_n[None, :] * 1

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension (the summed‑over sequence index)
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        # column/row offsets for this K‑tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # a : (M,K)   = left[batch_head, i, k]
        a_ptrs = a_ptr + pid_bh * stride_am \
                        + offs_m[:, None] * stride_ak \
                        + offs_k[None, :] * stride_ak2
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # b : (N,K)   = rightᵀ[batch_head, j, k]  (note row‑col swap)
        b_ptrs = b_ptr + pid_bh * stride_bm \
                        + offs_n[:, None] * stride_bk \
                        + offs_k[None, :] * stride_bk2
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # fp16 dot, accumulate into fp32
        tl.dot(a, b, out=acc)

    # store only the valid region
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Simple element‑wise multiplication kernel (fuse out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, N_ELEMS,
                        BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 3️⃣  Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32 CUDA)
        * mask_tensor  : [B, N, N]        (float32 or bool CUDA) – may be None
        * weights_dict : model parameters (torch tensors on CUDA)
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32 CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # called H in the paper
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(x,
                     normalized_shape=(dim,),
                     weight=w["norm.weight"],
                     bias=w["norm.bias"])

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, w["right_proj.weight"])  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask if it exists (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"])) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) – tiled GEMM in Triton
    #    We reshape to (B, H, N, N) → (B*H, N, N) and transpose right.
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*head dimension
    left_2  = left.view(B * hidden_dim, N, N)          # fp32
    right_2 = right.view(B * hidden_dim, N, N).transpose(1, 2).contiguous()  # transposed

    # cast to fp16 for the GEMM (still keep a fp32 copy for correctness)
    left_2_fp16  = left_2.to(torch.float16)
    right_2_fp16 = right_2.to(torch.float16)

    # allocate output (fp32)
    out_2 = torch.empty_like(left_2, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # launch parameters – tuned for Hopper (sm_90)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * hidden_dim,                     # batch*head dimension
        triton.cdiv(N, BLOCK_M),            # row tiles
        triton.cdiv(N, BLOCK_N),            # col tiles
    )

    # strides (in elements)
    a_stride_b = left_2_fp16.stride(0)   # B*H
    a_stride_m = left_2_fp16.stride(1)   # rows (i)
    a_stride_k = left_2_fp16.stride(2)   # cols (k)

    b_stride_b = right_2_fp16.stride(0)  # B*H
    b_stride_m = right_2_fp16.stride(1)  # rows (j) after transpose
    b_stride_k = right_2_fp16.stride(2)  # cols (k)

    c_stride_b = out_2.stride(0)
    c_stride_m = out_2.stride(1)
    c_stride_n = out_2.stride(2)

    trimmul_gemm_kernel[grid](
        left_2_fp16,               # a_ptr
        right_2_fp16,              # b_ptr (already transposed)
        out_2,                     # c_ptr
        B * hidden_dim,
        N,
        a_stride_b, a_stride_m, a_stride_k,
        b_stride_b, b_stride_m, b_stride_k,
        c_stride_b, c_stride_m, c_stride_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_2.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=w["to_out_norm.weight"],
                       bias=w["to_out_norm.bias"])

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with out_gate (fused Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)

    total = out_flat.numel()
    BLOCK = 1024
    grid_elem = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid_elem](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        BLOCK=BLOCK,
    )
    # out now holds out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B, N, N, dim]

    return out
