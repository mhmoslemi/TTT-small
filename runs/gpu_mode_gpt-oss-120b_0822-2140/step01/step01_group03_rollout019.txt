"""
TriMul “outgoing” forward pass – Triton‑accelerated core contraction.

The operation is
    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]          (k‑sum)

For each batch element and hidden dimension ``d`` this is a matrix
multiplication   (N×N) @ (N×N)ᵀ .  We fuse all (B·hidden_dim) GEMMs in a
single Triton kernel, using fp16 loads for bandwidth and fp32
accumulation for precision.  All other steps (layer‑norms, linear
projections, gating) are performed with PyTorch; they are cheap compared
to the O(N³) contraction.

The kernel works for the benchmark ranges:
    B ∈ {1,2}, N ≤ 1024, dim ∈ {128,384}, hidden_dim = 128.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A, B : [batch, M, K]), C : [batch, M, N]
# --------------------------------------------------------------
@triton.jit
def _trilinear_matmul_T_kernel(
    A_ptr,                      # *fp16   (batch, M, K)
    B_ptr,                      # *fp16   (batch, N, K)   (read as K×N)
    C_ptr,                      # *fp32   (batch, M, N)
    M, N, K,                    # matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note: Bk = original col stride)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_b = tl.program_id(2)          # batch index = B * hidden_dim
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this block
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk   # K‑dimension (original column)
        + offs_n[None, :] * stride_Bn   # N‑dimension (original row)
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)          # (BLOCK_M, BLOCK_K)

        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)          # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)        # (M×K)·(K×N) → (M×N)

        # advance pointers for next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # store result
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool/float)
        - weights      : dict of parameter tensors (all float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B, N, N, H]
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also produces out_gate (same hidden dim)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM A @ Bᵀ per hidden head
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 for better bandwidth, fp32 accumulation inside kernel
    left_fp16 = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_, dtype=torch.float32)

    # strides (in elements, not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For transposed view: Bk = original column stride, Bn = original row stride
    stride_Bk = stride_Bc
    stride_Bn = stride_Br
    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # --------------------------------------------------------------
    # Tiling – these values work well on Hopper/SM90 for N ≤ 1024
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * hidden
    )

    _trilinear_matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: layer‑norm → out‑gate → final linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to original channel dimension C

    return out
