"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with a fused
batched GEMM kernel.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reformulated as M = B * hidden independent GEMMs:
    (N×N) @ (N×N)ᵀ  →  (N×N)
and executed by a single Triton kernel.  The kernel works on
FP16 inputs (to cut memory traffic) while accumulating in FP32
(tensor‑core friendly).  All LayerNorms, Linear projections,
sigmoids and masking stay in PyTorch – they are negligible compared
to the O(N³) work.

The implementation follows the proven “permute‑to‑(B,H,N,N) → contiguous”
pattern, uses correct stride handling and launches a (128×128) tile
with K‑tile = 32, which matches the H200 tensor‑core sweet‑spot.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    N, M,                              # N = seq_len, M = B*hidden
    stride_ah, stride_ai, stride_ak,   # A strides (M, N, N)
    stride_bh, stride_bj, stride_bk,   # B strides (M, N, N)
    stride_ch, stride_ci, stride_cj,   # C strides (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance (pid_z) works on one (batch, hidden) slice.
    The kernel computes C = A @ Bᵀ where A,B ∈ (N,N) and C ∈ (N,N).
    A,B are FP16, C is FP32.
    """
    pid_z = tl.program_id(2)        # flattened batch*hidden
    pid_m = tl.program_id(0)        # block row   (i)
    pid_n = tl.program_id(1)        # block col   (j)

    # Offsets for the current 2‑D tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch, hidden) matrix pair
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension tiled by BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai      # row i
            + offs_k[None, :] * stride_ak      # col k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load Bᵀ[k, j]   shape (BLOCK_K, BLOCK_N)
        # Original B is indexed as B[j, k]; we load it transposed.
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk      # row k of Bᵀ
            + offs_n[None, :] * stride_bj      # col j of Bᵀ
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # FP16 dot → FP32 accumulation (tensor cores)
        acc += tl.dot(a, b)

    # Store the result (FP32)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, H, N, N)  FP16
    returns      : (B, H, N, N)  FP32 where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                     # flattened batch*head

    # reshape to (M, N, N) for stride simplicity
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty((M, N, N), dtype=torch.float32, device=left.device)

    # strides (int64) – note that after reshaping the logical order is
    # (M, N, N) → (batch*head, row, col)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tile sizes tuned for Hopper (SM90) – good occupancy & tensor‑core usage
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z (batch*head)
    )

    trimul_gemm_fp16_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        M,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps per block works well on H200
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Heavy contraction is executed by a fused Triton GEMM kernel;
    all other ops (LayerNorm, Linear, sigmoid, masking) are done in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections → hidden_dim (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)    # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ per hidden channel
    #    Convert layout to (B, H, N, N) and cast to FP16 for the kernel.
    # --------------------------------------------------------------
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_fp32 = trimul_gemm_fp16(left_fp16, right_fp16)                    # (B, H, N, N) FP32

    # Back to original layout (B, N, N, H)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)          # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                           # (B, N, N, dim)

    return out.to(torch.float32)
