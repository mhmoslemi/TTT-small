"""
TriMul “outgoing” forward pass – Triton‑fused batched GEMM.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication C = A @ Bᵀ where
    A = left .permute(0,3,1,2)  → (B, H, N, N)
    B = right.permute(0,3,1,2) → (B, H, N, N)

A and B are flattened to (B·H, N, N) and multiplied in FP16 with FP32
accumulation inside a Triton kernel.  All other ops (LayerNorm, Linear,
gates, final projection) are cheap and stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, M, K], [batch, N, K])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dims
    stride_az, stride_am, stride_ak,         # A strides (batch, row, col)
    stride_bz, stride_bk, stride_bn,         # B strides (batch, col, row) – original layout
    stride_cz, stride_cm, stride_cn,         # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed."""
    pid_batch = tl.program_id(2)          # which (B·H) matrix
    pid_m = tl.program_id(0)              # block‑row index
    pid_n = tl.program_id(1)              # block‑col index

    # ------------------------------------------------------------------
    # Offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile in A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_az \
                   + offs_m[:, None] * stride_am \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K). To read Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # FP32 accumulator (more stable than pure FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # dot: FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (down‑cast to FP16 to save bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ on (B·H, N, N) tensors in FP16.
    Returns C of the same shape / dtype (FP16).
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "square matrices required"

    c = torch.empty_like(a)

    # strides in **elements** (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # original layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tiling – these work well on Hopper/SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B_H                               # batch dimension (B*H)
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for 128×128 tiles
        num_stages=2,
    )
    return c


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a Triton‑fused
    batched GEMM.  Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            mask         : torch.Tensor [B, N, N]       (float32/bool)
            weights      : dict of torch.Tensor (float32)
            config       : dict with keys
                "dim"          – input channel dimension
                "hidden_dim"   – projection dimension
                "nomask"       – if True the mask can be ignored

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Rearrange to (B, H, N, N) and cast to FP16 for the GEMM
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM C = left @ rightᵀ   (still FP16)
    # --------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(left, right)    # (B*H, N, N)

    # --------------------------------------------------------------
    # 7. Restore original layout and cast back to FP32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Post‑projection LayerNorm + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # out_gate shape (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 9. Final linear projection back to the original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
