"""
TriMul “outgoing” kernel (AlphaFold‑3).

The costly step is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension *d* (i.e. A @ Bᵀ for
each (batch, d) pair).  The surrounding LayerNorm / Linear / gating ops are cheap
and stay in PyTorch.  The Triton kernel below computes the contraction with
FP16 loads and FP32 accumulation, using a 2‑D tiling (M×N) and a K‑loop.
Tile sizes (128,128,64) and 8 warps give good occupancy on Hopper (sm_90).

The host function `custom_kernel` performs the full forward pass and returns a
float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B, H, N, N) @ (B, H, N, N)ᵀ  →  (B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    (i.e. A @ Bᵀ)  A and B have shape (N, N), output C has shape (N, N).
    """

    pid_m = tl.program_id(0)                # row tile index   (i)
    pid_n = tl.program_id(1)                # col tile index   (j)
    pid_hb = tl.program_id(2)               # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------- tile offsets -------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------- K loop -------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A[b, h, i, k]  → shape (TILE_M, TILE_K)
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + head  * stride_a_head
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[b, h, j, k] → treat as (TILE_K, TILE_N)
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + head  * stride_b_head
            + offs_k[:, None] * stride_b_j   # k as first dim
            + offs_n[None, :] * stride_b_i   # j as second dim
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K (fp16 * fp16 → fp32)
        acc += tl.dot(a, b)    # (TILE_M,TILE_K) @ (TILE_K,TILE_N)

    # ------------------- store result -------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + head  * stride_c_head
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32/bfloat16)
        - mask         : torch.Tensor of shape [B, N, N]  (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) and gating
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])      # [B,N,N, H]
    right = F.linear(x, weights["right_proj.weight"])     # [B,N,N, H]

    # apply optional mask (mask is 0/1 float)
    if not config.get("nomask", True):
        mask_f = mask.to(dtype).unsqueeze(-1)            # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Heavy contraction: batched mat‑mul per hidden dim using Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous (row‑major)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in **elements** (not bytes)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we load transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # tile sizes – chosen to fit into shared memory on Hopper
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,
        num_stages=3,                     # pipeline depth (good for large N)
    )

    # back to original layout [B,N,N,H]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4. Final LayerNorm + gating + linear back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out.to(dtype)
