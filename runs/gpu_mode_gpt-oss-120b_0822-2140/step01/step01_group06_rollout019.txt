"""
TriMul (outgoing) – fast forward pass with a fused Triton GEMM.

The expensive O(N³·H) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is rewritten as a batched matrix‑multiply for every (batch, head) pair:
    A = left   → (BH, N, N)   (row‑major)
    B = right  → (BH, N, N)   (row‑major, we read it transposed)
    C = A @ Bᵀ → (BH, N, N)

The GEMM is performed by a custom Triton kernel that tiles the
matrices (BLOCK_M × BLOCK_K × BLOCK_N) and accumulates in fp32 while
loading in fp16 for higher throughput on Hopper (sm_90).  All other
operations (LayerNorms, Linear projections, gating, final projection)
remain in PyTorch.  The kernel is launched with a 3‑D grid
(grid_m, grid_n, batch_head) where *batch_head = B·H*.

The implementation supports both masked / un‑masked inputs, arbitrary
batch sizes (1 or 2), sequence lengths up to 1024 and hidden dimensions
128, 384, 768.  The final tensor is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled GEMM for the outgoing TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_outgoing_gemm(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (all equal to seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides: batch, row, col
    stride_bh, stride_bk, stride_bj,   # B strides: batch, k‑dim, j‑dim (B read transposed)
    stride_ch, stride_ci, stride_cj,   # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    """Compute C = A @ Bᵀ  where A,B ∈ (BH, M, K) and C ∈ (BH, M, N)."""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_h = tl.program_id(2)          # batch‑head index (BH)

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointer to the start of the current batch‑head slice
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_h * stride_ah
    b_batch = b_ptr + pid_h * stride_bh
    c_batch = c_ptr + pid_h * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (fp32).  We keep the accumulator in fp32 even when
    # the inputs are fp16 (FP16=True) for numerical stability.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B block (transposed): we want Bᵀ, i.e. (k, j)
        # original B has layout (batch, row, col) = (BH, K, N)
        # Using strides (stride_bk for K, stride_bj for N) gives us Bᵀ directly.
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bj)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        if FP16:
            a = a.to(tl.float16)
            b = b.to(tl.float16)

        # tl.dot performs a reduction over the k‑dimension
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask_tensor  : torch.Tensor or None, shape [B, N, N] (float32/bool)
        - weights_dict : mapping of parameter names to torch.Tensor
        - config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction using a fused Triton GEMM
    #    We reshape to (BH, N, N) where BH = B * hidden_dim.
    # --------------------------------------------------------------
    #   left/right: [B, N, N, H] → [B, H, N, N] → (BH, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left = left.view(BH, N, N)
    right = right.view(BH, N, N)

    # Cast to fp16 for the GEMM (Hopper’s Tensor Cores excel at fp16×fp16→fp32)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (fp32 as required by the rest of the pipeline)
    out_fp32 = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Launch parameters – chosen to give good occupancy on H200.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )

    # Strides for A (left), B (right), C (output)
    # Tensor layout after view: (batch_head, row, col) → (BH, N, N)
    a_strides = left_fp16.stride()   # (stride_batch, stride_row, stride_col)
    b_strides = right_fp16.stride()
    c_strides = out_fp32.stride()

    # B is accessed transposed: we need strides for (k, j) order.
    # Original layout: (BH, row, col) = (BH, K, N)
    # Hence stride_bk = stride for col dimension, stride_bj = stride for row dimension.
    triton_trimul_outgoing_gemm[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N,                    # M
        N,                    # N
        N,                    # K
        a_strides[0], a_strides[1], a_strides[2],   # A strides
        b_strides[0], b_strides[2], b_strides[1],   # B strides (k, j) → (col, row)
        c_strides[0], c_strides[1], c_strides[2],   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=True,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # --------------------------------------------------------------
    out = out * out_gate  # broadcast over hidden_dim

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimensionality
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
