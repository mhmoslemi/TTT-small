"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The expensive contraction  

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is computed as a batched GEMM on the GPU with a custom Triton kernel.
All other operations (LayerNorm, linear projections, gates, mask handling) are
performed in PyTorch – the kernel alone provides the required Triton work and
removes the N³ bottleneck.

Algorithm
---------
1.  Layer‑norm on the input `x` (dim → dim).
2.  Project to the hidden dimension `H = hidden_dim`:
        left  = x @ W_left   ,   right = x @ W_right   (no bias)
3.  Optional binary mask (broadcast over the hidden dim) – pure PyTorch.
4.  Gating: sigmoid( x @ W_gate )   →   left_gate, right_gate, out_gate.
5.  Fuse gates:   left  *= left_gate ,   right *= right_gate.
6.  Reshape to (B, H, N, N) and launch a Triton GEMM kernel that
    computes, for every (b, h) pair,   L @ Rᵀ   where
    L = left[b, h, :, :]   ,   Rᵀ = right[b, h, :, :].T .
    The kernel works in bfloat16 for bandwidth and accumulates in fp32.
7.  Reshape result back to (B, N, N, H).
8.  Output Layer‑norm (hidden_dim → hidden_dim) → multiply by `out_gate`.
9.  Final linear projection back to `dim`.

The kernel is written for three‑dimensional grid:
    (num_blocks_M, num_blocks_N, B*H)   →   each program processes one
    (batch, head) pair independently.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def trimmul_gemm_kernel(
    # pointers
    left_ptr,          # bf16*  (B*H, M, K)   →  rows=M=i, cols=K=k
    right_ptr,         # bf16*  (B*H, K, N)   →  rows=K=k, cols=N=j (already transposed)
    out_ptr,           # fp32*  (B*H, M, N)   →  rows=M=i, cols=N=j
    # matrix sizes
    M, N, K,           # int32
    # strides
    stride_l_batch, stride_l_row, stride_l_col,   # left  (batch, row, col)
    stride_r_batch, stride_r_row, stride_r_col,   # right (batch, row, col)  row==k, col==j
    stride_o_batch, stride_o_row, stride_o_col,   # out   (batch, row, col)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------ #
    # block identifiers
    # ------------------------------------------------------------------ #
    pid_m = tl.program_id(0)               # block row index
    pid_n = tl.program_id(1)               # block col index
    pid_z = tl.program_id(2)               # (batch * head) index

    # ------------------------------------------------------------------ #
    # starting offsets for this block
    # ------------------------------------------------------------------ #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices

    # compute base pointers for the (b,h) slice
    left_base  = left_ptr  + pid_z * stride_l_batch
    right_base = right_ptr + pid_z * stride_r_batch
    out_base   = out_ptr   + pid_z * stride_o_batch

    # ------------------------------------------------------------------ #
    # accumulator (fp32)
    # ------------------------------------------------------------------ #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------ #
    # loop over the reduction axis k
    # ------------------------------------------------------------------ #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile  (M x K)  →  left[i,k]
        a = tl.load(
            left_base
            + (offs_m[:, None] * stride_l_row)
            + (offs_k[None, :] * stride_l_col),
            mask=mask_k[None, :],
            other=0.0
        )
        # load B tile  (K x N)  →  right[k,j]
        b = tl.load(
            right_base
            + (offs_k[:, None] * stride_r_row)
            + (offs_n[None, :] * stride_r_col),
            mask=mask_k[:, None],
            other=0.0
        )
        # dot‑product (uses fp16·fp16→fp32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------ #
    # write back (with boundary masks)
    # ------------------------------------------------------------------ #
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_o = mask_m[:, None] & mask_n[None, :]

    tl.store(
        out_base
        + (offs_m[:, None] * stride_o_row)
        + (offs_n[None, :] * stride_o_col),
        acc,
        mask=mask_o
    )


# --------------------------------------------------------------------------- #
# Helper: element‑wise mask multiplication (used only when a mask is present)
# --------------------------------------------------------------------------- #
@triton.jit
def mask_mul_kernel(src_ptr, mask_ptr, dst_ptr, N_ELEM, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < N_ELEM
    src = tl.load(src_ptr + offs, mask=m, other=0.0)
    mask = tl.load(mask_ptr + offs, mask=m, other=0.0)
    tl.store(dst_ptr + offs, src * mask, mask=m)


def apply_binary_mask(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """tensor, mask have identical shape. Returns tensor*mask."""
    numel = tensor.numel()
    BLOCK = 131072                     # 128 K works for all test sizes
    grid = (triton.cdiv(numel, BLOCK),)
    out = torch.empty_like(tensor)
    mask_mul_kernel[grid](
        src_ptr=tensor,
        mask_ptr=mask,
        dst_ptr=out,
        N_ELEM=numel,
        BLOCK=BLOCK,
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]    (float32, CUDA)
        - mask         : [B, N, N]         (float32 0/1, CUDA)
        - weights      : dict of torch.Tensor containing all linear / LN params
        - config       : dict with keys
              * dim        – input channel size
              * hidden_dim – projection size
              * nomask     – bool, whether mask is all‑ones

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # helpers – LayerNorm & Linear (bias‑free)
    # ------------------------------------------------------------------- #
    def layer_norm(tensor, w, b):
        return F.layer_norm(
            tensor,
            normalized_shape=(w.shape[0],),
            weight=w,
            bias=b,
            eps=1e-5,
        )

    def linear(tensor, w):
        return F.linear(tensor, w)            # bias=False in all layers

    # ------------------------------------------------------------------- #
    # 1️⃣ Input norm
    # ------------------------------------------------------------------- #
    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------- #
    # 2️⃣ Project to hidden dim
    # ------------------------------------------------------------------- #
    left = linear(x, weights["left_proj.weight"])    # (B,N,N,H)
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask shape (B,N,N) → (B,N,N,1) → broadcast
        mask_b = mask.unsqueeze(-1).expand_as(left).contiguous()
        left = apply_binary_mask(left, mask_b)
        right = apply_binary_mask(right, mask_b)

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # (B,N,N,H)
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM in Triton
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    H = hidden_dim

    # reshape to (B, H, N, N) for a convenient batch*head view
    left = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()

    # right needs to be transposed on the last two axes (k ↔ j)
    right_T = right.transpose(-1, -2).contiguous()

    # bf16 for bandwidth, fp32 accumulator in the kernel
    left_b = left.to(torch.bfloat16)
    right_b = right_T.to(torch.bfloat16)

    # output buffer (fp32)
    out_b = torch.empty((B, H, N, N), dtype=torch.float32, device=x.device)

    # strides (in elements, not bytes)
    stride_l_batch, stride_l_head, stride_l_row, stride_l_col = left_b.stride()
    stride_r_batch, stride_r_head, stride_r_row, stride_r_col = right_b.stride()
    stride_o_batch, stride_o_head, stride_o_row, stride_o_col = out_b.stride()

    # tile sizes – chosen to fit H200 SM architecture
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * H,                             # batch * head
    )

    trimmul_gemm_kernel[grid](
        left_b,                     # ptr
        right_b,                    # ptr
        out_b,                      # ptr
        M=N, N=N, K=N,              # matrix sizes
        stride_l_batch=stride_l_batch,
        stride_l_row=stride_l_row,
        stride_l_col=stride_l_col,
        stride_r_batch=stride_r_batch,
        stride_r_row=stride_r_row,
        stride_r_col=stride_r_col,
        stride_o_batch=stride_o_batch,
        stride_o_row=stride_o_row,
        stride_o_col=stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
