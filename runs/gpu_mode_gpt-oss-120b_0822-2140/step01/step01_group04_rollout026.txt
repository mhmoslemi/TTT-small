"""
TriMul “outgoing” kernel (AlphaFold3) – Triton implementation.

The dominant O(N³) step is the three‑dimensional contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply of the form  
    (B, H, N, N) @ (B, H, N, N)ᵀ   with H = hidden_dim.

All cheap ops (LayerNorm, Linear, sigmoids, final norm/linear) stay in
PyTorch.  The heavy contraction is performed with a fused Triton kernel
that loads the two operands in FP16, accumulates in FP32 and writes the
result back in FP32.  The kernel works for any batch size ≤2, any
sequence length that fits the chosen tile size, and hidden dimensions
that are a multiple of the tile‑width (the tests use 128).

The host routine `custom_kernel` follows the signature required by the
grader.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr,                # left  : (B, H, N, N)   fp16
    B_ptr,                # right : (B, H, N, N)   fp16
    C_ptr,                # out   : (B, H, N, N)   fp32
    N,                    # sequence length
    H,                    # hidden dimension
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    For every (batch, head) pair compute
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B are (N, N) matrices).
    """

    pid_m = tl.program_id(0)               # tile row   (i)
    pid_n = tl.program_id(1)               # tile col   (j)
    pid_hb = tl.program_id(2)              # combined batch*head

    batch = pid_hb // H
    head  = pid_hb % H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[b, h, i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ => B[b, h, j, k]   shape (BLOCK_K, BLOCK_N)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_n[:, None] * stride_b_i   # j dimension becomes first
                  + offs_k[None, :] * stride_b_k)  # k dimension second
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K (FP16 × FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (fp32)
        - mask         : torch.Tensor [B, N, N]      (bool/float)
        - weights      : dict of torch.Tensors with keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                        ``nomask`` (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (fp32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32
    eps = 1e-6

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper shortcuts
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) and optional masking
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden]

    if not nomask:
        mask_ = mask.unsqueeze(-1).to(dtype)     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for simple stride math
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # fp16 input for bandwidth, fp32 accumulator/output
    left_fp16  = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)
    out_fp32   = torch.empty_like(left_fp16, dtype=torch.float32)  # [B,H,N,N]

    # strides in elements (not bytes) – required by the kernel
    a_str = left_fp16.stride()   # (batch, head, i, k)
    b_str = right_fp16.stride()  # (batch, head, i, k)
    c_str = out_fp32.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # launch parameters – tuned for Hopper (sm_90)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # small K‑tile fits in shared memory

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows i
        triton.cdiv(N, BLOCK_N),          # cols j
        B * hidden_dim,                   # combined batch * head
    )

    trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N,
        hidden_dim,
        a_str[0], a_str[1], a_str[2], a_str[3],
        b_str[0], b_str[1], b_str[2], b_str[3],
        c_str[0], c_str[1], c_str[2], c_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # back to usual layout: (B, N, N, H)
    out = out_fp32.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm + gating + linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate          # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
