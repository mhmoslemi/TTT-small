```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

```python
"""
TriMul “outgoing” implementation.
The expensive part is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply of shape (B·D, N, N) with the
transpose of the second operand.
We fuse this with a Triton kernel (left @ rightᵀ) and keep the
surrounding LayerNorm / Linear / gating logic in PyTorch for simplicity.
The kernel works in float32 (accumulation) and stores the result in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ   where left/right have shape
# (M, N, K)  with K == N.  M = batch * hidden_dim.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float32   [M,N,K]
    right_ptr,         # *float32   [M,N,K]
    out_ptr,           # *float32   [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile over rows   (i)
    pid_n = tl.program_id(1)          # tile over cols   (j)
    pid_z = tl.program_id(2)          # different matrix (batch*channel)

    # -------------------- offsets & masks --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------------------- pointers --------------------
    # left:   (M,N,K)   -> index (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                        + offs_m[:, None] * stride_ln \
                        + offs_k[None, :] * stride_lk

    # right transposed on the last two dims:
    # we need Bᵀ where B(z, j, k) -> we access as (z, k, j)
    b_ptrs = right_ptr + pid_z * stride_rm \
                         + offs_k[:, None] * stride_rk \
                         + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- main GEMM loop --------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # -------------------- store result --------------------
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    TriMul forward pass with a fused Triton GEMM for the N³ term.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask         : [B, N, N]        (float32/bool, 0/1)
            - weights      : dict of TT tensors
            - config       : {"dim": int, "hidden_dim": int}
    Returns:
        out : [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Fuse the N³ einsum with Triton:
    #    reshape to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------- #
    # (B,N,N,hidden) -> (B, hidden, N, N) -> (M, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_reshape = torch.empty_like(left_reshape, device=device)   # same dtype (fp32)

    # strides (in elements) for the three tensors
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * hidden_dim               # independent matrices
    )

    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden_dim, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # good default for fp32 on Hopper
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
```
