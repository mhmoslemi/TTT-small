"""
TriMul “outgoing” kernel (AlphaFold3).

The expensive part of the module is the three‑dimensional contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We implement this contraction with a single Triton kernel that computes,
for every (batch, head) pair, the product  

    C[b,h] = A[b,h] @ B[b,h].T          (A,B : N×N,   C : N×N)

All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
are cheap and stay in PyTorch.  The kernel works on FP16 values for
bandwidth and accumulates in FP32, producing an FP32 result that matches
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)^T
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    For each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ   (both A and B have shape (N, N))
    """

    pid_m = tl.program_id(0)            # tile row   (i)
    pid_n = tl.program_id(1)            # tile col   (j)
    pid_hb = tl.program_id(2)           # combined batch*head index

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in chunks of TILE_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ---- A : (TILE_M, TILE_K) ------------------------------------
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- Bᵀ : we need B[j,k] → load as (TILE_K, TILE_N) -----------
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dim
                  + offs_n[None, :] * stride_b_i)  # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K (FP16 * FP16 → FP32 accumulate)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensor           model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # convenience for weight lookup
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden_dim]

    if not nomask:
        # mask shape -> [B,N,N,1] to broadcast on hidden_dim
        mask_ = mask.unsqueeze(-1).to(dtype)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction via Triton
    #    reshape to (B, H, N, N) – contiguous in memory
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in elements (not bytes) – required by the kernel
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will read transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # ------------------------------------------------------------------
    # launch configuration – tuned for Hopper (sm_90)
    # ------------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,                     # 8 warps give good occupancy on H200
        num_stages=3,                    # prefetch depth
    )

    # back to the original layout: [B, N, N, hidden_dim]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating and projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                      # broadcast over hidden_dim
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
