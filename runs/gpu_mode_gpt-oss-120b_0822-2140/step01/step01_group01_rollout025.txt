"""
Optimized “outgoing” TriMul kernel (AlphaFold‑3 style).

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reformulated as a batched GEMM L @ Rᵀ for each (batch, hidden) slice.
We launch a Triton kernel that computes many independent GEMMs in parallel:

    (B, H, N, N) @ (B, H, N, N)ᵀ   →   (B, H, N, N)

All cheap operations (LayerNorm, linear projections, sigmoid gating, masking)
stay in PyTorch.  The heavy GEMM is performed in fp16 (to reduce memory traffic)
while accumulation is done in fp32 for numerical stability.  The kernel uses
64‑×‑128 tiles (BLOCK_M=128, BLOCK_N=128, BLOCK_K=64) which match the H200
SM resources and give good occupancy.

The entry‑point `custom_kernel` follows the exact API required by the
evaluator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers (fp16)
    N, M,                               # N = seq_len, M = B*hidden
    stride_ah, stride_ai, stride_ak,    # A strides (M, N, N)
    stride_bh, stride_bj, stride_bk,    # B strides (M, N, N)
    stride_ch, stride_ci, stride_cj,    # C strides (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Matrix multiplication of many (N×N) matrices:
    C[m] = A[m] @ B[m]ᵀ   for m = 0 … M-1
    All tensors are stored in fp16, accumulation in fp32.
    """
    pid_m = tl.program_id(0)      # block row   (i)
    pid_n = tl.program_id(1)      # block col   (j)
    pid_z = tl.program_id(2)      # flattened batch * hidden (m)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the selected (m‑th) matrix pair
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai          # i dimension
            + offs_k[None, :] * stride_ak          # k dimension
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # Original B is indexed as B[m, j, k]; we want Bᵀ[k, j]
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk          # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj          # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # Multiply‑accumulate (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # Write result back as fp16
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci          # i dimension
        + offs_n[None, :] * stride_cj          # j dimension
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul.
    Arguments:
        left, right : (B, H, N, N) torch.float16 tensors
    Returns:
        out : (B, H, N, N) torch.float16 tensor where out = left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                     # flattened batch*head dimension

    # Flatten to (M, N, N) – this does **not** copy data
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f)   # fp16 output

    # Strides in *elements* (not bytes)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tile sizes – tuned for Hopper SMs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )
    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    All cheap ops run in PyTorch; the O(N³) contraction is executed by a
    Triton batched‑GEMM kernel defined above.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim) – no bias
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)        # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ  (per hidden channel)
    #    Convert to (B, H, N, N) layout, cast to half for bandwidth.
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm = trimul_gemm(left_perm, right_perm)          # (B, H, N, N) fp16
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out
