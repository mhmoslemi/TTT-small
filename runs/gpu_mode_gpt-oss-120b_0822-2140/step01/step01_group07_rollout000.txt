"""
TriMul “outgoing” (AlphaFold‑3) forward pass.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply (B·D, N, N) × (B·D, N, N)ᵀ.
We fuse this GEMM with a Triton kernel; everything else (LayerNorm,
Linear, sigmoids, masking, final LayerNorm & projection) stays in PyTorch.
The kernel works in FP32 (accumulation) and uses a modest tile size that
fits well on Hopper (SM‑90) for the whole range of sequence lengths in the
benchmark.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ   where left/right have shape
# (M, N, K) with K == N.  M = batch * hidden_dim.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float32   [M, N, K]
    right_ptr,         # *float32   [M, N, K]
    out_ptr,           # *float32   [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the TriMul contraction.
    Each program instance (pid_z, pid_m, pid_n) computes a BLOCK_M × BLOCK_N
    tile of the output matrix for a given (batch * hidden) slice `z`.
    """
    pid_z = tl.program_id(0)      # index over M (= B*H)
    pid_m = tl.program_id(1)      # tile row   (i)
    pid_n = tl.program_id(2)      # tile col   (j)

    # ------------------------------------------------------------------ #
    # Offsets and masks for the current tile
    # ------------------------------------------------------------------ #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------ #
    # Pointers to the start of the tile (no broadcast on the K dimension)
    # ------------------------------------------------------------------ #
    # left:  (M,N,K)  ->  index (z, i, k)
    a_ptrs = (left_ptr
              + pid_z * stride_lm
              + offs_m[:, None] * stride_ln
              + offs_k[None, :] * stride_lk)

    # right is accessed as its transpose on the last two dims:
    # we need Bᵀ where B(z, j, k) -> we read (z, k, j)
    b_ptrs = (right_ptr
              + pid_z * stride_rm
              + offs_k[:, None] * stride_rk
              + offs_n[None, :] * stride_rn)

    # ------------------------------------------------------------------ #
    # Accumulator in FP32
    # ------------------------------------------------------------------ #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------ #
    # Main K loop (tiled)
    # ------------------------------------------------------------------ #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles; out‑of‑bounds elements are zero‑filled
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate dot product of the K‑tile
        acc += tl.dot(a, b)

        # Advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------ #
    # Write the result, respecting the boundary masks
    # ------------------------------------------------------------------ #
    out_ptrs = (out_ptr
                + pid_z * stride_om
                + offs_m[:, None] * stride_on
                + offs_n[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a fused Triton GEMM.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool/float, optional)
        - weights      : dict of torch.nn.Parameter tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    out : torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) →  [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) – same shape as left/right
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Fuse the N³ einsum with Triton:
    #    reshape to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------- #
    # (B,N,N,H) → (B, H, N, N) → (M, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_reshape = torch.empty_like(left_reshape, device=device)   # FP32

    # ------------------------------------------------------------------- #
    # 6. Strides (in elements) – required by the kernel
    # ------------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # ------------------------------------------------------------------- #
    # 7. Choose tile sizes – small enough to keep shared memory cheap,
    #    large enough to give good occupancy for the whole N‑range.
    # ------------------------------------------------------------------- #
    # The same tile config works well for all benchmarked N (256‑1024).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (M‑tiles, N‑tiles, batch*head)
    grid = (
        triton.cdiv(N, BLOCK_M),          # rows  (i)
        triton.cdiv(N, BLOCK_N),          # cols  (j)
        B * hidden_dim                    # independent GEMMs
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden_dim, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for FP32 on Hopper
        num_stages=2          # double‑buffering of K‑tiles
    )

    # ------------------------------------------------------------------- #
    # 8. Back‑to original layout → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    out = (out_reshape
           .view(B, hidden_dim, N, N)
           .permute(0, 2, 3, 1)
           .contiguous())

    # ------------------------------------------------------------------- #
    # 9. Output LayerNorm + gating + final projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
