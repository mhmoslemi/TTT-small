"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The only O(N³) work is the contraction  

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which we rewrite as a batched GEMM:

    (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)

where H = hidden_dim.  The surrounding LayerNorm / Linear / gating
operations are kept in PyTorch; the GEMM is fused in a single Triton
kernel that

* works on fp16 inputs (higher arithmetic intensity on Hopper)
* accumulates in fp32 (no loss of precision)
* processes one (batch·head) matrix per program‑id‑z tile.

The kernel is deliberately simple: tiled‑blocked matrix multiply with
BLOCK_M = BLOCK_N = 64, BLOCK_K = 64 and 8 warps – a good default for
fp16 on an H200 (sm_90).  The rest of the computation follows the
reference PyTorch implementation exactly.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ   (left/right : (M,N,N), out : (M,N,N))
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16   [M,N,N]
    right_ptr,         # *fp16   [M,N,N] (will be read transposed)
    out_ptr,           # *fp32   [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rk, stride_rn,   # right strides (note: K‑dim stride first)
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # which matrix (batch * head)

    # ------------------------------------------------------------------- #
    # Offsets & masks
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers (left is (M,N,K), right will be accessed as (M,K,N))
    # ------------------------------------------------------------------- #
    a_ptrs = left_ptr + pid_z * stride_lm \
                       + offs_m[:, None] * stride_ln \
                       + offs_k[None, :] * stride_lk

    b_ptrs = right_ptr + pid_z * stride_rm \
                        + offs_k[:, None] * stride_rk \
                        + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main GEMM loop
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                     # fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                     # fp16

        # promote to fp32 for accumulation (dot works on fp16 → fp32)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Store result
    # ------------------------------------------------------------------- #
    c_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool/float, may be None)
        - weights      : dict of model weights (all float32)
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Fuse the N³ einsum with Triton.
    #    Reshape to (M, N, N) where M = B * hidden.
    # --------------------------------------------------------------- #
    # (B,N,N,hidden) → (B,hidden,N,N) → (M,N,N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden

    left_fp16  = left_t.view(M, N, N).to(torch.float16)
    right_fp16 = right_t.view(M, N, N).to(torch.float16)

    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # strides in *elements* (not bytes) for the three tensors
    stride_lm, stride_ln, stride_lk = left_fp16.stride()
    stride_rm, stride_rk, stride_rn = right_fp16.stride()
    stride_om, stride_on, stride_ok = out_fp32.stride()

    # ------------------------------------------------------------------- #
    # Kernel launch configuration (tuned for Hopper)
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),          # rows i
        triton.cdiv(N, BLOCK_N),          # cols j
        M,                                # batch * head
    )

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rk, stride_rn,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                      # 8 warps → 256 threads, good for fp16
    )

    # --------------------------------------------------------------- #
    # 6. Reshape back to (B,N,N,hidden)
    # --------------------------------------------------------------- #
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 7. Output LayerNorm, gating and final linear (dim)
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B,N,N,dim]

    return out
