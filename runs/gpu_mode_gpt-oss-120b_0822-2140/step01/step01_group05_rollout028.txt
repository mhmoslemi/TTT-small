"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused
batched matrix‑multiply kernel written in Triton.

*   LayerNorm, linear projections and gating are done in PyTorch.
*   The N³ einsum   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is realised as a batched (N×N) @ (N×N)ᵀ for every hidden channel d.
    We treat each hidden channel as an extra batch dimension:
        (B, H, N, N) → (B·H, N, N)
*   Inside Triton the two matrices are cast to FP16, multiplied with
    tensor‑core `tl.dot`, accumulated in FP32 and written back as FP16.
    The result is cast back to FP32 before the final LayerNorm / linear.
*   The kernel works for any batch size (1 or 2), sequence length
    (128 – 1024) and hidden dimension (128, 384, 768) required by the
    test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (N×N) @ (N×N)ᵀ for many (batch*hidden) matrices
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A_ptr,               # FP16 matrix A, shape (B*H, M, K)
    B_ptr,               # FP16 matrix B, shape (B*H, N, K)
    C_ptr,               # FP16 output, shape (B*H, M, N)
    M, N, K,             # matrix dimensions (M == N == K == seq_len)
    stride_Ab, stride_Am, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bm, stride_Bk,   # strides for B (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)        # index over (B * hidden)
    pid_m = tl.program_id(1)            # row tile
    pid_n = tl.program_id(2)            # col tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointer base for this batch
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_batch * stride_Bb + offs_n[None, :] * stride_Bm + offs_k[:, None] * stride_Bk

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (FP16). Masks keep us inside the matrix bounds.
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float16)

        # Tensor‑core mma: FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        # Advance K‑pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Store result (FP16) – convert accumulator back to FP16
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data : tuple (input, mask, weights, config)
            - input  : torch.Tensor [B, N, N, dim]   (float32, CUDA)
            - mask   : torch.Tensor [B, N, N]       (bool or float, CUDA) or None
            - weights: dict of torch.Tensors (all float32, CUDA)
            - config : {'dim': int, 'hidden_dim': int, ...}
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim   = config["dim"]
    hidden = config["hidden_dim"]               # = hidden_dim
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Bring hidden dimension to the batch axis: (B, H, N, N)
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Flatten batch+head so that each head is processed as an independent matrix‑multiply
    left_flat  = left.view(B * hidden, N, N)   # FP32
    right_flat = right.view(B * hidden, N, N)  # FP32

    # Cast to FP16 for the kernel (tensor‑core friendly)
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output buffer (FP16)
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------
    # 6. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                     # batch*head axis
        triton.cdiv(N, BLOCK_M),        # M‑tiles
        triton.cdiv(N, BLOCK_N),        # N‑tiles
    )

    # Strides (in elements) of the three‑dimensional tensors
    a_batch, a_m, a_k = left_fp16.stride()
    b_batch, b_m, b_k = right_fp16.stride()
    c_batch, c_m, c_n = out_fp16.stride()

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_m, a_k,
        b_batch, b_m, b_k,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to FP32 and restore original layout (B, N, N, H)
    out = out_fp16.to(torch.float32).view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (hidden dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
