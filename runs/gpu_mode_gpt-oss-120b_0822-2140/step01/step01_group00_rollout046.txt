"""
TriMul “outgoing” (AlphaFold‑3) – pure‑forward implementation.

The expensive tensor contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix multiplication   left[b,d] @ right[b,d].T
with shape (N,N) for each (batch, hidden_dim) pair.
We rewrite the 4‑D tensors as (B*H, N, N) and launch a single
Triton kernel that performs a tiled GEMM (fp16/bf16 input → fp32 accumulation).
All other linear / layer‑norm / gating work is done with PyTorch which is
tiny compared with the N³ reduction.

The kernel
* tiles the M‑ and N‑dimensions (row/col of the output matrix)
* streams through the K‑dimension (the summed axis)
* uses `tl.dot` on bf16 blocks with fp32 accumulation
* writes the result as fp32.

The rest of the module follows the reference PyTorch code, but the
mask‑application is kept in pure‑torch (very cheap) while the core
reduction is fully Triton‑accelerated.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------------- #
# Triton batched GEMM (A @ Bᵀ) – used for the TriMul contraction
# ----------------------------------------------------------------------------- #
@triton.jit
def triton_gemm_outgoing_kernel(
    # pointers
    a_ptr,          # left   (B*H, M, K)  – bf16
    b_ptr,          # right  (B*H, N, K)  – bf16  (note: we will read transposed)
    c_ptr,          # out    (B*H, M, N) – fp32
    # problem size
    batch,          # total number of (B*H) matrices
    M, N, K,
    # strides (in elements, not bytes)
    stride_ab, stride_am, stride_ak,   # a: batch‑row‑col
    stride_bb, stride_bn, stride_bk,   # b: batch‑row‑col   (row = N, col = K)
    stride_cb, stride_cm, stride_cn,   # c: batch‑row‑col
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1️⃣ Compute which batch and tile we are working on
    # --------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m_tiles * num_n_tiles

    batch_idx = pid // tiles_per_batch
    tile_idx  = pid % tiles_per_batch
    tile_m = tile_idx // num_n_tiles
    tile_n = tile_idx % num_n_tiles

    # --------------------------------------------------------------
    # 2️⃣ Compute the base offsets for this tile
    # --------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = a_ptr + batch_idx * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    # We need Bᵀ: element (k, n) of Bᵀ is element (n, k) of B.
    b_ptrs = b_ptr + batch_idx * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # --------------------------------------------------------------
    # 3️⃣ Accumulate over K
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = tl.minimum(k + tl.arange(0, BLOCK_K), K - 1)

        # Load A block (M x K)
        a = tl.load(
            a_ptrs + (cur_k[None, :] - k) * stride_ak,
            mask=(offs_m[:, None] < M) & (cur_k[None, :] < K),
            other=0.0,
        )
        # Load Bᵀ block (K x N)  → we load transposed directly
        b = tl.load(
            b_ptrs + (cur_k[:, None] - k) * stride_bk,
            mask=(cur_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # dot: (M, K) @ (K, N) → (M, N)   (bf16 → fp32 accumulation)
        acc += tl.dot(a, b)

        # advance pointers for next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # 4️⃣ Write back
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch_idx * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

# ----------------------------------------------------------------------------- #
# Helper: launch the Triton kernel for a batch of GEMMs
# ----------------------------------------------------------------------------- #
def batched_gemm_outgoing(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : torch tensors of shape (B, H, N, N) and dtype torch.bfloat16
    returns c : torch.float32 tensor of shape (B, H, N, N)
    """
    assert a.shape == b.shape
    B, H, N, _ = a.shape
    batch = B * H
    M = N
    K = N
    # reshape to (batch, N, N)
    a_view = a.view(batch, N, N)
    b_view = b.view(batch, N, N)

    # output buffer (fp32)
    c = torch.empty_like(a_view, dtype=torch.float32)

    # tile sizes – tuned for Hopper (SM90) and the N range of the tests
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(batch * tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N), 1),)

    # strides in elements (not bytes)
    stride_ab = a_view.stride(0)
    stride_am = a_view.stride(1)
    stride_ak = a_view.stride(2)

    stride_bb = b_view.stride(0)
    stride_bn = b_view.stride(1)
    stride_bk = b_view.stride(2)

    stride_cb = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    triton_gemm_outgoing_kernel[grid](
        a_ptr=a_view,
        b_ptr=b_view,
        c_ptr=c,
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_ab=stride_ab,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bb=stride_bb,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_cb=stride_cb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return c.view(B, H, N, N)

# ----------------------------------------------------------------------------- #
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]
                  ) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]      (float32, CUDA)
        mask         : [B, N, N]           (float32, 0/1, CUDA) – may be ignored
        weights      : dict of tensors containing all linear / LN params
        config       : dict with keys
            dim          – input channel dimension
            hidden_dim   – projection dimension
            nomask       – bool, if True the mask is all‑ones

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """

    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input
    # --------------------------------------------------------------
    def layer_norm(x, w, b):
        return F.layer_norm(x, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (dim → hidden_dim)
    # --------------------------------------------------------------
    def linear(x, w):
        return F.linear(x, w)          # bias = 0 as per definition

    left = linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional binary mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_b = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_b
        right = right * mask_b

    # --------------------------------------------------------------
    # 4️⃣ Gating – sigmoid of a linear projection
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core TriMul reduction (batched GEMM via Triton)
    # --------------------------------------------------------------
    #   left/right : [B, N, N, H] → permute → [B, H, N, N] and cast to bf16
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16)

    # Batched GEMM: out_perm = left_perm @ right_permᵀ
    out_perm = batched_gemm_outgoing(left_perm, right_perm)    # fp32, shape (B,H,N,N)

    # Restore original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output layer‑norm (hidden_dim → hidden_dim)
    # --------------------------------------------------------------
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = linear(out, weights["to_out.weight"])

    return out
