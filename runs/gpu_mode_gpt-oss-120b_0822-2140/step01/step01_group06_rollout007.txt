"""
TriMul (outgoing) – fast forward pass using a custom Triton GEMM.

The only O(N³·H) work is the contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply (left @ rightᵀ) for every
(head = d) of every batch.  We reshape the tensors to (B*H, N, N)
and launch a tiled GEMM written in Triton.  All other
operations (LayerNorms, Linear projections, sigmoid gates,
final LayerNorm + linear) are performed with PyTorch.
The element‑wise multiplication with the output‑gate is fused
with a tiny Triton kernel to satisfy the “must use a kernel”
requirement.

The implementation works for any batch size ≤ 2, sequence length
up to 1024 and hidden dimensions that fit in H200 memory.
All arithmetic is carried out in fp16 inside the GEMM for speed,
while the accumulator uses fp32 to preserve numerical quality.
The final output is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where A,B ∈ (M, N, K) and K==N.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # dimensions
    stride_am, stride_ak,              # A strides: (M, K)
    stride_bk, stride_bn,              # Bᵀ strides: (K, N)
    stride_cm, stride_cn,              # C strides: (M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of M matrices (fp16 → fp32)."""
    pid_m = tl.program_id(0)   # row‑block id   (0 … ceil(M/BLOCK_M)-1)
    pid_n = tl.program_id(1)   # col‑block id   (0 … ceil(N/BLOCK_N)-1)

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks to avoid out‑of‑bounds loads / stores
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the K‑loop (inner dimension)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < K)),
                    other=0.0)          # fp16
        b = tl.load(b_ptrs, mask=((tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < K) & mask_n[None, :]),
                    other=0.0)          # fp16

        # Cast to fp16 explicitly (the loads already give fp16, but we keep the cast
        # for clarity) and perform the dot product.  tl.dot works on fp16/float16.
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result (cast back to fp32)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_store)


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (used for the out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elements,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32, CUDA)
        * mask_tensor   : torch.Tensor or None, shape [B, N, N] (bool/float, CUDA)
        * weights_dict  : dictionary with all linear / norm weights
        * config_dict   : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 2️⃣ Input LayerNorm (over last dimension)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                              # [B, N, N, dim]

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 5️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core contraction: batched GEMM left @ rightᵀ
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) and fuse batch & head into one dimension M = B*H
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    M = B * hidden_dim
    left_flat = left.view(M, N, N)     # (M, N, K)  with K == N
    right_flat = right.view(M, N, N)   # (M, N, K)

    # Cast to fp16 for the GEMM (the kernel accumulates in fp32)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Allocate output (fp32)
    out_hidden = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Launch Triton GEMM
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32          # empirically good for H200
    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (in elements) for the three tensors
    stride_am = left_fp16.stride(0)   # M‑stride
    stride_ak = left_fp16.stride(2)   # K‑stride (inner dim)
    # For Bᵀ we need strides (K, N): K comes from dim‑1 of original right,
    # N comes from dim‑2, hence we swap them.
    stride_bk = right_fp16.stride(1)  # K‑stride (original row stride)
    stride_bn = right_fp16.stride(2)  # N‑stride (original col stride)
    stride_cm = out_hidden.stride(0)
    stride_cn = out_hidden.stride(2)

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden,
        M,
        N,
        N,                     # K == N
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = (
        out_hidden.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                             # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                            # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8️⃣ Element‑wise multiplication with the output gate (fused Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        BLOCK=BLOCK,
    )
    # out now holds out * out_gate

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to the original dimensionality
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
