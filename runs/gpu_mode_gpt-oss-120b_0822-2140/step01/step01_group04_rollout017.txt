"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated contraction.

The heavy part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiplication
for each hidden dimension `d` (called *head* here):

    C[b,head,i,j] = A[b,head,i,:] @ B[b,head,j,:]ᵀ

The kernel below computes this A @ Bᵀ for all (batch,head) pairs.
All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
are kept in PyTorch because they are cheap compared to the O(N³) contraction.

The kernel works on tensors in **(B, H, N, N)** layout, uses FP16
loads for bandwidth, FP32 accumulation for accuracy and writes the result
back in FP32.  Tile sizes (64,64,64) were chosen to give good occupancy on
H200 (SM‑90) while keeping shared‑memory usage modest.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B, H, N, N) @ (B, H, N, N)ᵀ  →  (B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    For each (b, h) compute
        C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]      (i.e. A @ Bᵀ)
    """
    pid_m = tl.program_id(0)   # tile row   (i)
    pid_n = tl.program_id(1)   # tile col   (j)
    pid_hb = tl.program_id(2)  # combined batch * head

    batch = pid_hb // H
    head  = pid_hb %  H

    # -------------------------------------------------
    # tile offsets
    # -------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # -------------------------------------------------
    # loop over K dimension
    # -------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # -------- load A[b,h,i,k]  (TILE_M x TILE_K) ----------
        a_ptrs = (A_ptr +
                  batch * stride_a_batch +
                  head  * stride_a_head +
                  offs_m[:, None] * stride_a_i +
                  offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # -------- load Bᵀ[b,h,k,j] = B[b,h,j,k]  (TILE_K x TILE_N) ----------
        b_ptrs = (B_ptr +
                  batch * stride_b_batch +
                  head  * stride_b_head +
                  offs_k[:, None] * stride_b_j +   # k as first dim
                  offs_n[None, :] * stride_b_i)    # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # -------------------------------------------------
    # store result
    # -------------------------------------------------
    c_ptrs = (C_ptr +
              batch * stride_c_batch +
              head  * stride_c_head +
              offs_m[:, None] * stride_c_i +
              offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, dim)  (float32/bfloat16)
        - mask         : torch.Tensor of shape (B, N, N)      (bool/float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output of shape (B, N, N, dim) (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Optional mask (applied after projection)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_ = mask.to(dtype).unsqueeze(-1)           # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) → hidden_dim
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy contraction with Triton
    #    reshape to (B, H, N, N) and make contiguous
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B,H,N,N]

    # strides in *elements* (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)  (we will load transposed)
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile sizes – tuned for SM‑90, keep within shared memory limits
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,                     # enough for fp16 compute
    )

    # back to original layout   (B,N,N,H)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out.to(dtype)
