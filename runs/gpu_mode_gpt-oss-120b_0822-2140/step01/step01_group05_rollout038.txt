assistantfinal**Solution Overview**

The only part of *TriMul* that is truly cubic (`N³`) is the
`einsum('... i k d, ... j k d -> ... i j d')`.  
All other operations (LayerNorm, linear projections, sigmoid‑gates,
output‑norm and final linear) are cheap and are kept in PyTorch.

We fuse the cubic part into a single Triton kernel:

*   The hidden dimension is treated as an extra “batch’’ axis, i.e. we
    compute a **batched matrix‑multiply**  
    `out[b·h, i, j] = Σ_k left[b·h, i, k] * right[b·h, j, k]`  

    where `b·h = B * hidden_dim`.  
*   The kernel works on **tiles** of size `BLOCK_M × BLOCK_N` and iterates
    over the `K`‑dimension with tiles `BLOCK_K`.  
*   We use `tl.dot` which is automatically mapped to Tensor‑Core
    instructions on Hopper when the data type is `fp16`.  
    The inputs are stored as `fp32` (the model is trained in fp32) and are
    **re‑cast to fp16 inside the kernel** – this gives a large speed‑up
    while the final result is accumulated in `fp32` and written back as
    `fp32` to keep numerical stability.
*   The mask (if present) and the three sigmoid‑gates are applied **before**
    the kernel, because they are cheap element‑wise ops and doing them on
    the full 4‑D tensor avoids having to broadcast a 3‑D mask inside the
    kernel.

The kernel follows the *launch‑convention* required by Triton
(`grid = (n_tiles_m, n_tiles_n, B*hidden)`).

```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
matrix‑multiply kernel written in Triton.

The cubic part   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is executed as a batched (N×N) @ (N×N)ᵀ for every hidden channel d.
Inside the kernel we
* cast the fp32 inputs to fp16,
* use tl.dot (Tensor‑Core) for the inner product,
* accumulate in fp32,
* write the result back as fp32.

All other ops (LayerNorm, linear projections, sigmoid‑gates,
output‑norm and final linear) stay in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: batched (N,N) @ (N,N)ᵀ   for B*H matrices
# -------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers (fp32)
    M, N, K,                      # dimensions  (M=N=seq_len, K=seq_len)
    stride_aba, stride_ama, stride_aka,   # strides for A  (batch, M, K)
    stride_bba, stride_bmb, stride_bkb,   # strides for B  (batch, M, K)  (B is same shape as A)
    stride_cca, stride_cma, stride_cna,   # strides for C  (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched mat‑mul C = A @ Bᵀ  (all tensors are fp32 on the host)."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch * hidden

    # ---------------------------------------------------------
    # Tile offsets
    # ---------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the beginning of the tile
    a_ptrs = A_ptr + pid_b * stride_aba \
                  + offs_m[:, None] * stride_ama \
                  + offs_k[None, :] * stride_aka

    b_ptrs = B_ptr + pid_b * stride_bba \
                  + offs_n[None, :] * stride_bmb \
                  + offs_k[:, None] * stride_bkb   # note: we need Bᵀ, so k is first

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over K dimension
    # ---------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles, cast to fp16 for Tensor‑Core
        a_tile = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b_tile = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        a_fp16 = a_tile.to(tl.float16)
        b_fp16 = b_tile.to(tl.float16)

        # tl.dot works on fp16 → fp32 accumulation
        acc += tl.dot(a_fp16, b_fp16)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_aka
        b_ptrs += BLOCK_K * stride_bkb

    # ---------------------------------------------------------
    # Store result
    # ---------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cca \
                  + offs_m[:, None] * stride_cma \
                  + offs_n[None, :] * stride_cna

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------
# Host side entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor[B, N, N, dim] (float32, CUDA)
        mask    – torch.Tensor[B, N, N]   (float32 or bool)  or None
        weights – dict of torch.Tensor parameters (all float32, CUDA)
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)           # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the Triton kernel
    #    layout -> (B, hidden, N, N) -> (B*hidden, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_flat = right.view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)       # allocate output

    # ------------------------------------------------------------------
    # 6. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # program id 0  (row tiles)
        triton.cdiv(N, BLOCK_N),          # program id 1  (col tiles)
        B * hidden_dim,                   # program id 2  (batch*head)
    )

    # strides for A, B, C (batch, row, col)
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    batched_matmul_fp16_kernel[grid](
        left_flat,               # A_ptr
        right_flat,              # B_ptr
        out_flat,                # C_ptr
        N, N, N,                # M, N, K
        a_batch, a_row, a_col,   # strides A
        b_batch, b_row, b_col,   # strides B
        c_batch, c_row, c_col,   # strides C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to original layout
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
