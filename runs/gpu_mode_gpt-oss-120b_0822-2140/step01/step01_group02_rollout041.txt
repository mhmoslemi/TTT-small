The heavy part of the TriMul “outgoing’’ operator is the 3‑dimensional contraction  

\[
\text{out}_{b,i,j,d}= \sum_{k=0}^{N-1}\text{left}_{b,i,k,d}\;\text{right}_{b,j,k,d}
\]

which is exactly a batched matrix‑multiply where, for each batch *b* and hidden
dimension *d*, we compute  

```
out[b,d] = left[b,d] @ right[b,d].T          # shapes (N,N) @ (N,N)
```

All other operations (LayerNorm, Linear projections, gating, the final
projection) are cheap compared with the \(O(N^3\!·\!H)\) work and can stay in
PyTorch.

```python
"""
TriMul “outgoing” forward pass (AlphaFold3) with a fused Triton batched GEMM.

The contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batch of GEMMs with B·H independent (N×N) @ (N×N)ᵀ products.
We:
  1. Apply LayerNorm + linear projections + optional mask in PyTorch.
  2. Fuse the hidden dimension into the batch dimension (shape (B*H, N, N)).
  3. Run a Triton kernel that computes C = A @ Bᵀ in FP16 (FP32 accumulation).
  4. Reshape back, apply a second LayerNorm + gating, and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A,B shape [B·H, M, K], [B·H, N, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (int)
    stride_az, stride_am, stride_ak,    # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,    # strides for B (batch, row, col) – original layout
    stride_cz, stride_cm, stride_cn,    # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is interpreted transposed."""
    pid_batch = tl.program_id(2)               # batch index (B*H)
    pid_m = tl.program_id(0)                   # block row
    pid_n = tl.program_id(1)                   # block col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile in A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_az \
                   + offs_m[:, None] * stride_am \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles (mask out‑of‑bounds elements)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as FP16 to save bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : torch tensors of shape (B·H, N, N), dtype float16, contiguous.
    Returns c of the same shape & dtype (C = a @ bᵀ).
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N  # square matrices

    c = torch.empty_like(a)

    # Strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B_H,                       # batch (B*H)
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,          # M
        N,          # N
        N,          # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # Hopper handles 8 warps well for 64×64 tiles
    )
    return c


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]       (bool or float)
            - weights      : dict of torch.Tensor (float32)
            - config       : dict with keys
                  "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B,N,N,dim)  fp32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask can be bool or float; convert to same dtype as left/right
        mask_exp = mask.unsqueeze(-1).to(left.dtype)    # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Fold hidden dimension into the batch dimension for the GEMM
    #    Shape becomes (B*hidden, N, N) and we cast to fp16.
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out_fp16 = left @ rightᵀ   (still fp16)
    # ------------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden) and cast to fp32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N)          # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).to(torch.float32)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Post‑projection LayerNorm (hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate      # out_gate already has shape (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
```
