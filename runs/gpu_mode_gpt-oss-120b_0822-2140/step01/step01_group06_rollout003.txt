"""
TriMul (outgoing) – fast Triton implementation.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched GEMM for every (b,d) pair:
    (N×N) @ (N×N)ᵀ  →  (N×N).

We reshape the tensors to (B·H, N, N) and launch a single
triton kernel that performs a tiled GEMM for **all** (batch,head)
pairs in parallel (grid dim‑2 is the batch*head index).  All
other operations (layer‑norms, linear projections, gating,
final linear) stay in PyTorch – a tiny element‑wise Triton kernel
fuses the multiplication with the output‑gate.

The kernel uses fp16 loads (half‑precision arithmetic) but
accumulates in fp32, yielding the same numerical result as the
reference while being much faster on Hopper (H200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Element‑wise multiplication (used after the output‑gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr,                     # flattened pointer to a
    b_ptr,                     # flattened pointer to b
    out_ptr,                   # flattened pointer to output (may alias a)
    total_elems: tl.constexpr, # number of elements to process
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 2️⃣  Batched GEMM kernel for the O(N³·H) contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr,           # left  (BH, M, K)  – FP16
    b_ptr,           # right (BH, N, K)  – FP16, we read it transposed
    c_ptr,           # out   (BH, M, N) – FP32 result
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: batch*head, row, col strides
    stride_bn, stride_bk, stride_bb,   # B: batch*head, col (N) , row (K) strides – we treat Bᵀ
    stride_cm, stride_cn, stride_cb,   # C: batch*head, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch*head index

    # --------------------------------------------------------------
    # Offsets within the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers for this (batch,head) slice
    # --------------------------------------------------------------
    a_base = a_ptr + pid_b * stride_ab
    b_base = b_ptr + pid_b * stride_bb
    c_base = c_ptr + pid_b * stride_cb

    # A tile pointer (M x K)
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak

    # Bᵀ tile pointer (K x N) – note the swapped strides
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # Load tiles (fp16 → fp32 on the fly)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # dot returns fp32 because we ask for out_dtype=tl.float32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # Increment pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32 CUDA)
        * mask_tensor  : [B, N, N]       (bool/float32 CUDA) – may be None
        * weights_dict : all linear / LN parameters
        * config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32 CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = inp.shape
    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)       # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid after a bias‑less linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – tiled GEMM in Triton
    # ------------------------------------------------------------------
    #   reshape to (B*H, N, N) and make everything contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left  = left.view(BH, N, N).to(torch.float16)
    right = right.view(BH, N, N).to(torch.float16)

    # allocate output tensor (fp32)
    out = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # strides (in elements) for the three tensors
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # block size – tuned for Hopper (128×128 tile, 32‑wide K)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH,                        # batch*head
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        M=N,
        N=N,
        K=N,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_ab=stride_a_batch,
        stride_bn=stride_b_n,      # this is the stride for the "column" dim after transpose
        stride_bk=stride_b_k,
        stride_bb=stride_b_batch,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        stride_cb=stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # a reasonable default for H200
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Multiply with the output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        BLOCK=BLOCK,
    )
    # out now holds out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
