"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core contraction.

The operator works on a tensor x ∈ ℝ^{B×N×N×C}.  After a per‑channel LayerNorm
and two linear projections we obtain two tensors *left* and *right*
of shape (B, N, N, H) where H = hidden_dim.  The expensive step is the
tri‑multiplicative update

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is exactly a batched matrix‑multiply A @ Bᵀ for every (b, h) pair.
We fuse all (B·H) GEMMs in a single Triton kernel, using fp16 loads for
bandwidth and fp32 accumulation for numerical stability.

The rest of the module (gating, layer‑norms and the final projection) is
performed with regular PyTorch operations – they are cheap compared to the
N³ contraction.

Supported configurations:
    • B ∈ {1,2}
    • N ∈ {128,256,512,1024}
    • C ∈ {128,384,768}
    • H = hidden_dim = 128
Both masked and un‑masked variants are handled.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, M, K]), where K == N
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr,                      # *float16   (batch, M, K)
    B_ptr,                      # *float16   (batch, N, K)  (read as K×N)
    C_ptr,                      # *float32   (batch, M, N)
    M, N, K,                    # dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (see comment below)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where B is read transposed (i.e. we compute A @ Bᵀ)."""
    pid_b = tl.program_id(2)                 # batch*head index
    pid_m = tl.program_id(0)                 # block row
    pid_n = tl.program_id(1)                 # block column

    # ------------------------------------------------------------------
    # block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current block of A (M×K) and Bᵀ (K×N)
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk   # step along K (original column)
        + offs_n[None, :] * stride_Bn   # step along N (original row)
    )

    # ------------------------------------------------------------------
    # accumulate in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)                     # (M×K)·(K×N) → (M×N)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (all float32)
        - config       : dict containing {"dim", "hidden_dim", "nomask"}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → (B,N,N,H)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B,N,N,H]
    right = F.linear(x, right_proj_w)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core Tri‑multiplicative contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # Rearrange to (B, H, N, N) → (B*H, N, N) for the kernel
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (bandwidth) – keep fp32 accumulator
    left_fp16 = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # Output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (batch, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For transposed read of B we need:
    stride_Bk = stride_Bc   # step along K (original column)
    stride_Bn = stride_Br   # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # Tile sizes – empirically good for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * head
    )

    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim C

    return out
```

The implementation follows the lessons discussed:

* **Lesson 1 – permute to (B, H, N, N) before the kernel** – we do exactly that, which yields
  a contiguous layout and simplifies stride computation.
* **Lesson 2 – reformulate as batched GEMM** – the core contraction is performed by a single
  Triton kernel that computes `A @ Bᵀ` for every `(batch, head)` pair.
* **Lesson 3 – correct stride handling** – all strides are derived from the
  contiguous tensors and passed explicitly; the kernel never relies on implicit
  flat indexing.
* The rest of the module (layer‑norms, gating, final projection) stays in
  PyTorch because they are cheap compared to the N³ operation.
* Mixed‑precision (fp16 loads, fp32 accumulation) gives bandwidth efficiency while
  preserving numerical quality for the final float‑32 output.
