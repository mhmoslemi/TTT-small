"""
TriMul “outgoing” forward (AlphaFold‑3) with a fused batched matrix‑multiply
kernel written in Triton.

*   Input  : [B, N, N, D]                (float32, CUDA)
*   Mask   : optional [B, N, N] (0/1)   (broadcast on the hidden dim)
*   Weights: dict of LayerNorm, Linear & gate parameters (all float32)

The heavy part is the einsum
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix‑multiply (A @ Bᵀ) for every hidden head *h*.
We fuse the whole (B × hidden) batch into a single Triton kernel; the rest
of the module (LayerNorm, linear projections, sigmoids, final LayerNorm &
linear) stays in PyTorch because they are cheap compared to the N³ work.

The kernel works on tensors of shape (B*H, N, N) (row‑major) and uses a
64 × 64 × 32 tiling that matches the H200 tensor‑core layout.  All
computations are in FP32 (you may cast the inputs to FP16 before the kernel
for extra speed – the kernel itself keeps FP32 accumulation for numeric
stability).

The entry‑point `custom_kernel` follows the signature required by the
evaluator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply C = A @ Bᵀ
#   A : (B*H, M, K)   row‑major
#   B : (B*H, N, K)   row‑major  (the K‑dimension is the same as in A)
#   C : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes (M=N=seq_len, K=seq_len)
    stride_a_batch, stride_a_row, stride_a_col,   # strides for A
    stride_b_batch, stride_b_row, stride_b_col,   # strides for B
    stride_c_batch, stride_c_row, stride_c_col,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a whole batch (B*H) using tiling."""
    pid_b = tl.program_id(0)      # batch * hidden index
    pid_m = tl.program_id(1)      # tile row index
    pid_n = tl.program_id(2)      # tile col index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the first element of the tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_a_batch
        + offs_m[:, None] * stride_a_row
        + offs_k[None, :] * stride_a_col
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_b_batch
        + offs_k[:, None] * stride_b_col          # K dimension becomes rows of Bᵀ
        + offs_n[None, :] * stride_b_row          # N dimension becomes columns of Bᵀ
    )

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K loop
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # Load tiles (mask out‑of‑bounds entries with 0.0)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # tl.dot does (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b)

        # Move pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_col

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensor with the same keys as the
                         reference implementation
        - config       : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the Triton kernel:
    #    (B, N, N, H) -> (B*H, N, N)
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> contiguous -> view
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left.view(B * hidden, N, N)   # [B*H, N, N]
    right_flat = right.view(B * hidden, N, N) # [B*H, N, N]

    out_flat = torch.empty_like(left_flat)    # allocate result

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,             # batch * hidden dimension
        triton.cdiv(N, BLOCK_M),  # M‑tiles
        triton.cdiv(N, BLOCK_N),  # N‑tiles
    )

    # Strides for the three‑dimensional tensors (batch, row, col)
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        a_batch,
        a_row,
        a_col,
        b_batch,
        b_row,
        b_col,
        c_batch,
        c_row,
        c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to the original layout [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (over hidden dim) + final gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
