```python
"""
TriMul “outgoing” forward pass – Triton‑accelerated core contraction.

The operator works on a tensor x ∈ ℝ^{B×N×N×C}.  After a LayerNorm and two
linear projections we obtain `left` and `right` of shape
B×N×N×H (H = hidden_dim).  The expensive step is

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched GEMM: for each (b,h) we need the matrix product
    L_{b,h}  (N×N)  @  R_{b,h}ᵀ (N×N)  →  out_{b,h} (N×N)

The kernel below performs all B·H GEMMs in parallel, using fp16 loads and
fp32 accumulation.  The surrounding work (layer‑norms, linear layers,
sigmoids, optional mask, final projection) is done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B*H, N, N])
# -------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # *float16   (batch, M, K)   K == N
    B_ptr,                     # *float16   (batch, N, K)   read as K×N
    C_ptr,                     # *float32   (batch, M, N)
    M, N, K,                   # matrix dimensions (all == N except K==N)
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note Bk = original col stride, Bn = original row stride)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch*head index
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block col index

    # ---------- tile start indices ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---------- boundary masks ----------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------- pointer arithmetic ----------
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk   # iterate over K dimension of original B (its columns)
        + offs_n[None, :] * stride_Bn   # iterate over N dimension (original rows)
    )

    # ---------- accumulation ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- main K‑loop ----------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)               # (M×K)·(K×N) → (M×N)

        # advance pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---------- store result ----------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (`float32`)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights      : dict of model parameters (all `float32`)
        - config       : dict with keys
                         * ``dim``          – C
                         * ``hidden_dim``   – H
                         * ``nomask`` (optional) – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (`float32`).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # [B, N, N, H]
    right = F.linear(x, right_proj_w)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # shape [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # reshape to (B*H, N, N) and move to contiguous memory
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (bandwidth) – keep a fp32 output buffer
    left_fp16 = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)   # (B*H, N, N)

    # strides in **elements** (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For the transposed view we need:
    stride_Bk = stride_Bc            # original column stride -> K step
    stride_Bn = stride_Br            # original row stride    -> N step

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # -----------------------------------------------------------------------------
    # Launch parameters – tuned for Hopper (SM90)
    # -----------------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
        B * H                         # batch*head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (LayerNorm → out‑gate → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)    # back to original dim C, no bias

    return out
```

All other operations (LayerNorms, Linear layers, sigmoid gates, optional masking) stay in PyTorch, which is already optimal on the target hardware.
