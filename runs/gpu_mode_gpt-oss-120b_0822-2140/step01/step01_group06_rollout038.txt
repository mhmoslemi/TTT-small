"""
TriMul (outgoing) – fast forward pass.

The only O(N³·H) work is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
We reshape the tensors to (B, H, N, N) and launch a tiled GEMM kernel
that computes A @ Bᵀ for every (batch, head) pair.  The kernel works in
FP16 for the matrix‑multiply and accumulates in FP32, then stores the
result in FP32.  All other operations (layer‑norm, linear projections,
gates, final linear) are done with regular PyTorch for simplicity.

The implementation supports both masked / un‑masked inputs and works
for the dimensionalities required by the benchmark (dim = 128/384,
hidden_dim = 128, N up to 1024, B = 1/2).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A,B shape = (B*H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers to the three tensors
    BATCH, M, N, K,                    # B*H, rows, cols, inner‑dim
    stride_abh, stride_am, stride_ak,  # strides for A  (batch*head, row, col)
    stride_bbh, stride_bn, stride_bk,  # strides for B  (batch*head, row, col)
    stride_cbh, stride_cm, stride_cn,  # strides for C  (batch*head, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------------
    batch_head = tl.program_id(0)                 # 0 … BATCH‑1
    pid_m = tl.program_id(1)                      # block row
    pid_n = tl.program_id(2)                      # block col

    # ------------------------------------------------------------------
    # offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds (needed for the edge blocks)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main GEMM loop over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[i,k] – note: A is (batch_head, i, k)
        a_ptrs = (A_ptr
                  + batch_head * stride_abh
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ[k,j] – we need B transposed, i.e. B is (batch_head, j, k)
        # → load with k as the row index and j as the column index.
        b_ptrs = (B_ptr
                  + batch_head * stride_bbh
                  + offs_k[:, None] * stride_bk      # k dimension becomes row
                  + offs_n[None, :] * stride_bn)     # j dimension becomes col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        if FP16:
            a = a.to(tl.float16)
            b = b.to(tl.float16)
            acc += tl.dot(a, b).to(tl.float32)
        else:
            acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write‑back
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch_head * stride_cbh
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape (B, N, N, dim), dtype=torch.float32
        * mask_tensor  : torch.Tensor, shape (B, N, N)  (may be None)
        * weights_dict : contains the learned parameters (see doc‑string above)
        * config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}
    Returns
    -------
    torch.Tensor
        Output tensor, shape (B, N, N, dim), dtype=torch.float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                                    # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not cfg.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gates (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction via a tiled GEMM kernel
    #    reshape to (B, H, N, N) → (B*H, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # flatten the first two dimensions for the kernel launch
    batch_head = B * hidden_dim
    A = left
    B_ = right
    C = torch.empty_like(A)                       # result (B, H, N, N)

    # strides in element units (not bytes)
    stride_abh, stride_am, stride_ak = A.stride()
    stride_bbh, stride_bn, stride_bk = B_.stride()
    stride_cbh, stride_cm, stride_cn = C.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_head,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_gemm_kernel[grid](
        A_ptr=A,
        B_ptr=B_,
        C_ptr=C,
        BATCH=batch_head,
        M=N,
        N=N,
        K=N,
        stride_abh=stride_abh,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bbh=stride_bbh,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_cbh=stride_cbh,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=True,                     # compute in fp16, accumulate fp32
    )

    # ------------------------------------------------------------------
    # 6️⃣ Bring back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = C.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣ Post‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                            # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate                         # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
