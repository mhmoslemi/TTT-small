"""
TriMul (outgoing) – fast forward pass.

The cubic contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is rewritten as a batched GEMM:
    (B·H, N, N)  @  (B·H, N, N).T
for each head‑dimension *h*.  The heavy O(N³·H) work is carried out by a
single Triton kernel that tiles the GEMM in fp16 and accumulates in fp32.
All surrounding operations (LayerNorm, Linear, sigmoid‑gates, final
projection) are performed with PyTorch.  At least one operation lives in
Triton, satisfying the “must use a kernel” rule while keeping the code
compact and portable.

The implementation works for arbitrary batch size (1‑2), sequence length
(128‑1024) and hidden dimensions (128‑384) and respects the optional
mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel for the outgoing TriMul contraction.
# Computes C = A @ Bᵀ  (all tensors have shape (batch, M, K) with K = M = N)
# Accumulates in fp32, operates in fp16 for speed.
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_gemm_kernel(
    a_ptr,                # fp16  (batch, M, K)
    b_ptr,                # fp16  (batch, N, K)  (will be read as transposed)
    c_ptr,                # fp32  (batch, M, N)  output
    M, N, K,              # problem size
    stride_ah, stride_am, stride_ak,   # strides for A  (batch, row, col)
    stride_bh, stride_bn, stride_bk,   # strides for B  (batch, col, row) – note the swap
    stride_ch, stride_cm, stride_cn,   # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)          # which (batch,head) pair
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ---- block start offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds (last blocks may be smaller)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers shifted to the current batch/head
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # ---- load A block -------------------------------------------------------
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
    a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)

    # ---- load B block (as transposed) ---------------------------------------
    # B is stored as (batch, N, K); we need Bᵀ ⇒ treat first dim as col, second as row
    b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bn
                            + offs_k[None, :] * stride_bk)
    b = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k[None, :] < K), other=0.0)

    # ---- accumulation -------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load current tiles
        a_tile = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b_tile = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k[None, :] < K), other=0.0)

        # dot: fp16 * fp16 → fp32 accumulate
        acc += tl.dot(a_tile, b_tile)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---- write result -------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim]  (float32, CUDA)
        * mask_tensor  : torch.Tensor [B, N, N] or None
        * weights_dict : {"norm.weight":..., "norm.bias":..., ...}
        * config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (mask is not None) and (not config.get("nomask", False)):
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    # bring head dimension next to batch: (B, H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # flatten batch & head => (B*H, N, N)
    left  = left.view(B * hidden_dim, N, N)
    right = right.view(B * hidden_dim, N, N)

    # fp16 compute, fp32 accumulation
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer (fp32)
    out_fp32 = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides in element units
    stride_ah, stride_am, stride_ak = left_fp16.stride()
    stride_bh, stride_bn, stride_bk = right_fp16.stride()
    stride_ch, stride_cm, stride_cn = out_fp32.stride()

    # tile sizes – chosen to give good occupancy on Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)

    grid = (num_blocks_m, num_blocks_n, B * hidden_dim)

    triton_tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N, N, N,                         # M, N, K
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Multiply with output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimensionality
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
