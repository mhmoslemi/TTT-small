"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* the contraction is a matrix multiplication

    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We therefore launch a **batched GEMM** kernel that computes many independent
(N×N) × (N×N)ᵀ products in one Triton launch.  The kernel works in fp16
(both inputs) and accumulates in fp32 (the default of `tl.dot`).  All
pre‑/post‑processing (LayerNorm, linear projections, gating, masking) stays in
PyTorch because they are cheap compared with O(N³) work.

The kernel is written for Hopper (SM‑90) – block sizes 128×128×32 were found
to give the best occupancy / memory‑bandwidth trade‑off on an H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers (fp16 for a/b, fp32 for c)
    N,                          # sequence length
    stride_ah, stride_ai, stride_ak,   # A: (M, N, N)
    stride_bh, stride_bj, stride_bk,   # B: (M, N, N)
    stride_ch, stride_ci, stride_cj,   # C: (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  where each operand is (M, N, N).
    The kernel is launched with a 3‑D grid (pid_m, pid_n, pid_z) where
        pid_m : block row   (i)
        pid_n : block col   (j)
        pid_z : flattened batch × hidden dimension
    """
    pid_m = tl.program_id(0)          # i‑block
    pid_n = tl.program_id(1)          # j‑block
    pid_z = tl.program_id(2)          # batch*head

    # ------------------------------------------------------------------
    # Offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointer to the beginning of the (pid_z)-th matrix pair
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K (the shared “seq_len” index)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai      # i dimension
            + offs_k[None, :] * stride_ak      # k dimension
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B_T[k,j]  shape (BLOCK_K, BLOCK_N)  (load transposed)
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk      # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj      # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back C[i,j]
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul.
    Input/Output layout: (B, H, N, N)  (float16 for left/right, float32 for out)
    Returns: (B, H, N, N) float32
    """
    assert left.shape == right.shape
    B, H, N, _ = left.shape
    M = B * H                               # flatten batch & head

    # flatten to (M, N, N) – keep the memory layout contiguous
    left_f  = left.reshape(M, N, N).contiguous()
    right_f = right.reshape(M, N, N).contiguous()
    out_f   = torch.empty_like(left_f, dtype=torch.float32)

    # strides as int64 (required by Triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out_f.stride()

    # --------------------------------------------------------------
    # Tuned block sizes for Hopper (SM‑90)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch*head)
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out_f,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,               # good default for fp16 on Hopper
    )
    return out_f.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    All cheap operations (LayerNorm, Linear, Sigmoid, Mask) are done in
    PyTorch; the expensive N³ contraction is executed by the batched GEMM
    kernel above.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim) – no bias
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)    # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left  = left  * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, hidden, N, N) layout expected by the kernel.
    #    Cast to fp16 to cut memory traffic; kernel accumulates in fp32.
    # ------------------------------------------------------------------
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_fp32 = _trimul_gemm(left_fp16, right_fp16)    # (B, H, N, N) fp32

    # Back to the original layout (B, N, N, hidden)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)      # (dim, hidden)
    out = F.linear(out, to_out_w)                       # (B, N, N, dim)

    return out.to(torch.float32)
