"""
TriMul “outgoing” kernel (AlphaFold‑3).

The only expensive part is the triple‑wise contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is mathematically a batched matrix multiplication
A @ Bᵀ for every (batch, hidden) pair.
We fuse the whole reduction into a single Triton GEMM kernel that works
in FP16 (inputs) and accumulates in FP32, giving a huge speed‑up over the
reference torch.einsum while keeping the surrounding LayerNorm / Linear /
gate logic in regular PyTorch.

The kernel is launched on a 2‑D grid (M‑tiles × N‑tiles) for each
(batch × hidden) “head”.  All other operations remain in PyTorch, which is
fast compared with the N³ reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel (A @ Bᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_az, stride_am, stride_ak,   # a: batch, row (M), col (K)
    stride_bz, stride_bk, stride_bn,   # b: batch, row (K), col (N) – we treat as transposed
    stride_cz, stride_cm, stride_cn,   # c: batch, row (M), col (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent batches."""
    pid = tl.program_id(0)

    # -----------------------------------------------------------------------
    # decode pid → (batch, tile_m, tile_n)
    # -----------------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    tile_m = block_idx // num_blocks_n
    tile_n = block_idx % num_blocks_n

    # -----------------------------------------------------------------------
    # block start offsets
    # -----------------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_az
    b_batch_ptr = b_ptr + batch_idx * stride_bz
    c_batch_ptr = c_ptr + batch_idx * stride_cz

    # -----------------------------------------------------------------------
    # initialise accumulator in fp32
    # -----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # loop over K dimension in tiles
    # -----------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # K‑tile start
        k_tile = k * BLOCK_K
        offs_k = k_tile + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A block (M × K_tile)   -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)          # fp16 → fp32

        # load B block (N × K_tile) as if transposed:
        # original B layout is (K, N).  To get Bᵀ we need stride_k = N, stride_n = 1.
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)          # fp16 → fp32

        # accumulate
        acc += tl.dot(a, b)            # (M,K)·(K,N) → (M,N)

        # advance pointers for next K‑tile
        # (handled implicitly by recomputing a_ptrs/b_ptrs each iteration)
    # -----------------------------------------------------------------------
    # write back result
    # -----------------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Helper utilities
# --------------------------------------------------------------------------- #
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    """Fused LayerNorm over the last dimension."""
    return F.layer_norm(x, (weight.shape[0],), weight, bias, eps=1e-5)


def linear(x: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    """Linear projection without bias."""
    return F.linear(x, weight)


def launch_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B*H, N, N]  (float16, contiguous, row‑major)
    returns      : [B*H, N, N]  (float32)
    """
    BxH, N, _ = left.shape
    M = N
    K = N

    # strides for a (batch, M, K) – contiguous layout → (K*N, K, 1)
    stride_az, stride_am, stride_ak = left.stride()
    # strides for b (batch, K, N) – contiguous layout → (N*K, N, 1)
    stride_bz, stride_bk, stride_bn = right.stride()
    # output strides (batch, M, N)
    out_shape = (BxH, M, N)
    out = torch.empty(out_shape, dtype=torch.float32, device=left.device)
    stride_cz, stride_cm, stride_cn = out.stride()

    # tuning parameters (empirically good on Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = triton.cdiv(M, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_blocks = BxH * num_blocks_m * num_blocks_n

    grid = (total_blocks,)

    batched_gemm_transpose_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        M, N, K,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim]   (float32, CUDA)
        mask    – Tensor[B, N, N]        (float32 0/1, optional)
        weights – dict with all linear / LN parameters
        config  – dict with keys:
                  * dim         – input channel size
                  * hidden_dim  – projection size
                  * nomask      – bool, when True mask is all‑ones

    Returns
    -------
    Tensor[B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------- #
    x = layer_norm(inp, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections to hidden dim
    # ------------------------------------------------------------------- #
    left = linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ (optional) binary mask broadcast over hidden dimension
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → broadcast
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------- #
    #   reshape to (B, H, N, N) → (B*H, N, N) for a simple 3‑D GEMM
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    left = left.view(-1, N, N).to(torch.float16)   # [B*H, N, N]
    right = right.view(-1, N, N).to(torch.float16)

    # launch Triton GEMM
    out = launch_batched_gemm(left, right)        # [B*H, N, N] fp32

    # reshape back to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden → hidden)
    # ------------------------------------------------------------------- #
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
