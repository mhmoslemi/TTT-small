"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The only O(N³) part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed head d this is exactly a matrix multiplication
L = left[b,:, :, d] (N×N)  and  R = right[b,:, :, d] (N×N)
out[b,:, :, d] = L @ Rᵀ                     (N×N)

We launch a Triton kernel that computes many independent GEMMs
simultaneously (one per (batch, head) pair).  The tensors are
converted to FP16 for bandwidth‑limited loads, accumulated in FP32,
and written back as FP32.  All other operations (LayerNorm, Linear,
masking, gating) stay in PyTorch because they are negligible compared
to the O(N³) contraction.

The kernel works for any batch size B∈{1,2}, any sequence length N,
any model dimension dim and any hidden dimension hidden_dim.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr,                     # left  (B*H, N, N)   FP16
    b_ptr,                     # right (B*H, N, N)   FP16
    c_ptr,                     # out   (B*H, N, N)  FP32
    N,                         # sequence length
    stride_ah, stride_ai, stride_ak,   # strides of A
    stride_bh, stride_bk, stride_bj,   # strides of B
    stride_ch, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)        # block row  i
    pid_n = tl.program_id(1)        # block col  j
    pid_z = tl.program_id(2)        # flattened (batch * head)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to the (pid_z)-th matrix pair
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the contracted axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  : (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float16)               # ensure fp16

        # Bᵀ[k, j]  : (BLOCK_K, BLOCK_N)
        # original B is indexed as B[j, k]; loading it transposed matches dot()
        b_ptrs = (
            b_batch_ptr
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float16)

        # dot product accumulates in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store result
    c_ptrs = (
        c_batch_ptr
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, H, N, N)  FP16
    returns    : (B, H, N, N)  FP32  where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    # flatten batch*head dimension for kernel simplicity
    left_f = left.reshape(B * H, N, N)
    right_f = right.reshape(B * H, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bk, stride_bj = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable block sizes – chosen for Hopper (SM90) balance of shared‑mem & occupancy
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        B * H,                            # flattened batch*head
    )
    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bk, stride_bj,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Only the O(N³) contraction is executed in Triton; all other
    operations stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)        # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape: B,N,N)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.to(device).unsqueeze(-1)           # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ per hidden channel
    #    Convert to (B, H, N, N) layout expected by the kernel.
    # ------------------------------------------------------------------
    #   FP16 for the heavy memory traffic, FP32 accumulation inside kernel.
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm = trimul_gemm(left_perm, right_perm)      # (B,H,N,N) float32

    # back to (B,N,N,hidden_dim)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B,N,N,dim)

    # Ensure float32 output as required by the spec
    return out.to(torch.float32)
