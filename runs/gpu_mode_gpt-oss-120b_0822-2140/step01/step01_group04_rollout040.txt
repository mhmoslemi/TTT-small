"""
TriMul “outgoing” kernel (AlphaFold‑3).

The costly part of the module is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We move this contraction to a Triton kernel that computes,
for every (batch, head) pair, the product
    (N,N) @ (N,N)ᵀ
where the second operand is accessed transposed on‑the‑fly.
All remaining work (LayerNorms, Linear projections, gating) stays in
PyTorch – they are negligible compared with the O(N³) work.

The kernel works on tensors of shape (B, H, N, N) that are contiguous
(row‑major).  Each program instance computes a TILE_M × TILE_N tile of
the output matrix using FP16 loads, FP32 accumulation and writes the
result back in FP32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    N, H,                               # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_i,  # note swapped j/i
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, k, j]
    i.e. A @ Bᵀ  (both A and B have shape (N, N))
    """
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_hb = tl.program_id(2)  # combined batch*head

    batch = pid_hb // H
    head  = pid_hb % H

    # -------------------- tile offsets --------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # -------------------- loop over K --------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A : (i, k)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_i[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need (k, j) from the original (j, k) layout
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dimension
                  + offs_j[None, :] * stride_b_i)  # j as second dimension
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N)

    # -------------------- write back --------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_i[:, None] * stride_c_i
              + offs_j[None, :] * stride_c_j)

    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32 / bf16)
        - mask         : torch.Tensor [B, N, N] (bool / float) – may be all‑ones
        - weights      : dict of torch.Tensors containing all linear / LN params
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # helpers for weight lookup
    # --------------------------------------------------------------
    w = lambda name: weights[name].to(device)

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm over the channel dimension (dim)
    # --------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N, H]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N, H]

    # --------------------------------------------------------------
    # 3️⃣  Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; we broadcast on hidden dim
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣  Gating logits + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Heavy O(N³) contraction via Triton
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) – contiguous layout required
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in **elements** (not bytes)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, k)  – we will swap i/k when passing
    c_s = out_t.stride()     # (batch, head, i, j)

    # ------------------------------------------------------------------
    # launch parameters – tuned for Hopper (SM90)
    # ------------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # i‑tiles
        triton.cdiv(N, TILE_N),          # j‑tiles
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],   # note: stride_b_j = stride_i, stride_b_i = stride_k
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,          # Hopper benefits from more warps
        num_stages=2,         # double‑buffering
    )

    # back to (B, N, N, H)
    out = out_t.permute(0, 2, 3, 1)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣  Final LayerNorm + gating + linear back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))  # [B, N, N, dim]

    return out.to(dtype)
