"""
TriMul “outgoing” forward pass (no gradients).

The main cost is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply
    (B·H, N, N)  @  (B·H, N, N)ᵀ
with H = hidden_dim.  
We fuse this GEMM with a tiny Triton kernel that tiles the
M‑ and N‑ dimensions (rows / cols) while streaming over K.
All LayerNorm / Linear / gating work is kept in PyTorch – this
avoids the need for a triton‑side mean/variance reduction and
keeps the kernel simple and fast on Hopper (SM‑90).

Key ideas (lessons):
  * Reformulate the einsum as a batched GEMM → huge speed‑up.
  * Permute to (B, H, N, N) and flatten (B·H) before launching – the
    kernel only sees a 3‑D tensor, which matches the classic GEMM
    pattern.
  * Pass exact strides to the kernel – no “assume contiguous” hacks.
  * Use fp32 accumulation (tl.float32) even if the inputs are fp16
    (optional, we stay in fp32 for numerical safety).
  * Tile with BLOCK_M = BLOCK_N = 64 and BLOCK_K = 64; these sizes
    give good occupancy on H200 and fit nicely in a single warp‑wide
    dot product.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ   where left/right have shape
#   (M, N, K)   with K == N and M = B * hidden_dim.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp32  [M, N, K]
    right_ptr,         # *fp32  [M, N, K]
    out_ptr,           # *fp32  [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over output rows   (i)
    pid_n = tl.program_id(1)          # tile over output cols   (j)
    pid_z = tl.program_id(2)          # which matrix (batch*head)

    # ---------------------------------------------------------
    # 1) Compute the absolute offsets of the tile
    # ---------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---------------------------------------------------------
    # 2) Pointers to the current tile in the three tensors
    # ---------------------------------------------------------
    # left[z, i, k]   → (z * stride_lm) + (i * stride_ln) + (k * stride_lk)
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # rightᵀ[z, j, k] = right[z, k, j]
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # 3) Main K‑loop (tiled GEMM)
    # ---------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance K‑dimension
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ---------------------------------------------------------
    # 4) Write the tile back
    # ---------------------------------------------------------
    out_ptrs = out_ptr + pid_z * stride_om \
                        + offs_m[:, None] * stride_on \
                        + offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a fused Triton GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
            - weights      : dict of torch.Tensor – all linear / LN params
            - config       : dict with keys
                * "dim"          – input/output channel dimension
                * "hidden_dim"   – internal projection dimension

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Fuse the N³ contraction with Triton
    #    reshape: (B,N,N,H) → (B, H, N, N) → (M, N, N) where M = B*H
    # ------------------------------------------------------------------- #
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim

    left_ = left_.view(M, N, N)
    right_ = right_.view(M, N, N)

    out_ = torch.empty_like(left_, device=device)   # fp32

    # strides in elements (required by Triton)
    stride_lm, stride_ln, stride_lk = left_.stride()
    stride_rm, stride_rn, stride_rk = right_.stride()
    stride_om, stride_on, stride_ok = out_.stride()

    # ------------------------------------------------------------------- #
    # launch configuration – tile sizes chosen for Hopper
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),       # rows  (i)
        triton.cdiv(N, BLOCK_N),       # cols  (j)
        M                             # independent matrices
    )

    tri_mul_kernel[grid](
        left_,
        right_,
        out_,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,                  # 4 warps give good occupancy on SM90
    )

    # reshape back to the original layout: (B, N, N, H)
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6) Output LayerNorm + gating + final linear (dim -> dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                     # broadcast gate H
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
