"""
TriMul “outgoing” implementation – heavy reduction computed with a
batched GEMM Triton kernel.

The contraction  
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  
is exactly a matrix‑multiply for each (batch, head) pair:
    left  → (N,N)   ,   right → (N,N)ᵀ   →   out (N,N)

We
  1. perform all LayerNorm / Linear / gating in PyTorch (they are cheap),
  2. permute the tensors to shape (B, H, N, N) and launch a Triton kernel that
     computes B·H independent GEMMs (A @ Bᵀ) in fp16 with fp32 accumulation,
  3. permute the result back and finish the forward pass in PyTorch.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimension up to 384.  All tensors are expected to be on CUDA.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM A @ Bᵀ  (A,B : fp16, accumulate in fp32)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes
    stride_ob_a, stride_am, stride_ak,       # A strides (outer, row, col)
    stride_ob_b, stride_bk, stride_bn,       # B strides (outer, row, col)
    stride_ob_c, stride_cm, stride_cn,       # C strides (outer, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ Bᵀ for a batch of independent matrices."""
    pid_outer = tl.program_id(0)   # which (batch, head) pair
    pid_m = tl.program_id(1)       # block-row index
    pid_n = tl.program_id(2)       # block-col index

    # ------------------------------------------------------------------- #
    # Offsets inside the current block
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A[i,k] – shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_outer * stride_ob_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ[k,j] → we read B row‑wise (k, j)
        b = tl.load(
            b_ptr
            + pid_outer * stride_ob_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # a,b are fp16; tl.dot accumulates into fp32 (acc dtype)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------- #
    # Write out the result
    # ------------------------------------------------------------------- #
    tl.store(
        c_ptr
        + pid_outer * stride_ob_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Helper: element‑wise mask application (fallback to PyTorch when mask is 1)
# --------------------------------------------------------------------------- #
def apply_mask(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Broadcast mask of shape [B,N,N] over the last dim of x."""
    if mask is None:
        return x
    return x * mask.unsqueeze(-1).to(x.dtype)


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, CUDA)
        - mask         : [B, N, N]       (float32 0/1, CUDA) – may be None
        - weights      : dict of tensors (see code)
        - config       : dict with keys:
              * dim        – input channel size
              * hidden_dim – projection size
              * nomask     – bool, True → ignore mask

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # Unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ Layer‑norm on the input
    # ------------------------------------------------------------------- #
    def layer_norm(t, w, b):
        return F.layer_norm(t, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    def linear(t, w):
        return F.linear(t, w)  # bias‑free linear

    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------- #
    # 2️⃣ Projections
    # ------------------------------------------------------------------- #
    left = linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        left = apply_mask(left, mask)
        right = apply_mask(right, mask)

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM via Triton
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape
    outer = B * H                               # number of independent GEMMs

    # Permute to (outer, N, N) where outer = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Allocate output (float32) in permuted layout
    out_perm = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides (in elements) for the three tensors
    stride_ob_a = left_fp16.stride(0)   # outer stride
    stride_am = left_fp16.stride(1)     # row stride (N)
    stride_ak = left_fp16.stride(2)     # col stride (1)

    stride_ob_b = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)
    stride_bn = right_fp16.stride(2)

    stride_ob_c = out_perm.stride(0)
    stride_cm = out_perm.stride(1)
    stride_cn = out_perm.stride(2)

    # Tile sizes – chosen to fit H200 SMs and shared‑memory limits
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        outer,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_perm,
        M=N,
        N=N,
        K=N,
        stride_ob_a=stride_ob_a,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ob_b=stride_ob_b,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ob_c=stride_ob_c,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_perm.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm
    # ------------------------------------------------------------------- #
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
