"""
TriMul (outgoing) – fast forward pass.

The only O(N³·H) work is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
We reshape the tensors to (B·H, N, N) and launch a tiled GEMM kernel that
computes A @ Bᵀ for each (batch, head) pair.  The kernel works in fp16
but accumulates in fp32 for numerical stability.  All other operations
(LayerNorm, Linear, Sigmoid) stay in PyTorch; the final element‑wise
multiplication with the output‑gate is fused with a tiny Triton kernel
so that the solution satisfies the “must use a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Element‑wise multiplication (used for out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr,          # first operand (flattened)
    b_ptr,          # second operand (flattened)
    out_ptr,        # output (flattened, may alias a_ptr)
    total,          # total number of elements
    stride_a: tl.constexpr,
    stride_b: tl.constexpr,
    stride_out: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel: C = A @ Bᵀ   (A,B shaped (B·H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                         # matrix sizes (all equal to seq_len)
    stride_batch_a, stride_am, stride_ak,   # A strides
    stride_batch_b, stride_bk, stride_bn,   # B strides (note: Bᵀ → swap last two strides)
    stride_batch_c, stride_cm, stride_cn,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each batch index `pid` compute
       C[pid, i, j] = Σ_k A[pid, i, k] * B[pid, j, k]  (i.e. A @ Bᵀ)
    """
    pid = tl.program_id(0)          # batch‑head id
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block column

    # -------------------- block offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard against out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch‑head
    a_batch_ptr = a_ptr + pid * stride_batch_a
    b_batch_ptr = b_ptr + pid * stride_batch_b
    c_batch_ptr = c_ptr + pid * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch_ptr
            + (offs_m[:, None] * stride_am)
            + (offs_k[None, :] * stride_ak)
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B tile (remember we need Bᵀ → we treat B as (N, K) and transpose via strides)
        b_ptrs = (
            b_batch_ptr
            + (offs_k[:, None] * stride_bk)   # K‑stride (now the row of Bᵀ)
            + (offs_n[None, :] * stride_bn)   # N‑stride (now the column of Bᵀ)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # -------------------- write result --------------------
    c_ptrs = (
        c_batch_ptr
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn)
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32/bool/None, CUDA)
        * weights_dict : model parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = inp.device
    B, N, _, _ = inp.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction using a tiled GEMM kernel
    # --------------------------------------------------------------
    #   shape -> (B, H, N, N) → (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    BH, _, _ = left.shape[0] * left.shape[1], left.shape[2], left.shape[3]  # just for readability
    left_flat = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_flat = right.view(B * hidden_dim, N, N).to(torch.float16)

    # output buffer (fp16, will be cast back to fp32 later)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * hidden_dim,                     # batch‑head dimension
        triton.cdiv(N, BLOCK_M),           # rows
        triton.cdiv(N, BLOCK_N),           # cols
    )

    # NOTE: for Bᵀ we swap the last‑two strides of `right_flat`
    triton_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0),  # stride_batch_a
        left_flat.stride(1),  # stride_am   (row stride)
        left_flat.stride(2),  # stride_ak   (col stride)
        right_flat.stride(0), # stride_batch_b
        right_flat.stride(2), # stride_bk   (row stride of Bᵀ)
        right_flat.stride(1), # stride_bn   (col stride of Bᵀ)
        out_flat.stride(0),   # stride_batch_c
        out_flat.stride(1),   # stride_cm   (row stride)
        out_flat.stride(2),   # stride_cn   (col stride)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim] and cast to fp32
    out = out_flat.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate (fused Triton)
    # --------------------------------------------------------------
    out_flat2 = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat2.numel()
    BLOCK = 1024
    grid_mul = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid_mul](
        out_flat2,
        out_gate_flat,
        out_flat2,
        total,
        stride_a=1,
        stride_b=1,
        stride_out=1,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimensionality
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
