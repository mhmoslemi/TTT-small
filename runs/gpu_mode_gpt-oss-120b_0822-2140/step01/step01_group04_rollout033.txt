"""
TriMul “outgoing” kernel (AlphaFold3).

The expensive part of the module is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We compute it with a custom Triton kernel that multiplies, for every
(batch, hidden) pair, the (N,N) tensor `left` with the transpose of the
(N,N) tensor `right`.  All other operations (LayerNorm, Linear, gating,
final LayerNorm/Linear) stay in PyTorch – they are cheap compared to the
O(N³) contraction.

The kernel works on tensors of shape (B, H, N, N) that are contiguous in
memory (row‑major).  Each program instance computes a TILE_M × TILE_N
block of the output matrix for a single (batch, hidden) pair.
Inputs are cast to FP16 for bandwidth, accumulation is in FP32, and the
result is stored back as FP32 (matching the reference implementation).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)^T
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_k,   # note: B is accessed as B[j,k]
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,   # how many rows a block of programs works on
    GROUP_SIZE_N: tl.constexpr,   # how many cols a block of programs works on
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]   (i.e. A @ Bᵀ)
    A, B, C are stored as (B, H, N, N) in row‑major order.
    """

    pid = tl.program_id(0)               # combined (row block, col block)
    pid_hb = tl.program_id(1)            # combined batch * head

    # -------------------------------------------------------------
    # decode program ids
    # -------------------------------------------------------------
    batch = pid_hb // H
    head  = pid_hb %  H

    # split tile grid into a 2‑D grid (row, col)
    num_blocks_m = tl.cdiv(N, TILE_M)
    num_blocks_n = tl.cdiv(N, TILE_N)

    # group rows/cols to improve L2 reuse (see triton matmul example)
    pid_m = (pid // GROUP_SIZE_N) % num_blocks_m
    pid_n = (pid %  GROUP_SIZE_N)

    # global offsets for this tile
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # loop over K dimension
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A: (M, K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ: we need B[j, k] → load as (K, N)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_k   # k as first dim
                  + offs_n[None, :] * stride_b_j)  # j as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # -------------------------------------------------------------
    # store result
    # -------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for
    the O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim] (float32 / bf16)
        - mask         : torch.Tensor   [B, N, N]    (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch parameters (they are already on the correct device)
    # ------------------------------------------------------------------
    def w(name): return weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False) and optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden]
    right = F.linear(x, w("right_proj.weight"))  # [B,N,N,hidden]

    if not nomask:
        # mask shape -> [B,N,N,1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and apply to projections
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))   # [B,N,N,hidden]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy O(N³) contraction via Triton
    #    left, right  -> (B, H, N, N)  (contiguous)
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in elements (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, j, k)   note: we will address (k,j)
    c_s = out_t.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64
    # Grouping improves L2 cache reuse (taken from triton matmul example)
    GROUP_SIZE_M = 8
    GROUP_SIZE_N = 8

    grid = (
        triton.cdiv(N, TILE_M) * triton.cdiv(N, TILE_N),   # (row‑block, col‑block) flattened
        B * hidden_dim,                                   # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        GROUP_SIZE_N=GROUP_SIZE_N,
        num_warps=4,
        num_stages=2,
    )

    # back to original layout: [B, N, N, hidden]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate          # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B, N, N, dim]

    return out.to(dtype)
