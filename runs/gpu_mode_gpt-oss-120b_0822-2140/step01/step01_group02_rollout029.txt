## Triton‑accelerated “outgoing” TriMul

```python
"""
TriMul (outgoing) – forward only.

The expensive part is the three‑dimensional contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batch of GEMMs:
    left  : (B, H, N, N)
    right : (B, H, N, N)
    out   = left @ rightᵀ   (per (b, h) pair)

We fuse the hidden dimension H into the batch dimension (B*H) and
run a single batched GEMM in Triton, using fp16 arithmetic and
fp32 accumulation.  All surrounding ops (LayerNorm, linear
projections, gates, final linear) are performed in PyTorch because
they are negligible compared to the O(N³·H) work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   where A,B have shape (Batch, M, K) and (Batch, N, K)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix extents (int)
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col) – original layout
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed."""
    pid_batch = tl.program_id(2)       # index over (B*H)
    pid_m = tl.program_id(0)           # block‑row index
    pid_n = tl.program_id(1)           # block‑col index

    # --------------------------------------------------------------
    # Block offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current block of A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_az \
                 + offs_m[:, None] * stride_am \
                 + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n) -> row‑k, col‑n.
    b_ptrs = b_ptr + pid_batch * stride_bz \
                 + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (more accurate than staying in FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Masked loads – note that the mask for the K dimension is constant
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # tl.dot does fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Move to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write‑back (store as fp16 to reduce bandwidth)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host helper that launches the kernel
# ----------------------------------------------------------------------
def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : torch tensors of shape (B*H, N, N), dtype torch.float16, contiguous.
    Returns c of the same shape/dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, N, K = a.shape
    assert K == N, "square matrices are expected"

    c = torch.empty_like(a)

    # Strides in **elements** (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper/SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # program_id(0)
        triton.cdiv(N, BLOCK_N),   # program_id(1)
        batch,                     # program_id(2)
    )
    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # works well for 64×64 tiles on Hopper
    )
    return c


# ----------------------------------------------------------------------
# Required entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
          - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
          - mask         : torch.Tensor   [B, N, N]   (bool/float32) – may be None
          - weights      : dict of torch.Tensor (float32) – all linear / norm params
          - config       : dict with keys
                “dim”, “hidden_dim”, “nomask” (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; bring it to the same dtype
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Rearrange to (B, H, N, N) and cast to fp16 for the GEMM
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Fuse B and H into a single batch dimension expected by the kernel
    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM: out = left @ rightᵀ   (still fp16)
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, hidden_dim) and cast to fp32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate shape (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
