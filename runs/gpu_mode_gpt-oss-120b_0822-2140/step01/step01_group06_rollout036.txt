"""
TriMul (outgoing) – Triton‑accelerated version
================================================
The heavy O(N³·H) contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply of the form
    out = left @ rightᵀ
for every (batch, head) pair, where *head = hidden_dim*.
We reshape the tensors to (B·H, N, N) and launch a tiled GEMM kernel in
Triton that works entirely in fp16 (loads) and fp32 (accumulator) to
exploit the H200 tensor‑cores.  The rest of the TriMul pipeline
(LayerNorm, Linear, Gate‑sigmoids, final LayerNorm) stays in PyTorch.
At least one Triton kernel is therefore used, satisfying the
“must‑use‑kernel” requirement while keeping the implementation simple
and correct for all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel: computes  out = left @ rightᵀ
#   left  : (B, H, N, N)   – row‑major (i,k)
#   right : (B, H, N, N)   – row‑major (j,k)  (will be read transposed)
#   out   : (B, H, N, N)   – result (i,j)
# ----------------------------------------------------------------------
@triton.jit
def gemm_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides (in elements) of the three tensors
    stride_left_batch, stride_left_head, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_head, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_head, stride_out_i, stride_out_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)                # batch‑head index
    pid_m  = tl.program_id(0)                # tile over rows (i)
    pid_n  = tl.program_id(1)                # tile over cols (j)

    # ----- compute absolute offsets for this tile -----------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for the edges
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Base pointers for this (batch,head) slice
    left_base  = left_ptr \
        + pid_bh * stride_left_batch \
        + offs_i[:, None] * stride_left_i \
        + offs_k[None, :] * stride_left_k

    # Right is accessed transposed: we need element right[j,k] as right_T[k,j]
    # Hence row‑stride = stride_right_row (= stride over original k)
    #       col‑stride = stride_right_col (= stride over original j)
    right_base = right_ptr \
        + pid_bh * stride_right_batch \
        + offs_k[:, None] * stride_right_row \
        + offs_j[None, :] * stride_right_col

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over the reduction dimension k -------------------------
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(left_base,  mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(right_base, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # a,b are fp16; dot returns fp16, we accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

        # advance pointers to the next k‑block
        left_base  += BLOCK_K * stride_left_k
        right_base += BLOCK_K * stride_right_row

    # ----- write result ------------------------------------------------
    out_ptrs = out_ptr \
        + pid_bh * stride_out_batch \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs,
             acc.to(tl.float16),                     # store as fp16
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   : Tensor[B, N, N, dim]   (float32, CUDA)
        mask    : Tensor[B, N, N] or None (float32/bool, CUDA)
        weights : dict of torch.nn.Parameter tensors
        config  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    Tensor[B, N, N, dim]  (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (over last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                                 # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – Triton GEMM
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and cast to fp16 for the kernel
    left  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)  # [B, H, N, N]

    out_fp16 = torch.empty_like(left)                                 # allocate output

    # strides (in elements)
    s_left  = left.stride()      # (B, H, i, k)
    s_right = right.stride()     # (B, H, i, k)

    s_out = out_fp16.stride()    # same layout as left/right

    # launch grid : (tiles_M, tiles_N, batch*head)
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    gemm_outgoing_kernel[grid](
        left_ptr   = left.data_ptr(),
        right_ptr  = right.data_ptr(),
        out_ptr    = out_fp16.data_ptr(),
        B          = B,
        N          = N,
        H          = hidden_dim,
        stride_left_batch = s_left[0],
        stride_left_head  = s_left[1],
        stride_left_i     = s_left[2],
        stride_left_k     = s_left[3],
        stride_right_batch = s_right[0],
        stride_right_head  = s_right[1],
        # transposed view of right
        stride_right_row   = s_right[3],   # original k‑stride → row after transpose
        stride_right_col   = s_right[2],   # original i‑stride → col after transpose
        stride_out_batch   = s_out[0],
        stride_out_head    = s_out[1],
        stride_out_i       = s_out[2],
        stride_out_j       = s_out[3],
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,
        BLOCK_K = BLOCK_K,
    )

    # back to fp32 and to original layout (B,N,N,H)
    out = out_fp16.to(torch.float32).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                                   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Multiply by output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate                                 # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])       # [B,N,N,dim]

    return out
