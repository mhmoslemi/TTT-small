"""
TriMul (outgoing) – fast forward pass.
-------------------------------------------------
The only O(N³·H) work is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiply A @ Bᵀ for every (batch, head) pair.
We reshape the tensors to (B*H, N, N) and launch a single Triton
GEMM kernel that processes all heads in parallel.  All other
operations (LayerNorm, Linear, sigmoid‑gates, final LayerNorm and
the output‑gate multiplication) are done with PyTorch for simplicity.
The kernel works in fp16 (fast tensor cores) while accumulating in fp32
and writes fp32 results, guaranteeing the same numerical precision as
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ   (A,B : (BATCH, M, K), C : (BATCH, M, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                # A   (batch, M, K)  fp16
    b_ptr,                # B   (batch, N, K)  fp16  (note: we will load Bᵀ)
    c_ptr,                # C   (batch, M, N)  fp32
    BATCH, M, N, K,
    stride_az, stride_ak, stride_ai,      # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bj,      # strides for B (batch, col, row) – we treat Bᵀ
    stride_cz, stride_cm, stride_cn,      # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) decode the 3‑D grid (batch, tile_m, tile_n)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)

    # group tiles in M dimension to improve L2 reuse
    num_blocks_in_group = GROUP_SIZE_M * num_blocks_n
    group_id = pid_in_batch // num_blocks_in_group
    first_pid_m = group_id * GROUP_SIZE_M
    group_pid = pid_in_batch - group_id * num_blocks_in_group

    pid_m = first_pid_m + group_pid // num_blocks_n
    pid_n = group_pid % num_blocks_n

    # --------------------------------------------------------------
    # 2) compute the offsets of the tile in A, B and C
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch
    a_batch_ptr = a_ptr + batch_id * stride_az
    b_batch_ptr = b_ptr + batch_id * stride_bz
    c_batch_ptr = c_ptr + batch_id * stride_cz

    # ------------------------------------------------------------------
    # 3) iterate over K dimension
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A tile  (M x K)
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_ai)   # row stride
            + (offs_k[None, :] * stride_ak),   # col stride
            mask=mask_m[:, None],
            other=0.0,
        )
        # load Bᵀ tile (N x K)  – we read B as (batch, N, K) and treat it as transposed
        b = tl.load(
            b_batch_ptr
            + (offs_n[:, None] * stride_bj)   # row stride (original col)
            + (offs_k[None, :] * stride_bk),   # col stride (original row)
            mask=mask_n[:, None],
            other=0.0,
        )
        # cast to fp16, compute dot in fp16, accumulate in fp32
        a = a.to(tl.float16)
        b = b.to(tl.float16)
        acc += tl.dot(a, b)                # (BLOCK_M, BLOCK_N) fp32

        # advance K pointers
        a_batch_ptr += BLOCK_K * stride_ak
        b_batch_ptr += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # 4) write result
    # ------------------------------------------------------------------
    c = acc.to(tl.float32)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the GEMM kernel
# ----------------------------------------------------------------------
def batched_gemm(a_fp16: torch.Tensor, b_fp16: torch.Tensor) -> torch.Tensor:
    """
    a_fp16: (B*H, N, N)   fp16
    b_fp16: (B*H, N, N)   fp16
    returns: (B*H, N, N)  fp32
    """
    assert a_fp16.shape == b_fp16.shape
    batch, N, K = a_fp16.shape
    M = N
    assert K == N

    # output in fp32
    out_fp32 = torch.empty_like(a_fp16, dtype=torch.float32)

    # strides (in elements)
    stride_az, stride_ai, stride_ak = a_fp16.stride()
    stride_bz, stride_bj, stride_bk = b_fp16.stride()
    stride_cz, stride_cm, stride_cn = out_fp32.stride()

    # tiling parameters – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_SIZE_M = 8

    num_tiles_m = triton.cdiv(M, BLOCK_M)
    num_tiles_n = triton.cdiv(N, BLOCK_N)
    total_programs = batch * num_tiles_m * num_tiles_n

    grid = (total_programs,)

    batched_gemm_kernel[grid](
        a_fp16,
        b_fp16,
        out_fp32,
        batch,
        M,
        N,
        K,
        stride_az,
        stride_ak,
        stride_ai,
        stride_bz,
        stride_bk,
        stride_bj,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )
    return out_fp32


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   float32 CUDA
        - mask_tensor  : [B, N, N]       float32/bool CUDA   (or None)
        - weights_dict : model parameters (see skeleton)
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32 CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (over last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if supplied)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction using Triton GEMM
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) and make contiguous for correct strides
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # cast to fp16 for tensor‑core GEMM
    left_fp16 = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)

    # GEMM: left @ rightᵀ  (both (B*H, N, N))
    out_fp32 = batched_gemm(left_fp16, right_fp16)   # (B*H, N, N) fp32

    # reshape back to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
