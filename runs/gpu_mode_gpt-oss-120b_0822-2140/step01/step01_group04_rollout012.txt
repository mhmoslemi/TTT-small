"""
Outgoing TriMul (AlphaFold3) – forward only.

The costly O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that treats the hidden
dimension `d` as “heads”. For each (batch, head) pair we compute a
matrix‑multiply `left @ rightᵀ` where both operands are (N, N). All
other operations (LayerNorm, Linear, gating) remain in PyTorch because
they are negligible compared with the N³ work.

The kernel works on tensors of shape (B, H, N, N) laid out in
row‑major (contiguous) order, uses FP16 loads for bandwidth, FP32
accumulation for accuracy and writes the result back in FP32.
The host entry‑point `custom_kernel` follows the signature required by
the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul  C = A @ Bᵀ   (A,B shape (B,H,N,N))
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers to the three tensors
    N, H,                         # problem sizes (seq_len, hidden_dim)
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Compute, for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  with both operands (N, N).
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ---- offsets for the tile ----------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ---- loop over the reduction axis k -------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A : (TILE_M, TILE_K)  = A[batch, head, i, k]
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[j, k] -> load as (TILE_K, TILE_N)
        #   B layout: (batch, head, i, j)
        #   we treat i as j (row) and j as k (col) for the transposed view.
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j    # k acts as first dim
                  + offs_n[None, :] * stride_b_i)   # n (=j) acts as second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) -> (TILE_M,TILE_N)

    # ---- write result -------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights      : dict of torch.Tensor parameters
        - config       : dict with at least ``dim`` and ``hidden_dim`` entries,
                         optional ``nomask`` (True → no mask).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------
    def get(name):
        return weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=get("norm.weight"),
        bias=get("norm.bias"),
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free) and masking
    # ------------------------------------------------------------------
    # (B,N,N,dim) @ (hidden_dim, dim)^T -> (B,N,N,hidden_dim)
    left  = F.linear(x, get("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, get("right_proj.weight")) # [B,N,N,hidden_dim]

    if not nomask:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, get("left_gate.weight")))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, get("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, get("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Heavy O(N³) contraction – Triton batched mat‑mul
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in **elements** (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)
    c_s = out_t.stride()    # (batch, head, i, j)

    # launch configuration – tune TILE_* if needed
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    trimul_outgoing_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,
    )

    # back to the original layout (B, N, N, hidden_dim)
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Final LayerNorm, gating and projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=get("to_out_norm.weight"),
        bias=get("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, get("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
