"""
TriMul “outgoing” kernel (AlphaFold3).

The dominant cost is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`:
    (B, H, N, N) @ (B, H, N, N)ᵀ  → (B, H, N, N)

We keep all cheap operations (LayerNorm, Linear, gating) in PyTorch
and accelerate only this O(N³) part with a fused Triton kernel.
The kernel works on tensors of shape (B, H, N, N) stored in
row‑major order.  Each program instance computes a TILE_M × TILE_N
block of the output for a single (batch, head) pair, loading tiles of
the two input matrices in FP16, accumulating in FP32, and writing the
result back in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_j, stride_b_i,  # note: B is accessed as B[j,k]
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Computes for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B have shape (N, N)).
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head = pid_hb % H

    # -------------------------------------------------
    # Offsets for the current tile
    # -------------------------------------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over K dimension in blocks
    # -------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ----- load A[i,k] -------------------------------------------------
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + head * stride_a_head
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # ----- load Bᵀ[k,j]  (i.e. B[j,k]) -------------------------------
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + head * stride_b_head
            + offs_k[:, None] * stride_b_i   # k as first dimension
            + offs_j[None, :] * stride_b_j   # j as second dimension
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # ----- dot product over K -----------------------------------------
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # -------------------------------------------------
    # Store the result
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + head * stride_c_head
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (bool / float, may be all‑ones)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-6

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weights (they are already on the correct device)
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and optional masking
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # [B, N, N, H]
    right = F.linear(x, w("right_proj.weight"))  # [B, N, N, H]

    if not nomask:
        # mask shape -> [B,N,N,1] for broadcasting
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) – also no bias in the original implementation
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy contraction with Triton
    #    reshape to (B, H, N, N) and make contiguous for simple strides
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # allocate output tensor (FP32 accumulator, will store FP32)
    out_t = torch.empty_like(left_t, dtype=torch.float32)   # [B, H, N, N]

    # strides in elements (not bytes)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will read transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # tile sizes – tuned for Hopper (sm_90) with 128‑bit loads
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # program_id(0) – rows
        triton.cdiv(N, TILE_N),          # program_id(1) – cols
        B * hidden_dim,                  # program_id(2) – batch * head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[3], b_s[2],   # note swapped i/j for transposed access
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,                     # empirical choice for H200
    )

    # back to original layout: (B, N, N, H)
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Final LayerNorm + gating + linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate               # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))   # [B, N, N, dim]

    return out.to(input_tensor.dtype)   # keep same dtype as input (usually fp32)
