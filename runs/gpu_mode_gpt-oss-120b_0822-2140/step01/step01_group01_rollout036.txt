"""
tri_mul_outgoing_triton.py
~~~~~~~~~~~~~~~~~~~~~~~~~~

Implementation of the “outgoing” TriMul operator (AlphaFold‑3 style) where the
dominant O(N³) contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is reformulated as a batched matrix‑multiply   L @ Rᵀ   for every hidden
dimension *d* (and batch *b*).  The contraction is performed by a single
Triton kernel that computes many independent GEMMs in parallel:

    (B, H, N, N)  ⟶  (B·H, N, N)   A @ Bᵀ   →   (B·H, N, N)

All cheap operations (LayerNorms, linear projections, sigmoids, masking) are
handled in PyTorch – they are negligible compared with the N³ work.  The
kernel works in fp16 for the inputs (fast tensor‑core load/store) while
accumulating in fp32 to preserve numerical quality; the final result is cast
back to fp32.

The kernel parameters (BLOCK_M, BLOCK_N, BLOCK_K) have been tuned for the
H200 (SM‑90) topology:
    * BLOCK_M = BLOCK_N = 128  – matches the 128‑thread warp‑wide MMA size.
    * BLOCK_K = 64              – fits into shared memory and gives good
                                 occupancy for N up to 1024.

The implementation is fully self‑contained and can be launched from the
`custom_kernel` entry‑point required by the evaluator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # tensor pointers
    N, M,                                    # N = seq_len, M = B*H
    stride_ah, stride_ai, stride_ak,         # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,         # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,         # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for many independent (N×N) @ (N×N)ᵀ products.

    Each program instance works on one output tile (i‑range, j‑range) of a
    particular (batch*head) slice identified by pid_z.
    """

    pid_m = tl.program_id(0)   # tile row   i
    pid_n = tl.program_id(1)   # tile col   j
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # ------------------- tile offsets & masks -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------- base pointers for this slice -------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32 (the tensor cores produce fp16 → fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- K‑loop (tile over reduction dim) ---------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   – shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai      # i dimension
            + offs_k[None, :] * stride_ak      # k dimension
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B_T[k, j] – we need B[j, k] → load as (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk      # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj      # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 dot -> fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------- write‑back --------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel (fp16 inputs, fp32 output)
# ----------------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, H, N, N)   torch.float16
    returns    : (B, H, N, N)   torch.float32
    """
    assert left.shape == right.shape
    B, H, N, _ = left.shape
    M = B * H                                   # flattened batch*head

    # reshape to (M, N, N) – contiguous in memory
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64 (required by Triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # --------------------------------------------------------------
    # Tuning parameters – chosen for Hopper SM90
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z (B*H)
    )

    trimul_gemm_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        M,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps per block gives good occupancy on H200
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    The heavy O(N³) contraction is offloaded to a Triton kernel (batched GEMM).
    All surrounding operations (LayerNorm, linear projections, gating, mask)
    are performed with regular PyTorch ops.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: (B, N, N) → (B, N, N, 1)
        mask = mask.to(device).unsqueeze(-1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise product
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ per hidden channel
    #    Transform to (B, H, N, N) layout expected by the kernel.
    # ------------------------------------------------------------------
    # Convert to fp16 for the kernel (tensor‑core friendly)
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B,H,N,N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_fp32 = trimul_gemm_fp16(left_fp16, right_fp16)   # (B,H,N,N) fp32

    # Back to the original layout (B, N, N, H)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final projection back to original dimension (dim)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)       # (dim, hidden)
    out = F.linear(out, to_out_w)                        # (B, N, N, dim)

    # Ensure final dtype matches the reference implementation
    return out.to(torch.float32)
