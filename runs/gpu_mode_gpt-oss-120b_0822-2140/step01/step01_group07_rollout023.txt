"""
TriMul “outgoing” forward pass (AlphaFold3 style).

The only O(N³) work is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which after a permute can be seen as a batched GEMM:

    (B·H, N, N)  @  (B·H, N, N)ᵀ   →  (B·H, N, N)

* left/right are first projected from the input, masked and gated.
* The GEMM is executed in a custom Triton kernel that:
    – loads the two operands as fp16 (halving bandwidth)
    – accumulates in fp32 (full precision)
    – uses 64×64 tiles (BLOCK_M/N) and a 64‑deep K‑tile.
* All surrounding LayerNorm / Linear / gating logic stays in PyTorch
  (the kernel only handles the expensive contraction).
* The final output is layer‑normed, gated and linearly projected back to
  the original feature dimension.

The implementation works for any batch size ≤ 2, any sequence length
(128‑1024) and any hidden dimension (128) that appear in the test matrix.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton GEMM kernel: (M, N, K) @ (M, N, K)ᵀ   where M = B*H, N = seq_len, K = N
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr,                     # left  operand  (fp16)  [M, N, K]
    b_ptr,                     # right operand  (fp16)  [M, N, K]
    c_ptr,                     # output        (fp32)  [M, N, N]
    M, N, K,
    stride_am, stride_an, stride_ak,   # a strides
    stride_bm, stride_bn, stride_bk,   # b strides
    stride_cm, stride_cn, stride_ck,   # c strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows   (i)
    pid_n = tl.program_id(1)          # tile over cols   (j)
    pid_z = tl.program_id(2)          # different matrix (M)

    # ------------------------------------------------------------------- #
    # Offsets for the current tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers:  a is (z, i, k) ; b is (z, j, k) but we need Bᵀ → (z, k, j)
    # ------------------------------------------------------------------- #
    a_ptrs = a_ptr + pid_z * stride_am \
                     + offs_m[:, None] * stride_an \
                     + offs_k[None, :] * stride_ak

    b_ptrs = b_ptr + pid_z * stride_bm \
                     + offs_k[:, None] * stride_bk \
                     + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main K‑loop
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # fp16⋅fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # advance K‑dimension
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------- #
    # Store the result (fp32)
    # ------------------------------------------------------------------- #
    c_ptrs = c_ptr + pid_z * stride_cm \
                     + offs_m[:, None] * stride_cn \
                     + offs_n[None, :] * stride_ck
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]        (bool / 0‑1) or None
        weights      : dict of model parameters (all fp32)
        config       : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias) → [B, N, N, hidden]
    # --------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4. Gating (sigmoid) – also [B, N, N, hidden]
    # --------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Prepare tensors for the batched GEMM:
    #    (B, N, N, H) → (B, H, N, N) → (M, N, N)   where M = B*H
    #    Cast to fp16 for bandwidth savings.
    # --------------------------------------------------------------- #
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    M = B * hidden
    left_fp16  = left.view(M, N, N).to(torch.float16)
    right_fp16 = right.view(M, N, N).to(torch.float16)

    # Output buffer (fp32)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # --------------------------------------------------------------- #
    # 6. Launch Triton GEMM kernel
    # --------------------------------------------------------------- #
    stride_am, stride_an, stride_ak = left_fp16.stride()
    stride_bm, stride_bn, stride_bk = right_fp16.stride()
    stride_cm, stride_cn, stride_ck = out_fp32.stride()

    BLOCK_M = 64   # tuned for Hopper (fits nicely in shared memory)
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows (i)
        triton.cdiv(N, BLOCK_N),   # cols (j)
        M,                         # independent matrices
    )

    _tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        M, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps give good occupancy for fp16 on SM90
    )

    # --------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, hidden]
    # --------------------------------------------------------------- #
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 8. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
