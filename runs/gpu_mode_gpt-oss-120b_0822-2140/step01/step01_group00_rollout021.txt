"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

We implement the expensive contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

as a batched GEMM (left @ rightᵀ) inside Triton.  The surrounding
LayerNorm / Linear / gating operations remain in PyTorch – they are
cheap compared to the N³ reduction.  The kernel works on tensors that
have been permuted to shape (B, H, N, N) and uses fp16 inputs with fp32
accumulation.  The final result is cast back to float32 and
re‑permuted to the original layout.

The launch grid is 3‑D:
    (num_blocks_M, num_blocks_N, B*H)
so each (batch, head) pair runs an independent GEMM.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# Triton kernel – batched GEMM (left @ rightᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def trigmul_outgoing_kernel(
    # pointers
    left_ptr,          # fp16  (B, H, N, K)
    right_ptr,         # fp16  (B, H, N, K)   (we read it transposed)
    out_ptr,           # fp32  (B, H, N, N)

    # tensor sizes
    B, H, N, K,

    # strides (in elements, not bytes)
    stride_lbh, stride_lh, stride_li, stride_lk,   # left
    stride_rbh, stride_rh, stride_rj, stride_rk,   # right (original layout)
    stride_obh, stride_oh, stride_oi, stride_oj,   # out

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------- #
    # program ids
    # ------------------------------------------------------------------- #
    pid_bh = tl.program_id(2)                 # 0 … B*H‑1
    pid_m  = tl.program_id(0)                 # block row   (i)
    pid_n  = tl.program_id(1)                 # block column(j)

    # ------------------------------------------------------------------- #
    # block offsets
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds (last partial blocks)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # pointer to the current (batch, head) slice
    # ------------------------------------------------------------------- #
    left_base  = left_ptr  + pid_bh * stride_lbh
    right_base = right_ptr + pid_bh * stride_rbh
    out_base   = out_ptr   + pid_bh * stride_obh

    # ------------------------------------------------------------------- #
    # accumulator in fp32
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------- #
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left:   shape (M, K)   → (i, k)
        a_ptrs = (left_base
                  + offs_m[:, None] * stride_li   # row stride
                  + offs_k[None, :] * stride_lk)  # col stride
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # rightᵀ: we need element (k, j) = right[b,h,j,k]
        # right stored as (B, H, N, K) -> row‑stride = stride_rj, col‑stride = stride_rk
        b_ptrs = (right_base
                  + offs_k[:, None] * stride_rk   # now k is the *row* index
                  + offs_n[None, :] * stride_rj)  # j is the *column* index
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # tile‑wise dot product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------- #
    # write back (apply masking for partial blocks)
    # ------------------------------------------------------------------- #
    out_ptrs = (out_base
                + offs_m[:, None] * stride_oi
                + offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Helper: launch the Triton GEMM
# --------------------------------------------------------------------------- #
def trigmul_outgoing_tl(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, H, N, N]  (contiguous, fp16)
    returns out  : [B, H, N, N]  (fp32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, N, K = left.shape
    assert K == N

    # output in fp32
    out = torch.empty((B, H, N, N), dtype=torch.float32, device=left.device)

    # strides in *elements* (not bytes)
    stride_lbh, stride_lh, stride_li, stride_lk = left.stride()
    stride_rbh, stride_rh, stride_rj, stride_rk = right.stride()
    stride_obh, stride_oh, stride_oi, stride_oj = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # program_id(0)
        triton.cdiv(N, BLOCK_N),          # program_id(1)
        B * H,                            # program_id(2)
    )

    trigmul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        H=H,
        N=N,
        K=K,
        stride_lbh=stride_lbh,
        stride_lh=stride_lh,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_rbh=stride_rbh,
        stride_rh=stride_rh,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_obh=stride_obh,
        stride_oh=stride_oh,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point – pure PyTorch + Triton for the N³ part
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]      (float32, CUDA)
        mask         : [B, N, N]           (float32 0/1, CUDA) – may be all‑ones
        weights      : dict of model parameters
        config       : dict with keys
            dim, hidden_dim, nomask (bool)

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------- #
    # helper PyTorch ops
    # ------------------------------------------------------------------- #
    def layer_norm(t, w, b):
        return F.layer_norm(t, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    def linear(t, w):
        # w : [out, in]  (no bias)
        return F.linear(t, w)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------- #
    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------- #
    # 2️⃣ Projections to hidden dimension
    # ------------------------------------------------------------------- #
    left  = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask_b = mask.unsqueeze(-1)
        left  = left * mask_b
        right = right * mask_b

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul – reshape & launch Triton GEMM
    # ------------------------------------------------------------------- #
    #   shape conversion: (B, N, N, H) → (B, H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # cast to fp16 for the kernel (fp32 is also fine but slower on H200)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton kernel gives fp32 output
    out_fp32 = trigmul_outgoing_tl(left_fp16, right_fp16)   # [B, H, N, N] fp32

    # back to original layout (B, N, N, H)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden → hidden)
    # ------------------------------------------------------------------- #
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
