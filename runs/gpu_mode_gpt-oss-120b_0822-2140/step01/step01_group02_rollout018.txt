```python
"""
TriMul “outgoing” forward pass (no gradients) with a fused Triton GEMM.

The expensive step

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix multiplication (B·H independent GEMMs) that we evaluate
with a single Triton kernel.  All other ops (LayerNorms, linear projections,
sigmoids, final linear) are performed with regular PyTorch kernels because
they are negligible compared with the O(N³·H) work.

The kernel computes C = A @ Bᵀ where A,B have shape (batch, N, N) and dtype
float16.  Computation is done in FP16 with FP32 accumulation and the result
is stored back as FP16, then cast to FP32 on the host side.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers (float16)
    M, N, K,                           # matrix sizes
    stride_az, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # B strides (batch, row, col) – original layout (batch, N, K)
    stride_cz, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for tensors A,B of shape (batch, M, K) and (batch, N, K)
    producing C of shape (batch, M, N) = A @ Bᵀ.
    All tensors are fp16; accumulation is fp32.
    """
    pid_batch = tl.program_id(2)       # which (B*H) matrix
    pid_m = tl.program_id(0)           # block‑row
    pid_n = tl.program_id(1)           # block‑col

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the tile in A and Bᵀ
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_az \
                   + offs_m[:, None] * stride_am \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k,n),
    # i.e. row‑k, col‑n in the original layout.
    b_ptrs = b_ptr + pid_batch * stride_bz \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (more accurate than pure FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load a tile of Bᵀ
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # FP16×FP16 → FP32 dot product
        acc += tl.dot(a, b)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write‑back (store as FP16 to keep bandwidth low)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host helper that launches the kernel
# ----------------------------------------------------------------------
def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for fp16 tensors.
    a,b : (B*H, N, N)   fp16, contiguous
    Returns c of the same shape & dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, N, K = a.shape
    assert K == N, "left/right must be square"

    c = torch.empty_like(a)

    # Strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tunable tile sizes for Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch,                     # batch*head dimension
    )
    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # empirically good for 128×128 tiles on sm90
    )
    return c


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float32) – may be omitted
        - weights      : dict of parameter tensors (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension
              * "hidden_dim" – projection dimension
              * "nomask"      – True if mask should be ignored

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free as in the reference model)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Reshape for the batched GEMM:
    #     (B, N, N, hidden) -> (B*hidden, N, N)
    # --------------------------------------------------------------
    #   Move hidden into the batch dimension → (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM C = left @ rightᵀ  (still fp16)
    # --------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(left, right)   # (B*hidden, N, N)

    # --------------------------------------------------------------
    # 7️⃣ Restore the original layout and cast back to fp32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Post‑projection LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate shape (B,N,N,hidden)

    # --------------------------------------------------------------
    # 9️⃣ Final linear back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
