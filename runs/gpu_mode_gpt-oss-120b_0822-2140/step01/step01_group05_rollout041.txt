```python
"""
TriMul “outgoing” operator (AlphaFold‑3) – fast Triton implementation.

The expensive part is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel `d` this is a matrix product (N×N) @ (N×N)^T.
We:
  1. Layer‑norm the input.
  2. Project to the hidden dimension (`left`, `right`) and cast to fp16.
  3. (Optional) apply the pairwise mask.
  4. Apply per‑channel sigmoid gates.
  5. Fuse the batched matrix‑multiply for all (B·hidden) matrices in a
     single Triton kernel (fp16 inputs, fp32 accumulator).
  6. Layer‑norm the hidden‑dim result, apply the output gate and a final
     linear projection back to `dim`.

All heavy arithmetic lives in the Triton kernel; the surrounding
operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (N×N) @ (N×N)^T  for (B*H) matrices
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,               # const __fp16 *
    B_ptr,               # const __fp16 *
    C_ptr,               # __fp16 *
    M, N, K,             # matrix sizes (int32)
    stride_ab, stride_am, stride_ak,   # strides for A: (batch, row, col)
    stride_bb, stride_bm, stride_bk,   # strides for B: (batch, row, col)
    stride_cb, stride_cm, stride_cn,   # strides for C: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  Thread/block indices
    # --------------------------------------------------------------
    pid_b = tl.program_id(0)          # batch*hidden index
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # --------------------------------------------------------------
    #  Offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the tile
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # --------------------------------------------------------------
    #  Accumulator (float32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (tiles of size BLOCK_K)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (fp16).  Masks guard out‑of‑bounds elements.
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot: fp16 x fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    #  Write back (store as fp16, casting from fp32)
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor [B, N, N, dim] (float32, CUDA)
        mask    – torch.Tensor [B, N, N]   (bool/float32) or None
        weights – dict of all linear/LayerNorm parameters (float32, CUDA)
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # called hidden_dim in the paper
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    #    Cast to fp16 to reduce bandwidth for the big matmul.
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – keep gates in fp32 for stability
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate.to(left.dtype)
    right = right * right_gate.to(right.dtype)

    # ------------------------------------------------------------------
    # 5. Batched matrix‑multiply: out = left @ rightᵀ   (per hidden)
    #
    #    Layout: (B, N, N, H) -> (B, H, N, N) -> flatten to (B*H, N, N)
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and hidden dimension
    left_flat = left.view(B * hidden, N, N)
    right_flat = right.view(B * hidden, N, N)

    out_flat = torch.empty_like(left_flat)   # fp16 output

    # Tile sizes – tuned for SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                     # batch*hidden dimension (program_id 0)
        triton.cdiv(N, BLOCK_M),       # M tiles (program_id 1)
        triton.cdiv(N, BLOCK_N),       # N tiles (program_id 2)
    )

    # Strides (batch, row, col) for the three tensors
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden) and cast to fp32 for the following layers
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
