The heavy part of the “outgoing’’ TriMul is the 4‑D einsum  

```
out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
```  

For every hidden channel `d` this is a matrix‑multiply  
`(N×N) @ (N×N)ᵀ`.  
All other operations (LayerNorm, linear projections, gates, final linear) are cheap
and are left in PyTorch.

To make the kernel fast on the H200 we

* **Fuse the batched GEMM** for all `(batch × hidden)` matrices in one Triton
  kernel.  
* Work in **bfloat16** inside the kernel (the input tensors are `float32`);
  we accumulate in `float32` to keep numerical quality and store the result
  back in `float32`.
* Choose **BLOCK_M = BLOCK_N = 64, BLOCK_K = 64** – a good trade‑off for Hopper
  (each block fits in shared memory and lets the hardware keep >95 % occupancy).
* Pass **exact strides** (`batch, row, col`) to the kernel; the kernel therefore
  works for any `B`, `N`, `hidden_dim` without extra transposes.
* Keep the mask handling and gating in PyTorch (they are element‑wise and cheap).

The overall flow is

```

right_gate = sigmoid(Linear(x))

left  = left * left_gate
right = right * right_gate

# -------------------------------------------------
```

```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched GEMM
implemented in Triton.

The kernel computes, for every (batch, hidden) pair,
    C = A @ Bᵀ          # A,B ∈ ℝ^{N×N}
using bfloat16 arithmetic inside the kernel and fp32 accumulation.
All other operations (LayerNorm, Linear, gating) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many (batch*hidden) matrices
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,        # pointers
    M, N, K,                    # matrix dimensions (all = seq_len)
    stride_Ab, stride_Ar, stride_Ac,   # A strides: batch, row, col
    stride_Bb, stride_Br, stride_Bc,   # B strides
    stride_Cb, stride_Cr, stride_Cc,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch*hidden dimension
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for A and Bᵀ
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_n[None, :] * stride_Br + offs_k[:, None] * stride_Bc

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (tiles)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (stored as bfloat16), convert to fp32 for dot
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0).to(tl.float32)   # A tile  (M×K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0).to(tl.float32)   # Bᵀ tile (K×N)

        # Matrix‑multiply the tile
        acc += tl.dot(a, b)

        # Move to next K‑tile
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Bc

    # Write result back (still fp32)
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask  : torch.Tensor [B, N, N] or None
            - weights: dict of parameter tensors (already on CUDA)
            - config : {'dim': int, 'hidden_dim': int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------
    # 1. LayerNorm over dim
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # -------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # -------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    #    (B, H, N, N) -> flatten to (B*H, N, N)
    # -------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * hidden, N, N).to(torch.bfloat16)
    right_flat = right.view(B * hidden, N, N).to(torch.bfloat16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float32)   # result in fp32

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                 # batch*hidden dimension (program_id 0)
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
    )

    # Strides (in elements)
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                 # M
        N,                 # N
        N,                 # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 8. Output LayerNorm (hidden) + gating
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # -------------------------------------------------
    # 9. Final linear back to dim
    # -------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
