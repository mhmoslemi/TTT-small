```python
"""
TriMul “outgoing” forward pass – Triton‑fused batched GEMM.

Algorithm
---------
1. LayerNorm over the input channel dimension (dim).
2. Linear projections `left`, `right` (no bias) → shape (B, N, N, H).
3. Optional masking (applied before the GEMM).
4. Gating: sigmoid(linear(x)) for left, right and output.
5. Fuse gating into `left` and `right`.
6. Permute to (B, H, N, N) and reshape to (B*H, N, N).  Cast to fp16.
7. Triton kernel computes a batched GEMM:
        out[b] = left[b] @ right[b].T           (fp16×fp16 → fp32)
   The kernel works on tiles (BLOCK_M × BLOCK_N) with a K‑loop of size BLOCK_K.
8. Reshape back to (B, N, N, H), cast to fp32.
9. LayerNorm over hidden dimension, apply output gate and final linear
   projection back to `dim`.

The kernel uses FP32 accumulation for numerical stability and stores the
result as fp16 before casting back to fp32 on the host side.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ    (A,B shape [B, M, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # base pointers
    M, N, K,                            # matrix extents (int)
    stride_az, stride_am, stride_ak,    # strides for A: batch, row, col
    stride_bz, stride_bk, stride_bn,    # strides for B: batch, row, col (original layout)
    stride_cz, stride_cm, stride_cn,    # strides for C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiply C = A @ Bᵀ
    A : (batch, M, K)   fp16
    B : (batch, N, K)   fp16   (will be accessed as Bᵀ)
    C : (batch, M, N)   fp16   (accumulated in fp32)
    """
    pid_batch = tl.program_id(2)          # batch index  (B*H)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the current tile
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_az \
                  + offs_m[:, None] * stride_am \
                  + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K). For Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                  + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load Bᵀ tile
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (cast back to fp16 to reduce bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host helper that launches the kernel
# ----------------------------------------------------------------------
def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Arguments
    ----------
    a, b : torch.Tensor
        Shape (B*H, N, N), dtype float16, contiguous.

    Returns
    -------
    c : torch.Tensor
        Same shape / dtype as inputs.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, N, K = a.shape
    assert K == N, "square matrices required for TriMul"

    c = torch.empty_like(a)

    # Strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # (batch, N, K) layout
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – empirically good for Hopper (sm90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # fits nicely in registers

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over rows
        triton.cdiv(N, BLOCK_N),   # blocks over cols
        batch,                     # one program per (B*H)
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # Hopper prefers 8 warps for 64×64 tiles
    )
    return c


# ----------------------------------------------------------------------
# Main entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a fused Triton GEMM.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N] (float32/bool) or None
        - weights      : dict of torch.Tensor, all float32
        - config       : dict with keys
              * "dim"        – input channel dimension
              * "hidden_dim" – projection dimension (H)
              * "nomask"     – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], dtype float32
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B, N, N, H)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Mask (if present)
    # --------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) → (B, N, N, 1) broadcast over H
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Rearrange to (B, H, N, N) and cast to fp16
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & head dimensions => (B*H, N, N)
    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM: out = left @ rightᵀ   (still fp16)
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 7. Restore shape (B, N, N, H) and cast back to fp32
    # --------------------------------------------------------------
    out = (
        out_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )  # (B, N, N, H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm over hidden dim, apply output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate shape (B, N, N, H)

    # --------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
