"""
Triton‑accelerated “outgoing” TriMul kernel (AlphaFold3).

The forward pass performs
    out[b,i,j,c] = Σ_k left[b,i,k,h] · right[b,j,k,h]   (h = hidden_dim)

where `left/right` are linear projections of the input, gated with a sigmoid,
optionally masked, and the contraction is a batched GEMM `A @ Bᵀ` for every
batch‑head pair.  The heavy N³ work is carried out in a single Triton kernel
that tiles the matrices (M‑blocks × N‑blocks) and accumulates in fp32 while
reading / writing in fp16 for bandwidth efficiency.  The rest of the
operations (LayerNorm, linear layers, sigmoid gates) stay in PyTorch.

The kernel works for
    B ∈ {1,2}, N ≤ 1024, hidden_dim = 128 (as required by the benchmark) and
any `dim` (e.g. 128 or 384).  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_T_kernel(
    A_ptr,                      # *fp16   (batch, M, K)
    B_ptr,                      # *fp16   (batch, N, K) – will be accessed as K×N
    C_ptr,                      # *fp32   (batch, M, N)
    M, N, K,                    # matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one block of a batched matrix product:
        C[b, m:m+BM, n:n+BN] = Σ_k A[b, m:m+BM, k] · B[b, n:n+BN, k]ᵀ
    where A,B are fp16 and the accumulation is in fp32.
    """
    pid_b = tl.program_id(2)           # batch * hidden_dim index
    pid_m = tl.program_id(0)           # block row
    pid_n = tl.program_id(1)           # block col

    # ---------- block offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------- pointer arithmetic ----------
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    # B is read as (K, N) => we stride over K with stride_Bk and over N with stride_Bn
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk
        + offs_n[None, :] * stride_Bn
    )

    # ---------- accumulation ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)                       # (M×K)·(K×N) → (M×N)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---------- write ----------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator with a Triton‑fused core.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys
            * "dim"          – input channel dimension C
            * "hidden_dim"   – projection dimension H (fixed to 128 in benchmarks)
            * "nomask" (opt) – True → ignore mask, False → apply mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → shape [B,N,N,H]
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating – sigmoid(linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton: batched GEMM per hidden head
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden_dim

    # reshape to (B*H, N, N) and make contiguous for predictable strides
    left_view  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (half the bandwidth), accumulate in fp32
    left_fp16  = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # output buffer (fp32, same shape as left_view)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # ---------- strides (in element counts, not bytes) ----------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # For the transposed access pattern Bᵀ we need:
    stride_Bk = stride_Bc                # step along K (original column)
    stride_Bn = stride_Br                # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ---------- tiling parameters ----------
    # These values have been found to give good occupancy on Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * head dimension
    )

    _batched_matmul_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout (B,N,N,H)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final linear back to C
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B,N,N,C)

    return out
