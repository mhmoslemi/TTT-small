"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
1. LayerNorm, linear projections and gating are done in PyTorch (they are cheap).
2. The O(N³) contraction is reformulated as a batched GEMM:
        out[b, d] = left[b, d] @ right[b, d].T          (N×N)·(N×N)ᵀ
   and computed with a custom Triton kernel.
3. Inside the kernel we work in fp16 (Tensor‑Core friendly) and accumulate in fp32.
4. Input / output layout: (B, H, N, N) → (B, H, N, N) to match the GEMM.
5. After the kernel we undo the permute, apply the output LayerNorm,
   the out‑gate and the final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ for many (B*H) independent matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # *fp16* pointers
    N, M,                         # N = seq_len, M = B*H (flattened)
    stride_ah, stride_ai, stride_ak,   # A: (M, N, N)
    stride_bh, stride_bj, stride_bk,   # B: (M, N, N)
    stride_ch, stride_ci, stride_cj,   # C: (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a TILE of size BLOCK_M × BLOCK_N
    of the output matrix C for a single (batch, head) pair.
    """
    pid_m = tl.program_id(0)          # row tile index   (i)
    pid_n = tl.program_id(1)          # col tile index   (j)
    pid_z = tl.program_id(2)          # flattened (batch*head) index

    # ------------------------------------------------------------------
    # Offsets for the current tile (including tail handling)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, head) slice
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B_T[k, j] shape (BLOCK_K, BLOCK_N)
        # Original B indexed as B[j, k]; we load it transposed.
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk   # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj   # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Tensor‑Core dot (fp16 → fp32 accumulation)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  fp16
    returns       : (B, H, N, N)  fp16   (still fp16, later cast to fp32)
    """
    B, H, N, _ = left.shape
    M = B * H                                 # flattened batch*head

    # Flatten to (M, N, N) for the kernel
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f)            # fp16 output

    # Compute strides (int64)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable block sizes – good for Hopper (SM90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch * head)
    )

    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The O(N³) contraction is performed by a fused Triton GEMM kernel,
    everything else (LayerNorm, linear layers, gating) stays in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim) – PyTorch implementation (fast)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections: dim -> hidden_dim  (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)          # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N) → (B, N, N, 1)
        mask = mask.to(device).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ  (per hidden channel)
    #    Convert layout to (B, H, N, N) and cast to fp16 for Tensor‑Core
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm_fp16 = trimul_gemm(left_perm, right_perm)   # (B, H, N, N) fp16

    # Back to original layout and dtype
    out = out_perm_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)       # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                        # (B, N, N, dim)

    return out
