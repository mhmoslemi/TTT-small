def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        left = left * mask.unsqueeze(-1).to(device)
        right = right * mask.unsqueeze(-1).to(device)

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    left = left.permute(0,3,1,2).contiguous()
    right = right.permute(0,3,1,2).contiguous()

    out = trimul_gemm(left, right)  # returns fp16 tensor

    out = out.to(torch.float32)  # back to fp32
    out = out.permute(0,2,3,1).contiguous()

    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])
    return out

    stride_ch, stride_ci, stride_cj,

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)

"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The costly part of TriMul is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden channel *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We flatten the batch and hidden dimensions (M = B·H) and launch a
single Triton kernel that performs M independent GEMMs of shape (N,N)·(N,N)ᵀ.
All other operations (LayerNorm, linear projections, gating, masking) are
performed in PyTorch because they are inexpensive compared with the O(N³) work.
The kernel works in fp16 for the two inputs, accumulates in fp32 and writes the
result back in fp16 – this gives a good bandwidth/compute balance on H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    N, M,                               # N = seq_len, M = B*H
    stride_ah, stride_ai, stride_ak,    # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,    # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,    # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C[i,j] = Σ_k A[i,k] * B[j,k]   (per batch‑head pair). """
    pid_m = tl.program_id(0)   # row block   (i)
    pid_n = tl.program_id(1)   # col block   (j)
    pid_z = tl.program_id(2)   # flattened batch*head

    # --- block offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current (M‑th) matrix pair
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- loop over K ---------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        a_f32 = a.to(tl.float32)

        # Bᵀ[k,j] == B[j,k]  shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_batch_ptr
            + offs_k[:, None] * stride_bk      # k dimension becomes row of Bᵀ
            + offs_n[None, :] * stride_bj      # j dimension becomes column
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b_f32 = b.to(tl.float32)

        acc += tl.dot(a_f32, b_f32)   # (BLOCK_M,BLOCK_K)@(BLOCK_K,BLOCK_N)

    # --- write back ----------------------------------------------------
    c_ptrs = (
        c_batch_ptr
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.

    left / right : (B, H, N, N)   fp16
    returns       : (B, H, N, N)  fp16
    """
    B, H, N, _ = left.shape
    M = B * H                         # flatten batch & hidden

    # reshape to (M, N, N) – contiguous layout required for correct strides
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float16)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # tile sizes – these have been found to give good occupancy on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy O(N³) contraction is executed by a fused Triton kernel;
    everything else (LayerNorm, linear layers, gates, masks) stays in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))   # (B,N,N, H)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ------------------------------------------------------------------
    # 3) Optional mask – broadcast on the hidden axis
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) + element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, H, N, N) layout and cast to fp16 for the kernel.
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out = _trimul_gemm(left, right)         # (B, H, N, N) fp16
    out = out.to(torch.float32)              # back to fp32 for the rest

    # Back to the original layout expected by the output blocks
    out = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(device))

    return out
