"""
TriMul “outgoing” forward pass – highly‑optimized version.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is implemented as a batched GEMM on the (batch, head) dimension using a custom
Triton kernel.  All other operations (LayerNorm, linear projections, gating and
optional masking) stay in PyTorch because they are cheap compared with the
O(N³) reduction.

Key ideas
---------
* After the linear projections we reshape the tensors from [B,N,N,H] to
  [B*H, N, N] (permute → (B,H,N,N) → contiguous).  Each slice (bh) is an
  ordinary matrix‑matrix multiply:  C = A @ Bᵀ .
* The kernel works on bf16 inputs (fast on Hopper) and accumulates in fp32.
  The result is written back as bf16 and converted to fp32 for the remaining
  PyTorch post‑processing.
* Masking (if present) and gating are fused with the linear projections in
  PyTorch – they are memory‑bound and do not dominate the runtime.
* The kernel is launched with a 3‑D grid: (batch*head, ceil(N/M), ceil(N/N)),
  where M/N/K are the tiling parameters (64‑64‑32 works well for the
  tested sizes up to N=1024 and hidden_dim ≤ 384).

The implementation respects the required signature:
    custom_kernel(data) -> torch.Tensor
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) for the TriMul reduction
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions (all = seq_len)
    stride_am, stride_ak, stride_an,    # A strides: (batch_head, row, col)
    stride_bm, stride_bk, stride_bn,    # B strides (note: B is accessed transposed)
    stride_cm, stride_cn,               # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------- #
    # 1) program IDs
    # ------------------------------------------------------------------- #
    pid_batch = tl.program_id(0)          # which (batch * head) matrix
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # ------------------------------------------------------------------- #
    # 2) block offsets
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds in M and N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this matrix slice
    a_base = a_ptr + pid_batch * stride_am
    b_base = b_ptr + pid_batch * stride_bm
    c_base = c_ptr + pid_batch * stride_cm

    # ------------------------------------------------------------------- #
    # 3) accumulator (fp32)
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # 4) iterate over K dimension (tiled)
    # ------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load A block ------------------------------------------------
        # A is [M, K] (row‑major).  Offsets: (m, k)
        a = tl.load(
            a_base
            + (offs_m[:, None] * stride_ak)
            + (offs_k[None, :] * stride_an),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # ---- load B block (transposed) -----------------------------------
        # B is accessed as Bᵀ, i.e. we need element B[k, n] = original
        # right[b,h,k,n].  Offsets: (k, n) using swapped strides.
        b = tl.load(
            b_base
            + (offs_k[:, None] * stride_bk)          # row  = k  (original col)
            + (offs_n[None, :] * stride_bn),         # col  = n  (original row)
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # ---- dot product for the tile -----------------------------------
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------- #
    # 5) write back (bf16) – convert once per tile
    # ------------------------------------------------------------------- #
    c = acc.to(tl.bfloat16)
    tl.store(
        c_base
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Helper: element‑wise mask multiplication (used only when a mask is present)
# --------------------------------------------------------------------------- #
def apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """src, mask have identical shape – element‑wise multiplication."""
    return src * mask  # cheap; no Triton needed for correctness


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim]
        torch.Tensor,          # mask tensor [B, N, N]   (may be all‑ones)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32, on CUDA).
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------- #
    # layer‑norm & linear helpers (bias‑free linear layers)
    # ------------------------------------------------------------------- #
    def layer_norm(x, w, b):
        return F.layer_norm(x, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    def linear(x, w):
        # w : [out_dim, in_dim]
        return F.linear(x, w)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input normalisation
    # ------------------------------------------------------------------- #
    x = layer_norm(inp, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------- #
    # 2️⃣ Projections to hidden_dim
    # ------------------------------------------------------------------- #
    left = linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------- #
    if not nomask:
        # broadcast mask to [B,N,N,1] then to hidden_dim
        mask_ext = mask.unsqueeze(-1)                     # [B,N,N,1]
        mask_ext = mask_ext.expand(-1, -1, -1, hidden_dim)  # [B,N,N,H]
        left = apply_mask(left, mask_ext)
        right = apply_mask(right, mask_ext)

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core reduction – batched GEMM in Triton
    # ------------------------------------------------------------------- #
    # Cast to bf16 for speed; Triton kernel works on bf16 and accumulates fp32.
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # reshape to (B, H, N, N) → (B*H, N, N)
    left_perm = left_bf16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape  # B*H is the first dim after permute
    bh = B  # actually B == batch*head after permute

    # output buffer
    out_perm = torch.empty_like(left_perm, dtype=torch.bfloat16)

    # strides in elements (Triton works with element strides, not bytes)
    stride_am = left_perm.stride(0)
    stride_ak = left_perm.stride(2)   # row (i)
    stride_an = left_perm.stride(3)   # col (k)

    stride_bm = right_perm.stride(0)
    # B is accessed transposed: row‑stride = original col‑stride, col‑stride = original row‑stride
    stride_bk = right_perm.stride(3)  # original col (k) becomes row in Bᵀ
    stride_bn = right_perm.stride(2)  # original row (j) becomes col in Bᵀ

    stride_cm = out_perm.stride(0)
    stride_cn = out_perm.stride(3)    # output layout same as left_perm: (BH, N, N)

    # Tiling parameters – chosen to fit H200 shared memory & work‑group size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        bh,                                # batch*head dimension
        triton.cdiv(N, BLOCK_M),           # rows
        triton.cdiv(N, BLOCK_N),           # cols
    )

    tri_mul_gemm_kernel[grid](
        a_ptr=left_perm,
        b_ptr=right_perm,
        c_ptr=out_perm,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_an=stride_an,
        stride_bm=stride_bm,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to original layout [B, N, N, H] and to fp32 for the remaining ops
    out = out_perm.to(torch.float32).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6️⃣ Output layer‑norm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection (hidden_dim → dim)
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
