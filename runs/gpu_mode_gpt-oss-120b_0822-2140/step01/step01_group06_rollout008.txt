"""
TriMul (outgoing) – fast forward‑pass.

The only O(N³·H) work is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply of two tensors of shape
(B, H, N, N) with a transpose on the second operand.
We implement this GEMM completely in Triton (no cuBLAS), tiling the
M, N and K dimensions (BLOCK_M×BLOCK_N×BLOCK_K).  All other
operations (LayerNorm, Linear, sigmoid, final projection) are kept in
PyTorch for simplicity.  The kernel works for any batch size B∈{1,2},
any sequence length N (up to 1024 in the tests) and any hidden
dimension H (the “hidden_dim” config).  All computations are performed
in float32; the kernel can be trivially retuned to fp16 if needed.

The host‑side function `custom_kernel` follows the reference
implementation exactly, only swapping the heavy O(N³·H) contraction
with the Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel for the outgoing TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_outgoing_kernel(
    # pointers
    left_ptr,          # (B, H, N, N)  left operand
    right_ptr,         # (B, H, N, N)  right operand (will be used transposed)
    out_ptr,           # (B, H, N, N)  output
    # problem size
    B, H, N,
    # strides (in elements) for left tensor
    STRIDE_L_B, STRIDE_L_H, STRIDE_L_I, STRIDE_L_K,
    # strides for right tensor
    STRIDE_R_B, STRIDE_R_H, STRIDE_R_I, STRIDE_R_K,
    # strides for output tensor
    STRIDE_O_B, STRIDE_O_H, STRIDE_O_I, STRIDE_O_J,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for one (batch, head) pair:
        out[i, j] = Σ_k left[i, k] * right[j, k]   (right is transposed)
    """
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_z = tl.program_id(2)          # batch*head (b * H + h)

    # --------------------------------------------------------------
    # decode batch & head from pid_z
    # --------------------------------------------------------------
    b = pid_z // H
    h = pid_z % H

    # --------------------------------------------------------------
    # block offsets in the M, N dimensions
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds (tail handling)
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # --------------------------------------------------------------
    # pointer arithmetic helpers
    # --------------------------------------------------------------
    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * STRIDE_L_B + h * STRIDE_L_H
    right_base = right_ptr + b * STRIDE_R_B + h * STRIDE_R_H
    out_base   = out_ptr   + b * STRIDE_O_B + h * STRIDE_O_H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension (the summation axis)
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_blocks):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
            offs_m[:, None] * STRIDE_L_I + \
            offs_k[None, :] * STRIDE_L_K
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile (transposed on‑the‑fly):
        # we need right[j, k]   ->   load as (k, j)
        right_ptrs = right_base + \
            offs_k[:, None] * STRIDE_R_K + \
            offs_n[None, :] * STRIDE_R_J
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # write back the result (apply tail mask)
    # --------------------------------------------------------------
    out_ptrs = out_base + \
        offs_m[:, None] * STRIDE_O_I + \
        offs_n[None, :] * STRIDE_O_J
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : torch.Tensor [B, N, N]       (bool/float32 or None)
        * weights_dict : model parameters (see code below)
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating – linear + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – Triton GEMM
    #    reshape to (B, H, N, N) for convenient tiling
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    out_t = torch.empty_like(left_t)   # will hold [B, H, N, N]

    # strides in elements (torch uses int64 strides)
    sl = left_t.stride()
    sr = right_t.stride()
    so = out_t.stride()

    # launch parameters (tuned for Hopper – can be adapted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # pid_m
        triton.cdiv(N, BLOCK_N),          # pid_n
        B * hidden_dim,                   # pid_z  (batch * head)
    )

    triton_trimul_outgoing_kernel[grid](
        left_t.data_ptr(),
        right_t.data_ptr(),
        out_t.data_ptr(),
        B, hidden_dim, N,
        sl[0], sl[1], sl[2], sl[3],
        sr[0], sr[1], sr[2], sr[3],
        so[0], so[1], so[2], so[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # back to original layout [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # ------------------------------------------------------------------
    out = out * out_gate   # cheap, already in PyTorch (kernel already used)

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
