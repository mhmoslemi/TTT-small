"""
TriMul (outgoing) – fast forward pass.

The expensive contraction  
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]               (1)

is a batched matrix‑multiply for every (batch, head) pair:
    L = left.permute(0,3,1,2)   # (B,H,N,N)
    R = right.permute(0,3,1,2)  # (B,H,N,N)
    out[b,h,i,j] = L[b,h,i,:] @ R[b,h,j,:].T                     (2)

We implement (2) with a tiled GEMM kernel in Triton.  All other
operations (LayerNorm, linear projections, sigmoids, final Linear) are
performed in PyTorch – they are cheap compared with the O(N³·H) work.
The kernel works in FP16 for the matrix data and accumulates in FP32,
producing a FP32 result that is later processed by the remaining
PyTorch layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton GEMM kernel – computes   C = A @ Bᵀ   for each batch‑head pair.
# A : (BH, M, K)   (float16)
# B : (BH, N, K)   (float16)  – we load it as (BH, K, N) to get the transpose
# C : (BH, M, N)   (float32)
# -------------------------------------------------------------------------
@triton.jit
def triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    stride_ah, stride_am, stride_ak,    # strides for A (batch‑head, M, K)
    stride_bh, stride_bn, stride_bk,    # strides for B (batch‑head, N, K)
    stride_ch, stride_cm, stride_cn,    # strides for C (batch‑head, M, N)
    M, N, K,                            # problem size
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)               # batch‑head index
    pid_m  = tl.program_id(1)               # tile‑row index
    pid_n  = tl.program_id(2)               # tile‑col index

    # -----------------------------------------------------------------
    # Offsets of the current tile
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # Pointers to the beginning of the tile for A and B (B is transposed)
    # -----------------------------------------------------------------
    a_tile_ptr = a_ptr + pid_bh * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # load B as (K, N) so that a_tile @ b_tile gives (M, N)
    b_tile_ptr = b_ptr + pid_bh * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (FP16).  Mask out‑of‑bounds elements.
        a = tl.load(a_tile_ptr, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0, dtype=tl.float16)
        b = tl.load(b_tile_ptr, mask=mask_n[None, :] & (offs_k[:, None] < K), other=0.0, dtype=tl.float16)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # Advance pointers to the next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # -----------------------------------------------------------------
    # Store result (still FP32)
    # -----------------------------------------------------------------
    c_tile_ptr = c_ptr + pid_bh * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Entry point used by the benchmark harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (bool/float, CUDA)   – may be None
        * weights_dict : model weights (see keys in the docstring)
        * config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm over the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, w["right_proj.weight"])  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if not cfg.get("nomask", False) and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)               # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"])) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – Triton batched GEMM
    # --------------------------------------------------------------
    #   left , right : (B, N, N, H) → (B, H, N, N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # reshape to (BH, N, N) where BH = B * H
    left_reshape  = left_t.reshape(-1, N, N).to(torch.float16)
    right_reshape = right_t.reshape(-1, N, N).to(torch.float16)

    out_reshape = torch.empty_like(left_reshape, dtype=torch.float32)  # (BH, N, N)

    # strides (in elements) for the three tensors
    stride_lbh, stride_lm, stride_lk = left_reshape.stride()
    stride_rbh, stride_rn, stride_rk = right_reshape.stride()
    stride_cbh, stride_cm, stride_cn = out_reshape.stride()

    # Tiling parameters – tuned for Hopper tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_reshape.shape[0],                     # BH
        triton.cdiv(N, BLOCK_M),                  # tile rows
        triton.cdiv(N, BLOCK_N),                  # tile cols
    )

    triton_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        stride_lbh, stride_lm, stride_lk,
        stride_rbh, stride_rn, stride_rk,
        stride_cbh, stride_cm, stride_cn,
        N, N, N,                                   # M, N, K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )  # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Multiply by output gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])       # [B, N, N, dim]

    return out
