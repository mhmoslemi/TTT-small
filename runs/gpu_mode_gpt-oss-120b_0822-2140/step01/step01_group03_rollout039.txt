"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core contraction.

Algorithm
---------
1. Layer‑norm the (B,N,N,C) input.
2. Linear projections → left/right tensors of shape (B,N,N,H) (no bias).
3. Optional element‑wise mask on the projected tensors.
4. Gated‑linear transforms (sigmoid on three linear maps) applied per‑element.
5. Core operation (batched GEMM):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of H independent N×N matrix multiplications
   (A @ Bᵀ) for every head h; we fuse it in a Triton kernel that
   uses fp16 loads, fp32 accumulation and a 128×128 tile size (tuned for
   Hopper/SM90).
6. Layer‑norm + out‑gate on the result, then a final linear projection back to C.
All non‑triton steps are performed with PyTorch (torch.nn.functional) for
readability; the expensive contraction is fully on the GPU via Triton.

The kernel works for
    B ∈ {1,2}, N ≤ 1024, C ∈ {128,384}, H = hidden_dim = 128
as required by the benchmark suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (batch, M, K)   – fp16
#   B : (batch, N, K)   – fp16, read as K×N (transpose inside kernel)
#   C : (batch, M, N)   – fp32 accumulator, stored as fp32
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # *fp16   (batch, M, K)
    B_ptr,                     # *fp16   (batch, N, K)   (will be read as K×N)
    C_ptr,                     # *fp32   (batch, M, N)
    M, N, K,                   # matrix dims
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(2)            # batch*head
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block col

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk   # K dimension (original column)
        + offs_n[None, :] * stride_Bn   # N dimension (original row)
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main K‑loop
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)          # (M, K) -> fp32
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)          # (K, N) -> fp32 (transposed view)

        acc += tl.dot(a, b)        # (M,K)·(K,N) -> (M,N)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write result
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights      : dict containing the model parameters (float32)
        - config       : dict with keys
            * "dim"          – input channel dimension C
            * "hidden_dim"   – projection dimension H (always 128)
            * "nomask" (opt) – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # weight aliases (all are 1‑D tensors for LayerNorm or 2‑D for Linear)
    # --------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (C)
    # --------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on H)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM over the K dimension
    #    Convert (B,N,N,H) → (B*H,N,N) and launch the Triton kernel.
    # --------------------------------------------------------------
    B, N, _, H = left.shape            # H == hidden_dim

    # (B, H, N, N) → contiguous → (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel for bandwidth, keep fp32 accumulator
    left_fp16 = left_t.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides expressed in element counts (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()     # (batch, M, K)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()   # (batch, N, K)

    # For the transposed view Bᵀ we need:
    stride_Bk = stride_Bc    # step along original column (K dim)
    stride_Bn = stride_Br    # step along original row   (N dim)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # Tile sizes – these work well on Hopper/SM90 for N ≤ 1024, H = 128.
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,                     # M, N, K are all N
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)        # back to original C

    return out
