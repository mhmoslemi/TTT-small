"""
TriMul “outgoing” forward pass (AlphaFold3).

The costly part is the 4‑D einsum  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply
    (B·H, N, N)  @  (B·H, N, N)^T
with H = hidden_dim.  This kernel performs that GEMM in Triton,
loading the two operands as fp16 (half precision) and accumulating in fp32,
which is the sweet‑spot on Hopper (H200).  All surrounding LayerNorm,
Linear and gating logic stays in PyTorch for simplicity.

The kernel works for any batch size (1‑2), sequence length up to 1024,
and arbitrary `dim` / `hidden_dim` (the tests use 128/384 for dim and 128 for hidden_dim).
Masking is optional.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM  (M, N, K) @ (M, N, K).T
#   left  : (M, N, K)   fp16
#   right : (M, N, K)   fp16   (we read it transposed on the last two dims)
#   out   : (M, N, N)   fp32
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel_fp16(
    left_ptr,          # *fp16   [M,N,K]
    right_ptr,         # *fp16   [M,N,K]
    out_ptr,           # *fp32   [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over rows   (i)
    pid_n = tl.program_id(1)          # tile over cols   (j)
    pid_z = tl.program_id(2)          # different matrix (batch * head)

    # ------------------------------------------------------------------
    # offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # pointers (note the right operand is accessed transposed)
    # ------------------------------------------------------------------
    a_ptrs = left_ptr + pid_z * stride_lm \
                        + offs_m[:, None] * stride_ln \
                        + offs_k[None, :] * stride_lk

    b_ptrs = right_ptr + pid_z * stride_rm \
                         + offs_k[:, None] * stride_rk \
                         + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main GEMM loop
    # ------------------------------------------------------------------
    num_k_iter = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_iter):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # tl.dot promotes the result to fp32 automatically
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # store result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool/float, optional)
        - weights      : dict of torch.Tensor (all float32)
        - config       : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor : [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to fp16 and reshape for the GEMM
    # --------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B,N,N,hidden) -> (B, hidden, N, N) -> (M, N, N)  where M = B*hidden
    left_fp16 = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_fp16 = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (fp32)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 6. Launch Triton kernel
    # --------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_fp16.stride()
    stride_rm, stride_rn, stride_rk = right_fp16.stride()
    stride_om, stride_on, stride_ok = out_fp32.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows i
        triton.cdiv(N, BLOCK_N),          # cols j
        B * hidden_dim                    # independent GEMMs
    )

    tri_mul_kernel_fp16[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B * hidden_dim, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp16 on Hopper
    )

    # --------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm + gating + final Linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
