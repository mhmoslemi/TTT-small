```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused
batched‑matrix‑multiply kernel written in Triton.

The heavy part is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel d this is a matrix multiplication
    (N × N) @ (N × N)ᵀ
which we compute with a single Triton kernel that processes
all (batch × hidden) matrices together.
All other operations (LayerNorm, Linear, Sigmoid, masking) stay in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (A @ Bᵀ)   A,B shape = (batch, M, K)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr,            # *float32
    B_ptr,            # *float32
    C_ptr,            # *float32
    M, N, K,          # matrix sizes (M=N=seq_len, K=seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bm, stride_bk,   # strides for B (batch, M, K)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ Bᵀ   for a batch of matrices.

    A: (batch, M, K)
    B: (batch, N, K)   (note the K‑dimension is the last one)
    C: (batch, M, N)
    """
    pid_b = tl.program_id(2)          # which (batch * hidden) we are working on
    pid_m = tl.program_id(1)          # tile row index
    pid_n = tl.program_id(0)          # tile col index

    # ------------------------------------------------------------------
    # Offsets inside the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the start of the tile (batch offset added later)
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # B is stored as (batch, N, K); we need it as (K, N) for the dot
    b_ptrs = B_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bm

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # a : (BLOCK_M, BLOCK_K)   b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write results
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor   [B, N, N, dim]   (float32, on CUDA)
        mask    – torch.Tensor   [B, N, N] or None (float32/bool)
        weights – dict of torch.Tensor parameters (all on CUDA)
        config  – dict with at least ``dim`` and ``hidden_dim`` entries

    Returns
    -------
    torch.Tensor   [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # called hidden_dim in the reference code
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm (over dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections  (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, N, N]  ->  [B, N, N, 1] for broadcasting
        mask_ = mask.unsqueeze(-1).type_as(left)
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Batched matrix multiplication:   out = left @ rightᵀ
    #    Work on shape (B, hidden, N, N) -> treat hidden as extra batch.
    # --------------------------------------------------------------
    #   left/right : [B, N, N, hidden] -> [B, hidden, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & hidden into one dimension
    left_flat  = left_perm.view(B * hidden, N, N)   # (batch*H, M, K)
    right_flat = right_perm.view(B * hidden, N, N)  # (batch*H, N, K)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # tiles along N (output columns)
        triton.cdiv(N, BLOCK_M),   # tiles along M (output rows)
        B * hidden,                # batch * hidden dimension
    )

    # Strides for the three‑dim tensors (batch, row, col)
    a_batch, a_m, a_k = left_flat.stride()
    b_batch, b_m, b_k = right_flat.stride()
    c_batch, c_m, c_n = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_m, a_k,
        b_batch, b_m, b_k,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout: [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (over hidden) + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
