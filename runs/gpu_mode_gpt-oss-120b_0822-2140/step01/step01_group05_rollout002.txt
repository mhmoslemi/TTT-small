"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
matrix‑multiply kernel written in Triton.

The expensive operation is
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for every hidden channel `h`.  After permuting to (B, H, N, N) we
reshape to (B*H, N, N) and compute a batched (M×K) @ (N×K)ᵀ
using a single Triton kernel.  The kernel works in fp16 for the
loads, accumulates in fp32 and writes fp32, giving a large speed‑up
on Hopper (SM‑90) while preserving the required float‑32 output.

All LayerNorm, Linear and gating operations stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply (A @ Bᵀ) for many “batch‑heads”
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_fwd(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (all == seq_len)
    stride_ab, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bb, stride_bm, stride_bk,   # B strides (batch, row, col)
    stride_cb, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  for a batch of matrices.
    A, B, C have shapes (batch, M, K), (batch, N, K), (batch, M, N)
    """
    pid_b = tl.program_id(0)   # batch*hidden index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # ------------------------------------------------------------------
    # Offsets for the tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the tile
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_n[None, :] * stride_bm + offs_k[:, None] * stride_bk

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles – cast to fp16 for the dot product
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # tl.dot expects (M,K) and (K,N) -> (M,N); result is fp32 because acc is fp32
        acc += tl.dot(a, b)

        # Move pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32, CUDA)
            - mask: torch.Tensor[B, N, N] (bool/float32) or None
            - weights: dict of torch.Tensor containing all linear / norm parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # = hidden_dim in the paper
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched mat‑mul: out = left @ rightᵀ   (per hidden channel)
    # ------------------------------------------------------------------
    #   left, right : [B, N, N, hidden]  ->  [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    #   view as (B*hidden, N, N) – each hidden slice becomes a batch
    left_flat = left_perm.view(B * hidden, N, N)
    right_flat = right_perm.view(B * hidden, N, N)

    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for Hopper)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),        # row tiles
        triton.cdiv(N, BLOCK_N),        # col tiles
    )

    # Strides of the three‑dimensional tensors (batch, row, col)
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bm, stride_bk = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    _batched_matmul_fwd[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                 # M
        N,                 # N
        N,                 # K
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bm, stride_bk,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden dimension) + final gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
