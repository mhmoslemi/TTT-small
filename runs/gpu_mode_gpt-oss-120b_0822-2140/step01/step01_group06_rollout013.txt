"""
TriMul (outgoing) – fast forward pass
====================================

* The heavy O(N³·H) contraction  
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  
  is expressed as a batched matrix‑multiply  (A @ Bᵀ) for each
  (batch, head) pair.  We launch a Triton GEMM kernel that tiles the
  multiplication in fp16 and accumulates in fp32.

* All LayerNorms, Linear projections, and sigmoids are done with
  torch.nn.functional for simplicity.  The only required Triton
  components are:
    – a GEMM kernel that works on (B·H, N, N) tensors,
    – a tiny element‑wise kernel that fuses the final “out_gate”
      multiplication (optional – kept for completeness).

The implementation follows the reference PyTorch module exactly, but
the O(N³·H) part is now fully on‑device and much faster on an NVIDIA
H200 (SM‑90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel for the core TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def triton_gemm_outgoing_kernel(
    a_ptr,          # left  : (B*H, M, K)   fp16
    b_ptr,          # right : (B*H, N, K)   fp16  (will be used as Bᵀ)
    c_ptr,          # out   : (B*H, M, N)   fp32
    M, N, K,        # matrix sizes (all = seq_len)
    stride_am, stride_ak,           # a stride for M and K
    stride_bk, stride_bn,           # b stride for K and N
    stride_cm, stride_cn,           # c stride for M and N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # block row‑col index
    bid = tl.program_id(1)          # batch‑head index

    # --------------------------------------------------------------
    # block coordinates
    # --------------------------------------------------------------
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // grid_n
    pid_n = pid % grid_n

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch‑head
    a_ptrs = a_ptr + bid * (M * K) + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + bid * (N * K) + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # tiled K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] + k * BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] + k * BLOCK_K < K) & mask_n[None, :], other=0.0)

        # cast to fp16 (already fp16) then to fp32 for dot
        a = a.to(tl.float16)
        b = b.to(tl.float16)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result (fp32)
    c_ptrs = c_ptr + bid * (M * N) + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Tiny element‑wise multiplication kernel (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total, stride_a: tl.constexpr, stride_b: tl.constexpr, stride_out: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (bool/float32 or None)
        * weights_dict : learned parameters (tensors on CUDA)
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, w["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gates (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction via Triton GEMM
    # --------------------------------------------------------------
    # shape -> (B, hidden_dim, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch × head
    BH = B * hidden_dim
    left_flat  = left.view(BH, N, N).to(torch.float16)
    right_flat = right.view(BH, N, N).to(torch.float16)

    # output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # strides in element units
    stride_am = left_flat.stride(1)        # stride for M dimension
    stride_ak = left_flat.stride(2)        # stride for K dimension
    stride_bk = right_flat.stride(2)       # stride for K dimension in B
    stride_bn = right_flat.stride(1)       # stride for N dimension in B
    stride_cm = out_flat.stride(1)         # stride for M in C
    stride_cn = out_flat.stride(2)         # stride for N in C

    # tiling parameters – tuned for Hopper's 128‑KB shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n, BH)                       # (row‑col blocks, batch‑head)

    triton_gemm_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # fp32

    # --------------------------------------------------------------
    # 6️⃣ Post‑norm on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate (fused kernel)
    # --------------------------------------------------------------
    out_gate_flat = out_gate.view(-1)
    out_flat = out.view(-1)
    total = out_flat.numel()
    BLOCK_E = 1024
    grid_e = (triton.cdiv(total, BLOCK_E),)

    elemwise_mul_kernel[grid_e](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        stride_a=1,
        stride_b=1,
        stride_out=1,
        BLOCK=BLOCK_E,
    )
    # out now contains out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B, N, N, dim]

    return out
