"""
TriMul “outgoing” (AlphaFold‑3) forward pass.

The expensive part is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every hidden dimension d (called *head* below).  This can be written as a
batched matrix‑multiply per head:

    C[b,h] = A[b,h] @ B[b,h].T               # A,B : (N,N)

The kernel below computes C for all (batch,head) pairs in one launch.  The
input tensors are cast to fp16 for bandwidth, accumulation is performed in
fp32 (to keep numeric quality) and the result is written back as fp32.
All other operations (LayerNorm, Linear, gating, final projection) are kept
in PyTorch because they are comparatively cheap.

The kernel is tuned for Hopper (sm_90) – we use 128×128 tiles with 8 warps,
which gives good occupancy for N up to 1024 while keeping shared‑memory
footprint modest.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B,H,N,N) @ (B,H,N,N)ᵀ  →  (B,H,N,N)
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    For each (batch, head) pair compute
        C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]
    i.e. A @ Bᵀ  (both A and B have shape (N,N)).
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb % H

    # ----------- offsets for the current tile ----------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ------- load A (TILE_M × TILE_K) -------
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ------- load Bᵀ (TILE_K × TILE_N) -------
        # B is stored as (i,j) but we need (j,k).  Swapping the stride indices
        # gives the transposed view without an actual copy.
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first dim
                  + offs_n[None, :] * stride_b_i)  # j as second dim

        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot‑product over K (FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # ----------- store result ----------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 or bfloat16)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors containing the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` (int)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # convenience for weight lookup
    # ------------------------------------------------------------------
    def w(name): return weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)   →   [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))          # (B,N,N,H)
    right = F.linear(x, w("right_proj.weight"))         # (B,N,N,H)

    # optional mask (mask is broadcasted on the hidden dim)
    if not config.get("nomask", True):
        mask_ = mask.to(dtype).unsqueeze(-1)            # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid)   →   same shape
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy contraction: batched mat‑mul per head using Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B,H,N,N]

    # strides in *elements* (not bytes) – required by the kernel
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will read transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # tile sizes – tuned for Hopper
    TILE_M = 128
    TILE_N = 128
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch*head
    )

    trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,                     # 8 warps → good occupancy on H200
    )

    # back to original layout: (B,N,N,H)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm + gating + linear back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate          # broadcast over hidden_dim
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
