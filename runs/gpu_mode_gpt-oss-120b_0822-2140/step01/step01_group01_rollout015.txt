"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The dominant cost is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* this is a matrix multiplication
    L = left[b,:, :, d]      (N×N)
    R = right[b,:, :, d]     (N×N)
    out[b,:, :, d] = L @ Rᵀ (N×N)

We therefore launch a batched GEMM kernel that computes many independent
(N,N)·(N,N)ᵀ products in parallel.  The kernel works on FP16 inputs
(to cut memory traffic) while accumulating in FP32 for numeric stability.
All lightweight operations (LayerNorm, Linear, Sigmoid, gating and the final
projection) stay in PyTorch – they are negligible compared with the O(N³)
work.

The kernel shape is (B·H, N, N) where H = hidden_dim.  Tiling parameters
are chosen for Hopper (SM‑90): 128×128×64 blocks give good occupancy and
use the 4 × 128‑byte shared‑memory loads that the H200 prefers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel_fp16(
    a_ptr, b_ptr, c_ptr,          # [M, N, N] tensors (FP16)
    N, M,                         # N = seq_len, M = B*hidden_dim
    stride_ah, stride_ai, stride_ak,   # A strides (M,N,N)
    stride_bh, stride_bj, stride_bk,   # B strides (M,N,N)
    stride_ch, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for each z in [0, M):
        C[z] = A[z] @ B[z].T               # (N,N) @ (N,N).T  -> (N,N)
    A and B are stored as FP16, the accumulator is FP32.
    """
    pid_m = tl.program_id(0)    # row block of the output (i)
    pid_n = tl.program_id(1)    # col block of the output (j)
    pid_z = tl.program_id(2)    # flattened batch*head dimension

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the current (z) matrix pair
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i, k]  (BLOCK_M × BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # cast to FP32

        # Load B_T[k, j]  (BLOCK_K × BLOCK_N)
        # Original B is indexed B[j, k]; we load it transposed.
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk   # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj   # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)   # cast to FP32

        # Accumulate
        acc += tl.dot(a, b)

    # Write back C[i, j] (still FP32)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  (float16)
    returns      : (B, H, N, N)  (float32)
    """
    B, H, N, _ = left.shape
    M = B * H                                     # flatten batch*head

    # reshape to (M, N, N) – keep the strides, no copy
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tiling parameters (tuned for Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # pid_m
        triton.cdiv(N, BLOCK_N),      # pid_n
        M,                            # pid_z = batch * head
    )

    trimul_gemm_kernel_fp16[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    All cheap ops are done in PyTorch; the O(N³) contraction is performed
    by a fused batched GEMM Triton kernel (FP16 inputs, FP32 accumulation).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections -> (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)        # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        m = mask.to(device).unsqueeze(-1)          # (B, N, N, 1)
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) + element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Triton batched GEMM: out = left @ rightᵀ    (per hidden channel)
    #    Convert to (B, H, N, N) layout required by the kernel.
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) and cast to FP16 for bandwidth.
    left_perm  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # Batched GEMM, result is FP32
    out_perm = trimul_gemm_fp16(left_perm, right_perm)   # (B, H, N, N)

    # Back to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
