"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]   (N×N)
    R = right[b,:, :, d]  (N×N)
    out[b,:, :, d] = L @ Rᵀ            (N×N)

We therefore launch a Triton kernel that computes many independent
GEMMs (one per batch‑head pair).  All surrounding work (LayerNorm,
linear projections, gating, masking) is cheap and stays in PyTorch.
The kernel works on fp16 inputs and accumulates in fp32, returning an
fp32 tensor that is later fed through the output‑norm and linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers (fp16)
    N, M,                                    # N = seq_len, M = B*hidden_dim
    stride_ah, stride_ai, stride_ak,         # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,         # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,         # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A : (M, N, N)   fp16   – left  tensor   (B, H, N, N)
    B : (M, N, N)   fp16   – right tensor   (B, H, N, N)
    C : (M, N, N)   fp32   – output          (B, H, N, N)

    computes for each (m)  C[m] = A[m] @ B[m]ᵀ
    """

    pid_m = tl.program_id(0)            # block row   (i)
    pid_n = tl.program_id(1)            # block col   (j)
    pid_z = tl.program_id(2)            # flattened (batch*head)

    # ---------------------  block offsets ------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ----------------------  base pointers -----------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------  K loop  --------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  → (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # B_T[k, j]  → (BLOCK_K, BLOCK_N)
        # B is stored as (M, N, N) with axes (head, row, col);
        # we want the transpose on the last two axes, so we load with
        # k as the first index and j as the second.
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk           # k‑dim (row of Bᵀ)
            + offs_n[None, :] * stride_bj           # j‑dim (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot product, promote to fp32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # -----------------------  store result  ---------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  fp16
    returns      : (B, H, N, N)  fp32 where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                                   # flattened batch*head

    # flatten to (M, N, N)
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32)

    # strides (int64) of the *contiguous* tensors
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # --------------------  choose block sizes  -------------------------
    # Heuristics: larger blocks when N is big, keep K moderate.
    if N >= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),    # pid_m
        triton.cdiv(N, BLOCK_N),    # pid_n
        M,                          # pid_z  (batch*head)
    )
    trimul_gemm_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : (B, N, N, dim)   torch.float32
        - mask         : (B, N, N)        torch.float32 (or None)
        - weights      : dict of torch tensors
        - config       : dict with keys
            * dim          : input feature dimension
            * hidden_dim   : projection dimension
            * nomask       : bool, True when mask should be ignored

    Returns
    -------
    torch.Tensor of shape (B, N, N, dim)   (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_w,
                     bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)          # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)          # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, H, N, N) layout expected by the kernel.
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # fp16 for the heavy GEMM
    left_h  = left_perm.to(torch.float16)
    right_h = right_perm.to(torch.float16)

    out_perm_fp32 = trimul_gemm(left_h, right_h)         # (B, H, N, N) fp32

    # back to the original layout (B, N, N, H)
    out = out_perm_fp32.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
