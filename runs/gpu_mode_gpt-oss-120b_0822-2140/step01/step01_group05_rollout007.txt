"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
matrix‑multiply kernel written in Triton.

The heavy part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched GEMM (A @ Bᵀ) over the hidden dimension.
We treat every (batch, hidden) pair as an independent GEMM of size
(N × N) @ (N × N)ᵀ and compute all of them in one Triton kernel.
All other ops (LayerNorm, linear projections, gates) stay in
PyTorch – they are cheap compared with the N³ work.
The kernel works for fp16 inputs (to reduce memory traffic) while
accumulating in fp32 and returning fp32 results.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many (batch*hidden) matrices
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers
    M, N, K,                        # matrix sizes (M=N=seq_len, K=seq_len)
    stride_Ab, stride_Am, stride_Ak,   # strides for A  (batch, row, col)
    stride_Bb, stride_Bm, stride_Bk,   # strides for B  (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    A : (batch, M, K)   – left side
    B : (batch, N, K)   – right side (we use its transpose)
    C : (batch, M, N)   – output
    All tensors are contiguous in the last two dimensions.
    """
    pid_b = tl.program_id(0)   # batch*hidden index
    pid_m = tl.program_id(1)   # tile row   (M dim)
    pid_n = tl.program_id(2)   # tile col   (N dim)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_n[None, :] * stride_Bm          # note: we index B as (row=N, col=K)
        + offs_k[:, None] * stride_Bk
    )

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # a : (BLOCK_M, BLOCK_K) , b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input  – torch.Tensor [B, N, N, dim]   (float32, on CUDA)
        mask   – torch.Tensor [B, N, N] (bool/float) or None
        weights – dict of model parameters (all on CUDA)
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if any) – broadcasted on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ   (per hidden channel)
    #    ----------------------------------------------------
    #    left/right : [B, N, N, hidden]  ->  [B*hidden, N, N]
    #    we keep data in fp16 for bandwidth, accumulate in fp32.
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # (B, hidden, N, N) layout is more cache‑friendly for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left.view(B * hidden, N, N)
    right_flat = right.view(B * hidden, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # Triton launch parameters – tuned for Hopper (sm_90)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                          # batch*hidden
        triton.cdiv(N, BLOCK_M),             # rows
        triton.cdiv(N, BLOCK_N),             # cols
    )

    # strides for the three‑dimensional tensors (batch, row, col)
    a_batch, a_row, a_col = left_flat.stride()
    b_batch, b_row, b_col = right_flat.stride()
    c_batch, c_row, c_col = out_flat.stride()

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                 # M
        N,                 # N
        N,                 # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: [B, N, N, hidden]
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden) and final gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
