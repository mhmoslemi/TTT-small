**Implementation – “outgoing” TriMul with an FP16‑fused GEMM**

```python
"""
TriMul “outgoing” implementation (forward only).

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched GEMM (A @ Bᵀ) over the sequence dimension.
We fuse this GEMM in a Triton kernel that:
  * works on fp16 inputs (left/right) – half the memory traffic,
  * accumulates in fp32 (tl.dot does that automatically),
  * uses 64×64 tiles and 8 warps (good for Hopper/H200 fp16 GEMM).

All other operations (LayerNorm, Linear, sigmoid‑gates) are performed
with PyTorch’s highly‑optimised kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  out = left @ rightᵀ
#   left  : (M, N, K)   fp16
#   right : (M, N, K)   fp16
#   out   : (M, N, N)   fp32
#   M = B * hidden_dim,   K = N
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # *fp16  [M, N, K]
    right_ptr,         # *fp16  [M, N, K]
    out_ptr,           # *fp32  [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row  (i)
    pid_n = tl.program_id(1)          # tile col  (j)
    pid_z = tl.program_id(2)          # which matrix (batch * head)

    # --------------------------------------------------------------
    # Offsets within the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers (no transposition needed for left, we need rightᵀ)
    # --------------------------------------------------------------
    # left[z, i, k]
    a_ptrs = left_ptr + pid_z * stride_lm \
                     + offs_m[:, None] * stride_ln \
                     + offs_k[None, :] * stride_lk

    # rightᵀ : we read right as (z, k, j)
    b_ptrs = right_ptr + pid_z * stride_rm \
                      + offs_k[:, None] * stride_rk \
                      + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main GEMM loop over K
    # --------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)          # fp16 -> fp32 promotion inside dot
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # (BM, BK) @ (BK, BN)  -> fp32

        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # --------------------------------------------------------------
    # Store result (fp32)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N]  (float32/bool) or None
            - weights      : dict of pre‑loaded Parameter tensors
            - config       : dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Mask (if present) – applied **before** gating
    # ------------------------------------------------------------------- #
    if mask is not None:
        # mask shape: [B,N,N] → [B,N,N,1] for broadcast
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) – still in fp32
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Cast to fp16 for the GEMM, reshape to (M, N, N)
    # ------------------------------------------------------------------- #
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # From (B, N, N, H) → (B, H, N, N) → (M, N, N)
    left_rs  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_rs = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (fp32)
    out_rs = torch.empty_like(left_rs, dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Strides (in elements) for the three tensors
    # --------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_rs.stride()
    stride_rm, stride_rn, stride_rk = right_rs.stride()
    stride_om, stride_on, stride_ok = out_rs.stride()

    # --------------------------------------------------------------
    # Launch configuration – 64×64 tiles, 64‑k block, 8 warps
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows (i)
        triton.cdiv(N, BLOCK_N),          # cols (j)
        B * hidden_dim                    # independent matrices
    )

    tri_mul_outgoing_kernel[grid](
        left_rs,
        right_rs,
        out_rs,
        B * hidden_dim, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # best for fp16 GEMM on Hopper/H200
    )

    # --------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_rs.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm + gating + final Linear (dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B,N,N,dim]

    return out
```
