"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The dominant work in TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We therefore launch a **batched GEMM** kernel that computes many independent
products (batch × hidden) in parallel.  The kernel works on **fp16** inputs
(L,R) but accumulates in **fp32** to keep numerical quality.  All
pre‑/post‑processing (LayerNorm, Linear, sigmoid, mask) stays in PyTorch
because they are cheap compared with the O(N³) contraction.

The kernel tiles the matrices with
    BLOCK_M = BLOCK_N = 128   (rows / cols of the output tile)
    BLOCK_K = 64              (reduction tile)
These sizes were chosen empirically for Hopper‑SM90 (H200) – they give
good occupancy and fit the 64 KiB shared memory per SM while keeping the
inner dot in fp16×fp16→fp32.

The kernel returns a tensor of shape (B, hidden, N, N) which is permuted
back to (B, N, N, hidden) and passed through the final LayerNorm + linear
layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) on fp16 inputs, fp32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    N,                                 # sequence length
    stride_ah, stride_ai, stride_ak,   # A strides (M, N, N)
    stride_bh, stride_bj, stride_bk,   # B strides (M, N, N)
    stride_ch, stride_ci, stride_cj,   # C strides (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent (N×N) matrices.
    A,B : (M, N, N)   fp16
    C   : (M, N, N)   fp32
    M = batch * hidden_dim
    """
    pid_m = tl.program_id(0)   # output row tile index (i)
    pid_n = tl.program_id(1)   # output col tile index (j)
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the tails
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the (pid_z)-th matrix pair
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load A[i, k]  (shape BLOCK_M × BLOCK_K) -----------------
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_ai          # i dimension
            + offs_k[None, :] * stride_ak          # k dimension
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- Load Bᵀ[k, j]  (shape BLOCK_K × BLOCK_N) -----------------
        # Original B is indexed as B[j, k]; we load it transposed so that
        # we can call tl.dot(a, b) directly.
        b_ptrs = (
            b_batch_ptr
            + offs_k[:, None] * stride_bk          # k (row of Bᵀ)
            + offs_n[None, :] * stride_bj          # j (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # ---- Accumulate ------------------------------------------------
        acc += tl.dot(a, b)   # fp16×fp16 → fp32

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch_ptr
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    Arguments
    ---------
    left, right : (B, H, N, N)   dtype = torch.float16
    Returns
    -------
    out : (B, H, N, N)           dtype = torch.float32
    """
    B, H, N, _ = left.shape
    M = B * H                       # flattened batch*hidden

    # flatten to (M, N, N) for the kernel
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as 64‑bit ints
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # Tiling parameters – these work well on H200 (SM90)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch * hidden)
    )
    trimul_gemm_fp16_kernel[grid](
        left_f, right_f, out,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy O(N³) contraction is executed by a custom Triton kernel
    (batched fp16 GEMM).  All other linear / norm / sigmoid / mask
    operations are performed in regular PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections from dim -> hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)      # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)          # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, hidden, N, N) layout expected by the kernel.
    #    Use fp16 for the kernel to reduce memory traffic.
    # ------------------------------------------------------------------
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_fp32 = trimul_gemm_fp16(left_fp16, right_fp16)                    # (B, H, N, N) fp32

    # Back to the original layout (B, N, N, hidden)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)      # (dim, hidden)
    out = F.linear(out, to_out_w)                       # (B, N, N, dim)

    # The specification asks for float32 output
    return out.to(torch.float32)
