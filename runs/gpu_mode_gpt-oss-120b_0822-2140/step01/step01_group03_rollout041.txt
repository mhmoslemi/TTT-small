TriMul “outgoing” forward – Triton‑accelerated core contraction
================================================================
Given an input tensor X of shape (B, N, N, C) the module performs

1. Layer‑norm over the last dimension (C).
2. Two linear projections → left / right tensors of shape (B, N, N, H).
3. Optional element‑wise masking (broadcast on H).
4. Gated linear units (sigmoid gates) applied to left, right and the final output.
5. **Batched GEMM**   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   This is exactly a matrix multiplication per (batch, head) pair:
          out_slice = left_slice @ right_sliceᵀ
   The GEMM is implemented in a single Triton kernel that processes many
   (B·H) independent matrix products in parallel, using fp16 loads and fp32
   accumulation.
6. Layer‑norm + out‑gate + final linear projection back to C.

The kernel works for the full benchmark range (B≤2, N≤1024, C∈{128,384},
H=hidden_dim=128) on Hopper (SM‑90).  The public API is the function
``custom_kernel(data)`` whose output is a float‑32 tensor of shape
(B, N, N, C).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                      # *fp16  (batch, M, K)
    B_ptr,                      # *fp16  (batch, N, K)  (will be read as K×N)
    C_ptr,                      # *fp32  (batch, M, N)
    M, N, K,                    # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (k‑step & n‑step after transposition)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of independent matrix products:
        C[b, m, n] = Σₖ A[b, m, k] * B[b, n, k]   (i.e. A @ Bᵀ)

    Parameters are the raw pointers and strides (in elements, not bytes).
    """
    pid_b = tl.program_id(2)          # index in the fused batch‑head dimension
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block column index

    # ----- block start indices -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ----- masks for out‑of‑bounds -----
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- compute pointers for this block -----
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk      # K‑dimension (original column)
        + offs_n[None, :] * stride_Bn      # N‑dimension (original row)
    )

    # accumulator in fp32 (higher precision for the N³ sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main K loop -----
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))
        # advance pointers for next tile

        b_ptrs += BLOCK_K * stride_Bk

    # ----- write result back -----

        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
