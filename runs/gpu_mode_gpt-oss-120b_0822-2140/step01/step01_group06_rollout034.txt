"""
TriMul (outgoing) – fast forward pass.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply over the sequence dimension.
We reshape the tensors to (B·H, N, N) and launch a single Triton
kernel that performs a tiled GEMM A @ Bᵀ for every (batch, head) pair.
All other operations (LayerNorms, Linear projections, sigmoids,
and the final element‑wise multiplication with the output‑gate)
are done in PyTorch; the element‑wise product is also fused with a tiny
Triton kernel to satisfy the “must use a kernel” constraint.

The kernel works in fp16 for arithmetic but accumulates in fp32,
producing a float‑32 output that matches the reference PyTorch
implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Batched GEMM kernel  (A @ Bᵀ)  –  computes the O(N³·H) part
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,               # (BH, M, K)   – left   (fp16)
    b_ptr,               # (BH, K, N)   – rightᵀ (fp16)
    c_ptr,               # (BH, M, N)   – output (fp32)
    M, N, K,
    stride_am, stride_ak,        # strides for A
    stride_bk, stride_bn,        # strides for B (already transposed)
    stride_cm, stride_cn,        # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_head = tl.program_id(0)          # 0 … BH‑1
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block col

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds (needed when N is not a multiple of block size)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch‑head
    a_batch_ptr = a_ptr + batch_head * stride_am * M          # stride over batch_head = M*K
    b_batch_ptr = b_ptr + batch_head * stride_bk * K          # stride over batch_head = K*N

    # initialise accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k‑slice start
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < K

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                               + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)          # promote to fp32 for accumulation

        # B: (BLOCK_K, BLOCK_N)  (already transposed)
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # outer product + accumulate
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + (batch_head * stride_cm * M
                     + offs_m[:, None] * stride_cm
                     + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 2. Simple element‑wise multiplication kernel (used for out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elements,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing’’ TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        * input_tensor : [B, N, N, dim]   torch.float32 on CUDA
        * mask_tensor  : [B, N, N]       torch.bool / float32  (may be None)
        * weight_dict  : all Linear / LayerNorm parameters
        * config_dict  : {"dim":int, "hidden_dim":int}
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = inp.shape
    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )                                                   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction via Triton batched GEMM
    #    reshape to (B*H, N, N) → treat each head as an independent batch.
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (BH, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim

    left = left.view(BH, N, N)              # (BH, M, K)
    right = right.view(BH, N, N)            # (BH, K, N)  (we need Bᵀ)
    right_T = right.permute(0, 2, 1).contiguous()   # (BH, K, N)

    # cast to fp16 for speed, keep result in fp32
    left_fp16 = left.to(torch.float16)
    right_fp16 = right_T.to(torch.float16)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)   # (BH, M, N)

    # block sizes – tuned for Hopper (SM90) and the provided N ranges
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (in elements) for the contiguous tensors
    stride_am = left_fp16.stride(1)          # K dimension stride = N
    stride_ak = left_fp16.stride(2)          # inner stride = 1
    stride_bk = right_fp16.stride(1)         # K stride = N
    stride_bn = right_fp16.stride(2)         # N stride = 1
    stride_cm = out_fp32.stride(1)           # M stride = N
    stride_cn = out_fp32.stride(2)           # N stride = 1

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N, N, N,                               # M, N, K (all equal to sequence length)
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )                                                   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with out_gate (fused Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total_elems = out_flat.numel()

    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)
    elemwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total_elems,
        BLOCK=BLOCK,
    )
    # out now holds out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])          # [B, N, N, dim]

    return out
