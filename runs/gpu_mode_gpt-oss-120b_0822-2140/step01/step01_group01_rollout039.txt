"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension d this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We launch a single Triton kernel that performs
    (B·H) independent GEMMs  (N×N)·(N×N)ᵀ
using fp16 arithmetic (accumulate in fp32).  All surrounding ops
(LayerNorm, linear projections, gating, masking, output norm & linear)
are cheap and stay in PyTorch.

The kernel is a tuned version of the classic Triton matmul:
* BLOCK_M = BLOCK_N = 128, BLOCK_K = 32 – good for Hopper SM‑90.
* GROUP_SIZE_M improves L2 cache reuse.
* Tail handling via masks.
* Loads/stores are coalesced thanks to the (B,H,N,N) layout.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr,                         # const *fp16
    b_ptr,                         # const *fp16
    c_ptr,                         # *fp32
    N,                             # seq_len (int32)
    stride_ah, stride_ai, stride_ak,   # strides of A (M,N,N)
    stride_bh, stride_bj, stride_bk,   # strides of B (M,N,N)
    stride_ch, stride_ci, stride_cj,   # strides of C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    """Batched GEMM where each batch element computes
       C = A @ Bᵀ   (shapes (N,N) · (N,N)ᵀ)"""

    pid_z = tl.program_id(2)          # flattened (batch * hidden)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # 2‑D block indices within the current program (pid_z)
    block_id = tl.program_id(0) + tl.program_id(1) * num_pid_m
    pid_m = (block_id // GROUP_SIZE_M) % num_pid_m
    pid_n = block_id % num_pid_n

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # pointers to the start of the current batch‑hidden slice
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]  (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ[k,j]  (BLOCK_K, BLOCK_N) – note the transposition
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.

    Args:
        left  : (B, H, N, N)  fp16 tensor
        right : (B, H, N, N)  fp16 tensor

    Returns:
        out   : (B, H, N, N)  fp32 tensor  (each slice = left @ rightᵀ)
    """
    B, H, N, _ = left.shape
    M = B * H                                 # flattened batch*head

    # flatten to (M, N, N) – contiguity already guaranteed by callers
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # launch parameters (tuned for Hopper)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    GROUP_SIZE_M = 8

    grid = (
        triton.cdiv(N, BLOCK_M),          # pid_m (grouped)
        triton.cdiv(N, BLOCK_N),          # pid_n
        M,                                # pid_z (batch*head)
    )
    trimul_gemm_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=4,                     # lets Triton pick a reasonable count
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    data = (input_tensor, mask, weights, config)

    - input_tensor : (B, N, N, dim)  float32
    - mask         : (B, N, N)      (ignored if config[\"nomask\"] is True)
    - weights      : dict of torch.Tensor matching the PyTorch nn.Module
    - config       : dict with keys
        * dim, hidden_dim, nomask (bool)
    Returns:
        Tensor of shape (B, N, N, dim)  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim), no bias
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)        # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Mask (if present)
    # --------------------------------------------------------------
    if not nomask:
        m = mask.to(device).unsqueeze(-1)           # (B, N, N, 1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    # --------------------------------------------------------------
    # move hidden dim to 2nd position and cast to fp16 for speed
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm = trimul_gemm(left_perm, right_perm)      # (B, H, N, N) fp32

    # back to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 6. Output LayerNorm + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
