"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with a fused
batched GEMM kernel written in Triton.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply of shape (B·H, N, N) with the
transpose of the second operand.  We reshape the tensors to
(M, N, K) where   M = B·hidden_dim,   N = seq_len,   K = seq_len,
and compute the GEMM with a single Triton kernel.  All other
operations (LayerNorm, Linear projections, gating) stay in
PyTorch for simplicity.

The kernel works in FP16 for the loads (to reduce bandwidth) and
accumulates in FP32 (the default for `tl.dot`).  The final result
is stored as FP32, matching the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ rightᵀ   where left/right have shape (M,N,K)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16   [M,N,K]
    right_ptr,         # *fp16   [M,N,K]
    out_ptr,           # *fp32   [M,N,N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides (in elements)
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM `out = left @ rightᵀ`."""
    pid_m = tl.program_id(0)          # tile over rows   (i)
    pid_n = tl.program_id(1)          # tile over cols   (j)
    pid_z = tl.program_id(2)          # independent matrix (batch * head)

    # ------------------------------------------------------------------- #
    # Offsets within the current tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers (row‑major layout)
    # ------------------------------------------------------------------- #
    # left : (M, N, K)   index (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                       + offs_m[:, None] * stride_ln \
                       + offs_k[None, :] * stride_lk

    # right is accessed transposed on the last two axes:
    # we need Bᵀ where B(z, j, k) -> (z, k, j)
    b_ptrs = right_ptr + pid_z * stride_rm \
                        + offs_k[:, None] * stride_rk \
                        + offs_n[None, :] * stride_rn

    # ------------------------------------------------------------------- #
    # Accumulator
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main K‑loop
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
            dtype=tl.float16,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance K‑dimension
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Write back
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        - mask         : torch.Tensor   [B, N, N]       (bool/float) or None
        - weights      : dict of torch.Tensor   (all parameters, no bias)
        - config       : dict  {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    # left_proj.weight : (hidden_dim, dim)
    # right_proj.weight: (hidden_dim, dim)
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Fuse the N³ contraction with Triton
    #    reshape to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------- #
    # (B,N,N,H) -> (B,H,N,N) -> (M,N,N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Cast to fp16 for the kernel (half the memory traffic)
    left_reshape = left_reshape.to(torch.float16)
    right_reshape = right_reshape.to(torch.float16)

    out_reshape = torch.empty(
        (B * hidden_dim, N, N), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------- #
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # tuning parameters – they work well on Hopper for the supported sizes
    BLOCK_M = 64   # rows   of a tile
    BLOCK_N = 64   # cols   of a tile
    BLOCK_K = 64   # reduction tile

    grid = (
        triton.cdiv(N, BLOCK_M),          # tiles over i
        triton.cdiv(N, BLOCK_N),          # tiles over j
        B * hidden_dim,                   # independent GEMMs
    )

    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden_dim,
        N,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # Hopper can keep 8 warps busy for fp16 loads
    )

    # ------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------- #
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 8. Output LayerNorm + gating + final linear (dim)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
