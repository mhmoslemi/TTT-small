"""
TriMul “outgoing” kernel (AlphaFold3).

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply of shape (B·D, N, N) @ (B·D, N, N)ᵀ.
We fuse this reduction in a custom Triton GEMM that works in bfloat16
(≈ FP16‑speed on Hopper) while accumulating in FP32 for accuracy.
All other operations (LayerNorm, Linear, gating, masking) are kept in
PyTorch – they are cheap compared with the O(B·D·N³) reduction.
The kernel is launched with one program per (batch·head) × output‑tile,
so the whole workload stays on‑device and avoids any host‑side loops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# Triton GEMM (A @ Bᵀ) – batched over the leading dimension (batch * head)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr,               # A matrix pointer      (bf16)
    b_ptr,               # B matrix pointer      (bf16)
    c_ptr,               # C output pointer      (fp32)
    BATCH_HEAD,          # total number of independent matrices
    M, N, K,             # matrix dims (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bi, stride_bk,   # strides for B (batch, row, col)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -----------------------------------------------------------------------
    # 1‑D grid: each program handles one (batch_head, tile_m, tile_n)
    # -----------------------------------------------------------------------
    pid = tl.program_id(0)
    # decompose pid -> batch_head, tile_m, tile_n
    tile_m_per_batch = tl.cdiv(M, BLOCK_M)
    tile_n_per_batch = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = tile_m_per_batch * tile_n_per_batch

    batch_head = pid // tiles_per_batch
    tile_id   = pid % tiles_per_batch
    tile_m = tile_id // tile_n_per_batch
    tile_n = tile_id % tile_n_per_batch

    # offsets inside the current tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the first element of this (batch_head) matrix
    a_batch_ptr = a_ptr + batch_head * stride_ah
    b_batch_ptr = b_ptr + batch_head * stride_bh
    c_batch_ptr = c_ptr + batch_head * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------------
    #  Loop over K dimension
    # -----------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (row=m, col=k)
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B: (row=n, col=k) – we need Bᵀ, so we load as (col=k, row=n)
        b = tl.load(
            b_batch_ptr
            + (offs_n[:, None] * stride_bi + offs_k[None, :] * stride_bk),
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )

        # compute dot in bfloat16, accumulate in fp32
        a = a.to(tl.bfloat16)
        b = b.to(tl.bfloat16)
        acc += tl.dot(a, b)          # (M,N) += (M,K)·(N,K)

    # -----------------------------------------------------------------------
    #  Store result
    # -----------------------------------------------------------------------
    c = acc.to(tl.float32)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Helper to launch the GEMM kernel
# --------------------------------------------------------------------------- #
def batched_gemm_outgoing(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : [B, H, N, N]  (dtype = torch.bfloat16, contiguous)
    returns c : [B, H, N, N] (dtype = torch.float32)
    Implements c = a @ bᵀ for each (B,H) slice.
    """
    assert a.shape == b.shape
    B, H, N, _ = a.shape
    M = N
    K = N

    # flatten leading dimensions
    a_flat = a.view(-1, N, N)
    b_flat = b.view(-1, N, N)
    batch_head = a_flat.shape[0]

    # output buffer (fp32)
    c_flat = torch.empty_like(a_flat, dtype=torch.float32, device=a.device)

    # -------------------------------------------------------------------
    #  Block sizes – chosen to give good occupancy on Hopper (SM90)
    # -------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid dimensions
    tiles_m = triton.cdiv(M, BLOCK_M)
    tiles_n = triton.cdiv(N, BLOCK_N)
    total_tiles = batch_head * tiles_m * tiles_n
    grid = (total_tiles,)

    # launch
    batched_gemm_outgoing_kernel[grid](
        a_ptr=a_flat,
        b_ptr=b_flat,
        c_ptr=c_flat,
        BATCH_HEAD=batch_head,
        M=M,
        N=N,
        K=K,
        stride_ah=a_flat.stride(0),
        stride_ai=a_flat.stride(1),
        stride_ak=a_flat.stride(2),
        stride_bh=b_flat.stride(0),
        stride_bi=b_flat.stride(1),
        stride_bk=b_flat.stride(2),
        stride_ch=c_flat.stride(0),
        stride_ci=c_flat.stride(1),
        stride_cj=c_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return c_flat.view(B, H, N, N)


# --------------------------------------------------------------------------- #
# Main entry point required by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(
    data: Tuple[
        torch.Tensor,                 # input tensor   [B, N, N, dim]
        torch.Tensor,                 # mask          [B, N, N]
        Dict[str, torch.Tensor],      # weights dict
        Dict[str, Any],               # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Only the large pair‑wise reduction is executed on Triton.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1️⃣ LayerNorm on input
    # ------------------------------------------------------------------- #
    def layer_norm(x, w, b):
        return F.layer_norm(
            x,
            normalized_shape=(w.shape[0],),
            weight=w,
            bias=b,
            eps=1e-5,
        )

    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )                               # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------- #
    def linear(x, w):
        return F.linear(x, w)       # bias == None

    left  = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional binary mask (broadcasted over hidden_dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask_exp = mask.unsqueeze(-1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul reduction – batched GEMM in Triton
    # ------------------------------------------------------------------- #
    # Convert to bfloat16 for the GEMM (fast on Hopper)
    left_bf16  = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # Permute to [B, H, N, N] and make contiguous
    left_perm  = left_bf16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_bf16.permute(0, 3, 1, 2).contiguous()

    # Run GEMM: for each (b, h) we compute left @ rightᵀ
    out_fp32 = batched_gemm_outgoing(left_perm, right_perm)   # [B, H, N, N] FP32

    # Permute back to the original layout [B, N, N, H]
    out = out_fp32.permute(0, 2, 3, 1).contiguous()           # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])               # [B,N,N,dim]

    return out
