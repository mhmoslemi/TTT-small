"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The only O(N³) part is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension `d` this is a matrix multiplication
        L = left[b,:, :, d]        (N×N)
        R = right[b,:, :, d]       (N×N)
        out[b,:, :, d] = L @ Rᵀ    (N×N)

We pack the batch and hidden dimensions into a single axis `M = B*H` and
run a batched GEMM A @ Bᵀ with a custom Triton kernel.  The kernel works
in FP16 for the matrix tiles (the inputs are cast to fp16) and accumulates
in FP32, giving a good speed‑memory trade‑off on Hopper (sm_90).  All
remaining operations (LayerNorm, linear projections, sigmoids, masking)
stay in PyTorch – they are negligible compared with the N³ work.

The kernel returns a tensor of shape (B, H, N, N); we permute it back to
(B, N, N, H) before the final normalisation / linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    N, M,                               # N = seq_len, M = B*H
    stride_ah, stride_ai, stride_ak,   # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,   # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,   # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for every (batch * head) slice.
    A,B are (M, N, N); C is (M, N, N).
    """
    pid_m = tl.program_id(0)   # block row   (i)
    pid_n = tl.program_id(1)   # block col   (j)
    pid_z = tl.program_id(2)   # flattened batch*head

    # --- offsets within the N×N matrix ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --- base pointers for the current (M‑th) matrix pair ----------------
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # tiled iteration over the reduction axis k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        if FP16:
            a = a.to(tl.float16)

        # B_T[k,j] shape (BLOCK_K, BLOCK_N)
        # original B is indexed as B[j,k]; we load it transposed.
        b_ptrs = (
            b_batch_ptr
            + offs_k[:, None] * stride_bk      # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj      # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        if FP16:
            b = b.to(tl.float16)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch_ptr
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  FP32
    returns       : (B, H, N, N)  FP32   (C = left @ rightᵀ per head)
    """
    B, H, N, _ = left.shape
    M = B * H                               # flatten batch*head

    # flatten to (M, N, N) for a single launch
    left_f  = left.reshape(M, N, N).contiguous()
    right_f = right.reshape(M, N, N).contiguous()
    out_f   = torch.empty_like(left_f)

    # ----- cast to fp16 for the kernel (the kernel will accumulate in fp32) ---
    left_f  = left_f.half()
    right_f = right_f.half()

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out_f.stride()

    # tuned tile sizes for Hopper (sm_90).  They fit in shared memory and give
    # good occupancy for the tested N up to 1024.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    FP16 = True

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z (batch * head)
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out_f,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=FP16,
    )
    # reshape back to (B, H, N, N) – already fp32
    return out_f.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy O(N³) contraction is performed by a fused Triton GEMM kernel.
    All other operations (LayerNorm, projections, gates, masking) are done
    with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections : dim -> hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ per hidden channel
    #    Convert to (B, H, N, N) layout expected by the kernel.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    out_perm = _trimul_gemm(left_perm, right_perm)       # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
