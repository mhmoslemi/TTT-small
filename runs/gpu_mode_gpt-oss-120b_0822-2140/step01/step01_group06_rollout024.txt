"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched matrix‑multiply over the (i,k) and (j,k) axes.
For each pair (batch, head) we compute
    out = left @ rightᵀ
where left,right ∈ (B·H, N, N).  The heavy work is performed by a
custom Triton GEMM kernel that tiles M, N and K dimensions (BLOCK_M,
BLOCK_N, BLOCK_K).  All other operations (LayerNorm, Linear, sigmoid‑gates,
final LayerNorm, output‑gate multiplication and the final projection) are
expressed in PyTorch for simplicity; the kernel alone satisfies the “must
use a kernel” requirement.

The kernel works on fp16 inputs (saved memory‑bandwidth) while accumulating
in fp32 to retain numeric quality, and writes fp32 results.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM – computes C = A @ Bᵀ for each (batch, head) pair.
# A : (B·H, M, K)   – row‑major
# B : (B·H, N, K)   – row‑major, we read it as (K, N) to achieve the transpose
# C : (B·H, M, N)   – row‑major, fp32 output
# ----------------------------------------------------------------------
@triton.jit
def triton_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # problem size
    stride_am, stride_ak,         # strides for A (row, col)
    stride_bk, stride_bj,         # strides for B (k‑dim, j‑dim) – note transpose
    stride_cm, stride_cn,         # strides for C (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n          # which block of rows
    pid_n = pid % num_n           # which block of cols

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create masks for out‑of‑bounds (needed for the last partial block)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[offs_m, offs_k] – shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ -> B[offs_k, offs_n] – shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product accumulates in fp32
        acc += tl.dot(a, b)

    # write back C[offs_m, offs_n]   (fp32)
    c = acc
    tl.store(
        c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32, CUDA
        - mask_tensor  : torch.Tensor, shape [B, N, N] or None
        - weights_dict : mapping str → torch.Tensor with all learned parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], dtype float32, CUDA
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)          # True → ignore mask
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (over last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                           # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM via Triton
    #    reshape to (B·H, N, N) and compute left @ rightᵀ
    # ------------------------------------------------------------------
    # bring the head dimension next to batch
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # flatten batch & head
    left  = left.view(B * hidden_dim, N, N)   # [B·H, N, N]
    right = right.view(B * hidden_dim, N, N)   # [B·H, N, N]

    # cast to fp16 for bandwidth; accumulation stays fp32
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output placeholder (fp32)
    out_fp32 = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # strides (in elements) needed by the kernel
    stride_am, stride_ak = left_fp16.stride()[1:]   # (row‑stride, col‑stride) = (N, 1)
    stride_bk, stride_bj = right_fp16.stride()[2], right_fp16.stride()[1]
    # right_fp16 shape (B·H, N, N) → stride = (N*N, N, 1)
    # we want Bᵀ, thus:
    #   k‑dimension stride = stride of last dim (1)
    #   j‑dimension stride = stride of middle dim (N)
    stride_cm, stride_cn = out_fp32.stride()[1:]    # (row‑stride, col‑stride) = (N, 1)

    # tiling parameters – tuned for Hopper (sm_90)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_m = triton.cdiv(N, BLOCK_M)
    num_n = triton.cdiv(N, BLOCK_N)
    grid = (num_m * num_n,)

    triton_gemm_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N, N, N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [B, N, N, H]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                           # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # ------------------------------------------------------------------
    out = out * out_gate                        # still [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
