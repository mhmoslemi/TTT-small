"""
TriMul “outgoing” implementation (AlphaFold‑3) with a fused
batched matrix‑multiply kernel written in Triton.

The heavy part of the forward pass is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
For each hidden channel *d* this is a matrix multiplication
    (N×N) @ (N×N)ᵀ .
We pack the hidden dimension into the batch dimension (B·H matrices of size N×N)
and compute all of them in a single Triton kernel.  The rest of the
operations (LayerNorm, linear projections, gating, final projection) stay in
PyTorch so we keep numerical‑stability and avoid extra bookkeeping.

The kernel works for any batch size ≤2, any sequence length that fits in the
H200 shared memory (up to 1024 in the test suite) and any hidden size that
fits into the stride‑layout (hidden_dim is usually 128).  All inputs are
expected to be on CUDA and of dtype torch.float32; the kernel loads the
matrices as fp16 to reduce memory traffic and accumulates in fp32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel ---------------------------------------------------------
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr,                     # left  [B*H, N, N]  (fp16)
    B_ptr,                     # right [B*H, N, N]  (fp16) – will be read transposed
    C_ptr,                     # out   [B*H, N, N]  (fp32)
    M, N, K,                   # matrix sizes (M=N=seq_len, K=seq_len)
    stride_aba, stride_ama, stride_aka,   # strides for A  (batch, row, col)
    stride_bba, stride_bmb, stride_bkb,   # strides for B  (batch, row, col)
    stride_cca, stride_cma, stride_cna,   # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)            # batch*hidden index
    pid_m = tl.program_id(1)            # tile row
    pid_n = tl.program_id(2)            # tile col

    # ------------------------------------------------------------------
    # coordinate ranges for the tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the block for this batch
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_aba + offs_m[:, None] * stride_ama + offs_k[None, :] * stride_aka
    b_ptrs = B_ptr + pid_b * stride_bba + offs_k[:, None] * stride_bkb + offs_n[None, :] * stride_bmb   # note B is read transposed

    # ------------------------------------------------------------------
    # Accumulator in FP32 (more precise than FP16)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, apply bounds masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product over K tile
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_aka
        b_ptrs += BLOCK_K * stride_bkb

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_cca + offs_m[:, None] * stride_cma + offs_n[None, :] * stride_cna
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input  : torch.Tensor [B, N, N, dim]      (float32, CUDA)
        mask   : torch.Tensor [B, N, N] or None   (float32/bool, CUDA)
        weights: dict of torch.Tensor parameters   (float32, CUDA)
        config : dict with keys
                 - "dim"         : int  (input channel dimension)
                 - "hidden_dim"  : int  (projection dimension)

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm on last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply the gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched matmul kernel
    #    left/right : [B, N, N, hidden]  ->  [B*hidden, N, N]
    #    we move hidden to the batch dimension because the kernel can only
    #    handle three‑dimensional tensors.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden, N, N)
    right_flat = right.view(B * hidden, N, N)

    # Convert to fp16 for the kernel (memory‑bound part)
    left_flat_fp16 = left_flat.to(torch.float16)
    right_flat_fp16 = right_flat.to(torch.float16)

    # Output buffer (fp32 because we accumulate in fp32)
    out_flat = torch.empty((B * hidden, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 6. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per tile
    BLOCK_N = 64   # cols per tile
    BLOCK_K = 64   # reduction tile size

    grid = (
        B * hidden,                                   # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),                      # M tiles
        triton.cdiv(N, BLOCK_N),                      # N tiles
    )

    # stride extraction (batch, row, col)
    a_batch, a_row, a_col = left_flat_fp16.stride()
    b_batch, b_row, b_col = right_flat_fp16.stride()
    c_batch, c_row, c_col = out_flat.stride()

    # ------------------------------------------------------------------
    # 7. Call the kernel
    # ------------------------------------------------------------------
    _batched_matmul_kernel[grid](
        left_flat_fp16,
        right_flat_fp16,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original layout: [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (hidden dim) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
