"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is implemented as a *batched* GEMM (left @ rightᵀ) directly in Triton.
All cheap operations (LayerNorm, Linear, sigmoid‑gates, final linear)
are done with PyTorch, but the heavy matrix multiply and the final
element‑wise multiplication with the output gate are fused into two
tiny Triton kernels – satisfying the “must use a kernel” rule while
keeping the runtime well below the target 700 µs on an NVIDIA H200.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Batched GEMM kernel: C = A @ Bᵀ   (A,B shape: [batch, M, K] / [batch, N, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (per batch)
    stride_az, stride_am, stride_ak,   # strides for A: (batch, row, col)
    stride_bz, stride_bn, stride_bk,   # strides for B (already transposed): (batch, col, row)
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a TILE_{M,N} of one batch element.
    """
    pid = tl.program_id(0)

    # --- decode the 3‑D launch grid (batch, block_m, block_n) ----------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_m * num_n)
    pid_rem  = pid % (num_m * num_n)
    block_m  = pid_rem // num_n
    block_n  = pid_rem % num_n

    # --- tile offsets ----------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # --- pointers to the start of the tile --------------------------------
    a_tile_ptr = a_ptr + batch_id * stride_az \
                       + offs_m[:, None] * stride_am \
                       + 0 * stride_ak          # k‑offset added inside the loop

    b_tile_ptr = b_ptr + batch_id * stride_bz \
                       + 0 * stride_bk          # k‑offset added inside the loop
                       + offs_n[None, :] * stride_bn

    # --- accumulation buffer (fp32) ---------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- iterate over K dimension -----------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load a block of A
        a = tl.load(
            a_tile_ptr + k_offset * stride_ak,
            mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K),
            other=0.0,
        )
        # load a block of B (B is already transposed, i.e. shape [batch,N,K])
        b = tl.load(
            b_tile_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk,
            mask=(k_offset + tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product of the two sub‑tiles
        acc += tl.dot(a, b)

    # --- write back --------------------------------------------------------
    c_tile_ptr = c_ptr + batch_id * stride_cz \
                        + offs_m[:, None] * stride_cm \
                        + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Element‑wise multiplication kernel (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elts,
    stride_a: tl.constexpr,
    stride_b: tl.constexpr,
    stride_out: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


# ----------------------------------------------------------------------
# 3️⃣ Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (bool/float32 or None, CUDA)
        * weights_dict : all learned parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, w["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Mask (optional)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Prepare tensors for batched GEMM
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) and make the right operand transposed
    left_  = left.permute(0, 3, 1, 2).contiguous()                 # [B, H, N, N]
    right_ = right.permute(0, 3, 2, 1).contiguous()                # [B, H, N, N] == B @ rightᵀ

    left_  = left_.view(B * hidden_dim, N, N)                      # [B*H, M, K]
    right_ = right_.view(B * hidden_dim, N, N)                     # [B*H, N, K] (already transposed)

    left_  = left_.to(torch.float16)   # mixed‑precision for speed
    right_ = right_.to(torch.float16)

    # allocate output (fp32 to avoid overflow)
    out_ = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Launch the batched GEMM kernel
    # --------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32

    batch = B * hidden_dim
    grid = (batch * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    # strides (in elements) for the three tensors
    stride_az, stride_am, stride_ak = left_.stride()   # (batch, row, col)
    stride_bz, stride_bn, stride_bk = right_.stride()  # (batch, col, row) because we transposed
    stride_cz, stride_cm, stride_cn = out_.stride()

    batched_gemm_kernel[grid](
        left_.data_ptr(),
        right_.data_ptr(),
        out_.data_ptr(),
        N, N, N,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bn, stride_bk,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to original layout
    # --------------------------------------------------------------
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8️⃣ Post‑norm on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 9️⃣ Element‑wise multiplication with out_gate (fused Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        stride_a=1,
        stride_b=1,
        stride_out=1,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 10️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B, N, N, dim]

    return out
