"""
TriMul “outgoing” kernel (AlphaFold‑3).

The only O(N³) part of the module is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply per hidden dimension `d` :

    (B, H, N, N) @ (B, H, N, N)ᵀ   →   (B, H, N, N)

where H = hidden_dim.  The Triton kernel below computes this contraction
with:
  * tiling (TILE_M, TILE_N, TILE_K) = (64, 64, 64)
  * FP16 loads for bandwidth, FP32 accumulation for accuracy
  * one program instance per (tile‑row, tile‑col, batch×head)

All other operations (LayerNorm, Linear projections, gating, final
LayerNorm/Linear) are performed with PyTorch – they are negligible compared
to the cubic contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]   (i.e. A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # const float*   -> left  (B,H,N,N)
    B_ptr,                # const float*   -> right (B,H,N,N)
    C_ptr,                #       float*   -> out   (B,H,N,N)
    N, H,                 # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """Batched mat‑mul of (B,H,N,N) with the transpose of the second operand."""
    pid_m = tl.program_id(0)               # tile row   (i)
    pid_n = tl.program_id(1)               # tile col   (j)
    pid_hb = tl.program_id(2)              # combined batch*head

    batch = pid_hb // H
    head  = pid_hb % H

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)   # i indices
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)   # j indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, TILE_K)):
        offs_k = kk * TILE_K + tl.arange(0, TILE_K)   # k indices
        mask_k = offs_k < N

        # A : (TILE_M, TILE_K)   = left[batch, head, i, k]
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[j,k] → load as (TILE_K, TILE_N)
        # original layout: B[batch, head, j, k] (i.e. stride_b_i is over j,
        # stride_b_j over k).  By swapping the strides we get the transpose.
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k becomes first dim
                  + offs_n[None, :] * stride_b_i)  # j becomes second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)          # (TILE_M,TILE_K) @ (TILE_K,TILE_N)

    # ------------------------------------------------------------------
    # store result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (inp, mask, weights, config)
        - inp    : torch.Tensor [B, N, N, dim]   (float32 or bfloat16)
        - mask   : torch.Tensor [B, N, N]        (bool/float, may be all‑ones)
        - weights: dict of torch.Tensors (model parameters)
        - config : dict with keys ``dim`` and ``hidden_dim`` and optional
                   ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device
    dtype = torch.float32

    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper shortcuts for weight tensors
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # (hidden_dim, dim)
    right = F.linear(x, w("right_proj.weight"))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating sigmoids (still shape [B,N,N,hidden_dim])
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy cubic contraction – Triton kernel
    #    Reshape to (B, H, N, N) and make contiguous for simple strides.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # Strides in **elements**, not bytes (required by Triton)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j) – we will treat as transposed
    c_s = out_t.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # Launch configuration
    # ------------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch × head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,
    )

    # Back to original layout: [B, N, N, hidden_dim]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + gating + linear back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, w("to_out.weight"))  # (dim, hidden_dim)

    return out.to(dtype)
