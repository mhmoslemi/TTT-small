"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The heavy part is the 3‑dimensional contraction  

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be expressed as a batched GEMM A @ Bᵀ for each
(batch, head) pair, where A and B are (N×N) matrices.
A Triton kernel performs many of these GEMMs in parallel,
using fp16 loads and fp32 accumulation.  All other
operations (LayerNorm, linear projections, gates) stay in
native PyTorch for simplicity and numerical stability.

Supported shapes:
    input : [B, N, N, C]   (float32)
    mask  : [B, N, N]      (bool/float, optional)
    C = dim, hidden_dim = H (fixed to 128 in the benchmark)
Batch ≤ 2, N up to 1024.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _trilmul_gemm_T_kernel(
    A_ptr,                     # *fp16  (batch*H, M, K)   – M==N, K==N
    B_ptr,                     # *fp16  (batch*H, N, K)   – will be read as K×N (transposed)
    C_ptr,                     # *fp32  (batch*H, M, N)
    M, N, K,                   # matrix dims (all = N)
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (see comment below)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tiled matrix multiplication C = A @ Bᵀ for many independent
    batches.  The kernel is launched with a 3‑D grid:
        grid = (ceil(N/BM), ceil(N/BN), batch*H)
    """
    pid_b = tl.program_id(2)                 # batch*head index
    pid_m = tl.program_id(0)                 # block row
    pid_n = tl.program_id(1)                 # block col

    # ----- block coordinates -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointers to the current blocks -----
    a_ptrs = (A_ptr + pid_b * stride_Ab
              + offs_m[:, None] * stride_Am
              + offs_k[None, :] * stride_Ak)

    # B is read as (K, N) = Bᵀ, i.e. we stride across the original column
    # with stride_Bk and across the original row with stride_Bn.
    b_ptrs = (B_ptr + pid_b * stride_Bb
              + offs_k[:, None] * stride_Bk
              + offs_n[None, :] * stride_Bn)

    # ----- accumulator (fp32) -----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main K‑loop -----
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)      # (BM, BK)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)      # (BK, BN)
        acc += tl.dot(a, b)                     # (BM,BN)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ----- write result -----
    c_ptrs = (C_ptr + pid_b * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated TriMul “outgoing” forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights      : dict of model parameters (float32)
        - config       : dict with at least {"dim", "hidden_dim", "nomask"}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight shortcuts
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton (batched GEMM A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # reshape to (batch*head, N, N) and make contiguous
    left_view  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside kernel for bandwidth, fp32 accumulator
    left_fp16  = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()

    # To read Bᵀ we swap the row/col strides:
    stride_Bk = stride_Bc            # step along original column (K)
    stride_Bn = stride_Br            # step along original row    (N)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # tile sizes – empirically good for Hopper/SM90
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch*head
    )

    _trilmul_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → out‑gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)                 # back to original dim C

    return out
