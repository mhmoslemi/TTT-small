"""
TriMul “outgoing” kernel (AlphaFold3) – highly‑optimized Triton implementation.

The expensive part of the module is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.  This file
computes that contraction with a custom Triton kernel that evaluates,
for every (batch, head) pair, the product

    C[b,h] = A[b,h] @ B[b,h].T                # A,B : (N,N)

All other operations (layer‑norms, linear projections, gating) are cheap
and stay in PyTorch.  The kernel works in FP16 for bandwidth and uses
FP32 accumulation, matching the reference behavior while being fast on
Hopper (sm_90).

The host entry point `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output
and returns a tensor of shape [B, N, N, dim] (float32).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)^T
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem dimensions
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    For each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B are (N, N) matrices)
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A : (TILE_M, TILE_K)  – row‑major (i,k)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[j,k] → load as (TILE_K, TILE_N)
        #   B layout: (batch, head, i, j)
        #   we want elements B[batch, head, j, k]
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k as first axis
                  + offs_n[None, :] * stride_b_i)  # j as second axis
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K  → (TILE_M,TILE_N)  FP32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (fp32/bf16)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be all‑ones
        - weights      : dict of torch.Tensor, model parameters
        - config       : dict, must contain ``dim`` and ``hidden_dim`` and optionally
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    (x, mask, weights, config) = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (biases are stored separately)
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        x,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – shape → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # (hidden_dim, dim)
    right = F.linear(x, w("right_proj.weight"))

    # mask handling (applied before gating)
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – also shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction with Triton
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Output tensor (same layout)
    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # Strides in *elements* (required by Triton)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will read transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # Heuristic tile size – larger tiles for larger N improve occupancy
    if N <= 256:
        TILE_M = TILE_N = TILE_K = 64
        num_warps = 4
    elif N <= 512:
        TILE_M = TILE_N = TILE_K = 96
        num_warps = 6
    else:
        TILE_M = TILE_N = TILE_K = 128
        num_warps = 8

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch*head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=num_warps,
    )

    # Back to original layout: [B, N, N, hidden_dim]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm, gating, and projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate               # broadcast over hidden_dim
    out = F.linear(out, w("to_out.weight"))   # (dim, hidden_dim)

    return out.to(dtype)
