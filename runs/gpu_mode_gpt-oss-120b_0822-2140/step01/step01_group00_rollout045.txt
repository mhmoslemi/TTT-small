"""
Fast “outgoing” TriMul implementation.

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched GEMM:
    left  : (B, D, N, N)   → A   (M=N, K=N)
    right : (B, D, N, N)   → B   (N, K=N)  (we need Bᵀ)

out[b,d] = A @ Bᵀ      → (N, N)

The kernel tiles the GEMM with BLOCK_M×BLOCK_N tiles and
BLOCK_K‑wise accumulation.  All the surrounding LayerNorm / Linear /
gate logic stays in PyTorch – only the mask‑multiply (optional) and the
core N³ reduction are done in Triton.  The final result is returned in the
original layout [B, N, N, D] and dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# 1️⃣  Element‑wise mask * tensor (required “some Triton work”)
# --------------------------------------------------------------------------- #
@triton.jit
def mask_mul_kernel(src_ptr, mask_ptr, dst_ptr, N_ELEM,
                    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < N_ELEM

    s = tl.load(src_ptr + offs, mask=m, other=0.0)
    msk = tl.load(mask_ptr + offs, mask=m, other=0.0)
    tl.store(dst_ptr + offs, s * msk, mask=m)


def apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """src and mask have identical shape, both contiguous float32."""
    assert src.shape == mask.shape
    numel = src.numel()
    BLOCK = 131072                     # 128 K – fits most tensors comfortably
    grid = (triton.cdiv(numel, BLOCK),)

    out = torch.empty_like(src)
    mask_mul_kernel[grid](
        src_ptr=src,
        mask_ptr=mask,
        dst_ptr=out,
        N_ELEM=numel,
        BLOCK=BLOCK,
    )
    return out


# --------------------------------------------------------------------------- #
# 2️⃣  Core TriMul kernel (batched GEMM A @ Bᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
        a_ptr, b_ptr, c_ptr,                     # pointers
        B, D, M, N, K,                           # static sizes
        stride_ah, stride_ad, stride_am, stride_ak,   # A strides (batch, head, M, K)
        stride_bh, stride_bd, stride_bn, stride_bk,   # B strides (batch, head, N, K)
        stride_ch, stride_cd, stride_cm, stride_cn,   # C strides (batch, head, M, N)
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr):
    """
    Compute, for every (b, d) pair:
        C[b,d] = A[b,d] @ B[b,d].T          # A,B shape (M,N)
    where A,B are stored as (B, D, M, K) and (B, D, N, K) respectively.
    """
    # ----- decode the 3‑D grid -------------------------------------------
    pid_n = tl.program_id(0)            # block column
    pid_m = tl.program_id(1)            # block row
    pid_hd = tl.program_id(2)           # combined batch‑head index

    # decode batch and head from pid_hd
    d = pid_hd % D
    b = pid_hd // D

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the beginning of the (b,d) tiles
    a_base = a_ptr + b * stride_ah + d * stride_ad
    b_base = b_ptr + b * stride_bh + d * stride_bd
    c_base = c_ptr + b * stride_ch + d * stride_cd

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop (tile over the reduction dimension) --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # offsets inside the K‑tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for the K‑dimension
        mask_k = offs_k < K

        # -------- load A tile (M × K) ------------------------------------
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # -------- load Bᵀ tile (K × N) ------------------------------------
        # B is stored as (N, K); we need the transposed view (K, N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # -------- multiply‑accumulate ------------------------------------
        acc += tl.dot(a, b)               # (M,N) += (M,K)·(K,N)

    # ----- write result ----------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# 3️⃣  Custom kernel entry point (everything else stays in PyTorch)
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, CUDA)
        - mask         : [B, N, N]        (float32, 0/1, CUDA)
        - weights      : dict of torch.Tensor containing all LayerNorm / Linear params
        - config       : dict with keys
              * dim        – input channel dimension (C)
              * hidden_dim – projection dimension (C_hidden)
              * nomask     – bool, ``True`` → mask is all‑ones

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # helper functions
    # ------------------------------------------------------------------- #
    def layer_norm(t, w, b):
        return F.layer_norm(t, (w.shape[0],), weight=w, bias=b, eps=1e-5)

    def linear(t, w):
        # Linear without bias
        return F.linear(t, w)

    # ------------------------------------------------------------------- #
    # 1️⃣  Input LayerNorm
    # ------------------------------------------------------------------- #
    x = layer_norm(x, weights["norm.weight"], weights["norm.bias"])          # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2️⃣  Projections to hidden_dim
    # ------------------------------------------------------------------- #
    left = linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣  Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        # broadcast mask to the hidden dimension
        mask_exp = mask.unsqueeze(-1).expand_as(left).contiguous()
        left = apply_mask(left, mask_exp)
        right = apply_mask(right, mask_exp)

    # ------------------------------------------------------------------- #
    # 4️⃣  Gating
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))    # same shape

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣  Rearrange to (B, Head, N, N) for the GEMM kernel
    # ------------------------------------------------------------------- #
    # Head dimension == hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M = N
    K = N

    # allocate output tensor in (B, H, N, N)
    out_perm = torch.empty((B, H, N, N), dtype=torch.float32,
                           device=x.device)

    # strides (in elements) – note that .stride() returns element strides
    a_strides = left_perm.stride()   # (stride_batch, stride_head, stride_M, stride_K)
    b_strides = right_perm.stride()
    c_strides = out_perm.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_N),           # block columns
        triton.cdiv(N, BLOCK_M),           # block rows
        B * H,                             # batch * head
    )

    trimul_kernel[grid](
        a_ptr=left_perm,
        b_ptr=right_perm,
        c_ptr=out_perm,
        B=B,
        D=H,
        M=M,
        N=N,
        K=K,
        stride_ah=a_strides[0],
        stride_ad=a_strides[1],
        stride_am=a_strides[2],
        stride_ak=a_strides[3],
        stride_bh=b_strides[0],
        stride_bd=b_strides[1],
        stride_bn=b_strides[2],
        stride_bk=b_strides[3],
        stride_ch=c_strides[0],
        stride_cd=c_strides[1],
        stride_cm=c_strides[2],
        stride_cn=c_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6️⃣  Output LayerNorm (hidden → hidden)
    # ------------------------------------------------------------------- #
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    # ------------------------------------------------------------------- #
    # 7️⃣  Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣  Final projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
