"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We therefore launch a **batched GEMM** kernel that computes many independent
(N×N)·(N×N)ᵀ products in one launch.
All cheap operations (LayerNorm, linear projections, gating, masking) stay
in PyTorch.  The kernel works in fp16 (or bf16) for the two inputs, accumulates
in fp32 and writes fp32 results – this gives a large bandwidth reduction on
H200 while keeping the final precision identical to the reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  (B*H) independent GEMMs  (N×N) • (N×N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    N, M,                                # N = seq_len, M = B*H
    stride_ah, stride_ai, stride_ak,     # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,     # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,     # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    a,b : fp16 tensors (M,N,N)
    c   : fp32 output   (M,N,N)
    Each program instance computes a BLOCK_M × BLOCK_N tile of one
    (i,j) sub‑matrix of the result for a given (batch, head) pair.
    """
    pid_m = tl.program_id(0)      # tile row   (i)
    pid_n = tl.program_id(1)      # tile col   (j)
    pid_z = tl.program_id(2)      # batch*head index

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the selected (batch,head) slice
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]   -> (BLOCK_M, BLOCK_K)   fp16
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ[k,j]  -> we load B[j,k] and treat it as transposed
        # shape (BLOCK_K, BLOCK_N)   fp16
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk        # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj        # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot product, accumulation in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # write back C[i,j]  (fp32)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, H, N, N)  fp16
    returns    : (B, H, N, N)  fp32   each slice = left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                         # flattened batch*head

    # reshape to (M,N,N) – memory is already contiguous after permute()
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32)

    # strides as int64 (required by Triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # block sizes – tuned for H200 SMs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )
    _trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Heavy contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
    is executed by a custom Triton batched‑GEMM kernel; the surrounding
    LayerNorm / Linear / gating logic stays in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections : dim -> hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)          # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.to(device).unsqueeze(-1)       # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and reorder to (B, H, N, N) for the kernel
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton batched GEMM : out = left @ rightᵀ  (per hidden channel)
    #    Result is fp32.
    # ------------------------------------------------------------------
    out_fp32 = _trimul_gemm(left, right)          # (B, H, N, N) fp32

    # ------------------------------------------------------------------
    # 7. Back‑to original layout (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B,N,N,dim)

    return out.to(torch.float32)
