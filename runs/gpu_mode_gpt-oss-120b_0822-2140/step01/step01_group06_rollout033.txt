## Triton‑accelerated *outgoing* TriMul

```python
"""
TriMul (outgoing) – fast forward pass.

The expensive O(N³·H) part is implemented as a custom Triton batched GEMM:
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
i.e. a mat‑mul of A @ Bᵀ for every (batch, head) pair.
All other operations (LayerNorm, Linear, gating, final projection) are
performed with native PyTorch.  The kernel works with arbitrary batch
size B, sequence length N and hidden dimension H (called “hidden_dim” in the
config) and respects a possible mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def triton_gemm_outgoing_kernel(
    left_ptr,          # *float32   (B, H, N, N)
    right_ptr,         # *float32   (B, H, N, N)  (will be read as Bᵀ)
    out_ptr,           # *float32   (B, H, N, N)
    B, H, N,
    # strides (in elements, not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (b, h):
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    using a tiled GEMM (A @ Bᵀ).
    """
    pid = tl.program_id(0)          # tile over (i, j)
    pid_bh = tl.program_id(1)       # tile over (batch, head)

    # ------------------- decode batch / head -------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------- tile coordinates ---------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑index
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑index

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------- k‑loop initial pointers --------------------
    offs_k = tl.arange(0, BLOCK_K)

    # left[b, h, i, k]
    a_ptrs = (
        left_ptr
        + b * stride_l_b
        + h * stride_l_h
        + offs_m[:, None] * stride_l_i
        + offs_k[None, :] * stride_l_k
    )
    # right[b, h, j, k]  -> we need right[b, h, k, j] for the transpose
    b_ptrs = (
        right_ptr
        + b * stride_r_b
        + h * stride_r_h
        + offs_k[:, None] * stride_r_k
        + offs_n[None, :] * stride_r_j
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- main K‑loop -------------------------------
    for _k in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles, guard against OOB
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < N),
                    other=0.0)          # (M, K)
        b = tl.load(b_ptrs,
                    mask=mask_n[None, :] & (offs_k < N),
                    other=0.0)          # (K, N)

        # promote to float32 (inputs are float32 already, this is a no‑op)
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        acc += tl.dot(a_f32, b_f32)          # (M, N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # ------------------- write result -------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called H in the paper
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (over the last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                          # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Reshape for the batched GEMM (B, H, N, N) – contiguous layout
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Allocate output buffer
    out_perm = torch.empty_like(left_perm, dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 6️⃣ Launch the Triton GEMM kernel
    # --------------------------------------------------------------
    # Tile sizes – tuned for Hopper (SM90) memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),   # tiles over (i, j)
        B * hidden_dim,                                      # (batch, head) dimension
    )

    # strides (in elements)
    s_l = left_perm.stride()   # (stride_b, stride_h, stride_i, stride_k)
    s_r = right_perm.stride()
    s_o = out_perm.stride()

    triton_gemm_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, hidden_dim, N,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7️⃣ Back to the original layout [B, N, N, H]
    # --------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1)          # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Post‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 9️⃣ Multiply with the output gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
