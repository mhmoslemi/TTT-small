"""
custom_kernel – Triton‑accelerated forward pass of the TriMul “outgoing” operator.

Algorithm
---------
1️⃣ Layer‑norm the input (last dim = dim).
2️⃣ Linear projections → left / right tensors (B×N×N×H) with no bias.
3️⃣ (Optional) mask the projected tensors.
4️⃣ Gated‑linear transforms: sigmoid gates applied element‑wise.
5️⃣ Core contraction (batched GEMM):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is computed as a batched matrix multiplication A @ Bᵀ for each
   (batch, head) pair using a Triton kernel that tiles the M, N and K
   dimensions (BLOCK_M, BLOCK_N, BLOCK_K).  Inputs are cast to fp16,
   accumulation is fp32, output is fp32.
6️⃣ Post‑processing – LayerNorm, out‑gate, final linear projection back to dim.

The implementation keeps the heavyweight tensor ops in PyTorch (layer‑norm,
linear, sigmoid) and moves only the O(B·H·N³) contraction to Triton,
which is the dominant cost.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A, B : [batch, M, K]),  C : [batch, M, N]
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # fp16 *, shape (batch, M, K)
    B_ptr,                     # fp16 *, shape (batch, N, K)  (will be read as K×N)
    C_ptr,                     # fp32 *, shape (batch, M, N)
    M, N, K,                   # matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # strides of A  (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides of B  (batch, col‑in‑K, row‑in‑N)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the right operand is transposed (A @ Bᵀ)."""
    pid_b = tl.program_id(2)          # batch * head index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # --------------------------------------------------------------
    # block start offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # pointers for this block
    # --------------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    # B is stored as (batch, N, K); we need Bᵀ => (batch, K, N)
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # --------------------------------------------------------------
    # accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float32)          # (M×K) in fp32
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (K×N) in fp32
        acc += tl.dot(a, b)                           # (M×N)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (bool or float)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # weight shortcuts
    # --------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating – sigmoid applied to separate linear layers
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM left @ rightᵀ per head
    # --------------------------------------------------------------
    B, N, _, H = left.shape                       # H == hidden_dim

    # reshape to (B*H, N, N) and make contiguous so that strides are simple
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 for bandwidth, fp32 accumulation
    left_fp16 = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # ------------------------------------------------------------------
    # strides (in elements, not bytes) – needed by the kernel
    # ------------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (batch, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()  # (batch, row, col)

    # For the transposed view of B we swap the K and N strides:
    stride_Bk = stride_Bc   # step along K (original column)
    stride_Bn = stride_Br   # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # launch parameters – tuned for Hopper, N ≤ 1024, H = 128
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,                               # M, N, K (all = N)
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)    # back to original dim

    return out
