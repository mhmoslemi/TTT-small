"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast Triton implementation.

The heavy work is the 3‑dimensional contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

for every batch * hidden_dim slice.  This is exactly a batched matrix‑multiply
A @ Bᵀ where each slice has shape (N,N) and there are B·H such slices.
We fuse all slices in one Triton kernel, use fp16 loads for bandwidth,
fp32 accumulation for accuracy and a simple tiling (128×128×32) that is
well‑suited to the Hopper SM‑90 architecture.

All other operations (LayerNorm, Linear, gating, final projection) are kept
in PyTorch – they are cheap compared with the O(B·H·N³) core and avoid
unnecessary kernel launches.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, N, N]), batch = B*H
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                      # *fp16  (batch, M, K)
    B_ptr,                      # *fp16  (batch, N, K)  (will be read as K×N)
    C_ptr,                      # *fp32  (batch, M, N)
    M, N, K,                    # matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)        # batch index = B*H
    pid_m = tl.program_id(0)        # block row
    pid_n = tl.program_id(1)        # block col

    # ------------------------------------------------------------------
    # block offsets (in element indices)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # pointers for the current block
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    # B is accessed as (K,N) i.e. the transpose of the logical (N,K)
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk
        + offs_n[None, :] * stride_Bn
    )

    # ------------------------------------------------------------------
    # accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K-loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)          # (M,K) in fp32
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)          # (K,N) in fp32
        acc += tl.dot(a, b)        # (M,N) += (M,K)*(K,N)

        # advance pointers for next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # store result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask         : torch.Tensor, shape [B, N, N]    (bool/float)
        - weights      : dict of parameter tensors (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight tensors
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also shape (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM with Triton
    #    out = left @ rightᵀ   per hidden dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                    # H == hidden_dim

    # (B,H,N,N) layout is convenient for the kernel
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    # flatten batch & head so kernel sees a single batch dimension
    left_flat  = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # fp16 inside the kernel for bandwidth, keep fp32 accumulator
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # ------------------------------------------------------------------
    # strides (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (batch, M, K)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()  # (batch, N, K)

    # For the transposed view of B we swap the last two strides:
    stride_Bk = stride_Bc   # step along the original column (K)
    stride_Bn = stride_Br   # step along the original row    (N)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ------------------------------------------------------------------
    # Tiling – 128×128 tiles with a K‑tile of 32 work well on SM‑90
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch*head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden_dim)
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing
    #    – LayerNorm over hidden_dim
    #    – multiply by out_gate
    #    – final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B,N,N,dim)

    return out
