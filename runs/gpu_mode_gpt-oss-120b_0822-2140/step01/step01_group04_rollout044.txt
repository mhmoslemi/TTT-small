"""
TriMul “outgoing” kernel (AlphaFold3) – Triton + PyTorch

The expensive part of the module is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply per hidden dimension `d`
( A @ Bᵀ with A,B ∈ ℝ^{B·H×N×N} ).
The kernel below computes, for every (batch, head) pair, a TILE_M×TILE_N
block of the result using FP16 loads and FP32 accumulation.
All cheap operations (LayerNorm, Linear, gating, final projection) stay in
PyTorch, keeping the implementation simple yet fast on Hopper GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B,H,N,N) @ (B,H,N,N)^T  →  (B,H,N,N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # left  (B,H,N,N)  fp16
    B_ptr,                     # right (B,H,N,N)  fp16
    C_ptr,                     # output (B,H,N,N) fp32
    N,                         # sequence length
    H,                         # hidden dimension
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    For each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. a batched mat‑mul with the second operand transposed on the last two axes.
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # -------- A : (TILE_M, TILE_K) ----------------------------
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_i[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # -------- Bᵀ : we need B[j,k] -> load as (TILE_K, TILE_N) --
        # note the intentional swap of stride_b_i / stride_b_j
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k dimension uses original j‑stride
                  + offs_j[None, :] * stride_b_i)  # j dimension uses original i‑stride
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot product over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_i[:, None] * stride_c_i
              + offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool/float, optional)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helper to fetch parameters (they are already on the correct device)
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension ``dim``
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False) + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))          # [B,N,N,hidden_dim]
    right = F.linear(x, w("right_proj.weight"))         # [B,N,N,hidden_dim]

    if not nomask:
        # mask shape: [B,N,N] → [B,N,N,1] for broadcasting
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))    # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy O(N³) contraction via Triton
    #    Transform to (B, H, N, N) layout for easier indexing.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Cast to fp16 for bandwidth‑limited loads (kernel accumulates in fp32)
    left_t_fp16  = left_t.to(torch.float16)
    right_t_fp16 = right_t.to(torch.float16)

    # Output buffer (fp32 accumulation)
    out_t = torch.empty_like(left_t_fp16, dtype=torch.float32)   # [B, H, N, N]

    # Strides in number of elements (required by the kernel)
    a_s = left_t_fp16.stride()   # (batch, head, i, k)
    b_s = right_t_fp16.stride()  # (batch, head, i, j)
    c_s = out_t.stride()         # (batch, head, i, j)

    # Tile sizes – tuned for Hopper (SM90) occupancy.
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # rows
        triton.cdiv(N, TILE_N),          # cols
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t_fp16,
        right_t_fp16,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,
    )

    # Back to original layout: (B, N, N, hidden_dim)
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Final LayerNorm + gating + projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    # out_gate was computed on the original x, shape [B,N,N,hidden_dim]
    out = out * out_gate
    out = F.linear(out, w("to_out.weight"))   # [B,N,N,dim]

    return out.to(dtype)
