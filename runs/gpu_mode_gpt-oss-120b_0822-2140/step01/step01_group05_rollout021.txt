"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a fused batched
matrix‑multiply kernel written in Triton.

The expensive step is
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

For every hidden channel *h* this is a matrix multiplication
    (N×N) @ (N×N)^T .
We treat each (batch, hidden) pair as an independent batch and compute all
the products with a single Triton kernel that tiles the M, N and K axes.
All other operations (LayerNorm, linear projections, gating, final
Linear) stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (A @ Bᵀ)   A,B shape = (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # base pointers
    M, N, K,                           # matrix dimensions (M=N=seq_len, K=seq_len)
    stride_ab, stride_am, stride_ak,   # strides of A  (batch, row, col)
    stride_bb, stride_bm, stride_bn,   # strides of B  (batch, row, col) – note col stride = stride_bn
    stride_cb, stride_cm, stride_cn,   # strides of C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                # batch*hidden index
    pid_m = tl.program_id(1)                # tile row
    pid_n = tl.program_id(0)                # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for A (row‑major) and Bᵀ (accessed as B[row=k, col=n])
    a_ptrs = (
        A_ptr
        + pid_b * stride_ab
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bb
        + offs_k[:, None] * stride_bm   # k is now the “row” index of Bᵀ
        + offs_n[None, :] * stride_bn   # n is the “col” index of Bᵀ
    )

    # accumulator in FP32 (even if inputs are FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A and Bᵀ
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # dot over K
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bm

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]    (float32, on CUDA)
            - mask  : torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all on CUDA)
            - config : {'dim': int, 'hidden_dim': int}
    Returns:
        torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weight_dict, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]          # hidden_dim == c_z in AlphaFold‑3

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weight_dict["norm.weight"],
        bias=weight_dict["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weight_dict["left_proj.weight"])
    right = F.linear(x, weight_dict["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weight_dict["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weight_dict["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weight_dict["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matmul: out = left @ rightᵀ  (per hidden channel)
    #    Shape transformation:
    #        left , right : [B, N, N, H] -> [B, H, N, N] -> [B*H, N, N]
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape                       # batch, seq_len
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * hidden, N, N)       # [B*H, N, N]
    right_flat = right.view(B * hidden, N, N)

    out_flat = torch.empty_like(left_flat)        # same dtype (float32)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # program_id(0) – columns
        triton.cdiv(N, BLOCK_M),   # program_id(1) – rows
        B * hidden,                # program_id(2) – batch*hidden
    )

    # stride tuples (batch, row, col)
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bm, stride_bn = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bm, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden) + gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weight_dict["to_out_norm.weight"],
        bias=weight_dict["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weight_dict["to_out.weight"])

    return out
