"""
TriMul (outgoing) – fast forward pass with a fused Triton GEMM.

The only O(N³·H) work (the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] )
is performed by a custom Triton kernel that treats each (batch, head)
pair as an independent matrix‑multiply A @ Bᵀ.  All other operations
(LayerNorm, Linear, Sigmoid gates, final projection) stay in PyTorch
to keep the implementation short and numerically stable.

The kernel works on fp16 inputs and accumulates in fp32, finally storing
the result in fp16.  After the kernel we cast back to fp32, apply the
post‑norm, the output‑gate (element‑wise) and the final linear projection.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # flattened pointers
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_ah, stride_am, stride_ak,   # batch, row, col strides for A
    stride_bh, stride_bk, stride_bn,   # batch, row, col strides for B
    stride_ch, stride_cm, stride_cn,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    a: (M, K)   b: (N, K)   c: (M, N)
    One program instance processes a single (batch, head) pair.
    """
    pid = tl.program_id(0)                      # index of (batch, head)

    # Pointers to the current matrix pair
    a_batch = a_ptr + pid * stride_ah
    b_batch = b_ptr + pid * stride_bh
    c_batch = c_ptr + pid * stride_ch

    # Tile offsets within the M×N output matrix
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = tl.arange(0, BLOCK_K)

        # Load A tile  (M×K)  and B tile (N×K) – note B is transposed later
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
            dtype=tl.float16,
        )
        b = tl.load(
            b_batch + (offs_n[:, None] * stride_bn) + (offs_k[None, :] * stride_bk),
            mask=(offs_n[:, None] < N) & (offs_k[None, :] < K),
            other=0.0,
            dtype=tl.float16,
        )
        # b is loaded as (N, K); we need (K, N) for the dot, so transpose via view
        b = tl.trans(b)   # now shape (K, N)

        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_batch += BLOCK_K * stride_ak
        b_batch += BLOCK_K * stride_bk

    # Write back the fp16 result
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask)


# ----------------------------------------------------------------------
# Public entry point used by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask_tensor  : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weight_dict  : mapping of all Linear / LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device

    # --------------------------------------------------------------
    # 2️⃣ Input LayerNorm (over dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, w["right_proj.weight"])  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Gates (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, w["out_gate.weight"]))      # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Prepare tensors for the Triton GEMM
    #    Move the hidden dimension to the second axis and flatten
    #    (batch, head) -> total_heads.
    # --------------------------------------------------------------
    # shape -> (B, hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    total_heads = B * hidden_dim
    left_flat = left.view(total_heads, N, N)   # (Htot, N, N)
    right_flat = right.view(total_heads, N, N)

    # Cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Allocate output buffer (fp16)
    out_fp16 = torch.empty_like(left_fp16, device=device)

    # --------------------------------------------------------------
    # 7️⃣ Launch Triton GEMM kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides in *elements* (not bytes)
    stride_ah = left_fp16.stride(0)   # batch*head stride
    stride_am = left_fp16.stride(1)   # rows stride (M)
    stride_ak = left_fp16.stride(2)   # cols stride (K)

    stride_bh = right_fp16.stride(0)
    stride_bn = right_fp16.stride(1)  # rows of Bᵀ (i.e. N)
    stride_bk = right_fp16.stride(2)  # cols of Bᵀ (i.e. K)

    stride_ch = out_fp16.stride(0)
    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)

    grid = (total_heads,)
    outgoing_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,          # M
        N,          # N
        N,          # K
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to float32 and reshape to (B, N, N, hidden_dim)
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8️⃣ Post‑norm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 9️⃣ Apply the output gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B, N, N, dim]

    return out
