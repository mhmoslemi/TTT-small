"""
TriMul (outgoing) – fast forward pass
------------------------------------
*   LayerNorm on the input (dim)
*   Linear projections → left/right  (dim ➜ hidden_dim)
*   Optional mask   (broadcasted on the last dim)
*   Gating (sigmoid of three more linears)
*   Core O(N³·H) operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Implemented as a **batched GEMM** on the (B·H) matrix pairs
    using a custom Triton kernel (fp16 compute, fp32 accumulation).
*   LayerNorm on the hidden dimension
*   Element‑wise multiplication with the output‑gate (fused Triton
    element‑wise kernel)
*   Final linear projection back to the original dimension.
All heavy arithmetic (the N³ contraction and the final element‑wise
product) runs on Triton; everything else uses fast PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Batched GEMM kernel (left @ rightᵀ) over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    M, N, K,
    # strides (in elements) for a, b (transposed) and c
    stride_ah, stride_am, stride_ak,          # a: (batch_head, row, col)
    stride_bh, stride_bk, stride_bn,          # b (already transposed inside kernel)
    stride_ch, stride_cm, stride_cn,          # c
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    a : (B*H, M, K)          fp16
    b : (B*H, N, K)          fp16   (original right is (B*H, K, N); we treat it transposed)
    c : (B*H, M, N)          fp32   (accumulated result)
    """
    bh_id = tl.program_id(0)          # index over B*H
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # ------------------------------------------------------------------
    #   Compute global offsets for this thread block
    # ------------------------------------------------------------------
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = off_m < M
    mask_n = off_n < N

    # pointers for this batch‑head
    a_batch = a_ptr + bh_id * stride_ah
    b_batch = b_ptr + bh_id * stride_bh
    c_batch = c_ptr + bh_id * stride_ch

    # ------------------------------------------------------------------
    #   Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #   Loop over K dimension (tiling)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # current tile offsets in K dimension
        cur_k = k * BLOCK_K + off_k
        mask_k = cur_k < K

        # ---- load A tile ------------------------------------------------
        # a[off_m, cur_k]
        a_ptrs = a_batch + off_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # ---- load B tile (transposed) -----------------------------------
        # we need rightᵀ → treat original right as (B*H, K, N)
        # after transpose we load b[cur_k, off_n]
        b_ptrs = b_batch + cur_k[:, None] * stride_bk + off_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # ---- matrix multiply --------------------------------------------
        acc += tl.dot(a, b)          # fp16×fp16 → fp32 accumulation

    # ------------------------------------------------------------------
    #   Write back result
    # ------------------------------------------------------------------
    c_ptrs = c_batch + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Element‑wise multiplication kernel (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elements,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 3️⃣  Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (bool/float32 or None)
        * weights_dict : model parameters
        * config_dict  : {'dim': int, 'hidden_dim': int, ...}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)       # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid of three linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Core contraction – batched GEMM on (B·hidden_dim) pairs
    # ------------------------------------------------------------------
    #   reshape to (B, H, N, N) → contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    #   cast to fp16 for the GEMM kernel
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # output buffer (fp32)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # --------------------------------------------------------------
    # launch parameters
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                 # program_id(0) – batch * head
        triton.cdiv(N, BLOCK_M),       # program_id(1) – block rows
        triton.cdiv(N, BLOCK_N),       # program_id(2) – block cols
    )

    # strides (in elements)
    stride_ah = left_fp16.stride(0)
    stride_am = left_fp16.stride(2)          # row stride
    stride_ak = left_fp16.stride(3)          # col stride

    stride_bh = right_fp16.stride(0)
    # we load right transposed: row‑stride = original col stride, col‑stride = original row stride
    stride_bk = right_fp16.stride(3)          # corresponds to K (original col)
    stride_bn = right_fp16.stride(2)          # corresponds to N (original row)

    stride_ch = out_fp32.stride(0)
    stride_cm = out_fp32.stride(2)
    stride_cn = out_fp32.stride(3)

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N,                # M
        N,                # N
        N,                # K
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    #   back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣  Post‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣  Fuse out_gate (element‑wise) with Triton
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total = out_flat.numel()

    BLOCK = 1024
    grid_elem = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid_elem](
        out_flat,
        gate_flat,
        out_flat,
        total,
        BLOCK=BLOCK,
    )
    # now out contains out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣  Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
