"""
TriMul “outgoing” kernel (AlphaFold‑3)

The dominant work is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel *d* this is a matrix multiplication
    L = left[b,:, :, d]   (N×N)
    R = right[b,:, :, d]  (N×N)
    out[b,:, :, d] = L @ Rᵀ                (N×N)

We batch the GEMMs over (batch × hidden) → M independent
multiplications of shape (N,N)·(N,N)ᵀ and compute them with a single
Triton kernel.  The kernel:

* works on fp16 data (the linear layers produce fp16 tensors) to reduce
  memory traffic,
* accumulates in fp32 (the default behaviour of `tl.dot` with fp16
  arguments on Hopper),
* uses a 128×128×32 tile which matches the H200 shared‑memory / register
  budget,
* handles arbitrary N (not necessarily a multiple of the tile size) via
  masks.

All cheap operations (LayerNorm, linear projections, sigmoids, masking,
output‑norm + final linear) stay in PyTorch.  The kernel is launched on
the flattened dimension `M = batch * hidden_dim` and returns a tensor in
layout (B, H, N, N) which is permuted back to (B, N, N, H) before the
output‑norm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr,               # left  (M, N, N)  fp16
    b_ptr,               # right (M, N, N)  fp16
    c_ptr,               # out   (M, N, N)  fp16
    N,                   # seq_len
    stride_ah, stride_ai, stride_ak,   # A strides (M, N, N)
    stride_bh, stride_bj, stride_bk,   # B strides (M, N, N)
    stride_ch, stride_ci, stride_cj,   # C strides (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a BLOCK_M × BLOCK_N tile of the
    matrix product  C = A @ Bᵀ  for one (batch, hidden) pair (pid_z).
    """
    pid_m = tl.program_id(0)      # tile row  (i dimension)
    pid_n = tl.program_id(1)      # tile col  (j dimension)
    pid_z = tl.program_id(2)      # flattened (batch * hidden)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the selected (M‑th) matrix pair
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop (tile over the reduction dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]  shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + \
                 offs_m[:, None] * stride_ai + \
                 offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ[k, j]  shape (BLOCK_K, BLOCK_N)
        # original B is indexed as B[j, k]; we load it transposed
        b_ptrs = b_batch + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # tl.dot with fp16 inputs accumulates in fp32 on Hopper
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  dtype=fp16
    returns       : (B, H, N, N)  dtype=fp16  where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                                   # flatten batch & hidden

    # flatten to (M, N, N)
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f)              # fp16 output

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – chosen for H200 (large shared memory, many registers)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )

    _trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps works well on sm90
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)

# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    A small portion (the O(N³) contraction) is executed in a fused Triton
    kernel.  All other linear / norm / sigmoid / masking operations are
    performed with native PyTorch – they are cheap compared to the
    matrix‑multiply work.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections: dim → hidden_dim (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)               # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ  (per hidden channel)
    #    Convert to (B, H, N, N) layout, cast to fp16 for the kernel.
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_h  = left_perm.to(torch.float16)
    right_h = right_perm.to(torch.float16)

    out_h = _trimul_gemm_fp16(left_h, right_h)           # (B, H, N, N), fp16

    # back to original layout and fp32
    out = out_h.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)       # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                        # (B, N, N, dim)

    return out
