## Triton‑accelerated “outgoing” TriMul

```python
"""
TriMul “outgoing” forward pass (no gradients) with a fused Triton
batched GEMM C = A @ Bᵀ.

Algorithm
---------
1. Layer‑norm over the input `dim`.
2. Linear projections → left/right (shape B,N,N,hidden_dim).
3. Optional mask multiplication.
4. Compute per‑head gates (sigmoid of linear proj) and apply them.
5. Reshape to (B*hidden_dim, N, N) and cast to FP16.
6. Batched GEMM (FP16×FP16 → FP32 accumulation) where each batch element
   computes   out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k].
7. Reshape back to (B,N,N,hidden_dim) → FP32.
8. Post‑norm, gate, final linear projection back to `dim`.
All heavy O(N³·hidden_dim) work is performed inside a single Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (B_H, M, K)   (row‑major)
#   B : (B_H, N, K)   (row‑major) – we treat it logically transposed
#   C : (B_H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix extents
    stride_az, stride_am, stride_ak,         # strides for A (batch, row, col)
    stride_bz, stride_bn, stride_bk,         # strides for B (batch, row, col)
    stride_cz, stride_cm, stride_cn,         # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed."""
    pid_batch = tl.program_id(2)          # index of (batch * head)
    pid_m = tl.program_id(0)              # block‑row index
    pid_n = tl.program_id(1)              # block‑col index

    # ------------------------------------------------------------------
    # Block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current tiles of A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_az \
                   + offs_m[:, None] * stride_am \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n),
    # i.e. row‑k, col‑n in the original layout.
    b_ptrs = b_ptr + pid_batch * stride_bz \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop (tiling over the reduction dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = tl.arange(0, BLOCK_K) < K  # mask for the last partial tile
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & cur_k[None, :],
                    other=0.0)                # (BLOCK_M, BLOCK_K) FP16
        b = tl.load(b_ptrs,
                    mask=cur_k[:, None] & mask_n[None, :],
                    other=0.0)                # (BLOCK_K, BLOCK_N) FP16

        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as FP16 to keep bandwidth low)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host helper that launches the kernel
# ----------------------------------------------------------------------
def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Perform (B*H,N,N) @ (B*H,N,N)ᵀ in FP16 with FP32 accumulation.
    a, b : contiguous FP16 tensors of shape (B*H, N, N)
    Returns c of the same shape and dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "square matrices required"

    c = torch.empty_like(a)

    # Strides expressed in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – 128×128 tiles give good occupancy on Hopper,
    # BLOCK_K=32 fits nicely into registers.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M (i‑dimension)
        triton.cdiv(N, BLOCK_N),   # blocks along N (j‑dimension)
        B_H,                       # one program per (batch*head)
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bn,
        stride_bk,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps works well for 128×128 tiles on sm90
        num_stages=3,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor, shape [B, N, N] (bool or float)
        weights      : dict of weight tensors (all float32)
        config       : dict with keys
                       - "dim"          (int)
                       - "hidden_dim"   (int)
                       - "nomask"       (bool, optional, default True)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Projections without bias
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (mask shape: B,N,N).  Broadcast to last dim.
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Reshape to (B*hidden, N, N) and cast to FP16
    # --------------------------------------------------------------
    # (B, N, N, hidden) -> (B, hidden, N, N) -> (B*hidden, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM: out_fp16 = left_fp16 @ right_fp16ᵀ
    # --------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left_fp16, right_fp16)   # (B*hidden, N, N)

    # --------------------------------------------------------------
    # 7. Restore original layout and dtype (float32)
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Post‑projection LayerNorm + gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # broadcasting works: (B,N,N,hidden)

    # --------------------------------------------------------------
    # 9. Final linear back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
