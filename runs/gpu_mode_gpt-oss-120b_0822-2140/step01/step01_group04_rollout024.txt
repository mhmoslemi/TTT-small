"""
TriMul “outgoing” kernel (AlphaFold3)

The expensive part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply per hidden dimension `d`:
for each (batch, head) pair we compute  (N,N) @ (N,N)^T.

The implementation below
* keeps the cheap LayerNorm / Linear / gating ops in PyTorch,
* reshapes the tensors to (B, H, N, N) and launches a custom Triton kernel
  that performs the batched mat‑mul with FP16 loads / FP32 accumulation,
* finally restores the original layout and applies the output LayerNorm,
  gating and projection.

The kernel is written for Hopper (sm_90) but works on any recent CUDA GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:   C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]    (i.e. A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    N, H,                               # problem sizes
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Compute for every (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    where A and B have shape (N, N) and C has shape (N, N).
    """
    pid_m = tl.program_id(0)          # tile row   → i
    pid_n = tl.program_id(1)          # tile col   → j
    pid_hb = tl.program_id(2)         # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------- tile offsets -------------------
    offs_i = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_j = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (high precision)
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------- loop over K -------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A: (TILE_M, TILE_K)  – rows i, cols k
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_i[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ: we need B[j,k] → treat k as first dim, j as second
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k axis
                  + offs_j[None, :] * stride_b_i)  # j axis
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # ------------------- store result -------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_i[:, None] * stride_c_i
              + offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32 or bf16)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights      : dict of torch.Tensors (already on the correct device)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch parameters
    # ------------------------------------------------------------------
    def W(name):
        return weights[name]

    eps = 1e-6

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=W("norm.weight"),
        bias=W("norm.bias"),
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, W("left_proj.weight"))   # (hidden_dim, dim)
    right = F.linear(x, W("right_proj.weight"))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; convert to same dtype as proj tensors
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, W("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, W("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, W("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched mat‑mul in Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # allocate output tensor
    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in elements (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j) – we will read transposed
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile size – 64 works well for N up to 1024 on Hopper
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    # launch grid: (rows, cols, batch*head)
    grid = (
        triton.cdiv(N, TILE_M),
        triton.cdiv(N, TILE_N),
        B * hidden_dim,
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,                # good default for Hopper
    )

    # reshape back to [B, N, N, H]
    out = out_t.permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + gating + projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=W("to_out_norm.weight"),
        bias=W("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, W("to_out.weight"))  # (dim, hidden_dim)

    return out.to(dtype)
