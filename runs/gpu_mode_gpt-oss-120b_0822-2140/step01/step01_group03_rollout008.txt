"""
TriMul “outgoing” Triton implementation (AlphaFold‑3 style)

Algorithm
---------
1️⃣  Layer‑norm the input tensor (B,N,N,C).
2️⃣  Linear projections → left/right tensors (B,N,N,H) (no bias).
3️⃣  (optional) mask the projected tensors (broadcast on H).
4️⃣  Gated‑linear transforms (sigmoid gates) applied element‑wise.
5️⃣  Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM A @ Bᵀ for each (batch, head) pair.
   The kernel works on a view (B*H, N, N) and uses
   fp16 loads with fp32 accumulation.
6️⃣  Layer‑norm → out‑gate → final linear projection back to C.

The kernel is tiled (BLOCK_M × BLOCK_N) with a K‑tile of size BLOCK_K.
Block sizes (64,64,32) give good occupancy on Hopper (sm_90) for the
tested sequence lengths (≤1024) and hidden_dim = 128.
All other ops stay in PyTorch.  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : (B*H, N, N)  fp16
#   C    : (B*H, N, N)  fp32
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                     # *fp16  (batch, M, K)
    B_ptr,                     # *fp16  (batch, N, K)  (read as K×N)
    C_ptr,                     # *fp32  (batch, M, N)
    M, N, K,                   # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (k‑step, n‑step)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch index = B*H
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk
        + offs_n[None, :] * stride_Bn
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)               # (M, K_tile)
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)               # (K_tile, N)

        acc += tl.dot(a, b)             # (M, N)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ extract weight tensors (all are float32)
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = (
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # F.linear expects (..., in_features)
    left = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1]
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core contraction via Triton
    #    out = left @ rightᵀ   (per hidden channel)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N)  →  each head is an independent GEMM
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (bandwidth) – keep a copy to avoid inplace cast
    left_fp16 = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides in element units (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # B will be read as K×N, so:
    stride_Bk = stride_Bc    # step along K (original column)
    stride_Bn = stride_Br    # step along N (original row)

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # --------------------------------------------------------------
    # Tile sizes – tuned for Hopper, satisfy shared‑memory limits.
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H,                     # batch * hidden
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N,
        N,
        N,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)     # back to original dim C

    return out
