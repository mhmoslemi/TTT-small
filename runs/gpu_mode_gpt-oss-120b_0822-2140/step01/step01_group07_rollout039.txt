"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply with shape
    (B·H, N, N) @ (B·H, N, N)ᵀ      (H = hidden_dim)

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(because they are cheap) and fuse only the N³ part in Triton.
The kernel loads the two operands in fp16 (halving bandwidth) while
accumulating in fp32 for numerical stability.  Tiling is
BLOCK_M × BLOCK_N × BLOCK_K = 64 × 64 × 32, which gives good occupancy on
Hopper (SM‑90).  The kernel works for any batch size ≤2, any sequence length
up to 1024 and any hidden dimension that fits in the “head” dimension of the
batched GEMM.

The host side reshapes the tensors to (M, N, N) where
    M = B * hidden_dim,
then launches a 3‑D grid:
    pid_z – different (batch, head) matrix
    pid_m – row tile (i)
    pid_n – column tile (j)
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM   C = A @ Bᵀ
#  A : (M, N, K)    B : (M, N, K)   (here K == N)
#  C : (M, N, N)
# All loads are fp16, accumulator is fp32.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel_fp16(
    a_ptr,            # *fp16   [M, N, K]
    b_ptr,            # *fp16   [M, N, K]
    c_ptr,            # *fp32   [M, N, N]
    M, N, K,
    stride_am, stride_an, stride_ak,   # A strides
    stride_bm, stride_bn, stride_bk,   # B strides
    stride_cm, stride_cn, stride_ck,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # which (batch,head) matrix  (0 … M‑1)
    pid_m = tl.program_id(0)          # row tile   (i)
    pid_n = tl.program_id(1)          # col tile   (j)

    # ------------------------------------------------------------------- #
    # offsets for the current tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # pointers for A (row‑major) and Bᵀ (we read B as (k, j) to get the transpose)
    # ------------------------------------------------------------------- #
    a_ptrs = a_ptr + pid_z * stride_am \
                    + offs_m[:, None] * stride_an \
                    + offs_k[None, :] * stride_ak

    b_ptrs = b_ptr + pid_z * stride_bm \
                    + offs_k[:, None] * stride_bk \
                    + offs_n[None, :] * stride_bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # tiled K‑loop
    # ------------------------------------------------------------------- #
    num_iters = tl.cdiv(K, BLOCK_K)
    for _ in range(num_iters):
        # load A and Bᵀ as fp16, respecting the edge masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → fp32
        acc += tl.dot(a, b, out=acc)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------- #
    # write result (fp32)
    # ------------------------------------------------------------------- #
    c_ptrs = c_ptr + pid_z * stride_cm \
                    + offs_m[:, None] * stride_cn \
                    + offs_n[None, :] * stride_ck

    tl.store(c_ptrs,
              acc,
              mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module with a fused Triton GEMM.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask         : torch.Tensor[B, N, N]        (bool/float, may be None)
        - weights      : dict of torch.Tensor – all model parameters
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Fuse the N³ contraction with Triton
    #    Reshape to (M, N, N) where M = B * hidden_dim
    # --------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N) → (M,N,N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim

    left_fp16  = left_t.view(M, N, N).to(torch.float16)
    right_fp16 = right_t.view(M, N, N).to(torch.float16)

    # output is kept in fp32 (final datatype)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # strides in *elements* (not bytes) – required by the kernel
    stride_am, stride_an, stride_ak = left_fp16.stride()
    stride_bm, stride_bn, stride_bk = right_fp16.stride()
    stride_cm, stride_cn, stride_ck = out_fp32.stride()

    # tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # rows (i)
        triton.cdiv(N, BLOCK_N),      # cols (j)
        M,                            # batch*head dimension
    )

    tri_mul_kernel_fp16[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        M, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # good default for fp16 on SM90
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
