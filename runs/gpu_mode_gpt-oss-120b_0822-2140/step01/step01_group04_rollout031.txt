"""
TriMul “outgoing” kernel (AlphaFold3 – forward only).

The expensive part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply for every hidden dimension *d*.
We implement this contraction with a Triton kernel that computes, for each
(batch, head) pair, the product  A @ Bᵀ  where

    A = left  : (B, H, N, N)   (i‑k matrix)
    B = right : (B, H, N, N)   (j‑k matrix, transposed on‑the‑fly)

All remaining operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
stay in PyTorch – they are cheap compared to the O(N³) contraction.
The kernel uses FP16 loads, FP32 accumulation and writes the result in FP32,
matching the reference implementation while keeping memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)^T
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    N, H,                               # problem size
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Computes for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B have shape (N, N))
    """
    pid_m = tl.program_id(0)   # tile row   (i)
    pid_n = tl.program_id(1)   # tile col   (j)
    pid_hb = tl.program_id(2)  # combined batch*head

    batch = pid_hb // H
    head  = pid_hb %  H

    # tile offsets
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A[b, h, i, k]   shape (TILE_M, TILE_K)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[b, h, j, k] → treat as (TILE_K, TILE_N)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k is now first dim
                  + offs_n[None, :] * stride_b_i)  # j is second dim
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # store
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32/bfloat16)
        - mask         : torch.Tensor [B, N, N] (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False) and gating
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden_dim]

    # optional mask (broadcasted over hidden_dim)
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Heavy contraction with Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and cast to FP16 for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)  # [B,H,N,N]

    # allocate output in FP32
    out_t = torch.empty((B, hidden_dim, N, N), device=device, dtype=dtype)

    # strides in *elements* (not bytes)
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)   we will load transposed
    c_s = out_t.stride()    # (batch, head, i, j)

    # tile sizes – tuned for Hopper (SM90) where 64×64 tiles give good occupancy
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # i‑tiles
        triton.cdiv(N, TILE_N),          # j‑tiles
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,                     # 4 warps gives good SM utilization on H200
        num_stages=2,                    # double‑buffering
    )

    # back to original layout  [B,N,N,hidden_dim]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 4. Final LayerNorm, gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                      # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out.to(dtype)
