"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The heavy part of TriMul is the 4‑D contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a batched matrix multiply
    L = left[b,:, :, d]          (N×N)
    R = right[b,:, :, d]         (N×N)
    out[b,:, :, d] = L @ Rᵀ      (N×N)

We launch a single Triton kernel that computes *M = B·hidden_dim* independent
GEMMs (A @ Bᵀ) in mixed‑precision:
* inputs are loaded as float16 (half‑precision, 2 × lower bandwidth)
* the dot product is performed in fp16 but accumulated in fp32 (TL’s default)
* the final result is written as fp32.

All cheap operations (LayerNorms, linear projections, sigmoids, masking) stay
in PyTorch.  The kernel is the only non‑trivial GPU piece, satisfying the
requirement “at least part of the work is in Triton”.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# Input / output are 4‑D tensors with layout (M, N, N) where M = B*hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_fp16_kernel(
    a_ptr,                     # left  (M,N,N)  fp16
    b_ptr,                     # right (M,N,N)  fp16
    c_ptr,                     # out   (M,N,N)  fp32
    N,                         # seq_len
    stride_ah, stride_ai, stride_ak,   # strides of a (M,N,N)
    stride_bh, stride_bj, stride_bk,   # strides of b (M,N,N)
    stride_ch, stride_ci, stride_cj,   # strides of c (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row  (i)
    pid_n = tl.program_id(1)          # block col  (j)
    pid_z = tl.program_id(2)          # flattened batch*hidden (M)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (M‑th) pair of matrices
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32 (to avoid overflow)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i, k]  : (BLOCK_M, BLOCK_K)   fp16
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load Bᵀ[k, j] : (BLOCK_K, BLOCK_N)   fp16
        # Original B is indexed as B[j, k]; we load transposed
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk       # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj       # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Accumulate
        acc += tl.dot(a, b)   # fp16*fp16 → fp32

    # Store the result (fp32)
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the mixed‑precision kernel
# ----------------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  fp16  (already on the device)
    returns      : (B, H, N, N)  fp32
    """
    B, H, N, _ = left.shape
    M = B * H                               # flattened batch*head

    # flatten to (M, N, N)
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – chosen for Hopper (SM90) where 128² tiles fit in
    # shared memory and give good occupancy.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch * head)
    )
    trimul_gemm_fp16_kernel[grid](
        left_f, right_f, out,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    data = (input_tensor, mask, weights, config)
    - input_tensor : [B, N, N, dim]   (float32)
    - mask         : [B, N, N]        (float32 or bool) – may be ignored
    - weights      : dict of torch tensors (LayerNorm/Gate/Linear weights)
    - config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
    - out : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections: dim -> hidden_dim (no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)      # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Apply optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool; convert to float for multiplication
        mask_f = mask.to(device).unsqueeze(-1).to(left.dtype)
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ per hidden channel
    #    Convert to (B, hidden, N, N) and half‑precision for bandwidth.
    # --------------------------------------------------------------
    # (B,N,N,hidden) -> (B, hidden, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16  = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Batched GEMM in fp16 → fp32
    out_fp32 = trimul_gemm_fp16(left_fp16, right_fp16)   # (B, hidden, N, N)

    # Back to original layout (B,N,N,hidden)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) and out‑gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
