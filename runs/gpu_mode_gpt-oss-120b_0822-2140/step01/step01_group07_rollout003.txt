"""
TriMul (outgoing) – fast Triton‑fused contraction.

The expensive part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiply of shape
(M, N, N) with the transpose of the second operand, where
    M = B * hidden_dim.
The kernel below computes this GEMM tile‑wise (BLOCK_M × BLOCK_N
with a K‑tile of size BLOCK_K).  All arithmetic is done in FP32,
inputs may be FP16 to lower bandwidth, but the final result is
returned in FP32 as required by the reference implementation.

Only the contraction is Triton‑fused; the surrounding LayerNorm,
Linear, and gating logic stay in PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ   (batched GEMM)
# left  : (M, N, K)   where K == N
# right : (M, N, K)   we need the transpose on the last two dims, i.e.
#          rightᵀ (M, K, N)  → accessed as (M, k, j)
# out   : (M, N, N)
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,          # *float32  [M, N, K]
    right_ptr,         # *float32  [M, N, K]
    out_ptr,           # *float32  [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over output rows   (i)
    pid_n = tl.program_id(1)          # tile over output cols   (j)
    pid_z = tl.program_id(2)          # independent matrix (M)

    # ------------------------------------------------------------------- #
    # Offsets for the current tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Pointers for the left operand (no transpose)
    #   left[pid_z, i, k]
    # ------------------------------------------------------------------- #
    a_ptrs = left_ptr + pid_z * stride_lm \
                       + offs_m[:, None] * stride_ln \
                       + offs_k[None, :] * stride_lk

    # ------------------------------------------------------------------- #
    # Pointers for the right operand (transposed on the last two dims)
    #   we need right[pid_z, j, k] → accessed as (pid_z, k, j)
    # ------------------------------------------------------------------- #
    b_ptrs = right_ptr + pid_z * stride_rm \
                        + offs_k[:, None] * stride_rk \
                        + offs_n[None, :] * stride_rn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Main GEMM loop over K
    # ------------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance K‑dimension
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------- #
    # Write result
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok

    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator with a fused
    Triton GEMM for the N³ contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor (no bias linear layers)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------- #
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]          # called hidden_dim in the paper

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------- #
    # shape after linear: [B, N, N, hidden]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3. Optional mask (broadcast on channel dim)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Fuse the N³ contraction with Triton
    #    reshape to (M, N, N) where M = B * hidden
    # --------------------------------------------------------------- #
    # (B,N,N,hidden) -> (B, hidden, N, N) -> (M, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden

    left_reshape = left_t.view(M, N, N)
    right_reshape = right_t.view(M, N, N)

    # output buffer
    out_reshape = torch.empty_like(left_reshape, device=device)

    # --------------------------------------------------------------- #
    # 6. Launch Triton GEMM
    # --------------------------------------------------------------- #
    stride_lm, stride_ln, stride_lk = left_reshape.stride()
    stride_rm, stride_rn, stride_rk = right_reshape.stride()
    stride_om, stride_on, stride_ok = out_reshape.stride()

    # Tune these block sizes for Hopper – they give good occupancy and
    # keep shared‑memory usage modest.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows  (i)
        triton.cdiv(N, BLOCK_N),          # cols  (j)
        M                                 # independent matrices
    )

    _tri_mul_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps performs well on Hopper for fp32
    )

    # --------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, hidden]
    # --------------------------------------------------------------- #
    out = out_reshape.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 8. Output LayerNorm, gating and final linear (dim)
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
