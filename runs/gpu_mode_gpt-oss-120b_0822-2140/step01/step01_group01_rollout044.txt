"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched GEMM (A @ Bᵀ) for every (batch, hidden) pair.
We launch a Triton kernel that computes many independent GEMMs in FP16
while accumulating in FP32 to preserve precision.  All other ops
(LayerNorm, linear projections, sigmoids, masking, final norm/linear)
are cheap and stay in PyTorch.

The kernel works for any batch size (1‑2), sequence length up to 1024,
and hidden dimensions 128/384.  It expects the inputs to be contiguous
in the layout (B, H, N, N) and returns the same layout, which the host
code permutes back to (B, N, N, H).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many (N,N) matrices
#   A, B : (M, N, N)  FP16
#   C    : (M, N, N)  FP32 accumulator, written as FP32
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    N, M,                         # N = seq_len, M = batch * hidden
    stride_ah, stride_ai, stride_ak,   # A strides (M,N,N)
    stride_bh, stride_bj, stride_bk,   # B strides (M,N,N)
    stride_ch, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # block row (i)
    pid_n = tl.program_id(1)      # block col (j)
    pid_z = tl.program_id(2)      # flattened (batch * hidden)

    # Offsets of the current tile within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch,head) matrix pair
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i,k]  → (BLOCK_M, BLOCK_K)  FP16
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load Bᵀ[k,j] i.e. original B[j,k] → (BLOCK_K, BLOCK_N) FP16
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk      # k as row of Bᵀ
            + offs_n[None, :] * stride_bj      # j as column of Bᵀ
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)   # accumulates in FP32 automatically

    # Store result as FP32
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul.
    left / right : (B, H, N, N)  FP16
    returns       : (B, H, N, N)  FP32
    """
    B, H, N, _ = left.shape
    M = B * H

    # flatten to (M, N, N) – stride information stays correct
    left_f  = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64 (required by triton)
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # Tunable tile sizes – good fit for Hopper SMs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z (batch * hidden)
    )
    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    All cheap ops stay in PyTorch, the O(N³) contraction is executed in
    a fused Triton kernel (trimul_gemm).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1. Input LayerNorm ----------
    w_norm = weights["norm.weight"].to(device)
    b_norm = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=w_norm, bias=b_norm)

    # ---------- 2. Linear projections ----------
    w_left = weights["left_proj.weight"].to(device)   # (hidden, dim)
    w_right = weights["right_proj.weight"].to(device)
    left = F.linear(x, w_left)      # (B, N, N, hidden)
    right = F.linear(x, w_right)

    # ---------- 3. Optional mask ----------
    if not nomask:
        mask = mask.to(device).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ---------- 4. Gating ----------
    w_left_gate = weights["left_gate.weight"].to(device)
    w_right_gate = weights["right_gate.weight"].to(device)
    w_out_gate = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Triton GEMM (left @ rightᵀ) ----------
    # Convert to (B, H, N, N) and half‑precision for bandwidth savings
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_perm = trimul_gemm(left_perm, right_perm)   # (B, H, N, N) FP32

    # back to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Output LayerNorm + out_gate ----------
    w_out_norm = weights["to_out_norm.weight"].to(device)
    b_out_norm = weights["to_out_norm.bias"].to(device)
    out = F.layer_norm(out, (hidden_dim,), weight=w_out_norm, bias=b_out_norm)
    out = out * out_gate

    # ---------- 7. Final linear projection ----------
    w_to_out = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, w_to_out)                    # (B, N, N, dim)

    return out.to(torch.float32)
