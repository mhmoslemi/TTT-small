"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation
-----------------------------------------------------------------
The costly part of the TriMul forward pass is the three‑dimensional
contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We accelerate this step with a custom Triton kernel that computes,
for every (batch, head) pair, the product

    C[b,h,i,j] = A[b,h,i,k] @ B[b,h,j,k]   (i.e. A @ Bᵀ)

where `A` = left and `B` = right are stored in
shape (B, H, N, N).  All other operations (LayerNorm, Linear,
gating, final LayerNorm/Linear) are cheap and stay in PyTorch.

The kernel works on FP16 inputs (to reduce memory traffic) while
accumulating in FP32 for numerical stability.  Tile sizes (64,64,64)
and 8 warps give good occupancy on Hopper (SM‑90) and bring the
runtime of the N³ contraction down to a few hundred micro‑seconds
for the target sizes (N ≤ 1024, H = hidden_dim ≤ 384).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    N, H,                              # problem dimensions
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ  (both A and B have shape (N, N))
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch * head

    batch = pid_hb // H
    head  = pid_hb % H

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # ---- load A (TILE_M × TILE_K) ----
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- load Bᵀ (TILE_K × TILE_N) ----
        # B is stored as (batch, head, i, j).  We need B[j,k] → treat
        # j as column, k as row ⇒ load with k as first dim.
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k index
                  + offs_n[None, :] * stride_b_i)  # j index
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # ---- dot product over K ----
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32/float16)
        - mask         : torch.Tensor [B, N, N]     (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helpers for weight lookup
    # ------------------------------------------------------------------
    def w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False) → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))   # (dim, hidden_dim)ᵀ
    right = F.linear(x, w("right_proj.weight"))

    # optional mask (broadcast over hidden_dim)
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(dtype)            # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) → same shape
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy N³ contraction via Triton
    #    Rearrange to (B, H, N, N) so that each kernel instance works on a
    #    single (batch, head) matrix pair.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B,H,N,N]

    # strides in elements (not bytes) – required by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)  (we will transpose on‑the‑fly)
    c_s = out_t.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # launch configuration
    # ------------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # i‑tiles
        triton.cdiv(N, TILE_N),          # j‑tiles
        B * hidden_dim,                  # batch * heads
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=8,          # 8 warps gives full SM‑90 occupancy for 64×64 tiles
    )

    # back to original layout [B,N,N,hidden_dim]
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    # out_gate has shape [B,N,N,hidden_dim] – broadcast over dim
    out = out * out_gate
    out = F.linear(out, w("to_out.weight"))   # (hidden_dim, dim)ᵀ

    # Ensure float32 output (the reference uses float32 throughout)
    return out.to(dtype)
