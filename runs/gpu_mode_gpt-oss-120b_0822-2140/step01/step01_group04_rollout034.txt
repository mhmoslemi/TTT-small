"""
TriMul “outgoing” (AlphaFold3) forward pass.

The only O(N³) operation is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

which can be seen as a batched matrix‑multiply per hidden dimension d:
    (B, H, N, N) @ (B, H, N, N)ᵀ → (B, H, N, N)

where H = hidden_dim.  We implement (1) with a fused Triton kernel.
All cheap operations (LayerNorm, Linear, sigmoid gating, final LayerNorm/Linear)
stay in PyTorch.  The kernel works in FP16 for bandwidth and accumulates in FP32,
matching the reference implementation.

The host function ``custom_kernel`` follows the required signature:
    (input_tensor, mask, weights, config) → output_tensor
and returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ  (A,B shape: B×H×N×N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    N, H,                         # problem size (seq_len, hidden_dim)
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    TILE_M: tl.constexpr,
    TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
    BLOCK_SIZE_K: tl.constexpr,   # K‑loop block (usually same as TILE_K)
):
    """
    Compute, for each (batch, head) pair:
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]
    i.e. A @ Bᵀ   (both A and B are N×N matrices)
    """

    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch*head index

    batch = pid_hb // H
    head  = pid_hb %  H

    # ------------------- offsets for this tile -------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Iterate over K dimension in blocks of BLOCK_SIZE_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_SIZE_K)):
        offs_k = k * BLOCK_SIZE_K + tl.arange(0, BLOCK_SIZE_K)
        mask_k = offs_k < N

        # ----- load A[b, h, i, k] -----
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ----- load Bᵀ : B[b, h, j, k]  (notice k is the first dim) -----
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k dimension
                  + offs_n[None, :] * stride_b_i)  # j dimension
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot‑product over K (FP16 × FP16 → FP32 accumulation)
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N) → (TILE_M,TILE_N)

    # ----------------------- store result -------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)

    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (bool/float, may be all‑ones)
        - weights      : dict of torch.Tensor, model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (True if mask can be ignored)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # helpers for weight lookup (biases are stored with the same key name)
    # ------------------------------------------------------------------
    w = lambda name: weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    eps = 1e-6
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = F.linear(x, w("left_proj.weight"))      # [B,N,N,H]
    right = F.linear(x, w("right_proj.weight"))     # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; we broadcast on hidden dim
        mask_f = mask.to(dtype).unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction using Triton
    #    Rearrange to (B, H, N, N) for contiguous memory layout.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in *elements* (not bytes) – required by the kernel
    a_s = left_t.stride()   # (batch, head, i, k)
    b_s = right_t.stride()  # (batch, head, i, j)  – we load transposed inside kernel
    c_s = out_t.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # Triton launch configuration
    #   TILE_M/N/K are tuned for Hopper (SM90) – 64 works well for N up to 1024.
    # ------------------------------------------------------------------
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64          # also used as BLOCK_SIZE_K inside the kernel

    grid = (
        triton.cdiv(N, TILE_M),               # rows
        triton.cdiv(N, TILE_N),               # cols
        B * hidden_dim,                       # batch*head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        BLOCK_SIZE_K=TILE_K,
        num_warps=4,          # 4 warps gives good occupancy on H200
        num_stages=2,         # double‑buffering
    )

    # back to original layout (B,N,N,H)
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, w("to_out.weight"))    # [B,N,N,dim]

    return out.to(dtype)
