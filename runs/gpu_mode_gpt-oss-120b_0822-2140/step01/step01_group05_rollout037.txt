"""
TriMul “outgoing” (AlphaFold‑3) – fast Triton‑fused batched GEMM

*   The expensive part of the forward pass is the three‑dimensional
    einsum   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
    For every hidden channel *d* this is a matrix multiplication
    (N×N) @ (N×N)ᵀ .  We treat each hidden channel as an extra batch dimension
    (B·H) and compute all those GEMMs in a single Triton kernel.

*   All other operations (LayerNorm, linear projections, gating,
    final LayerNorm + linear) stay in PyTorch – they are cheap compared
    with the O(B·H·N³) work of the GEMM.

*   Mixed‑precision:  left/right are cast to FP16 before the kernel,
    the kernel loads FP16 values and accumulates in FP32 (Tensor‑Core
    friendly on Hopper).  The result is stored as FP32 so the rest of
    the network works in FP32.

*   The kernel tiles the output matrix with
    BLOCK_M × BLOCK_N tiles and walks the reduction dimension K with
    BLOCK_K‑wide chunks.  The tiling parameters are chosen adaptively
    based on the sequence length to keep occupancy high for the whole
    benchmark range (256 → 1024).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (B*H) independent GEMMs   A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def outgoing_gemm_kernel(
    A_ptr,                     # fp16, shape (B*H, M, K)
    B_ptr,                     # fp16, shape (B*H, N, K)   (the second matrix is NOT transposed,
                                   # we index it as (j, k) and treat K as the reduction dim)
    C_ptr,                     # fp32, shape (B*H, M, N)
    M, N, K,                   # matrix sizes (M=N=seq_len, K=seq_len)
    stride_Ab, stride_Am, stride_Ak,   # strides for A  (batch, M, K)
    stride_Bb, stride_Bm, stride_Bk,   # strides for B  (batch, N, K)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C  (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  program IDs
    # ------------------------------------------------------------------
    pid_b = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------
    #  tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    #  pointers that will be incremented in the K‑loop
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_n[None, :] * stride_Bm + offs_k[:, None] * stride_Bk

    # ------------------------------------------------------------------
    #  accumulator in FP32 (Tensor‑Core accumulation)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  main reduction loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    #  write back
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor  [B, N, N, dim]   (float32, CUDA)
        mask    – torch.Tensor  [B, N, N] or None
        weights – dict of model parameters (float32, CUDA)
        config  – dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]          # called hidden_dim in the paper
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm over the original channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # [B, N, N, 1]
        left  = left  * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to FP16 for the GEMM
    # --------------------------------------------------------------
    left_h  = left.half()
    right_h = right.half()

    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_t  = left_h.permute(0, 3, 1, 2).contiguous()
    right_t = right_h.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_t.view(B * hidden, N, N)
    right_flat = right_t.view(B * hidden, N, N)

    # output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # --------------------------------------------------------------
    # 6. Choose tiling parameters (heuristic for the given N)
    # --------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden,                     # pid_b
        triton.cdiv(N, BLOCK_M),        # pid_m
        triton.cdiv(N, BLOCK_N),        # pid_n
    )

    # ------------------------------------------------------------------
    # 7. Strides (batch, row, col) for each tensor
    # ------------------------------------------------------------------
    a_batch, a_m, a_k = left_flat.stride()
    b_batch, b_m, b_k = right_flat.stride()
    c_batch, c_m, c_n = out_flat.stride()

    outgoing_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        a_batch, a_m, a_k,
        b_batch, b_m, b_k,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 9. Output LayerNorm (hidden) + gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 10. Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
