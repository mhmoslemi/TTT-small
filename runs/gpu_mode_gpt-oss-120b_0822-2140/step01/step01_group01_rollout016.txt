"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is exactly a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We therefore launch a *batched* GEMM kernel that computes M = B·hidden_dim
independent (N,N) @ (N,N)ᵀ products.  The kernel works in fp16 for the
inputs (to cut bandwidth) while the accumulator is fp32 (required for
numerical stability).  All other operations (LayerNorm, linear projections,
sigmoids, masking) stay in PyTorch because they are cheap compared with the
O(N³) work.

The kernel is written for the Hopper SM90 (NVIDIA H200) – block sizes are
chosen to give good occupancy on that architecture.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr,               # half *, shape (M, N, N)
    b_ptr,               # half *, shape (M, N, N)
    c_ptr,               # float*, shape (M, N, N)
    N,                   # seq_len
    stride_ah, stride_ai, stride_ak,   # A strides (M,N,N)
    stride_bh, stride_bj, stride_bk,   # B strides (M,N,N)
    stride_ch, stride_ci, stride_cj,   # C strides (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One GEMM per (batch * hidden) slice.
    The computation is:
        C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k]          (B is accessed transposed)
    """

    pid_m = tl.program_id(0)          # block row   i
    pid_n = tl.program_id(1)          # block col   j
    pid_z = tl.program_id(2)          # flattened (batch * hidden) index

    # ------------------------------------------------------------------
    # Offsets for this block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to the slice m = pid_z
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (the summation axis)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k] : (BLOCK_M, BLOCK_K)   stored as fp16
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # promote to fp32

        # Bᵀ[k,j] : (BLOCK_K, BLOCK_N)   we load B[j,k] and treat it as transposed
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk   # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj   # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM wrapper.

    Arguments
    ---------
    left, right : torch.Tensor of shape (B, H, N, N), dtype float16 or float32
                  (they must have the same dtype)

    Returns
    -------
    out : torch.Tensor of shape (B, H, N, N), dtype float32
    """
    B, H, N, _ = left.shape
    M = B * H                               # flatten batch * hidden

    # flatten to (M, N, N) – keep the same dtype as the inputs
    left_f  = left.reshape(M, N, N).contiguous()
    right_f = right.reshape(M, N, N).contiguous()

    # output will be fp32
    out = torch.empty_like(left_f, dtype=torch.float32, device=left.device)

    # strides as int64
    stride_ah, stride_ai, stride_ak = left_f.stride()
    stride_bh, stride_bj, stride_bk = right_f.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # Tunable block sizes – empirically good for sm_90
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch*hidden)
    )

    trimul_gemm_kernel[grid](
        left_f, right_f, out,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy contraction is performed by a custom Triton GEMM kernel.
    All remaining operations are done in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)  dim → hidden_dim
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)        # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B, N, N) → (B, N, N, 1)
        mask = mask.to(device).unsqueeze(-1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, hidden, N, N) layout expected by the kernel.
    #    Cast to fp16 for the kernel to reduce memory traffic.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # cast to half – the kernel will accumulate in fp32
    left_h  = left_perm.to(torch.float16)
    right_h = right_perm.to(torch.float16)

    out_perm = trimul_gemm(left_h, right_h)            # (B, H, N, N), fp32

    # back to (B, N, N, hidden)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    return out.to(torch.float32)
