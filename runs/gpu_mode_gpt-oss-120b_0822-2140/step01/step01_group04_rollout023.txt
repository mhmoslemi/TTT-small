"""
TriMul “outgoing” kernel (AlphaFold3).

The expensive part of the module is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden dimension `d`.
We move this contraction to Triton while keeping the cheap LayerNorm /
Linear / gating ops in PyTorch.  The kernel works on tensors of shape
(B, H, N, N) (contiguous, row‑major) and computes, for each (b, h),

    C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]   (i.e. A @ Bᵀ)

Inputs are cast to FP16 for bandwidth, the accumulation is FP32, and the
result is written back in FP32 to match the reference implementation.
"""


import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (B, H, N, N) @ (B, H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                      # pointers
    N, H,                                     # problem sizes
    stride_a_batch, stride_a_head,
    stride_a_i, stride_a_k,                   # A strides (batch, head, i, k)
    stride_b_batch, stride_b_head,
    stride_b_i, stride_b_j,                   # B strides (batch, head, i, j)
    stride_c_batch, stride_c_head,
    stride_c_i, stride_c_j,                   # C strides (batch, head, i, j)
    TILE_M: tl.constexpr, TILE_N: tl.constexpr,
    TILE_K: tl.constexpr,
):
    """
    Compute for each (b, h):
        C[b, h, i, j] = Σ_k A[b, h, i, k] * B[b, h, j, k]

    A and B are stored in row‑major (C‑contiguous) layout.
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch * head

    batch = pid_hb // H
    head = pid_hb % H

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * TILE_M + tl.arange(0, TILE_M)
    offs_n = pid_n * TILE_N + tl.arange(0, TILE_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((TILE_M, TILE_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main K loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, TILE_K)):
        offs_k = k * TILE_K + tl.arange(0, TILE_K)
        mask_k = offs_k < N

        # A : (i, k)
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + head  * stride_a_head
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Bᵀ : we need B[j, k] → load as (k, j)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + head  * stride_b_head
                  + offs_k[:, None] * stride_b_j   # k dimension first
                  + offs_n[None, :] * stride_b_i)  # j dimension second
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot over K
        acc += tl.dot(a, b)   # (TILE_M,TILE_K) @ (TILE_K,TILE_N)

    # --------------------------------------------------------------
    # store result
    # --------------------------------------------------------------
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + head  * stride_c_head
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N]  (may be all ones)
        - weights      : dict of torch.Tensors containing all linear &
                         layer‑norm parameters (biases are included)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", True):
        # mask is broadcast over hidden_dim
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Heavy O(N³) contraction using Triton
    #    reshape to (B, H, N, N) and make contiguous
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_t = torch.empty_like(left_t, dtype=dtype)    # [B, H, N, N]

    # strides in element units (convert to int64 for Triton)
    a_s = left_t.stride()    # (batch, head, i, k)
    b_s = right_t.stride()   # (batch, head, i, j) – we will load transposed
    c_s = out_t.stride()     # (batch, head, i, j)

    # tile sizes – chosen to give good occupancy on Hopper GPUs
    TILE_M = 64
    TILE_N = 64
    TILE_K = 64

    grid = (
        triton.cdiv(N, TILE_M),          # i tiles
        triton.cdiv(N, TILE_N),          # j tiles
        B * hidden_dim,                  # batch * head
    )

    _trimul_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        N,
        hidden_dim,
        a_s[0], a_s[1], a_s[2], a_s[3],
        b_s[0], b_s[1], b_s[2], b_s[3],
        c_s[0], c_s[1], c_s[2], c_s[3],
        TILE_M=TILE_M,
        TILE_N=TILE_N,
        TILE_K=TILE_K,
        num_warps=4,                     # 4 warps works well for 64×64 tiles
    )

    # back to original layout (B, N, N, hidden_dim)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, gating and linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                       # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out.to(dtype)
