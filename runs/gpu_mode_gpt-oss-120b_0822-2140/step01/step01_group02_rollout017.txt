```python
"""
TriMul “outgoing” forward pass (no gradients).

The dominant work is the 3‑D contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be written as a batched GEMM:
    left_slice  : (N, N)   = left[b, :, :, d]
    right_T     : (N, N)   = right[b, :, :, d].transpose(-2,-1)
    out_slice   = left_slice @ right_T

We fuse *all* (batch × hidden_dim) GEMMs in one Triton kernel.
All other operations (LayerNorm, Linear projections, sigmoid gates,
masking, final projection) are performed in PyTorch – they are cheap
compared with the O(N³·H) work.

The kernel works in FP16 internally (the model is trained in FP32 but the
contraction is tolerant to reduced precision) and accumulates in FP32.
An `@triton.autotune` decorator explores a small space of tile sizes to
pick the fastest launch on Hopper/H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (batched GEMM, A and B of shape [B, M, K])
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 64,  "BLOCK_K": 32, "num_warps": 4}, num_stages=2),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32, "num_warps": 4}, num_stages=2),
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 128, "BLOCK_K": 32, "num_warps": 4}, num_stages=2),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 64,  "BLOCK_K": 32, "num_warps": 4}, num_stages=2),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def batched_gemm_kernel(
    a_ptr,                # *float16
    b_ptr,                # *float16
    c_ptr,                # *float16
    M, N, K,              # matrix sizes (int)
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, col, row)   <-- note bk/bn order!
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed.

    Computes, for every batch `b`:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]
    which is equivalent to A @ Bᵀ.
    """
    pid_batch = tl.program_id(2)               # batch dimension (B·H)
    pid_m = tl.program_id(0)                   # block row
    pid_n = tl.program_id(1)                   # block col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in A
    a_ptrs = a_ptr + pid_batch * stride_az \
                     + offs_m[:, None] * stride_am \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # For Bᵀ we need elements (k, n) of the original (batch, N, K)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                     + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main accumulation loop over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # mask for the K‑dimension of this tile
        mask_k = (cur_k + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                 # FP16×FP16 → FP32

        # move pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as FP16 to reduce memory traffic)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for FP16 tensors.

    a, b : float16 tensors of shape (B, N, N) (contiguous)
    returns c of the same shape & dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, N, K = a.shape
    assert K == N

    c = torch.empty_like(a)

    # Strides in **elements** (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()  # (batch, N, K)

    # For the transposed view we need (batch, col, row) strides:
    #   column index -> original K dimension, row index -> original N dimension
    # So we pass stride_bz (batch), stride_bk (col), stride_bn (row)
    stride_bz_t = stride_bz
    stride_bk_t = stride_bk
    stride_bn_t = stride_bn

    stride_cz, stride_cm, stride_cn = c.stride()

    grid = (
        triton.cdiv(N, 128),   # we will let autotune pick actual tile size
        triton.cdiv(N, 128),
        B,
    )
    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,          # M
        N,          # N
        N,          # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz_t,
        stride_bk_t,
        stride_bn_t,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=128,   # dummy; real value overridden by autotune config
        BLOCK_N=128,
        BLOCK_K=32,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Required entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a fused Triton GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) – may be dummy
            - weights      : dict of torch.Tensors (float32) containing every
                             linear / layer‑norm weight/bias used below
            - config       : dict with keys
                * "dim"          – model dimension (C)
                * "hidden_dim"   – hidden dimension (C_z)
                * "nomask" (bool) – if True the mask should be ignored

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)      # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Bring hidden dimension into the batch axis and cast to fp16
    # ------------------------------------------------------------------
    # shape -> (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # merge batch*hidden so each GEMM is independent
    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out_fp16 = left @ rightᵀ
    # ------------------------------------------------------------------
    out_fp16 = batched_gemm_fp16(left, right)          # (B*hidden, N, N)

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden) and cast to fp32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Post‑projection LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # out_gate already has shape (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 9. Final linear back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
