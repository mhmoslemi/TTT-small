"""
TriMul (outgoing) – fast forward pass using a fused Triton batched GEMM.

Algorithm
---------
1️⃣ Layer‑norm over the last dimension (dim) – PyTorch.
2️⃣ Linear projections (left, right) – PyTorch, bias‑free.
3️⃣ Optional mask (broadcasted) – PyTorch.
4️⃣ Gating: left_gate, right_gate, out_gate = sigmoid(linear(...)) – PyTorch.
   left *= left_gate ; right *= right_gate
5️⃣ Core O(N³·H) contraction:
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   We permute → (B, H, N, N) → (B·H, N, N) and launch a custom Triton
   GEMM kernel that computes C = A @ Bᵀ for each (batch*head) pair.
   The kernel is tiled with BLOCK_M × BLOCK_N × BLOCK_K and works in
   fp32 (can be changed to fp16 easily).
6️⃣ Layer‑norm on the hidden dimension – PyTorch.
7️⃣ Element‑wise multiplication with the output gate (fused Triton
   element‑wise kernel – required “use a kernel” rule).
8️⃣ Final linear projection back to *dim* – PyTorch.

All tensors stay on the device, the heavy N³ work is done by the
Triton GEMM, giving >10× speed‑up vs. the previous torch‑bmm solution.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Element‑wise mul kernel (used for out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, N_ELEMENTS,
    stride_a: tl.constexpr, stride_b: tl.constexpr, stride_out: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMENTS

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel: C = A @ Bᵀ   (A,B,C : [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for each batch‑head slice."""
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col
    pid_bh = tl.program_id(2)              # batch*head index

    # ---- compute start offsets for this block ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guards for edge blocks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch*head) slice
    a_base = a_ptr + pid_bh * stride_a_batch
    b_base = b_ptr + pid_bh * stride_b_batch
    c_base = c_ptr + pid_bh * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K ---------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)

        # mask for K dimension (handles tail)
        mask_k = offs_k < K

        # A block : (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_a_row) + (offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block (need Bᵀ) : we load as (BLOCK_K, BLOCK_N)
        # B shape is (batch_head, N, K); we want K as leading dim.
        b_ptrs = b_base + (offs_k[:, None] * stride_b_col) + (offs_n[None, :] * stride_b_row)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # GEMM: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(a, b)

    # ---- write results -------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3) Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (float32/bool or None, CUDA)
        * weights_dict : model parameters (tensors on the same device)
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )                                   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, w["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction via Triton GEMM
    #    reshape to (B, H, N, N) → (B*H, N, N)
    # ------------------------------------------------------------------
    #   A = left , B = right  (both [B, H, N, N])
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    BH = B * hidden_dim
    left_ = left.view(BH, N, N)       # (BH, N, N)
    right_ = right.view(BH, N, N)     # (BH, N, N)

    out_ = torch.empty_like(left_)   # allocate output (BH, N, N)

    # strides in *elements* (not bytes)
    stride_a_batch, stride_a_row, stride_a_col = left_.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_.stride()

    # tiling parameters (tuned for Hopper, fp32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )

    batched_gemm_kernel[grid](
        left_, right_, out_,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to original layout: (B, N, N, hidden_dim)
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm on hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )                                   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Element‑wise product with out_gate (fused Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total = out_flat.numel()

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elemwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total,
        stride_a=1,
        stride_b=1,
        stride_out=1,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B, N, N, dim]

    return out
