assistantfinal**TriMul “outgoing” forward pass (AlphaFold‑3)**  
The heavy part of the module is the triple‑wise contraction  

\[
\text{out}_{b,i,j,h}= \sum_{k=0}^{N-1}
      \text{left}_{b,i,k,h}\; \text{right}_{b,j,k,h}
\]

which can be seen as **B·H independent matrix‑multiplications**  
\(A_{b,h}= \text{left}_{b,:, :,h}\) (shape N×N) and  
\(B_{b,h}= \text{right}_{b,:, :,h}\) (shape N×N).  
We compute \(A_{b,h}\;B_{b,h}^{\mathsf T}\) with a tiled GEMM written in
Triton.  All other operations (layer‑norms, linear projections, gates,
masking) stay in PyTorch – they are cheap compared to the \(O(N^{3})\)
inner loop.

The kernel works for any batch size ≤ 2, sequence length up to 1024 and
hidden dimension 128 (the benchmark configuration).  Mixed‑precision
(fp16 loads, fp32 accumulation) reduces memory bandwidth while keeping the
final result in FP32.

```python
"""
Triton‑accelerated forward pass of the TriMul “outgoing” operator
(AlphaFold‑3).

Algorithm
---------
1️⃣ Layer‑norm the input (last dim = dim).
2️⃣ Linear projections → left / right (shape B×N×N×H).
3️⃣ (Optional) apply the pair‑wise mask.
4️⃣ Gated‑linear transforms (sigmoid gates) on left / right and compute an
   output‑gate.
5️⃣ Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM: for each (b,h) we compute
      out_slice = left_slice @ right_slice.T.
   Implemented with a tiled Triton kernel (fp16 loads, fp32 acc).
6️⃣ Post‑processing: Layer‑norm → out‑gate → final linear projection back
   to the original channel dimension.

The kernel is written for NVIDIA Hopper (SM90) – block sizes are tuned for
the H200 but work on any recent GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr,                     # *float16  (batch, M, K)   → left
    B_ptr,                     # *float16  (batch, N, K)   → right (accessed transposed)
    C_ptr,                     # *float32  (batch, M, N)   → output
    M, N, K,                   # matrix dimensions (M=N=N_seq, K=N_seq)
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (note: Bk = col‑stride, Bn = row‑stride)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of GEMMs A @ Bᵀ with fp16 inputs and fp32 accumulation.
    """
    pid_b = tl.program_id(2)          # batch * hidden_dim index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # ---- block start indices ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- boundary masks ----
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- compute pointers for the current block ----
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk      # k‑dimension (original column)
        + offs_n[None, :] * stride_Bn      # n‑dimension (original row)
    )

    # ---- accumulator in fp32 ----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- main K‑loop ----
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        ).to(tl.float32)
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)                     # (M×K)·(K×N) → (M×N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ---- write result ----
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” module using a Triton kernel for the
    O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        - mask         : torch.Tensor, shape [B, N, N]       (bool/float)
        - weights      : dict of torch.Tensor parameters (float32)
        - config       : dict with keys
              "dim"        – input channel dimension C
              "hidden_dim"– hidden dimension H
              "nomask"    – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # weight tensors (all stored as float32)
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # F.linear expects (..., in_features).  Input shape is (..., dim)
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton (batched GEMM A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # B=batch, N=seq_len, H=hidden_dim

    # reshape to (batch*H, N, N) and make contiguous for clean strides
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # fp16 inside the kernel for bandwidth, fp32 accumulator
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # ---- strides in element units (not bytes) ----
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()   # (batch, row, col)
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()  # (batch, row, col)

    # For the transposed view of B we need:
    #   k‑step  = original column stride
    #   n‑step  = original row stride
    stride_Bk = stride_Bc
    stride_Bn = stride_Br

    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # ---- tuning parameters (good for Hopper, works for all test sizes) ----
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H,                     # batch * hidden
    )

    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (layer‑norm → out‑gate → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dim C

    return out
```
