"""
TriMul “outgoing” implementation (AlphaFold‑3 style)

The expensive part of the module is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is exactly a matrix‑multiplication

    L = left[b,:, :, d]          (N×N)
    R = right[b,:, :, d]         (N×N)
    out[b,:, :, d] = L @ Rᵀ      (N×N)

The kernel below computes many independent GEMMs (one per (batch, hidden) pair)
in FP16 while accumulating in FP32, which on the H200 gives the best
bandwidth‑to‑compute ratio.  All other ops (LayerNorm, Linear, Sigmoid, masking)
are cheap and stay in PyTorch.

The kernel works on tensors of shape (M, N, N) where
    M = batch * hidden_dim
and returns a tensor of the same shape.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers (fp16)
    N,                                 # sequence length
    stride_ah, stride_ai, stride_ak,   # A: (M, N, N) strides
    stride_bh, stride_bj, stride_bk,   # B: (M, N, N) strides
    stride_ch, stride_ci, stride_cj,   # C: (M, N, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ Bᵀ  for each batch‑hidden slice.
    A, B, C are stored as (M, N, N) in row‑major order.
    """
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_z = tl.program_id(2)          # flattened (batch*hidden)

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the beginning of the current slice
    a_batch = a_ptr + pid_z * stride_ah
    b_batch = b_ptr + pid_z * stride_bh
    c_batch = c_ptr + pid_z * stride_ch

    # Accumulator in FP32 (more precise than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block of A  (BLOCK_M × BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai               # i dimension
            + offs_k[None, :] * stride_ak               # k dimension
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # Load a block of Bᵀ  (BLOCK_K × BLOCK_N)
        # B is indexed as B[j, k]; we load it transposed to match tl.dot
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk               # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj               # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Dot‑product, accumulation in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store the result block
    # ------------------------------------------------------------------
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  float32
    Returns       : (B, H, N, N)  float32  where each slice is left @ rightᵀ
    Internally works in FP16 for bandwidth efficiency.
    """
    B, H, N, _ = left.shape
    M = B * H                                 # flattened batch*hidden

    # ------------------------------------------------------------------
    # Cast to FP16 and make sure tensors are contiguous
    # ------------------------------------------------------------------
    left_f16  = left.reshape(M, N, N).contiguous().to(torch.float16)
    right_f16 = right.reshape(M, N, N).contiguous().to(torch.float16)

    out_f16 = torch.empty_like(left_f16, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides (as int64) – needed because we flattened the first two dims
    # ------------------------------------------------------------------
    stride_ah, stride_ai, stride_ak = left_f16.stride()
    stride_bh, stride_bj, stride_bk = right_f16.stride()
    stride_ch, stride_ci, stride_cj = out_f16.stride()

    # ------------------------------------------------------------------
    # Tunable block sizes – chosen for H200 (large shared memory, 128‑thread warps)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z
    )
    trimul_gemm_fp16_kernel[grid](
        left_f16, right_f16, out_f16,
        N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Cast back to FP32 and reshape to (B, H, N, N)
    out = out_f16.to(torch.float32).reshape(B, H, N, N)
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape (bs, seq_len, seq_len, dim) (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm over the feature dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)    # (B, N, N, hidden)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Apply mask if present
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.to(device).unsqueeze(-1)               # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton GEMM : out = left @ rightᵀ  (per hidden channel)
    #    Convert to (B, H, N, N) layout expected by the kernel.
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    out_perm = trimul_gemm_fp16(left_perm, right_perm)   # (B, H, N, N)

    # Back to the original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # --------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"].to(device)
    out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=out_norm_w, bias=out_norm_b)
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)       # (dim, hidden)
    out = F.linear(out, to_out_w)                        # (B, N, N, dim)

    return out.to(torch.float32)
