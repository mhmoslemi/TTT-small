"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm over the last dimension (C).
2. Linear projections → left / right tensors (shape B×N×N×H).
3. (Optional) apply the pairwise mask on the projected tensors.
4. Compute three sigmoid gates (left, right, out) and multiply them element‑wise.
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM `left_slice @ right_sliceᵀ` for every (b, h).
   The GEMM is performed by a Triton kernel using fp16 loads and fp32
   accumulation (mixed‑precision for bandwidth).  Tile sizes are tuned for
   Hopper (SM90): BLOCK_M=N=64, BLOCK_K=128.
6. Layer‑norm on the hidden dimension, apply the out‑gate, and a final
   linear projection back to C.
The function returns a float‑32 tensor of shape [B, N, N, C].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [batch, M, K])  ->  C : [batch, M, N]
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    A_ptr,                      # *fp16   (batch, M, K)
    B_ptr,                      # *fp16   (batch, N, K)   (will be read as K×N)
    C_ptr,                      # *fp32   (batch, M, N)
    M, N, K,                    # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (B is stored as [batch,N,K])
    stride_Cb, stride_Cm, stride_Cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(2)          # batch*head dimension
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # -------------------- block offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current tile
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float32)                     # (M × K)

        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)                     # (K × N)

        acc += tl.dot(a, b)                   # (M × N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # -------------------- write back --------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                 # input tensor [B, N, N, C] (float32)
        torch.Tensor,                 # mask tensor   [B, N, N]   (bool/float)
        Dict[str, torch.Tensor],      # weights dict (float32)
        Dict,                         # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator with a Triton‑accelerated
    batched GEMM kernel.
    """
    # -------------------- unpack --------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------- weight aliases --------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # -------------------- 1️⃣ Layer‑norm --------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -------------------- 2️⃣ Linear projections (no bias) --------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # -------------------- 3️⃣ Mask (optional) --------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -------------------- 4️⃣ Gating --------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------- 5️⃣ Core contraction (batched GEMM) --------------------
    B, N, _, H = left.shape                     # H == hidden_dim

    # reshape to (B*H, N, N) and make contiguous for stride correctness
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 inside the kernel (bandwidth) – keep fp32 accumulator
    left_fp16 = left_view.to(torch.float16)
    right_fp16 = right_view.to(torch.float16)

    # output buffer (fp32)
    out_buf = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides in *elements* (not bytes)
    stride_Ab, stride_Am, stride_Ak = left_fp16.stride()
    stride_Bb, stride_Br, stride_Bc = right_fp16.stride()
    # B is stored as (batch, N, N) → for Bᵀ we need:
    stride_Bk = stride_Bc          # step along K (original column)
    stride_Bn = stride_Br          # step along N (original row)
    stride_Cb, stride_Cm, stride_Cn = out_buf.stride()

    # -------------------- launch parameters (tuned for SM90) --------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H                      # batch * head
    )

    _batched_gemm_T_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_buf.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------- 6️⃣ Post‑processing --------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # back to original dimension C

    return out
