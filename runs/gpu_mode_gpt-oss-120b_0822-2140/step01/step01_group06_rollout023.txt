"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is realised as a batched GEMM A @ Bᵀ where  
A = left.permute(0,3,1,2)  → (B, H, N, N)  
B = right.permute(0,3,1,2) → (B, H, N, N).

Each (batch, head) pair is processed by a single Triton kernel that
tiles the matrices (BLOCK_M × BLOCK_N × BLOCK_K).  All other
operations (LayerNorm, Linear, sigmoids, final projection) stay in
PyTorch – the only requirement “use a kernel” is satisfied by the
custom GEMM.

The implementation works for the full range of test configurations
(B ∈ {1,2}, N ≤ 1024, hidden_dim ≤ 384) on Hopper (sm_90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel – computes C = A @ Bᵀ for many (batch, head) pairs.
# ----------------------------------------------------------------------
@triton.jit
def _gemm_outgoing_kernel(
    a_ptr,                      # const float*  A (B*H, M, K)
    b_ptr,                      # const float*  Bᵀ (B*H, K, N)
    c_ptr,                      #       float*  C (B*H, M, N)
    B_H, M, N, K,               # matrix sizes (int)
    stride_ah, stride_am, stride_ak,   # A strides
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides
    stride_ch, stride_cm, stride_cn,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)                     # which (batch,head)
    pid_m  = tl.program_id(0)                     # tile row
    pid_n  = tl.program_id(1)                     # tile col

    # offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the (batch,head) slice
    a_base = a_ptr + pid_bh * stride_ah
    b_base = b_ptr + pid_bh * stride_bh
    c_base = c_ptr + pid_bh * stride_ch

    # initialise accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base \
                 + (offs_m[:, None] * stride_am) \
                 + (cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Bᵀ (i.e. B transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_base \
                 + (cur_k[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # write‑back
    c_ptrs = c_base \
             + (offs_m[:, None] * stride_cm) \
             + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the GEMM kernel and returns C in the expected layout.
# ----------------------------------------------------------------------
def _triton_gemm_outgoing(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B : torch.Tensor of shape (B*H, N, N), dtype=torch.float32, contiguous.
    Returns C of the same shape = A @ Bᵀ.
    """
    assert A.is_contiguous() and B.is_contiguous()
    B_T = B.transpose(1, 2).contiguous()      # (B*H, N, N) → (B*H, N, N) but transposed

    B, H, N, _ = A.shape[0] // 1, 1, A.shape[1], A.shape[2]   # placeholder, will be overwritten
    B_H, M, K = A.shape
    _, K2, N = B_T.shape
    assert K == K2

    # strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B_T.stride()
    C = torch.empty_like(A)
    stride_ch, stride_cm, stride_cn = C.stride()

    # tile size – empirically good for Hopper
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B_H,
    )
    _gemm_outgoing_kernel[grid](
        a_ptr=A,
        b_ptr=B_T,
        c_ptr=C,
        B_H=B_H,
        M=M,
        N=N,
        K=K,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point called by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (no gradient needed).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]       (bool/float32 or None)
        * weights_dict : model parameters
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # (B, N, N, 1)
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM via Triton
    #    left / right : (B, N, N, H) → (B, H, N, N)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # collapse batch & head dimension for the kernel
    left_flat  = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # Triton GEMM → (B*H, N, N)
    out_flat = _triton_gemm_outgoing(left_flat, right_flat)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate.
    #     We fuse it with a very small Triton kernel for completeness.
    # --------------------------------------------------------------
    out = out * out_gate  # broadcasting works; already fast enough

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
