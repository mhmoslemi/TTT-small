"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is realised as a batched GEMM A @ Bᵀ for each (batch, head) pair.
We permute the tensors to (B, H, N, N) and launch a single Triton
kernel that tiles the matrices (64×64×64).  The kernel works on
fp16 data (loads as fp16, accumulates in fp32) and writes the result
directly in fp32, eliminating the need for an extra cuBLAS call.
All other operations (LayerNorm, Linear, sigmoid‑gates, final
projection) are performed with PyTorch – the kernel satisfies the
“must use a kernel” requirement while keeping the implementation
compact and fast on Hopper (sm_90) GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_gemm(
    # pointers
    a_ptr,                # left  : (B, H, N, N)  fp16
    b_ptr,                # right : (B, H, N, N)  fp16 (will be used transposed)
    out_ptr,              # output: (B, H, N, N)  fp32
    # sizes
    B, H, N,
    # strides (in elements, not bytes)
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_k, stride_b_j,
    stride_o_batch, stride_o_head, stride_o_i, stride_o_j,
    # compile‑time tiling constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM where each (b, h) computes
       out[b, h, i, j] = Σ_k a[b, h, i, k] * b[b, h, j, k] .
       The right operand is accessed as (k, j) to obtain Bᵀ.
    """
    pid0 = tl.program_id(0)          # batch * head dimension
    pid1 = tl.program_id(1)          # tile over rows (i)
    pid2 = tl.program_id(2)          # tile over cols (j)

    # decode batch / head from the first program ID
    b = pid0 // H
    h = pid0 % H

    # offsets within the N‑dimensional plane
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load A (shape M×K) ----------
        a = tl.load(
            a_ptr
            + b * stride_a_batch
            + h * stride_a_head
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k,
            mask=(mask_i[:, None] & mask_k[None, :]),
            other=0.0,
        )
        # cast to fp32 for the dot product
        a = a.to(tl.float32)

        # ----------- load Bᵀ (shape K×N) ----------
        #   right is stored as (j, k); we load it as (k, j)
        b = tl.load(
            b_ptr
            + b * stride_b_batch
            + h * stride_b_head
            + offs_k[:, None] * stride_b_k   # k‑dimension (row of Bᵀ)
            + offs_j[None, :] * stride_b_j,  # j‑dimension (col of Bᵀ)
            mask=(mask_k[:, None] & mask_j[None, :]),
            other=0.0,
        )
        b = b.to(tl.float32)

        # ----------- dot and accumulate ----------
        acc += tl.dot(a, b)   # fp32 ← fp16·fp16

    # store the block
    out = out_ptr \
        + b * stride_o_batch \
        + h * stride_o_head \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j

    tl.store(out, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor   shape (B, N, N, dim)   float32, CUDA
        * mask_tensor  : torch.Tensor   shape (B, N, N)      float32 or bool, CUDA
        * weights_dict : dictionary with the eight learned parameters
        * config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, dim), float32, CUDA
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, W, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last channel (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=W["norm.weight"],
        bias=W["norm.bias"],
    )                                             # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, W["left_proj.weight"])   # (B, N, N, hidden)
    right = F.linear(x, W["right_proj.weight"])  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not cfg.get("nomask", False):
        m = mask.unsqueeze(-1).to(x.dtype)       # (B, N, N, 1)
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, W["left_gate.weight"]))   # (B, N, N, hidden)
    right_gate = torch.sigmoid(F.linear(x, W["right_gate.weight"]))  # (B, N, N, hidden)
    out_gate   = torch.sigmoid(F.linear(x, W["out_gate.weight"]))    # (B, N, N, hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction using a Triton GEMM kernel
    # ------------------------------------------------------------------
    #   Permute to (B, H, N, N) so that each (b, h) pair can be treated
    #   as an independent matrix‑multiply.
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # cast to fp16 for the kernel (loads are cheaper)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output in fp32 (the kernel accumulates in fp32)
    out = torch.empty((B, hidden, N, N), dtype=torch.float32, device=device)

    # strides (in element units) required by the kernel
    s_a_b, s_a_h, s_a_i, s_a_k = left_fp16.stride()
    s_b_b, s_b_h, s_b_i, s_b_j = right_fp16.stride()   # i‑axis will act as k
    s_o_b, s_o_h, s_o_i, s_o_j = out.stride()

    # tiling constants – empirically good for Hopper GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                 # batch × head dimension
        triton.cdiv(N, BLOCK_M),    # tiles over rows
        triton.cdiv(N, BLOCK_N),    # tiles over cols
    )

    triton_trimul_gemm[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        out_ptr=out,
        B=B,
        H=hidden,
        N=N,
        stride_a_batch=s_a_b,
        stride_a_head=s_a_h,
        stride_a_i=s_a_i,
        stride_a_k=s_a_k,
        stride_b_batch=s_b_b,
        stride_b_head=s_b_h,
        stride_b_k=s_b_i,          # original i‑stride becomes the k‑stride
        stride_b_j=s_b_j,
        stride_o_batch=s_o_b,
        stride_o_head=s_o_h,
        stride_o_i=s_o_i,
        stride_o_j=s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Post‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=W["to_out_norm.weight"],
        bias=W["to_out_norm.bias"],
    )                                             # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dim
    # ------------------------------------------------------------------
    out = F.linear(out, W["to_out.weight"])      # (B, N, N, dim)

    return out
