## Triton‑fused TriMul (outgoing) implementation

```python
"""
TriMul “outgoing” forward pass.
The only expensive operation is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiply (B·H, N, N) @ (B·H, N, N)^T.
We fuse this contraction into a single Triton kernel that
* loads the two operands as float16,
* performs fp16×fp16→fp32 dot products inside Tensor‑cores,
* accumulates in float32,
* writes the result back as float32.
All surrounding LayerNorm / Linear / gating logic stays in PyTorch
for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM  (M, N, K)  @  (M, N, K)^T   →  (M, N, N)
#   M = batch * hidden_dim
#   N = sequence length
#   K = N
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16   [M, N, K]
    right_ptr,         # *fp16   [M, N, K]
    out_ptr,           # *fp32   [M, N, N]
    M, N, K,
    stride_lm, stride_ln, stride_lk,   # left strides
    stride_rm, stride_rn, stride_rk,   # right strides
    stride_om, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program ids
    # --------------------------------------------------------------
    pid_z = tl.program_id(0)          # which (batch·head) matrix
    pid_m = tl.program_id(1)          # row tile   (i)
    pid_n = tl.program_id(2)          # col tile   (j)

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # pointer arithmetic
    # --------------------------------------------------------------
    # left:   (M, N, K)   index (z, i, k)
    a_ptrs = left_ptr + pid_z * stride_lm \
                       + offs_m[:, None] * stride_ln \
                       + offs_k[None, :] * stride_lk

    # right transposed on the last two dims:
    # we need right[z, k, j]  (swap N ↔ K)
    b_ptrs = right_ptr + pid_z * stride_rm \
                        + offs_k[:, None] * stride_rk \
                        + offs_n[None, :] * stride_rn

    # --------------------------------------------------------------
    # accumulate in fp32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                # fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                # fp16

        # fp16 dot → fp16, cast to fp32 and accumulate
        acc += tl.dot(a, b).to(tl.float32)

        # advance K‑dim pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # --------------------------------------------------------------
    # store
    # --------------------------------------------------------------
    out_ptrs = out_ptr + pid_z * stride_om \
                       + offs_m[:, None] * stride_on \
                       + offs_n[None, :] * stride_ok
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   shape [B, N, N, dim]  (float32)
        - mask         : torch.Tensor   shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensor (all fp32)
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Fuse the N³ contraction with Triton
    #    reshape to (M, N, N) where M = B * hidden_dim
    # ------------------------------------------------------------------- #
    # (B,N,N,hidden) -> (B, hidden, N, N) -> (M, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim

    left_fp16  = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    out_fp32 = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # strides in **elements** (not bytes) – PyTorch already reports that
    stride_lm, stride_ln, stride_lk = left_fp16.stride()
    stride_rm, stride_rn, stride_rk = right_fp16.stride()
    stride_om, stride_on, stride_ok = out_fp32.stride()

    # ------------------------------------------------------------------- #
    # 6. Launch kernel
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,                                   # one program per (batch·head)
        triton.cdiv(N, BLOCK_M),             # rows
        triton.cdiv(N, BLOCK_N),             # cols
    )

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        M, N, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # good for fp16 on Hopper
    )

    # reshape back to [B, N, N, hidden]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 7. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
```
