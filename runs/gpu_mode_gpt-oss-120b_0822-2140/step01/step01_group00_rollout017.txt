"""
TriMul “outgoing” implementation (forward only).

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reformulated as a batched matrix‑multiply:
    L_{b,d} : (N, N)   = left .permute(0,3,1,2)   # (B, H, N, N)
    R_{b,d} : (N, N)   = right.permute(0,3,1,2)   # (B, H, N, N)
    out_{b,d} = L_{b,d} @ R_{b,d}.T
A custom Triton GEMM kernel performs the multiply in bfloat16,
accumulating in fp32.  All other ops (layer‑norm, linear, gating,
masking) stay in PyTorch.  The kernel is launched with a 1‑D grid that
covers (batch × heads) × tiled‑M × tiled‑N blocks.

The final output shape is (B, N, N, dim) and dtype‑float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------------------- #
# 1. Element‑wise mask kernel (optional – masks are 0/1)
# --------------------------------------------------------------------------- #
@triton.jit
def mask_mul_kernel(
    src_ptr,          # float32*
    mask_ptr,         # float32*   (broadcasted to src shape)
    dst_ptr,          # float32*
    N_ELEM: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    tl.store(dst_ptr + offs, src * m, mask=mask)


def apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """src and mask are the same shape, contiguous, float32."""
    assert src.shape == mask.shape
    total = src.numel()
    BLOCK = 131072                     # ~128 K elements per program
    grid = (triton.cdiv(total, BLOCK),)

    out = torch.empty_like(src)
    mask_mul_kernel[grid](
        src_ptr=src,
        mask_ptr=mask,
        dst_ptr=out,
        N_ELEM=total,
        BLOCK=BLOCK,
    )
    return out


# --------------------------------------------------------------------------- #
# 2. Triton GEMM kernel: C = A @ B.T   (A,B : (B,H,N,N))
# --------------------------------------------------------------------------- #
@triton.jit
def triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix sizes (all = seq_len)
    stride_am, stride_ah, stride_ak,  # A strides: (batch*head, head, K)
    stride_bm, stride_bh, stride_bk,  # B strides (still B,H,N,K)
    stride_cm, stride_ch, stride_cn,  # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------- #
    # 1‑D grid → (batch, head, block_m, block_n)
    # ------------------------------------------------------------------- #
    total_blocks_m = tl.cdiv(M, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_head = total_blocks_m * total_blocks_n

    # index of (batch, head) pair
    batch_head_id = pid // blocks_per_head
    block_idx     = pid % blocks_per_head
    block_m = block_idx // total_blocks_n
    block_n = block_idx % total_blocks_n

    # recover batch and head from linear id
    batch = batch_head_id // tl.load(tl.constexpr_ptr(0))   # placeholder, will be replaced at launch
    head  = batch_head_id %  tl.load(tl.constexpr_ptr(0))   # placeholder

    # ------------------------------------------------------------------- #
    # Compute offsets for the tile
    # ------------------------------------------------------------------- #
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = a_ptr + \
        batch * stride_am + head * stride_ah + \
        offs_m[:, None] * stride_ak + tl.arange(0, BLOCK_K)[None, :] * 1   # K‑dim stride is 1 after permute

    b_ptrs = b_ptr + \
        batch * stride_bm + head * stride_bh + \
        offs_n[None, :] * stride_bk + tl.arange(0, BLOCK_K)[:, None] * 1   # note: we will read B^T, so K dimension is contiguous

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over K dimension
    # ------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < K) & (offs_n[None, :] < N), other=0.0)

        # a : (BLOCK_M, BLOCK_K), b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * 1        # stride over K (contiguous)
        b_ptrs += BLOCK_K * stride_bk # stride over K in B (since B is stored as (B,H,N,K))

    # ------------------------------------------------------------------- #
    # Write back
    # ------------------------------------------------------------------- #
    c_ptrs = c_ptr + \
        batch * stride_cm + head * stride_ch + \
        offs_m[:, None] * stride_cn + offs_n[None, :]

    mask_m = offs_m < M
    mask_n = offs_n < N
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# 3. Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]
                 ) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    data = (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   (float32, CUDA)
        mask         : [B, N, N]        (float32 0/1, CUDA) – may be ignored
        weights      : dict of torch.Tensor (see above)
        config       : dict containing
            dim, hidden_dim, nomask (bool)
    Returns:
        out : [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Helper functions (layer‑norm & linear) using supplied params
    # ------------------------------------------------------------------- #
    def layer_norm(x, w, b):
        return F.layer_norm(
            x,
            normalized_shape=(w.shape[0],),
            weight=w,
            bias=b,
            eps=1e-5,
        )

    def linear(x, w):
        # weight shape: [out, in], bias already folded into LN if needed
        return F.linear(x, w)

    # ------------------------------------------------------------------- #
    # 1️⃣ Input layer‑norm
    # ------------------------------------------------------------------- #
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )                               # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Projections to hidden_dim
    # ------------------------------------------------------------------- #
    left  = linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------- #
    if not nomask:
        mask_bc = mask.unsqueeze(-1).expand_as(left).contiguous()
        left  = apply_mask(left,  mask_bc)
        right = apply_mask(right, mask_bc)

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid of linear)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5️⃣ Core TriMul – batched GEMM in Triton
    # ------------------------------------------------------------------- #
    # cast to bfloat16 for speed, accumulate in fp32
    left_b  = left.to(torch.bfloat16)
    right_b = right.to(torch.bfloat16)

    # (B,N,N,H) -> (B,H,N,N)
    left_perm  = left_b.permute(0, 3, 1, 2).contiguous()
    right_perm = right_b.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    out_perm = torch.empty((B, H, N, N), dtype=torch.float32, device=left_perm.device)

    # strides (in elements, not bytes)
    a_stride_bh = left_perm.stride(0)   # batch*head stride
    a_stride_h  = left_perm.stride(1)   # head stride
    a_stride_m  = left_perm.stride(2)   # row (i) stride -> K dimension (contiguous)
    a_stride_k  = left_perm.stride(3)   # column (k) stride = 1 after contiguity

    b_stride_bh = right_perm.stride(0)
    b_stride_h  = right_perm.stride(1)
    b_stride_n  = right_perm.stride(2)   # row (j) stride
    b_stride_k  = right_perm.stride(3)   # column (k) stride = 1

    c_stride_bh = out_perm.stride(0)
    c_stride_h  = out_perm.stride(1)
    c_stride_m  = out_perm.stride(2)
    c_stride_n  = out_perm.stride(3)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_M = 8

    total_blocks_m = triton.cdiv(N, BLOCK_M)
    total_blocks_n = triton.cdiv(N, BLOCK_N)
    blocks_per_head = total_blocks_m * total_blocks_n
    grid = (B * H * blocks_per_head,)

    # Pass batch‑head count as a constexpr via a dummy pointer trick:
    # Triton cannot accept Python scalars as constexpr, so we embed them
    # in the kernel through a small tensor.
    bh_counts = torch.tensor([H], dtype=torch.int32, device=left_perm.device)

    triton_gemm_kernel[grid](
        a_ptr=left_perm,
        b_ptr=right_perm,
        c_ptr=out_perm,
        M=N, N=N, K=N,
        stride_am=a_stride_bh,
        stride_ah=a_stride_h,
        stride_ak=a_stride_k,          # K‑stride == 1
        stride_bm=b_stride_bh,
        stride_bh=b_stride_h,
        stride_bk=b_stride_k,          # K‑stride == 1
        stride_cm=c_stride_bh,
        stride_ch=c_stride_h,
        stride_cn=c_stride_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_M,
    )

    # Back to (B,N,N,H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] float32

    # ------------------------------------------------------------------- #
    # 6️⃣ Output layer‑norm (hidden_dim → hidden_dim)
    # ------------------------------------------------------------------- #
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣ Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = linear(out, weights["to_out.weight"])

    return out
