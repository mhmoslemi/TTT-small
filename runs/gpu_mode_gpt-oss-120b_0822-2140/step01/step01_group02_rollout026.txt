## Triton + PyTorch implementation

```python
"""
TriMul “outgoing” forward pass (no gradients).

The O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched GEMM C = A @ Bᵀ where
    A = left  : (B·H, N, N)
    B = right : (B·H, N, N)   (no bias)
    C = out   : (B·H, N, N)

We fuse the whole (batch × hidden_dim) GEMM space in a single Triton
kernel.  All surrounding operations (LayerNorm, linear projections,
sigmoid gates, final LayerNorm + linear) stay in PyTorch because they are
trivial compared with the cubic work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers (float16)
    M, N, K,                             # matrix sizes (int)
    stride_az, stride_am, stride_ak,     # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,     # strides for B (batch, row, col)
    stride_cz, stride_cm, stride_cn,     # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed."""
    pid_batch = tl.program_id(2)      # index over (B * hidden_dim)
    pid_m = tl.program_id(0)          # block‑row index
    pid_n = tl.program_id(1)          # block‑col index

    # ------------------------------------------------------------------
    # Offsets of the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the current tile of A  (row‑major)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_batch * stride_az \
                  + offs_m[:, None] * stride_am \
                  + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # ------------------------------------------------------------------
    # Pointers for the current tile of Bᵀ .
    # B is stored as (batch, N, K); we need element (k, n) => row k, col n.
    # ------------------------------------------------------------------
    b_ptrs = b_ptr + pid_batch * stride_bz \
                  + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K (tile‑wise)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load Bᵀ tile
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (cast back to fp16 to reduce traffic)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : (B*H, N, N)   float16 contiguous
    returns c of same shape and dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "square matrices are required"

    c = torch.empty_like(a)

    # Strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()      # original layout (B, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32          # fits nicely in SM90 registers / tensor‑cores

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B_H,
    )
    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # optimal on Hopper for 128×128 tiles
    )
    return c


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N]   (bool or float)
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys
              "dim"          – input feature dimension
              "hidden_dim"   – projection dimension (H)
              "nomask" (bool) – True → ignore mask
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Rearrange to (B, H, N, N) and cast to FP16 for the kernel
    # --------------------------------------------------------------
    #   (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and head dimension so the kernel sees (B*H, N, N)
    left_fp16 = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    # --------------------------------------------------------------
    # 6. Batched GEMM: C = left @ rightᵀ   (still in FP16)
    # --------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(left_fp16, right_fp16)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # --------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Post‑projection LayerNorm + output gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate        # out_gate shape: (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
