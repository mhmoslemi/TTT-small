"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM in Triton.  The tensors are
permuted to (B, H, N, N) and flattened to (B*H, N, N).  Each
(batch‑head) pair performs a tiled fp16 GEMM with an fp32
accumulator; the result is cast back to fp32 and the remaining
operations (LayerNorm, linear layers, sigmoids, gating) are kept
in PyTorch.

The kernel works for any batch size ≤ 2, sequence length up to 1024,
dim = 128/384 and hidden_dim = 128, and works with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A,B are fp16, accumulator fp32)
# --------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # base pointers
    stride_a_batch, stride_am, stride_ak,  # strides for A (batch, row, col)
    stride_b_batch, stride_bk,  stride_bn, # strides for B (batch, row=k, col=j)
    stride_c_batch, stride_cm, stride_cn,  # strides for C (batch, row, col)
    M, N, K,                               # matrix sizes (all == N in our case)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------- tile coordinates --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard against out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # pointers for the current tile of A and B
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_am
            + k_off[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + k_off[:, None] * stride_bk   # k dimension of B (the “row” after transpose)
            + offs_n[None, :] * stride_bn
        )

        # load tiles (fp16) – mask out‑of‑bounds elements
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product – results are accumulated in fp32
        acc += tl.dot(a, b)

    # -------------------- write back --------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]   (float32, CUDA)
        * mask_tensor  : [B, N, N]        (float32/bool/None, CUDA)
        * weights_dict : learned parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ----------------------------------------------------------------
    # unpack
    # ----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ----------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dimension)
    # ----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                                          # [B, N, N, dim]

    # ----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])           # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------
    # 3️⃣ Apply mask if present
    # ----------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)            # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------
    # 4️⃣ Gates (linear + sigmoid)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – batched GEMM in Triton
    # ----------------------------------------------------------------
    # permute to (B, H, N, N) and flatten batch*head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.reshape(B * hidden_dim, N, N).to(torch.float16)
    right_flat = right_perm.reshape(B * hidden_dim, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # strides in *elements* (not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * hidden_dim, 1),            # batch dimension (pid 0)
        triton.cdiv(N, BLOCK_M),                  # row tiles   (pid 1)
        triton.cdiv(N, BLOCK_N),                  # col tiles   (pid 2)
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        N,          # M
        N,          # N
        N,          # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 and reshape to original layout
    out = out_flat.to(torch.float32).reshape(B, hidden_dim, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, hidden]

    # ----------------------------------------------------------------
    # 6️⃣ Post‑norm on hidden dimension
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                                  # [B,N,N,hidden]

    # ----------------------------------------------------------------
    # 7️⃣ Element‑wise multiplication with the output gate
    # ----------------------------------------------------------------
    out = out * out_gate                               # fused elementwise (tiny cost)

    # ----------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dim
    # ----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])      # [B,N,N,dim]

    return out
