"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy computation is the 4‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel `d` this is a matrix multiplication
(N×N) @ (N×N)ᵀ.  We treat each hidden channel of every batch as an
independent matrix‑multiply and fuse all of them in a single Triton
kernel.  The kernel works on half‑precision inputs (to cut memory traffic)
while accumulating in fp32, and finally writes fp32 results.

All non‑linearities, LayerNorms and the final linear projection stay in
PyTorch – they are cheap compared with the O(N³) work of the einsum.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A @ Bᵀ  (A,B : [B*H, N, N])
#   * inputs are fp16, accumulator fp32, output fp32
#   * three‑dimensional grid: (pid_m, pid_n, pid_batch)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                # tensor pointers
    M, N, K,                            # matrix dimensions (M=N=seq_len, K=seq_len)
    stride_Ab, stride_Am, stride_Ak,    # A strides (batch, row, col)
    stride_Bb, stride_Bm, stride_Bk,    # B strides (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,    # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile‑row
    pid_n = tl.program_id(1)          # tile‑col
    pid_b = tl.program_id(2)          # batch‑*‑head index

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the tile (including batch offset)
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_n[None, :] * stride_Bm + offs_k[:, None] * stride_Bk

    # ------------------------------------------------------------------
    # Accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a block of A and B (both fp16).  Mask out‑of‑bounds elements.
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (offs_k[None, :] < K)),
                    other=0.0).to(tl.float32)   # cast to fp32 for the dot
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < K) & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Dot over the K‑tile
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Store result (fp32).  Apply mask to avoid writing OOB.
    # ------------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple (input, mask, weights, config)
        input   : torch.Tensor  [B, N, N, dim]   (float32, CUDA)
        mask    : torch.Tensor  [B, N, N]        (float32/bool) or None
        weights : dict of torch.Tensor parameters (float32, CUDA)
        config  : dict with at least `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape               # B, seq_len, seq_len, dim

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections → left/right  (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape [B,N,N] → [B,N,N,1]
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the fused GEMM
    #    (B, N, N, hidden) → (B*hidden, N, N)  and cast to fp16
    # ------------------------------------------------------------------
    #   (B, hidden, N, N) contiguous → view as (B*hidden, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_fp16  = left.view(B * hidden, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden, N, N).to(torch.float16)

    # output will be fp32
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 6. Triton launch configuration
    # ------------------------------------------------------------------
    # Tile sizes were chosen to keep shared memory under 64 KB on Hopper.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m dimension
        triton.cdiv(N, BLOCK_N),   # pid_n dimension
        B * hidden,                # pid_b dimension
    )

    # Strides (batch, row, col)
    a_batch, a_row, a_col = left_fp16.stride()
    b_batch, b_row, b_col = right_fp16.stride()
    c_batch, c_row, c_col = out_fp32.stride()

    batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N,                # M
        N,                # N
        N,                # K
        a_batch, a_row, a_col,
        b_batch, b_row, b_col,
        c_batch, c_row, c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to the original layout [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (hidden) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
