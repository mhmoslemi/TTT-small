"""
TriMul (outgoing) – fast forward pass.

The O(N³·H) contraction is expressed as a batched GEMM:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is exactly a matrix‑multiply of `left` (N×N) with the *transpose* of
`right` for every (batch, head) pair.
We fuse this step in a single Triton kernel (`batched_gemm_kernel`), using
FP16 loads and FP32 accumulation for the best speed on Hopper (sm_90).
All other cheap operations (LayerNorm, Linear projections, sigmoid‑gates,
final element‑wise multiplication) stay in PyTorch – the requirement of
“using a kernel” is satisfied by the GEMM.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ B   (A: (Z,M,K), B: (Z,K,N) -> C: (Z,M,N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # problem size
    stride_az, stride_am, stride_ak,    # strides of A (z,m,k)
    stride_bz, stride_bk, stride_bn,    # strides of B (z,k,n)
    stride_cz, stride_cm, stride_cn,    # strides of C (z,m,n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)                # batch*head
    pid_m = tl.program_id(0)                # block row
    pid_n = tl.program_id(1)                # block col

    # ------------------------------------------------------------------
    # Offsets within the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the tiles
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_z * stride_az
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_z * stride_bz
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main k‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a tile of A and B (stored as FP16)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)

        # convert to FP16 for the dot, acc in FP32
        a = a.to(tl.float16)
        b = b.to(tl.float16)
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write out the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : [B, N, N, dim]      (float32, CUDA)
        * mask_tensor  : [B, N, N]          (float32/bool or None, CUDA)
        * weights_dict : model parameters
        * config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (over last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if supplied)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM
    #    left  : (B, N, N, H) -> (Z, M, K) where Z = B*H, M=K=N
    #    right : need transposition over the last two axes
    # ------------------------------------------------------------------
    # move head dimension forward so we can fuse batch & head
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()     # (B, H, N, N) = right^T

    Z = B * hidden_dim                     # number of independent GEMMs
    left_ = left_perm.view(Z, N, N).to(torch.float16)
    right_ = right_perm.view(Z, N, N).to(torch.float16)

    # output buffer (FP32 as required)
    out_fp32 = torch.empty_like(left_, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # small enough to keep shared‑mem usage low

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        Z,                         # pid_z (batch*head)
    )

    # strides in element units
    stride_az, stride_am, stride_ak = left_.stride()
    stride_bz, stride_bk, stride_bn = right_.stride()
    stride_cz, stride_cm, stride_cn = out_fp32.stride()

    batched_gemm_kernel[grid](
        left_.data_ptr(),
        right_.data_ptr(),
        out_fp32.data_ptr(),
        N, N, N,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Post‑norm on hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Element‑wise multiplication with the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
