## Triton‑based outgoing TriMul implementation

```python
"""
TriMul “outgoing” forward pass (no gradient).

The expensive part is the triple‑nested contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be written as a batched GEMM:
    C = A @ Bᵀ   with  A,B ∈ ℝ^{(B·H) × N × N}
We fuse this GEMM in a single Triton kernel (fp16 input,
fp32 accumulation).  All surrounding ops (LayerNorm,
Linear, gating) remain in PyTorch because they are cheap.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape [Batch, M, K] with K == M == N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # base pointers
    M, N, K,                           # matrix dimensions
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col) – original layout
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where B is logically transposed."""
    pid_batch = tl.program_id(2)               # index over (B·H)
    pid_m = tl.program_id(0)                   # block row
    pid_n = tl.program_id(1)                   # block col

    # ------------------------------------------------------------------
    # Block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current block of A
    a_ptrs = a_ptr + pid_batch * stride_az \
                 + offs_m[:, None] * stride_am \
                 + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                 + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # FP16×FP16 → FP32 accumulate
        acc += tl.dot(a, b)

        # Advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back (store as FP16 to reduce bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_outgoing_fp16(a: torch.Tensor,
                               b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    a, b : (B*H, N, N)   float16, contiguous
    returns c : same shape and dtype
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N, "square matrices required"

    c = torch.empty_like(a)

    # Strides expressed in *elements*
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bn, stride_bk = b.stride()   # original layout (batch, N, K)
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile size – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32    # fits nicely into registers / Tensor‑core WMMA

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B_H,                       # batch*head dimension
    )

    batched_gemm_outgoing_kernel[grid](
        a,
        b,
        c,
        N,               # M
        N,               # N
        N,               # K
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,     # empirically good for 64×64 tiles on Hopper
    )
    return c


# ----------------------------------------------------------------------
# Public entry point required by the kata
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a fused Triton GEMM.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask         : torch.Tensor  [B, N, N]       (float32 / bool)
        weights      : dict of torch.Tensor (float32)
        config       : dict with keys
                       - "dim"          (int)
                       - "hidden_dim"   (int)
                       - "nomask"       (bool, optional)

    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Reshape for the fused GEMM: (B, H, N, N) -> (B*H, N, N)
    # ------------------------------------------------------------------
    #   permute -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    #   collapse batch & head
    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Batched GEMM C = left @ rightᵀ   (still fp16)
    # ------------------------------------------------------------------
    out_fp16 = _batched_gemm_outgoing_fp16(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden_dim) and cast to fp32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Post‑projection LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate            # out_gate shape (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
