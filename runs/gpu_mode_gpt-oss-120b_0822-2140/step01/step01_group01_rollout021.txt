"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The costly part of TriMul is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* it is a matrix multiplication
    L = left[b,:, :, d] (N×N)      R = right[b,:, :, d] (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We flatten the batch and hidden dimensions:
    M = B * hidden_dim
and launch a tiled Triton kernel that computes M independent GEMMs
(N×N)·(N×N)ᵀ.  The kernel works in fp16 (to exploit Tensor‑cores) and
accumulates in fp32 for numerical stability.  All other operations
(LayerNorm, linear projections, gating, masking) are cheap and are
performed in PyTorch.

The kernel is auto‑tuned for three block‑size configurations that
cover the whole range of sequence lengths used in the tests.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# --------------------------------------------------------------
@triton.jit
def trimul_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    N, M,                           # N = seq_len, M = B * hidden_dim
    stride_ah, stride_ai, stride_ak,    # A strides (M, N, N)
    stride_bh, stride_bj, stride_bk,    # B strides (M, N, N)
    stride_ch, stride_ci, stride_cj,    # C strides (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM A @ Bᵀ where A,B are (M,N,N) and C is (M,N,N)."""
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # flattened batch*hidden dimension

    # ------------ offsets for the current tile -----------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to the first element of the (pid_z)th matrix pair
    a_batch_ptr = a_ptr + pid_z * stride_ah
    b_batch_ptr = b_ptr + pid_z * stride_bh
    c_batch_ptr = c_ptr + pid_z * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------- loop over K dimension (tiles) --------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   shape (BLOCK_M, BLOCK_K)  (fp16)
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B_T[k, j] shape (BLOCK_K, BLOCK_N)  (fp16)
        # original B is indexed as B[j, k]; we load it transposed.
        b_ptrs = (
            b_batch_ptr
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # fp16 × fp16 -> fp32 accumulation (Tensor‑core friendly)
        acc += tl.dot(a, b)

    # -------------- write back ----------------------------------
    c_ptrs = (
        c_batch_ptr
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Helper that launches the kernel (auto‑tuned block sizes)
# --------------------------------------------------------------
def trimul_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, H, N, N)   fp16
    returns    : (B, H, N, N)   fp32  (each slice = left @ rightᵀ)
    """
    B, H, N, _ = left.shape
    M = B * H                                 # flattened batch*hidden
    out = torch.empty((B, H, N, N), dtype=torch.float32, device=left.device)

    # strides (int64)
    stride_ah, stride_ai, stride_ak = left.stride()
    stride_bh, stride_bj, stride_bk = right.stride()
    stride_ch, stride_ci, stride_cj = out.stride()

    # ------------------------------------------------------------------
    # Auto‑tuned block configurations (cover small to large seq_len)
    # ------------------------------------------------------------------
    configs = [
        (64, 64, 64),   # good for N ≤ 256
        (128, 128, 64), # medium
        (128, 128, 128) # large N
    ]

    # pick the first config that fits the current N
    BLOCK_M, BLOCK_N, BLOCK_K = configs[-1]   # default large
    for bm, bn, bk in configs:
        if N <= bm:
            BLOCK_M, BLOCK_N, BLOCK_K = bm, bn, bk
            break

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        M,                         # batch*hidden
    )

    trimul_gemm_fp16_kernel[grid](
        left, right, out,
        N, M,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bj, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out  # fp32


# --------------------------------------------------------------
# Main entry point required by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    The O(N³) contraction is performed by a Triton fp16 kernel;
    all other operations are done with PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim) – no bias
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)        # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask.to(device).unsqueeze(-1)          # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – also dim → hidden_dim
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, H, N, N) fp16 layout expected by the kernel.
    # -----------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # fp16 for the kernel (Tensor‑cores)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # (B, H, N, N) fp32 result
    out_perm = trimul_gemm_fp16(left_fp16, right_fp16)

    # back to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7. Final projection back to dim
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)       # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                        # (B, N, N, dim)

    return out  # fp32 tensor, ready for the next layer
