"""
TriMul “outgoing” forward pass (αFold‑3 style).

The only O(N³) work is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be written as a batched GEMM C = A @ Bᵀ for each head d.
The kernel below performs this GEMM in FP16 (FP32 accumulation) for all
(b, head) pairs in a single launch.  All lightweight ops (LayerNorm,
Linear, sigmoid, gating) stay in PyTorch – they are negligible compared
to the N³ work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batched, FP16 → FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (int)
    stride_az, stride_am, stride_ak,   # strides for A  (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B  (batch, row, col)
    stride_cz, stride_cm, stride_cn,   # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is logically transposed."""
    pid_batch = tl.program_id(2)               # (B*H) index
    pid_m = tl.program_id(0)                   # block‑row
    pid_n = tl.program_id(1)                   # block‑col

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of this tile in A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_az \
                 + offs_m[:, None] * stride_am \
                 + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    # B is stored as (batch, N, K).  For Bᵀ we need element (k, n)
    b_ptrs = b_ptr + pid_batch * stride_bz \
                 + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator (fp32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load a tile of A
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load a tile of Bᵀ
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        # FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write‑back (store as fp16 to reduce bandwidth)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for FP16 tensors.

    a, b : (B*H, N, N)  fp16, contiguous
    returns c of same shape / dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, K = a.shape
    assert K == N and b.shape == (B_H, N, N)

    c = torch.empty_like(a)

    # Strides are in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tunable tile sizes – 128×128 works well on Hopper/H200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32          # fits in registers on sm90

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B_H,                       # batch*head dimension
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # empirically good for 128×128 tiles
    )
    return c


# ----------------------------------------------------------------------
# Public entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, dim] (float32)
            - mask         : torch.Tensor[B, N, N] or None
            - weights      : dict of fp32 weight tensors
            - config       : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape: B,N,N)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of additional linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Rearrange to (B, hidden, N, N) → (B*hidden, N, N) and cast to fp16
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left  = left.view(B * hidden, N, N).to(torch.float16)
    right = right.view(B * hidden, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Heavy O(N³) contraction via the Triton kernel
    # ------------------------------------------------------------------
    out_fp16 = _batched_gemm_fp16(left, right)      # (B*hidden, N, N)

    # ------------------------------------------------------------------
    # 7. Back‑to (B, N, N, hidden) and fp32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Post‑projection LayerNorm (hidden) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                               # broadcast (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 9. Final linear back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
