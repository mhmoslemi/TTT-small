"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The dominant work of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L = left[b,:, :, d]           (N×N)
    R = right[b,:, :, d]          (N×N)
    out[b,:, :, d] = L @ Rᵀ       (N×N)

We flatten the batch and hidden dimensions (M = B·H) and compute M independent
GEMMs of size (N×N)·(N×N)ᵀ with a tiled Triton kernel.  All other operations
(LayerNorms, linear projections, sigmoids, masking) are cheap and are kept
in PyTorch.

The kernel loads tiles of A and Bᵀ, accumulates in fp32 and writes the result.
Block sizes 64×64×64 work well on Hopper (SM90) and fit comfortably in the
shared memory budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent (N,N) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr,                # left  tensor (M, N, N)
    b_ptr,                # right tensor (M, N, N)
    c_ptr,                # output tensor (M, N, N)
    N,                    # sequence length
    stride_am, stride_ak, stride_ai,   # A strides: (M,N,N) -> (batch, row, col)
    stride_bm, stride_bk, stride_bj,   # B strides: (M,N,N)
    stride_cm, stride_ck, stride_cj,   # C strides: (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of GEMMs:
        C[m] = A[m] @ B[m].T   where A,B,C ∈ (M,N,N)
    Each program instance works on a tile (i0:i0+BLOCK_M, j0:j0+BLOCK_N) of one
    matrix identified by program_id(2) (the flattened batch*hidden index).
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # which (batch,head) pair

    # tile start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointer for the selected matrix pair
    a_batch = a_ptr + pid_z * stride_am
    b_batch = b_ptr + pid_z * stride_bm
    c_batch = c_ptr + pid_z * stride_cm

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the summed‑over axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_batch
            + offs_m[:, None] * stride_ai   # row stride
            + offs_k[None, :] * stride_ak   # col stride
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ[k, j]   B is stored as (row=j, col=k); we load transposed
        b_ptrs = (
            b_batch
            + offs_k[:, None] * stride_bk   # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bj   # j dimension (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product, accumulation in fp32
        acc += tl.dot(a, b)

    # store the tile
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_ck
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, H, N, N)  fp32
    returns       : (B, H, N, N)  fp32 where each slice is left @ rightᵀ
    """
    B, H, N, _ = left.shape
    M = B * H                         # flattened batch*head dimension

    # reshape to (M, N, N) – this does not copy data because the tensor is
    # contiguous after the permute+contiguous in the caller.
    left_f = left.reshape(M, N, N)
    right_f = right.reshape(M, N, N)

    out = torch.empty_like(left_f)

    # strides as int64 (required by Triton)
    stride_am, stride_ai, stride_ak = left_f.stride()
    stride_bm, stride_bj, stride_bk = right_f.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        M,                         # pid_z  (batch * head)
    )

    _trimul_gemm_kernel[grid](
        left_f,
        right_f,
        out,
        N,
        stride_am, stride_ak, stride_ai,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # Hopper can handle 4‑warps per block comfortably
    )

    # reshape back to (B, H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor : torch.Tensor of shape (B, N, N, dim)
        - mask         : torch.Tensor of shape (B, N, N)   (may be unused)
        - weights      : dict of pre‑loaded model parameters
        - config       : dict with keys:
              * dim           – input feature dimension
              * hidden_dim    – hidden dimension for the multiplicative update
              * nomask (bool) – whether the mask should be ignored

    Returns
    -------
    torch.Tensor of shape (B, N, N, dim)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections: dim -> hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w)        # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape (B, N, N, 1) – same dtype as left/right
        mask_f = mask.to(device).unsqueeze(-1).type_as(left)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also dim -> hidden_dim
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = left @ rightᵀ   (per hidden channel)
    #    Convert to (B, H, N, N) before the kernel.
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    out_perm = _trimul_gemm(left_perm, right_perm)    # (B, H, N, N)

    # back to original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                    # (B, N, N, dim)

    # ensure fp32 output as required
    return out.to(torch.float32)
