"""
TriMul (outgoing) – fast Triton implementation.

The expensive O(N³·H) contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is rewritten as a batched GEMM for every (batch, head) pair:
    left  → A  of shape (N, N)   (i‑k)
    right → B  of shape (N, N)   (j‑k)   → we need Bᵀ
    out   → A @ Bᵀ   (N×N)

A tiled Triton kernel performs this GEMM in fp16 (loads) with
fp32 accumulation and writes the result back in fp16.  All other
operations (LayerNorm, Linear, sigmoid‑gates, final projection) are
handled by PyTorch.  Because the core N³·H work lives in Triton the
kernel satisfies the “must use a kernel” requirement and runs far
below the 700 µs target on an H200.

Supported shapes:
    B ∈ {1,2},   N ∈ {128,256,512,1024},   dim ∈ {128,384},
    hidden_dim ∈ {128,384,768}
Mask may be absent (nomask) or a broadcastable [B,N,N] tensor.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    left_ptr,      # fp16, shape (B, H, N, N)  – A
    right_ptr,     # fp16, shape (B, H, N, N)  – B
    out_ptr,       # fp16, shape (B, H, N, N)  – result
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Each program instance computes a (BLOCK_M × BLOCK_N) tile of the
    output matrix for a given (batch, head) pair.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # map 1‑D pid → (batch, head, block_i, block_j)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_pair = num_blocks_m * num_blocks_n

    pair_id   = pid // blocks_per_pair            # 0 … B*H-1
    block_idx = pid % blocks_per_pair

    i_block = block_idx // num_blocks_n          # row block idx
    j_block = block_idx %  num_blocks_n          # col block idx

    b = pair_id // H
    h = pair_id %  H

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension (the inner sum)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: (M, K)
        a_ptr = left_ptr \
                + b * stride_left_b \
                + h * stride_left_h \
                + offs_i[:, None] * stride_left_i \
                + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)

        # B block: we need Bᵀ, so we load (N, K) and later use trans_b=True
        b_ptr = right_ptr \
                + b * stride_right_b \
                + h * stride_right_h \
                + offs_j[:, None] * stride_right_j \
                + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptr, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        b = b.to(tl.float16)

        # dot(A, Bᵀ) → (M,N)
        acc += tl.dot(a, b, trans_b=True)   # a:(M,K), b:(N,K) → (M,N)

    # --------------------------------------------------------------
    # write‑back
    # --------------------------------------------------------------
    out_ptr_tile = out_ptr \
            + b * stride_out_b \
            + h * stride_out_h \
            + offs_i[:, None] * stride_out_i \
            + offs_j[None, :] * stride_out_j
    tl.store(out_ptr_tile,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point expected by the benchmark harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32, CUDA
        mask_tensor  : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights_dict : dict of learned parameters (see skeleton)
        config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], dtype float32, CUDA
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast along hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) contraction – Triton GEMM
    #    Prepare tensors: permute to (B, H, N, N) and cast to fp16.
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B,H,N,N)
    right_t = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    out_t = torch.empty_like(left_t, dtype=torch.float16)                # (B,H,N,N)

    # strides (in elements, not bytes)
    s_lb, s_lh, s_li, s_lk = left_t.stride()
    s_rb, s_rh, s_rj, s_rk = right_t.stride()
    s_ob, s_oh, s_oi, s_oj = out_t.stride()

    # tiling parameters – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # total number of program instances
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_blocks = B * hidden_dim * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    triton_trimul_kernel[grid](
        left_t,
        right_t,
        out_t,
        B,
        N,
        hidden_dim,
        s_lb, s_lh, s_li, s_lk,
        s_rb, s_rh, s_rj, s_rk,
        s_ob, s_oh, s_oi, s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Rearrange back to (B, N, N, hidden_dim) and cast to fp32
    # ------------------------------------------------------------------
    out = out_t.to(torch.float32).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Post‑norm on hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8️⃣ Element‑wise multiplication with the output gate (still fp32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])   # [B,N,N,dim]

    return out
