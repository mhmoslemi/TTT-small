        # compute weights
        maxS = np.max(S)
        exp_beta = np.exp(beta * (S - maxS))  # avoid overflow

                l = -lag
                v[l:] -= h[:n-l]   # i+lag = i - l term
                v[:n-l] -= h[l:]   # i-lag = i + l term

        step = step_size / np.sqrt(iter + 1)

        # compute weight
        maxS = np.max(S_raw)

        # gradient
        grad = np.zeros_like(h)
        # compute contributions
        for idx, lag in enumerate(range(-(n-1), n)):
            weight = wnorm[idx]
            if weight == 0:
                continue
            # compute vector v = dSℓ/dh
            if lag >= 0:
                # v = ones; subtract h[lag:] from first part, subtract h[:n-lag] from second part
                v = np.ones(n)
                v[:n-lag] -= h[lag:]
                v[lag:] -= h[:n-lag]
            else:
                l = -lag
                v = np.ones(n)
                v[l:] -= h[:n-l]
                v[:n-l] -= h[l:]
            grad += weight * v
        # step size
        step = step0 / np.sqrt(iter + 1)
        # gradient descent
        h_new = h - step * grad
        # projection onto constraints
        h_new = project_onto_constraints(h_new, target_sum)
        # evaluate C5
        c5 = evaluate(h_new, dx)
        if c5 < best_val:
            best_val = c5
            best_h = h_new.copy()
        # update
        h = h_new
        # optionally increase beta gradually
        if iter % 500 == 0 and iter > 0:
            beta = min(beta + beta_inc, 200)  # maybe saturate
        iter += 1
        # early break if best_val <= 0.3807 maybe
        if best_val <= 0.3807:
            break
    return best_h, best_val

def simulated_annealing(h0, budget_s):
    start = time.time()
    h = h0.copy()
    cur_c5 = evaluate(h)
    best_h = h.copy()
    best_c5 = cur_c5
    n = len(h)
    target_sum = n/2
    T = 0.1  # initial temperature
    T_min = 1e-6
    decay = 0.9995
    # step scale: maximum possible delta magnitude for a move
    max_step = 0.5  # because we cap at [0,1]
    while time.time() - start < budget_s:
        # propose move
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # compute allowed delta range
        # want to add delta to i, subtract from j.
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        delta = rng.uniform(delta_low, delta_high)
        # optionally scale by step factor that decays
        # we may just use the bound as allowed, no scaling.
        # apply
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_c5 = evaluate(h_new)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        # update temperature
        T = max(T * decay, T_min)
    return best_h, best_c5
