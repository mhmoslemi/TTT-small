def gradient_descent(h0, time_limit, S):
    h = h0.copy().astype(float)
    n = len(h)
    dx = 2./n
    best_h = h.copy()
    # compute bestC initially using compute_c5
    bestC = compute_c5(h)
    start = time.time()
    iter = 0
    while time.time() - start < time_limit:
        # compute auto-correlation:
        corr = np.correlate(h, h, mode='full') * dx
        min_corr = np.min(corr)
        # get active lags
        tol = 1e-12
        active_indices = np.where(corr <= min_corr + tol)[0]
        # compute lag list
        active_lags = active_indices - (n - 1) # convert index to lag
        # compute gradient
        g = np.zeros_like(h)
        for L in active_lags:
            if L == 0:
                g += 1.0
                continue
            if L > 0:
                # h_i contributes to g[i] from neighbors at i+L and i-L
                # For all i where i+L < n:
                g[:n-L] += h[L:]  # h_{i+L}
                g[L:] += h[:n-L]  # h_{i-L}
            else: # L < 0
                k = -L
                g[k:] += h[:n-k]
                g[:n-k] += h[k:]

        # step size schedule decreasing: step = step0 * (1 - iter/max_iter)
        # we may set step0 = 0.05
        step0 = 0.1
        # approximate max_iters based on time left: we can compute a decreasing schedule around 0
        step = step0 * (1 - (time.time() - start) / time_limit)
        if step < 1e-5:
            step = 1e-5

        # update candidate
        cand = h + step * g
        # project to constraints
        S = n/2.0  # sum constraint
        cand_proj = project_onto_simplex_box(cand, S)

        candC = compute_c5(cand_proj)
        if candC < bestC - 1e-12:
            bestC = candC
            best_h = cand_proj.copy()
        # Accept if decreasing else revert
        curC = compute_c5(h)  # but we have curC maybe
        if candC < curC:
            h = cand_proj
        # else keep h unchanged
        iter += 1
        # optional break if no improvement for many steps

    return best_h, bestC
