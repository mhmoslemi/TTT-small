```python
    rng = np.random.default_rng(seed)

    # Choose n
    n = 200 # even
    weight = n // 2

    # Initialize h randomly: choose weight random positions to set 1.
    h = np.zeros(n, dtype=np.int8)
    ones_idx = rng.choice(n, size=weight, replace=False)
    h[ones_idx] = 1

    # compute O for k=1..n-1:
    # O = np.array([np.dot(h[:n - k], h[k:]) for k in range(1, n)], dtype=int)
    O = np.zeros(n, dtype=int)
    # O[0] = weight (not needed)
    for k in range(1, n):
        O[k] = int(np.dot(h[:n - k], h[k:]))

    min_O = O[1:].min()
    cur_Umax = weight - min_O
    best_Umax = cur_Umax
    best_h = h.copy()

    start = time.time()
    # schedule parameters
    # We'll set initial temperature T0 based on typical Umax range e.g., 10.
    T0 = 5.0
    # compute total iterations maybe 200000?
    # We'll use time-based loop:
    i = 0
    while time.time() - start < budget_s * 0.9:
        # random swap
        # pick a random 1-index and 0-index
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1 - h)
        i1 = rng.choice(ones)
        i2 = rng.choice(zeros)

        # compute delta
        delta = np.zeros(n, dtype=int)
        # for k from 1..n-1 compute loss+gain
        # maybe vectorize: compute arrays of neighbor indices? but loop is okay.
        for k in range(1, n):
            loss = 0
            # i1 + k
            ip = i1 + k
            if ip < n and ip != i2 and h[ip]:
                loss += 1
            im = i1 - k
            if im >= 0 and im != i2 and h[im]:
                loss += 1
            gain = 0
            jp = i2 + k
            if jp < n and jp != i1 and h[jp]:
                gain += 1
            jm = i2 - k
            if jm >= 0 and jm != i1 and h[jm]:
                gain += 1
            delta[k] = gain - loss

        # new O = O + delta
        O_new = O + delta
        # compute min_O_new ignoring index 0
        min_O_new = O_new[1:].min()
        cur_Umax_new = weight - min_O_new

        # accept?
        t = budget_s - (time.time() - start)
        # temperature schedule: linearly decreasing from T0
        # calculate fraction progress
        progress = (time.time() - start) / (budget_s * 0.9)
        T = T0 * (1 - progress)
        if cur_Umax_new < cur_Umax + 1e-12 or rng.random() < np.exp(-(cur_Umax_new - cur_Umax) / max(T, 1e-8)):
            # Accept swap
            h[i1] = 0
            h[i2] = 1
            O = O_new
            cur_Umax = cur_Umax_new
            if cur_Umax < best_Umax:
                best_Umax = cur_Umax
                best_h = h.copy()
        i += 1

    # compute final bound
    c5 = best_Umax * 2.0 / n

    # return float array real 0/1: h_values as list of floats
    return best_h.astype(float).tolist(), float(c5), int(n)
```
