def gradient_descent(h, current_val, start_time, time_limit):
    n = len(h)
    dx = 2.0 / n
    step = 0.5
    for it in range(50000):
        if time.time() - start_time > time_limit: break
        # compute O(s) vector
        O_vec = np.correlate(h, 1 - h, mode='full') * dx
        s_idx = np.argmax(O_vec)
        O_val = O_vec[s_idx]
        if O_val < current_val - 1e-12:
            current_val = O_val
        # shift
        s = s_idx - (n-1)
        # compute gradient g_i = (1 - h_{i+s}) if i+s in range else 1
        g = np.empty_like(h)
        if s >= 0:
            # overlapping indices
            g[:n-s] = 1 - h[s:]   # use h[i+s]
            g[n-s:] = 1.0
        else:
            s_abs = -s
            # i+s >=0 -> i>=s_abs
            g[s_abs:] = 1 - h[:n-s_abs]
            g[:s_abs] = 1.0
        # now project gradient to zero-sum direction
        # descent direction d = -g + lambda*1 where lambda = mean(g)
        lam = np.mean(g)
        d = -g + lam
        # line search max step without violating bounds
        # compute alpha_max such that 0 <= h + alpha*d <=1
        # For each i, if d_i > 0, bound1 = (1 - h_i)/d_i else bound1 = -h_i/d_i
        # compute possible steps and take smallest positive.
        pos = d > 0
        neg = d < 0
        max_alpha = 1e9
        if np.any(pos):
            max_alpha = min(max_alpha, np.min((1.0 - h[pos]) / d[pos]))
        if np.any(neg):
            max_alpha = min(max_alpha, np.min((-h[neg]) / d[neg]))
        # limit step size to step * max_alpha maybe
        alpha = min(step, max_alpha)
        # update
        h = h + alpha * d
        # optional small perturbation to avoid stagnation: add random noise?
        # compute new value
        new_val = evaluate_erdos_solution(h)
        if new_val < current_val:
            current_val = new_val
        else:
            # maybe revert if not improved? but we maintain descent direction anyway ensures Omax can't increase? Actually Omax could increase for other shifts. Use simple accept if improved else revert.
            if new_val > current_val + 1e-6:
                # revert
                h = h - alpha*d
        # gradually reduce step
        step *= 0.9995
    return h, current_val

def binary_greedy_optimize(h, start_time, time_limit):
    n = len(h)
    dx = 2.0/ n
    best_val = evaluate_erdos_solution(h)
    # ensure h is binary
    h = (h > 0.5).astype(np.float64) # or round.
    # local improvement loop
    while time.time() - start_time < time_limit:
        O_vec = np.correlate(h, 1 - h, mode='full') * dx
        s_idx = np.argmax(O_vec)
        O_val = O_vec[s_idx]
        if O_val < best_val - 1e-12:
            best_val = O_val
        if O_val <= best_val + 1e-12:
            # no improvement? maybe break
            break
        s = s_idx - (n-1)
        if s < 0: s = -s  # we can treat symmetrical
        # find swap
        # overlapping indices
        inside = np.arange(0, n-s)
        tail = np.arange(n-s, n)
        # candidates inside where h[i]==0 and partner is 1
        cand_in = inside[(h[inside]==0) & (h[inside + s]==1)]
        # candidates tail where h[i]==1
        cand_tail = tail[h[tail]==1]
        if len(cand_in) == 0 or len(cand_tail) == 0:
            # try alternative swap inside both i,j within overlapping region: find i with h_i=1, h_{i+s}=0 and j with h_j=0, h_{j+s}=1
            cand_i = inside[(h[inside]==1) & (h[inside + s]==0)]
            cand_j = inside[(h[inside]==0) & (h[inside + s]==1)]
            if len(cand_i) > 0 and len(cand_j) > 0:
                i = np.random.choice(cand_i)
                j = np.random.choice(cand_j)
                # swap
                h[i] = 0
                h[j] = 1
                new_val = evaluate_erdos_solution(h)
                if new_val < best_val - 1e-12:
                    best_val = new_val
                else:
                    # revert revert
                    h[i] = 1
                    h[j] = 0
            else:
                # cannot improve further
                break
        else:
            i = np.random.choice(cand_tail)
            j = np.random.choice(cand_in)
            # swap
            h[i] = 0
            h[j] = 1

                best_val = new_val

                # revert
                h[i] = 1

                # maybe break if too many fails?
                break
    return h, best_val
