import numpy as np
from scipy.optimize import minimize_scalar

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h to satisfy 0 <= h_i <= 1 and sum(h) = n/2."""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        # Don't overshoot
        if diff > 0:
            step = min(step, (1.0 - h[mask]).min())
        else:
            step = max(step, h[mask].min())
        h[mask] += step
    h = np.clip(h, 0, 1)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Find a coordinate to adjust
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def line_search_1d(h, d, n, n_coarse=50, n_fine=20):
    """Line search along direction d from point h. Returns best h and value."""
    d = d - d.mean()  # zero-sum
    if np.max(np.abs(d)) < 1e-15:
        return h, evaluate_c5(h)
    
    # Valid range
    pos = d > 1e-14
    neg = d < -1e-14
    t_max = np.inf
    t_min = -np.inf
    if np.any(pos):
        t_max = np.min((1.0 - h[pos]) / d[pos])
    if np.any(neg):
        t_min = np.min(h[neg] / (-d[neg]))
    
    t_min = max(t_min, -1.0)  # bound the search
    t_max = min(t_max, 1.0)
    
    if t_min >= t_max - 1e-14:
        return h, evaluate_c5(h)
    
    best_t = 0.0
    best_val = evaluate_c5(h)
    
    # Coarse search
    ts = np.linspace(t_min, t_max, n_coarse)
    vals = []
    for t in ts:
        h_new = h + t * d
        h_new = np.clip(h_new, 0, 1)
        vals.append(evaluate_c5(h_new))
    vals = np.array(vals)
    idx = np.argmin(vals)
    best_val = vals[idx]
    best_t = ts[idx]
    
    # Fine search around best
    span = (t_max - t_min) / n_coarse
    ts_fine = np.linspace(best_t - span, best_t + span, n_fine)
    ts_fine = ts_fine[(ts_fine >= t_min) & (ts_fine <= t_max)]
    for t in ts_fine:
        h_new = h + t * d
        h_new = np.clip(h_new, 0, 1)
        val = evaluate_c5(h_new)
        if val < best_val:
            best_val = val
            best_t = t
    
    h_new = h + best_t * d
    h_new = np.clip(h_new, 0, 1)
    return h_new, best_val

def swap_direction(h, n, rng):
    """Generate a swap direction: transfer mass from i to j."""
    i, j = rng.integers(0, n, size=2)
    while j == i:
        j = rng.integers(0, n)
    d = np.zeros(n)
    d[i] = -1.0
    d[j] = 1.0
    return d

def random_direction(n, rng):
    d = rng.standard_normal(n)
    d -= d.mean()
    return d

def block_direction(h, n, rng):
    """Transfer mass from a block to another block."""
    b1 = rng.integers(0, n)
    b2 = rng.integers(0, n)
    d = np.zeros(n)
    # Block 1: source
    start1 = b1
    end1 = min(b1 + rng.integers(1, n//4 + 1), n)
    # Block 2: destination
    start2 = b2
    end2 = min(b2 + rng.integers(1, n//4 + 1), n)
    d[start1:end1] = -1.0
    d[start2:end2] = (end1 - start1) / max(1, (end2 - start2))
    d -= d.mean()
    return d

def thue_morse_bits(n):
    """Generate Thue-Morse sequence of length n."""
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def binary_random(n, rng):
    """Random binary sequence with exactly n/2 ones."""
    ones = rng.integers(0, n, size=n//2)
    h = np.zeros(n)
    h[ones] = 1.0
    return h

def line_search_fast(h, d, n, n_coarse=80, n_fine=30):
    """Fast line search using precomputed correlation components."""
    d = d - d.mean()
    if np.max(np.abs(d)) < 1e-15:
        return h, evaluate_c5(h)
    
    # Precompute
    A = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    corr_dh = np.correlate(d, h, mode="full")   # length 2n-1
    corr_dd = np.correlate(d, d, mode="full")    # length 2n-1

    # C = corr_dd (with negative sign in the formula)

        return np.max(A + t * B - t*t * corr_dd) * dx

    neg = d < -1e-14
    t_max = np.inf

            best_val = v
            best_t = t
    
    return h + best_t * d, best_val

def optimize_local(h, n, rng, time_limit, start_time):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    h_curr = h.copy()
    no_improve = 0
    max_no_improve = 300
    
    while time.time() - start_time < time_limit:
        r = rng.random()
        if r < 0.25:
            g = subgradient_c5(h_curr, n)
            d = -(g - np.mean(g))
        elif r < 0.45:
            d = rng.standard_normal(n)
        elif r < 0.65:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (i + 1) % n
            d = np.zeros(n)
            d[i] = -1.0
            d[j] = 1.0
        elif r < 0.80:
            bsize = int(rng.integers(2, n // 4 + 1))
            i = int(rng.integers(0, max(1, n - bsize)))
            j = int(rng.integers(0, max(1, n - bsize)))
            d = np.zeros(n)
            d[i:i+bsize] = -1.0
            d[j:j+bsize] = 1.0
        else:
            d = np.zeros(n)
            for _ in range(rng.integers(2, 5)):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    d[i] -= 1.0
                    d[j] += 1.0
        
        h_new, c5_new = line_search_fast(h_curr, d, n)
        h_curr = h_new
        
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Perturb
            h_curr = project_feasible(h_curr + 0.1 * rng.standard_normal(n), n)
            no_improve = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    n0 = len(initial_h_values)

    t_phase1_end = t0 + budget_s * 0.45
    best_h, best_c5 = optimize_local(h0, n0, rng, t_phase1_end, t0)
    best_n = n0
    
    # Phase 2: Basin hopping
    t_phase2_end = t0 + budget_s * 0.75
    while time.time() < t_phase2_end:
        perturbation_size = 0.15
        h_pert = project_feasible(best_h + perturbation_size * rng.standard_normal(n0), n0)
        t_sub_start = time.time()
        t_sub_end = min(t_sub_start + 3.0, t_phase2_end)
        h_loc, c5_loc = optimize_local(h_pert, n0, rng, t_sub_end, t_sub_start)
        if c5_loc < best_c5:
            best_h = h_loc.copy()
            best_c5 = c5_loc
    
    # Phase 3: Higher resolution attempts
    t_phase3_end = t0 + budget_s * 0.95
    for n_try in [128, 256]:
        if time.time() > t_phase3_end:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        t_sub_start = time.time()

        if c5_loc < best_c5:
            best_h = h_loc.copy()
            best_c5 = c5_loc
            best_n = n_try
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, best_n
