import numpy as np
import time
from scipy.optimize import minimize

def compute_corr_values(h):
    N = len(h)
    dx = 2.0 / N
    # Full correlation
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def exact_score(h):
    corr = compute_corr_values(h)
    return float(np.max(corr))

def project_feasible(h):
    h = np.array(h, dtype=float)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    N = len(h)
    target = N / 2.0
    # Bisection for shift t: find t such that sum(clip(h - t, 0, 1)) = target
    def sum_at(t):
        return np.sum(np.clip(h - t, 0.0, 1.0))
    lo = -2.0
    hi = 2.0
    # Ensure sum_at(lo) >= target and sum_at(hi) <= target
    # sum_at(-2) = sum(clip(h+2, 0, 1)) = sum(1) = N >= N/2 = target ✓
    # sum_at(2) = sum(clip(h-2, 0, 1)) = 0 <= target ✓
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if sum_at(mid) > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    out = np.clip(h - t, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(out)
    if abs(diff) > 1e-14:
        # Find indices not at bounds
        if diff > 0:
            mask = out < 1.0 - 1e-12
        else:
            mask = out > 1e-12
        if np.any(mask):
            n_mask = np.sum(mask)
            adj = diff / n_mask
            out[mask] += adj
            out = np.clip(out, 0.0, 1.0)
            # If still off, fix one element
            diff2 = target - np.sum(out)
            if abs(diff2) > 1e-14:
                # Find an element with enough room
                if diff2 > 0:
                    idx = np.where(out < 1.0 - abs(diff2) - 1e-15)[0]
                else:
                    idx = np.where(out > abs(diff2) + 1e-15)[0]
                if len(idx) > 0:
                    out[idx[0]] += diff2
    return out

def smooth_obj_and_grad(h, beta):
    N = len(h)
    dx = 2.0 / N
    # Compute all 2N-1 correlation values
    corr_vals = compute_corr_values(h)
    f = corr_vals  # already multiplied by dx
    
    # Logsumexp with shift for stability
    f_max = np.max(f)
    shifted = beta * (f - f_max)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    obj = f_max + np.log(sum_exp) / beta
    
    # Softmax weights
    weights = exp_vals / sum_exp
    
    # Now compute gradient
    # For each of the 2N-1 correlation terms, compute its gradient
    # and accumulate weighted sum
    grad = np.zeros(N)
    
    # The correlation output is ordered:
    # index 0 to N-2: lag = -(N-1), ..., -1 (left shifts)
    # index N-1: lag = 0 (center)
    # index N to 2N-2: lag = 1, ..., N-1 (right shifts)
    
    # For center (index N-1):
    w_center = weights[N - 1]
    grad += w_center * dx * (1.0 - 2.0 * h)
    
    # For left shifts (index i, i = 0 to N-2, lag = i - (N-1) = -(N-1-i))
    # lag = -(N-1-i), so t = N-1-i
    # L_t = sum_{j=0}^{N-1-t} h[j]*(1-h[j+t])
    # dL_t/dh[j] = (1-h[j+t]) for j in [0, N-1-t]
    # dL_t/dh[j] = -h[j-t] for j in [t, N-1]
    for i in range(N - 1):
        t = N - 1 - i
        w = weights[i]
        # dL_t/dh
        grad[:N - t] += w * dx * (1.0 - h[t:N])
        grad[t:] += w * dx * (-h[:N - t])
    
    # For right shifts (index i, i = N to 2N-2, lag = i - (N-1) = i-N+1)
    # lag = s = i-N+1, so s = 1 to N-1
    # R_s = sum_{j=0}^{N-1-s} h[j+s]*(1-h[j])
    # dR_s/dh[j] = (1-h[j-s]) for j in [s, N-1]
    # dR_s/dh[j] = -h[j+s] for j in [0, N-1-s]
    for i in range(N, 2 * N - 1):
        s = i - N + 1
        w = weights[i]
        # dR_s/dh
        grad[s:] += w * dx * (1.0 - h[:N - s])
        grad[:N - s] += w * dx * (-h[s:])
    
    return obj, grad

def smooth_obj_slsqp(h, beta):
    obj, _ = smooth_obj_and_grad(h, beta)
    return obj

def smooth_grad_slsqp(h, beta):
    _, grad = smooth_obj_and_grad(h, beta)
    return grad

def sum_constraint(h):
    N = len(h)
    return np.sum(h) - N / 2.0

def sum_constraint_grad(h):
    return np.ones(len(h))

def transfer(h, new_N):
    old_N = len(h)
    if old_N == new_N:
        return project_feasible(h)
    # Linear interpolation on bin centers
    old_centers = (np.arange(old_N) + 0.5) * 2.0 / old_N
    new_centers = (np.arange(new_N) + 0.5) * 2.0 / new_N
    h_interp = np.interp(new_centers, old_centers, h, left=h[0], right=h[-1])
    return project_feasible(h_interp)

def random_perturb(h, scale, rng):
    v = rng.normal(0, scale, len(h))
    v -= v.mean()
    cand = h + v
    return project_feasible(cand)

def optimize_at_resolution(h_start, N, betas, maxiter_per_beta, time_fn, reserve):
    best_h = project_feasible(h_start)
    best_c = exact_score(best_h)
    
    N_local = len(best_h)
    target = N_local / 2.0
    
    for beta in betas:
        remaining = time_fn()
        if remaining < reserve + 30:
            break
        try:
            res = minimize(
                smooth_obj_slsqp,
                best_h,
                args=(beta,),
                method='SLSQP',
                jac=smooth_grad_slsqp,
                bounds=[(0.0, 1.0)] * N_local,
                constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_grad}],
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-12, 'disp': False}
            )
            if res.x is not None:
                cand = project_feasible(res.x)
                c = exact_score(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
        except Exception:
            pass
    
    return best_h, best_c

def anneal_search(h_start, rng, time_fn, reserve, time_limit):
    N = len(h_start)
    state = project_feasible(h_start.copy())
    state_c = exact_score(state)
    best = state.copy()
    best_c = state_c
    
    T = 0.001
    T_min = 1e-6
    cooling = 0.999
    
    start_time = time.perf_counter()
    while True:
        remaining = time_fn()
        if remaining < reserve + 5:
            break
        elapsed_stage = time.perf_counter() - start_time
        if elapsed_stage > time_limit:
            break
        
        # Generate perturbation
        scale = 0.05 * T / 0.001  # scale with temperature
        cand = random_perturb(state, max(0.01, min(0.3, scale)), rng)
        cand_c = exact_score(cand)
        
        delta = cand_c - state_c
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            state = cand
            state_c = cand_c
            if state_c < best_c:
                best = state.copy()
                best_c = state_c
        
        T = max(T_min, T * cooling)
    
    return best, best_c

def binary_local_search(h_start, N, time_fn, reserve, time_limit):
    # Quantize to binary
    h = project_feasible(h_start)
    N_use = len(h)
    if N_use % 2 != 0:
        return None, None
    
    # Create binary: take top N/2 values
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(N_use)
    h_bin[sorted_idx[:N_use // 2]] = 1.0
    c = exact_score(h_bin)
    
    best = h_bin.copy()
    best_c = c
    start_time = time.perf_counter()
    
    improved = True
    while improved:
        improved = False
        if time_fn() < reserve + 10:
            break
        if time.perf_counter() - start_time > time_limit:
            break
        
        ones_idx = np.where(best == 1.0)[0]
        zeros_idx = np.where(best == 0.0)[0]
        
        # Try all swaps (might be expensive for large N)
        # Use limited swaps for efficiency
        max_swaps = 20000
        swap_count = 0
        for i in ones_idx:
            if swap_count > max_swaps:
                break
            for j in zeros_idx:
                swap_count += 1
                if swap_count > max_swaps:
                    break
                old_vi = best[i]
                old_vj = best[j]
                best[i] = 0.0
                best[j] = 1.0
                c_new = exact_score(best)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    improved = True
                    ones_idx = np.where(best == 1.0)[0]
                    zeros_idx = np.where(best == 0.0)[0]
                    # Restart loops with new indices
                    break
                else:
                    best[i] = old_vi
                    best[j] = old_vj
            if best[i] == 0.0:
                # We made a swap and broke inner loop
                break
    
    return best, best_c

R_ab[k] = sum_n a[n] * b[n-k]

For numpy's correlate(a, v):
It computes sum_n a[n+k] * v[n] for each k.

For 'full' mode with arrays of equal length N:

Let me verify: for i = N-1, k=0: out = sum_n a[n]*v[n]. ✓

- i < N-1, let s = (N-1)-i >= 1: sum_{n=s}^{N-1} h[n-s]*(1-h[n]) = sum_{m=0}^{N-1-s} h[m]*(1-h[m+s])

def corr_values_dx(h):
    """Compute the 2N-1 correlation values, each multiplied by dx."""
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_c5(h):
    """Compute exact C5 = max correlation * dx."""
    return float(np.max(corr_values_dx(h)))

def project_feasible(h):
    """Project h to [0,1]^N with sum = N/2."""
    h = np.asarray(h, dtype=np.float64).copy()
    h[~np.isfinite(h)] = 0.5
    np.clip(h, 0.0, 1.0, out=h)
    N = len(h)
    target = N * 0.5
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(out)
    if abs(diff) > 1e-13:
        if diff > 0:

        else:

def smooth_obj(h, beta):
    """Log-sum-exp smoothed max objective."""
    f = corr_values_dx(h)
    f_max = np.max(f)
    shifted = beta * (f - f_max)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        return f_max
    return f_max + np.log(s) / beta

def smooth_grad(h, beta):
    """Gradient of the smoothed objective."""
    N = len(h)
    dx = 2.0 / N
    f = corr_values_dx(h)
    f_max = np.max(f)
    shifted = beta * (f - f_max)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        w = np.full_like(f, 1.0 / len(f))
    else:
        w = e / s
    
    grad = np.zeros(N)
    
    # Center term (index N-1)
    wc = w[N - 1]
    grad += wc * dx * (1.0 - 2.0 * h)
    
    # Left terms (indices 0 to N-2), t = N-1-i
    for i in range(N - 1):
        t = N - 1 - i
        wi = w[i] * dx
        if wi < 1e-15:

        grad[:N - t] += wi * (1.0 - h[t:])
        grad[t:] += wi * (-h[:N - t])
    
    # Right terms (indices N to 2N-2), s = i-N+1
    for i in range(N, 2 * N - 1):
        s = i - N + 1
        wi = w[i] * dx
        if wi < 1e-15:
            continue
        grad[s:] += wi * (1.0 - h[:N - s])
        grad[:N - s] += wi * (-h[s:])
    
    return grad

    return smooth_obj(h, beta)

def slsqp_grad(h, beta):

def transfer_to(h, new_N):
    """Transfer h to a new resolution using linear interpolation."""
    old_N = len(h)
    if old_N == new_N:
        return project_feasible(h)

                    c = cc
        except Exception:
