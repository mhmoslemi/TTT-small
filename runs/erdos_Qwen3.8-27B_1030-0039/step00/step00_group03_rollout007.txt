from scipy.optimize import minimize, brentq
import time

class TimeBudget:
    def __init__(self, total, reserve=5.0):
        self.start = time.time()
        self.total = total
        self.reserve = reserve
    def remaining(self):
        return self.total - (time.time() - self.start) - self.reserve
    def exhausted(self):
        return self.remaining() <= 0

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_objective(h, beta, budget):
    if budget.exhausted():
        raise ValueError("Time budget exhausted")
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m * beta > 700:
        return m + np.log(len(corr)) / beta
    e = np.exp(beta * corr - beta * m)
    Z = np.sum(e)
    return (m + np.log(Z)) / beta

def smooth_gradient(h, beta, budget):
    if budget.exhausted():
        raise ValueError("Time budget exhausted")
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if m * beta > 700:
        # Fallback: just use exact gradient at max lag
        e = np.zeros(len(corr))
        e[np.argmax(corr)] = 1.0
    else:
        e = np.exp(beta * corr - beta * m)
        Z = np.sum(e)
        e = e / Z

def project_feasible(x, n):
    S = n / 2.0
    # Solve: sum(clip(x_i + t, 0, 1)) = S
    def f(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - S
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    # f(lo) <= 0, f(hi) >= 0
    if f(lo) > 0:
        return np.clip(x + lo, 0, 1)
    if f(hi) < 0:
        return np.clip(x + hi, 0, 1)
    t = brentq(f, lo, hi, xtol=1e-14)
    return np.clip(x + t, 0, 1)

    new_h = np.interp(new_x, old_x, old_h)
    return project_feasible(new_h, new_n)

    return float(np.max(corr) * dx)

def _corr_values(h):

    dx = 2.0 / n

def smooth_psi(h, beta):
    corr = _corr_values(h)
    m = np.max(corr)
    scaled = beta * (corr - m)
    max_s = np.max(scaled)
    if max_s > 500:
        return m + np.log(len(corr)) / beta
    e = np.exp(scaled - max_s)
    return m + (max_s + np.log(np.sum(e))) / beta

def smooth_grad(h, beta):
    corr = _corr_values(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    scaled = beta * (corr - m)
    max_s = np.max(scaled)

def exact_refine(h0, n, budget, max_rounds=500):

    best_score = evaluate_exact(best_h)
    no_gain_rounds = 0

        if no_gain_rounds > 20:
            break
        
        improved_this_round = False

        for i in range(n):
            if budget.exhausted():
                return best_h, best_score
            for j in range(i+1, n):
                # Try both directions
                # Direction 1: increase i, decrease j
                max_d1 = min(1.0 - best_h[i], best_h[j])
                # Direction 2: increase j, decrease i
                max_d2 = min(1.0 - best_h[j], best_h[i])
                
                best_d = 0.0

                        tmp = best_h.copy()
                        tmp[i] += d
                        tmp[j] -= d
                        s = evaluate_exact(tmp)
                        if s < best_s:
                            best_s = s
                            best_d = d

                    # Scan direction 2
                    for d in np.linspace(0, max_d2, 8)[1:]:
                        tmp = best_h.copy()
                        tmp[j] += d
                        tmp[i] -= d

                            best_s = s

import numpy as np
from scipy.optimize import minimize, brentq
import time

class TimeBudget:
    def __init__(self, total, reserve=5.0):
        self.start = time.time()
        self.total = total
        self.reserve = reserve
    def remaining(self):
        return self.total - (time.time() - self.start) - self.reserve
    def exhausted(self):
        return self.remaining() <= 0

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_psi(h, beta):
    corr = _corr_values(h)
    m = np.max(corr)
    scaled = beta * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        return m + np.log(len(corr)) / beta
    e = np.exp(scaled - ms)
    return m + (ms + np.log(np.sum(e))) / beta

def smooth_grad(h, beta):
    corr = _corr_values(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    scaled = beta * (corr - m)
    ms = np.max(scaled)
    if ms > 500:
        p = np.zeros(len(corr))
        p[np.argmax(corr)] = 1.0
    else:
        e = np.exp(scaled - ms)
        Z = np.sum(e)
        p = e / Z
    g = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-14:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            k = n - lag
            g[0:k] += w * dx * (1.0 - h[lag:n])
            g[lag:n] += w * dx * (-h[0:k])
        else:
            pp = -lag
            k = n - pp
            g[pp:n] += w * dx * (1.0 - h[0:k])
            g[0:k] += w * dx * (-h[pp:n])
    return g

def project_feasible(x, n):
    S = n / 2.0
    def f(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - S
    lo = -np.max(x) - 0.5
    hi = 1.0 - np.min(x) + 0.5
    if f(lo) >= 0:
        return np.clip(x + lo, 0, 1)
    if f(hi) <= 0:
        return np.clip(x + hi, 0, 1)
    t = brentq(f, lo, hi, xtol=1e-15, rtol=1e-15, maxiter=200)
    return np.clip(x + t, 0, 1)

def transfer_h(old_h, new_n):
    old_n = len(old_h)
    if old_n == new_n:
        return old_h.copy()
    if old_n == 1:
        return np.full(new_n, old_h[0])
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return project_feasible(new_h, new_n)

def smooth_optimize(h0, n, budget, betas, maxiters):
    best_h = h0.copy()
    best_score = evaluate_exact(best_h)
    S = n / 2.0
    for bi in range(len(betas)):
        if budget.exhausted():
            break
        beta = betas[bi]
        maxit = maxiters[bi]
        def fun(x):
            if budget.exhausted():
                raise ValueError("timeout")
            return smooth_psi(x, beta)
        def jac(x):
            if budget.exhausted():
                raise ValueError("timeout")
            return smooth_grad(x, beta)
        def con_eq(x):
            if budget.exhausted():
                raise ValueError("timeout")
            return np.sum(x) - S
        try:
            res = minimize(fun, best_h, jac=jac, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': con_eq}],
                           options={'maxiter': maxit, 'ftol': 1e-14})
            x_new = project_feasible(res.x, n)
            s = evaluate_exact(x_new)
            if s < best_score - 1e-14:
                best_h = x_new
                best_score = s
        except Exception:
            pass
    return best_h, best_score
