def smooth_obj_and_grad(x, beta):
    N = len(x); dx=2/N
    fvals = np.empty(2*N-1)
    fvals[0]=dx*np.dot(x,1-x)
    idx=1
    for s in 1..N-1:
       overlap = N-s
       a = dx*np.dot(x[:overlap], 1-x[s:])
       b = dx*np.dot(x[s:], 1-x[:overlap])
       fvals[idx]=a; fvals[idx+1]=b; idx+=2
    maxf = np.max(fvals)
    shifted = beta*(fvals-maxf)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    if not finite or Z=0: weights = ones/melse weights=exps/Z
    obj = maxf + np.log(Z)/beta
    grad = np.zeros(N)
    w0 = weights[0]
    if w0 != 0: grad += w0*dx*(1-2*x)
    idx=1
    for s in 1..N-1:
       wa=weights[idx]; wb=weights[idx+1]
       overlap=N-s
       if wa:
          grad[:overlap] += wa*dx*(1-x[s:])
          grad[s:] += -wa*dx*x[:overlap]
       if wb:
          grad[:overlap] += -wb*dx*x[s:]
          grad[s:] += wb*dx*(1-x[:overlap])
       idx +=2
    return obj, grad

def exact_c5(vals):
    h = np.asarray(vals, dtype=float).reshape(-1)
    if h.size == 0:
        return float("inf")
    if not np.all(np.isfinite(h)):
        return float("inf")
    h = np.clip(h, 0.0, 1.0)
    N = h.size
    dx = 2.0 / float(N)
    c = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    if not np.isfinite(c):
        return float("inf")
    return c


def overlap_values(h):
    N = h.size
    dx = 2.0 / float(N)
    out = np.empty(2 * N - 1, dtype=float)
    out[0] = dx * float(np.dot(h, 1.0 - h))
    idx = 1
    for s in range(1, N):
        m = N - s
        out[idx] = dx * float(np.dot(h[:m], 1.0 - h[s:]))
        out[idx + 1] = dx * float(np.dot(h[s:], 1.0 - h[:m]))
        idx += 2
    return out


def smooth_obj_only(x, beta):
    N = x.size
    if N == 0:
        return float("inf")
    fvals = overlap_values(x)
    maxf = float(np.max(fvals))
    shifted = beta * (fvals - maxf)
    exps = np.exp(shifted)
    Z = float(np.sum(exps))
    if (not np.isfinite(Z)) or Z <= 0.0:
        return maxf
    logn = float(np.log(float(fvals.size)))
    return float(maxf + (float(np.log(Z)) - logn) / beta)


def smooth_grad_slsqp(x, beta):
    N = x.size
    fvals = overlap_values(x)
    maxf = float(np.max(fvals))
    exps = np.exp(beta * (fvals - maxf))
    Z = float(np.sum(exps))
    if (not np.isfinite(Z)) or Z <= 0.0:
        weights = np.full(fvals.size, 1.0 / float(fvals.size))
    else:
        weights = exps / Z

    dx = 2.0 / float(N)
    g = np.zeros(N, dtype=float)
    w0 = weights[0]
    if w0 > 1e-16:
        g += w0 * dx * (1.0 - 2.0 * x)

    idx = 1
    for s in range(1, N):
        m = N - s
        wa = weights[idx]
        wb = weights[idx + 1]
        if wa > 1e-16:
            g[:m] += wa * dx * (1.0 - x[s:])
            g[s:] += -wa * dx * x[:m]
        if wb > 1e-16:
            g[:m] += -wb * dx * x[s:]
            g[s:] += wb * dx * (1.0 - x[:m])
        idx += 2
    return g


def sum_constraint(x, target):
    return float(np.sum(x) - float(target))


def sum_constraint_jac(x, target):
    return np.ones_like(x, dtype=float)


def exact_obj_slsqp(x, target):
    c = exact_c5(x)
    s = float(np.sum(x)) - float(target)
    return c + 10.0 * s * s


def exact_project_obj(x):

def check_smooth_gradient():

            xp = x.copy()
            xm = x.copy()

    return True

def cos_feasible(N, mode):

def transfer_length(h, new_N):
    h = np.asarray(h, dtype=float).reshape(-1)
    new_N = int(new_N)
    if h.size == new_N:
        return make_feasible(h)
    if h.size == 0 or new_N <= 0:
        return np.full(max(1, new_N), 0.5)
    if new_N % h.size == 0:
        factor = new_N // h.size
        y = np.repeat(h, factor)
        return make_feasible(y)
    if h.size % new_N == 0:
        factor = h.size // new_N
        y = h.reshape(new_N, factor).mean(axis=1)
        return make_feasible(y)
    src = (np.arange(h.size) + 0.5) * (2.0 / float(h.size))
    dst = (np.arange(new_N) + 0.5) * (2.0 / float(new_N))
    y = np.interp(dst, src, h, left=float(h[0]), right=float(h[-1]))
    return make_feasible(y)

def projected_gradient_pass(x0, rng, t0, budget, reserve, betas, max_inner):

def pair_local_search(x0, rng, max_evals, t0, budget, reserve):
    x = make_feasible(x0)
    best_h = make_feasible(x)
    best_c = exact_c5(best_h)
    N = x.size
    evals = 0
    max_evals = int(max_evals)
    fractions = [1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.34, 1.0]
    while evals < max_evals and remaining_budget(t0, budget) > reserve + 5.0:
        i = int(rng.integers(N))
        j = int(rng.integers(N))
        if i == j:
            continue
        hi = float(x[i])
        hj = float(x[j])
        max_pos = min(1.0 - hi, hj)
        max_neg = min(hi, 1.0 - hj)
        if max_pos < 1e-14 and max_neg < 1e-14:
            continue
        for frac in fractions:
            dp = float(frac) * max_pos
            if dp > 1e-16:
                cand = x.copy()
                cand[i] += dp
                cand[j] -= dp
                cand = make_feasible(cand)
                c = exact_c5(cand)
                evals += 1
                if np.isfinite(c) and c < best_c - 1e-14:
                    best_c = c
                    best_h = cand
                    x = cand
                    break
            dn = float(frac) * max_neg
            if dn > 1e-16:
                cand = x.copy()
                cand[i] -= dn
                cand[j] += dn
                cand = make_feasible(cand)
                c = exact_c5(cand)

                if np.isfinite(c) and c < best_c - 1e-14:

                    best_h = cand
                    x = cand
                    break
            if evals >= max_evals:
                break
            if remaining_budget(t0, budget) < reserve + 5.0:
                break

        if np.isfinite(c2) and c2 < best_c - 1e-14:

        h_cur = st2
        if smooth_ok and remaining_budget(t0, budget) > reserve + 220.0:

                best_c = c_s
                best_h = h_s.copy()
            h_cur = h_s

        if remaining_budget(t0, budget) > reserve + 120.0:
            h_e, c_e = exact_slsqp_pass(h_cur, 30, t0, budget, reserve)
            if np.isfinite(c_e) and c_e < best_c - 1e-14:
                best_c = c_e

            h_cur = h_e

        if grad_ok and remaining_budget(t0, budget) > reserve + 80.0:
            h_p, c_p = projected_gradient_pass(h_cur, rng, t0, budget, reserve, pg_betas, 50)
            if np.isfinite(c_p) and c_p < best_c - 1e-14:
                best_c = c_p
                best_h = h_p.copy()
            h_cur = h_p

        pevals = pair_evals_for(N2)
        if remaining_budget(t0, budget) > reserve + 30.0:
            h_pr, c_pr = pair_local_search(best_h, rng, pevals, t0, budget, reserve)
            if np.isfinite(c_pr) and c_pr < best_c - 1e-14:
                best_c = c_pr
                best_h = h_pr.copy()

        back = transfer_length(best_h, N0)
        cb = exact_c5(back)
        if np.isfinite(cb) and cb < best_c - 1e-14:
            best_c = cb
            best_h = back.copy()

    N3 = 3 * N0
    if N3 <= 180 and remaining_budget(t0, budget) > reserve + 180.0:
        st3 = transfer_length(best_h, N3)
        c3 = exact_c5(st3)
        if np.isfinite(c3) and c3 < best_c - 1e-14:
            best_c = c3

        h_cur = st3
        if smooth_ok and remaining_budget(t0, budget) > reserve + 150.0:
            h_s, c_s = slsqp_smooth_pass(h_cur, [30.0, 120.0, 400.0], 30, 1e-9, grad_ok, t0, budget, reserve)

                best_h = h_s.copy()
            h_cur = h_s

        if remaining_budget(t0, budget) > reserve + 90.0:
            h_e, c_e = exact_slsqp_pass(h_cur, 25, t0, budget, reserve)
            if np.isfinite(c_e) and c_e < best_c - 1e-14:

            if np.isfinite(c_p) and c_p < best_c - 1e-14:

            h_cur = h_p

        pevals = pair_evals_for(N3)

    if remaining_budget(t0, budget) > reserve + 70.0:
        h_co, c_co = cobyla_exact_pass(best_h, t0, budget, reserve, 80)

        pevals = pair_evals_for(best_h.size)
