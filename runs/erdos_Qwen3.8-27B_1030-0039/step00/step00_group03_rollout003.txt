def is_feasible(h, tol=1e-10):
    if len(h) == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1 + tol):
        return False
    n = len(h)
    if abs(np.sum(h) - n/2) > tol * n:

def repair_feasible(raw_h):
    n = len(raw_h)
    target = n / 2.0
    
    # Find offset tau such that sum(clip(raw_h + tau, 0, 1)) = target
    def f(tau):
        return np.sum(np.clip(raw_h + tau, 0, 1)) - target
    
    lo = -raw_h.max()
    hi = 1.0 - raw_h.min()
    
    if f(lo) > 0 or f(hi) < 0:
        return None
    
    # Bisect (safer than brentq for piecewise linear)
    for _ in range(200):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-15:
            break
    
    tau = (lo + hi) / 2
    result = np.clip(raw_h + tau, 0, 1)
    
    # Check feasibility
    if abs(np.sum(result) - target) > 1e-8:
        # Try to fix with small adjustments
        diff = target - np.sum(result)
        if diff > 0:
            # Need to increase some entries
            mask = result < 1.0
            n_inc = mask.sum()
            if n_inc > 0:
                result[mask] += diff / n_inc
                result = np.clip(result, 0, 1)
        else:
            mask = result > 0
            n_dec = mask.sum()
            if n_dec > 0:
                result[mask] += diff / n_dec
                result = np.clip(result, 0, 1)

        return 0.0, exact_score(h)

def repair_feasible(raw_h):
    raw_h = np.asarray(raw_h, dtype=np.float64)
    n = len(raw_h)
    if n == 0:
        return None
    target = n * 0.5
    
    # Bisection to find tau
    lo = -raw_h.max() - 0.1
    hi = 1.0 - raw_h.min() + 0.1
    
    def f(tau):
        return float(np.sum(np.clip(raw_h + tau, 0.0, 1.0)) - target)
    
    if f(lo) >= 0:
        # Already too high even at minimum tau
        # This shouldn't happen if target is in [0, n]
        pass
    if f(hi) <= 0:
        pass
    
    # Bisection
    for _ in range(200):
        mid = (lo + hi) * 0.5
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-15:
            break
    
    tau = (lo + hi) * 0.5
    result = np.clip(raw_h + tau, 0.0, 1.0)
    
    # Fix any residual sum error
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    
    if not is_feasible(result, tol=1e-7):
        return None
    return result

def pair_line_optimize(h, i, j, max_iter=35):
    """Find optimal t for pair (i,j). Returns (best_t, new_h, new_score)."""
    n = len(h)
    dx = 2.0 / n
    
    lb = max(-h[i], h[j] - 1.0)
    ub = min(1.0 - h[i], h[j])
    if ub - lb < 1e-14:
        return 0.0, None, None
    
    invphi = 0.5 * (np.sqrt(5.0) - 1.0)
    a, b = lb, ub
    
    # Initial two points
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    
    work = np.empty_like(h)
    
    np.copyto(work, h)
    work[i] += c
    work[j] -= c
    fc = float(np.max(np.correlate(work, 1.0 - work, mode='full')) * dx)
    
    np.copyto(work, h)
    work[i] += d
    work[j] -= d
    fd = float(np.max(np.correlate(work, 1.0 - work, mode='full')) * dx)
    
    for _ in range(max_iter):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            np.copyto(work, h)
            work[i] = h[i] + c
            work[j] = h[j] - c
            fc = float(np.max(np.correlate(work, 1.0 - work, mode='full')) * dx)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            np.copyto(work, h)
            work[i] = h[i] + d
            work[j] = h[j] - d
            fd = float(np.max(np.correlate(work, 1.0 - work, mode='full')) * dx)
    
    t_opt = 0.5 * (a + b)
    # Clamp to feasible
    t_opt = max(lb, min(ub, t_opt))
    
    new_h = h.copy()
    new_h[i] = h[i] + t_opt
    new_h[j] = h[j] - t_opt
    # Round to exact 0 or 1 if very close
    if new_h[i] < 1e-11:
        new_h[i] = 0.0
    elif new_h[i] > 1.0 - 1e-11:
        new_h[i] = 1.0
    if new_h[j] < 1e-11:
        new_h[j] = 0.0
    elif new_h[j] > 1.0 - 1e-11:
        new_h[j] = 1.0
    
    new_score = exact_score(new_h)
    return t_opt, new_h, new_score

    n = len(h)
