                    hi = min(hi_bound, h[j])
                    if hi - lo < 1e-14:
                        continue

                    ci = np.zeros(2 * N - 1)
                    m1 = (lags <= i) & (lags > i - N)
                    m2 = (lags >= -i) & (lags < N - i)
                    ci[m1] = 1.0 - h[i - lags[m1].astype(int)]
                    ci[m2] -= h[i + lags[m2].astype(int)]

                    cj = np.zeros(2 * N - 1)
                    m1j = (lags <= j) & (lags > j - N)
                    m2j = (lags >= -j) & (lags < N - j)
                    cj[m1j] = 1.0 - h[j - lags[m1j].astype(int)]
                    cj[m2j] -= h[j + lags[m2j].astype(int)]

                    c = dx * (ci - cj)

                    a, b = lo, hi
                    c_pt = b - gr * (b - a)
                    d_pt = a + gr * (b - a)
                    fc = np.max(corr + c_pt * c)
                    fd = np.max(corr + d_pt * c)
                    for _ in range(12):
                        if fc < fd:
                            b = d_pt
                            d_pt = c_pt
                            fd = fc
                            c_pt = b - gr * (b - a)
                            fc = np.max(corr + c_pt * c)
                        else:
                            a = c_pt
                            c_pt = d_pt
                            fc = fd
                            d_pt = a + gr * (b - a)
                            fd = np.max(corr + d_pt * c)

                    best_delta = (a + b) / 2.0
                    new_c5 = np.max(corr + best_delta * c)

                    if new_c5 < best_c5 - 1e-13:
                        h[i] += best_delta
                        h[j] -= best_delta
                        corr += best_delta * c
                        best_c5 = float(new_c5)
                        improved = True

            sweep += 1
            if sweep % 100 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                best_c5 = float(np.max(corr))
            if not improved:
                break

        return h, best_c5

    deadline = start_time + budget_s

    # Phase 1: Optimize the given initial solution
    d1 = min(deadline - 60.0, start_time + budget_s * 0.3)
    h, c5 = pairwise_optimize(h0, d1)
    best_h = h.copy()
    best_c5 = c5

    # Phase 2: Try larger N via interpolation
    for N_new in [64, 96, 128, 160, 200]:
        if time.time() > deadline - 60.0:
            break
        x_old = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, N_new)
        h_int = np.interp(x_new, x_old, best_h)
        target_sum = N_new / 2.0
        h_int = project(h_int, target_sum)
        if abs(np.sum(h_int) - target_sum) > 1e-6:
            continue
        alloc = min(budget_s * 0.15, deadline - time.time() - 30.0)
        if alloc < 10.0:
            break
        h_try, c5_try = pairwise_optimize(h_int, time.time() + alloc)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try

    # Phase 3: Random restarts on best solution
    if time.time() < deadline - 30.0:
        for attempt in range(3):
            if time.time() > deadline - 30.0:
                break
            scale = 0.02 * (attempt + 1)
            h_pert = best_h + rng.normal(0.0, scale, len(best_h))
            h_pert = project(h_pert, len(best_h) / 2.0)
            alloc = min(120.0, deadline - time.time() - 20.0)
            if alloc < 10.0:
                break
            h_try, c5_try = pairwise_optimize(h_pert, time.time() + alloc)
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try

    # Phase 4: Final polish
    if time.time() < deadline - 5.0:
        h, c5 = pairwise_optimize(best_h, deadline - 2.0, max_sweeps=200)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return best_h, best_c5, len(best_h)
