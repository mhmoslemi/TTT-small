import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def get_gradient(h, lag, n, dx):
    offset = lag - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        k = offset
        if k == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += dx * (1.0 - h[k:])
            grad[k:] += dx * (-h[:n-k])
    else:
        k = -offset
        if k == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        else:
            grad[k:] += dx * (1.0 - h[:n-k])
            grad[:n-k] += dx * (-h[k:])
    return grad

def softmax_gradient(h, n, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    max_val = np.max(vals)
    exps = np.exp((vals - max_val) / temp)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for lag in range(len(weights)):
        if weights[lag] > 1e-12:
            grad += weights[lag] * get_gradient(h, lag, n, dx)
    return grad + np.mean(grad)

def project_onto_constraint(h, target_sum, tol=1e-12):
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target_sum
        if abs(diff) < tol:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(free):
            break
        h[free] -= diff / np.sum(free)
    return h

def optimize_from_start(h, n, dx, max_time=50.0):
    target_sum = n / 2.0
    h = project_onto_constraint(h.copy(), target_sum)
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    start = time.time()
    temp = 0.01
    step_size = 0.005
    stall = 0
    while time.time() - start < max_time:
        grad = softmax_gradient(h, n, dx, temp)
        h_new = h - step_size * grad
        h_new = project_onto_constraint(h_new, target_sum)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5 - 1e-10:
            h = h_new
            current_c5 = new_c5
            stall = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stall += 1
            step_size *= 0.85
        if stall > 30:
            temp = max(0.003, temp * 0.7)
            step_size = 0.005
            stall = 0
        temp = max(0.003, temp * 0.999)
    return best_h, best_c5

def polish_scipy(h, n, dx, max_iter=500):
    target_sum = n / 2.0
    def obj(x):
        hh = np.clip(x, 0.0, 1.0)
        return compute_c5(hh, dx)
    def obj_grad(x):
        hh = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hh, 1.0 - hh, mode="full")
        lags = np.where(corr == np.max(corr))[0]
        grad = np.zeros(n)
        for lag in lags:
            grad = grad + get_gradient(hh, lag, n, dx)
        grad = grad / len(lags)
        return grad
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n)]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.fun < compute_c5(h, dx):
            res_h = project_onto_constraint(np.clip(res.x, 0.0, 1.0), target_sum)
            return res_h, compute_c5(res_h, dx)
    except Exception:
        pass
    return h, compute_c5(h, dx)
