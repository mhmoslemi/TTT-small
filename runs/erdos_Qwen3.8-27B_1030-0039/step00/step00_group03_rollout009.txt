import numpy as np
from scipy.optimize import minimize, brentq
import time

class TimeLimit(Exception):
    pass

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    n = len(x)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01
    if F(t_lo) >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if F(t_hi) <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-15)
    out = np.clip(x + t, 0.0, 1.0)
    # Fine-tune for exact sum
    diff = target_sum - out.sum()
    if abs(diff) > 1e-12:
        mask = (out > 1e-9) & (out < 1.0 - 1e-9)
        if mask.sum() > 0:
            out[mask] += diff / mask.sum()
            out = np.clip(out, 0.0, 1.0)
    return out

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    p = e / Z
    val = (m + np.log(Z)) / beta
    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += w * (-h[d:])
            grad[d:] += w * (1.0 - h[:n-d])
        else:
            k = -d
            grad[k:] += w * (-h[:n-k])
            grad[:n-k] += w * (1.0 - h[k:])
    return val, grad

def transfer_to_new_n(h_old, n_new):
    m = len(h_old)
    x_old = np.linspace(0.0, 2.0, m)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new / 2.0)

def run_slsqp_stage(h, deadline, beta_list=None):
    if beta_list is None:
        beta_list = [20.0, 100.0, 500.0]
    n = len(h)
    target_sum = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    
    for beta in beta_list:
        rem = deadline - time.time()
        if rem < 15:
            break
        maxiter = max(20, min(150, int(rem * 3)))
        
        def _f(x):
            if time.time() > deadline:
                raise TimeLimit()
            v, _ = smooth_obj_grad(x, beta)
            return v
        def _g(x):
            if time.time() > deadline:
                raise TimeLimit()
            _, g = smooth_obj_grad(x, beta)
            return g
        def _c(x):
            return x.sum() - target_sum
        
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': _c}]
        try:
            res = minimize(_f, h_curr, jac=_g, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
                sc = evaluate_exact(h_new)
                if sc < best_score:
                    best_h = h_new.copy()
                    best_score = sc
                h_curr = h_new.copy()
        except TimeLimit:
            break
        except Exception:
            break
    return best_h, best_score

def pair_descent(h, deadline, max_sweeps=20, n_eval=30):
    n = len(h)
    h = h.copy()
    best_score = evaluate_exact(h)
    target_sum = n / 2.0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return h, best_score
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-14:
                    continue
                # Line search
                best_d = 0.0
                best_s = best_score
                for d in np.linspace(lo, hi, n_eval):
                    h[i] += d
                    h[j] -= d
                    s = evaluate_exact(h)
                    h[i] -= d
                    h[j] += d
                    if s < best_s - 1e-14:
                        best_s = s
                        best_d = d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    new_s = evaluate_exact(h)
                    if new_s < best_score - 1e-14:
                        best_score = new_s
                        improved = True
        if not improved:
            break
    return h, best_score

import numpy as np
from scipy.optimize import minimize, brentq
import time

class TimeLimit(Exception):
    pass

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    n = len(x)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if fhi <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-15)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    p = e / Z
    val = (m + np.log(Z)) / beta
    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += w * (-h[d:])
            grad[d:] += w * (1.0 - h[:n-d])
        else:
            k = -d
            grad[k:] += w * (-h[:n-k])
            grad[:n-k] += w * (1.0 - h[k:])
    return val, grad

    h_interp = np.interp(x_new, x_old, h_old)

    target_sum = n / 2.0

            v, _ = smooth_obj_grad(x, beta)
            return v

            if time.time() > deadline:
                raise TimeLimit()
            _, g = smooth_obj_grad(x, beta)
            return g
        def _c(x):
            return x.sum() - target_sum
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': _c}]
        try:
            res = minimize(_f, h_curr, jac=_g, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})

                    best_score = sc

def pair_descent(h, deadline, max_sweeps=15):

    h = h.copy()
    best_score = evaluate_exact(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
