def transfer_up(h, factor):

def transfer_down(h, factor):

    n = len(h)

    h2 = h[:new_n*factor].reshape(new_n, factor).mean(axis=1)

def binary_local_search(h_bin, max_swaps, t0, budget):

    N = len(h_bin)

        ones = np.where(h_curr == 1)[0]

                h_curr[j] = 1

                        break
                else:
                    h_curr[i] = 1
                    h_curr[j] = 0
            if swaps_done >= max_swaps:
                break
        if not improved:
            break

def repair_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.5)

        else:
            mask = h_out > 1e-10

def compute_fvals(h):
    N = len(h)
    dx = 2.0 / N
    omh = 1.0 - h
    fvals = np.empty(2 * N - 1)
    fvals[N - 1] = dx * np.sum(h * omh)
    for k in range(1, N):
        fvals[N - 1 + k] = dx * np.sum(h[k:] * omh[:-k])
        fvals[N - 1 - k] = dx * np.sum(h[:-k] * omh[k:])
    return fvals

    fvals = compute_fvals(x)
    max_f = np.max(fvals)
    exps = np.exp(beta * (fvals - max_f))
    sum_exps = np.sum(exps)
    if sum_exps <= 0:
        return max_f
    return max_f + np.log(sum_exps) / beta

    N = len(x)
    dx = 2.0 / N
    fvals = compute_fvals(x)
    max_f = np.max(fvals)
    exps = np.exp(beta * (fvals - max_f))
    sum_exps = np.sum(exps)
    if sum_exps <= 0:
        return np.zeros(N)
    w = exps / sum_exps
    omh = 1.0 - x
    grad = np.zeros(N)
    grad += w[N - 1] * dx * (1.0 - 2.0 * x)
    for k in range(1, N):

        if wk > 1e-15:
            m = N - k
            grad[k:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * x[k:]
        wk2 = w[N - 1 - k]
        if wk2 > 1e-15:
            m = N - k
            grad[:m] += wk2 * dx * omh[k:]
            grad[k:] -= wk2 * dx * x[:m]
    return grad

    dx = 2.0 / N
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
