import numpy as np
from scipy.optimize import brentq
from scipy.signal import correlate
import time


class _TimeExpired(Exception):
    pass


def _score(h, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def _repair(raw):
    x = np.asarray(raw, dtype=np.float64)
    if x.ndim != 1 or len(x) == 0 or not np.all(np.isfinite(x)):
        return None
    n = len(x)
    S = n / 2.0

    def F(tau):
        return float(np.sum(np.clip(x + tau, 0.0, 1.0)) - S)

    L = float(-np.max(x)) - 0.01
    U = 1.0 - float(np.min(x)) + 0.01
    if L >= U:
        return None
    if F(L) > 0 or F(U) < 0:
        L, U = -1.0, 2.0
        if F(L) > 0 or F(U) < 0:
            return None
    try:
        tau = brentq(F, L, U, xtol=1e-13, rtol=1e-14, maxiter=300)
    except Exception:
        return None
    z = np.clip(x + tau, 0.0, 1.0)
    if abs(float(np.sum(z)) - S) > 1e-6:
        return None
    return z


def _golden(func, lo, hi, max_iter=45, tol=1e-12):
    if hi - lo < tol:
        return (lo + hi) / 2.0
    ip = (np.sqrt(5.0) + 1.0) / 2.0
    a, b = lo, hi
    c = b - (b - a) / ip
    d = a + (b - a) / ip
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / ip
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / ip
            fd = func(d)
    return (a + b) / 2.0


def _pair_descent_one(h, i, j, n):
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-13:
        return 0.0, _score(h, n)

    def g(t):
        h[i] += t
        h[j] -= t
        s = _score(h, n)
        h[i] -= t
        h[j] += t
        return s

    t_g = _golden(g, L, U, max_iter=40, tol=1e-13)
    candidates = [0.0, t_g, L, U, (L + U) * 0.5]
    best_t = 0.0
    best_v = g(0.0)
    for tc in candidates:
        tc = max(L, min(U, tc))
        v = g(tc)
        if v < best_v:
            best_v = v
            best_t = tc
    return best_t, best_v


def _spot_pair_descent(h, n, deadline, max_passes=20):
    cur = h.copy().astype(np.float64)
    cur_s = _score(cur, n)
    imp_thresh = 1e-10
    for _pass in range(max_passes):
        if time.time() > deadline - 3.0:
            break
        improved = False
        if n <= 70:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            rng = np.random.RandomState(7919 + _pass)
            seen = set()
            pairs = []
            target = min(400, n * (n - 1) // 4)
            while len(pairs) < target:
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j and (i, j) not in seen:
                    seen.add((i, j))
                    pairs.append((i, j))
        for (i, j) in pairs:
            if time.time() > deadline - 2.0:
                break
            t_opt, new_s = _pair_descent_one(cur, i, j, n)
            if abs(t_opt) > 1e-12 and new_s < cur_s - imp_thresh:
                cur[i] += t_opt
                cur[j] -= t_opt
                cur = np.clip(cur, 0.0, 1.0)
                if abs(float(np.sum(cur)) - n / 2.0) > 1e-8 * n:
                    cur[i] -= t_opt
                    cur[j] += t_opt
                    rep = _repair(cur)
                    if rep is not None:
                        cur = rep
                        new_s = _score(cur, n)
                        if new_s >= cur_s:
                            continue
                    else:
                        continue
                cur_s = new_s
                improved = True
        if not improved:
            break
    return cur, cur_s

def _sa_step(h, n, T, deadline):
    rng = np.random.RandomState()
    i = rng.randint(n)
    j = rng.randint(n)
    if i == j:
        return h, _score(h, n)
    t = rng.uniform(-0.4, 0.4)
    hi_new = h[i] + t
    hj_new = h[j] - t
    if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
        return h, _score(h, n)
    h[i] = hi_new
    h[j] = hj_new
    s = _score(h, n)
    return h, s


def _simulated_anneal(h, n, T_start, T_end, n_iter, deadline):
    cur = h.copy().astype(np.float64)
    cur_s = _score(cur, n)
    best = cur.copy()
    best_s = cur_s
    rng = np.random.RandomState(42)
    for it in range(n_iter):
        if time.time() > deadline - 5.0:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iter - 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        t = rng.uniform(-0.5, 0.5)
        hi_new = cur[i] + t
        hj_new = cur[j] - t
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        cur[i] = hi_new
        cur[j] = hj_new
        new_s = _score(cur, n)
        delta = new_s - cur_s
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_s = new_s
            if new_s < best_s:
                best_s = new_s
                best = cur.copy()
        else:
            cur[i] = hi_new - t
            cur[j] = hj_new + t
    return best, best_s


def _interpolate_to(h_src, n_tgt):

    src_pos = np.linspace(0, n_src - 1, n_src)
    tgt_pos = np.linspace(0, n_tgt - 1, n_tgt)
    h_new = np.interp(tgt_pos, src_pos, h_src)
