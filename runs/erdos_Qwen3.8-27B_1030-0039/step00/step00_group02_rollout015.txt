import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    """Project onto {0<=x<=1, sum(x)=n/2} using iterative water-filling."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = diff / cnt
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = (-diff) / cnt
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_max(c_dx, temperature):
    """Compute log-sum-exp of c_dx with temperature."""
    m = np.max(c_dx)
    if m * temperature > 40:
        active = c_dx >= m - 1e-10
        return m + np.log(np.sum(active)) / temperature
    shifted = (c_dx - m) * temperature
    return m + np.log(np.sum(np.exp(shifted))) / temperature

def smooth_gradient(h, temperature):
    """Compute gradient of smoothed max correlation objective."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full')
    c_dx = c * dx
    
    m = np.max(c_dx)
    if m * temperature > 40:
        active = c_dx >= m - 1e-10
        w = np.zeros_like(c_dx)
        w[active] = 1.0 / np.sum(active)
    else:
        shifted = (c_dx - m) * temperature
        exps = np.exp(shifted)
        w = exps / np.sum(exps)
    
    # grad[j] = dx * sum_m w[m] * dc(m)/dh[j]
    # For shift k = m-(n-1):
    # dc/dh[j] = (1-h[j+k]) * I[max(0,-k)<=j<=min(n-1,n-1-k)] - h[j-k] * I[max(0,k)<=j<=min(n-1,n-1+k)]
    
    grad = np.zeros(n)
    for m_idx in range(2*n-1):
        wm = w[m_idx]
        if abs(wm) < 1e-15:
            continue
        k = m_idx - (n-1)
        # First term: (1-h[j+k]) for j in [max(0,-k), min(n-1, n-1-k)]
        j_lo = max(0, -k)
        j_hi = min(n-1, n-1-k)
        if j_lo <= j_hi:
            if k >= 0:
                # h[j+k] for j in [j_lo, j_hi], j+k in [j_lo+k, j_hi+k] = [j_lo+k, n-1]
                grad[j_lo:j_hi+1] += wm * (1.0 - h[j_lo+k:j_hi+1+k])
            else:
                # h[j+k] for j in [j_lo, j_hi], j+k in [j_lo+k, j_hi+k] = [0, j_hi+k]
                grad[j_lo:j_hi+1] += wm * (1.0 - h[j_lo+k:j_hi+1+k])
        # Second term: -h[j-k] for j in [max(0,k), min(n-1, n-1+k)]
        j2_lo = max(0, k)
        j2_hi = min(n-1, n-1+k)
        if j2_lo <= j2_hi:
            if k >= 0:
                # h[j-k] for j in [j2_lo, j2_hi], j-k in [0, j2_hi-k]
                grad[j2_lo:j2_hi+1] -= wm * h[j2_lo-k:j2_hi+1-k]
            else:
                # h[j-k] for j in [j2_lo, j2_hi], j-k in [j2_lo-k, j2_hi-k]
                grad[j2_lo:j2_hi+1] -= wm * h[j2_lo-k:j2_hi+1-k]
    
    grad *= dx
    return grad

def optimize_smoothed(h, n, temperature, n_iters=2000, lr=0.01):
    """Projected gradient descent on smoothed objective."""
    best_h = h.copy()
    best_val = evaluate_c5(h)
    h = h.copy()
    
    for it in range(n_iters):
        grad = smooth_gradient(h, temperature)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        val = evaluate_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        # Decay learning rate
        if it % 500 == 499:
            lr *= 0.5
    
    return best_h, best_val

def coordinate_descent(h, n, n_sweeps=10):
    """Paired coordinate descent with line search."""
    h = h.copy()
    best_val = evaluate_c5(h)
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Try to find optimal transfer from i to j or j to i
                # We want to maximize min reduction in max correlation
                # Simple: try a few delta values
                h_orig = h[i]
                h_j_orig = h[j]
                
                for direction in [1, -1]:

                    # Max delta: if direction=1: min(1-h[i], h[j])
                    # Max delta: if direction=-1: min(h[i], 1-h[j])
                    if direction == 1:
                        max_d = min(1.0 - h[i], h[j])
                    else:
                        max_d = min(h[i], 1.0 - h[j])
                    
                    if max_d < 1e-10:
                        continue
                    
                    # Try a few values of delta
                    for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                        delta = frac * max_d
                        h[i] = h_orig + direction * delta
                        h[j] = h_j_orig - direction * delta
                        if h[i] < -1e-12 or h[i] > 1+1e-12 or h[j] < -1e-12 or h[j] > 1+1e-12:
                            h[i] = h_orig
                            h[j] = h_j_orig

                        val = evaluate_c5(h)
                        if val < best_val - 1e-12:
                            best_val = val
                            improved = True
                        else:
                            h[i] = h_orig
                            h[j] = h_j_orig
                    h[i] = h_orig
                    h[j] = h_j_orig
        
        if not improved:
            break
    
    return h, best_val

import numpy as np
import time

def project_feasible(h, n):
    """Project h onto the feasible set {0<=x_i<=1, sum(x)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def c5_value(h):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_smoothed(h, temp):
    """Gradient of smoothed max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    if m * temp > 35:
        w = np.zeros(len(c))
        mask = c >= m - 1e-10
        w[mask] = 1.0 / mask.sum()
    else:
        es = np.exp((c - m) * temp)
        w = es / es.sum()
    
    g = np.zeros(n)
    for mi in range(len(c)):
        wm = w[mi]
        if wm < 1e-15:
            continue
        k = mi - (n - 1)
        # First term: (1-h[j+k]) for j in [max(0,-k), min(n-1, n-1-k)]
        jlo = max(0, -k)
        jhi = min(n - 1, n - 1 - k)
        if jlo <= jhi:
            idx = np.arange(jlo, jhi + 1)
            g[idx] += wm * (1.0 - h[idx + k])
        # Second term: -h[j-k] for j in [max(0,k), min(n-1, n-1+k)]
        j2lo = max(0, k)
        j2hi = min(n - 1, n - 1 + k)
        if j2lo <= j2hi:
            idx2 = np.arange(j2lo, j2hi + 1)
            g[idx2] -= wm * h[idx2 - k]
    
    return g * dx

            cnt = mask.sum()

        else:
