    """Project v onto {x: 0 <= x_i <= 1, sum(x) = target}."""
    n = len(v)
    lo, hi = -5.0, 5.0

        if s > target:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    max_c = np.max(c)

import numpy as np

from scipy.optimize import minimize_scalar

def project_onto_capped_simplex(v, target):
    n = len(v)
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    return np.clip(v - mu, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def apply_transfer(h, i, j, t):
    h[i] += t
    h[j] -= t

def revert_transfer(h, i, j, t):
    h[i] -= t
    h[j] += t

def eval_transfer(h, i, j, t):
    apply_transfer(h, i, j, t)
    c5 = evaluate_c5(h)
    revert_transfer(h, i, j, t)
    return c5

def eval_swap(h, i, j):

    c5 = evaluate_c5(h)
    h[i], h[j] = h[j], h[i]
    return c5

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    max_c = np.max(c)
    shifted = np.exp(alpha * (c - max_c))
    s = np.sum(shifted)
    F = (max_c + np.log(s)) / alpha
    w = shifted / s
    grad = np.zeros(n)
    offsets = np.arange(2 * n - 1) - (n - 1)
    for i in range(n):
        v1 = (i + offsets >= 0) & (i + offsets < n)
        v2 = (i - offsets >= 0) & (i - offsets < n)
        grad[i] = np.dot(w[v1], 1.0 - h[i + offsets[v1]]) - np.dot(w[v2], h[i - offsets[v2]])
    grad *= dx
    return F, grad
