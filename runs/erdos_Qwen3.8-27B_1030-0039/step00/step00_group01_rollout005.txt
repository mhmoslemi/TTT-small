The constraints C(k) ≤ T are bilinear (involving products h_i * h_j), so it's not a linear feasibility problem. But we can use the following approach:

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum=None):
    """Project h to [0,1]^n with sum = target_sum using water filling."""
    if target_sum is None:
        target_sum = len(h) / 2.0
    n = len(h)
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add mass: add to smallest h first
        order = np.argsort(h)
        remaining = diff
        for idx in order:
            add = min(1.0 - h[idx], remaining)
            h[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass: remove from largest h first
        order = np.argsort(-h)
        remaining = -diff
        for idx in order:
            rem = min(h[idx], remaining)
            h[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break
    return h

def compute_gradient(h, active_lags):
    """Compute subgradient of max C(k) at h, using given active lags."""
    n = len(h)
    g = np.zeros(n)
    for lag in active_lags:
        if lag == 0:
            g += (1.0 - 2.0 * h)
        else:
            i_idx = np.arange(n)
            idx1 = i_idx - lag  # i - lag
            idx2 = i_idx + lag  # i + lag
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            term = np.zeros(n)
            term[valid1] += (1.0 - h[idx1[valid1]])
            term[valid2] -= h[idx2[valid2]]
            g += term
    return g

def two_point_line_search(h, i, j):
    """Line search along direction: h[i] -= t, h[j] += t."""
    h_i = h[i]
    h_j = h[j]
    t_max = min(h_i, 1.0 - h_j)
    if t_max <= 1e-12:
        return 0, evaluate_c5(h)
    
    best_t = 0.0
    best_s = np.max(np.correlate(h, 1.0 - h, mode="full"))
    
    # Coarse search
    n_evals = 20
    for t in np.linspace(0, t_max, n_evals)[1:]:
        h[i] = h_i - t
        h[j] = h_j + t
        s = np.max(np.correlate(h, 1.0 - h, mode="full"))
        if s < best_s:
            best_s = s
            best_t = t
    
    # Fine search around best_t
    if 0 < best_t < t_max:
        step = t_max / n_evals
        t_lo = max(0.0, best_t - step)
        t_hi = min(t_max, best_t + step)
        for t in np.linspace(t_lo, t_hi, 10):
            h[i] = h_i - t
            h[j] = h_j + t
            s = np.max(np.correlate(h, 1.0 - h, mode="full"))
            if s < best_s:
                best_s = s
                best_t = t
    
    # Apply
    h[i] = h_i - best_t
    h[j] = h_j + best_t
    return best_t, best_s

def continuous_refine(h, max_iters=10000, time_limit=None, start_time=None):
    """Iterative local optimization using gradient-guided 2-point transfers."""
    n = len(h)
    h = project_feasible(h.copy())
    current_score = evaluate_c5(h)
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        # Compute correlation and find active set
        corr = np.correlate(h, 1.0 - h, mode="full")
        M = np.max(corr)
        
        # Active set: lags within 2% of max
        threshold = M * 0.98
        active_idx = np.where(corr >= threshold)[0]
        lags = active_idx - (n - 1)
        
        # Compute gradient
        g = compute_gradient(h, lags)
        g -= np.mean(g)  # zero-sum projection
        
        # Find best 2-point transfer
        can_dec = h > 1e-10
        can_inc = h < 1.0 - 1e-10
        
        if not np.any(can_dec) or not np.any(can_inc):
            break
        
        # Score each position for being a source (decrease) or destination (increase)
        source_score = g.copy()
        source_score[~can_dec] = -np.inf
        dest_score = g.copy()
        dest_score[~can_inc] = np.inf
        
        i = np.argmax(source_score)
        j = np.argmin(dest_score)
        
        if source_score[i] <= dest_score[j] + 1e-10:
            # No clear improvement; try random pairs
            dec_cands = np.where(can_dec)[0]
            inc_cands = np.where(can_inc)[0]
            best_improvement = 0
            best_pair = None
            for _ in range(20):
                ri = dec_cands[np.random.randint(len(dec_cands))]
                rj = inc_cands[np.random.randint(len(inc_cands))]
                imp = g[ri] - g[rj]
                if imp > best_improvement:
                    best_improvement = imp
                    best_pair = (ri, rj)
            if best_pair is None or best_improvement <= 0:
                # Try multi-point: top 5 sources to top 5 destinations
                top_src = np.argsort(source_score)[-5:]
                top_dst = np.argsort(dest_score)[:5]
                # Transfer small amount from each source to each destination
                total_mass = min(np.sum(h[top_src]) * 0.1, (len(top_dst) * (1.0 - np.mean(h[top_dst]))) * 0.1)
                if total_mass < 1e-10:
                    # Stuck, add small random perturbation
                    p1 = np.random.randint(n)
                    p2 = np.random.randint(n)
                    if h[p1] > 0.01 and h[p2] < 0.99:
                        h[p1] -= 0.01
                        h[p2] += 0.01
                        h = project_feasible(h)
                        current_score = evaluate_c5(h)
                    continue
                per_transfer = total_mass / (len(top_src) * len(top_dst))
                for si in top_src:
                    for dj in top_dst:
                        amt = min(per_transfer, h[si])
                        h[si] -= amt
                        h[dj] += amt
                h = project_feasible(h)
                current_score = evaluate_c5(h)
                continue
            i, j = best_pair
        
        # Line search
        t, new_score = two_point_line_search(h, i, j)
        
        if new_score < current_score:
            current_score = new_score
        elif t == 0:
            # No movement was made; try perturbation
            p1 = np.random.randint(n)
            p2 = np.random.randint(n)
            if h[p1] > 0.01 and h[p2] < 0.99:
                h[p1] -= 0.01
                h[p2] += 0.01
                h = project_feasible(h)
    
    return h, current_score

    n = len(h)
    h = project_feasible(h.copy())
    current_score = evaluate_c5(h)
    
    while True:
        if time_limit_fn is not None and time_limit_fn():

        corr = np.correlate(h, 1.0 - h, mode="full")

        active_idx = np.where(corr >= threshold)[0]
        lags = active_idx - (n - 1)
        
        g = compute_gradient(h, lags)
        g -= np.mean(g)

        k = max(2, n // 16)
        can_dec = h > 1e-10
        can_inc = h < 1.0 - 1e-10
        
        src_scores = g.copy()
        src_scores[~can_dec] = -np.inf
        dst_scores = g.copy()
```
