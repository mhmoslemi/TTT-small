def soft_max_c5(h, temperature=0.01):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # Soft max
    m = np.max(c)
    return m + temperature * np.log(np.sum(np.exp((c - m) / temperature)))

def gradient_soft_max(h, temperature=0.01):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    exp_c = np.exp((c - m) / temperature)
    weights = exp_c / np.sum(exp_c)
    # Gradient is sum of weights * gradients of each c_k
    grad = np.zeros(N)
    for k in range(2*N-1):
        if weights[k] < 1e-8:
            continue
        lag = k - (N-1)
        i1s, i1e = max(0, lag), min(N, N+lag)
        if i1e > i1s:
            idx = np.arange(i1s, i1e)
            grad[idx] += weights[k] * dx * (1.0 - h[idx - lag])
        i2s, i2e = max(0, -lag), min(N, N-lag)
        if i2e > i2s:
            idx = np.arange(i2s, i2e)
            grad[idx] -= weights[k] * dx * h[idx + lag]
    return grad

import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_polytope(h, target_sum):
    N = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0.0, 1.0)
    return x

def gradient_active(h, active_set):
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    for k in active_set:
        lag = k - (N - 1)
        i1s, i1e = max(0, lag), min(N, N + lag)
        if i1e > i1s:
            idx = np.arange(i1s, i1e)
            grad[idx] += 1.0 - h[idx - lag]
        i2s, i2e = max(0, -lag), min(N, N - lag)
        if i2e > i2s:
            idx = np.arange(i2s, i2e)
            grad[idx] -= h[idx + lag]
    return grad * dx

def gradient_softmax(h, temp=0.02):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(c)
    exp_c = np.exp((c - m) / temp)
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(N)

        lag = k - (N - 1)

            grad[idx] += w[k] * dx * (1.0 - h[idx - lag])
        i2s, i2e = max(0, -lag), min(N, N - lag)
        if i2e > i2s:
            idx = np.arange(i2s, i2e)
            grad[idx] -= w[k] * dx * h[idx + lag]
    return grad
