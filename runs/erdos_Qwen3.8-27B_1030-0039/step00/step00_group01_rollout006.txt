        return h

    if diff > 0:
        # Need to increase some values
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 0:
            h += diff * room / total_room
        else:
            h[:] = 0.5

        if total_room > 0:
            h += diff * room / total_room
        else:

    h = np.clip(h, 0.0, 1.0)

def _active_gradient(h, m, n):
    """Compute gradient of corr[m] with respect to h."""
    dx = 2.0 / n
    offset = m - (n - 1)
    grad = np.zeros(n)
    
    if offset >= 0:
        # corr[m] = sum_{i=0}^{n-1-offset} h[i+offset] * (1-h[i])
        for i in range(n - offset):
            # Term: h[i+offset] * (1-h[i])
            # d/dh[k]: (1-h[i]) if k=i+offset, -h[i+offset] if k=i
            grad[i + offset] += (1 - h[i]) * dx
            grad[i] -= h[i + offset] * dx

            grad[i + offset] += (1 - h[i]) * dx
            grad[i] -= h[i + offset] * dx

                break

        if beta < beta_max:
            beta *= 1.1

    target = n / 2.0

        room = 1.0 - h

import numpy as np
from scipy.special import logsumexp
import time

def _fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            room = 1.0 - h
            total = np.sum(room)
            if total > 1e-14:
                h += residual * room / total
            else:
                h[:] = 0.5
        else:
            total = np.sum(h)
            if total > 1e-14:
                h += residual * h / total
            else:
                h[:] = 0.5
        h = np.clip(h, 0.0, 1.0)
        # Final fix
        resid2 = target - np.sum(h)
        if abs(resid2) > 1e-12:
            idx = np.argmax(1.0 - h) if resid2 > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + resid2, 0.0, 1.0)
    return h

def _compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _eval_c5(h):
    return float(np.max(_compute_corr(h)))

def _grad_at_m(h, m, n, dx):
    """Gradient of the m-th correlation component w.r.t. h."""
    offset = m - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        idx_range = range(n - offset)
    else:
        idx_range = range(-offset, n)
    for i in idx_range:
        j = i + offset
        grad[j] += (1.0 - h[i])
        grad[i] -= h[j]
    return grad * dx

def _smooth_objective_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = _compute_corr(h)
    bc = beta * corr
    bc_max = np.max(bc)
    bc -= bc_max
    w = np.exp(bc)
    sw = np.sum(w)
    obj = (bc_max + np.log(sw)) / beta
    # Gradient
    w = w / sw
    grad = np.zeros(n)
    for m in range(len(corr)):
        if w[m] > 1e-10:
            grad += w[m] * _grad_at_m(h, m, n, dx)
    return obj, grad, corr

def _project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h + lam, 0.0, 1.0)
    resid = target - np.sum(h_p)
    if abs(resid) > 1e-10:
        if resid > 0:
            idx = np.argmax(1.0 - h_p)
        else:
            idx = np.argmax(h_p)
        h_p[idx] = np.clip(h_p[idx] + resid, 0.0, 1.0)
    return h_p

def _optimize_smooth(h, n, t_limit, beta_start=5.0, beta_end=300.0):
    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    beta = beta_start
    step = 0.5 / n  # initial step size
    
    while time.time() - t0 < t_limit:
        _, grad, corr = _smooth_objective_and_grad(h, n, beta)
        c5 = np.max(corr)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        # Feasible gradient direction
        g = grad - np.mean(grad)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            beta = min(beta_end, beta * 1.2)
            continue
        
        g = g / gn
        obj_curr = logsumexp(beta * (corr - np.max(corr))) / beta + np.max(corr)
        
        # Backtracking
        alpha = step
        improved = False
        for _ in range(25):
            h_new = _project_feasible(h - alpha * g, n)
            corr_new = _compute_corr(h_new)
            obj_new = logsumexp(beta * (corr_new - np.max(corr_new))) / beta + np.max(corr_new)
            if obj_new < obj_curr - 1e-14:
                h = h_new
                step = min(step * 1.05, 5.0 / n)
                improved = True
                break
            alpha *= 0.5
        if not improved:
            step *= 0.8
        
        beta = min(beta_end, beta * 1.02)
    
    return best_h, best_c5

def _line_search_refine(h, n, t_limit, rng):
    """Refine using exact line search along selected directions."""
    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    dx = 2.0 / n
    
    # Precompute correlation
    corr = _compute_corr(h)
    
    n_iter = 0
    while time.time() - t0 < t_limit and n_iter < 500:
        n_iter += 1
        # Choose a random pair (i, j)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Direction: transfer mass from j to i
        # h_i += alpha, h_j -= alpha
        # Feasible range:
        # h_i + alpha in [0,1] → alpha in [-h_i, 1-h_i]
        # h_j - alpha in [0,1] → alpha in [h_j-1, h_j]
        alpha_lo = max(-h[i], h[j] - 1.0)
        alpha_hi = min(1.0 - h[i], h[j])
        
        if alpha_hi - alpha_lo < 1e-10:
            continue
        
        # Evaluate at multiple points
        n_samples = 30
        alphas = np.linspace(alpha_lo, alpha_hi, n_samples)
        best_alpha = 0.0
        best_val = np.max(corr)
        
        for a in alphas:
            h_trial = h.copy()
            h_trial[i] += a
            h_trial[j] -= a
            h_trial = np.clip(h_trial, 0.0, 1.0)
            corr_t = _compute_corr(h_trial)
            val = np.max(corr_t)
            if val < best_val:
                best_val = val
                best_alpha = a
            if val < best_c5:
                best_c5 = val
        
        if best_alpha != 0 and best_val < best_c5 + 1e-10:
            h[i] += best_alpha
            h[j] -= best_alpha
            h = np.clip(h, 0.0, 1.0)
            # Re-fix sum (should be exact but just in case)
            h = _fix_sum(h, n)
            corr = _compute_corr(h)
            if _eval_c5(h) < best_c5:
                best_c5 = _eval_c5(h)
                best_h = h.copy()
    
    return best_h, best_c5

def _binary_search(h, n, t_limit, rng):
    """Local search for binary solutions."""
    t0 = time.time()
    # Binarize: sort and threshold
    h_sorted = np.sort(h)
    thresh = h_sorted[n // 2 - 1] + 1e-6
    h_bin = (h >= thresh).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        # Remove excess ones (the smallest ones that are 1)
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by original h value (keep the ones with highest h)
        order = np.argsort(-h[ones_idx])
        for k in range(n_ones - n // 2):
            h_bin[ones_idx[order[k]]] = 0
    elif n_ones < n // 2:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for k in range(n // 2 - n_ones):
            h_bin[zeros_idx[order[k]]] = 1
    
    h_cur = h_bin.copy()
    best_c5 = _eval_c5(h_cur)
    best_h = h_cur.copy()
    
    # Simulated annealing
    temperature = 0.01
    ones = np.where(h_cur == 1)[0]
    zeros = np.where(h_cur == 0)[0]
    
    while time.time() - t0 < t_limit:
        i_zeros = rng.randint(len(zeros))
        i_ones = rng.randint(len(ones))
        iz = zeros[i_zeros]
        io = ones[i_ones]
        
        h_trial = h_cur.copy()
        h_trial[iz] = 1
        h_trial[io] = 0
        
        c5_trial = _eval_c5(h_trial)
        delta = c5_trial - best_c5

            h_cur = h_trial
            # Update sets
            zeros = np.where(h_cur == 0)[0]
            ones = np.where(h_cur == 1)[0]
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_cur.copy()
        
        temperature *= 0.99
        if temperature < 1e-8:
            temperature = 0.005  # restart annealing
    
    return best_h, best_c5
