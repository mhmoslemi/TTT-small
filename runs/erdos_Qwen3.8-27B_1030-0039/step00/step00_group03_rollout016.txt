def smooth_gradient(h, p, state):
    n = len(h)

    grad = np.zeros(n)
    for m in range(2 * n - 1):
        d = m - (n - 1)
        pm = p[m] * dx
        if d == 0:
            grad[:] += pm * (1.0 - 2.0 * h)
        elif d > 0:
            k = d
            nk = n - k
            grad[k:n] += pm * (1.0 - h[0:nk])
            grad[0:nk] += pm * (-h[k:n])
        else:
            ps = -d
            nk = n - ps
            grad[0:nk] += pm * (1.0 - h[ps:n])
            grad[ps:n] += pm * (-h[0:nk])
    return grad

    Z = np.sum(w)

def _clip_sum(t, x, target):

def exact_line_search(h, a, b, n_evals, state):
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-14:
        return h, evaluate_exact(h)
    best_h = h.copy()
    best_score = evaluate_exact(h)
    deltas = np.linspace(lo, hi, n_evals)

        check_time(state)
        tmp = h.copy()
        tmp[a] += delta
        tmp[b] -= delta
        score = evaluate_exact(tmp)
        if score < best_score - 1e-14:
            best_score = score
            best_h = tmp.copy()
    return best_h, best_score

    n = len(h)
