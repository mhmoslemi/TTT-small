def pair_optimize(h, i, j):
    """For coordinates i and j, find optimal transfer t."""
    n = len(h)
    dx = 2.0 / n
    hi = h[i]
    hj = h[j]
    
    # t range: h[i]+t >= 0, h[j]-t >= 0, h[i]+t <= 1, h[j]-t <= 1
    t_min = max(-hi, hj - 1)  # h[i]+t >= 0 => t >= -hi; h[j]-t <= 1 => t >= hj-1
    t_max = min(1 - hi, hj)    # h[i]+t <= 1 => t <= 1-hi; h[j]-t >= 0 => t <= hj
    
    if t_min >= t_max:
        return 0.0
    
    def max_corr(t):
        h_new = h.copy()
        h_new[i] += t
        h_new[j] -= t
        return np.max(np.correlate(h_new, 1 - h_new, mode="full")) * dx
    
    # Bisection to find optimal t
    # Actually, let's use golden section or ternary search

    # Coarse search
    n_coarse = 50
