import numpy as np
import time
import random

class TimeGuard:
    def __init__(self, deadline):
        self.deadline = deadline
    def remaining(self):
        return self.deadline - time.time()
    def check(self, reserve=1.0):
        if time.time() > self.deadline - reserve:
            raise TimeoutError()

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_feasible(raw, target=None):
    raw = np.asarray(raw, dtype=np.float64)
    n = len(raw)
    if target is None:
        target = n / 2.0
    if n == 0:
        return raw.copy()
    lo = float(-np.max(raw)) - 1.0
    hi = 1.0 - float(np.min(raw)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(raw + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    result = np.clip(raw + tau, 0.0, 1.0)
    diff = target - float(np.sum(result))
    if abs(diff) > 1e-12:
        mask = (result > 1e-9) & (result < 1.0 - 1e-9)
        cnt = int(np.sum(mask))
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def is_feasible(h, tol=1e-6):

    if n == 0:
        return False
    if not np.all(np.isfinite(h)):

    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > tol * max(1, n):
        return False
    return True

        rem = [x for x in range(n) if x not in set(idx)]
        h[rng.choice(rem)] = 0.5
    return h

    hn[i] += t
    hn[j] -= t

def local_polish(h, guard, max_passes=300, improve_tol=1e-11, max_iter_gs=20):
    n = len(h)
    h = h.copy()
    current_score = exact_score(h)
    for pass_num in range(max_passes):
        guard.check(reserve=5.0)
        improved = False
        for i in range(n):
            guard.check(reserve=2.0)
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-12:
                    continue
                h_new, s_new = pair_optimize(h, n, i, j, max_iter=max_iter_gs)
                if s_new < current_score - improve_tol:
                    h = h_new
                    current_score = s_new
                    improved = True
        if not improved:

    return h, current_score

    n = len(h)

    s_cur = score
    h_best = h.copy()
    s_best = score
    steps_per_chunk = 5000
    total_chunks = max(1, max_steps // steps_per_chunk)
    for chunk in range(total_chunks):
        guard.check(reserve=10.0)
        frac = chunk / max(1, total_chunks - 1)
        temp = t_start * (t_end / t_start) ** frac
        for _ in range(steps_per_chunk):
            h_new, s_new = sa_step(h_cur, n, guard, rng, temp)

                h_cur = h_new
                s_cur = s_new
                if s_new < s_best:
                    h_best = h_new.copy()
                    s_best = s_new
    return h_best, s_best
