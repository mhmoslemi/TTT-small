import numpy as np
import time

def precompute_pairs(n):
    """Precompute (a_idx, b_idx) pairs for each lag r."""
    pairs = []
    for r in range(2*n - 1):
        t = r - (n - 1)
        n_min = max(0, -t)
        n_max = min(n - 1, n - 1 - t)
        if n_min > n_max:
            pairs.append((np.array([], dtype=int), np.array([], dtype=int)))
            continue
        n_vals = np.arange(n_min, n_max + 1)
        a_idx = n_vals + t
        b_idx = n_vals
        pairs.append((a_idx, b_idx))
    return pairs

def evaluate_c5(h, n):
    """Compute C5 exactly."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_dvals(h, n, pairs):
    """Compute all D_r values."""
    dx = 2.0 / n
    dvals = np.zeros(2*n - 1)
    for r, (a_idx, b_idx) in enumerate(pairs):
        if len(a_idx) == 0:
            continue
        dvals[r] = dx * np.sum(h[a_idx] * (1.0 - h[b_idx]))
    return dvals

def compute_smooth_grad(h, n, pairs, dvals, tau):
    """Compute gradient of smooth max tau*log(sum exp(D_r/tau))."""
    K = 2*n - 1
    # Softmax weights
    d_min = np.min(dvals)
    exp_d = np.exp((dvals - d_min) / tau)
    P = exp_d / np.sum(exp_d)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(K):
        if P[r] < 1e-15:
            continue
        a_idx, b_idx = pairs[r]
        if len(a_idx) == 0:
            continue
        w = P[r] * dx
        # ∂D_r/∂h[a] += w * (1-h[b])
        # ∂D_r/∂h[b] -= w * h[a]
        np.add.at(grad, a_idx, w * (1.0 - h[b_idx]))
        np.add.at(grad, b_idx, -w * h[a_idx])
    
    return grad

def project_tangent(grad, n):
    """Project gradient onto sum-zero subspace."""
    return grad - np.mean(grad)

import numpy as np
import time

def precompute_pairs(n):
    pairs = []
    for r in range(2 * n - 1):
        t = r - (n - 1)
        n_min = max(0, -t)
        n_max = min(n - 1, n - 1 - t)
        if n_min > n_max:
            pairs.append((np.empty(0, dtype=int), np.empty(0, dtype=int)))
        else:
            n_vals = np.arange(n_min, n_max + 1)
            pairs.append((n_vals + t, n_vals))
    return pairs

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_dvals(h, pairs, dx):
    K = len(pairs)
    dvals = np.zeros(K)
    for r in range(K):
        a, b = pairs[r]
        if len(a) > 0:
            dvals[r] = dx * np.sum(h[a] * (1.0 - h[b]))
    return dvals

def compute_grad(h, pairs, dvals, tau, dx):
    K = len(pairs)
    d_min = np.min(dvals)
    exp_d = np.exp((dvals - d_min) / tau)
    P = exp_d / np.sum(exp_d)
    grad = np.zeros(len(h))
    for r in range(K):
        if P[r] * dx < 1e-15:
            continue
        a, b = pairs[r]
        if len(a) == 0:
            continue
        w = P[r] * dx
        np.add.at(grad, a, w * (1.0 - h[b]))
        np.add.at(grad, b, -w * h[a])
    return grad

def line_search(h, direction, n, pairs, dx, max_steps=20):
    """Find best step along direction (sum-zero, bounded)."""
    # Compute max alpha
    alpha_max = np.inf
    pos = direction > 1e-12
    neg = direction < -1e-12
    if np.any(pos):
        alpha_max = min(alpha_max, np.min((1.0 - h[pos]) / direction[pos]))
    if np.any(neg):
        alpha_max = min(alpha_max, np.min(h[neg] / (-direction[neg])))
    if not np.isfinite(alpha_max) or alpha_max <= 0:
        return None, None
    
    # Try geometric sequence
    best_h = h
    best_score = np.inf
    for i in range(max_steps):
        alpha = alpha_max * (0.8 ** i)
        if alpha <= 0:
            break
        h_new = h + alpha * direction
        # Project to [0,1] and fix sum
        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        target = n / 2.0
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Redistribute
            frac = diff / n
            h_new = np.clip(h_new + frac, 0, 1)
            # Second pass if needed
            diff2 = target - np.sum(h_new)
            if abs(diff2) > 1e-10:
                # Fine adjustment
                mask = (h_new > 0) & (h_new < 1)
                if np.any(mask):
                    h_new[mask] += diff2 / np.sum(mask)
                    h_new = np.clip(h_new, 0, 1)
                else:
                    h_new = h  # can't fix, reject
                    continue
        score = evaluate_c5(h_new, n)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
    return best_h, best_score

def local_optimize(h_start, n, pairs, dx, tau_start, tau_min, tau_decay, 
                   max_iters, rng, time_limit, t0):
    h = h_start.copy()
    current_score = evaluate_c5(h, n)
    best_h = h.copy()
    best_score = current_score
    tau = tau_start
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if tau < tau_min:
            break
        
        dvals = compute_dvals(h, pairs, dx)
        grad = compute_grad(h, pairs, dvals, tau, dx)
        direction = -(grad - np.mean(grad))
        
        step_norm = np.linalg.norm(direction)
        if step_norm < 1e-10:
            stall_count += 1
            tau *= tau_decay
            continue
        stall_count = 0
        
        h_new, score_new = line_search(h, direction, n, pairs, dx)

            current_score = score_new

                best_score = current_score
                best_h = h.copy()
            tau *= max(tau_decay, 0.95)  # slow decay on success
        else:
            tau *= tau_decay
    
    return best_h, best_score

def random_perturb(h, n, sigma, rng):
    h_new = h + sigma * rng.standard_normal(n)
    # Project to feasible set
    h_new = np.clip(h_new, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-10:

            if np.any(mask):
                h_new[mask] += diff2 / np.sum(mask)
                h_new = np.clip(h_new, 0, 1)
    return h_new

def pair_swap(h, n, rng, step_min=0.01, step_max=0.1):
    """Random feasible pair swap."""
    h_new = h.copy()
    for _ in range(10):
        i, j = rng.choice(n, 2, replace=False)
        if h_new[i] == h_new[j]:
            continue
        step = rng.uniform(step_min, step_max)
        # Move mass from higher to lower
        if h_new[i] > h_new[j]:
            delta = min(step, h_new[i] - h_new[j])
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta = min(step, h_new[j] - h_new[i])
            h_new[j] -= delta
            h_new[i] += delta
    return h_new

def upsample(h, n, factor):
    """Upsample by repeating each value."""
    if factor == 1:
        return h.copy()
    # Simple repeat
    h_new = np.repeat(h, factor)
    n_new = len(h_new)
    # Check sum
    target = n_new / 2.0
    current_sum = np.sum(h_new)
    # Should be exactly factor * n/2 = factor * target_original
    # target_original = n/2, new target = factor*n/2
    # Current sum = factor * sum(h) = factor * n/2 = new target. 
    # So it should be exact!
    return h_new

import numpy as np

def precompute_pairs(n):
    pairs = []
    for r in range(2 * n - 1):
        t = r - (n - 1)
        n_min = max(0, -t)
        n_max = min(n - 1, n - 1 - t)
        if n_min > n_max:
            pairs.append((np.empty(0, dtype=np.int32), np.empty(0, dtype=np.int32)))
        else:
            n_vals = np.arange(n_min, n_max + 1, dtype=np.int32)
            pairs.append((n_vals + t, n_vals))
    return pairs

def evaluate_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    for r in range(K):

        if len(a) > 0:

        else:
            dvals[r] = 0.0

import numpy as np
import time

def precompute_pairs(n):
    pairs = []
    for r in range(2 * n - 1):
        t = r - (n - 1)
        n_min = max(0, -t)
        n_max = min(n - 1, n - 1 - t)
        if n_min > n_max:
            pairs.append((np.empty(0, dtype=np.int32), np.empty(0, dtype=np.int32)))
        else:
            n_vals = np.arange(n_min, n_max + 1, dtype=np.int32)
            pairs.append((n_vals + t, n_vals))
    return pairs

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def score_internal(h, pairs, dx):
    max_d = 0.0
    for r in range(len(pairs)):
        a, b = pairs[r]
        if len(a) > 0:
            d = dx * np.dot(h[a], 1.0 - h[b])
            if d > max_d:
                max_d = d
    return max_d

def compute_dvals(h, pairs, dx):
    K = len(pairs)
    dvals = np.zeros(K)
    for r in range(K):
        a, b = pairs[r]
        if len(a) > 0:
            dvals[r] = dx * np.dot(h[a], 1.0 - h[b])
    return dvals

def compute_soft_grad(h, pairs, dvals, tau, dx):
    K = len(pairs)
    d_max = np.max(dvals)
    exp_d = np.exp((dvals - d_max) / tau)
    P = exp_d / np.sum(exp_d)
    grad = np.zeros(len(h))
    for r in range(K):
        w = P[r] * dx
        if w < 1e-16:
            continue
        a, b = pairs[r]
        if len(a) == 0:
            continue
        np.add.at(grad, a, w * (1.0 - h[b]))
        np.add.at(grad, b, -w * h[a])
    return grad

def project_feasible(z, n):
    target = n / 2.0
    h = np.asarray(z, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s_mid = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def gradient_descent(h_start, n, pairs, dx, tau_start, tau_min, tau_decay, max_iters, time_limit, t_start):
    h = h_start.copy()
    current_score = score_internal(h, pairs, dx)
    best_h = h.copy()
    best_score = current_score
    tau = tau_start
    no_improve = 0
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        if tau < tau_min:
            break
        dvals = compute_dvals(h, pairs, dx)
        grad = compute_soft_grad(h, pairs, dvals, tau, dx)
        direction = -(grad - np.mean(grad))
        step_norm = np.linalg.norm(direction)
        if step_norm < 1e-12:
            tau *= tau_decay
            no_improve += 1
            if no_improve > 5:
                break
            continue
        alpha_max = np.inf
        pos_m = direction > 1e-14
        neg_m = direction < -1e-14
        if np.any(pos_m):
            alpha_max = min(alpha_max, np.min((1.0 - h[pos_m]) / direction[pos_m]))
        if np.any(neg_m):
            alpha_max = min(alpha_max, np.min(h[neg_m] / (-direction[neg_m])))
        if not np.isfinite(alpha_max) or alpha_max <= 0:
            tau *= tau_decay
            continue
        improved = False
        best_trial_score = current_score
        best_trial_h = h.copy()
        for i in range(30):
            alpha = alpha_max * (0.7 ** i)
            if alpha * step_norm < 1e-13:
                break
            h_trial = h + alpha * direction
            h_trial = np.clip(h_trial, 0.0, 1.0)
            s_diff = np.sum(h_trial) - n / 2.0
            if abs(s_diff) > 1e-10:
                h_trial -= s_diff / n
                h_trial = np.clip(h_trial, 0.0, 1.0)
            sc = score_internal(h_trial, pairs, dx)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_h = h_trial.copy()
            if sc < current_score:
                improved = True
                break
        if improved and best_trial_score < current_score - 1e-14:
            h = best_trial_h
            current_score = best_trial_score
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best_h = h.copy()
            tau = max(tau * 0.97, tau_min)
        else:
            tau *= tau_decay
            no_improve += 1
            if no_improve > 15:
                break
    return best_h, best_score

def random_perturb(h, n, sigma, rng):
    noise = sigma * rng.standard_normal(n)
    h_new = h + noise
    return project_feasible(h_new, n)

def random_binary(n, rng):
    h = np.zeros(n, dtype=np.float64)
    if n % 2 == 0:
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    else:
        half = (n - 1) // 2
        ones = rng.choice(n, half, replace=False)
        h[ones] = 1.0
        remaining = np.setdiff1d(np.arange(n), ones)
        mid = rng.choice(remaining)
        h[mid] = 0.5
    return h

def upsample(h, factor):
    return np.repeat(h, factor)

def structured_binary(n, rng):
    ratio = rng.choice([0.38196601125, 0.61803398875, 0.5, 0.41421356, 0.30901699])
    phases = [ratio * n * (1 + 0.3 * rng.standard_normal()) % 1.0 for _ in range(3)]
    ph = phases[0] % 1.0
    x = np.arange(n, dtype=np.float64) / n
    h = np.where((x + ph % 1.0) % 1.0 < ratio, 1.0, 0.0)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    time_limit = budget_s * 0.95
    rng = np.random.default_rng(seed)

    h_init = globals().get('initial_h_values', None)

    h_best = project_feasible(h_best, n_best)

    pairs_best = precompute_pairs(n_best)
    dx_best = 2.0 / n_best

    phase1_end = t_start + time_limit * 0.25
    h_pol, sc_pol = gradient_descent(
        h_best, n_best, pairs_best, dx_best,

    if sc_pol < score_best:
        h_best, score_best = h_pol, sc_pol
        pairs_best = precompute_pairs(len(h_best))
        dx_best = 2.0 / len(h_best)

    phase2_end = t_start + time_limit * 0.50
    n_cur = len(h_best)
    pairs_cur = precompute_pairs(n_cur)
    dx_cur = 2.0 / n_cur
    while time.time() < phase2_end:
        attempt_budget = max(2.0, min(8.0, (phase2_end - time.time()) * 0.3))
        if time.time() + attempt_budget > phase2_end:
