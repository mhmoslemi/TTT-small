from scipy.optimize import minimize
import time

def repair_feasible(h):
    """Project h onto the feasible set: 0<=h<=1, sum(h)=N/2."""
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    target = N / 2.0
    
    # Handle non-finite
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    
    # Bisection to find t such that sum(clip(h - t, 0, 1)) = target
    def sum_at_t(t):
        return np.sum(np.clip(h - t, 0.0, 1.0))
    
    # Find range
    t_lo = -1.0  # sum at t=-1 is sum(min(h+1,1)) = sum(1) for h>=0... = N
    t_hi = 1.0   # sum at t=1 is sum(max(h-1,0)) = 0
    
    # Actually let's be more careful
    # sum_at_t is non-increasing in t
    # At t = -1: clip(h+1, 0, 1) = 1 for all h in [0,1], so sum = N
    # At t = 1: clip(h-1, 0, 1) = 0 for all h in [0,1], so sum = 0
    # target = N/2 is between 0 and N, so there exists a valid t
    
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_t(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2.0
    h_out = np.clip(h - t, 0.0, 1.0)
    
    # Final correction for floating point
    d = target - np.sum(h_out)
    if abs(d) > 1e-14:
        # Find interior points
        interior = np.where((h_out > 1e-12) & (h_out < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            # Distribute d among interior points
            per = d / len(interior)
            h_out[interior] += per
            # If any went out of bounds, fix
            h_out = np.clip(h_out, 0.0, 1.0)
            # Re-correct
            d2 = target - np.sum(h_out)
            if abs(d2) > 1e-14 and len(interior) > 0:
                h_out[interior[0]] += d2
                h_out = np.clip(h_out, 0.0, 1.0)
    
    return h_out


def evaluate_exact(h):
    """Compute the exact C5 objective."""
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_objective(h, beta):
    """Compute log-sum-exp / beta of the correlation terms."""
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = dx * corr
    max_val = np.max(vals)
    if not np.isfinite(max_val):
        return 1e10
    shifted = vals - max_val
    exp_vals = np.exp(beta * shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp <= 0:
        return max_val
    return max_val + np.log(sum_exp) / beta


def smooth_gradient(h, beta):
    """Compute gradient of log-sum-exp / beta."""
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = dx * corr
    max_val = np.max(vals)
    shifted = vals - max_val
    exp_vals = np.exp(beta * shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp <= 0:
        return np.zeros(N)
    weights = exp_vals / sum_exp
    
    # Gradient: sum_k weights[k] * dx * grad_k
    # grad_k[m] = [0<=m-k<N]*(1-h[m-k]) - [0<=m+k<N]*h[m+k]
    # where k = index - (N-1)
    
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        k = idx - (N - 1)
        w = weights[idx]
        if w < 1e-15:
            continue
        # Part 1: [0<=m-k<N]*(1-h[m-k])
        # m ranges from max(0,k) to min(N-1, N-1+k) when k>=0
        # m ranges from 0 to N-1+k when k<0, but m-k must be in [0,N-1]
        
        if k >= 0:
            # m from k to N-1: h[m-k] is valid
            m_start, m_end = k, N
            grad[m_start:m_end] += w * dx * (1.0 - h[m_start - k:m_end - k])
            # Part 2: -[0<=m+k<N]*h[m+k]
            # m from 0 to N-1-k
            m_start2, m_end2 = 0, N - k
            grad[m_start2:m_end2] -= w * dx * h[m_start2 + k:m_end2 + k]
        else:
            # k < 0, let j = -k > 0
            j = -k
            # Part 1: [0<=m+k<N] means [0<=m-j<N], m from j to N-1
            # h[m+k] = h[m-j]
            m_start, m_end = j, N
            grad[m_start:m_end] += w * dx * (1.0 - h[m_start - j:m_end - j])
            # Part 2: -[0<=m-k<N] = -[0<=m+j<N], m from 0 to N-1-j
            # h[m-k] = h[m+j]
            m_start2, m_end2 = 0, N - j
            grad[m_start2:m_end2] -= w * dx * h[m_start2 + j:m_end2 + j]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def repair_feasible(h):
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    target = N / 2.0
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h_out = np.clip(h - t, 0.0, 1.0)
    d = target - np.sum(h_out)
    if abs(d) > 1e-13:
        interior = np.where((h_out > 1e-12) & (h_out < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            h_out[interior] += d / len(interior)
            h_out = np.clip(h_out, 0.0, 1.0)
            d2 = target - np.sum(h_out)
            if abs(d2) > 1e-13 and len(interior) > 0:
                room = np.minimum(1.0 - h_out[interior], h_out[interior])
                idxs = interior[room >= abs(d2)]
                if len(idxs) > 0:
                    h_out[idxs[0]] += d2
                else:
                    h_out[interior[0]] += d2
                h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def so_fun(x, beta):
    N = len(x)
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = dx * corr
    mv = np.max(vals)
    if not np.isfinite(mv):
        return 1e10
    shifted = beta * (vals - mv)
    se = np.sum(np.exp(shifted))
    if se < 1e-300:
        return float(mv)
    return float(mv + np.log(se) / beta)

def so_grad(x, beta):
    N = len(x)
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode="full")
    vals = dx * corr
    mv = np.max(vals)
    shifted = beta * (vals - mv)
    exp_vals = np.exp(shifted)
    se = np.sum(exp_vals)
    if se < 1e-300:
        return np.zeros(N)
    w = exp_vals / se
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        k = idx - (N - 1)
        wi = w[idx]
        if wi < 1e-15:
            continue
        wk = wi * dx
        if k >= 0:
            ms, me = k, N
            grad[ms:me] += wk * (1.0 - x[ms - k:me - k])
            ms2, me2 = 0, N - k
            if me2 > ms2:
                grad[ms2:me2] -= wk * x[ms2 + k:me2 + k]
        else:
            j = -k
            ms, me = j, N
            grad[ms:me] += wk * (1.0 - x[ms - j:me - j])
            ms2, me2 = 0, N - j
            if me2 > ms2:
                grad[ms2:me2] -= wk * x[ms2 + j:me2 + j]
    return grad

def con_fun(x, target):
    return float(np.sum(x) - target)

def con_jac(x, target):
    return np.ones(len(x))

def transfer_to(h, new_N):
    old_N = len(h)
    if new_N == old_N:
        return h.copy()
    if new_N % old_N == 0:
        return np.repeat(h, new_N // old_N)
    if old_N % new_N == 0:
        return h.reshape(-1, old_N // new_N).mean(axis=1)
    oc = (np.arange(old_N, dtype=np.float64) + 0.5) * 2.0 / old_N
    nc = (np.arange(new_N, dtype=np.float64) + 0.5) * 2.0 / new_N
    return np.interp(nc, oc, h)

def run_slsqp_pass(h0, beta, bounds, constraint, maxiter, ftol):
    try:
        sol = minimize(
            so_fun, h0, args=(beta,), method='SLSQP',
            jac=so_grad, bounds=bounds, constraints=[constraint],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if sol.x is not None and np.all(np.isfinite(sol.x)):
            return sol.x
    except Exception:
        pass
    return None
