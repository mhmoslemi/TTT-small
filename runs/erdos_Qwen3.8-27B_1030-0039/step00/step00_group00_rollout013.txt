def repair_feasible(h, N):
    """Project h onto {0 <= x <= 1, sum(x) = N/2}"""
    h = np.asarray(h, dtype=np.float64).copy()
    if h.shape[0] != N:
        # Resample
        if h.size == 0:
            return np.full(N, 0.5)
        new_h = np.interp(np.linspace(0, 1, N, endpoint=False), np.linspace(0, 1, h.size, endpoint=False), h)
        h = new_h
    np.clip(h, 0.0, 1.0, out=h)
    # Bisection for t
    target = N / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        q = np.sum(np.clip(h - mid, 0.0, 1.0))
        if q > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(h - t, 0.0, 1.0)
    # Final correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            mask = h < 1.0 - 1e-12
        else:

    N = len(h)

def compute_smooth_objective_and_grad(h, beta):

    N = len(h)
    dx = 2.0 / N
    
    # Compute all 2N-1 correlation terms (raw, unscaled)
    # Using the structure: center + forward + backward
    # Center (lag 0): sum h_i (1-h_i)

    # Forward lags s=1..N-1: sum_{i=0}^{N-1-s} h_{i+s}(1-h_i)

        terms[N - 1 - s] = np.sum(h[:-s] * (1.0 - h[s:]))  # backward lag s

    # Log-sum-exp with max subtraction for stability
    g_max = np.max(g)

        terms[N - 1 - s] = np.dot(h[s:], 1.0 - h[:N-s])
        terms[N - 1 + s] = np.dot(h[:N-s], 1.0 - h[s:])
    
    g = terms * dx
    g_max = np.max(g)
    shifted = beta * (g - g_max)
    exp_vals = np.exp(shifted)

            grad[:N-s] += wB * (1.0 - h[s:])
            grad[s:] += wB * (-h[:N-s])
