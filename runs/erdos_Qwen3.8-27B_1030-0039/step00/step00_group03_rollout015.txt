def compute_smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    y = beta * corr
    m = np.max(y)
    exp_y = np.exp(y - m)
    Z = np.sum(exp_y)
    psi = (m + np.log(Z)) / beta
    p = exp_y / Z
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-15:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            s = d
            k = n - s
            grad[0:k] += w * (1.0 - h[0:k])
            grad[s:n] -= w * h[0:k]
        else:
            p_shift = -d
            k = n - p_shift
            grad[p_shift:n] += w * (1.0 - h[p_shift:n])
            grad[0:k] -= w * h[p_shift:n]
    return psi, grad

        if non_sat.any():
            result[non_sat] += diff / np.sum(non_sat)
            result = np.clip(result, 0.0, 1.0)
    return result
