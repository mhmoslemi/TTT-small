    c = np.correlate(h, f, mode='full') * dx  # (2n-1,)

    c_shifted = c - np.max(c)

def project(h, n):
    target = n / 2.0
    # Binary search for lambda such that sum(clip(h + lambda, 0, 1)) = target
    lo = -np.max(h) - 1
    hi = 1 - np.min(h) + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h + mid, 0, 1)
        s = h_proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix any tiny numerical error
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        # Add to elements not at boundary
        free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if free.any():
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0, 1)
        else:
            # Fallback: distribute to all
            h_proj += diff / n
            h_proj = np.clip(h_proj, 0, 1)
    
    # Final fixup: ensure exact sum
    # If still slightly off, adjust one element
    remaining = target - h_proj.sum()
    if abs(remaining) > 1e-14:
        idx = np.argmin(np.abs(h_proj - 0.5))  # adjust the most central one
        h_proj[idx] += remaining
        h_proj = np.clip(h_proj, 0, 1)
    
    return h_proj

def _pair_move(h, n, rng, max_pairs=200):
    """Try random pair moves to improve."""
    h = h.copy()
    best = _evaluate_c5(h, n)
    for _ in range(max_pairs):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # h[i] += delta, h[j] -= delta
        # valid: 0 <= h[i]+delta <= 1, 0 <= h[j]-delta <= 1
        # delta in [-h[i], 1-h[j]] for i, and delta in [h[j]-1, h[j]] for j
        # Combined: delta in [-h[i], 1-h[j]] ∩ [-h[j], h[j]-1]... 
        # Wait: h[j] - delta >= 0 → delta <= h[j]
        #       h[j] - delta <= 1 → delta >= h[j] - 1
        #       h[i] + delta >= 0 → delta >= -h[i]
        #       h[i] + delta <= 1 → delta <= 1 - h[i]
        # So delta in [max(-h[i], h[j]-1), min(1-h[i], h[j])]
        dlo = max(-h[i], h[j] - 1.0)
        dhi = min(1.0 - h[i], h[j])
        if dlo >= dhi:
            continue
        # Try a few delta values
        for t in (0.3, 0.5, 0.7):
            delta = dlo + t * (dhi - dlo)
            h[i] += delta
            h[j] -= delta
            v = _evaluate_c5(h, n)
            if v < best - 1e-13:
                best = v
            else:
                h[i] -= delta
                h[j] += delta
                break
        if h[i] < h[i] - delta or h[j] > h[j] + delta:
            pass  # should not happen
    return h, best

Let me look at this from a different angle. The problem is:
minimize max_k Tr_k(h)
where Tr_k(h) = ∑_i h[i]·(1-h[i-d_k])·dx

This is a "minimax" problem. A common approach is:
1. Introduce t ≥ Tr_k(h) for all k
2. Minimize t

Using Lagrange multipliers / bisection: for a given t, check if there exists h with Tr_k(h) ≤ t for all k. This is a feasibility problem.

But the feasibility problem is:
∑_i h[i](1-h[i-d_k]) ≤ t/dx for all k
0 ≤ h[i] ≤ 1
sum(h) = n/2

The first constraint is quadratic in h, so it's not a standard linear programming feasibility problem.

Another approach: use the fact that h[i](1-h[i-d]) = h[i] - h[i]h[i-d]. So:
Tr_k = (n/2)·dx - ∑_i h[i]h[i-d_k]·dx

Minimizing Tr_k is the same as maximizing ∑_i h[i]h[i-d_k]. And minimizing max_k Tr_k is the same as maximizing min_k ∑_i h[i]h[i-d_k].

So the problem is:
maximize min_k R_k(h)
where R_k(h) = ∑_i h[i]·h[i-d_k] (for valid i)
subject to 0 ≤ h[i] ≤ 1, sum(h) = n/2

This is a "maximin" problem. The function R_k is quadratic in h (specifically, it's a quadratic form with a structured matrix). The "maximin" of quadratic functions is non-convex.

But wait: R_k(h) = h^T A_k h where A_k is a matrix with A_k[i, i-d_k] = 1 (for valid entries). This is a symmetric matrix (if we include both A_k[i,j] and A_k[j,i]). Actually:

R_k = ∑_i h[i]·h[i-d] = ∑_{i: valid} h[i]·h[i-d]

This is not a standard quadratic form h^T M h because the "valid" condition means the sum is over a subset. But we can write it as h^T M_k h where M_k is a matrix with M_k[i, i-d_k] = 1/2 and M_k[i-d_k, i] = 1/2 (to make it symmetric), and the diagonal handled appropriately.

Actually, R_k = ∑_i h[i]·h[i-d_k]. If i-d_k ≠ i (i.e., d_k ≠ 0), then each pair (i, i-d_k) appears once. We can write:
R_k = (1/2) ∑_i (h[i]·h[i-d_k] + h[i-d_k]·h[i]) = h^T M_k h

where M_k[i, i-d_k] = M_k[i-d_k, i] = 1/2 for valid indices.

For d_k = 0: R_0 = ∑_i h[i]^2 (the self-correlation).

So the problem is:

subject to 0 ≤ h ≤ 1, 1^T h = n/2

where M_k are symmetric matrices (some positive semidefinite, some not necessarily).

This is still non-convex in general. But it gives us insight into the structure.
