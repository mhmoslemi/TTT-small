    pass

def _check_time(t0, budget, reserve=5.0):
    if time.time() - t0 > budget - reserve:
        raise TimeLimit()

def _exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def _lse_objective(x, beta, t0, budget):
    _check_time(t0, budget)
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    f = corr * dx
    m = np.max(f)
    exp_f = np.exp(beta * (f - m))
    Z = np.sum(exp_f)
    return (m + np.log(Z)) / beta

def _lse_jacobian(x, beta, t0, budget):
    _check_time(t0, budget)
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    f = corr * dx
    m = np.max(f)
    exp_f = np.exp(beta * (f - m))
    Z = np.sum(exp_f)
    p = exp_f / Z
    grad = np.zeros(n)
    for d in range(-n + 1, n):
        j = n - 1 + d
        w = p[j] * dx
        if abs(w) < 1e-15:
            continue
        if d >= 0:
            k = n - d
            if k > 0:
                grad[0:k] += w * (1.0 - x[d:])
                grad[d:n] += -w * x[0:k]
        else:
            p2 = -d
            k = n - p2
            if k > 0:
                grad[p2:n] += w * (1.0 - x[0:k])
                grad[0:k] += -w * x[p2:n]
    return grad
