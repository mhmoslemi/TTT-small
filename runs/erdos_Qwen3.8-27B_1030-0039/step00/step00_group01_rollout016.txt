def smooth_logsumexp_grad(h, n, tau):
    """Gradient of the smoothed objective."""
    corr = compute_corr(h)
    dx = 2.0 / n
    cmax = np.max(corr)
    exp_vals = np.exp((corr - cmax) / tau)
    p = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-8:
            continue
        offset = k - (n - 1)
        if offset == 0:
            grad += p[k] * (1.0 - 2.0 * h)
        else:
            g = np.zeros(n)
            idx1 = np.arange(n) + offset
            v1 = (idx1 >= 0) & (idx1 < n)
            g[v1] += 1.0 - h[idx1[v1]]
            idx2 = np.arange(n) - offset
            v2 = (idx2 >= 0) & (idx2 < n)
            g[v2] -= h[idx2[v2]]
            grad += p[k] * g
    return dx * grad

def optimize_smoothed(h, n, max_iter=5000, tau_init=0.1, tau_min=0.01):
    """Optimize using smoothed gradient descent with decreasing tau."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = tau_init
    
    for it in range(max_iter):
        # Decrease tau gradually
        tau = max(tau_min, tau_init * (0.999 ** it))
        
        # Compute gradient
        grad = smooth_logsumexp_grad(h, n, tau)
        
        # Line search: try several step sizes
        c5_current = compute_c5(h)
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005]:
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_current - 1e-12:
                h = h_new
                c5_current = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            # Annaproduction: small random perturbation
            h = h + 0.005 * np.random.randn(n)
            h = project_feasible(h, n)
    
    return best_h, best_c5

def binary_sa(n, iterations=50000, seed=42):
    """Simulated annealing for binary solutions."""
    rng = np.random.RandomState(seed)
    target_ones = n // 2
    
    # Start with random balanced binary
    h = np.zeros(n)
    ones = rng.choice(n, target_ones, replace=False)
    h[ones] = 1.0
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    t_start = 0.5
    t_end = 0.01
    alpha = (t_end / t_start) ** (1.0 / max(iterations, 1))
    
    for it in range(iterations):
        temp = t_start * (alpha ** it)
        
        i = rng.randint(n)
        h[i] = 1.0 - h[i]
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
    
    return best_h, best_c5

def interpolate_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy(), n_new)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
