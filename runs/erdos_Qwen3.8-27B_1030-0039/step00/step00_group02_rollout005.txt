    c_max = np.max(c_scaled)

    weights = exp_c / np.sum(exp_c)
    return weights

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_correlations(h)) * dx)

def project(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - float(np.sum(h))
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0 - 1e-12

            if cnt == 0:

        else:

            if cnt == 0:

    return np.clip(h, 0.0, 1.0)

def gradient_method(h, target_sum, max_iters, time_limit, start_time):
    n = len(h)
    h = h.copy()
    T = 30.0
    lr = 0.01
    best_c5 = compute_c5(h)

        curr_c5 = float(np.max(corr) * (2.0/n))
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        
        # Compute softmax weights
        c_scaled = T * corr
        c_max = np.max(c_scaled)
        exp_c = np.exp(c_scaled - c_max)
        weights = exp_c / np.sum(exp_c)
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-10:
                continue
            shift = k - (n - 1)
            j_min = max(0, shift)
            j_max = min(n-1, n-1+shift)
            for i in range(j_min, j_max+1):

                jp = i + shift
