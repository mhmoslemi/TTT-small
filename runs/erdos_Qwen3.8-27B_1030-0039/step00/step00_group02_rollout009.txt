def smooth_c5(h, n, tau):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    val = c_max + np.log(np.sum(np.exp(tau * (c - c_max)))) / tau
    return val * dx

    c = np.correlate(h, 1-h, mode='full')

- In (1-h[i+lag_k]): if i+lag_k = i (lag_k=0), change = -α; if i+lag_k = j (lag_k = j-i), change = +α (since h[j] decreased by α, so 1-h[j] increased by α)
- In h[i-lag_k]: if i-lag_k = i (lag_k=0), change = +α; if i-lag_k = j (lag_k = i-j), change = -α (since h[j] decreased by α)

So ΔS[i,k] = (-α if lag_k=0 else 0) + (α if lag_k=j-i else 0) - (α if lag_k=0 else 0) - (-α if lag_k=i-j else 0)
= -2α*δ(lag_k,0) + αδ(lag_k,j-i) + αδ(lag_k,i-j)

where δ is the Kronecker delta.

import numpy as np
import time

def project_h(h, target_sum):
    """Euclidean projection onto {0 <= x <= 1, sum(x) = target_sum}"""
    def sum_clipped(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    lo = -float(h.max()) - 1.0
    hi = 1.0 - float(h.min()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if sum_clipped(mid) < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def exact_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def precompute_slope_matrix(h, n):
    n_c = 2 * n - 1
    lags = np.arange(n_c) - (n - 1)
    i_idx = np.arange(n)[:, None]
    lag_idx = lags[None, :]
    sum_idx = i_idx + lag_idx
    mask1 = (sum_idx >= 0) & (sum_idx < n)
    term1 = np.zeros((n, n_c))
    term1[mask1] = 1.0 - h[sum_idx[mask1]]
    diff_idx = i_idx - lag_idx
    mask2 = (diff_idx >= 0) & (diff_idx < n)
    term2 = np.zeros((n, n_c))
    term2[mask2] = h[diff_idx[mask2]]
    return term1 - term2

def coord_descent(h, n, target_sum, max_passes, time_check=None, time_limit=None):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = exact_c5(h, n)
    n_c = 2 * n - 1
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi2 = 1.0 / (phi * phi)
    
    for p in range(max_passes):
        if time_check is not None and time.time() - time_check > time_limit:
            break
        S = precompute_slope_matrix(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        pass_improved = False
        
        for i in range(n):
            for j in range(i + 1, n):
                s = S[i] - S[j]
                t_lo = max(-h[i], h[j] - 1.0)
                t_hi = min(1.0 - h[i], h[j])
                if t_hi - t_lo < 1e-14:
                    continue
                
                a, b = t_lo, t_hi
                cr = b - inv_phi2 * (b - a)
                cd = a + inv_phi2 * (b - a)
                fc = float(np.max(c + s * cr))
                fd = float(np.max(c + s * cd))
                
                for _ in range(35):
                    if fc < fd:
                        b = cd
                        cd = cr
                        fd = fc
                        cr = b - inv_phi2 * (b - a)
                        fc = float(np.max(c + s * cr))
                    else:
                        a = cr
                        cr = cd
                        fc = fd
                        cd = a + inv_phi2 * (b - a)
                        fd = float(np.max(c + s * cd))
                
                t_opt = 0.5 * (a + b)
                val = float(np.max(c + s * t_opt))
                
                if val < float(c.max()) - 1e-13:
                    h[i] += t_opt
                    h[j] -= t_opt
                    np.clip(h, 0.0, 1.0, out=h)
                    c = np.correlate(h, 1.0 - h, mode='full')
                    pass_improved = True
                    new_c5 = float(np.max(c)) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
        
        if not pass_improved:
            break
    
    return h, best_c5

def smooth_grad_descent(h, n, target_sum, tau, max_iters, time_check, time_limit):
    h = h.copy()
    best_c5 = exact_c5(h, n)
    best_h = h.copy()
    dx = 2.0 / n
    
    # Backtracking line search parameters
    c1 = 1e-4
    rho = 0.5
    
    for it in range(max_iters):
        if time.time() - time_check > time_limit:
            break
        
        S = precompute_slope_matrix(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(c)
        exp_c = np.exp(tau * (c - cmax))
        w = exp_c / exp_c.sum()
        grad = dx * (w @ S.T)
        
        # Projected gradient step with line search
        lr = 0.1
        for ls in range(20):
            h_trial = h - lr * grad
            h_trial = project_h(h_trial, target_sum)
            val = (cmax + np.log(np.sum(np.exp(tau * (np.correlate(h_trial, 1.0 - h_trial, mode='full') - cmax)))) / tau) * dx
            
            if val < (cmax + np.log(np.sum(exp_c)) / tau) * dx - c1 * lr * np.dot(grad, grad):
                h = h_trial
                break
            lr *= rho
        
        val = exact_c5(h, n)
        if val < best_c5:
            best_c5 = val
            best_h = h.copy()
    
    return best_h, best_c5
