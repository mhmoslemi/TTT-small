    def f(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    lo, hi = -2.0, 2.0

        lo -= 1.0

        hi += 1.0

        return np.full(n, 0.5)
    t = brentq(f, lo, hi, xtol=1e-15)
    return np.clip(h + t, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode="full") * dx
    return scores

    scores = compute_scores_scaled(x)
    m = np.max(scores)
    exp_s = np.exp(beta * (scores - m))
    Z = np.sum(exp_s)
    return (m + np.log(Z)) / beta

    n = len(x)

Let me also think about the exchange refinement more carefully. The key insight is that for a pair (a, b) with exchange amount δ, the score is a piecewise function of δ. For each lag d, the score s_d(δ) is at most quadratic in δ (as I analyzed above). The overall score is max_d s_d(δ), which is the max of several quadratics.

For the 1D optimization over δ, I'll use a simple approach:
1. Determine the valid range of δ: [δ_min, δ_max]
2. Sample ~20-30 points in this range
3. Evaluate the exact score at each point
4. Pick the best
5. Optionally refine with golden section or fine grid

For selecting which pairs to try, I'll use:
- Pairs with high "gradient magnitude" (from the smooth gradient)
- Random pairs as a fallback
- Pairs where one element is near 0 and the other near 1 (to allow large δ)

    dx = 2.0 / n
    scores = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(scores))

    n = len(h)
    target = n / 2.0

        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    lo, hi = -2.0, 2.0
    for _ in range(20):
        if f(lo) <= 0: break
        lo *= 2.0
    for _ in range(20):
        if f(hi) >= 0: break
        hi *= 2.0
    if f(lo) > 0 or f(hi) < 0:
        return np.full(n, 0.5)
    t = brentq(f, lo, hi, xtol=1e-15)
    return np.clip(h + t, 0.0, 1.0)

def get_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_value(x, beta, state):
    check_time(state)
    scores = get_scores(x)
    m = np.max(scores)
    if m < 0: m = 0
    shifted = beta * (scores - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    Z = np.sum(np.exp(shifted))
    if Z == 0:
        return m
    return m + np.log(Z) / beta

def smooth_grad(x, beta, state):
    check_time(state)
    n = len(x)
    dx = 2.0 / n
    scores = get_scores(x)
    m = np.max(scores)
    shifted = beta * (scores - m)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    if Z == 0:
        return np.zeros(n)
    p = exp_s / Z
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * x)
        elif lag < 0:
            d = -lag
            klen = n - d
            grad[:klen] += w * (1.0 - x[d:])
            grad[d:] += w * (-x[:klen])
        else:
            d = lag
            klen = n - d
            grad[:klen] += w * (-x[d:])
            grad[d:] += w * (1.0 - x[:klen])
    
    return grad

def sum_constraint(x, target):
    return np.sum(x) - target

def run_slsqp(h, beta, state, maxiter=150):
    n = len(h)
    target = n / 2.0
    try:
        result = minimize(
            smooth_value, h, method='SLSQP',
            jac=smooth_grad, args=(beta, state),
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (target,)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return result.x
    except (TimeLimit, Exception):
        return h

def exchange_refine(h, state, max_pairs=2000):
    """Refine using exact exchange moves."""
    n = len(h)
    best_score = evaluate_exact(h)
    h_curr = h.copy()
    no_gain = 0
    pairs_tried = 0
    
    while pairs_tried < max_pairs:
        check_time(state)
        if no_gain > 50:
            break

        # Use smooth gradient with high beta as proxy
        try:
            grad = smooth_grad(h_curr, 200.0, state)
        except:
            grad = np.random.randn(n) * 0.01
        
        # Select pair: maximize |grad[a] - grad[b]| subject to feasibility
        # Simple approach: pick a with high positive grad, b with high negative grad
        # Or just random
        if np.random.random() < 0.3:
            # Use gradient-informed selection
            a = int(np.argmax(grad))
            b = int(np.argmin(grad))
        else:
            a = np.random.randint(n)
            b = np.random.randint(n)
        
        if a == b:
            continue
        
        # Determine valid range for delta
        # h[a] + delta in [0,1], h[b] - delta in [0,1]
        d_min = max(-h_curr[a], h_curr[b] - 1.0)
        d_max = min(1.0 - h_curr[a], h_curr[b])
        
        if d_max - d_min < 1e-10:
            no_gain += 1
            continue
        
        # Sample points in [d_min, d_max]
        n_samples = 15
        deltas = np.linspace(d_min, d_max, n_samples)

        best_s = best_score
        
        for delta in deltas:
            if abs(delta) < 1e-15:

            h_test = h_curr.copy()
            h_test[a] += delta
            h_test[b] -= delta

            if s_test < best_s:

                best_d = delta

            h_curr[a] += best_d
            h_curr[b] -= best_d
            best_score = best_s
            no_gain = 0
        else:
            no_gain += 1
        
        pairs_tried += 1
    
    return h_curr, best_score
