import numpy as np
import time


def evaluate_exact(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(z, n):
    target = n / 2.0
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - result.sum()
    if residual > 1e-10:
        order = np.argsort(1.0 - result)
        for idx in order:
            if residual <= 1e-12:
                break
            add = min(1.0 - result[idx], residual)
            result[idx] += add
            residual -= add
    elif residual < -1e-10:
        order = np.argsort(result)
        for idx in order:
            if -residual <= 1e-12:
                break
            rem = min(result[idx], -residual)
            result[idx] -= rem
            residual += rem
    return result


def compute_smooth_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    dvals = corr * dx
    max_d = dvals.max()
    w = np.exp((dvals - max_d) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n), dvals
    P = w / w_sum
    grad = np.zeros(n)
    for r in range(len(dvals)):
        p_r = P[r]
        if p_r < 1e-14:
            continue
        t = r - (n - 1)
        # Range 1: k in [max(0,-t), min(n, n-t)): grad[k] += p_r * dx * (1 - h[k+t])
        r1_lo = max(0, -t)
        r1_hi = min(n, n - t)
        if r1_hi > r1_lo:

            grad[idx1] += p_r * dx * (1.0 - h[idx1 + t])
        # Range 2: k in [max(0,t), min(n, n+t)): grad[k] -= p_r * dx * h[k-t]

            idx2 = np.arange(r2_lo, r2_hi)

def line_search(h, d, n, score_cur, max_alpha):

    neg_mask = d < -1e-15

def random_feasible(n, rng):
    z = rng.uniform(0.0, 1.0, n)
    return project_feasible(z, n)


def make_pair_swap(h, n, rng, max_delta):
    h_new = h.copy()
    i = int(rng.integers(n))
    j = int(rng.integers(n))
    if i == j:
        return h_new
    delta = rng.uniform(0.01, max_delta)
    if h[i] >= h[j]:
        d = min(h[i] - h[j], delta)
        h_new[i] -= d
        h_new[j] += d
    else:
        d = min(h[j] - h[i], delta)
        h_new[j] -= d
        h_new[i] += d
    return h_new


def local_refine(h_start, n, tau_init, max_iter, t0, budget_s):

            if not time_ok(10):
                break
            h_try = make_pair_swap(best_h, best_n, rng, 0.05)
            score_try = evaluate_exact(h_try, best_n)
            if score_try < best_score - 1e-10:
                best_h = h_try
                best_score = score_try

    # Phase 5: Final ultra-fine gradient polish
    if time_ok(10):
        tau_final = max(1e-4, 0.002 * best_score)
        best_h, best_score = local_refine(best_h, best_n, tau_final, 100, t0, budget_s)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_score = evaluate_exact(best_h, best_n)

    return best_h, final_score, int(best_n)
