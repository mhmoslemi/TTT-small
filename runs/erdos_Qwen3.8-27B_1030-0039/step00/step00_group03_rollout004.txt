import numpy as np
from scipy.optimize import minimize, brentq
import time

class TimeLimit:
    def __init__(self, deadline):
        self.deadline = deadline
    def check(self):
        if time.time() > self.deadline:
            raise TimeoutError()
    def remaining(self):
        return max(0, self.deadline - time.time())

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h_raw, n):
    """Project h_raw onto [0,1]^n with sum = n/2."""
    target = n / 2.0
    # Use bisection on offset t: find t such that sum(clip(h_raw + t, 0, 1)) = target
    lo = h_raw.min() - 2.0  # ensure all clipped to 0
    hi = 1.0 - h_raw.max() + 2.0  # ensure all clipped to 1
    
    def f(t):
        return np.sum(np.clip(h_raw + t, 0.0, 1.0)) - target
    
    lo_val = f(lo)
    hi_val = f(hi)
    if lo_val > 0:
        return np.clip(h_raw, 0.0, 1.0)  # already above target, shouldn't happen
    if hi_val < 0:
        return np.clip(h_raw, 0.0, 1.0)
    
    t = brentq(f, lo, hi, xtol=1e-15)
    return np.clip(h_raw + t, 0.0, 1.0)

def smooth_objective_and_grad(x, beta, n, limit):
    limit.check()
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    bcorr = beta * corr
    m = np.max(bcorr)
    w = np.exp(bcorr - m)
    Z = np.sum(w)
    psi = (m + np.log(Z)) / beta
    p = w / Z
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wkt = p[k] * dx
        if d >= 0:
            if d < n:
                grad[d:] += wkt * (1.0 - x[:n - d])
            if n - d > 0:
                grad[:n - d] += wkt * (-x[d:])
        else:
            pp = -d
            if pp < n:
                grad[:n - pp] += wkt * (1.0 - x[pp:])
            if n - pp > 0:
                grad[pp:] += wkt * (-x[:n - pp])
    
    return psi, grad

def constraint_sum(x, n):
    return np.sum(x) - n / 2.0

def smooth_optimize(h0, n, limit, betas=None, maxiter=None):
    """Run SLSQP with log-sum-exp smooth objective."""
    if maxiter is None:
        maxiter = 150
    if betas is None:
        betas = [10.0, 50.0, 200.0, 500.0]
    
    best_h = h0.copy()
    best_score = evaluate_exact(h0)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n)}]
    
    current = best_h.copy()
    
    for beta in betas:
        if limit.remaining() < 5.0:
            break
        try:
            res = minimize(
                lambda x: smooth_objective_and_grad(x, beta, n, limit)[0],
                current,
                jac=lambda x: smooth_objective_and_grad(x, beta, n, limit)[1],
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': maxiter, 'ftol': 1e-12}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                candidate = project_feasible(res.x, n)

    if d >= 0:
        if d < n:
            grad[d:] += dx * (1.0 - h[:n - d])
        if (n - d) > 0:
            grad[:n - d] += dx * (-h[d:])
    else:
        pp = -d
        if pp < n:
            grad[:n - pp] += dx * (1.0 - h[pp:])
        if (n - pp) > 0:
            grad[pp:] += dx * (-h[:n - pp])
    return grad

def coordinate_refine(h0, n, limit, max_rounds):
    best_h = h0.copy()
    best_score = evaluate_exact(h0)
    no_improve = 0
    rng = np.random.RandomState(42)
    for rnd in range(max_rounds):
        if limit.remaining() < 3.0:
            break
        if no_improve > 300:
            break
        ag = _active_lag_gradient(best_h, n)
        if rnd < 30:
            pairs = []
            for i in range(n):
                for j in range(i + 1, n):
                    pairs.append((i, j))
        else:
            pairs = []
            for _ in range(n * (n - 1) // 2):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j:
                    pairs.append((min(i, j), max(i, j)))
        pair_grad_diff = np.array([abs(ag[p[0]] - ag[p[1]]) for p in pairs])
        order = np.argsort(-pair_grad_diff)
        improved = False
        for idx in order:
            if limit.remaining() < 3.0:
                break
            if no_improve > 300:
                break
            i, j = pairs[idx]
            for direction in (1, -1):
                if direction == 1:
                    lo_d = max(-best_h[j], -0.5)
                    hi_d = min(1.0 - best_h[i], 0.5)
                else:
                    lo_d = max(-best_h[i], -0.5)
                    hi_d = min(1.0 - best_h[j], 0.5)
                if hi_d - lo_d < 1e-12:
                    continue
                deltas = np.linspace(lo_d, hi_d, 7)
                best_ld = 0.0
                best_ls = best_score
                for dv in deltas:
                    if abs(dv) < 1e-12:
                        continue
                    tmp = best_h.copy()
                    if direction == 1:
                        tmp[i] += dv
                        tmp[j] -= dv
                    else:
                        tmp[i] -= dv
                        tmp[j] += dv
                    sc = evaluate_exact(tmp)
                    if sc < best_ls:
                        best_ls = sc
                        best_ld = dv
                if best_ld != 0.0 and best_ls < best_score - 1e-12:
                    if direction == 1:
                        best_h[i] += best_ld
                        best_h[j] -= best_ld
                    else:
                        best_h[i] -= best_ld
                        best_h[j] += best_ld
                    best_h = project_feasible(best_h, n)
                    best_score = best_ls
                    improved = True
                    no_improve = 0
            if not improved:
                no_improve += 1
    return best_h, best_score

def transfer_resolution(h_old, new_n):
    m = len(h_old)
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h_old)
    return project_feasible(h_new, new_n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5.0
    limit = _TimeLimit(deadline)
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        base_n = len(parent)
    except Exception:
        parent = None
        base_n = 45

    if parent is not None and len(parent) > 0:
        best_h = project_feasible(parent, base_n)
        best_score = evaluate_exact(best_h)
        best_n = base_n
    else:
        best_h = np.full(base_n, 0.5)
        best_score = evaluate_exact(best_h)
        best_n = base_n

    if limit.remaining() > 40:
        h1, s1 = smooth_optimize(best_h.copy(), base_n, limit,
                                 [8.0, 30.0, 100.0, 300.0, 800.0, 2000.0], 200)
        if s1 < best_score:
            best_h, best_score, best_n = h1, s1, base_n

    for sigma in (0.02, 0.05, 0.10, 0.20):
        if limit.remaining() < 25:
            break
        pert = best_h + sigma * rng.randn(base_n)
        pert = project_feasible(pert, base_n)
        hp, sp = smooth_optimize(pert, base_n, limit,
                                 [10.0, 50.0, 200.0, 600.0], 120)
        if sp < best_score:
            best_h, best_score, best_n = hp, sp, base_n

    if limit.remaining() > 20:
        comp = 1.0 - best_h
        comp = project_feasible(comp, base_n)
        hc, sc = smooth_optimize(comp, base_n, limit,
                                 [10.0, 50.0, 200.0], 100)
        if sc < best_score:
            best_h, best_score, best_n = hc, sc, base_n

    if limit.remaining() > 20:
        rev = best_h[::-1].copy()
        hr, sr = smooth_optimize(rev, base_n, limit,
                                 [10.0, 50.0, 200.0], 100)
        if sr < best_score:
            best_h, best_score, best_n = hr, sr, base_n

    if limit.remaining() > 30:
        hc2, sc2 = coordinate_refine(best_h.copy(), best_n, limit, 500)
        if sc2 < best_score:
            best_h, best_score, best_n = hc2, sc2, best_n

    for tgt_n in (base_n * 2, base_n * 3, base_n * 4):
        if limit.remaining() < 50:
            break
        if tgt_n > 200:
            tgt_n = 181
        if tgt_n <= best_n:
            continue
        ht = transfer_resolution(best_h, tgt_n)
        st = evaluate_exact(ht)
        hn, sn = smooth_optimize(ht, tgt_n, limit,
                                 [8.0, 30.0, 100.0, 300.0, 800.0], 150)
        if sn < best_score:
            best_h, best_score, best_n = hn, sn, tgt_n
        if limit.remaining() > 20 and best_n == tgt_n:
            hcr, scr = coordinate_refine(best_h.copy(), best_n, limit, 300)
            if scr < best_score:
                best_h, best_score, best_n = hcr, scr, best_n

    if limit.remaining() > 15:
        if best_n < 90:
            tgt2 = min(90, base_n * 2)
            ht2 = transfer_resolution(best_h, tgt2)
            hn2, sn2 = smooth_optimize(ht2, tgt2, limit,
                                       [10.0, 50.0, 200.0, 500.0], 150)
            if sn2 < best_score:
                best_h, best_score, best_n = hn2, sn2, tgt2
        hf2, sf2 = coordinate_refine(best_h.copy(), best_n, limit, 800)
        if sf2 < best_score:
            best_h, best_score, best_n = hf2, sf2, best_n

    if limit.remaining() > 5:
        for _ in range(3):
            if limit.remaining() < 5:
                break
            pert_f = best_h + 0.01 * rng.randn(best_n)
            pert_f = project_feasible(pert_f, best_n)
            hpf, spf = coordinate_refine(pert_f, best_n, limit, 500)
            if spf < best_score:
                best_h, best_score, best_n = hpf, spf, best_n

    best_h = project_feasible(best_h, best_n)
    best_score = evaluate_exact(best_h)
    return (best_h, best_score, best_n)
