def evaluate_c5(h):

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)

        else:

    n = len(h)
    h = h.copy()
    best_val = evaluate_c5(h)

        for i in range(n):

                    c += (1.0 - h[j2])
                j1 = i - shift

Let shift = m - (N-1), so m = shift + (N-1), shift ranges from -(N-1) to (N-1).

coefs[shift + (N-1)] = (1 - h[i+shift]) * [0 ≤ i+shift < N] + (-h[i-shift]) * [0 ≤ i-shift < N]

So I can precompute this as:
- First term: for shifts where 0 ≤ i+shift < N, the value is (1-h[i+shift])
  This is a window of h centered at i, of length N (but truncated to valid range)
- Second term: for shifts where 0 ≤ i-shift < N, the value is -h[i-shift]
  This is also a window.

I can compute coefs using array slicing:
```python
# shifts range from -(N-1) to (N-1)
shifts = np.arange(-(N-1), N)
# First term: 1-h[i+shift] where 0 <= i+shift < N
valid1 = (i + shifts >= 0) & (i + shifts < N)
term1 = np.zeros(2*N-1)
term1[valid1] = 1.0 - h[i + shifts[valid1]]
# Second term: -h[i-shift] where 0 <= i-shift < N
valid2 = (i - shifts >= 0) & (i - shifts < N)
term2 = np.zeros(2*N-1)
term2[valid2] = -h[i - shifts[valid2]]
coefs = term1 + term2
```

This is O(N) vectorized operations, much faster than a Python loop.

Let me try a different approach: use scipy.optimize with a smoothed objective.

The log-sum-exp smoothing:
f_β(h) = (1/β) * log(sum_m exp(β * c[m] * dx))

As β → ∞, this approaches max_m c[m] * dx.

The gradient of f_β with respect to h[i]:
∂f_β/∂h[i] = (1/β) * sum_m [β * c[m]' * dx * exp(β * c[m] * dx)] / sum_m [exp(β * c[m] * dx)]
= dx * sum_m c[m]' * exp(β * c[m] * dx) / sum_m exp(β * c[m] * dx)
= dx * E_m[c[m]' | m ~ softmax(β * c * dx)]

where c[m]' is the derivative of c[m] with respect to h[i].

But computing the gradient requires knowing c[m]' for all m, which is 2N-1 values per coordinate. For N=90, that's 179 values per coordinate, 90 coordinates = 16K values per gradient evaluation. That's fine.

However, the log-sum-exp can overflow for large β. I should use the log-sum-exp trick:
LSE(x) = max(x) + log(sum(exp(x - max(x))))

Let me implement this approach:

```python
def smooth_c5(h, beta=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = beta * corr * dx
    max_val = np.max(vals)
    lse = max_val + np.log(np.sum(np.exp(vals - max_val)))
    return lse / beta  # This approximates max(corr * dx)

def smooth_c5_grad(h, beta=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = beta * corr * dx
    max_val = np.max(vals)
    weights = np.exp(vals - max_val)
    weights /= np.sum(weights)
    
    # Gradient: dx * sum_m weights[m] * d(c[m])/d(h[i])
    # d(c[m])/d(h[i]) = (1-h[i+shift]) * [valid] - h[i-shift] * [valid]
    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    for i in range(n):
        valid1 = (i + shifts >= 0) & (i + shifts < n)
        valid2 = (i - shifts >= 0) & (i - shifts < n)
        dc_dhi = np.zeros(2*n-1)
        dc_dhi[valid1] += (1.0 - h[i + shifts[valid1]])
        dc_dhi[valid2] += (-h[i - shifts[valid2]])
        grad[i] = dx * np.dot(weights, dc_dhi)
    return grad
```

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:

        n_mask = np.sum(mask)
        if n_mask == 0:
            break
        h[mask] += diff / n_mask
        h = np.clip(h, 0.0, 1.0)
    return h

def get_coefs(h, i):
    n = len(h)
    shifts = np.arange(-(n-1), n)
    v1 = (i + shifts >= 0) & (i + shifts < n)
    v2 = (i - shifts >= 0) & (i - shifts < n)
    coefs = np.zeros(2*n - 1)
    coefs[v1] += (1.0 - h[i + shifts[v1]])
    coefs[v2] += (-h[i - shifts[v2]])
    return coefs

    n = len(h)
    coefs = get_coefs(h, i)
    old = h[i]
    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode="full")
    h[i] = old
    
    # Minimize max(base + coefs*t) over t in [0,1]
    # Convex PL minimization
    # Quick: check if min is at boundary
    v0 = float(np.max(base))
    v1 = float(np.max(base + coefs))
    
    # Search for interior minimum
    # The derivative changes sign at the optimum
    # At t, the active set is argmax_m (base[m] + coefs[m]*t)
    # If all active coefs > 0, decrease t
    # If all active coefs < 0, increase t
    
    # Practical: grid search + bisection
    ts = np.linspace(0.0, 1.0, 31)
    vals = np.max(base[None, :] + coefs[:, None] * ts[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    t_best = float(ts[best_idx])

            t_best = float(ts_f[np.argmin(vals_f)])

    if v1 <= vb:

import cvxpy as cp

def linearization_step(h0, n):
    """One step of the linearization (cutting plane) method."""
    # Compute c0 and Jacobian
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    
    # Compute Jacobian J[k, m] = d(c[m])/d(h[k]) at h0
    shifts = np.arange(-(n-1), n)
    J = np.zeros((n, 2*n-1))
    for k in range(n):
        v1 = (k + shifts >= 0) & (k + shifts < n)
        v2 = (k - shifts >= 0) & (k - shifts < n)
        J[k, v1] += (1.0 - h0[k + shifts[v1]])
        J[k, v2] += (-h0[k - shifts[v2]])
    
    # Set up LP
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        g >= -h0 + 1e-10,
        g <= 1.0 - h0 - 1e-10,
        cp.sum(g) == 0
    ]
    # c0[m] + sum_k J[k,m] * g[k] <= t
    # J.T @ g + c0 <= t
    constraints.append(J.T @ g + corr0 <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or CBC, GLPK, etc.
    
    if g.value is not None:
        h_new = h0 + g.value
        h_new = project_feasible(h_new, n)
        return h_new
    return h0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_mask = np.sum(mask)
        if n_mask == 0:
            break
        h[mask] += diff / n_mask
        h = np.clip(h, 0.0, 1.0)
    return h

def get_jacobian(h, n):
    """Compute J[k, m] = d(c[m])/d(h[k]) at h."""
    shifts = np.arange(-(n-1), n)
    J = np.zeros((n, 2*n-1))
    for k in range(n):
        v1 = (k + shifts >= 0) & (k + shifts < n)
        v2 = (k - shifts >= 0) & (k - shifts < n)
        J[k, v1] += (1.0 - h[k + shifts[v1]])
        J[k, v2] += (-h[k - shifts[v2]])
    return J

def lp_step(h, n, alpha=1.0):
    """One LP-based step."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    J = get_jacobian(h, n)
    
    g = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        g >= -h,
        g <= 1.0 - h,
        cp.sum(g) == 0,
        J.T @ g + corr <= t_var
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CBC)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return h
    
    if g.value is not None and not np.any(np.isnan(g.value)):
        h_new = h + alpha * g.value
        h_new = project_feasible(h_new, n)
        return h_new
    return h

            v2 = (i - shifts >= 0) & (i - shifts < n)
            coefs = np.zeros(2*n - 1)
            coefs[v1] += (1.0 - h[i + shifts[v1]])
            coefs[v2] += (-h[i - shifts[v2]])
            
            old = h[i]
            h[i] = 0.0
            base = np.correlate(h, 1.0 - h, mode="full")
            h[i] = old
            
            ts = np.linspace(0.0, 1.0, 31)
            vals = np.max(base[None, :] + coefs[:, None] * ts[None, :], axis=1)
            best_idx = int(np.argmin(vals))
            t_best = float(ts[best_idx])
            
            if 0 < best_idx < 30:

        val = evaluate_c5(h)
        if val < best_val:

        if max_change < 1e-12:
