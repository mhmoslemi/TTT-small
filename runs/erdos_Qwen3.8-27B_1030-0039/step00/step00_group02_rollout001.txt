```python
def compute_gradient_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = 1e-4
    active = np.where(corr >= max_val - tol)[0]
    
    grad_h = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        # dc[k]/dh[j] = (1 - h[j-lag]) * [0<=j-lag<n] - h[j+lag] * [0<=j+lag<n]
        # This is: sum over j of (1-h[j-lag]) - h[j+lag] (with valid index conditions)
        
        # Vectorized:
        # For the first term: (1-h[j-lag]) for valid j-lag
        # For the second term: -h[j+lag] for valid j+lag
        
        if lag >= 0:
            # j-lag valid when j >= lag, i.e., j in [lag, n-1]
            # j+lag valid when j <= n-1-lag, i.e., j in [0, n-1-lag]
            idx1 = np.arange(lag, n)
            idx2 = np.arange(0, n-lag)
            grad_h[idx1] += (1.0 - h[idx1 - lag])
            grad_h[idx2] -= h[idx2 + lag]
        else:
            # lag < 0, let L = -lag > 0
            L = -lag
            # j-lag = j+L valid when j+L < n, i.e., j in [0, n-1-L]
            # j+lag = j-L valid when j >= L, i.e., j in [L, n-1]
            idx1 = np.arange(0, n-L)
            idx2 = np.arange(L, n)
            grad_h[idx1] += (1.0 - h[idx1 + L])
            grad_h[idx2] -= h[idx2 - L]
    
    grad_h /= len(active)
    return grad_h * dx
```

This is O(n) per active k, and we have at most 2n-1 active k's (but usually much fewer). So total is O(n * |active|).

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_gradient_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = max(1e-4, 0.01 * max_val)
    active = np.where(corr >= max_val - tol)[0]
    if len(active) == 0:
        return np.zeros(n)
    grad_h = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                idx1 = np.arange(lag, n)
                grad_h[idx1] += (1.0 - h[idx1 - lag])
                idx2 = np.arange(0, n - lag)
                grad_h[idx2] -= h[idx2 + lag]
        else:
            L = -lag
            if L < n:
                idx1 = np.arange(0, n - L)
                grad_h[idx1] += (1.0 - h[idx1 + L])
                idx2 = np.arange(L, n)
                grad_h[idx2] -= h[idx2 - L]
    grad_h /= len(active)
    return grad_h * dx

def project_simplex_box(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        idx_max = np.argmax(x < 1.0) if diff > 0 else np.argmin(x > 0.0)
        if diff > 0:
            x[idx_max] = min(x[idx_max] + diff, 1.0)
        else:
            x[idx_max] = max(x[idx_max] + diff, 0.0)
    return x

def subgradient_optimize(h, target_sum, max_iters, alpha0, decay, time_check):
    n = len(h)
    h = h.copy()
    best_h_local = h.copy()
    best_c5_local = compute_c5(h)
    for t in range(max_iters):
        if time_check():
            break
        alpha = alpha0 / (1.0 + decay * t)
        g = compute_gradient_fast(h)
        h_new = h - alpha * g
        h_new = project_simplex_box(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5_local:
            best_c5_local = c5
            best_h_local = h_new.copy()
        h = h_new
    return best_h_local, best_c5_local

def coordinate_descent_optimize(h, target_sum, max_sweeps, time_check):
    n = len(h)
    h = h.copy()
    best_c5_local = compute_c5(h)
    for it in range(max_sweeps):
        if time_check():
            break
        improved = False
        for j in range(n):
            if time_check():
                break
            def make_obj(idx):
                def obj_1d(x):
                    h_copy = h.copy()
                    h_copy[idx] = x
                    h_copy = project_simplex_box(h_copy, target_sum)
                    return compute_c5(h_copy)
                return obj_1d
            obj = make_obj(j)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            if res.fun < best_c5_local - 1e-12:
                h[j] = res.x
                h = project_simplex_box(h, target_sum)
                best_c5_local = res.fun
                improved = True
            else:
                h = project_simplex_box(h, target_sum)
        if not improved:
            break
    return h, best_c5_local
```
