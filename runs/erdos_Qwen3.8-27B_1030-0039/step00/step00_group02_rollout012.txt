    """Generate an m-sequence.
    poly_bits: integer representing the feedback taps (excluding the leading 1)
    e.g., for x^4 + x + 1, poly_bits = 0b0011 = 3 (taps at positions 0 and 1)

    period = 2**degree - 1
    if length is None:

    # LFSR state: degree bits
    state = 1  # Initial state (non-zero)
    seq = []
    
    for _ in range(period):

                feedback ^= (state >> bit) & 1
        # Shift right and add feedback to top

    return np.array(seq[:length])

import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def make_binary(values, n, rng):
    if len(values) == n:
        h = (values >= 0.5).astype(float)
    else:
        x_old = np.linspace(0.0, 2.0, len(values))
        x_new = np.linspace(0.0, 2.0, n)
        interp = np.interp(x_new, x_old, values)
        h = (interp >= 0.5).astype(float)
    target = n // 2
    current = int(h.sum())
    if current > target:
        idx = np.where(h == 1.0)[0]
        rng.shuffle(idx)
        for k in range(current - target):
            h[idx[k]] = 0.0
    elif current < target:
        idx = np.where(h == 0.0)[0]
        rng.shuffle(idx)
        for k in range(target - current):
            h[idx[k]] = 1.0
    return h


def random_binary(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h


def sturmian_binary(n, slope, rng):
    h = np.zeros(n)
    ones_needed = n // 2
    seq = np.zeros(n, dtype=int)
    s = 0.0
    for i in range(n):
        s += slope
        if s >= 1.0:
            s -= 1.0
            seq[i] = 1
    if seq.sum() != ones_needed:
        idx_ones = np.where(seq == 1)[0]
        idx_zeros = np.where(seq == 0)[0]
        if seq.sum() > ones_needed:
            rng.shuffle(idx_ones)
            for k in range(int(seq.sum()) - ones_needed):
                seq[idx_ones[k]] = 0
        else:
            rng.shuffle(idx_zeros)
            for k in range(ones_needed - int(seq.sum())):
                seq[idx_zeros[k]] = 1
    return seq.astype(float)


def simulated_annealing(h, time_limit, rng):
    n = len(h)
    h = h.astype(float).copy()
    start = time.time()
    current = evaluate_c5(h)
    best = current
    best_h = h.copy()
    T = 0.05
    T_min = 0.0005
    cooling = 0.9995
    no_improve = 0
    patience = 3000

    while time.time() - start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        a = int(rng.integers(len(ones)))
        b = int(rng.integers(len(zeros)))
        pa, pb = ones[a], zeros[b]
        h[pa], h[pb] = h[pb], h[pa]
        new_val = evaluate_c5(h)
        delta = new_val - current
        if delta < 0:
            current = new_val
            no_improve += 1
            if current < best:
                best = current
                best_h = h.copy()
                no_improve = 0
        else:
            if T > T_min and rng.random() < np.exp(-delta / T):
                current = new_val
                no_improve += 1
            else:
                h[pa], h[pb] = h[pb], h[pa]
                no_improve += 1
        T *= cooling
        if no_improve > patience:
            T = 0.03
            no_improve = 0
        if T < T_min:
            T = T_min

    if best < current:
        return best_h, best
    return h, current


def best_improvement(h, max_passes=300):
    n = len(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    def check_update(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = None

    n_list = sorted(set([48, 50, 52, 54, 56, 60, 64, 72, 80]))
    if h_init is not None:
        ni = len(h_init)
        if ni % 2 == 0:
            n_list.insert(0, ni)
        else:
            n_list.insert(0, ni + 1)
    n_list = sorted(set(n_list))

    total_candidates = len(n_list) * 3 + 4
    per_candidate = (budget_s * 0.85) / total_candidates

    for n in n_list:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 10:
            break

        alloc = min(per_candidate, remaining / max(1, (len(n_list) - n_list.index(n)) * 3))
        alloc = max(alloc, 2.0)

        inits = []
        if h_init is not None:
            inits.append(make_binary(h_init, n, np.random.default_rng(seed + n)))
        inits.append(random_binary(n, np.random.default_rng(seed + 1000 + n)))
        slope = 0.5 + 0.01 * (n % 7)
        inits.append(sturmian_binary(n, slope, np.random.default_rng(seed + 2000 + n)))

        for k, h0 in enumerate(inits):
            elapsed2 = time.time() - start_time
            remaining2 = budget_s - elapsed2
            if remaining2 < 5:
                break
            t_alloc = min(alloc, remaining2 * 0.4)
            if t_alloc < 1.0:
                t_alloc = max(1.0, remaining2 * 0.3)
            srng = np.random.default_rng(seed + n * 10 + k)
            h_res, c5_res = simulated_annealing(h0, t_alloc, srng)
            check_update(h_res, c5_res)

    # Additional random restarts at various n
    for n in [48, 50, 52, 56, 60, 64]:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 15:
            break
        t_alloc = min(15.0, remaining * 0.3)
        if t_alloc < 1.0:
            break
        h0 = random_binary(n, np.random.default_rng(seed + 5000 + n))
        srng = np.random.default_rng(seed + 6000 + n)
        h_res, c5_res = simulated_annealing(h0, t_alloc, srng)
        check_update(h_res, c5_res)

    # Final best-improvement polish on the best solution found
    if best_h is not None:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining > 5 and len(best_h) <= 80:
            npass = max(1, int(remaining // 3))
            npass = min(npass, 200)
            best_h, best_c5 = best_improvement(best_h, max_passes=npass)

    if best_h is None:
        n = 48
        best_h = random_binary(n, rng)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
