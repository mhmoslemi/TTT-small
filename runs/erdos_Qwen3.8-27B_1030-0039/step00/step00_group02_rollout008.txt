    # Set k_ones values to 1, rest to 0, adjust one value
    h_rounded[:] = 0.0
    h_rounded[sorted_idx[:k_ones]] = 1.0

            best_h = h_new.copy()

import numpy as np
import time

def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _to_01(h, target_sum):
    n = len(h)
    h_new = np.zeros(n)
    k_full = int(np.floor(target_sum))
    frac = target_sum - k_full
    sorted_idx = np.argsort(h)[::-1]
    h_new[sorted_idx[:k_full]] = 1.0
    if frac > 1e-10:
        h_new[sorted_idx[k_full]] = frac
    return h_new

def _search_combinatorial(h, time_limit):
    n = len(h)
    c5_curr = _c5_value(h)
    t_start = time.time()
    stagnant = 0

        zeros = np.where(h <= 0.001)[0]

            break
        
        best_improvement = 0.0
        best_pair = None
        
        for i in ones:
            if time.time() - t_start > time_limit:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = _c5_value(h)
                h[i] = 1.0
                h[j] = 0.0
                imp = c5_curr - c
                if imp > best_improvement + 1e-12:
                    best_improvement = imp
                    best_pair = (i, j)

            c5_curr = _c5_value(h)

            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
