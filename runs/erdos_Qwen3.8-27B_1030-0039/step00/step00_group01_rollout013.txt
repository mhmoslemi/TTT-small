    dx = 2.0 / n

        if wk < 1e-12:
            continue
        lag = k - (n - 1)

        h_try = h + a * d
        h_try = _project(h_try, n)
        s = _eval(h_try)
        if s < best_s:
            best_s = s
            best_h = h_try
    return best_h, best_s

def _local_search_01(h01, n, rng, max_iters, deadline):
    c5_cur = _eval(h01)
    best_local = h01.copy()
    best_c5_local = c5_cur
    h_cur = h01.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h_cur >= 1.0 - 1e-9)[0]
        zeros = np.where(h_cur <= 1e-9)[0]
        fracs = np.where((h_cur > 1e-9) & (h_cur < 1.0 - 1e-9))[0]
        
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_save = h_cur.copy()
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        s = _eval(h_cur)
        if s < c5_cur:
            c5_cur = s
            if s < best_c5_local:
                best_c5_local = s
                best_local = h_cur.copy()
        else:
            h_cur = h_save
    
    return best_local, best_c5_local

def _fractional_tune(h, n, rng, max_iters, deadline):
    c5_cur = _eval(h)
    best_local = h.copy()
    best_c5_local = c5_cur
    h_cur = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        h_save = h_cur.copy()
        delta = rng.normal(0, 0.005)
        h_cur[i] = h_cur[i] + delta
        h_cur = _project(h_cur, n)
        s = _eval(h_cur)
        if s < c5_cur:
            c5_cur = s
            if s < best_c5_local:
                best_c5_local = s
                best_local = h_cur.copy()
        else:
            h_cur = h_save
    
    return best_local, best_c5_local

def _gradient_descent(h, n, rng, max_iters, deadline):
    best_h = h.copy()
    best_c5 = _eval(h)
    h_cur = h.copy()
    c5_cur = best_c5
    tau = 0.003
    scale = 0.08
    stuck = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        g = _grad_smooth(h_cur, tau)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        d = -g / gn
        
        h_try, s_try = _try_step(h_cur, d, n, scale)
        if s_try < c5_cur - 1e-12:
            h_cur = h_try
            c5_cur = s_try
            stuck = 0
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
        else:
            stuck += 1
            if stuck > 30:
                tau = max(tau * 0.7, 1e-5)
                scale *= 0.7
                if stuck > 100:
                    h_cur = h_cur + rng.normal(0, 0.01, n)
                    h_cur = _project(h_cur, n)
                    c5_cur = _eval(h_cur)
                    tau = 0.005
                    scale = 0.05
                    stuck = 0
            else:
                tau = max(tau * 0.95, 1e-5)
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.93
    
    h_init = globals().get('initial_h_values', None)
    if h_init is not None:
        h = np.array(h_init, dtype=np.float64).copy()
    else:
        h = np.full(63, 0.5)
    n = len(h)
    h = _project(h, n)
    
    best_h = h.copy()
    best_c5 = _eval(h)
    
    phase1_deadline = t0 + (deadline - t0) * 0.35
    best_h, best_c5 = _gradient_descent(best_h, n, rng, 200000, phase1_deadline)
    
    # Also try from the original
    h2 = _project(h.copy(), n)
    h2, c5_2 = _gradient_descent(h2, n, rng, 100000, phase1_deadline)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    phase2_deadline = t0 + (deadline - t0) * 0.65
    # 0-1 local search
    target = n // 2
    idx = np.argsort(best_h)[::-1]
    h01 = np.zeros(n)
    if n % 2 == 0:
        h01[idx[:target]] = 1.0
    else:
        h01[idx[:target]] = 1.0
        h01[idx[target]] = 0.5
    c5_01 = _eval(h01)
    
    if c5_01 < best_c5:
        best_c5 = c5_01
        best_h = h01.copy()
    
    h01_best, c5_01_best = _local_search_01(h01, n, rng, 500000, phase2_deadline)
    if c5_01_best < best_c5:
        best_c5 = c5_01_best
        best_h = h01_best.copy()
    
    # Try a few random 0-1 starts
    for trial in range(5):
        if time.time() > phase2_deadline:
            break
        h_rand = np.zeros(n)
        if n % 2 == 0:
            perm = rng.permutation(n)
            h_rand[perm[:n // 2]] = 1.0
        else:
            perm = rng.permutation(n)
            h_rand[perm[:n // 2]] = 1.0
            h_rand[perm[n // 2]] = 0.5
        h_rand_best, c5_rand = _local_search_01(h_rand, n, rng, 50000, phase2_deadline)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand_best.copy()
    
    # Phase 3: Fractional fine-tuning
    phase3_deadline = deadline
    for _ in range(3):
        if time.time() > phase3_deadline:
            break
        h_tuned, c5_tuned = _fractional_tune(best_h.copy(), n, rng, 200000, phase3_deadline)
        if c5_tuned < best_c5:
            best_c5 = c5_tuned
            best_h = h_tuned.copy()
    
    # Final gradient polish
    if time.time() < deadline - 5:
        h_polished, c5_pol = _gradient_descent(best_h.copy(), n, rng, 100000, deadline)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_polished.copy()
    
    # Ensure feasibility
    best_h = _project(best_h, n)
    final_c5 = _eval(best_h)
    if final_c5 > best_c5 + 1e-9:
        best_h = best_h  # keep as is, re-evaluate
        final_c5 = _eval(best_h)
    
    return best_h, final_c5, n
