    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on additve offset
    lo = -max(h) - 0.1
    hi = 1.0 - min(h) + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    offset = (lo + hi) / 2.0
    result = np.clip(h + offset, 0.0, 1.0)
    # Fix sum exactly
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to add mass: prioritize elements with most room
            order = np.argsort(-(1.0 - result))

            # Need to remove mass: prioritize elements with most value
            order = np.argsort(result)

                break

                result[i] += add
                diff -= add
            else:
                sub = min(-diff, result[i])
                result[i] -= sub
                diff += sub
    return result

            j_arr = np.arange(j_lo, j_hi + 1)
            grad[j_arr] -= w[r] * dx * h[j_arr + t]

    for r in range(2 * n - 1):
        if w[r] < 1e-14:
            continue
        t = r - (n - 1)
        i_lo = max(t, 0)
        i_hi = min(n - 1 + t, n - 1)
        if i_lo <= i_hi:
            i_arr = np.arange(i_lo, i_hi + 1)
            grad[i_arr] += w[r] * dx * (1.0 - h[i_arr - t])
        j_lo = max(0, -t)
        j_hi = min(n - 1, n - 1 - t)
        if j_lo <= j_hi:
            j_arr = np.arange(j_lo, j_hi + 1)
            grad[j_arr] -= w[r] * dx * h[j_arr + t]
    return smooth_val, grad

def optimize_smooth(h_start, n, tau_init, lr_init, n_iter, rng, deadline):
    dx = 2.0 / n
    h = project_feasible(h_start, n)
    current_score = evaluate_c5(h, n)
    best_local = h.copy()
    best_local_score = current_score
    tau = tau_init
    lr = lr_init
    no_improve = 0
    stall_limit = 150

    for it in range(n_iter):
        if time.time() > deadline:
            break
        smooth_val, grad = compute_obj_and_grad(h, n, tau)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            tau *= 0.5
            if tau < 1e-8:
                break
            continue
        direction = -grad / gnorm
        direction = direction - np.mean(direction)
        base_alpha = lr / max(1.0, np.linalg.norm(direction))
        best_alpha = 0.0
        best_smooth = smooth_val
        best_h_trial = None
        for k in range(20):
            alpha = base_alpha * (0.7 ** k)
            h_trial = project_feasible(h + alpha * direction, n)
            sv, _ = compute_obj_and_grad(h_trial, n, tau)
            if sv < best_smooth - 1e-10:
                best_smooth = sv
                best_alpha = alpha
                best_h_trial = h_trial
            if k > 8 and sv > best_smooth:
                break
        if best_h_trial is not None:
            h = best_h_trial
            new_score = evaluate_c5(h, n)
            if new_score < best_local_score - 1e-10:
                best_local = h.copy()
                best_local_score = new_score
                no_improve = 0
            else:
                no_improve += 1
            tau = max(tau * 0.998, 1e-7)
        else:
            no_improve += 1
            tau = max(tau * 0.6, 1e-8)
            if tau < 1e-7:
                noise = rng.normal(0, 0.15, n)
                noise -= noise.mean()
                h = project_feasible(h + noise, n)
                current_score = evaluate_c5(h, n)
                tau = tau_init * 0.3
                no_improve = 0
        if no_improve > stall_limit:
            noise_scale = 0.25 * max(1, (no_improve - stall_limit) // 50)
            noise = rng.normal(0, noise_scale, n)
            noise -= noise.mean()
            h = project_feasible(h + noise, n)
            current_score = evaluate_c5(h, n)
            tau = tau_init * 0.5
            no_improve = 0

    return best_local, best_local_score

def pairwise_optimize(h, n, n_moves, rng, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_score = evaluate_c5(h, n)
    target = n / 2.0
    accepted = 0
    for it in range(n_moves):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        magnitude = rng.uniform(0.005, 0.08)
        if rng.random() < 0.5:
            delta = magnitude
        else:
            delta = -magnitude
        if h[i] + delta < -1e-12 or h[i] + delta > 1.0 + 1e-12:
            continue
        if h[j] - delta < -1e-12 or h[j] - delta > 1.0 + 1e-12:
            continue
        h_trial = h.copy()
        h_trial[i] += delta
        h_trial[j] -= delta
        score = evaluate_c5(h_trial, n)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_trial.copy()
            h = h_trial.copy()
            accepted += 1
    return best_h, best_score

def multi_pair_optimize(h, n, n_moves, rng, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_score = evaluate_c5(h, n)
    for it in range(n_moves):
        if time.time() > deadline:
            break
        k = rng.integers(2, 5)
        indices = rng.choice(n, size=k, replace=False)
        src = indices[0]
        dsts = indices[1:]
        total = h[src]
        if total < 1e-10:
            continue
        fracs = rng.dirichletion(np.ones(k - 1))
        h_trial = h.copy()
        for idx, fr in zip(dsts, fracs):
            amt = total * fr * rng.uniform(0.1, 0.5)
            if amt > h[src]:
                amt = h[src]
            if h_trial[idx] + amt > 1.0:
                amt = max(0.0, 1.0 - h_trial[idx])
            h_trial[idx] += amt
            h_trial[src] -= amt
        if h_trial[src] < -1e-12:
            continue
        if np.any(h_trial < -1e-12) or np.any(h_trial > 1.0 + 1e-12):
            continue
        score = evaluate_c5(h_trial, n)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_trial.copy()
            h = h_trial.copy()
    return best_h, best_score

def threshold_and_smooth(h, n, rng, deadline):
    """Try thresholding to binary-like, then smooth."""
    candidates = []
    thresholds = [0.3, 0.4, 0.5, 0.6, 0.7]
    for thr in thresholds:
        h_bin = (h > thr).astype(np.float64)
        h_bin = project_feasible(h_bin, n)
        h_smooth = project_feasible(h_bin + rng.normal(0, 0.05, n), n)
        h_smooth = project_feasible(h_smooth, n)
        score = evaluate_c5(h_smooth, n)
        candidates.append((h_smooth, score))
    candidates.sort(key=lambda x: x[1])
    return candidates[0]

def upsample(h, n_old, n_new):
    """Upsample from n_old to n_new using linear interpolation + resample."""
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, np.concatenate([h[:1], h, h[-1:]]),
                         period=2.0) if False else np.interp(x_new, x_old, h)
    # Handle wrap-around for periodic extension
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_interp = np.interp(x_new, x_ext, h_ext)
    return project_feasible(h_interp, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s * 0.93
    rng = np.random.default_rng(seed)

    h0 = globals().get('initial_h_values', None)
    if h0 is not None:
        h0 = np.array(h0, dtype=np.float64).copy()
        n0 = len(h0)
    else:
        n0 = 63
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_score = evaluate_c5(best_h, n0)
    best_n = n0

    tau_init = 0.012
    lr_init = 0.8

    # Phase 1: Gradient descent from initial (30% of time)
    d1 = min(t_start + budget_s * 0.25, deadline)
    h1, s1 = optimize_smooth(h0, n0, tau_init, lr_init, 5000, rng, d1)
    if s1 < best_score:
        best_h, best_score = h1, s1

    # Phase 2: Pairwise refinement (10% of time)
    d2 = min(budgets_time(t_start, budget_s, 0.35), deadline)
    h2, s2 = pairwise_optimize(best_h, best_n, 200000, rng, d2)
    if s2 < best_score:
        best_h, best_score = h2, s2

    # Phase 3: Multiple restarts with perturbation (30% of time)
    n_restarts = 8
    restart_start = t_start + budget_s * 0.35
    restart_end = t_start + budget_s * 0.65
    restart_time = max(1.0, (restart_end - restart_start) / n_restarts)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        rd = min(t_start + budget_s * 0.35 + (r + 1) * restart_time, deadline)
        noise_scale = 0.4 * (0.8 ** r)
        noise = rng.normal(0, noise_scale, best_n)
        noise -= noise.mean()
        h_start_r = project_feasible(best_h + noise, best_n)
        # Sometimes add structured perturbation
        if r % 3 == 2:
            freq = rng.uniform(2.0, 8.0)
            x = np.linspace(0, 2.0, best_n, endpoint=False)
            structured = 0.1 * np.sin(2 * np.pi * freq * x)
            structured -= structured.mean()
            h_start_r = project_feasible(h_start_r + structured, best_n)
        hr, sr = optimize_smooth(h_start_r, best_n, tau_init * 0.6, lr_init * 0.6,
                                  2000, rng, rd)
        if sr < best_score:
            best_h, best_score = hr, sr
        # Quick pairwise polish
        if sr < best_score + 0.01:
            hp, sp = pairwise_optimize(hr, best_n, 50000, rng, rd)
            if sp < best_score:
                best_h, best_score = hp, sp

    # Phase 4: Threshold-based restarts (10% of time)
    d4 = min(t_start + budget_s * 0.75, deadline)
    for r in range(3):
        if time.time() > deadline:
            break
        h_th, s_th = threshold_and_smooth(best_h, best_n, rng, d4)
        if s_th < best_score + 0.02:
            h_opt, s_opt = optimize_smooth(h_th, best_n, tau_init * 0.5, lr_init * 0.5,
                                           2000, rng, d4)
            if s_opt < best_score:
                best_h, best_score = h_opt, s_opt

    # Phase 5: Pairwise + multi-pair fine-tuning (10% of time)
    d5 = min(t_start + budget_s * 0.85, deadline)
    h5, s5 = pairwise_optimize(best_h, best_n, 300000, rng, d5)
    if s5 < best_score:
        best_h, best_score = h5, s5
    h5b, s5b = multi_pair_optimize(best_h, best_n, 100000, rng, d5)
    if s5b < best_score:
        best_h, best_score = h5b, s5b

    # Phase 6: Final gradient polish (10% of time)
    d6 = deadline
    h6, s6 = optimize_smooth(best_h, best_n, 0.003, 0.2, 10000, rng, d6)
    if s6 < best_score:
        best_h, best_score = h6, s6

    # Final: try upsample to larger n if time permits
    if time.time() < deadline - 30:
        for n_try in [n0 * 2]:
            if time.time() > deadline - 20:
                break
            h_up = upsample(best_h, best_n, n_try)
            h_up_opt, s_up = optimize_smooth(h_up, n_try, tau_init * 0.4, lr_init * 0.4,
                                             3000, rng, deadline)
            if s_up < best_score:
                best_h = h_up_opt
                best_score = s_up
                best_n = n_try

    # Final pairwise polish
    if time.time() < deadline:
        h_final, s_final = pairwise_optimize(best_h, best_n, 100000, rng, deadline)
        if s_final < best_score:
            best_h, best_score = h_final, s_final

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_score = evaluate_c5(best_h, best_n)
    if final_score > best_score + 1e-10:
        # Something went wrong in final projection, use previous
        pass

    return best_h, final_score, best_n

def budgets_time(t_start, budget_s, frac):
    return t_start + budget_s * frac
