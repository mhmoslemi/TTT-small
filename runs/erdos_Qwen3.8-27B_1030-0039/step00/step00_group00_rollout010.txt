import numpy as np
from scipy.optimize import minimize
import time


def get_parent():
    try:
        vals = initial_h_values
        if vals is not None and len(np.atleast_1d(vals)) > 0:
            return np.array(np.atleast_1d(vals), dtype=float, copy=True)
    except Exception:
        pass
    return None


def repair_feasible(h, N):
    h = np.array(h, dtype=float, copy=True).ravel()
    if h.shape[0] != N:
        if h.size == 0:
            h = np.full(N, 0.5)
        else:
            h = np.interp(np.linspace(0, h.size - 1, N), np.arange(h.size), h)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0

    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    t = (lo + hi) / 2.0
    result = np.clip(h - t, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            cands = np.where(result < 1.0 - 1e-12)[0]
        else:
            cands = np.where(result > 1e-12)[0]
        if len(cands) > 0:
            per = diff / len(cands)
            result[cands] += per
            result = np.clip(result, 0.0, 1.0)
            diff2 = target - np.sum(result)
            if abs(diff2) > 1e-14:
                if diff2 > 0:
                    cands2 = np.where(result < 1.0)[0]
                else:
                    cands2 = np.where(result > 0.0)[0]
                if len(cands2) > 0:
                    result[cands2[0]] += diff2
                    result = np.clip(result, 0.0, 1.0)
    return result


def evaluate_exact(h):
    h = np.asarray(h, dtype=float)
    N = len(h)
    dx = 2.0 / N
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals))


def smooth_obj_and_grad(h, beta):
    N = len(h)
    dx = 2.0 / N
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(vals)
    scaled = beta * (vals - max_val)
    exp_vals = np.exp(scaled)
    sum_exp = np.sum(exp_vals)
    if sum_exp <= 1e-300:
        obj = max_val
        weights = np.ones(2 * N - 1) / (2 * N - 1)
    else:
        obj = max_val + np.log(sum_exp) / beta
        weights = exp_vals / sum_exp

    grad = np.zeros(N)
    wc = weights[N - 1]
    for j in range(N):
        g = wc * (1.0 - 2.0 * h[j])
        for s in range(1, j + 1):
            i = j - s
            g += weights[N - 1 + s] * (1.0 - h[i])
            g += -weights[N - 1 - s] * h[i]
        for s in range(1, N - j):
            i = j + s
            g += -weights[N - 1 + s] * h[i]
            g += weights[N - 1 - s] * (1.0 - h[i])
        grad[j] = dx * g
    return obj, grad


def _slsqp_obj(x, beta):
    val, _ = smooth_obj_and_grad(x, beta)
    return val


def _slsqp_jac(x, beta):
    _, g = smooth_obj_and_grad(x, beta)
    return g


def _sum_fun(x, target):

def _gradient_check(beta, N):
    rng = np.random.default_rng(12345)
    h = rng.random(N) * 0.6 + 0.2
    h = repair_feasible(h, N)
    _, g_analytic = smooth_obj_and_grad(h, beta)

        h_plus = h.copy()
        h_plus[j] += eps
        h_plus = repair_feasible(h_plus, N)
        obj_plus, _ = smooth_obj_and_grad(h_plus, beta)
        h_minus = h.copy()
        h_minus[j] -= eps
        h_minus = repair_feasible(h_minus, N)
        obj_minus, _ = smooth_obj_and_grad(h_minus, beta)
        g_num[j] = (obj_plus - obj_minus) / (2 * eps)
    diff = np.max(np.abs(g_analytic[:min(N,10)] - g_num))
    scale = np.max(np.abs(g_analytic)) + 1e-10
    return diff / scale < 1e-4


def transfer_length(h, new_N):
    old_N = len(h)
    if old_N == new_N:
        return repair_feasible(h, new_N)
    if new_N % old_N == 0:
        factor = new_N // old_N
        h_new = np.repeat(h, factor)
        return repair_feasible(h_new, new_N)
    if old_N % new_N == 0:
        factor = old_N // new_N
        h_new = h.reshape(new_N, factor).mean(axis=1)
        return repair_feasible(h_new, new_N)
    src_centers = (np.arange(old_N) + 0.5) * 2.0 / old_N
    dst_centers = (np.arange(new_N) + 0.5) * 2.0 / new_N
    h_new = np.interp(dst_centers, src_centers, h, left=h[0], right=h[-1])
    return repair_feasible(h_new, new_N)


def perturb_feasible(h, rng, scale):
    v = rng.normal(0.0, scale, len(h))
    v -= v.mean()
    return repair_feasible(h + v, len(h))


def do_slsqp_pass(h_start, beta, N, maxiter, ftol):
    target = N / 2.0
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h_start, args=(beta,), method='SLSQP',
                       jac=_slsqp_jac, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return repair_feasible(res.x, N)
    except Exception:
        pass
    return None


def do_anneal(h_start, rng, max_iters, t0, t_min, scale, deadline_fn):
    N = len(h_start)
    h_cur = h_start.copy()
    c_cur = evaluate_exact(h_cur)
    c_best = c_cur
    h_best = h_cur.copy()
    for it in range(max_iters):
        if deadline_fn():
            break
        h_cand = perturb_feasible(h_cur, rng, scale)
        c_cand = evaluate_exact(h_cand)
        if c_cand <= c_cur:
            h_cur = h_cand
            c_cur = c_cand
            if c_cand < c_best:
                c_best = c_cand
                h_best = h_cand.copy()
        else:
            T = t_min + (t0 - t_min) * max(0.0, 1.0 - it / float(max_iters))
            T = max(T, 1e-8)
            if c_cand - c_cur < 1e-12:
                h_cur = h_cand
                c_cur = c_cand
                if c_cand < c_best:
                    c_best = c_cand
                    h_best = h_cand.copy()
            else:
                ratio = np.exp(-(c_cand - c_cur) / T)
                if rng.random() < ratio:
                    h_cur = h_cand
                    c_cur = c_cand
        if it % 100 == 99:
            h_cur = h_best.copy()
            c_cur = c_best
    return h_best, c_best


def try_binary_local(h_start, N, rng, max_swaps, deadline_fn):
    if N % 2 != 0:
        return None, None
    order = np.argsort(h_start)[::-1]
    h = np.zeros(N)
    h[order[:N // 2]] = 1.0
    c = evaluate_exact(h)
    improved = True
    swaps = 0
    while improved and swaps < max_swaps:
        improved = False
        if deadline_fn():
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for a in ones:
            if deadline_fn():
                break
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                c_new = evaluate_exact(h)
                if c_new < c - 1e-12:
                    c = c_new
                    improved = True
                    swaps += 1
                else:
                    h[a] = 1.0
                    h[b] = 0.0
    return h, c
