For a transfer of t from i to j, the correlation at lag k changes linearly:

And C₅(t) = max_k [c_k(0) + t * d_k] * dx

To find min_t C₅(t), I need to find the minimum of a convex piecewise-linear function.

The minimum occurs where the subgradient contains 0, i.e., where there exist k1, k2 with d_{k1} ≤ 0 ≤ d_{k2} and c_{k1}(0) + t*d_{k1} = c_{k2}(0) + t*d_{k2}.

2. Sort by d_k
3. For the minimum, we need the "upper envelope" of the lines c_k + t*d_k
4. The minimum of the upper envelope is where it reaches its lowest point

- We can use bisection on the derivative (subgradient)

The subgradient at t is {dx * d_k : k is argmax of c_k + t*d_k}.
At the minimum, 0 should be in the subgradient, meaning there's an argmax with d_k ≤ 0 and one with d_k ≥ 0 (or one with d_k = 0).

- If f is increasing at t=0 (all active slopes > 0), minimum is at t=0
- Otherwise, binary search

```python
def find_min_convex_pl(a, b, t_lo, t_hi, n_iter=50):
    """Find min of max_k (a[k] + b[k]*t) for t in [t_lo, t_hi].
    a, b are arrays of length m. Returns (t_opt, f_opt)."""
    def f(t):
        return np.max(a + b * t)
    
    def active_slopes(t):
        vals = a + b * t
        mx = np.max(vals)
        active = np.where(np.abs(vals - mx) < 1e-10)[0]
        return b[active], mx
    
    # Check endpoints
    f_lo = f(t_lo)
    f_hi = f(t_hi)
    
    if f_lo <= f_hi:
        # Check slope at t_lo
        slopes, _ = active_slopes(t_lo)
        if np.min(slopes) >= -1e-12:
            return t_lo, f_lo
    else:
        slopes, _ = active_slopes(t_hi)
        if np.max(slopes) <= 1e-12:
            return t_hi, f_hi
    
    # Binary search
    lo, hi = t_lo, t_hi
    for _ in range(n_iter):
        mid = (lo + hi) / 2
        slopes, f_mid = active_slopes(mid)
        if np.min(slopes) >= -1e-12:
            hi = mid
        elif np.max(slopes) <= 1e-12:
            lo = mid
        else:
            # 0 is in the subgradient, we're at the minimum
            return mid, f_mid
    
    t_opt = (lo + hi) / 2
    return t_opt, f(t_opt)
```

```python
def compute_gradients(h):
    n = len(h)
    n_corr = 2*n - 1
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For coordinate p and lag lags[k]:
    # First term: 1-h[p - lags[k]] if 0 <= p-lags[k] < n
    # Second term: -h[p + lags[k]] if 0 <= p+lags[k] < n
    
    p_idx = np.arange(n)[:, None]  # (n, 1)
    l = lags[None, :]  # (1, 2n-1)
    
    # p - lags[k]: range [p-(n-1), p+(n-1)]
    minus = p_idx - l  # (n, 2n-1)
    plus = p_idx + l   # (n, 2n-1)
    
    grad = np.zeros((n, n_corr))
    
    mask_minus = (minus >= 0) & (minus < n)
    mask_plus = (plus >= 0) & (plus < n)
    
    # First term: (1-h[minus]) where valid
    h_minus = np.where(mask_minus, h[np.clip(minus, 0, n-1)], 0.0)
    grad += (1.0 - h_minus) * mask_minus
    
    # Second term: -h[plus] where valid
    h_plus = np.where(mask_plus, h[np.clip(plus, 0, n-1)], 0.0)
    grad -= h_plus * mask_plus
    
    return grad
```

```python
import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    if s < target_sum:
        deficit = target_sum - s
        idx = np.argsort(h)
        for i in idx:
            d = min(1.0 - h[i], deficit)
            h[i] += d
            deficit -= d
            if deficit <= 1e-12:
                break
    else:
        excess = s - target_sum
        idx = np.argsort(-h)
        for i in idx:
            d = min(h[i], excess)
            h[i] -= d
            excess -= d
            if excess <= 1e-12:
                break
    return h

def compute_grad_matrix(h):
    n = len(h)
    n_corr = 2 * n - 1
    lags = np.arange(-(n-1), n)
    p_idx = np.arange(n)[:, None].astype(float)
    l = lags[None, :].astype(float)
    minus = p_idx - l
    plus = p_idx + l
    n_int = np.clip(minus.astype(int), 0, n-1)
    p_int = np.clip(plus.astype(int), 0, n-1)
    mask_m = (minus >= 0) & (minus < n)
    mask_p = (plus >= 0) & (plus < n)
    grad = np.zeros((n, n_corr))
    grad[mask_m] += (1.0 - h[n_int[mask_m]])
    grad[mask_p] -= h[p_int[mask_p]]
    return grad

def find_min_1d_convex(a, b, t_lo, t_hi, max_iter=60):
    m = len(a)
    def f(t):
        return np.max(a + b * t)
    
    f_lo = f(t_lo)
    f_hi = f(t_hi)
    
    if f_lo <= f_hi:
        vals = a + b * t_lo
        mx = np.max(vals)
        active = b[np.abs(vals - mx) < 1e-10]
        if len(active) > 0 and np.min(active) >= -1e-12:
            return t_lo, f_lo
    else:
        vals = a + b * t_hi
        mx = np.max(vals)
        active = b[np.abs(vals - mx) < 1e-10]
        if len(active) > 0 and np.max(active) <= 1e-12:
            return t_hi, f_hi
    
    lo, hi = t_lo, t_hi
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        vals = a + b * mid
        mx = np.max(vals)
        active = b[np.abs(vals - mx) < 1e-10]
        if len(active) == 0:
            hi = mid
        elif np.min(active) >= -1e-12:
            hi = mid
        elif np.max(active) <= 1e-12:
            lo = mid
        else:
            return mid, mx
        if hi - lo < 1e-12:
            break
    t_opt = (lo + hi) * 0.5
    return t_opt, f(t_opt)

def local_search(h, dx, target_sum, max_passes=200, time_limit_s=None, t0=None, tol=1e-10):

        grad = compute_grad_matrix(h)
        corr = np.correlate(h, 1-h, mode="full")

        perm = np.random.permutation(n)
        
        for ii in range(n):
            i = perm[ii]
```

```python
            if time.time() >= deadline - 5:
                break
            h_pert = best_h + 0.05 * np.random.normal(0, 1, len(best_h))
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            dx_p = 2.0 / len(best_h)
            h_pert, _ = local_search_fast(h_pert, dx_p, len(best_h) / 2.0, t0, time.time() + 15.0, max_passes=50)
            try_candidate(h_pert, len(best_h))

    return best_h, best_c5, best_n
```
