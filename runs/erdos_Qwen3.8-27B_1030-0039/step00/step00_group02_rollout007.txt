    def sum_clipped(tau):

        else:
            hi = mid

    max_val = np.max(vals)

def pgd_optimize(h0, target_sum, beta, max_iter=5000, time_limit=60.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    step = 0.5
    n = len(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Softmax gradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = beta * corr * (2.0 / n)
        max_v = np.max(vals)
        exp_v = np.exp(vals - max_v)
        weights = exp_v / np.sum(exp_v)
        
        # Compute gradient
        dx = 2.0 / n
        grad = np.zeros(n)
        arange = np.arange(n)
        for m in range(2 * n - 1):
            d = m - n + 1
            w = weights[m] * dx
            if w < 1e-14:
                continue
            idx1 = arange - d
            v1 = (idx1 >= 0) & (idx1 < n)
            grad[v1] += w * (1.0 - h[idx1[v1]])
            idx2 = arange + d
            v2 = (idx2 >= 0) & (idx2 < n)
            grad[v2] -= w * h[idx2[v2]]
        
        # Backtracking line search
        f_current = compute_c5(h)
        alpha = step
        improved = False
        for _ in range(30):
            h_new = project_box_sum(h - alpha * grad, target_sum)
            f_new = compute_c5(h_new)
            if f_new < f_current - 1e-14:
                h = h_new
                improved = True
                if f_new < best_c5:
                    best_c5 = f_new
                    best_h = h.copy()
                break
            alpha *= 0.5
        
        if not improved:
            step = max(step * 0.5, 1e-6)
        
        # Occasionally boost step
        if it % 200 == 0 and step < 0.1:
            step = 0.3
    
    return best_h
