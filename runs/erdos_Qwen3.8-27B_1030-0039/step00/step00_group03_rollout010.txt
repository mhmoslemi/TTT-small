import numpy as np
from scipy.optimize import minimize, brentq
import time

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def project_feasible(h, n):
    """Project h to feasible set: 0 <= h[i] <= 1, sum(h) = n/2"""
    target = n / 2.0
    # Use bisection on offset t: find t such that sum(clip(h + t, 0, 1)) = target
    def f(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    lo = -1.0 - np.max(h)
    hi = 1.0 - np.min(h)
    if f(lo) > 0:
        return np.clip(h + lo, 0.0, 1.0), False
    if f(hi) < 0:
        return np.clip(h + hi, 0.0, 1.0), False
    try:
        t = brentq(f, lo, hi, xtol=1e-14)
    except Exception:
        t = brentq(f, lo, hi, xtol=1e-12)
    result = np.clip(h + t, 0.0, 1.0)
    # Fix tiny numerical errors in sum
    diff = target - np.sum(result)
    if abs(diff) < 1e-10:
        # Distribute tiny correction
        mask = (result > 1e-10) & (result < 1 - 1e-10)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result, True

def smooth_objective_and_grad(h, beta):
    """Compute log-sum-exp smooth objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    m = np.max(y)
    w = np.exp(y - m)
    Z = np.sum(w)
    psi = (m + np.log(Z)) / beta
    p = w / Z  # softmax weights
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        weight = p[k] * dx
        if weight < 1e-15:
            continue
        if lag >= 0:
            # f_k = dx * sum_{r=0}^{n-1-lag} h[r+lag] * (1-h[r])

            # d/dh[r] for r in lag..n-1: -h[r-lag]

                grad[:upper] += weight * (1.0 - h[lag:n])

            # Reindex: let r = n-p, so n = r+p, r ranges 0 to n-1-p
            # f_k = dx * sum_{r=0}^{n-1-p} h[r] * (1-h[r+p])
            # d/dh[r] for r in 0..n-1-p: (1-h[r+p])
            # d/dh[r] for r in p..n-1: -h[r-p]
            if upper > 0:
                grad[:upper] += weight * (1.0 - h[p_neg:n])
            if p_neg < n:
                grad[p_neg:] += weight * (-h[:upper])
    
    return psi, grad

def transfer_resolution(h_old, n_new):
    """Transfer h from old resolution to new resolution via interpolation + projection."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new, valid = project_feasible(h_new, n_new)
    return h_new

def coordinate_descent(h, max_rounds=200, tol=1e-12):
    """Pair exchange coordinate descent. Preserves sum exactly."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_score = evaluate_exact(h)
    
    for round_idx in range(max_rounds):
        improved = False
        # Try all pairs (or a subset for large n)
        if n <= 60:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        else:
            # Random subset
            rng = np.random.RandomState(round_idx)
            pairs = []
            for _ in range(3 * n * n // 4):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    pairs.append((min(i,j), max(i,j)))
            pairs = list(set(pairs))
        
        for i, j in pairs:
            # h[i] + delta, h[j] - delta
            # Constraints: 0 <= h[i]+delta <= 1, 0 <= h[j]-delta <= 1
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo >= hi:
                continue
            # Sample several delta values
            deltas = np.linspace(lo, hi, 9)
            best_d = 0.0
            best_s = best_score
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                h[i] += d
                h[j] -= d
                s = evaluate_exact(h)
                if s < best_s - tol:
                    best_s = s
                    best_d = d
            # Reset
            h[i] -= best_d
            h[j] += best_d
            # Apply best
            if best_d != 0.0:
                h[i] += best_d
                h[j] -= best_d
                best_score = best_s
                improved = True
        
        if not improved:
            break
    
    return h, best_score

import numpy as np
from scipy.optimize import minimize, brentq
import time


def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    if n == 1:
        val = min(max(target, 0.0), 1.0)
        return np.array([val]), True
    def f_offset(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    lo = -1.0 - np.max(h)
    hi = 1.0 - np.min(h)
    flo = f_offset(lo)
    fhi = f_offset(hi)
    if flo > 0:
        return np.clip(h + lo, 0.0, 1.0), True
    if fhi < 0:
        return np.clip(h + hi, 0.0, 1.0), True
    try:
        t = brentq(f_offset, lo, hi, xtol=1e-15, rtol=1e-15, maxiter=200)
    except Exception:
        t = brentq(f_offset, lo, hi, xtol=1e-13, maxiter=200)
    result = np.clip(h + t, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 0:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0 and abs(diff) < cnt * 1e-10:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result, True


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    y = beta * f
    m = np.max(y)
    w = np.exp(y - m)
    Z = np.sum(w)
    psi = (m + np.log(Z)) / beta
    p = w / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        weight = p[k] * dx
        if weight < 1e-16:
            continue
        if lag >= 0:
            upper = n - lag

            if upper > 0:
                grad[:upper] += weight * (1.0 - h[pn:n])
            if pn <= n - 1:
                grad[pn:] += weight * (-h[:upper])

def transfer_to_resolution(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new, _ = project_feasible(h_new, n_new)
    return h_new


def coordinate_descent(h, n_rounds=200, n_samples=13, tol=1e-13):
    n = len(h)
    h = h.copy()
    if n < 2:
        return h, evaluate_exact(h)
    best_score = evaluate_exact(h)
    for rnd in range(n_rounds):
        improved_any = False
        if n <= 70:
            indices = list(range(n))
        else:
            rng_local = np.random.RandomState(rnd * 7 + 3)
            indices = rng_local.permutation(n).tolist()
        for ii in range(len(indices)):
            i = indices[ii]
            for jj in range(ii + 1, len(indices)):
                j = indices[jj]
                lo = max(-h[i], h[j] - 1.0)

                if hi - lo < 1e-14:

                deltas = np.linspace(lo, hi, n_samples)

                    h[i] += best_d
                    h[j] -= best_d

        h[i] -= q1
        h[j] += q1
        h[i] += q2
        h[j] -= q2
        s2 = evaluate_exact(h)
        h[i] -= q2
        h[j] += q2
        if s1 < s2:
            hi = q2
        else:
            lo = q1
        if hi - lo < 1e-13:
            break
    if best_d != 0.0:
        h[i] += best_d
        h[j] -= best_d
    return h, best_s


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 8.0
    rng = np.random.RandomState(seed)

    def time_left():
        return deadline - time.time()

    try:
        h_init = np.array(initial_h_values, dtype=np.float64).flatten().copy()
        n_init = len(h_init)
    except Exception:
        n_init = 45
        h_init = np.full(n_init, 0.5)

    h_init, _ = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_score = evaluate_exact(h_init)
    best_n = n_init

    candidates = []

    def add_candidate(hc, nc):
        hc, v = project_feasible(hc, nc)
        if not v or not np.all(np.isfinite(hc)):
            return
        s = evaluate_exact(hc)
        candidates.append((s, hc, nc))
        return s

    # Initial candidates
    add_candidate(h_init, n_init)
    h_refl = 1.0 - h_init[::-1].copy()
    add_candidate(h_refl, n_init)
    for sig in [0.02, 0.05, 0.1, 0.2]:
        add_candidate(h_init + rng.randn(n_init) * sig, n_init)
    h_wave = 0.5 + 0.3 * np.sin(np.linspace(0, 4 * np.pi, n_init))
    add_candidate(h_wave, n_init)
    h_wave2 = 0.5 + 0.4 * np.cos(np.linspace(0, 6 * np.pi, n_init))
    add_candidate(h_wave2, n_init)

    resolutions = sorted(set([n_init, int(1.3 * n_init), int(1.6 * n_init), int(2.0 * n_init)]))
    resolutions = [r for r in resolutions if 10 <= r <= 220]
    if n_init not in resolutions:
        resolutions.insert(0, n_init)

    for nr in resolutions:
        if nr != n_init:
            h_tr = transfer_to_resolution(h_init, nr)
            add_candidate(h_tr, nr)

    candidates.sort(key=lambda x: x[0])
    if len(candidates) > 12:
        candidates = candidates[:12]

    beta_schedule = [6.0, 20.0, 60.0, 150.0, 400.0, 1000.0]
    processed_count = 0
    max_candidates = 30

    while time_left() > 40 and processed_count < max_candidates:
        if not candidates:
            h_pert = best_h + rng.randn(best_n) * 0.03
            add_candidate(h_pert, best_n)
            if not candidates:
                break
            candidates.sort(key=lambda x: x[0])

        score_c, h_c, n_c = candidates.pop(0)
        processed_count += 1

        if score_c > best_score + 0.05:
            if len(candidates) < 10:
                candidates.append((score_c, h_c, n_c))
            continue

        h_work = h_c.copy()
        h_work, v = project_feasible(h_work, n_c)
        if not v:
            continue
        cur_score = evaluate_exact(h_work)

        n = n_c
        maxiter_base = 300 if n <= 60 else (200 if n <= 120 else 150)

        for bi, beta in enumerate(beta_schedule):
            if time_left() < 25:
                break

            def obj_func(x):
                val, _ = smooth_obj_grad(x, beta)
                return val

            def grad_func(x):
                _, g = smooth_obj_grad(x, beta)
                return g

            try:
                bounds = [(0.0, 1.0)] * n
                cons_eq = [{'type': 'eq',
                            'fun': lambda x: np.sum(x) - n / 2.0,
                            'jac': lambda x: np.ones(n)}]
                mi = maxiter_base
                res = minimize(obj_func, h_work, jac=grad_func, method='SLSQP',
                               bounds=bounds, constraints=cons_eq,
                               options={'maxiter': mi, 'ftol': 1e-14, 'disp': False})
                if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = res.x
                    h_new, v2 = project_feasible(h_new, n)
                    if v2:
                        s_new = evaluate_exact(h_new)
                        if s_new < cur_score - 1e-13:
                            cur_score = s_new
                            h_work = h_new
                        else:
                            h_work = h_new
            except Exception:
                continue

        if time_left() > 30:
            h_cd, s_cd = coordinate_descent(h_work, n_rounds=80, n_samples=11, tol=1e-13)
            if s_cd < cur_score - 1e-13:
                cur_score = s_cd
                h_work = h_cd

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_h = h_work.copy()
            best_n = n_c

        if time_left() > 60:
            add_candidate(h_work + rng.randn(n_c) * 0.02, n_c)
            add_candidate(h_work + rng.randn(n_c) * 0.06, n_c)
            for nr in resolutions:
                if nr != n_c:
                    h_tr = transfer_to_resolution(h_work, nr)
                    add_candidate(h_tr, nr)
            candidates.sort(key=lambda x: x[0])
            if len(candidates) > 15:
                candidates = [candidates[0]] + candidates[1:15]

    # Final intensive refinement
    if time_left() > 15:
        h_fin, s_fin = coordinate_descent(best_h, n_rounds=300, n_samples=15, tol=1e-14)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    if time_left() > 10:
        n_try = best_n
        h_fin2, s_fin2 = coordinate_descent(best_h, n_rounds=200, n_samples=17, tol=1e-14)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2

    # Final validation
    best_h, v = project_feasible(best_h, best_n)
    if not v:
        best_h = np.full(best_n, 0.5)
        best_score = evaluate_exact(best_h)
    else:
        final_score = evaluate_exact(best_h)
        if final_score < best_score + 1e-12:
            best_score = final_score

    return best_h, best_score, best_n
