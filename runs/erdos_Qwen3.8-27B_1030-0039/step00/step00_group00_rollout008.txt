```python
import numpy as np
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, N):
    target = N / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, N):
    n_corr = 2 * N - 1
    G = np.zeros((n_corr, N))
    k_arr = np.arange(-(N - 1), N)
    for j in range(N):
        for m in range(n_corr):
            k = k_arr[m]
            n_lo = max(0, -k)
            n_hi = min(N - 1, N - 1 - k)
            val = 0.0
            n1 = j - k
            if n_lo <= n1 <= n_hi:
                val += 1.0 - h[n1]
            if n_lo <= j <= n_hi:
                jk = j + k
                if 0 <= jk < N:
                    val -= h[jk]
            G[m, j] = val
    return G

def pair_swap_sweep(h, N, n_candidates=50):
    c = np.correlate(h, 1 - h, mode="full")
    G = compute_grad_matrix(h, N)
    current_max = float(np.max(c))
    
    for j in range(N):
        h_j = h[j]
        Gj = G[:, j]
        best_delta = 0.0
        best_l = -1
        best_val = current_max
        
        for l in range(N):
            if l == j:
                continue
            h_l = h[l]
            d_lo = max(-h_j, h_l - 1.0)
            d_hi = min(1.0 - h_j, h_l)
            if d_lo >= d_hi - 1e-12:
                continue
            g = Gj - G[:, l]
            # Quick check: if all g are the same sign, the function is monotone
            # and the minimum is at an endpoint
            candidates = np.linspace(d_lo, d_hi, n_candidates)
            new_vals = np.max(c[None, :] + candidates[:, None] * g[None, :], axis=1)
            idx = int(np.argmin(new_vals))
            if new_vals[idx] < best_val - 1e-14:
                best_val = float(new_vals[idx])
                best_delta = float(candidates[idx])
                best_l = l
        
        if best_l >= 0 and abs(best_delta) > 1e-14:
            h[j] += best_delta
            h[best_l] -= best_delta
            h = np.clip(h, 0.0, 1.0)
            c = np.correlate(h, 1 - h, mode="full")
            current_max = float(np.max(c))
    
    return h, current_max

def smooth_grad_step(h, N, beta=50.0, lr=0.05):
    dx = 2.0 / N
    c = np.correlate(h, 1 - h, mode="full")
    scaled = beta * c
    max_s = float(np.max(scaled))
    w = np.exp(scaled - max_s)
    w /= float(np.sum(w))
    G = compute_grad_matrix(h, N)
    grad = (G.T @ w) * dx
    grad -= float(np.mean(grad))
    h_new = h - lr * grad
    return project_feasible(h_new, N)

def upsample(h, N_new):
    N_old = len(h)
    if N_new == N_old:
        return h.copy()
    old_x = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    new_x = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    return np.interp(new_x, old_x, h)
```
