            #      grad[s:n] += w * (-h[0:n-s])

        else:

            grad[sp:] += w * (1.0 - h[:n-sp])
            grad[:n-sp] -= w * h[sp:]

import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project(v, c):
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > c:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def get_weights(corr, t):
    x = t * corr
    x = x - np.max(x)
    x = np.clip(x, -500.0, 500.0)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(corr) / len(corr)
    return e / s


def compute_grad(h, dx, t):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    w = get_weights(corr, t)
    g = np.zeros(n)
    nm1 = n - 1
    for j in range(2 * n - 1):
        wi = w[j]
        if wi < 1e-14:
            continue
        s = j - nm1
        if s >= 0:
            m = n - s
            g[:m] += wi * omh[s:]
            g[s:] -= wi * h[:m]
        else:
            sp = -s
            m = n - sp
            g[sp:] += wi * omh[:m]
            g[:m] -= wi * h[sp:]
    return g


def gd_phase(h, dx, c, t_list, lr0, t_budget):
    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for t in t_list:
        if t_budget - (time.time() - t0) < 20:
            break
        lr = lr0
        for _ in range(10):
            if t_budget - (time.time() - t0) < 10:
                break
            for _ in range(200):
                g = compute_grad(h, dx, t)
                h = project(h - lr * g, c)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= 0.6
    return best_h, best_c5


def ls_continuous(h, dx, rng, t_budget):
    t0 = time.time()
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h, dx)
    while t_budget - (time.time() - t0) > 5:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-1.0, 1.0]) * rng.uniform(0.005, 0.05)
        oi, oj = h[i], h[j]
        h[i] = oi + d
        h[j] = oj - d
        if 0.0 <= h[i] and h[i] <= 1.0 and 0.0 <= h[j] and h[j] <= 1.0:
            c5n = compute_c5(h, dx)
            if c5n < c5:
                c5 = c5n
            else:
                h[i], h[j] = oi, oj
        else:
            h[i], h[j] = oi, oj
    return h, c5


def ls_binary(h, dx, rng, t_budget):
    t0 = time.time()
    n = len(h)
    c = int(np.round(np.sum(h)))
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) < 1 or len(zeros) < 1:
        return h, compute_c5(h, dx)
    c5 = compute_c5(h, dx)
    while t_budget - (time.time() - t0) > 5:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c5n = compute_c5(h, dx)
        if c5n < c5:
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            c5 = c5n
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, c5


def to_binary(h, c):

    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    k = int(round(c))
    hb[idx[:k]] = 1.0
    return hb

    def rem():

            best_h = h4.copy()
        else:
            break

    return best_h, best_c5, len(best_h)

import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project(v, c):
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > c:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def get_weights(corr, t):
    x = t * corr
    x = x - np.max(x)
    x = np.clip(x, -500.0, 500.0)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(corr) / len(corr)
    return e / s


def compute_grad(h, dx, t):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    w = get_weights(corr, t)
    g = np.zeros(n)
    nm1 = n - 1
    for j in range(2 * n - 1):
        wi = w[j]
        if wi < 1e-14:
            continue
        s = j - nm1
        if s >= 0:
            m = n - s
            g[:m] += wi * omh[s:]
            g[s:] -= wi * h[:m]
        else:
            sp = -s
            m = n - sp
            g[sp:] += wi * omh[:m]
            g[:m] -= wi * h[sp:]
    return g


def gd_phase(h, dx, c, t_list, lr0, t_start, budget_s):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for t in t_list:
        if budget_s - (time.time() - t_start) < 20:
            break
        lr = lr0
        for _ in range(10):
            if budget_s - (time.time() - t_start) < 10:
                break
            for _ in range(200):
                g = compute_grad(h, dx, t)
                h = project(h - lr * g, c)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= 0.6
    return best_h, best_c5


def ls_continuous(h, dx, rng, t_start, budget_s):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h, dx)
    while budget_s - (time.time() - t_start) > 5:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-1.0, 1.0]) * rng.uniform(0.005, 0.05)
        oi, oj = h[i], h[j]
        h[i] = oi + d
        h[j] = oj - d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            c5n = compute_c5(h, dx)
            if c5n < c5:
                c5 = c5n
            else:
                h[i], h[j] = oi, oj
        else:
            h[i], h[j] = oi, oj
    return h, c5


def to_binary(h, c):
    n = len(h)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    k = int(round(c))
    hb[idx[:k]] = 1.0
    return hb


def ls_binary(h, dx, rng, t_start, budget_s):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h, dx)
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    while budget_s - (time.time() - t_start) > 5:

            zeros = np.where(h < 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, c5
