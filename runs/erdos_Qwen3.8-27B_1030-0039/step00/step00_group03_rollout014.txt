Actually numpy defines it as:

Let me just use the standard definition. The key point is: when we change h_i and h_j, each term in the correlation that involves either h_i or h_j changes. The correlation C(h, 1-h) has terms of the form h_p * (1 - h_q) for various (p, q) pairs depending on the lag.

When we do h_i += t and h_j -= t:
- Any term h_p * (1-h_q) where p=i or q=i gets contribution +t or -t (linear in t)
- Any term where p=j or q=j gets -t or +t (linear in t)
- Any term where both p and j are involved: h_j * (1-h_i) or h_i * (1-h_j) -> these get quadratic terms

Specifically:
- h_i(t) * (1-h_j(t)) = (h_i+t)(1-h_j+t) = h_i(1-h_j) + t(1-h_j+h_i) + t^2
- h_j(t) * (1-h_i(t)) = (h_j-t)(1-h_i-t) = h_j(1-h_i) - t(1-h_i+h_j) + t^2

So each such term gains +t^2, making it convex. Terms with only one endpoint changing are linear. So the max of all these is convex. Good.

    raw = np.asarray(raw, dtype=np.float64)
    n = len(raw)
    S = n / 2.0

        return None

        return np.sum(np.clip(raw + tau, 0.0, 1.0)) - S
    lo = -np.max(raw) - 1.0
    hi = 1.0 - np.min(raw) + 1.0
    Flo = F(lo)
    Fhi = F(hi)
    if Flo > 0 or Fhi < 0:
        return None
    try:
        tau = brentq(F, lo, hi, xtol=1e-14, rtol=1e-14, maxiter=200)
    except Exception:
        return None
    z = np.clip(raw + tau, 0.0, 1.0)
    if abs(np.sum(z) - S) > 1e-8:
        return None
    return z

def golden_minimize(f, a, b, max_iter=40, tol=1e-10):
    """Minimize convex univariate f on [a,b] using golden section."""
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)
    return (a + b) / 2.0

    """Coordinate descent using exact pair-exchange line minimization."""
    h = h.copy()
    n = len(h)
    score = exact_score(h)

        if time_limit is not None and start_time is not None:

                break

                L = max(-h[i], h[j] - 1.0)
                U = min(1.0 - h[i], h[j])

                    continue

                    return pair_line_score(h, i, j, t, n)
                t_opt = golden_minimize(f_t, L, U, max_iter=30, tol=1e-12)
                # Clip to feasible
                t_opt = max(L + 1e-15, min(U - 1e-15, t_opt))
                new_score = pair_line_score(h, i, j, t_opt, n)
                if new_score < score - improve_tol:
                    h[i] += t_opt
                    h[j] -= t_opt
                    score = new_score

                    no_improve = 0
                else:

        if not improved:

            score = new_score

import numpy as np
from scipy.optimize import brentq
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_feasible(raw):
    raw = np.asarray(raw, dtype=np.float64)
    n = len(raw)
    S = n / 2.0
    if np.any(~np.isfinite(raw)):
        return None
    def F(tau):
        return np.sum(np.clip(raw + tau, 0.0, 1.0)) - S
    lo = -np.max(raw) - 1.0
    hi = 1.0 - np.min(raw) + 1.0
    if F(lo) > 0 or F(hi) < 0:
        return None
    try:
        tau = brentq(F, lo, hi, xtol=1e-14, rtol=1e-14, maxiter=200)
    except Exception:
        return None
    z = np.clip(raw + tau, 0.0, 1.0)
    if abs(np.sum(z) - S) > 1e-8:
        return None
    return z

def golden_minimize(f, a, b, max_iter=35, tol=1e-12):
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)
    return (a + b) / 2.0

    hn = h.copy()
    hn[i] += t
    hn[j] -= t
    dx = 2.0 / n
    corr = np.correlate(hn, 1.0 - hn, mode="full")

    n = len(h)
    score = exact_score(h)

                break

                    return pair_line_eval(h, i, j, t, n)

                t_opt = max(L + 1e-12, min(U - 1e-12, t_opt))

                    h[j] -= t_opt

        amp = rng.uniform(0.05, 0.35)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        val += amp * np.sin(2.0 * np.pi * k * x / n + phase)
    val += 0.5
    return repair_feasible(val)

def make_binary_seed(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    else:
        k = (n - 1) // 2
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
        remaining = np.setdiff1d(np.arange(n), idx)
        h[int(remaining[0])] = 1.0
        h[int(remaining[1])] = 0.0  # will be repaired
    return repair_feasible(h)

def transfer_to_size(h_src, n_tgt):
    n_src = len(h_src)
    if n_src == n_tgt:
        return h_src.copy()
    src_pos = np.linspace(0.0, 1.0, n_src)
    tgt_pos = np.linspace(0.0, 1.0, n_tgt)
    h_interp = np.interp(tgt_pos, src_pos, h_src)
    return repair_feasible(h_interp)
