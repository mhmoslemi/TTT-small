def project_to_feasible(v, N):
    """Project v onto {h: 0<=h<=1, sum(h)=N/2}"""
    v = np.asarray(v, dtype=np.float64)
    # Replace non-finite
    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, 0.0, 1.0)
    target = N / 2.0
    # Bisection for t
    # sum(clip(v - t, 0, 1)) is non-increasing in t
    lo = -2.0
    hi = 2.0
    # Ensure bracket
    while sum(np.clip(v - lo, 0, 1)) < target:
        lo -= 1.0
    while sum(np.clip(v - hi, 0, 1)) > target:
        hi += 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    t = 0.5 * (lo + hi)
    h = np.clip(v - t, 0.0, 1.0)
    # Correct sum
    d = target - sum(h)
    if abs(d) > 1e-13:

import numpy as np
import time
from scipy.optimize import minimize


def get_parent():
    try:
        vals = np.asarray(initial_h_values, dtype=np.float64)
        if vals.ndim == 1 and len(vals) > 0 and np.all(np.isfinite(vals)):
            return vals.copy()
    except Exception:
        pass
    return None


def project_to_feasible(v, N):
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, 0.0, 1.0)
    target = N / 2.0
    s = sum(v)
    if abs(s - target) < 1e-14:
        return v
    lo = -2.0
    hi = 2.0
    while sum(np.clip(v - lo, 0.0, 1.0)) < target:
        lo -= 1.0
    while sum(np.clip(v - hi, 0.0, 1.0)) > target:
        hi += 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        sm = sum(np.clip(v - mid, 0.0, 1.0))
        if sm > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    t = 0.5 * (lo + hi)
    h = np.clip(v - t, 0.0, 1.0)
    d = target - sum(h)
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            step = d / len(idx)
            h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
            d2 = target - sum(h)
            if abs(d2) > 1e-14:
                if d2 > 0:
                    idx2 = np.where(h < 1.0 - 1e-10)[0]
                else:
                    idx2 = np.where(h > 1e-10)[0]
                if len(idx2) > 0:
                    h[idx2[0]] += d2
    return h


def evaluate_exact(h):
    N = len(h)
    if N == 0:
        return 1e10
    dx = 2.0 / N
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return 1e10
    target = N / 2.0
    if abs(float(np.sum(h)) - target) > 1e-7:
        return 1e10
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = float(np.max(c))
    if not np.isfinite(val):
        return 1e10
    return val


def smooth_surrogate(x, beta):
    N = len(x)
    dx = 2.0 / N
    fvals = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(fvals)
    if not np.isfinite(m):
        return 1e10
    shifted = beta * (fvals - m)
    e = np.exp(shifted)
    s = np.sum(e)
    if s <= 0 or not np.isfinite(s):
        return m
    return m + np.log(s) / beta


def smooth_grad(x, beta):
    N = len(x)
    dx = 2.0 / N
    fvals = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(fvals)
    if not np.isfinite(m):
        return np.zeros(N)
    shifted = beta * (fvals - m)
    e = np.exp(shifted)
    s = np.sum(e)
    if s <= 0 or not np.isfinite(s):
        w = np.ones(2 * N - 1) / (2 * N - 1)
    else:
        w = e / s
    grad = np.zeros(N)
    # Center term (index N-1)
    wc = w[N - 1]
    grad += wc * dx * (1.0 - 2.0 * x)
    for s in range(1, N):
        ws = w[N - 1 + s]
        wm = w[N - 1 - s]
        if ws > 1e-15:
            # c[N-1+s] = dx * sum_{i=0}^{N-1-s} h[i+s]*(1-h[i])
            # grad[j] for j=s..N-1: +dx*(1-h[j-s])
            # grad[j] for j=0..N-1-s: -dx*h[j+s]
            grad[s:] += ws * dx * (1.0 - x[:N - s])
            grad[:N - s] += ws * (-dx) * x[s:]
        if wm > 1e-15:
            # c[N-1-s] = dx * sum_{i=0}^{N-1-s} h[i]*(1-h[i+s])
            # grad[j] for j=0..N-1-s: +dx*(1-h[j+s])
            # grad[j] for j=s..N-1: -dx*h[j-s]
            grad[:N - s] += wm * dx * (1.0 - x[s:])
            grad[s:] += wm * (-dx) * x[:N - s]
    return grad


def sum_eq_fun(x, target):
    return float(np.sum(x)) - target


def sum_eq_jac(x, target):
    return np.ones(len(x))


def run_slsqp_pass(h0, beta, N, target, maxiter, ftol=1e-12):
    bounds = [(0.0, 1.0)] * N
    constraints = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        sol = minimize(
            smooth_surrogate, h0, args=(beta,),
            method='SLSQP', jac=smooth_grad,
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        return sol.x
    except Exception:
        return h0


def transfer_length(h_src, N_dst):
    N_src = len(h_src)
    if N_src == N_dst:
        return h_src.copy()
    # Linear interpolation on bin centers
    src_centers = (np.arange(N_src) + 0.5) * (2.0 / N_src)
    dst_centers = (np.arange(N_dst) + 0.5) * (2.0 / N_dst)
    h_interp = np.interp(dst_centers, src_centers, h_src)
    return h_interp


def random_perturb(h, rng, scale):

    return v


def make_random_feasible(N, rng):

    # Subtract mean to center around 0.5
    v = v - np.mean(v) + 0.5

def make_binary(h, N):
    """Quantize h to binary with exactly N/2 ones."""
    if N % 2 != 0:
        return None
    threshold_idx = N // 2
    sorted_idx = np.argsort(h)[::-1][:threshold_idx]
    h_bin = np.zeros(N)
    h_bin[sorted_idx] = 1.0
    return h_bin


def binary_local_search(h_bin, max_iters, rng, start_time, budget):
    N = len(h_bin)
    dx = 2.0 / N
    h = h_bin.copy()
    cur_c = evaluate_exact(h)
    best_h = h.copy()
    best_c = cur_c
    improved = True
    iters = 0
    while improved and iters < max_iters:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Shuffle for diversity
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for a in ones:
            for b in zeros:
                if h[a] != 1.0 or h[b] != 0.0:
                    continue
                h[a] = 0.0
                h[b] = 1.0
                c_new = evaluate_exact(h)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    cur_c = c_new
                    improved = True
                    iters += 1
                    # Check time
                    if time.perf_counter() - start_time > budget * 0.85:
                        return best_h, best_c
                    break
                else:
                    h[a] = 1.0
                    h[b] = 0.0
                iters += 1
                if time.perf_counter() - start_time > budget * 0.85:
                    return best_h, best_c
    return best_h, best_c


def simulated_anneal(h0, rng, N, start_time, time_limit, T0=0.0005, Tmin=0.00002, cooling=0.998):
    h = h0.copy()
    cur_c = evaluate_exact(h)
    best_h = h.copy()
    best_c = cur_c
    T = T0
    total_moves = max(100, int(time_limit * 300))
    for it in range(total_moves):
        if time.perf_counter() - start_time >= time_limit:
            break
        if T < Tmin:
            break
        # Generate perturbation
        scale = 0.05 + 0.1 * T / T0
        v = h + scale * rng.standard_normal(N)
        h_new = project_to_feasible(v, N)
        c_new = evaluate_exact(h_new)
        if c_new <= cur_c or rng.random() < np.exp(-(c_new - cur_c) / max(T, 1e-10)):
            h = h_new
            cur_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h.copy()
        T *= cooling
    return best_h, best_c
