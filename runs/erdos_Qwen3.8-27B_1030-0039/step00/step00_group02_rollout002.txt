    n = len(x)
    lo, hi = x.min() - 1.0, x.max() + 1.0

        if s > S:

        else:
            hi = mid
    lam = (lo + hi) / 2.0

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def compute_gradient(h, weights):
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        for k in range(2 * n - 1):
            w = weights[k]
