import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def compute_F(h):
    """F[i, m] = contribution coefficient for changing h[i] at correlation lag m"""
    n = len(h)
    F = np.zeros((n, 2 * n - 1))
    omh = 1.0 - h
    for i in range(n):
        for m in range(2 * n - 1):
            s = m - (n - 1)
            if 0 <= i - s < n:
                F[i, m] += omh[i - s]
            if 0 <= i + s < n:
                F[i, m] -= h[i + s]
    return F

def cde_step(h, corr_raw, F, i, j, grid_size=60):
    """Try transferring from j to i, return (best_t, new_corr_raw) or (0, None) if no improvement."""
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, None
    D = F[i] - F[j]
    cur_max = float(np.max(corr_raw))
    ts = np.linspace(lo, hi, grid_size)
    # vectorized: new_maxes[k] = max(corr_raw + ts[k]*D)
    new_maxes = np.max(corr_raw[None, :] + ts[:, None] * D[None, :], axis=1)
    idx = int(np.argmin(new_maxes))
    if new_maxes[idx] < cur_max - 1e-14:
        return ts[idx], corr_raw + ts[idx] * D
    return 0.0, None

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def compute_F(h):
    n = len(h)
    F = np.zeros((n, 2 * n - 1))
    omh = 1.0 - h
    for i in range(n):
        row = F[i]
        for m in range(2 * n - 1):
            s = m - (n - 1)
            if 0 <= i - s < n:
                row[m] += omh[i - s]
            if 0 <= i + s < n:
                row[m] -= h[i + s]
    return F

def make_feasible(h, n):
    h = h.astype(float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        target = 0.5 * n
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
    return np.clip(h, 0.0, 1.0)
