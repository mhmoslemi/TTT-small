from scipy.optimize import minimize_scalar
import time

def project_feasible(y, N):
    """Project y onto {h in [0,1]^N : sum(h) = N/2} using bisection."""
    target = N / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0

        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx

def get_active(c, n_active=10, tol=1e-6):
    """Get indices of active (near-max) correlation values."""
    max_val = c.max()
    threshold = max_val - tol
    active = np.where(c >= threshold)[0]
    if len(active) > n_active:
        # Take the top n_active
        top_vals = np.argsort(c)[-n_active:]
        active = top_vals
    return active, max_val

def compute_descent_direction(h, active, alpha=1.0):
    """Compute descent direction from active set gradients."""
    N = len(h)
    grad = np.zeros(N)
    for m in active:
        lag = m - (N - 1)
        n_start = max(0, lag)
        n_end = min(N, N + lag)
        # First term: for i in [n_start, n_end), grad[i] += (1 - h[i-lag])
        for i in range(n_start, n_end):
            grad[i] += (1.0 - h[i - lag])
        # Second term: for i where i+lag in [n_start, n_end), grad[i] -= h[i+lag]

        i_end = min(N, n_end - lag)
        for i in range(i_start, i_end):
            grad[i] -= h[i + lag]
    return -grad / len(active)

def compute_descent_direction(h, active, N):
    grad = np.zeros(N)
    for m in active:
        lag = m - (N - 1)
        # First term: for i in valid range, grad[i] += (1 - h[i-lag])
        n_start = max(0, lag)
        n_end = min(N, N + lag)
        # h[i-lag] for i in [n_start, n_end)
        # i-lag ranges from [n_start-lag, n_end-lag) = [max(0,lag)-lag, min(N,N+lag)-lag)
        # = [max(0,lag)-lag, min(N,N+lag)-lag)
        # If lag > 0: [0, N-lag)
        # If lag < 0: [N+lag, N)  wait no
        # If lag >= 0: n_start = lag, n_end = N, so i-lag ranges from 0 to N-lag-1
        # If lag < 0: n_start = 0, n_end = N+lag, so i-lag ranges from -lag to N+lag-(-lag)-1 = N-1
        
        if lag >= 0:
            # i in [lag, N), i-lag in [0, N-lag)
            grad[lag:N] += (1.0 - h[0:N-lag])
            # Second term: i+lag in [lag, N) => i in [0, N-lag)
            grad[0:N-lag] -= h[lag:N]
        else:
            # lag < 0, let l = -lag > 0
            l = -lag
            # i in [0, N-l), i-lag = i+l in [l, N)
            grad[0:N-l] += (1.0 - h[l:N])
            # Second term: i+lag = i-l in [0, N-l) => i in [l, N)
            grad[l:N] -= h[0:N-l]
    return -grad / len(active)

import numpy as np
import time


def project_feasible(y, N):
    target = N / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return c.max() * dx


def get_active(c, n_max=12, rel_tol=0.015):
    c_max = c.max()
    thresh = c_max * (1.0 - rel_tol)
    active = np.where(c >= thresh)[0]
    if len(active) > n_max:
        active = np.argsort(c)[-n_max:]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    return active


def descent_direction(h, active, N):
    grad = np.zeros(N)
    for m in active:
        lag = m - (N - 1)
        if lag >= 0:
            nv = N - lag
            if nv > 0:
                grad[lag:] += (1.0 - h[:nv])
                grad[:nv] -= h[lag:]
        else:
            l = -lag
            nv = N - l
            if nv > 0:
                grad[:nv] += (1.0 - h[l:])
                grad[l:] -= h[:nv]
    return -grad / float(len(active))
