def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    bc = beta * c
    m = np.max(bc)
    e = np.exp(bc - m)
    p = e / np.sum(e)  # softmax weights
    
    # Central difference gradient
    eps = 1e-7
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        c_plus = np.correlate(h_plus, 1.0 - h_plus, mode='full') * dx
        p_plus = beta * c_plus
        m_plus = np.max(p_plus)
        e_plus = np.exp(p_plus - m_plus)
        obj_plus = (m_plus + np.log(np.sum(e_plus))) / beta
        
        h_minus = h.copy()
        h_minus[i] -= eps
        c_minus = np.correlate(h_minus, 1.0 - h_minus, mode='full') * dx
        p_minus = beta * c_minus
        m_minus = np.max(p_minus)
        e_minus = np.exp(p_minus - m_minus)
        obj_minus = (m_minus + np.log(np.sum(e_minus))) / beta
        
        grad[i] = (obj_plus - obj_minus) / (2 * eps)
    
    return grad

def project_feasible(h):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2} using bisection on offset."""
    n = len(h)
    target = n / 2.0
    
    def F(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    
    # Find bounds
    t_lo = -np.max(h) - 1.0
    t_hi = 1.0 - np.min(h) + 1.0
    
    # Make sure F(t_lo) < 0 < F(t_hi)
    if F(t_lo) > 0:
        t_lo = -10.0
    if F(t_hi) < 0:
        t_hi = 10.0
    
    try:
        t_star = brentq(F, t_lo, t_hi, xtol=1e-12, rtol=1e-14)
    except:
        # Fallback: just clip and normalize
        h_clipped = np.clip(h, 0.0, 1.0)
        s = np.sum(h_clipped)
        if s > 0:
            h_clipped = h_clipped * (target / s)
            h_clipped = np.clip(h_clipped, 0.0, 1.0)
        else:
            h_clipped = np.full(n, 0.5)
        return h_clipped
    
    return np.clip(h + t_star, 0.0, 1.0)

def transfer_resolution(h_old, new_n):
    """Linearly interpolate h_old to new_n points, then project."""
    old_n = len(h_old)
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp)

import numpy as np
from scipy.optimize import minimize, brentq
import time

def evaluate_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective_val(c, beta):
    bc = beta * c
    m = np.max(bc)
    e = np.exp(bc - m)
    return (m + np.log(np.sum(e))) / beta

def smooth_obj_func(x, beta, deadline):
    if time.time() > deadline:
        raise TimeoutError
    c = compute_c(x)
    return smooth_objective_val(c, beta)

def smooth_grad_func(x, beta, deadline):
    if time.time() > deadline:
        raise TimeoutError
    n = len(x)
    eps = 1e-7
    grad = np.zeros(n)
    for i in range(n):
        if time.time() > deadline:
            raise TimeoutError
        if x[i] + eps > 1.0:
            h_p = x.copy()
            h_m = x.copy()
            h_m[i] -= eps
            c_p = compute_c(x)
            c_m = compute_c(h_m)
            grad[i] = (smooth_objective_val(c_p, beta) - smooth_objective_val(c_m, beta)) / eps
        elif x[i] - eps < 0.0:
            h_p = x.copy()
            h_p[i] += eps
            c_m = compute_c(x)
            c_p = compute_c(h_p)
            grad[i] = (smooth_objective_val(c_p, beta) - smooth_objective_val(c_m, beta)) / eps
        else:
            h_p = x.copy()
            h_m = x.copy()
            h_p[i] += eps
            h_m[i] -= eps
            c_p = compute_c(h_p)
            c_m = compute_c(h_m)
            grad[i] = (smooth_objective_val(c_p, beta) - smooth_objective_val(c_m, beta)) / (2.0 * eps)

    if not np.all(np.isfinite(h)):

    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(np.sum(h) - 0.5 * n) > tol * n:
        return False
    return True

def _F_offset(t, h, target):
    return np.sum(np.clip(h + t, 0.0, 1.0)) - target

def project_feasible(h):
    n = len(h)
    target = 0.5 * n
    t_lo = -float(np.max(h)) - 0.5
    t_hi = 1.0 - float(np.min(h)) + 0.5
    try:
        f_lo = _F_offset(t_lo, h, target)
        f_hi = _F_offset(t_hi, h, target)
        if f_lo > 0 or f_hi < 0:
            h_clip = np.clip(h, 0.0, 1.0)
            s = np.sum(h_clip)
            if s > 0:
                h_clip = h_clip * (target / s)

            else:
                h_clip = np.full(n, 0.5)
            return h_clip

    old_n = len(h_old)
