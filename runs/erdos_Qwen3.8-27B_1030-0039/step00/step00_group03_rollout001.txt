def golden_section_min(forward_func, lo, hi, max_iter=35, guard=None):
    """Minimize a unimodal function on [lo, hi] using golden section search."""
    if lo >= hi:
        return 0.5 * (lo + hi)
    
    invphi = (5.0**0.5 - 1.0) / 2.0  # ~0.618
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = forward_func(c)
    fd = forward_func(d)
    
    for _ in range(max_iter):
        if guard is not None and guard.expired():
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = forward_func(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = forward_func(d)
        if hi - lo < 1e-10:
            break
    
    t_best = (lo + hi) / 2.0
    return t_best

def simulated_annealing(h_start, guard, max_steps=5000, T_start=None, T_end=1e-5):

    h = h_start.copy()

    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    
    if T_start is None:
        # Estimate from typical improvement
        T_start = 0.02
    
    T = T_start
    cooldown = (T_end / T_start) ** (1.0 / max(1, max_steps))
    
    rng = np.random.RandomState(123)
    
    for step in range(max_steps):
        if guard.expired():
            break
        
        # Propose a pair move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        L = max(-h[i], h[j] - 1.0)
        U = min(1.0 - h[i], h[j])
        if L >= U - 1e-12:
            continue
        
        t = L + rng.random() * (U - L)
        
        h[i] += t
        h[j] -= t
        
        # Verify feasibility
        if h[i] < -1e-10 or h[i] > 1.0 + 1e-10 or h[j] < -1e-10 or h[j] > 1.0 + 1e-10:
            h[i] -= t
            h[j] += t
            T *= cooldown
            continue
        
        new_score = exact_score(h)
        
        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

            if delta < T:
                if np.log(rng.random()) < -delta / T:

                else:
                    h[i] -= t

    return best_h, best_score

def repair_feasible(raw_h, n=None):
    if n is None:
        n = len(raw_h)
    h = np.asarray(raw_h, dtype=np.float64).copy()
    if h.ndim != 1 or len(h) != n:
        return None

        return None
    S = n / 2.0
    lo = -np.max(h)
    hi = 1.0 - np.min(h)

        return None
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s_val = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s_val < S:
            lo = mid
        else:
            hi = mid

            else:

            residual -= adj
    if abs(float(np.sum(result)) - S) > 1e-6:

def golden_section_min(h, i, j, lo, hi, max_iter=30, guard=None):
    if hi - lo < 1e-14:
        return 0.0
    invphi = (5.0 ** 0.5 - 1.0) / 2.0
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = eval_pair_move(h, i, j, c)
    fd = eval_pair_move(h, i, j, d)
    for _ in range(max_iter):
        if guard is not None and guard.expired():
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - invphi * (hi - lo)
            fc = eval_pair_move(h, i, j, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = eval_pair_move(h, i, j, d)
        if hi - lo < 1e-12:
            break
    t_mid = (lo + hi) / 2.0
    s_mid = eval_pair_move(h, i, j, t_mid)
    s_lo = eval_pair_move(h, i, j, lo)
    s_hi = eval_pair_move(h, i, j, hi)
    if s_lo < s_mid and s_lo < s_hi:
        return lo
    if s_hi < s_mid and s_hi < s_lo:

    return t_mid

def simulated_annealing(h_start, guard, max_steps=10000, T_start=None, T_end=1e-4):

            continue

        h[i] += t
        h[j] -= t
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)

        if new_score < current_score:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:

                else:
                    h[i] -= t
                    h[j] += t
                    h[i] = np.clip(h[i], 0.0, 1.0)
                    h[j] = np.clip(h[j], 0.0, 1.0)
        T *= cooldown
    return best_h, best_score

def make_binary_random(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:

def make_fourier_seed(n, freq1, freq2, amp, phase1, phase2, offset, rng):

        h += amp * 0.5 * np.sin(2.0 * np.pi * freq2 * x + phase2)

def transfer_interpolate(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    src_pos = np.linspace(0.0, 1.0, n_src)
    tgt_pos = np.linspace(0.0, 1.0, n_target)
    h_tgt = np.interp(tgt_pos, src_pos, h_src)
    return h_tgt
