    for j in range(len(c)):
        w = p[j] * dx
        if w < 1e-18:
            continue
        d = j - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            # c_j = dx * sum_{i=d}^{n-1} h[i] * (1-h[i-d])
            # d(c_j)/d(h[i]) for i in [d, n-1]: (1-h[i-d])
            # d(c_j)/d(h[i]) for i in [0, n-1-d]: -h[i+d]
            grad[d:] += w * (1.0 - h[:n-d])
            grad[:n-d] += w * (-h[d:])
        else:
            k = -d
            # c_j = dx * sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
            # d/d(h[i]) for i in [k, n-1]: (1-h[i-k])
            # d/d(h[i]) for i in [0, n-1-k]: -h[i+k]
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += w * (-h[k:])
    return val, grad

def topk_objective_and_grad(h, K, tau):
    """Soft top-K objective: approximately the mean of the top K correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    # Sort indices
    idx = np.argsort(c)[::-1]
    top = c[idx[:K]]
    val = top.mean()
    # Use softmax over top-K for gradient
    m = top.max()
    e = np.exp((c - m) / tau)
    p = e / e.sum()
    grad = np.zeros(n)
    for j in range(len(c)):
        w = p[j] * dx / K
        if w < 1e-18:
            continue
        d = j - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += w * (1.0 - h[:n-d])
            grad[:n-d] += w * (-h[d:])
        else:
            k = -d
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += w * (-h[k:])
    return val, grad

def slsqp_optimize(h, deadline, betas=None):
    """SLSQP optimization with LSE objective."""
    n = len(h)
    target = n / 2.0
    if betas is None:
        betas = [20.0, 80.0, 400.0]
    
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    
    for beta in betas:
        rem = deadline - time.time()
        if rem < 15:
            break
        
        def obj(x):
            if time.time() > deadline:
                raise TimeLimitError()
            v, _ = lse_objective_and_grad(x, beta)
            return v
        
        def grad(x):
            if time.time() > deadline:
                raise TimeLimitError()
            _, g = lse_objective_and_grad(x, beta)
            return g
        
        def constraint(x):
            if time.time() > deadline:
                raise TimeLimitError()
            return x.sum() - target
        
        bounds = list(zip(np.zeros(n), np.ones(n)))
        maxiter = max(20, min(300, int(rem * 3)))
        
        try:
            res = minimize(obj, h_curr, jac=grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(np.clip(res.x, 0, 1), target)
                sc = evaluate_exact(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_curr = h_new.copy()
        except (TimeLimitError, Exception):
            break
    
    return best_h, best_score

        def grad(x):
            if time.time() > deadline:
                raise _TL()
            corr_x = np.correlate(x, 1.0 - x, mode="full") * (2.0 / len(x))
            ca = corr_x[active]
            m = ca.max()
            e = np.exp(beta * (ca - m))
            p = e / e.sum()
            g = np.zeros(len(x))
            for ii, idx in enumerate(active):
                w = p[ii] * (2.0 / len(x))
                d = idx - (len(x) - 1)
                nn = len(x)
                if d == 0:
                    g += w * (1.0 - 2.0 * x)
                elif d > 0:
                    g[d:] += w * (1.0 - x[:nn-d])
                    g[:nn-d] -= w * x[d:]
                else:
                    k = -d
                    g[k:] += w * (1.0 - x[:nn-k])
                    g[:nn-k] -= w * x[k:]
            return g
        
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
        
        try:
            res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = _proj(np.clip(res.x, 0, 1), target)
                sc = _eval(h_new)
                if sc < best_score - 1e-13:
                    best_h, best_score = h_new.copy(), sc
                    h = h_new.copy()
                else:
                    break
        except Exception:
            break
    
    # Final coordinate descent
    if time.time() < deadline - 3:
        h, sc = _pair_descent(h, deadline)
        if sc < best_score:
            best_h, best_score = h, sc
    
    return best_h, best_score
