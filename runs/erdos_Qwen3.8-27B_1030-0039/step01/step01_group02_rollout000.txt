def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    if n == 1:
        return np.array([max(0.0, min(1.0, target))]), True
    
    def f_offset(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    
    lo = -float(np.max(h)) - 0.1
    hi = 1.0 - float(np.min(h)) + 0.1
    flo = f_offset(lo)
    fhi = f_offset(hi)
    
    if flo >= 0:
        t = lo
    elif fhi <= 0:
        t = hi
    else:
        try:
            t = brentq(f_offset, lo, hi, xtol=1e-15, rtol=8.881784197001252e-16, maxiter=200)
        except Exception:

            a, b = lo, hi
            for _ in range(100):

                    a = mid
                else:
                    b = mid

def transfer_resolution(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:

    x_old = np.linspace(0.0, 2.0, n_old)

def smooth_score_and_grad(h, beta, top_k=None):
    """Compute smooth log-sum-exp objective and gradient, optionally using only top-k lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    
    m = len(f)
    if top_k is not None and top_k < m:
        # Select top-k indices
        idx = np.argpartition(f, -top_k)[-top_k:]
        f_sel = f[idx]
        # Softmax on selected
        s = beta * f_sel
        s_max = np.max(s)
        w = np.exp(s - s_max)
        Z = np.sum(w)
        psi = (s_max + np.log(Z)) / beta
        p_full = np.zeros(m)
        p_full[idx] = w / Z
    else:
        s = beta * f
        s_max = np.max(s)
        w = np.exp(s - s_max)
        Z = np.sum(w)
        psi = (s_max + np.log(Z)) / beta
        p_full = w / Z
    
    # Gradient computation
    grad = np.zeros(n)
    for k in range(m):
        weight = p_full[k] * dx
        if weight < 1e-18:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            upper = n - lag
            if upper > 0:
                grad[:upper] += weight * (-h[lag:n])
                grad[lag:] += weight * (1.0 - h[:upper])
        else:
            pn = -lag
            upper = n - pn
            if upper > 0:
                grad[:upper] += weight * (1.0 - h[pn:n])
                grad[pn:] += weight * (-h[:upper])
    return psi, grad

def subgradient_step(h, grad, step_size, n):

def exact_pair_line_search(h, i, j, cur_score, n_samples=20):
    """Exact line search along the direction that transfers mass from j to i."""
    n = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return 0.0, cur_score
    
    best_d = 0.0
    best_s = cur_score
    
    # Golden section / ternary search
    ratio = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    for _ in range(n_samples):
        if b - a < 1e-13:
            break
        c = b - ratio * (b - a)
        d_ = a + ratio * (b - a)
        h[i] += c
        h[j] -= c
        s_c = evaluate_exact(h)
        h[i] -= c
        h[j] += c
        h[i] += d_
        h[j] -= d_
        s_d = evaluate_exact(h)
        h[i] -= d_
        h[j] += d_
        
        if s_c < s_d:
            b = d_
        else:
            a = c
        
        if s_c < best_s - 1e-15:
            best_s = s_c
            best_d = c
        if s_d < best_s - 1e-15:
            best_s = s_d
            best_d = d_
    
    # Final evaluation at best
    if best_d != 0.0:
        h[i] += best_d
        h[j] -= best_d
        final_s = evaluate_exact(h)
        if final_s < best_s:
            best_s = final_s
        else:
            h[i] -= best_d
            h[j] += best_d
    
    return best_d, best_s

    h = h.copy()
    cur_score = evaluate_exact(h)
    best_score = cur_score

        else:
            # Use gradient differences to prioritize
            grad_diff = np.abs(grad[:, None] - grad[None, :])
            np.fill_diagonal(grad_diff, 0)
            # Get top pairs
            flat_idx = np.argsort(grad_diff.ravel())[::-1]
            pairs = []
            seen = set()
            for fi in flat_idx[:min(500, n*(n-1)//2)]:
                ii, jj = fi // n, fi % n
                if ii > jj:
                    ii, jj = jj, ii
                if (ii, jj) not in seen and ii != jj:
                    seen.add((ii, jj))
                    pairs.append((ii, jj))
                    if len(pairs) >= 80:
                        break
        
        for (i, j) in pairs:
            if time.time() > deadline - reserve:
                break
            _, cur_score = exact_pair_line_search(h, i, j, cur_score, n_samples=16)
            if cur_score < best_score - 1e-14:

                improved = True
        
        if not improved:
            break
    
    return h, best_score
