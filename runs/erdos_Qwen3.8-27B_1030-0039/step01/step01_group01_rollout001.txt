class TimeLimit(Exception):

_SLSQP_BETA = 50.0
_SLSQP_TARGET = 0.0
_SLSQP_DEADLINE = 0.0
_SLSQP_N = 0

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.array(x, dtype=np.float64).copy()
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.1
    t_hi = 1.0 - x.min() + 0.1
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if fhi <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-15, maxiter=80)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-12) & (out < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx

    m = c.max()

def _slsqp_g(x):
    if time.time() > _SLSQP_DEADLINE:
        raise TimeLimit()
    n = _SLSQP_N
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx

    p = e / Z
    grad = np.zeros(n)

            grad += w * (1.0 - 2.0 * x)

            grad[:n - d] += w * (1.0 - x[d:])
            grad[d:] += w * (-x[:n - d])
        else:
            k = -d
            grad[k:] += w * (1.0 - x[:n - k])
            grad[:n - k] += w * (-x[k:])
    return grad

            idx_j = rng.randint(0, n, size=num_pairs)
            mask_valid = idx_i < idx_j
            idx_i = idx_i[mask_valid]
            idx_j = idx_j[mask_valid]
            pairs = list(zip(idx_i, idx_j))
            rng.shuffle(pairs)
        else:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
