            m = N - r
            if m <= 0:
                continue
            contribs = h[r:] * (1.0 - h[:m])
        
        # Top M contributing pairs
        top_idx = np.argsort(contribs)[::-1][:top_m_per_shift]
        for ti in top_idx:
            if r > 0:
                p1, p2 = int(ti), int(ti + r)
            else:
                p1, p2 = int(ti), int(ti + r)
            key = (min(p1, p2), max(p1, p2))
            if key not in seen:
                seen.add(key)
                pairs.append((key[0], key[1], float(contribs[ti])))
    
    return pairs

import time
import numpy as np


def remaining_budget(t0, budget):
    return float(budget) - (time.perf_counter() - float(t0))


def get_parent_values():
    try:
        vals = initial_h_values
    except Exception:
        return None
    if vals is None:
        return None
    arr = np.asarray(vals, dtype=float).reshape(-1)
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.5)
    return arr


def make_feasible(v):
    arr = np.asarray(v, dtype=float).reshape(-1)
    if arr.size == 0:
        return np.array([0.5], dtype=float)
    N = arr.size
    arr = np.where(np.isfinite(arr), arr, 0.5)
    arr = np.clip(arr, 0.0, 1.0)
    target = 0.5 * float(N)
    cur = float(np.sum(arr))
    if abs(cur - target) < 1e-10:
        return arr.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(arr - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(arr - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(300):
        cur = float(np.sum(h))
        d = target - cur
        if abs(d) < 1e-14:
            break
        if d > 0:
            idx = np.where(h < 1.0 - 1e-13)[0]
            if idx.size == 0:
                break
            h[idx] = np.minimum(h[idx] + d / float(idx.size), 1.0)
        else:
            idx = np.where(h > 1e-13)[0]
            if idx.size == 0:
                break
            h[idx] = np.maximum(h[idx] + d / float(idx.size), 0.0)
    return h


def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
        return float('inf')
    h = np.clip(h, 0.0, 1.0)
    N = h.size
    dx = 2.0 / float(N)
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    return c if np.isfinite(c) else float('inf')


def eval_pair_buf(buf, i, j, t, q):
    buf[i] = t
    buf[j] = q - t
    N = buf.size
    dx = 2.0 / float(N)
    return float(np.max(np.correlate(buf, 1.0 - buf, mode='full'))) * dx


def golden_pair(h, i, j, cur_c, work, max_gs=28):
    q = h[i] + h[j]
    lo = max(0.0, q - 1.0)
    hi = min(1.0, q)
    if hi - lo < 1e-13:
        return h[i], cur_c
    t_orig = h[i]
    mid = 0.5 * (lo + hi)
    best_t = t_orig
    best_c = cur_c
    for pt in (lo, mid, hi):
        if abs(pt - t_orig) < 1e-10:
            continue
        c = eval_pair_buf(work, i, j, pt, q)
        if c < best_c - 1e-11:
            best_c = c
            best_t = pt
    if best_t == t_orig:
        return t_orig, cur_c
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc = eval_pair_buf(work, i, j, c1, q)
    fd = eval_pair_buf(work, i, j, d1, q)
    for _ in range(max_gs):
        if b - a < 1e-10:
            break
        if fc < fd:
            b = d1
            d1 = c1
            fd = fc
            c1 = b - gr * (b - a)
            fc = eval_pair_buf(work, i, j, c1, q)
        else:
            a = c1
            c1 = d1
            fc = fd
            d1 = a + gr * (b - a)
            fd = eval_pair_buf(work, i, j, d1, q)
    tg = 0.5 * (a + b)
    cg = eval_pair_buf(work, i, j, tg, q)
    if cg < best_c - 1e-12:
        best_c = cg
        best_t = tg
    return best_t, best_c


def get_active_pairs(h, top_k=25, top_m=40):
    N = h.size
    dx = 2.0 / float(N)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sort_idx = np.argsort(corr)[::-1]
    pairs = []
    seen = set()
    for si in sort_idx[:top_k]:
        r = int(si) - (N - 1)
        if r == 0:
            continue
        ar = abs(r)
        if ar >= N:
            continue
        m = N - ar
        if m <= 1:
            continue
        if r > 0:
            contribs = h[:m] * (1.0 - h[ar:])
        else:
            contribs = h[ar:] * (1.0 - h[:m])
        top = np.argsort(contribs)[::-1][:top_m]
        for ti in top:
            if r > 0:
                p1, p2 = int(ti), int(ti + ar)
            else:
                p1, p2 = int(ti + ar), int(ti)
            key = (min(p1, p2), max(p1, p2))
            if key not in seen:
                seen.add(key)
                pairs.append(key)
    return pairs


def active_pass(h, cur_c, work, t0, budget, reserve, top_k=25, top_m=40):
    pairs = get_active_pairs(h, top_k, top_m)
    improved = 0
    for (i, j) in pairs:
        if remaining_budget(t0, budget) < reserve + 2.0:
            break
        q = h[i] + h[j]
        t_new, c_new = golden_pair(h, i, j, cur_c, work, max_gs=28)
        if c_new < cur_c - 1e-11:
            h[i] = t_new
            h[j] = q - t_new
            cur_c = c_new
            improved += 1
    return h, cur_c, improved


def all_pairs_pass(h, cur_c, work, t0, budget, reserve, max_pairs=None):
    N = h.size
    total_pairs = N * (N - 1) // 2
    if max_pairs is None:
        max_pairs = total_pairs
    count = 0
    improved = 0
    for i in range(N - 1):
        if remaining_budget(t0, budget) < reserve + 2.0:
            break
        for j in range(i + 1, N):
            if count >= max_pairs:
                break
            count += 1
            if remaining_budget(t0, budget) < reserve + 1.0:
                break
            q = h[i] + h[j]
            t_new, c_new = golden_pair(h, i, j, cur_c, work, max_gs=22)
            if c_new < cur_c - 1e-11:
                h[i] = t_new
                h[j] = q - t_new
                cur_c = c_new
                improved += 1
    return h, cur_c, improved


def random_pair_pass(h, cur_c, work, rng, n_pairs, t0, budget, reserve):
    N = h.size
    improved = 0
    for _ in range(n_pairs):
        if remaining_budget(t0, budget) < reserve + 2.0:
            break
        i = int(rng.integers(N))
        j = int(rng.integers(N))
        if i == j:
            continue
        q = h[i] + h[j]
        t_new, c_new = golden_pair(h, i, j, cur_c, work, max_gs=25)
        if c_new < cur_c - 1e-11:
            h[i] = t_new
            h[j] = q - t_new
            cur_c = c_new
            improved += 1
    return h, cur_c, improved


def perturb_feasible(h, rng, sigma):
    x = h + float(sigma) * rng.standard_normal(h.size)
    return make_feasible(x)


def transfer_length(h, new_N):
    h = np.asarray(h, dtype=float).reshape(-1)
    new_N = int(new_N)
    if h.size == new_N:
        return make_feasible(h)
    if new_N <= 0:
        return np.full(1, 0.5)
    if new_N % h.size == 0:
        factor = new_N // h.size
        y = np.repeat(h, factor)
        return make_feasible(y)
    if h.size % new_N == 0:
        factor = h.size // new_N
        y = h.reshape(new_N, factor).mean(axis=1)
        return make_feasible(y)
    src = (np.arange(h.size) + 0.5) * (2.0 / float(h.size))
    dst = (np.arange(new_N) + 0.5) * (2.0 / float(new_N))
    y = np.interp(dst, src, h, left=float(h[0]), right=float(h[-1]))
    return make_feasible(y)


def offset_pairs_pass(h, cur_c, work, t0, budget, reserve, offsets=None):
    N = h.size
    if offsets is None:
        offsets = list(range(1, min(N, 30)))
    improved = 0
    for d in offsets:
        if remaining_budget(t0, budget) < reserve + 2.0:
            break
        for i in range(N - d):
            j = i + d

                h[i] = t_new
                h[j] = q - t_new
                cur_c = c_new
                improved += 1
    return h, cur_c, improved


def finalize_solution(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
        h = np.array([0.5], dtype=float)
    h = make_feasible(h)
    h = np.clip(h, 0.0, 1.0)
    c = exact_c5(h)
    if not np.isfinite(c):
        h = make_feasible(np.full(h.size, 0.5))
        c = exact_c5(h)
    return h, float(c), int(h.size)

    reserve = min(30.0, max(5.0, 0.04 * budget))
    rng = np.random.default_rng(int(seed))

    parent = get_parent_values()
    if parent is None:
        N0 = 135
        parent = make_feasible(0.5 + 0.35 * np.cos(2.0 * np.pi * 2.0 * np.linspace(0, 1, N0)) + 0.15 * rng.standard_normal(N0))
    else:
        N0 = int(parent.size)

    best_h = make_feasible(parent)
    best_c = exact_c5(best_h)
    work = np.zeros(N0, dtype=float)

    no_improve_count = 0
    for pass_num in range(60):

            break

        h, c, n_imp = active_pass(h, c, work, t0, budget, reserve, top_k=25, top_m=40)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 3 and pass_num > 5:
                break

    # Phase 2: Offset-based passes (small offsets)
    small_offsets = list(range(1, 15))
    for pass_num in range(8):

        elif pass_num > 2:
            break

    # Phase 3: All pairs passes
    total_pairs = N0 * (N0 - 1) // 2
    for pass_num in range(6):
        if remaining_budget(t0, budget) < reserve + 40.0:
            break
        h = best_h.copy()
        c = best_c
        h, c, n_imp = all_pairs_pass(h, c, work, t0, budget, reserve, max_pairs=total_pairs)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        elif pass_num > 1:
            break

    # Phase 4: Interleaved active + random
    for pass_num in range(30):
        if remaining_budget(t0, budget) < reserve + 40.0:
            break
        h = best_h.copy()
        c = best_c
        h, c, _ = active_pass(h, c, work, t0, budget, reserve, top_k=30, top_m=50)

        if c < best_c - 1e-12:

        h = perturb_feasible(best_h, rng, sigmas[restart % len(sigmas)])

            h, c, _ = active_pass(h, c, work_r, t0, budget, reserve, top_k=25, top_m=40)
            if remaining_budget(t0, budget) > reserve + 30.0:

            if remaining_budget(t0, budget) > reserve + 20.0:
