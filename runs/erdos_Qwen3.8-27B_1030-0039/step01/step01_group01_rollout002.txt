def project_feasible(x, target_sum):
    n = len(x)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01

    fhi = F(t_hi)

import numpy as np
from scipy.optimize import minimize, brentq
import time

class TimeLimit(Exception):
    pass

_SLSQP_STATE = {}

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if fhi <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-15)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            out = out.copy()
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    beta_c = beta * c
    m = beta_c.max()
    e = np.exp(beta_c - m)
    Z = e.sum()
    p = e / Z
    val = (m + np.log(Z)) / beta
    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            for i in range(n - d):
                grad[i + d] += w * (1.0 - h[i])
                grad[i] += w * (-h[i + d])
        else:
            k = -d
            for i in range(k, n):
                grad[i - k] += w * (1.0 - h[i])
                grad[i] += w * (-h[i - k])
    return val, grad

def _slsqp_f(x):
    if time.time() > _SLSQP_STATE['deadline']:
        raise TimeLimit()
    v, _ = smooth_obj_grad(x, _SLSQP_STATE['beta'])
    return v

def _slsqp_g(x):
    if time.time() > _SLSQP_STATE['deadline']:
        raise TimeLimit()
    _, g = smooth_obj_grad(x, _SLSQP_STATE['beta'])
    return g

def _slsqp_c(x):
    return x.sum() - _SLSQP_STATE['target']

def run_slsqp(h_init, betas, deadline, target):
    n = len(h_init)
    h_curr = h_init.copy()
    best_h = h_init.copy()
    best_score = evaluate_exact(h_init)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': _slsqp_c}]
    
    for beta in betas:
        rem = deadline - time.time()
        if rem < 10:
            break
        _SLSQP_STATE['beta'] = beta
        _SLSQP_STATE['target'] = target
        _SLSQP_STATE['deadline'] = deadline
        maxiter = max(30, min(200, int(rem * 1.5)))
        try:
            res = minimize(_slsqp_f, h_curr, jac=_slsqp_g, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(np.clip(res.x, 0, 1), target)
                sc = evaluate_exact(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_curr = h_new.copy()
        except TimeLimit:
            break
        except Exception:
            break
    return best_h, best_score

def golden_pair_line(h, i, j, deadline):
    n = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, evaluate_exact(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] += c; h[j] -= c
    fc = evaluate_exact(h)
    h[i] -= c; h[j] += c
    h[i] += d; h[j] -= d
    fd = evaluate_exact(h)
    h[i] -= d; h[j] += d
    for _ in range(25):
        if time.time() > deadline:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            h[i] += c; h[j] -= c
            fc = evaluate_exact(h)
            h[i] -= c; h[j] += c
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            h[i] += d; h[j] -= d
            fd = evaluate_exact(h)
            h[i] -= d; h[j] += d
    best_t = (a + b) / 2.0
    h[i] += best_t; h[j] -= best_t
    fs = evaluate_exact(h)
    h[i] -= best_t; h[j] += best_t
    return best_t, fs

def pair_descent(h, deadline, max_sweeps=20):
    n = len(h)
    h = h.copy()
    best_score = evaluate_exact(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_score
            for j in range(i + 1, n):
                d_best, s_best = golden_pair_line(h, i, j, deadline)
                if d_best != 0.0 and s_best < best_score - 1e-13:
                    h[i] += d_best
                    h[j] -= d_best
                    h = np.clip(h, 0.0, 1.0)
                    diff = n / 2.0 - h.sum()
                    if abs(diff) > 1e-12:
                        h = project_feasible(h, n / 2.0)
                    new_s = evaluate_exact(h)
                    if new_s < best_score - 1e-13:
                        best_score = new_s
                        improved = True
        if not improved:
            break
    return h, best_score

def transfer_to_new_n(h_old, n_new):
    m = len(h_old)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0/m
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new / 2.0)

def run_de(h_best, deadline, target, seed):
    n = len(h_best)
    rng = np.random.RandomState(seed)
    pop_size = max(30, min(60, n + 15))
    cr = 0.8
    f_start = 0.9
    f_end = 0.3
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    scores = np.zeros(pop_size)
    
    # Individual 0: current best
    pop[0] = h_best
    scores[0] = evaluate_exact(h_best)
    
    # Individual 1: reversed
    pop[1] = project_feasible(h_best[::-1].copy(), target)
    scores[1] = evaluate_exact(pop[1])
    
    # Individual 2: reflected+complement
    pop[2] = project_feasible(1.0 - h_best[::-1].copy(), target)
    scores[2] = evaluate_exact(pop[2])
    
    # Rest: random perturbations
    for i in range(3, pop_size):
        sigma = 0.1 + 0.3 * (i / pop_size)
        trial = h_best + rng.randn(n) * sigma
        pop[i] = project_feasible(trial, target)
        scores[i] = evaluate_exact(pop[i])
    
    best_idx = np.argmin(scores)
    best_score = scores[best_idx]
    best_h = pop[best_idx].copy()
    
    max_gens = 5000
    gen = 0
    stale_count = 0
    
    while gen < max_gens:
        if time.time() > deadline:
            break
        gen += 1
        f = f_start + (f_end - f_start) * gen / max_gens
        f = max(f_end, f)
        
        improved_this_gen = False
        for i in range(pop_size):
            if time.time() > deadline:
                return best_h, best_score
            
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            while r1 == r2 or r1 == i or r2 == i:
                r1 = rng.randint(pop_size)
                r2 = rng.randint(pop_size)
            
            # Mutation: DE/best/1 + noise
            trial = pop[best_idx] + f * (pop[r1] - pop[r2]) + rng.randn(n) * 0.01
            trial = np.clip(trial, 0.0, 1.0)
            
            # Crossover
            cross_mask = rng.rand(n) < cr
            cross_mask[rng.randint(n)] = True  # ensure at least one
            trial[cross_mask] = pop[i][cross_mask]
            
            # Project
            trial = project_feasible(trial, target)
            trial_score = evaluate_exact(trial)
            
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score - 1e-14:
                    best_score = trial_score
                    best_h = trial.copy()
                    best_idx = i
                    improved_this_gen = True
        
        if improved_this_gen:
            stale_count = 0
        else:
            stale_count += 1
            if stale_count > 100:
                # Inject diversity
                for i in range(pop_size // 4):

                    trial = h_best + 
rng.randn(n) * 0.2
                    pop[idx] = project_feasible(trial, target)
                    scores[idx] = evaluate_exact(pop[idx])
                stale_count = 0
    
    return best_h, best_score
