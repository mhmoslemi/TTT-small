def pair_line_search(h, i, j, max_evals=25):
    """Golden section search along pair (i,j). Returns (best_h, best_c) or (None, inf)."""
    N = h.size
    hi_val = float(h[i])
    hj_val = float(h[j])
    q = hi_val + hj_val
    lo = max(0.0, q - 1.0)
    hi = min(1.0, q)

        return None, float('inf')

    gr = (np.sqrt(5.0) - 1.0) / 2.0

        h_temp = h.copy()
        h_temp[i] = t
        h_temp[j] = q - t
        return eval_c5_only(h_temp)
    
    # Golden section
    a = lo
    b = hi

    d_pt = a + gr * (b - a)
    fc = f(c_pt)
    fd = f(d_pt)
    evals = 2
    best_t = c_pt if fc < fd else d_pt
    best_f = min(fc, fd)
    
    for _ in range(max_evals - 2):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = f(c_pt)
            evals += 1
            if fc < best_f:
                best_f = fc
                best_t = c_pt
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = f(d_pt)
            evals += 1
            if fd < best_f:
                best_f = fd
                best_t = d_pt
    
    # Also check endpoints
    for t_check in [lo, hi, lo + 0.1 * width, hi - 0.1 * width]:
        t_check = max(lo, min(hi, t_check))
        h_temp = h.copy()
        h_temp[i] = t_check
        h_temp[j] = q - t_check
        fc = eval_c5_only(h_temp)
        if fc < best_f:
            best_f = fc
            best_t = t_check

        h_new[j] = q - best_t

        h_new[j] = max(0.0, min(1.0, h_new[j]))
        return h_new, best_f
    return None, float('inf')

def active_pairs_from_shift(corr_contrib, shift, h, top_k=20):
    """Get pairs of indices that contribute most to a given shift."""
    N = h.size
    s = int(abs(shift))
    if s <= 0:
        # shift 0: h[i] * (1-h[i]) for all i
        contribs = h * (1.0 - h)
        idx = np.argsort(contribs)[-top_k:]
        pairs = []
        for i in idx:
            pairs.append((int(i), (int(i) + 1) % N))
        return pairs
    if s < N:
        m = N - s
        left_contrib = h[:m] * (1.0 - h[s:])
        right_contrib = h[s:] * (1.0 - h[:m])
        # For left: pairs are (k, k+s)
        # For right: pairs are (k+s, k)
        top_left = np.argsort(left_contrib)[-top_k:]
        top_right = np.argsort(right_contrib)[-top_k:]
        pairs = []
        for k in top_left:
            pairs.append((int(k), int(k + s)))
        for k in top_right:
            pairs.append((int(k + s), int(k)))
        return pairs
    return []

                    pairs.add((min(int(i), j), max(int(i), j)))

                right_contrib = h[s_abs:] * (1.0 - h[:m])
                top_l = np.argsort(left_contrib)[-top_k_per_shift:]
                top_r = np.argsort(right_contrib)[-top_k_per_shift:]
                for k in top_l:
                    pairs.add((int(k), int(k + s_abs)))
                for k in top_r:
                    pairs.add((int(k + s_abs), int(k)))

                for k in top_l:
                    pairs.add((int(k), int(k + s_p)))

        idx = np.random.default_rng(0).permutation(len(pair_list))[:max_pairs]

        deadline = t0 + budget - reserve

    no_improve = 0
    total_passes = 0

    while time.perf_counter() < deadline - 5.0:
        total_passes += 1
        improved = False

        if total_passes % 4 == 3:
            pairs = random_pairs(rng, N, 250)
            me = 16
        else:
            pairs = get_active_pairs(h, top_k=8, max_pairs=350, slack=0.025)
            me = 22 if time.perf_counter() < t0 + 0.6 * budget else 16

        perm = rng.permutation(len(pairs))
        for idx_perm in perm:
            if time.perf_counter() > deadline - 10.0:
                break
            pi, pj = pairs[int(idx_perm)]
            h_new, c_new = pair_line_search(h, pi, pj, max_evals=me)
            if h_new is not None and np.isfinite(c_new) and c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                h = h + 0.02 * rng.normal(0.0, 1.0, N)
                h = make_feasible(h)
                no_improve = 0
            elif no_improve >= 8:
                h = h + 0.05 * rng.normal(0.0, 1.0, N)
                h = make_feasible(h)
                no_improve = 0
            elif no_improve >= 14:
                h = make_feasible(parent)
                no_improve = 0

    if time.perf_counter() < deadline - 80.0:
        N2 = 2 * N
        if N2 <= 400:
            h2 = transfer_up(h, 2)
            c2 = exact_c5(h2)
            N2_size = h2.size
            for _ in range(800):
                if time.perf_counter() > deadline - 60.0:
                    break
                i2 = int(rng.integers(N2_size))
                j2 = int(rng.integers(N2_size))
                if i2 == j2:
                    continue
                h2_new, c2_new = pair_line_search(h2, i2, j2, max_evals=15)
                if h2_new is not None and np.isfinite(c2_new) and c2_new < c2 - 1e-12:
                    h2 = h2_new
                    c2 = c2_new
            src2 = (np.arange(N2) + 0.5) * (2.0 / float(N2))
            dst2 = (np.arange(N) + 0.5) * (2.0 / float(N))
            h_back = np.interp(dst2, src2, h2, left=float(h2[0]), right=float(h2[-1]))
            h_back = make_feasible(h_back)
            c_back = exact_c5(h_back)
            if np.isfinite(c_back) and c_back < best_c - 1e-12:
                best_c = c_back
                best_h = h_back.copy()
                h = h_back.copy()

    if time.perf_counter() < deadline - 15.0:
        pairs_final = get_active_pairs(best_h, top_k=20, max_pairs=500, slack=0.015)
        perm_f = rng.permutation(len(pairs_final))
        for idx_f in perm_f:
            if time.perf_counter() > deadline:
                break
            pi, pj = pairs_final[int(idx_f)]
            h_new, c_new = pair_line_search(best_h, pi, pj, max_evals=30)
            if h_new is not None and np.isfinite(c_new) and c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = exact_c5(best_h)
    if not np.isfinite(final_c) or final_c > best_c + 1e-8:
        best_h = make_feasible(best_h)
        final_c = exact_c5(best_h)
    if not np.isfinite(final_c):
        best_h = make_feasible(np.full(N0, 0.5))
        final_c = exact_c5(best_h)

    return best_h, float(final_c), int(best_h.size)
