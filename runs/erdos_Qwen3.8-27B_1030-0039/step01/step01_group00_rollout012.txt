    lo = -2.0 - float(np.max(h_raw))
    hi = 2.0 + 1.0 - float(np.min(h_raw))

import numpy as np
from scipy.optimize import minimize, brentq
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h_raw, n):
    target = n / 2.0
    def _f(t):
        return float(np.sum(np.clip(h_raw + t, 0.0, 1.0))) - target
    lo = -2.0 - float(np.max(h_raw))
    hi = 2.0 + 1.0 - float(np.min(h_raw))
    flo = _f(lo)
    fhi = _f(hi)
    if flo >= 0.0:
        return np.clip(h_raw + lo, 0.0, 1.0)
    if fhi <= 0.0:
        return np.clip(h_raw + hi, 0.0, 1.0)
    t = brentq(_f, lo, hi, xtol=1e-15, rtol=1e-15, maxiter=200)
    return np.clip(h_raw + t, 0.0, 1.0)


def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def smooth_obj_grad(x, beta, n, deadline):
    if time.time() > deadline:
        raise TimeoutError()
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    bcorr = beta * corr
    m = np.max(bcorr)
    w = np.exp(bcorr - m)
    Z = np.sum(w)
    obj = float((m + np.log(Z)) / beta)
    p = w / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wkt = p[k] * dx
        if wkt < 1e-15:
            continue
        if d >= 0:
            if d < n:
                grad[d:] += wkt * (1.0 - x[:n - d])
            if (n - d) > 0:
                grad[:n - d] += wkt * (-x[d:])
        else:
            pp = -d
            if pp < n:
                grad[:n - pp] += wkt * (1.0 - x[pp:])
            if (n - pp) > 0:
                grad[pp:] += wkt * (-x[:n - pp])
    return obj, grad


def smooth_slsqp(h0, n, deadline, betas, maxiter_per_beta=120):
    best_h = h0.copy()
    best_score = exact_score(h0)
    bounds = [(0.0, 1.0)] * n

    def con_eq(x):
        return float(np.sum(x)) - n / 2.0

    constraints = [{'type': 'eq', 'fun': con_eq}]
    current = best_h.copy()

    for beta in betas:
        if time.time() > deadline - 8.0:
            break

        def obj_func(x):
            if time.time() > deadline:
                raise TimeoutError()
            o, _ = smooth_obj_grad(x, beta, n, deadline)
            return o

        def grad_func(x):
            if time.time() > deadline:
                raise TimeoutError()
            _, g = smooth_obj_grad(x, beta, n, deadline)
            return g

        try:
            res = minimize(obj_func, current, jac=grad_func,
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': maxiter_per_beta, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                candidate = project_feasible(res.x, n)
                score = exact_score(candidate)
                if score < best_score:
                    best_h = candidate.copy()
                    best_score = score
                current = candidate.copy()
        except (TimeoutError, Exception):
            pass

    return best_h, best_score

    best_score = exact_score(best_h)
    rng = np.random.RandomState(seed)
    no_improve = 0

    for rnd in range(max_rounds):
        if time.time() > deadline - 5.0:
            break
        if no_improve > 120:
            break

        ag = active_gradient(best_h, n)
        n_pairs = min(200, n * (n - 1) // 2)
        improved = False

        for _ in range(n_pairs):
            if time.time() > deadline - 5.0:
                break

            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue

            for direction in (1, -1):
                if direction == 1:
                    lo_d = -best_h[j] + 1e-10
                    hi_d = 1.0 - best_h[i] - 1e-10
                else:
                    lo_d = -best_h[i] + 1e-10
                    hi_d = 1.0 - best_h[j] - 1e-10

                if hi_d - lo_d < 1e-10:
                    continue

                deltas = np.linspace(lo_d, hi_d, 7)
                best_ld = 0.0
                best_ls = best_score
                for dv in deltas:
                    if abs(dv) < 1e-10:
                        continue
                    tmp = best_h.copy()
                    if direction == 1:
                        tmp[i] += dv
                        tmp[j] -= dv
                    else:
                        tmp[i] -= dv
                        tmp[j] += dv
                    sc = exact_score(tmp)
                    if sc < best_ls:
                        best_ls = sc
                        best_ld = dv

                if best_ld != 0.0 and best_ls < best_score - 1e-12:
                    if direction == 1:
                        best_h[i] += best_ld
                        best_h[j] -= best_ld
                    else:
                        best_h[i] -= best_ld
                        best_h[j] += best_ld
                    best_h = project_feasible(best_h, n)
                    best_score = best_ls
                    improved = True
                    no_improve = 0
                    break

            if improved:
                break

        if not improved:
            no_improve += 1

    return best_h, best_score


def gradient_descent_subgrad(h0, n, deadline, max_iter=2000, lr_init=0.01):
    best_h = h0.copy()
    best_score = exact_score(best_h)
    lr = lr_init
    n_pairs_stuck = 0

    for it in range(max_iter):
        if time.time() > deadline - 5.0:
            break

        ag = active_gradient(best_h, n)
        diff = ag[np.argmax(ag)] - ag[np.argmin(ag)]
        if diff < 1e-10:
            break

        d_best = np.argmax(ag)
        s_best = np.argmin(ag)

        if ag[d_best] - ag[s_best] > 0:
            lo_d = -best_h[s_best] + 1e-10
            hi_d = 1.0 - best_h[d_best] - 1e-10
        else:
            lo_d = -best_h[d_best] + 1e-10
            hi_d = 1.0 - best_h[s_best] - 1e-10

        if hi_d - lo_d < 1e-10:
            break

        step = min(lr, hi_d - lo_d) * 0.5
        if ag[d_best] > ag[s_best]:
            tmp = best_h.copy()
            tmp[d_best] += step
            tmp[s_best] -= step
        else:
            tmp = best_h.copy()
            tmp[d_best] -= step
            tmp[s_best] += step

        tmp = project_feasible(tmp, n)
        sc = exact_score(tmp)
        if sc < best_score - 1e-12:
            best_h = tmp
            best_score = sc
            n_pairs_stuck = 0
        else:
            n_pairs_stuck += 1
            lr = max(lr * 0.8, 1e-6)
            if n_pairs_stuck > 50:
                break

    return best_h, best_score


def transfer_resolution(h_old, new_n):
    m = len(h_old)
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h_old)
    return project_feasible(h_new, new_n)


def fourier_profile(n, rng, D=6):
    x = 2.0 * np.arange(n) / n
    profile = np.ones(n) * 0.5
    for r in range(1, D + 1):
        amp = rng.uniform(0.05, 0.35) / r
        phase = rng.uniform(0, 2 * np.pi)
        profile += amp * np.cos(np.pi * r * x + phase)
    return project_feasible(profile, n)


def block_profile(n, rng):
    h = np.zeros(n)
    blocks = rng.randint(3, 8)
    boundaries = sorted(rng.randint(1, n, size=blocks - 1))
    for b in range(blocks):
        lo = boundaries[b - 1] if b > 0 else 0
        hi = boundaries[b] if b < blocks - 1 else n
        if rng.random() > 0.5:
            h[lo:hi] = rng.uniform(0.6, 1.0)
        else:
            h[lo:hi] = rng.uniform(0.0, 0.4)
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12.0
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=np.float64)
        base_n = len(parent)
    except Exception:
        parent = None
        base_n = 45

    if parent is not None and len(parent) > 0 and np.all(np.isfinite(parent)):
        best_h = project_feasible(parent, base_n)
        best_score = exact_score(best_h)
        best_n = base_n
    else:
        best_h = np.full(base_n, 0.5)
        best_score = exact_score(best_h)
        best_n = base_n

    # Phase 1: Quick optimization at base resolution
    if deadline - time.time() > 50:
        h1, s1 = smooth_slsqp(best_h.copy(), best_n, deadline,
                              [5, 20, 60, 200, 600, 1500], 100)
        if s1 < best_score:
            best_h, best_score, best_n = h1, s1, best_n

    if deadline - time.time() > 30:
        hg, sg = gradient_descent_subgrad(best_h.copy(), best_n, deadline, max_iter=500, lr_init=0.02)
        if sg < best_score:
            best_h, best_score, best_n = hg, sg, best_n

    if deadline - time.time() > 20:
        hc, sc = coord_descent(best_h.copy(), best_n, deadline, max_rounds=300, seed=seed)
        if sc < best_score:
            best_h, best_score, best_n = hc, sc, best_n

    # Phase 2: Transfer to higher resolution
    target_resolutions = []
    if best_n <= 60:
        target_resolutions = [90, 120]
    elif best_n <= 90:
        target_resolutions = [120, 150]
    else:
        target_resolutions = [best_n + 30, best_n + 60]

    for tgt_n in target_resolutions:
        if time.time() > deadline - 100:
            break
        if tgt_n <= best_n:
            continue

        ht = transfer_resolution(best_h, tgt_n)
        st = exact_score(ht)

        if deadline - time.time() > 70:
            hn, sn = smooth_slsqp(ht, tgt_n, deadline,
                                  [5, 15, 50, 150, 500, 1200], 80)
            if sn < best_score:
                best_h, best_score, best_n = hn, sn, tgt_n

        if deadline - time.time() > 40:
            hg2, sg2 = gradient_descent_subgrad(best_h.copy(), best_n, deadline, max_iter=400, lr_init=0.015)
            if sg2 < best_score:
                best_h, best_score, best_n = hg2, sg2, best_n

        if deadline - time.time() > 30:
            hd, sd = coord_descent(best_h.copy(), best_n, deadline, max_rounds=400, seed=seed + 7)
            if sd < best_score:
                best_h, best_score, best_n = hd, sd, best_n

    # Phase 3: Random restarts
    restart_values = [0.03, 0.05, 0.08, 0.12, 0.2, 0.35]
    for idx, sigma in enumerate(restart_values):
        if time.time() > deadline - 35:
            break
        pert = best_h + sigma * rng.randn(best_n)
        pert = project_feasible(pert, best_n)
        hp, sp = smooth_slsqp(pert, best_n, deadline,
                              [10, 40, 150, 500], 70)
        if sp < best_score:
            best_h, best_score, best_n = hp, sp, best_n
        if deadline - time.time() > 25:
            hcr, scr = coord_descent(best_h.copy(), best_n, deadline, max_rounds=200, seed=seed + idx)
            if scr < best_score:
                best_h, best_score, best_n = hcr, scr, best_n

    # Phase 4: Structure-based restarts
    for _ in range(4):
        if time.time() > deadline - 35:
            break
        for gen_func in [fourier_profile, block_profile]:
            if time.time() > deadline - 20:
                break
            prof = gen_func(best_n, rng)
            hp, sp = smooth_slsqp(prof, best_n, deadline,
                                  [10, 40, 150], 60)
            if sp < best_score:
                best_h, best_score, best_n = hp, sp, best_n

    # Phase 5: Complement and reversal
    if time.time() > deadline - 35:
        comp = project_feasible(1.0 - best_h, best_n)
        hc, sc = smooth_slsqp(comp, best_n, deadline, [10, 50, 200], 70)
        if sc < best_score:
            best_h, best_score, best_n = hc, sc, best_n

    if time.time() > deadline - 35:
        rev = best_h[::-1].copy()
        hr, sr = smooth_slsqp(rev, best_n, deadline, [10, 50, 200], 70)
        if sr < best_score:
            best_h, best_score, best_n = hr, sr, best_n

    # Phase 6: Final intensive coordinate descent + subgradient
    if time.time() > deadline - 20:
        hf, sf = coord_descent(best_h.copy(), best_n, deadline, max_rounds=600, seed=seed + 42)
        if sf < best_score:
            best_h, best_score, best_n = hf, sf, best_n

    if time.time() > deadline - 15:
        hg3, sg3 = gradient_descent_subgrad(best_h.copy(), best_n, deadline, max_iter=1000, lr_init=0.01)
        if sg3 < best_score:
            best_h, best_score, best_n = hg3, sg3, best_n

    if time.time() > deadline - 15:
        hf2, sf2 = coord_descent(best_h.copy(), best_n, deadline, max_rounds=400, seed=seed + 84)
        if sf2 < best_score:
            best_h, best_score, best_n = hf2, sf2, best_n

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_score = exact_score(best_h)

    return (best_h, best_score, best_n)
