def minmax_slsqp(h0, n, limit, C_init, maxiter=200):
    """
    Minimize C subject to C_k(h) <= C for all k, sum(h) = n/2, 0 <= h <= 1.
    Variables: [h_0, ..., h_{n-1}, C]
    """
    def obj(y):
        limit.check()
        return float(y[-1])
    
    def obj_grad(y):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def ineq(y):
        limit.check()
        h = y[:n]
        C = y[-1]
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return C - corr  # >= 0
    
    def ineq_jac(y):
        limit.check()
        h = y[:n]
        dx = 2.0 / n
        m = 2 * n - 1
        jac = np.zeros((m, n + 1))
        for k in range(m):
            d = k - (n - 1)
            g = corr_grad_single(h, d, n)
            jac[k, :n] = -g  # dg_k/dh = -dC_k/dh
            jac[k, n] = 1.0   # dg_k/dC = 1
        return jac
    
    def eq(y):
        return float(np.sum(y[:n])) - n / 2.0
    
    def eq_jac(y):
        g = np.ones(n + 1)
        g[-1] = 0.0
        return g
    
    y0 = np.append(h0, C_init)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac},
    ]
    
    try:
        res = minimize(obj, y0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_res = project_feasible(res.x[:n], n)
            return h_res, exact_score(h_res)
    except Exception:
        pass
    return h0.copy(), exact_score(h0)

def active_coord_descent(h0, n, limit, max_rounds=500, tol=1e-10):
    """
    Coordinate descent focusing on active lags.
    """
    best_h = h0.copy()
    best_score = exact_score(best_h)
    no_improve = 0
    rng = np.random.RandomState(42)
    
    for rnd in range(max_rounds):
        limit.check()
        if no_improve > 200:
            break
        
        corr = corr_array(best_h)
        c_max = float(np.max(corr))
        # Active lags: within 1% of max
        active_mask = corr > c_max - 0.005
        active_indices = np.where(active_mask)[0]
        
        # Compute gradient at the worst lag
        worst_k = int(np.argmax(corr))
        d_worst = worst_k - (n - 1)
        g_worst = corr_grad_single(best_h, d_worst, n)
        
        # Also compute average gradient over active lags
        g_avg = np.zeros(n)
        for ak in active_indices:
            d_ak = ak - (n - 1)
            g_avg += corr_grad_single(best_h, d_ak, n)
        g_avg /= len(active_indices)
        
        # Try pair transfers
        # Priority: pairs where one is "high" and the other is "low" in the gradient
        # This is a heuristic for which transfers reduce the active lags most
        
        n_pairs = min(n * (n-1) // 2, 500)
        pairs = []
        for _ in range(n_pairs):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        
        # Sort by gradient difference (bigger diff = more likely to help)
        pair_scores = np.array([abs(g_worst[p[0]] - g_worst[p[1]]) for p in pairs])
        order = np.argsort(-pair_scores)
        
        improved = False
        for idx in order[:200]:  # Try top 200 pairs
            limit.check()
            i, j = pairs[idx]
            
            # Try transferring mass from i to j and vice versa
            for sign in (1, -1):
                if sign == 1:
                    # h[i] += t, h[j] -= t
                    lo_t = max(-best_h[i], -1.0)
                    hi_t = min(1.0 - best_h[i] + best_h[j], 1.0)
                    lo_t = max(lo_t, -best_h[j])  # h[j] - t >= 0
                    hi_t = min(hi_t, 1.0 - best_h[i])  # h[i] + t <= 1
                else:
                    # h[i] -= t, h[j] += t
                    lo_t = max(-best_h[j], -1.0)
                    hi_t = min(1.0 - best_h[j] + best_h[i], 1.0)
                    lo_t = max(lo_t, -best_h[i])
                    hi_t = min(hi_t, 1.0 - best_h[j])
                
                if hi_t - lo_t < 1e-12:
                    continue
                
                # Try several values of t
                ts = np.linspace(lo_t, hi_t, 9)
                best_t = 0.0
                best_ts = best_score
                for t in ts:
                    if abs(t) < 1e-12:
                        continue
                    tmp = best_h.copy()
                    if sign == 1:
                        tmp[i] += t
                        tmp[j] -= t
                    else:
                        tmp[i] -= t
                        tmp[j] += t
                    s = exact_score(tmp)
                    if s < best_ts - 1e-12:

                    best_score = best_ts
                    improved = True
                    no_improve = 0
