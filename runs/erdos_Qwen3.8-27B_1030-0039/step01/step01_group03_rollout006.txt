                h = np.clip(h + d / float(N), 0.0, 1.0)
        else:
            idx = np.where(h + d >= 0.0)[0]
            if idx.size > 0:
                h[idx[0]] += d
            else:
                h = np.clip(h + d / float(N), 0.0, 1.0)
    return h


def exact_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
        return float("inf")
    h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    N = h.size
    dx = 2.0 / float(N)
    c = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    return c if np.isfinite(c) else float("inf")


def lag_values_and_gradients(h):
    N = h.size
    dx = 2.0 / float(N)
    n_lags = 2 * N - 1
    vals = np.empty(n_lags, dtype=float)
    G = np.zeros((n_lags, N), dtype=float)
    vals[0] = dx * float(np.dot(h, 1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for s in range(1, N):
        m = N - s
        vals[idx] = dx * float(np.dot(h[:m], 1.0 - h[s:]))
        G[idx, :m] = dx * (1.0 - h[s:s + m])
        G[idx, s:] += -dx * h[:m]
        vals[idx + 1] = dx * float(np.dot(h[s:], 1.0 - h[:m]))
        G[idx + 1, s:] = dx * (1.0 - h[:m])
        G[idx + 1, :m] += -dx * h[s:s + m]
        idx += 2
    return vals, G


def solve_tangent_lp(h, vals, G, active_idx, near_idx, M, trust_r, eta=0.25):
    N = h.size
    n_vars = N + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = -1.0
    A_rows = []
    b_rows = []
    for k in active_idx:
        row = np.zeros(n_vars)
        row[:N] = G[k]
        row[-1] = 1.0
        A_rows.append(row)
        b_rows.append(0.0)
    for k in near_idx:
        row = np.zeros(n_vars)
        row[:N] = G[k]
        row[-1] = eta
        A_rows.append(row)
        slack = M - vals[k]
        if slack < -1e-12:
            slack = 0.0
        b_rows.append(slack)
    delta_max = max(0.005, 0.1 / N)
    row = np.zeros(n_vars)
    row[-1] = 1.0
    A_rows.append(row)
    b_rows.append(delta_max)
    if len(A_rows) == 0:
        return None, 0.0
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(N):
        lo = -min(trust_r, h[i] + 1e-12)
        hi = min(trust_r, 1.0 - h[i] + 1e-12)
        bounds.append((lo, hi))
    bounds.append((0.0, delta_max))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
    except Exception:
        return None, 0.0
    if not res.success:
        return None, 0.0
    d = res.x[:N].copy()
    delta = float(res.x[N])
    d[np.abs(d) < 1e-14] = 0.0
    if not np.all(np.isfinite(d)) or delta < 1e-10:
        return None, 0.0
    return d, delta


def line_search(h, d, current_c, t0, budget, reserve):
    N = h.size
    pos_mask = d > 1e-10
    neg_mask = d < -1e-10
    alpha_max = 1.0
    if np.any(pos_mask):
        alpha_max = min(alpha_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        alpha_max = min(alpha_max, float(np.min(h[neg_mask] / (-d[neg_mask]))))
    alpha_max = max(alpha_max, 1e-6)
    alphas = [alpha_max, 0.7 * alpha_max, 0.5 * alpha_max, 0.35 * alpha_max,
              0.25 * alpha_max, 0.16 * alpha_max, 0.08 * alpha_max,
              0.04 * alpha_max, 0.02 * alpha_max, 0.01 * alpha_max,
              0.005 * alpha_max, 0.001 * alpha_max]
    best_h = h.copy()
    best_c = current_c
    for a in alphas:
        if a <= 1e-10:
            continue
        if time_left(t0, budget) < reserve + 2.0:
            break
        cand = h + a * d
        cand = make_feasible(cand)
        c = exact_c5(cand)
        if np.isfinite(c) and c < best_c - 1e-13:
            best_c = c
            best_h = cand
    return best_h, best_c


def tangent_descent_loop(h, c, t0, budget, reserve, max_iters=2000):
    h = make_feasible(h)
    c = exact_c5(h)
    best_h = h.copy()
    best_c = c
    N = h.size
    trust_r = 0.15
    stagnant = 0
    for it in range(max_iters):
        if time_left(t0, budget) < reserve + 10.0:
            break
        vals, G = lag_values_and_gradients(h)
        M = float(np.max(vals))
        active_tol = max(1e-6, 2e-4 * M)
        near_tol = 5.0 * active_tol
        active_idx = np.where(vals >= M - active_tol)[0]
        near_idx = np.where((vals < M - active_tol) & (vals >= M - near_tol))[0]
        if len(near_idx) > 40:
            near_idx = near_idx[np.argsort(vals[near_idx])[-30:]]
        d, delta = solve_tangent_lp(h, vals, G, active_idx, near_idx, M, trust_r)
        if d is None or delta < 1e-8:
            trust_r *= 0.7
            if trust_r < 0.005:
                stagnant += 1
                if stagnant > 5:
                    break
            continue
        new_h, new_c = line_search(h, d, c, t0, budget, reserve)
        if new_c < c - 1e-13:
            h = new_h
            c = new_c
            best_h = h.copy()
            best_c = c
            stagnant = 0
            trust_r = min(0.2, trust_r * 1.1)
        else:
            stagnant += 1
            trust_r *= 0.8
            if trust_r < 0.003:
                if stagnant > 8:
                    break
    return best_h, best_c


def pair_refine(h, c, rng, n_pairs, t0, budget, reserve):
    h = make_feasible(h)
    c = exact_c5(h)
    best_h = h.copy()
    best_c = c
    N = h.size
    for _ in range(n_pairs):
        if time_left(t0, budget) < reserve + 5.0:
            break
        i = int(rng.integers(N))
        j = int(rng.integers(N))
        if i == j:
            continue
        q = h[i] + h[j]
        lo_t = max(0.0, q - 1.0)
        hi_t = min(1.0, q)
        if hi_t - lo_t < 1e-10:
            continue
        t0_val = h[i]
        for frac in [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0]:
            t_new = lo_t + frac * (hi_t - lo_t)
            if abs(t_new - t0_val) < 1e-10:
                continue
            cand = h.copy()
            cand[i] = t_new
            cand[j] = q - t_new
            cand = make_feasible(cand)
            c_new = exact_c5(cand)
            if np.isfinite(c_new) and c_new < best_c - 1e-13:
                best_c = c_new
                best_h = cand
                h = cand
                c = c_new
                break
    return best_h, best_c


def transfer(h, new_N):
    h = np.asarray(h, dtype=float).reshape(-1)
    new_N = int(new_N)
    if h.size == new_N:
        return make_feasible(h)
    if h.size == 0 or new_N <= 0:
        return np.full(max(1, new_N), 0.5)
    if new_N % h.size == 0:
        factor = new_N // h.size
        y = np.repeat(h, factor)
        return make_feasible(y)
    if h.size % new_N == 0:
        factor = h.size // new_N
        y = h.reshape(new_N, factor).mean(axis=1)
        return make_feasible(y)
    src = (np.arange(h.size) + 0.5) * (2.0 / float(h.size))
    dst = (np.arange(new_N) + 0.5) * (2.0 / float(new_N))
    y = np.interp(dst, src, h, left=float(h[0]), right=float(h[-1]))
    return make_feasible(y)


def perturb_feasible(h, rng, sigma):
    x = h + float(sigma) * rng.normal(0.0, 1.0, h.size)
    return make_feasible(x)


def gradient_descent_pass(h, c, t0, budget, reserve, max_iters=500):
    h = make_feasible(h)
    c = exact_c5(h)
    best_h = h.copy()
    best_c = c
    N = h.size
    for it in range(max_iters):
        if time_left(t0, budget) < reserve + 10.0:
            break
        vals, G = lag_values_and_gradients(h)
        M = float(np.max(vals))
        active_tol = max(1e-6, 1e-3 * M)
        active_mask = vals >= M - active_tol
        n_active = int(np.sum(active_mask))
        if n_active == 0:
            n_active = 1
            active_idx = np.argsort(vals)[-1:]
        else:
            active_idx = np.where(active_mask)[0]
        avg_grad = np.mean(G[active_idx], axis=0)
        avg_grad -= np.mean(avg_grad)
        norm = float(np.linalg.norm(avg_grad))
        if norm < 1e-12:
            break
        u = -avg_grad / norm
        found = False
        for L in [0.05, 0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            cand = make_feasible(h + L * u)
            c_new = exact_c5(cand)
            if np.isfinite(c_new) and c_new < best_c - 1e-13:
                best_c = c_new
                best_h = cand
                h = cand
                c = c_new
                found = True
                break
        if not found:
            break
    return best_h, best_c
