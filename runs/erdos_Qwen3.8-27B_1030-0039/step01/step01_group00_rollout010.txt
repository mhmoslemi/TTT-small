    def __init__(self, deadline):
        self.deadline = deadline
    def check(self):
        if time.time() > self.deadline:
            raise TimeoutError()
    def remaining(self):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

    hi = 3.0

        fhi = _f(hi)
