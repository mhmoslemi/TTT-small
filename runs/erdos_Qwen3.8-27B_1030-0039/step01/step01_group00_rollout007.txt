from scipy.optimize import linprog, brentq

class _TL:
    def __init__(self, deadline):
        self.dl = deadline

        if time.time() > self.dl:

    def rem(self):
        return max(0.0, self.dl - time.time())

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def _proj(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    def _f(t):
        return float(np.sum(np.clip(h + t, 0.0, 1.0))) - target
    flo, fhi = _f(lo), _f(hi)
    if flo >= 0.0 or fhi <= 0.0:
        return np.clip(h, 0.0, 1.0)
    try:
        t = brentq(_f, lo, hi, xtol=1e-15, rtol=1e-15)
    except Exception:
        return np.clip(h, 0.0, 1.0)
    return np.clip(h + t, 0.0, 1.0)

def _grad_lag(h, n, d):
    dx = 2.0 / n
    g = np.zeros(n)
    if d >= 0:
        g[d:] += (1.0 - h[:n - d])
        g[:n - d] += (-h[d:])
    else:
        p = -d
        g[:n - p] += (1.0 - h[p:])
        g[p:] += (-h[:n - p])
    return dx * g

def _lp_dir(h, n, d_lags, tau):
    K = len(d_lags)
    if K == 0:
        return None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A = np.zeros((K, n + 1))
    for i, d in enumerate(d_lags):
        g = _grad_lag(h, n, d)
        A[i, :n] = g
        A[i, n] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-tau, tau)] * n + [(-10.0, 10.0)]
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

            z_val = res.x[n]
            delta = res.x[:n]
            if z_val < -1e-9 and np.max(np.abs(delta)) > 1e-10:

    except Exception:
        pass
    return None

def _line_search(h, delta, cur_c, n, limit):

def _transfer(h_old, n_new):
    n_old = len(h_old)
    h_new = np.zeros(n_new)
    for i in range(n_old):
        lo = int(i * n_new / n_old)
        hi = int((i + 1) * n_new / n_old)
        hi = max(hi, lo + 1)
        h_new[lo:hi] = h_old[i]
    return _proj(h_new, n_new)
