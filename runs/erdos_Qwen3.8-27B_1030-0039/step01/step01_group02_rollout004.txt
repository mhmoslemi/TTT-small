def grad_lag(h, p, n):
    """Gradient of g_lag where |lag| = p."""
    dx = 2.0 / n
    if p >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    upper = n - p
    grad[:upper] += dx * (1.0 - h[p:])
    grad[p:] -= dx * h[:upper]
    return grad

def solve_active_qp(h, n, lags, mu):
    """Solve the active-set linearized QP."""
    if not HAS_CVXPY:
        return None
    
    target = n / 2.0
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    obj = cp.Minimize(t + mu * cp.sum_squares(h_var - h))
    
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    for lag_idx in lags:
        # lag_idx is index into the (2n-1) array, actual lag = lag_idx - (n-1)
        lag = lag_idx - (n - 1)
        p = abs(lag)
        if p >= n:
            continue
        if lag >= 0:
            g_val = dx * np.sum(h[:n-p] * (1.0 - h[p:]))
        else:
            g_val = dx * np.sum(h[p:] * (1.0 - h[:n-p]))
        g = grad_lag(h, p, n)
        cons.append(g_val + g @ (h_var - h) <= t)
    
    prob = cp.Problem(obj, cons)
    
    for solver in [cp.OSQP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None and np.all(np.isfinite(h_var.value)):
                return np.asarray(h_var.value, dtype=np.float64)
        except Exception:
            continue
    return None

    lags_arr = compute_lag_values(h)
