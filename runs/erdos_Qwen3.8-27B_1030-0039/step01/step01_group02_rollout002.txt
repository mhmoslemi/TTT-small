**Subgradient method for the max of bilinear functions:**

f(h) = max_k g_k(h) where g_k(h) = dx * sum_{valid i} h[i]*(1-h[i+k]) (or equivalent)

A subgradient of f at h is: any convex combination of the gradients of the active g_k's.

The gradient of g_k with respect to h[i] is:
- If i is in the "first" position: dx * (1 - h[i+k]) for the pair (i, i+k)
- If i is in the "second" position: -dx * h[i-k] for the pair (i-k, i)
- Could be both if i appears in multiple pairs at the same lag

More precisely, for a given lag k (positive), g_k(h) = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
d g_k / d h[j] = dx * [sum_{i: i=j} (1-h[i+k]) - sum_{i: i+k=j} h[i]]
= dx * [(1-h[j+k]) * 1_{j ≤ n-1-k} - h[j-k] * 1_{j ≥ k}]

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    if n == 1:
        val = min(max(target, 0.0), 1.0)
        return np.array([val]), True
    
    def f_offset(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    
    lo = -1.0 - np.max(h)
    hi = 1.0 - np.min(h)
    
    # Ensure brackets
    flo = f_offset(lo)
    fhi = f_offset(hi)
    if flo > 0:
        return np.clip(h + lo, 0.0, 1.0), True
    if fhi < 0:
        return np.clip(h + hi, 0.0, 1.0), True
    
    try:
        t = brentq(f_offset, lo, hi, xtol=1e-15, rtol=1e-15, maxiter=200)
    except Exception:
        t = brentq(f_offset, lo, hi, xtol=1e-13, maxiter=200)
    
    result = np.clip(h + t, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result, True

def smooth_obj_and_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    m = len(f)
    
    # Compute softmax weights, optionally over top-k
    y = beta * f
    y_max = np.max(y)
    w = np.exp(y - y_max)
    Z = np.sum(w)
    
    if top_k is not None and top_k < m:
        # Only use top-k outputs
        idx = np.argpartition(f, -top_k)[-top_k:]
        w[idx < top_k] = 0  # wrong, need to zero out non-top
        # Actually, zero out non-top indices
        top_mask = np.zeros(m, dtype=bool)
        top_mask[idx] = True
        w[~top_mask] = 0
        Z = np.sum(w)
    
    p = w / Z
    psi = (y_max + np.log(Z)) / beta
    
    grad = np.zeros(n)
    for k in range(m):
        weight = p[k] * dx
        if weight < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            upper = n - lag
            if upper > 0:
                grad[:upper] += weight * (-h[lag:n])
            if lag <= n - 1:

            upper = n - pn
            if upper > 0:
                grad[:upper] += weight * (1.0 - h[pn:n])
            if pn <= n - 1:
                grad[pn:] += weight * (-h[:upper])
    
    return psi, grad
