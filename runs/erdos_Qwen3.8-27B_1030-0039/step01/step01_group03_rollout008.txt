def solve_tangent_lp(h, active_grads, near_grads, near_slacks, r, eta, delta_max):
    N = h.size
    n_active = len(active_grads)
    n_near = len(near_grads)
    
    # Variables: d[0..N-1], delta
    # Objective: minimize -delta → c = [0,...,0, -1]
    c = np.zeros(N + 1)
    c[-1] = -1.0
    
    # Inequality constraints (A_ub @ x <= b_ub):
    # For each active j: G_j · d + delta <= 0 → [G_j, 1] @ [d, delta] <= 0
    # For each near j: G_j · d + eta*delta <= slack_j → [G_j, eta] @ [d, delta] <= slack_j
    # delta >= 0: -delta <= 0 → [0, -1] @ [d, delta] <= 0
    
    rows = []
    rhs = []
    for j in range(n_active):
        row = np.zeros(N + 1)
        row[:N] = active_grads[j]
        row[N] = 1.0
        rows.append(row)
        rhs.append(0.0)
    for j in range(n_near):
        row = np.zeros(N + 1)
        row[:N] = near_grads[j]
        row[N] = eta
        rows.append(row)
        rhs.append(near_slacks[j])
    # delta >= 0
    row = np.zeros(N + 1)
    row[N] = -1.0
    rows.append(row)
    rhs.append(0.0)
    
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(N):
        r_i = min(r, h[i], 1.0 - h[i])
        bounds.append((-r_i, r_i))
    bounds.append((0.0, delta_max))
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        d = result.x[:N]
        delta = result.x[N]
        # Fix sum(d) to be exactly 0
        d -= np.mean(d)
        return d, float(delta)
    return None, 0.0

    c_obj[N] = -1.0
    n_ineq = n_a + n_n + 1
    A_ub = np.zeros((n_ineq, n_vars))
    b_ub = np.zeros(n_ineq)
    for j in range(n_a):
        A_ub[j, :N] = dx * active_grads[j]
        A_ub[j, N] = 1.0
    for j in range(n_a, n_a + n_n):
        A_ub[j, :N] = dx * near_grads[j - n_a]
        A_ub[j, N] = eta
        b_ub[j] = near_slacks[j - n_a]
    A_ub[-1, N] = -1.0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(N):
        r_i = min(r, h[i] + 1e-12, 1.0 - h[i] + 1e-12)
        r_i = max(0.0, r_i)
        bounds.append((-r_i, r_i))
    bounds.append((0.0, delta_max))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return None, 0.0
    if res.success and np.isfinite(res.x[N]):
        d = res.x[:N].copy()
        d -= np.mean(d)
        return d, float(res.x[N])
    return None, 0.0


def tangent_descent(h, c, t0, budget, reserve, max_iters=20000):
    N = h.size
    dx = 2.0 / float(N)
    best_h = h.copy()
    best_c = c
    r = 0.05
    stall_count = 0
    max_stall = 30
    for it in range(max_iters):
        if remaining_budget(t0, budget) < reserve + 5.0:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        M = float(np.max(corr))
        active_tol = max(1e-7, 3e-4 * M)
        active_mask = corr >= M - active_tol
        active_indices = np.where(active_mask)[0]
        near_tol = 8.0 * active_tol
        near_mask = (corr >= M - near_tol) & (~active_mask)
        near_indices = np.where(near_mask)[0]
        if near_indices.size > 25:
            nv = corr[near_indices]
            top_k = np.argpartition(nv, -25)[-25:]
            near_indices = near_indices[top_k]
        n_a = active_indices.size
        if n_a == 0:
            break
        if remaining_budget(t0, budget) < reserve + 2.0:
            break
        active_grads = np.array([gradient_for_output_index(h, idx) for idx in active_indices])
        if near_indices.size > 0:
            near_grads = np.array([gradient_for_output_index(h, idx) for idx in near_indices])
            near_slacks = (M - corr[near_indices]).astype(float)
        else:
            near_grads = np.zeros((0, N))
            near_slacks = np.zeros(0)
        d, delta = solve_tangent_lp(h, active_grads, near_grads, near_slacks, r, 0.2, 0.02, dx)
        if d is None:
            stall_count += 1
            if stall_count > max_stall:
                break
            r *= 0.7
            if r < 1e-5:
                break
            continue
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-12 or delta < 5e-7:
            stall_count += 1
            if stall_count > max_stall:
                break
            r *= 0.7
            if r < 1e-5:
                break
            continue
        stall_count = 0
        pos = d > 1e-14
        neg = d < -1e-14
        alpha_max_step = 1.0
        if np.any(pos):
            a_candidates = (1.0 - h[pos]) / d[pos]
            alpha_max_step = min(alpha_max_step, float(np.min(a_candidates)))
        if np.any(neg):
            a_candidates = h[neg] / (-d[neg])
            alpha_max_step = min(alpha_max_step, float(np.min(a_candidates)))
        alpha_max_step = max(1e-8, min(1.0, alpha_max_step))
        factors = [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
        accepted = False
        used_alpha = 0.0
        for factor in factors:
            al = alpha_max_step * factor
            if al < 1e-10:
                break
            cand = h + al * d
            cand = make_feasible(cand)
            cand_c = exact_c5(cand)
            if np.isfinite(cand_c) and cand_c < best_c - 1e-12:
                best_c = cand_c
                best_h = cand.copy()
                h = cand.copy()
                accepted = True
                used_alpha = al
                break
        if accepted:
            if used_alpha > 0.5 * alpha_max_step:
                r = min(r * 1.15, 0.25)
            else:
                r = max(r * 0.9, 1e-5)
        else:
            r *= 0.55
            if r < 1e-6:
                break
    return best_h, best_c


def pair_local_search(h, c, rng, max_evals, t0, budget, reserve):
    best_h = h.copy()
    best_c = c
    N = h.size
    evals = 0
    max_evals = int(max_evals)
    fractions = [0.5, 0.3, 0.15, 0.07, 0.03, 0.01, 0.005]
    while evals < max_evals and remaining_budget(t0, budget) > reserve + 3.0:
        i = int(rng.integers(N))
        j = int(rng.integers(N))
        if i == j:
            continue
        hi_val = float(best_h[i])
        hj_val = float(best_h[j])
        max_pos = min(1.0 - hi_val, hj_val)
        max_neg = min(hi_val, 1.0 - hj_val)
        if max_pos < 1e-14 and max_neg < 1e-14:
            continue
        found = False
        for frac in fractions:
            dp = frac * max_pos
            if dp > 1e-15:
                cand = best_h.copy()
                cand[i] += dp
                cand[j] -= dp
                cand = make_feasible(cand)
                cc = exact_c5(cand)
                evals += 1
                if np.isfinite(cc) and cc < best_c - 1e-13:
                    best_c = cc
                    best_h = cand
                    found = True
                    break
            dn = frac * max_neg
            if dn > 1e-15:
                cand = best_h.copy()
                cand[i] -= dn
                cand[j] += dn
                cand = make_feasible(cand)
                cc = exact_c5(cand)
                evals += 1
                if np.isfinite(cc) and cc < best_c - 1e-13:
                    best_c = cc
                    best_h = cand
                    found = True
                    break
            if evals >= max_evals:
                break
            if remaining_budget(t0, budget) < reserve + 3.0:
                break
    return best_h, best_c


def transfer_length(h, new_N):
    h = np.asarray(h, dtype=float).reshape(-1)
    new_N = int(new_N)
    if h.size == new_N:
        return make_feasible(h)
    if h.size == 0 or new_N <= 0:
        return np.full(max(1, new_N), 0.5)
    if new_N % h.size == 0:
        factor = new_N // h.size
        y = np.repeat(h, factor)
        return make_feasible(y)
    if h.size % new_N == 0:
        factor = h.size // new_N
        y = h.reshape(new_N, factor).mean(axis=1)
        return make_feasible(y)
    src = (np.arange(h.size, dtype=float) + 0.5) * (2.0 / float(h.size))
    dst = (np.arange(new_N, dtype=float) + 0.5) * (2.0 / float(new_N))
    y = np.interp(dst, src, h, left=float(h[0]), right=float(h[-1]))
    return make_feasible(y)


def perturb_feasible(h, rng, sigma):
    h = np.asarray(h, dtype=float).reshape(-1)
    x = h + float(sigma) * rng.normal(0.0, 1.0, h.size)
    return make_feasible(x)


def cosine_feasible(N, mode, amp):
    t = (np.arange(N, dtype=float) + 0.5) / float(N)
    x = 0.5 + amp * np.cos(2.0 * np.pi * float(mode) * t)
    return make_feasible(x)


def finalize_solution(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
        h = np.array([0.5], dtype=float)
    h = make_feasible(h)
    c = exact_c5(h)
    if not np.isfinite(c):
        h = make_feasible(np.full(h.size, 0.5))
        c = exact_c5(h)
    h = np.clip(h, 0.0, 1.0)
    h = make_feasible(h)
    c = exact_c5(h)
    if not np.isfinite(c):
        h = make_feasible(np.full(h.size, 0.5))
        c = exact_c5(h)
    return h, float(c), int(h.size)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    budget = float(budget_s)
    reserve = min(30.0, max(5.0, 0.05 * budget))
    rng = np.random.default_rng(int(seed))

    parent = get_parent_values()
    if parent is None:
        N0 = 135
        h0 = cosine_feasible(N0, 3, 0.3)
    else:
        h0 = make_feasible(parent)
        N0 = int(h0.size)

    best_h = h0.copy()
    best_c = exact_c5(best_h)
    if not np.isfinite(best_c):
        best_h = make_feasible(np.full(max(1, N0), 0.5))
        best_c = exact_c5(best_h)

    grad_ok = False
    if remaining_budget(t0, budget) > reserve + 5.0:
        try:
            grad_ok = check_gradient_correctness()
        except Exception:
            grad_ok = False

    if remaining_budget(t0, budget) > reserve + 60.0:
        h_t, c_t = tangent_descent(best_h, best_c, t0, budget, reserve, 20000)
        if np.isfinite(c_t) and c_t < best_c - 1e-14:
            best_c = c_t
            best_h = h_t.copy()

    if remaining_budget(t0, budget) > reserve + 30.0:
        pevals = int(max(2000, min(20000, 500000.0 / float(best_h.size))))
        h_p, c_p = pair_local_search(best_h, best_c, rng, pevals, t0, budget, reserve)
        if np.isfinite(c_p) and c_p < best_c - 1e-14:
            best_c = c_p
            best_h = h_p.copy()

    if remaining_budget(t0, budget) > reserve + 200.0:
        pert = perturb_feasible(best_h, rng, 0.03)
        c_pt = exact_c5(pert)
        h_t, c_t = tangent_descent(pert, c_pt, t0, budget, reserve, 15000)
        if np.isfinite(c_t) and c_t < best_c - 1e-14:
            best_c = c_t
            best_h = h_t.copy()
        if remaining_budget(t0, budget) > reserve + 20.0:
            h_p, c_p = pair_local_search(best_h, best_c, rng, 5000, t0, budget, reserve)
            if np.isfinite(c_p) and c_p < best_c - 1e-14:
                best_c = c_p
                best_h = h_p.copy()

    if remaining_budget(t0, budget) > reserve + 250.0:
        cos_start = cosine_feasible(N0, 2, 0.35)
        c_cs = exact_c5(cos_start)
        h_t, c_t = tangent_descent(cos_start, c_cs, t0, budget, reserve, 15000)
        if np.isfinite(c_t) and c_t < best_c - 1e-14:
            best_c = c_t
            best_h = h_t.copy()
        if remaining_budget(t0, budget) > reserve + 20.0:
            h_p, c_p = pair_local_search(best_h, best_c, rng, 5000, t0, budget, reserve)
            if np.isfinite(c_p) and c_p < best_c - 1e-14:
                best_c = c_p
                best_h = h_p.copy()

    N2 = 2 * N0
    if N2 <= 300 and remaining_budget(t0, budget) > reserve + 250.0:
        st2 = transfer_length(best_h, N2)
        c2 = exact_c5(st2)
        if np.isfinite(c2) and c2 < best_c - 1e-14:
            best_c = c2
            best_h = st2.copy()
        h_t, c_t = tangent_descent(st2, c2, t0, budget, reserve, 15000)
        if np.isfinite(c_t) and c_t < best_c - 1e-14:
            best_c = c_t
            best_h = h_t.copy()
        if remaining_budget(t0, budget) > reserve + 30.0:
            h_p, c_p = pair_local_search(best_h, best_c, rng, 8000, t0, budget, reserve)
            if np.isfinite(c_p) and c_p < best_c - 1e-14:
                best_c = c_p
                best_h = h_p.copy()
        back = transfer_length(best_h, N0)
        cb = exact_c5(back)
        if np.isfinite(cb) and cb < best_c - 1e-14:
            best_c = cb
            best_h = back.copy()

    N3 = 3 * N0
    if N3 <= 360 and remaining_budget(t0, budget) > reserve + 200.0:
        st3 = transfer_length(best_h, N3)
        c3 = exact_c5(st3)
        h_t, c_t = tangent_descent(st3, c3, t0, budget, reserve, 10000)
        if np.isfinite(c_t) and c_t < best_c - 1e-14:
            best_c = c_t
            best_h = h_t.copy()
        back = transfer_length(best_h, N0)
        cb = exact_c5(back)
        if np.isfinite(cb) and cb < best_c - 1e-14:
            best_c = cb
            best_h = back.copy()

    if remaining_budget(t0, budget) > reserve + 100.0:
        h_t, c_t = tangent_descent(best_h, best_c, t0, budget, reserve, 10000)
        if np.isfinite(c_t) and c_t < best_c - 1e-14:
            best_c = c_t
            best_h = h_t.copy()

    if remaining_budget(t0, budget) > reserve + 30.0:
        pevals = int(max(3000, min(30000, 600000.0 / float(best_h.size))))
        h_p, c_p = pair_local_search(best_h, best_c, rng, pevals, t0, budget, reserve)
        if np.isfinite(c_p) and c_p < best_c - 1e-14:
            best_c = c_p
            best_h = h_p.copy()

    if remaining_budget(t0, budget) > reserve + 20.0:
        pert2 = perturb_feasible(best_h, rng, 0.01)
        c_p2 = exact_c5(pert2)
        if c_p2 < best_c:
            best_c = c_p2
            best_h = pert2.copy()
        h_t, c_t = tangent_descent(best_h, best_c, t0, budget, reserve, 3000)
        if np.isfinite(c_t) and c_t < best_c - 1e-14:
            best_c = c_t
            best_h = h_t.copy()

    if remaining_budget(t0, budget) > reserve + 10.0:
        pevals = int(max(2000, min(15000, 300000.0 / float(best_h.size))))
        h_p, c_p = pair_local_search(best_h, best_c, rng, pevals, t0, budget, reserve)
        if np.isfinite(c_p) and c_p < best_c - 1e-14:
            best_c = c_p
            best_h = h_p.copy()

    return finalize_solution(best_h)
