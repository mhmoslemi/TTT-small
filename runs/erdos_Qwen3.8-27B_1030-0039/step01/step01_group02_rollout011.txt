def score_exact(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def is_feasible(h, tol=1e-8):

def smooth_obj_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    m_out = len(f)
    
    if top_k is not None and top_k < m_out:
        # Select top_k outputs
        idx = np.argpartition(-f, top_k)[:top_k]
        # Degenerate case: all same value
        if np.max(f[idx]) - np.min(f[idx]) < 1e-15:
            # All weights equal
            active_vals = f[idx]
            active_idx = idx
        else:
            active_vals = f[idx]
            active_idx = idx
    else:
        active_vals = f
        active_idx = np.arange(m_out)
    
    # Smooth max with top-k
    y = beta * active_vals
    m = np.max(y)
    w = np.exp(y - m)
    Z = np.sum(w)
    psi = (m + np.log(Z)) / beta
    p = w / Z
    
    grad = np.zeros(n)
    for ii, k in enumerate(active_idx):
        weight = p[ii] * dx
        if weight < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            upper = n - lag
            if upper > 0:
                grad[:upper] += weight * (-h[lag:n])
            if lag <= n - 1:
                grad[lag:] += weight * (1.0 - h[:upper])

            if upper > 0:
                grad[:upper] += weight * (1.0 - h[pn:n])
            if pn <= n - 1:
                grad[pn:] += weight * (-h[:upper])
    
    return psi, grad

def sa_binary_search(h_start, n, iterations, t_start, deadline, reserve_s=5.0):
    """Simulated annealing on binary vectors."""
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    
    # Initialize from h_start: select top n/2 positions
    target = n // 2
    if target <= 0 or target >= n:
        return None
    
    # Sort by value, pick top n/2
    order = np.argsort(-h_start)
    bits = np.zeros(n, dtype=np.int8)
    bits[order[:target]] = 1
    
    cur_score = binary_objective(bits)
    best_bits = bits.copy()
    best_score = cur_score
    
    # Temperature schedule
    t0 = 0.15
    t_end = 0.001
    time_limit = max(1.0, deadline - reserve_s - time.time())
    
    it = 0
    while it < iterations:
        if time.time() > deadline - reserve_s:
            break
        
        # Check time every 500 iterations
        if it % 500 == 0 and it > 0:
            if time.time() > deadline - reserve_s:
                break
            # Estimate remaining iterations
            elapsed = time.time() - t_start
            if elapsed > 0.1:
                rate = it / elapsed
                remaining_time = deadline - reserve_s - time.time()
                if rate > 0 and remaining_time / rate < 100:
                    break
        
        # Temperature
        frac = it / max(1, iterations)
        temp = t0 * (t_end / t0) ** frac
        
        # Propose move
        ones = np.where(bits == 1)[0]
        zeros = np.where(bits == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        if r < 0.6:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            bits[i] = 0
            bits[j] = 1
            new_score = binary_objective(bits)
            delta = new_score - cur_score
            if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_bits = bits.copy()
            else:
                bits[i] = 1
                bits[j] = 0
        else:
            # Block move: pick a window, keep same number of ones
            block_size = min(rng.randint(4, 16), n - 1)
            start = rng.randint(0, n - block_size + 1)
            end = start + block_size
            window = bits[start:end]
            k_ones = np.sum(window)
            if k_ones == 0 or k_ones == block_size:
                # Can't change anything
                it += 1
                continue
            # Randomly redistribute
            perm = rng.permutation(block_size)
            new_window = np.zeros(block_size, dtype=np.int8)
            new_window[perm[:k_ones]] = 1
            if np.array_equal(window, new_window):
                it += 1
                continue
            bits[start:end] = new_window
            new_score = binary_objective(bits)
            delta = new_score - cur_score
            if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_bits = bits.copy()
            else:
                bits[start:end] = window
        
        it += 1
    
    if best_bits is not None and best_score < 0.5:
        return best_bits, best_score
    return None

def coordinate_descent_exact(h, n_rounds=50, n_samples=13, tol=1e-13, deadline=None, reserve=10.0):
    """Exact pair coordinate descent."""
    n = len(h)
    h = h.copy()
    if n < 2:
        return h, score_exact(h)
    
    best_score = score_exact(h)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.time() > deadline - reserve:
            break
        
        improved_any = False
        if n <= 60:
            indices = list(range(n))
        else:
            rng_local = np.random.RandomState(rnd * 31 + 7)
            indices = rng_local.permutation(n).tolist()
        
        for ii in range(len(indices)):
            if deadline is not None and (ii % 20 == 0) and time.time() > deadline - reserve:
                break
            i = indices[ii]
            for jj in range(ii + 1, len(indices)):
                j = indices[jj]
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-14:
                    continue
                deltas = np.linspace(lo, hi, n_samples)
                best_d = 0.0
                best_s = best_score
                for d in deltas:
                    if abs(d) < 1e-14:
                        continue
                    h[i] += d
                    h[j] -= d
                    s = score_exact(h)
                    if s < best_s - tol:
                        best_s = s
                        best_d = d
                h[i] -= best_d
                h[j] += best_d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    best_score = best_s
                    improved_any = True
        
        if not improved_any:
            break
    
    return h, best_score

def score_exact(h):
    n = len(h)
    if n < 1:
        return 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, -10.0, 10.0)
    
    def _f(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    
    lo = -float(np.max(h)) - 1.0
    hi = 1.0 - float(np.min(h)) + 1.0
    
    flo = _f(lo)
    fhi = _f(hi)
    
    if flo >= 0:
        result = np.clip(h + lo, 0.0, 1.0)
    elif fhi <= 0:
        result = np.clip(h + hi, 0.0, 1.0)
    else:
        try:
            t = brentq(_f, lo, hi, xtol=1e-14, rtol=1e-14, maxiter=300)

                if _f(mid) > 0:
                    lo = mid
                else:
                    hi = mid

            result = np.clip(result, 0.0, 1.0)

def sum_constraint_fun(x, target):
    return np.sum(x) - target

    return np.ones(len(x))

def smooth_obj_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    m_out = len(f)
    
    if top_k is not None and top_k < m_out:
        idx = np.argpartition(-f, top_k)[:top_k]
        active_vals = f[idx]
        active_idx = idx
    else:
        active_vals = f
        active_idx = np.arange(m_out)
    
    y = beta * active_vals
    m = np.max(y)
    w = np.exp(y - m)
    Z = np.sum(w)
    psi = (m + np.log(Z)) / beta
    p = w / Z
    
    grad = np.zeros(n)
    for ii in range(len(active_idx)):
        k = active_idx[ii]
        weight = p[ii] * dx
        if weight < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            upper = n - lag
            if upper > 0:
                grad[:upper] += weight * (-h[lag:n])
            if lag <= n - 1:
                grad[lag:] += weight * (1.0 - h[:upper])
        else:
            pn = -lag
            upper = n - pn
            if upper > 0:
                grad[:upper] += weight * (1.0 - h[pn:n])
            if pn <= n - 1:
                grad[pn:] += weight * (-h[:upper])
    
    return psi, grad

def transfer_to_resolution(h_old, n_new):

    x_new = np.linspace(0.0, 2.0, n_new)
