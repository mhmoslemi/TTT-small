        return float(np.sum(np.clip(h_raw + t, 0.0, 1.0))) - target

        return h_low

        h = np.clip(h_raw + t, 0.0, 1.0)

import numpy as np
from scipy.optimize import linprog, brentq
import time

class Deadline:
    def __init__(self, t):
        self.t = t
    def remaining(self):
        return max(0.0, self.t - time.time())
    def expired(self):
        return time.time() >= self.t

def project_feasible(h_raw, n):
    target = n / 2.0
    def _f(t):
        return float(np.sum(np.clip(h_raw + t, 0.0, 1.0))) - target
    h_clipped = np.clip(h_raw, 0.0, 1.0)
    if abs(float(np.sum(h_clipped)) - target) < 1e-10:
        return h_clipped
    lo = float(np.min(h_raw)) - 1.5
    hi = 1.5 - float(np.max(h_raw))
    flo = _f(lo)
    fhi = _f(hi)
    if flo * fhi <= 0.0:
        t = brentq(_f, lo, hi, xtol=1e-15, rtol=1e-15, maxiter=200)
        h = np.clip(h_raw + t, 0.0, 1.0)
    else:
        h = h_clipped.copy()
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lag(h, d):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += (-h[d:])
    else:
        p = -d
        if p < n:
            g[:n - p] += (1.0 - h[p:])
            g[p:] += (-h[:n - p])
    return g * dx

def lp_direction(h, active_lags, limit):
    n = len(h)
    K = len(active_lags)
    if K == 0:
        return None, 0.0
    grads = np.zeros((K, n))
    for i, m in enumerate(active_lags):
        grads[i] = grad_for_lag(h, m - (n - 1))
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-1.0, 1.0)] * n + [(-5.0, 5.0)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, 0.0
    if res.status != 0 or res.x is None:
        return None, 0.0
    delta = res.x[:n]
    z_val = float(res.x[n])
    if z_val >= -1e-8:
        return None, z_val
    max_d = float(np.max(np.abs(delta)))
    if max_d < 1e-12:
        return None, z_val
    delta = delta / max_d
    return delta, z_val

def line_search(h, direction, current_score, limit):
    n = len(h)
    target = n / 2.0
    pos_mask = direction > 1e-12
    neg_mask = direction < -1e-12
    alpha_max = 0.5
    if np.any(pos_mask):
        cands = (1.0 - h[pos_mask]) / direction[pos_mask]
        alpha_max = min(alpha_max, float(np.min(cands)))
    if np.any(neg_mask):
        cands = h[neg_mask] / (-direction[neg_mask])
        alpha_max = min(alpha_max, float(np.min(cands)))
    if alpha_max < 1e-10 or not np.isfinite(alpha_max):
        return current_score, h.copy()
    best_score = current_score
    best_alpha = 0.0
    n_grid = 25
    alphas = np.linspace(0.0, alpha_max, n_grid)
    for a in alphas:
        if a < 1e-12:
            continue
        h_t = h + a * direction
        h_t = np.clip(h_t, 0.0, 1.0)
        s = float(np.sum(h_t))
        if abs(s - target) > 1e-10:
            h_t = project_feasible(h_t, n)
        sc = exact_score(h_t)
        if sc < best_score - 1e-14:
            best_score = sc
            best_alpha = a
    if best_alpha > 0.0:
        lo = max(0.0, best_alpha - alpha_max / n_grid * 1.5)
        hi = min(alpha_max, best_alpha + alpha_max / n_grid * 1.5)
        alphas2 = np.linspace(lo, hi, 15)
        for a in alphas2:
            if a < 1e-12:
                continue
            h_t = h + a * direction
            h_t = np.clip(h_t, 0.0, 1.0)
            s = float(np.sum(h_t))
            if abs(s - target) > 1e-10:
                h_t = project_feasible(h_t, n)
            sc = exact_score(h_t)
            if sc < best_score - 1e-14:
                best_score = sc
                best_alpha = a
    if best_alpha > 0.0:
        h_new = h + best_alpha * direction
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        final_sc = exact_score(h_new)
        return min(best_score, final_sc), h_new
    return current_score, h.copy()

def lp_descent(h, n, limit, max_iters, K_frac=0.15, tol_frac=0.03):
    h_cur = h.copy()
    score_cur = exact_score(h_cur)
    best_h = h_cur.copy()
    best_score = score_cur
    no_improve = 0
    for it in range(max_iters):
        if limit.remaining() < 3.0:
            break
        if no_improve > 80:
            break
        corrs = compute_corrs(h_cur)
        C_val = float(np.max(corrs))
        tol = max(1e-5, tol_frac * C_val)
        active_mask = corrs >= (C_val - tol)
        active_lags = np.where(active_mask)[0]
        K = max(6, min(25, int(K_frac * (2 * n - 1))))
        if len(active_lags) > K:
            order = np.argsort(-corrs[active_lags])
            active_lags = active_lags[order[:K]]
        direction, z_val = lp_direction(h_cur, list(active_lags), limit)
        if direction is not None:
            new_score, new_h = line_search(h_cur, direction, score_cur, limit)
            if new_score < score_cur - 1e-11:
                h_cur = new_h
                score_cur = new_score
                if score_cur < best_score:
                    best_h = h_cur.copy()
                    best_score = score_cur
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 20:
                break
    return best_h, best_score

def coord_descent(h, n, limit, max_rounds, rng):
    h_cur = h.copy()

    best_h = h_cur.copy()
    best_score = score_cur
    no_improve = 0
    corrs_cache = compute_corrs(h_cur)

        if no_improve > 150:
            break

        C_val = float(np.max(corrs_cache))

        grad_active = np.zeros(n)
        for m in active_lags[:12]:

        if rnd < 40:

        improved = False
        for (i, j) in pairs:
            if limit.remaining() < 2.0:
                break
            for direction in (1, -1):
                if direction == 1:
                    lo_t = max(-h_cur[j], -0.4)
                    hi_t = min(1.0 - h_cur[i], 0.4)

                if hi_t - lo_t < 1e-12:

                arr_t = np.linspace(lo_t, hi_t, 9)

                    else:
                        h_t[i] -= tv
                        h_t[j] += tv
                    h_t = np.clip(h_t, 0.0, 1.0)

                    if sc < best_ls:

                        best_t = tv
                if best_t != 0.0 and best_ls < score_cur - 1e-11:
                    if direction == 1:
                        h_cur[i] += best_t
                        h_cur[j] -= best_t
                    else:
                        h_cur[i] -= best_t
                        h_cur[j] += best_t
                    h_cur = np.clip(h_cur, 0.0, 1.0)

                    score_cur = exact_score(h_cur)
                    if score_cur < best_score:

                    improved = True

        if not improved:

    return best_h, best_score

def transfer_resolution(h_old, new_n):
    old_n = len(h_old)
    if old_n == new_n:
        return project_feasible(h_old.copy(), new_n)
    old_x = np.arange(old_n) / old_n
    new_x = np.arange(new_n) / new_n
    h_new = np.interp(new_x, old_x, h_old)
    return project_feasible(h_new, new_n)

def make_variants(h, n, rng):
    variants = []
    variants.append(h.copy())
    variants.append(1.0 - h)
    variants.append(h[::-1].copy())
    for shift in (1, 2, 3, 7, 11, 13):
        if shift < n:
            variants.append(np.roll(h, shift))
        if -shift > -n:
            variants.append(np.roll(h, -shift))
    for sigma in (0.05, 0.1, 0.15, 0.25):
        v = h + sigma * rng.randn(n)
        variants.append(project_feasible(v, n))
    fourier_v = np.ones(n) * 0.5
    for k in range(1, 5):
        a = rng.randn() * 0.3
        b = rng.randn() * 0.3
        fourier_v += a * np.sin(2 * np.pi * k * np.arange(n) / n)
        fourier_v += b * np.cos(2 * np.pi * k * np.arange(n) / n)
    variants.append(project_feasible(fourier_v, n))
    binary_v = (rng.rand(n) < 0.5).astype(float)
    variants.append(project_feasible(binary_v, n))
    return variants
