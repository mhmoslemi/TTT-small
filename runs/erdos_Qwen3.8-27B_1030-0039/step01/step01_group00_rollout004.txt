def project_feasible(h_raw, n):
    target = n / 2.0
    h_clip = np.clip(h_raw, 0.0, 1.0)
    current_sum = np.sum(h_clip)
    if abs(current_sum - target) < 1e-12:
        return h_clip
    # Water-filling: add shift t, clip
    lo = -max(np.max(h_raw), target / n) - 2.0
    hi = 1.0 - min(np.min(h_raw), 1.0 - target / n) + 2.0
    def _f(t):
        return float(np.sum(np.clip(h_raw + t, 0.0, 1.0))) - target
    lo_v = _f(lo)

    _state['n'] = best_n

    h_c = best_h.copy()
    s_c = best_score

    if _remaining() > 40:
        h_c, s_c = smooth_optimize(h_c, best_n, [5.0, 20.0, 60.0, 200.0, 600.0, 1500.0], 200)
        if s_c < best_score:
            best_h, best_score, best_n = h_c.copy(), s_c, best_n

    if _remaining() > 60:
        h_a, s_a = active_set_optimize(best_h.copy(), best_n, top_k=5, max_rounds=40, slsqp_maxiter=300)
        if s_a < best_score:
            best_h, best_score, best_n = h_a.copy(), s_a, best_n

    if _remaining() > 30:
        h_cd, s_cd = coord_descent(best_h.copy(), best_n, max_rounds=300, top_k=5)
        if s_cd < best_score:
            best_h, best_score, best_n = h_cd.copy(), s_cd, best_n

    if _remaining() > 40:
        h_a2, s_a2 = active_set_optimize(best_h.copy(), best_n, top_k=7, max_rounds=30, slsqp_maxiter=250)
        if s_a2 < best_score:
            best_h, best_score, best_n = h_a2.copy(), s_a2, best_n

    for sigma in (0.05, 0.10, 0.20, 0.35):
        if _remaining() < 30:
            break
        pert = best_h + sigma * rng.randn(best_n)
        pert = project_feasible(pert, best_n)
        hp, sp = smooth_optimize(pert, best_n, [10.0, 50.0, 200.0, 600.0], 150)
        if _remaining() > 15:
            hp, sp = active_set_optimize(hp, best_n, top_k=5, max_rounds=20, slsqp_maxiter=200)
        if sp < best_score:
            best_h, best_score, best_n = hp.copy(), sp, best_n

    if _remaining() > 25:
        comp = 1.0 - best_h
        comp = project_feasible(comp, best_n)
        hc, sc = smooth_optimize(comp, best_n, [20.0, 100.0, 400.0], 100)
        if sc < best_score:
            best_h, best_score, best_n = hc.copy(), sc, best_n

    for fs in range(3):
        if _remaining() < 30:
            break
        h_f = _fourier_init(best_n, seed + 100 * fs, D=5)
        hf, sf = smooth_optimize(h_f, best_n, [5.0, 30.0, 100.0, 400.0], 150)
        if _remaining() > 10:
            hf, sf = active_set_optimize(hf, best_n, top_k=5, max_rounds=15, slsqp_maxiter=200)
        if sf < best_score:
            best_h, best_score, best_n = hf.copy(), sf, best_n

    for tgt_n in sorted(set([best_n * 2, best_n * 3, 90, 120, 150])):
        if _remaining() < 60:
            break
        if tgt_n <= best_n or tgt_n > 200:

        hn, sn = smooth_optimize(ht, tgt_n, [5.0, 20.0, 60.0, 200.0, 600.0], 150)

            hn, sn = active_set_optimize(hn, tgt_n, top_k=5, max_rounds=25, slsqp_maxiter=200)
        if _remaining() > 10:
            hn, sn = coord_descent(hn, tgt_n, max_rounds=200, top_k=5)
        if sn < best_score:
            best_h, best_score, best_n = hn.copy(), sn, tgt_n

    if _remaining() > 20:
        _state['n'] = best_n
        h_cd2, s_cd2 = coord_descent(best_h.copy(), best_n, max_rounds=600, top_k=7)
        if s_cd2 < best_score:
            best_h, best_score, best_n = h_cd2.copy(), s_cd2, best_n

    if _remaining() > 15:
        h_a3, s_a3 = active_set_optimize(best_h.copy(), best_n, top_k=3, max_rounds=30, slsqp_maxiter=400)
        if s_a3 < best_score:
            best_h, best_score, best_n = h_a3.copy(), s_a3, best_n

    if _remaining() > 10:
        h_cd3, s_cd3 = coord_descent(best_h.copy(), best_n, max_rounds=400, top_k=3)
        if s_cd3 < best_score:
            best_h, best_score, best_n = h_cd3.copy(), s_cd2 if False else s_cd3

    for _ in range(3):
        if _remaining() < 5:
            break
        pert_f = best_h + 0.02 * rng.randn(best_n)
        pert_f = project_feasible(pert_f, best_n)
        hpf, spf = coord_descent(pert_f, best_n, max_rounds=300, top_k=5)
        if spf < best_score:
            best_h, best_score, best_n = hpf.copy(), spf, best_n

    _state['n'] = best_n
    best_h = project_feasible(best_h, best_n)
    best_score = evaluate_exact(best_h)

    return (best_h, best_score, best_n)
