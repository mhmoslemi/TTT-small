    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        c = corr * dx
        max_val = c.max()
        
        # Find active lags (within 1% of max)
        active = np.where(c >= max_val * 0.999)[0]
        
        # Compute gradient of the max (average gradient of active lags)
        grad = np.zeros(n)
        for j in active:
            d = j - (n - 1)
            if d == 0:
                grad += (1.0 - 2.0 * h_curr)
            elif d > 0:
                grad[:n-d] += -h_curr[d:]
                grad[d:] += (1.0 - h_curr[:n-d])
            else:
                k = -d
                grad[k:] += -h_curr[:n-k]
                grad[:n-k] += (1.0 - h_curr[k:])
        grad /= len(active)
        
        # Project gradient to maintain sum constraint
        grad -= np.mean(grad)
        
        # Barzilai-Borwein or simple line search
        # Try a few step sizes
        best_step_score = best_score
        best_step_h = best_h
        for alpha_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = alpha_mult * 0.01  # base step size

            h_try = project_feasible(h_try, target)

            if sc < best_step_score:
                best_step_score = sc
                best_step_h = h_try.copy()

            h_curr = best_step_h.copy()
            best_h = h_curr.copy()
            best_score = best_step_score

    n = len(h)

    best_h = h.copy()
    best_score = evaluate_exact(h)
    alpha = base_alpha
    no_improve_count = 0
    
    while time.time() < deadline:
        c = compute_corr(h_curr)
        max_val = c.max()
        
        # Active lags: within tolerance of max
        active = np.where(c >= max_val - 0.01)[0]
        if len(active) == 0:
            active = np.where(c == max_val)[0]
        
        # Average gradient of active lags
        grad = np.zeros(n)
        for j in active:
            d = j - (n - 1)
            grad += grad_single_lag(h_curr, d)
        grad /= len(active)
        
        # Project to maintain sum
        grad -= np.mean(grad)
        
        # Try the step
        h_new = project_feasible(h_curr - alpha * grad, target)
        new_score = evaluate_exact(h_new)
        
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
            h_curr = h_new.copy()
            no_improve_count = 0
            alpha = min(alpha * 1.1, 0.1)  # Increase step slightly
        else:
            no_improve_count += 1
            alpha *= 0.7
            if alpha < 1e-6:

                h_curr = h_curr + np.random.randn(n) * 0.02
                h_curr = project_feasible(h_curr, target)
                alpha = base_alpha
                no_improve_count = 0
            
            if no_improve_count > patience:
                break
    
    return best_h, best_score

def project_feasible(x, target_sum):

    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01

        g[d:] += 1.0 - h[:n-d]
    else:
        k = -d
        g[k:] -= h[:n-k]
        g[:n-k] += 1.0 - h[k:]
    return g

        grad += w * g
    return val, grad

def projected_gradient_method(h, deadline, beta=100.0, n_steps=5000):
    """Projected gradient with Barzilai-Borwein step size."""
    n = len(h)
    target = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    alpha = 0.01
    prev_grad = None
    prev_h = None
    prev_dx = None
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        val, grad = smooth_max_grad(h_curr, beta)
        grad -= np.mean(grad)  # project to maintain sum
        
        # BB step size
        if prev_grad is not None:
            s = h_curr - prev_h
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            if yy > 1e-14 and abs(sy) > 1e-14:
                alpha_bb = sy / yy
                alpha = max(0.001, min(0.1, alpha_bb))
        
        prev_h = h_curr.copy()
        prev_grad = grad.copy()
        
        h_new = project_feasible(h_curr - alpha * grad, target)
        new_score = evaluate_exact(h_new)
        
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
        
        h_curr = h_new
    
    return best_h, best_score

Hmm, but the BB step size with log-sum-exp might be tricky because the smoothed objective changes with h. Let me simplify and just use a backtracking line search.

Let me also think about the pair descent more carefully. The key issue is efficiency. For n=45, there are 990 pairs. For each pair, the golden section search takes ~30 function evaluations, each of which is O(n) to compute the correlation (since we only change 2 elements, we can update the correlation incrementally).

Wait, can we update the correlation incrementally? If we change h[i] by δ and h[j] by -δ, how does the correlation change?

out[k] = sum_m h[m]*(1-h[m+k])

If h[i] → h[i]+δ and h[j] → h[j]-δ:
- out[k] changes by: δ*(1-h[i+k]) - δ*(1-h[i-k]) (if i and j contribute to out[k])
  Plus: -h[i]*δ + ... (from the h[m+k] terms)
  
This is getting complicated. Let me just do the full O(n²) evaluation for each pair step. With n=45, that's 2025 operations per evaluation, and 30 evaluations per pair, so 60750 operations per pair. With 990 pairs, that's ~60M operations per sweep. In Python with numpy, that's maybe 1-2 seconds per sweep. We can do 100+ sweeps in a few minutes.

Actually, I realize that the golden section search for each pair is overkill. Since the objective as a function of the pair move is at most piecewise quadratic (the max of linear/quadratic functions), we can find the minimum more efficiently.

But for simplicity, let me just use the golden section search with fewer iterations (20 instead of 30).

1. Projected gradient with adaptive steps (for initial polish and high-res optimization)

def project_feasible(x, target_sum):
    n = len(x)
