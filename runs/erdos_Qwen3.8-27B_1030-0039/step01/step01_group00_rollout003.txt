from scipy.optimize import minimize, brentq
from scipy.special import softmax
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project(h_raw, n):
    target = n / 2.0
    s = np.sum(h_raw)
    if abs(s - target) < 1e-12:
        return np.clip(h_raw, 0.0, 1.0)
    # Binary search for shift
    lo = -max(h_raw) - 1.0
    hi = 1.0 - min(h_raw) + 1.0
    def _f(t):
        return np.sum(np.clip(h_raw + t, 0.0, 1.0)) - target
    flo = _f(lo)
    fhi = _f(hi)
    if flo * fhi > 0:
        # Fallback: uniform
        return np.full(n, 0.5)
    t = brentq(_f, lo, hi, xtol=1e-15, rtol=8.9e-16, maxiter=100)
    return np.clip(h_raw + t, 0.0, 1.0)

def _corr_and_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def _smooth_obj_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    bcorr = beta * corr
    m = np.max(bcorr)
    w = np.exp(bcorr - m)
    Z = np.sum(w)
    val = (m + np.log(Z)) / beta
    p = w / Z
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        pk = p[k] * beta * dx
        if abs(pk) < 1e-15:

        if d >= 0:
            if d < n:
                grad[d:] += pk * (1.0 - h[:n - d])
            if n - d > 0:
                grad[:n - d] += pk * (-h[d:])
        else:
            pp = -d
            if pp < n:
                grad[:n - pp] += pk * (1.0 - h[pp:])
            if n - pp > 0:
                grad[pp:] += pk * (-h[:n - pp])
    return val, grad

def _smooth_optimize(h0, n, betas, maxiter_per_beta, deadline):

    bounds = [(0.0, 1.0)] * n
