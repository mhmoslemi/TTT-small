def pair_line_search(h, i, j, n_iter=20, c0=None):
    """Golden section search on pair (i,j). Returns (h_new, c_new)."""
    N = h.size
    q = h[i] + h[j]
    lo = max(0.0, q - 1.0)
    hi = min(1.0, q)
    if hi - lo < 1e-12:
        return h, c0 if c0 is not None else exact_c5(h)
    
    if c0 is None:
        c0 = exact_c5(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    hc = h.copy()
    hc[i] = c
    hc[j] = q - c
    fc = exact_c5(hc)
    
    hd = h.copy()
    hd[i] = d
    hd[j] = q - d
    fd = exact_c5(hd)
    
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            hc[i] = c
            hc[j] = q - c
            fc = exact_c5(hc)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            hd[i] = d
            hd[j] = q - d
            fd = exact_c5(hd)
    
    # Check the best point
    best_f = min(fc, fd, c0)
    if best_f == c0:
        return h, c0
    elif best_f == fc:
        return hc, fc
    else:
        return hd, fd

def select_active_pairs(h, n_pairs=50):
    """Select pairs most likely to improve based on active shifts."""
    N = h.size
    overlaps = all_overlaps(h)
    c_max = float(np.max(overlaps))
    # Active shifts: those within 1% of max
    active_threshold = c_max * 0.99
    active_shifts = np.where(overlaps >= active_threshold)[0]
    
    pairs = set()
    for s in active_shifts:
        # For shift s, the terms are h[i](1-h[i+s]) for i=0,...,N-1-s
        # High-contribution terms have large h[i] and large (1-h[i+s])
        m = N - s

        top_idx = np.argpartition(contributions, -n_top)[-n_top:]
        for idx in top_idx:

def transfer_length(h, new_N):

    N = h.size

    y = np.interp(dst, src, h, left=h[0], right=h[-1])
    return make_feasible(y)
