def smooth_grad(x, beta, n, limit):
    limit.check()
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    bcorr = beta * corr
    m = np.max(bcorr)
    w = np.exp(bcorr - m)
    Z = np.sum(w)
    p = w / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = p[k] * dx
        if abs(wk) < 1e-15:
            continue
        if d >= 0:
            if d < n:
                grad[d:] += wk * (1.0 - x[:n-d])
                if n - d > 0:
                    grad[:n-d] += wk * (-x[d:])
        else:
            pp = -d
            if pp < n:
                grad[:n-pp] += wk * (1.0 - x[pp:])
                if n - pp > 0:
                    grad[pp:] += wk * (-x[:n-pp])
    return grad

# Smooth optimization with SLSQP
def smooth_optimize(h0, n, limit, betas, maxiter_per_beta):
    best_h = h0.copy()
    best_score = exact_score(h0)
    bounds = [(0.0, 1.0)] * n
    current = best_h.copy()
    for beta in betas:
        if limit.remaining() < 5.0:
            break
        # Define objective and gradient using beta
        def obj(x):
            return smooth_obj(x, beta, n, limit)
        def grad(x):
            return smooth_grad(x, beta, n, limit)
        
        def con_eq(x):
            return float(np.sum(x)) - n / 2.0
        def con_eq_grad(x):
            return np.ones(n)
        
        constraints = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_grad}]
        try:
            res = minimize(obj, current, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                candidate = project_feasible(res.x, n)
                score = exact_score(candidate)
                if score < best_score:
                    best_h = candidate.copy()
                    best_score = score
                current = candidate.copy()
        except Exception:
            break
    return best_h, best_score

# Coordinate descent
def coordinate_refine(h0, n, limit, max_rounds, top_pairs=200):
    best_h = h0.copy()
    best_score = exact_score(h0)
    no_improve = 0
    rng = np.random.RandomState(42)
    
    for rnd in range(max_rounds):
        if limit.remaining() < 3.0:
            break
        if no_improve > 200:
            break
        
        # Compute gradient of the max correlation
        dx = 2.0 / n
        corr = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        k_max = int(np.argmax(corr))
        d = k_max - (n - 1)
        ag = corr_grad(d, best_h, dx)
        
        # Also consider near-active lags
        sorted_lags = np.argsort(-corr)
        active_lags = []
        for idx in sorted_lags[:5]:
            d_idx = int(idx) - (n - 1)
            if abs(corr[idx] - corr[k_max]) < 0.01:
                active_lags.append(d_idx)
        
        # Weighted gradient from active lags
        if len(active_lags) > 1:
            weights = np.exp(100 * (corr[:2*n-1] - corr[k_max]))
            weights = weights / weights.sum()
            ag_w = np.zeros(n)
            for d_idx in range(2*n-1):
                if weights[d_idx] > 1e-6:
                    ag_w += weights[d_idx] * corr_grad(d_idx - (n-1), best_h, dx)
            ag = ag_w
        
        # Generate candidate pairs
        if rnd < 20:
            # All pairs (for small n)
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        else:
            # Sample pairs weighted by gradient difference
            pairs = []
            for _ in range(min(top_pairs, n * (n-1) // 4)):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j:
                    pairs.append((min(i,j), max(i,j)))
        
        # Sort by gradient difference (most promising first)
        pair_metrics = np.array([abs(ag[p[0]] - ag[p[1]]) for p in pairs])
        order = np.argsort(-pair_metrics)
        
        improved = False
        for idx in order[:top_pairs]:
            if limit.remaining() < 3.0:
                break
            if no_improve > 200:
                break
            i, j = pairs[idx]
            
            # Try both directions
            for direction in (1, -1):
                if direction == 1:
                    # h[i] += t, h[j] -= t
                    lo_d = max(-best_h[j], -0.4)
                    hi_d = min(1.0 - best_h[i], 0.4)
                else:
                    lo_d = max(-best_h[i], -0.4)
                    hi_d = min(1.0 - best_h[j], 0.4)
                
                if hi_d - lo_d < 1e-12:
                    continue
                
                # Coarse-to-fine search
                best_ld = 0.0
                best_ls = best_score
                for n_grid in (21, 51):
                    deltas = np.linspace(lo_d, hi_d, n_grid)
                    for dv in deltas:
                        if abs(dv) < 1e-14:
                            continue
                        tmp = best_h.copy()
                        if direction == 1:
                            tmp[i] += dv
                            tmp[j] -= dv
                        else:
                            tmp[i] -= dv
                            tmp[j] += dv
                        sc = exact_score(tmp)
                        if sc < best_ls:
                            best_ls = sc
                            best_ld = dv
                
                if best_ld != 0.0 and best_ls < best_score - 1e-13:
                    if direction == 1:
                        best_h[i] += best_ld
                        best_h[j] -= best_ld
                    else:
                        best_h[i] -= best_ld
                        best_h[j] += best_ld
                    best_h = project_feasible(best_h, n)
                    best_score = exact_score(best_h)
                    improved = True
                    no_improve = 0
                    break  # Restart pair ordering
            else:
                no_improve += 1
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_score

# Resolution transfer
def transfer_resolution(h_old, new_n):
    m = len(h_old)
    old_x = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0/m  # cell centers
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(new_x, old_x, h_old)
    return project_feasible(h_new, new_n)

# Random 0-1 pattern
def random_binary(n, rng):
    h = np.zeros(n)
    indices = rng.choice(n, size=n//2, replace=False)
    h[indices] = 1.0
    return project_feasible(h, n)
