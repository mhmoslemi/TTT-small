```python
import numpy as np
from scipy.optimize import minimize, brentq
import time

_SLSQP_BETA = 0.0
_SLSQP_DEADLINE = 0.0
_SLSQP_TARGET = 0.0
_GRAD_ENABLED = True

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    n = len(x)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target
    t_lo = float(-x.max()) - 0.01
    t_hi = float(1.0 - x.min()) + 0.01
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if fhi <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-14, rtol=1e-14, maxiter=60)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-12:
        mask = (out > 1e-8) & (out < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            out = out.copy()
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

def smooth_obj_func(h):
    if time.time() > _SLSQP_DEADLINE:
        return 0.0  # Won't be used past deadline
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    beta = _SLSQP_BETA
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    return (m + np.log(Z)) / beta

def smooth_grad_func(h):
    if time.time() > _SLSQP_DEADLINE:
        return np.zeros(len(h))
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    beta = _SLSQP_BETA
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    p = e / Z
    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - d] += w * (-h[d:])
            grad[d:] += w * (1.0 - h[:n - d])
        else:
            k = -d
            grad[k:] += w * (-h[:n - k])
            grad[:n - k] += w * (1.0 - h[k:])
    return grad

def slsqp_eq_constraint(h):
    return h.sum() - _SLSQP_TARGET

def run_slsqp_stage(h, deadline, beta_list):
    global _SLSQP_BETA, _SLSQP_DEADLINE, _SLSQP_TARGET
    n = len(h)
    target = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    _SLSQP_DEADLINE = deadline
    _SLSQP_TARGET = target
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': slsqp_eq_constraint}]
    for beta in beta_list:
        if time.time() > deadline - 15:
            break
        _SLSQP_BETA = beta
        maxiter = max(30, min(300, int((deadline - time.time()) * 1.5)))
        try:
            res = minimize(smooth_obj_func, h_curr, jac=smooth_grad_func,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(np.clip(res.x, 0, 1), target)
                sc = evaluate_exact(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_curr = h_new.copy()
        except Exception:
            break
    return best_h, best_score
```
