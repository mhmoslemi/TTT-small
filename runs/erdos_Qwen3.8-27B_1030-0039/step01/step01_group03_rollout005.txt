def exact_c5(h):

    dx = 2.0 / float(N)
    c = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    return float(c) if np.isfinite(c) else float('inf')

def make_feasible(arr):
    arr = np.asarray(arr, dtype=float).reshape(-1)
    if arr.size == 0:
        return arr
    N = arr.size
    arr = np.nan_to_num(arr, nan=0.5, posinf=1.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1.0)
    target = 0.5 * float(N)
    
    # Bisection on global shift
    cur = float(np.sum(arr))
    if abs(cur - target) < 1e-14:
        h = arr.copy()
    else:
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            s = float(np.sum(np.clip(arr + mid, 0.0, 1.0)))
            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(arr + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fine adjustment
    for _ in range(200):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            n_free = int(np.sum(mask))
            step = diff / n_free
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            n_free = int(np.sum(mask))
            step = -diff / n_free
            h[mask] = np.clip(h[mask] - step, 0.0, 1.0)
    
    # Final exact fix
    for _ in range(50):

        if abs(diff) < 1e-15:

def transfer_length(h, new_N):

    new_N = int(new_N)
    if h.size == new_N:
        return make_feasible(h)
    if h.size == 0 or new_N <= 0:
        return np.full(max(1, new_N), 0.5)
    if new_N % h.size == 0:
        factor = new_N // h.size
        y = np.repeat(h, factor)
        return make_feasible(y)
    if h.size % new_N == 0:
        factor = h.size // new_N
        y = h.reshape(new_N, factor).mean(axis=1)
        return make_feasible(y)
    src = (np.arange(h.size) + 0.5) * (2.0 / float(h.size))
    dst = (np.arange(new_N) + 0.5) * (2.0 / float(new_N))
    y = np.interp(dst, src, h, left=float(h[0]), right=float(h[-1]))

        if remaining_budget(t0, budget) < reserve + 5.0:
            break
        if evals % 2000 == 0:
            cvals = correlations(best_h)
            cmax = float(np.max(cvals))
            active_tol = max(1e-7, 2e-4 * cmax)
            active_mask = cvals >= cmax - active_tol
            active_idx = np.where(active_mask)[0]
    return best_h, best_c


def random_feasible(rng, N):
    x = 0.5 + 0.3 * rng.normal(0.0, 1.0, N)
    return make_feasible(x)


def cos_feasible(N, mode, amp):
    t = (np.arange(N) + 0.5) / float(N)
    x = 0.5 + float(amp) * np.cos(2.0 * np.pi * float(mode) * t)
    return make_feasible(x)


def finalize_solution(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
        h = np.array([0.5], dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h = make_feasible(h)
    c = exact_c5(h)
    if not np.isfinite(c):
        h = make_feasible(np.full(h.size, 0.5))
        c = exact_c5(h)
    return h, float(c), int(h.size)


def get_parent():
    try:
        vals = initial_h_values
    except Exception:
        return None
    if vals is None:
        return None
    arr = np.asarray(vals, dtype=float).reshape(-1)
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.5)
    return arr


def remaining_budget(t0, budget):
    return float(budget) - (time.perf_counter() - float(t0))
