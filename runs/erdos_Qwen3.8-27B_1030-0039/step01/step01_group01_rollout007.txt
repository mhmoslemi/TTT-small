```python
def cycle_lag_descent(h, deadline, K=3, alpha=0.01, n_iter=10000):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Compute all c_d
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full') * dx
        # Find top K lags
        top_k = np.argsort(corr)[-K:]
        # Average gradient over top K
        grad = np.zeros(n)
        for d_idx in top_k:
            d = d_idx - (n - 1)
            # grad of c_d
            if d == 0:
                grad += (1.0 - 2.0 * h) * dx
            elif d > 0:
                grad[:n-d] += (-h[d:]) * dx
                grad[d:] += (1.0 - h[:n-d]) * dx
            else:
                k = -d
                grad[k:] += (-h[:n-k]) * dx
                grad[:n-k] += (1.0 - h[k:]) * dx
        grad /= K
        # Gradient step
        h -= alpha * grad
        # Project to [0,1] and sum constraint
        h = project_feasible(h, target)
        # Check improvement
        score = evaluate_exact(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize, brentq
import time

class TimeLimit(Exception):
    pass

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    n = len(x)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if fhi <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-15)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def grad_of_single_lag(h, d):
    """Gradient of c_d(h) = sum valid h[i](1-h[i+d]) * dx"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif d > 0:
        if d < n:
            grad[:n-d] += (-h[d:]) * dx
            grad[d:] += (1.0 - h[:n-d]) * dx
    else:
        k = -d
        if k < n:
            grad[k:] += (-h[:n-k]) * dx
            grad[:n-k] += (1.0 - h[k:]) * dx
    return grad

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    p = e / Z
    val = (m + np.log(Z)) / beta
    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += w * (-h[d:])
            grad[d:] += w * (1.0 - h[:n-d])
        else:
            k = -d
            grad[k:] += w * (-h[:n-k])
            grad[:n-k] += w * (1.0 - h[k:])
    return val, grad

def nesterov_subgrad(h, deadline, K=3, alpha0=0.05, n_iter=5000):
    """Nesterov-accelerated subgradient on averaged top-K lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h_curr)
    t = 1.0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Nesterov momentum
        t_new = 1.0 + (1 + np.sqrt(1 + 4 * t * t)) / 2
        mu = (t - 1) / t_new
        y = h_curr + mu * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute top K lags
        corr = compute_corr(y)
        top_idx = np.argpartition(corr, -K)[-K:]
        
        # Average gradient
        grad = np.zeros(n)
        for idx in top_idx:
            d = idx - (n - 1)
            grad += grad_of_single_lag(y, d)
        grad /= K
        
        # Step
        alpha = alpha0 / np.sqrt(it + 1)
        h_new = y - alpha * grad
        h_new = project_feasible(h_new, target)
        
        h_prev = h_curr
        h_curr = h_new
        t = t_new
        
        score = evaluate_exact(h_curr)
        if score < best_score:
            best_score = score
            best_h = h_curr.copy()
    
    return best_h, best_score

def run_slsqp_stage(h, deadline, beta_list):
    n = len(h)
    target_sum = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    for beta in beta_list:
        rem = deadline - time.time()
        if rem < 12:
            break
        maxiter = max(20, min(200, int(rem * 2)))
        def _f(x):
            if time.time() > deadline:
                raise TimeLimit()
            v, _ = smooth_obj_grad(x, beta)
            return v
        def _g(x):
            if time.time() > deadline:
                raise TimeLimit()
            _, g = smooth_obj_grad(x, beta)
            return g
        def _c(x):
            return x.sum() - target_sum
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': _c}]
        try:
            res = minimize(_f, h_curr, jac=_g, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
                sc = evaluate_exact(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_curr = h_new.copy()
        except TimeLimit:
            break
        except Exception:
            break
    return best_h, best_score

def golden_pair_line(h, i, j):
    n = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, evaluate_exact(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] += c; h[j] -= c
    fc = evaluate_exact(h)
    h[i] -= c; h[j] += c
    h[i] += d; h[j] -= d
    fd = evaluate_exact(h)
    h[i] -= d; h[j] += d
    for _ in range(30):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[i] += c; h[j] -= c
            fc = evaluate_exact(h)
            h[i] -= c; h[j] += c
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[i] += d; h[j] -= d
            fd = evaluate_exact(h)
            h[i] -= d; h[j] += d
    best_d = (a + b) / 2.0
    h[i] += best_d; h[j] -= best_d
    fs = evaluate_exact(h)
    h[i] -= best_d; h[j] += best_d
    return best_d, fs

def pair_descent(h, deadline, max_sweeps=15):
    n = len(h)
    h = h.copy()
    best_score = evaluate_exact(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_score
            for j in range(i + 1, n):
                d_best, s_best = golden_pair_line(h, i, j)
                if d_best != 0.0 and s_best < best_score - 1e-13:
                    h[i] += d_best
                    h[j] -= d_best
                    h = np.clip(h, 0.0, 1.0)
                    diff = n / 2.0 - h.sum()

def transfer_to_new_n(h_old, n_new):
    m = len(h_old)
    x_old = np.linspace(0.0, 2.0, m)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new / 2.0)

def thue_morse_h(n):
    """Generate Thue-Morse inspired pattern mapped to [0,1]."""
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Map to [0, 1]: 0 -> 0, 1 -> 1, then clip and project
    h = tm.astype(float)
    return project_feasible(h, n / 2.0)

def periodic_h(n, period, duty=0.5):
    """Generate a periodic pattern."""
    i = np.arange(n)
    pattern = np.zeros(period)
    pattern[:int(duty * period)] = 1.0
    h = np.tile(pattern, n // period + 1)[:n]
    return project_feasible(h, n / 2.0)

def random_ternary_h(n, rng):
    """Random h with values in {0, 0.5, 1} satisfying sum constraint."""
    # We need sum = n/2. With values 0, 0.5, 1:
    # Let a = #zeros, b = #halves, c = #ones
    # a + b + c = n, b/2 + c = n/2
    # So c = n/2 - b/2 = (n-b)/2, a = n - b - c = n - b - (n-b)/2 = (n-b)/2
    # So a = c = (n-b)/2, b must have same parity as n
    n_half = n // 2
    if n % 2 == 0:
        # n even: b can be 0, 2, 4, ..., n
        b_options = list(range(0, n + 1, 2))
    else:
        # n odd: b must be odd: 1, 3, 5, ..., n
        b_options = list(range(1, n + 1, 2))
    b = rng.choice(b_options)
    a = (n - b) // 2
    c = (n - b) // 2
    h = np.concatenate([np.zeros(a), np.full(b, 0.5), np.ones(c)])
    rng.shuffle(h)
    return h.copy()

def Fourier_h(n, rng):
    """Fourier-based random pattern."""
    f = rng.randint(1, max(2, n // 3) + 1)
    phi = rng.uniform(0, 2 * np.pi)
    A = rng.uniform(0.3, 0.5)
    i = np.arange(n)
    h = 0.5 + A * np.cos(2 * np.pi * f * i / n + phi)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)

def sa_ternary(h_init, deadline, n_iter=20000, T0=1.0, T_end=0.01):
    """Simulated annealing on ternary lattice."""
    n = len(h_init)
    target = n / 2.0
    # Convert to ternary
    h = np.round(h_init * 2) / 2  # map to {0, 0.5, 1}
    # Fix sum
    h = project_feasible(h, target)
    # Snap to nearest allowed values
    h = np.round(h * 2) / 2
    # Now fix sum exactly
    # Current sum
    cur_sum = h.sum()
    diff = target - cur_sum
    # Adjust: if diff > 0, need to increase some values
    # Find indices where we can increase
    if abs(diff) > 1e-10:
        # Just re-randomize
        rng = np.random.RandomState(0)
        h = random_ternary_h(n, rng)
    
    cur_score = evaluate_exact(h)
    best_h = h.copy()
    best_score = cur_score
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / n_iter)
        # Pick two random indices
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # Try a move: change h[i] and h[j] to maintain sum
        # Possible moves: (0->0.5, 0.5->0), (0.5->1, 1->0.5), (0->1, 1->0), etc.
        # Simple: swap h[i] and h[j]
        h[i], h[j] = h[j], h[i]
        new_score = evaluate_exact(h)
        delta = new_score - cur_score
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    
    return best_h, best_score
```

```python
def gradient_line_search(h, grad, target, max_alpha=1.0, c1=1e-4, rho=0.5, max_iter=50):
    """Backtracking line search along -grad direction."""
    n = len(h)
    f0 = evaluate_exact(h)
    alpha = max_alpha
    for _ in range(max_iter):
        h_trial = h - alpha * grad
        h_trial = project_feasible(h_trial, target)
        f_trial = evaluate_exact(h_trial)
        if f_trial <= f0 - c1 * alpha * np.dot(grad, h - h_trial):
            return h_trial, f_trial, alpha
        alpha *= rho
    return h, f0, 0.0
```

```python
import numpy as np
from scipy.optimize import minimize, brentq
import time


class TimeLimit(Exception):
    pass


def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(x, target_sum):
    n = len(x)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if fhi <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-15)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_of_single_lag(h, d):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif 0 < d < n:
        grad[:n-d] += (-h[d:]) * dx
        grad[d:] += (1.0 - h[:n-d]) * dx
    elif -n < d < 0:
        k = -d
        grad[k:] += (-h[:n-k]) * dx
        grad[:n-k] += (1.0 - h[k:]) * dx
    return grad


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    p = e / Z
    val = (m + np.log(Z)) / beta
    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += w * (-h[d:])
            grad[d:] += w * (1.0 - h[:n-d])
        else:
            k = -d
            grad[k:] += w * (-h[:n-k])
            grad[:n-k] += w * (1.0 - h[k:])
    return val, grad


def slsqp_obj(x, beta, deadline):
    if time.time() > deadline:
        raise TimeLimit()
    v, _ = smooth_obj_grad(x, beta)
    return v


def slsqp_jac(x, beta, deadline):
    if time.time() > deadline:
        raise TimeLimit()
    _, g = smooth_obj_grad(x, beta)
    return g


def slsqp_con(x, target):
    return x.sum() - target


def run_slsqp_stage(h, deadline, beta_list):
    n = len(h)
    target_sum = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    for beta in beta_list:
        rem = deadline - time.time()
        if rem < 12:
            break
        maxiter = max(15, min(150, int(rem * 1.5)))
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (target_sum,)}]
        try:
            res = minimize(slsqp_obj, h_curr, jac=slsqp_jac,
                          args=(beta, deadline), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
                sc = evaluate_exact(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_curr = h_new.copy()
        except TimeLimit:
            break
        except Exception:
            break
    return best_h, best_score


def nesterov_subgrad(h, deadline, K=3, alpha0=0.08, n_iter=8000):
    n = len(h)
    target = n / 2.0
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h_curr)
    t = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t_new = 1.0 + (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mu = (t - 1.0) / t_new
        y = h_curr + mu * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = compute_corr(y)
        top_idx = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(n)
        for idx in top_idx:
            d = idx - (n - 1)
            grad += grad_of_single_lag(y, d)
        grad /= float(K)
        alpha = alpha0 / np.sqrt(float(it + 1))
        h_new = y - alpha * grad
        h_new = project_feasible(h_new, target)
        h_prev = h_curr
        h_curr = h_new
        t = t_new
        score = evaluate_exact(h_curr)
        if score < best_score:
            best_score = score
            best_h = h_curr.copy()
    return best_h, best_score


def golden_pair_line(h, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, evaluate_exact(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] += c; h[j] -= c
    fc = evaluate_exact(h)
    h[i] -= c; h[j] += c
    h[i] += d; h[j] -= d
    fd = evaluate_exact(h)
    h[i] -= d; h[j] += d
    for _ in range(25):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[i] += c; h[j] -= c
            fc = evaluate_exact(h)
            h[i] -= c; h[j] += c
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[i] += d; h[j] -= d
            fd = evaluate_exact(h)
            h[i] -= d; h[j] += d
    best_d = (a + b) / 2.0
    h[i] += best_d; h[j] -= best_d
    fs = evaluate_exact(h)
    h[i] -= best_d; h[j] += best_d
    return best_d, fs


def pair_descent(h, deadline, max_sweeps=8):
    n = len(h)
    h = h.copy()
    best_score = evaluate_exact(h)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_score
            for j in range(i + 1, n):
                d_best, s_best = golden_pair_line(h, i, j)
                if d_best != 0.0 and s_best < best_score - 1e-13:
                    h[i] += d_best
                    h[j] -= d_best
                    h = np.clip(h, 0.0, 1.0)
                    diff = n / 2.0 - h.sum()
                    if abs(diff) > 1e-12:
                        h = project_feasible(h, n / 2.0)
                    new_s = evaluate_exact(h)
                    if new_s < best_score - 1e-13:
                        best_score = new_s
                        improved = True
        if not improved:
            break
    return h, best_score


def random_pair_descent(h, deadline, n_pairs=150, max_sweeps=20, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_score = evaluate_exact(h)
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    npairs = min(n_pairs, len(all_pairs))
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        idx = rng.choice(len(all_pairs), size=npairs, replace=False)
        improved = False
        for k in idx:
            i, j = all_pairs[k]
            d_best, s_best = golden_pair_line(h, i, j)
            if d_best != 0.0 and s_best < best_score - 1e-13:
                h[i] += d_best
                h[j] -= d_best
                h = np.clip(h, 0.0, 1.0)
                diff = n / 2.0 - h.sum()
                if abs(diff) > 1e-12:
                    h = project_feasible(h, n / 2.0)
                new_s = evaluate_exact(h)
                if new_s < best_score - 1e-13:
                    best_score = new_s
                    improved = True
        if not improved:
            break
    return h, best_score


def transfer_to_new_n(h_old, n_new):
    m = len(h_old)
    x_old = np.linspace(0.0, 2.0, m)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new / 2.0)


def thue_morse_h(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm.astype(float)
    return project_feasible(h, n / 2.0)


def periodic_h(n, period, duty):
    i = np.arange(n)
    base = np.array([1.0 if (pp % period) < int(duty * period) else 0.0 for pp in range(period)])
    h = np.tile(base, n // period + 1)[:n]
    return project_feasible(h, n / 2.0)


def random_ternary_h(n, rng):
    if n % 2 == 0:
        b_options = list(range(0, n + 1, 2))
    else:
        b_options = list(range(1, n + 1, 2))
    b = int(rng.choice(b_options))
    a = (n - b) // 2
    c = (n - b) // 2
    h = np.concatenate([np.zeros(a), np.full(b, 0.5), np.ones(c)])
    rng.shuffle(h)
    return h.copy()


def fourier_h(n, rng):
    f = int(rng.randint(1, max(2, n // 3) + 1))
    phi = rng.uniform(0, 2 * np.pi)
    A = rng.uniform(0.3, 0.5)
    i = np.arange(n)
    h = 0.5 + A * np.cos(2 * np.pi * f * i / n + phi)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n / 2.0)


def sa_ternary(h_init, deadline, n_iter=15000, T0=0.5, Tend=0.01, seed=42):
    n = len(h_init)
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h = np.round(h_init * 2.0) / 2.0
    h = project_feasible(h, target)
    h = np.round(h * 2.0) / 2.0
    if abs(h.sum() - target) > 1e-10:
        h = random_ternary_h(n, rng)
    cur_score = evaluate_exact(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (Tend / T0) ** (float(it) / float(n_iter))
        i = int(rng.randint(0, n))
        j = int(rng.randint(0, n))
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = h[j], h[i]
        new_score = evaluate_exact(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
    return best_h, best_score
```
