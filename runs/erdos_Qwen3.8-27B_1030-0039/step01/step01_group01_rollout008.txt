    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n - d] += w * (-h[d:])
            grad[d:] += w * (1.0 - h[:n - d])
        else:
            k = -d
            grad[k:] += w * (-h[:n - k])
            grad[:n - k] += w * (1.0 - h[k:])
    return val, grad

    bh = h.copy()

        def _g(x):

            return g
        def _c(x):
            return x.sum() - target

                    bs = sc

        except _TO:

def _pair_line(h, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, _eval(h)
    def f(t):
        h[i] += t; h[j] -= t
        v = _eval(h)
        h[i] -= t; h[j] += t
        return v
    gr = (5.0 ** 0.5 - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c); fd = f(d)
    for _ in range(25):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    mt = (a + b) / 2.0
    h[i] += mt; h[j] -= mt
    ns = _eval(h)
    return mt, ns

def _pair_desc(h, deadline, nsweeps=100):
