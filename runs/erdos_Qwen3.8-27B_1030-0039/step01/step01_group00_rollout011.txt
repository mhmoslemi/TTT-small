def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h_raw, n):
    target = n / 2.0
    # Find shift t such that sum(clip(h_raw + t, 0, 1)) = target
    def loss(t):
        return np.sum(np.clip(h_raw + t, 0.0, 1.0)) - target
    lo = -max(h_raw) - 1.0
    hi = 1.0 - min(h_raw) + 1.0
    # loss(lo) should be ≤ 0 (all clamped to 0), loss(hi) should be ≥ 0 (all clamped to 1)
    # Actually: at t=lo, h_raw+lo ≤ 0, so clip gives 0, sum=0, loss=-target<0
    # At t=hi, h_raw+hi ≥ 1, so clip gives 1, sum=n, loss=n-target=target>0

        return np.clip(h_raw + lo, 0.0, 1.0)

    t = brentq(loss, lo, hi, xtol=1e-15, rtol=1e-15)

    h = x
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(corr - C_target, 0)

    # But we want sum of softplus(beta * (corr - C_target))
    # For corr > C_target: softplus(beta*(corr-C_target)) ≈ beta*(corr-C_target) for large beta*(corr-C_target)
    # For corr ≤ C_target: softplus(beta*(corr-C_target)) ≈ 0
    a = beta * (corr - C_target)
    # stable softplus: max(a, 0) + log(1 + exp(-|a|))
    sp = np.maximum(a, 0) + np.log1p(np.exp(-np.abs(a)))

import numpy as np
import time
from scipy.optimize import minimize, brentq

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h_raw, n):
    target = n / 2.0
    def loss(t):
        return float(np.sum(np.clip(h_raw + t, 0.0, 1.0))) - target
    lo = float(np.min(h_raw)) - 2.0
    hi = 1.0 - float(np.max(h_raw)) + 2.0
    vlo, vhi = loss(lo), loss(hi)
    if vlo >= 0:
        return np.clip(h_raw + lo, 0.0, 1.0)
    if vhi <= 0:
        return np.clip(h_raw + hi, 0.0, 1.0)
    t = brentq(loss, lo, hi, xtol=1e-14, rtol=1e-14)
    return np.clip(h_raw + t, 0.0, 1.0)

def smooth_score_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    bc = beta * corr
    m = float(np.max(bc))
    w = np.exp(bc - m)
    Z = float(np.sum(w))
    score = (m + np.log(Z)) / beta
    p = w / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        d = k - (n - 1)
        wkt = pk * dx
        if d >= 0:
            if d < n:
                grad[d:] += wkt * (1.0 - h[:n-d])
                grad[:n-d] += wkt * (-h[d:])
        else:
            pp = -d
            if pp < n:
                grad[:n-pp] += wkt * (1.0 - h[pp:])
                grad[pp:] += wkt * (-h[:n-pp])
    return score, grad

def _eq_constraint(x):
    n = len(x)
    return float(np.sum(x)) - n / 2.0

def _eq_jacobian(x):
    n = len(x)
    return np.ones(n)

def smooth_optimize(h0, n, betas, maxiter_per_beta, deadline):
    best_h = h0.copy()
    best_score = exact_score(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jacobian}]
    current = best_h.copy()
    for beta in betas:
        if time.time() > deadline:
            break
        def obj(x):
            s, _ = smooth_score_grad(x, beta)
            return s
        def grad(x):
            _, g = smooth_score_grad(x, beta)
            return g
        try:
            res = minimize(obj, current, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                candidate = project_feasible(res.x, n)
                sc = exact_score(candidate)
                if sc < best_score - 1e-12:
                    best_h = candidate.copy()
                    best_score = sc
                current = candidate.copy()
        except Exception:
            break
    return best_h, best_score

def target_optimize(h0, n, C_target, betas, maxiter_per_beta, deadline):
    best_h = h0.copy()
    best_score = exact_score(h0)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jacobian}]
    current = best_h.copy()
    for beta in betas:
        if time.time() > deadline:
            break
        def obj_t(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode="full") * dx
            a = beta * (corr - C_target)
            sp = np.maximum(a, 0) + np.log1p(np.exp(-np.abs(a)))
            return float(np.sum(sp))
        def grad_t(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode="full") * dx
            a = beta * (corr - C_target)
            sig = 1.0 / (1.0 + np.exp(-a))  # sigmoid
            # grad of sum(softplus(beta*(corr_k - C))) = sum_k sigmoid(beta*(corr_k - C)) * beta * dcorr_k/dx
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                sk = sig[k]
                if sk < 1e-14:
                    continue
                wkt = sk * beta * dx
                d = k - (n - 1)
                if d >= 0:
                    if d < n:
                        grad[d:] += wkt * (1.0 - x[:n-d])
                        grad[:n-d] += wkt * (-x[d:])
                else:
                    pp = -d
                    if pp < n:
                        grad[:n-pp] += wkt * (1.0 - x[pp:])
                        grad[pp:] += wkt * (-x[:n-pp])
            return grad
        try:
            res = minimize(obj_t, current, jac=grad_t, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                candidate = project_feasible(res.x, n)
                sc = exact_score(candidate)
                if sc < best_score - 1e-12:
                    best_h = candidate.copy()
                    best_score = sc
                current = candidate.copy()
        except Exception:
            break
    return best_h, best_score
