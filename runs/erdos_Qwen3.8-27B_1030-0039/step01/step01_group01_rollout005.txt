    if time.time() > state['deadline']:

    v, _ = smooth_obj_grad(x, beta)
    return v

    if time.time() > state['deadline']:
        raise TimeLimit()
    _, g = smooth_obj_grad(x, state['beta'])
    return g

def slsqp_c(x, state):
    return x.sum() - state['target_sum']

def run_slsqp(h, deadline, beta_list):
    n = len(h)
    target_sum = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    for beta in beta_list:
        if time.time() > deadline:
            break
        rem = deadline - time.time()
        if rem < 15:
            break
        state = {'beta': beta, 'deadline': deadline, 'target_sum': target_sum}
        maxiter = max(20, min(300, int(rem * 2.5)))
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': slsqp_c, 'args': (state,)}]
        try:
            res = minimize(slsqp_f, h_curr, jac=slsqp_g, args=(state,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})

                if s_p3 < best_score - 1e-12:
                    best_h, best_score, best_n = h_p3, s_p3, n_try

    rem = deadline - time.time()
    if rem > 40:
        n_try2 = int(best_n * 2)
        if n_try2 != best_n and n_try2 <= 500:
            h_t2 = transfer_to_new_n(best_h, n_try2)
            h_t2, s_t2 = run_slsqp(h_t2, deadline - 5, [40.0, 120.0])
            if s_t2 < best_score - 1e-12:
                best_h, best_score, best_n = h_t2, s_t2, n_try2

    rem = deadline - time.time()
    if rem > 15:
        h_f, s_f = pair_descent(best_h, deadline, max_sweeps=6)
        if s_f < best_score - 1e-13:
            best_h, best_score, best_n = h_f, s_f, best_n

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)
    final_score = evaluate_exact(best_h)

    if not np.all(np.isfinite(best_h)) or best_n < 1:
        best_h = np.full(45, 0.5)
        final_score = evaluate_exact(best_h)
        best_n = 45

    return (best_h, final_score, best_n)
