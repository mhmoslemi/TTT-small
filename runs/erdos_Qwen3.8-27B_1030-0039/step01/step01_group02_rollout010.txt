def evaluate_exact(h):

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    if n == 1:
        val = min(max(target, 0.0), 1.0)
        return np.array([val]), True
    def f_offset(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    lo = -1.0 - np.max(h)
    hi = 1.0 - np.min(h)
    flo = f_offset(lo)
    fhi = f_offset(hi)
    if abs(flo) < 1e-14:
        return np.clip(h + lo, 0.0, 1.0), True
    if abs(fhi) < 1e-14:
        return np.clip(h + hi, 0.0, 1.0), True
    if flo > 0:
        return np.clip(h + lo, 0.0, 1.0), True
    if fhi < 0:
        return np.clip(h + hi, 0.0, 1.0), True
    try:

    except Exception:
        t = brentq(f_offset, lo, hi, xtol=1e-12, maxiter=500)
    result = np.clip(h + t, 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = np.sum(mask)

def get_active_gradients(h, K):
    """Get top-K lag values and their gradients."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    values = corr * dx
    # Get top-K indices
    idx = np.argsort(values)[::-1][:K]
    # Compute gradients for active lags
    grads = np.zeros((len(idx), n))
    for ii, k_idx in enumerate(idx):
        lag = k_idx - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                g[i] += dx * (1.0 - h[i + lag])
                g[i + lag] += -dx * h[i]
        else:
            p = -lag
            for i in range(p, n):
                g[i] += dx * (1.0 - h[i - p])
                g[i - p] += -dx * h[i]
        grads[ii] = g
    return values[idx], grads

def qp_trust_step(h, K, trust, mu, deadline):

    n = len(h)
    active_vals, active_grads = get_active_gradients(h, K)
    
    # Build QP: min t + mu * ||x - h||^2
    # s.t. active_grads[i] @ x <= t + (active_grads[i] @ h - active_vals[i])
    #      0 <= x <= 1
    #      sum(x) = n/2
    #      ||x - h||^2 <= trust^2  (relaxed via penalty for simplicity)
    
    x = cp.Variable(n)
    t = cp.Variable()
    
    offset = active_grads @ h - active_vals
    constraints = []
    for i in range(len(active_vals)):
        constraints.append(active_grads[i] @ x - t <= offset[i])
    
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    constraints.append(cp.sum(x) == n / 2.0)

    prob = cp.Problem(objective, constraints)

                if prob.status in ['optimal', 'optimal_inaccurate']:
                    break
            except Exception:
                continue

            return None, None
        
        x_star = x.value
        if x_star is None or not np.all(np.isfinite(x_star)):
            return None, None
        
        return x_star, t.value
    except Exception:

            best_score = s

    # Refine around best_alpha
    lo = max(0, best_alpha - (max(alpha_max, alpha_min) / 10) * 2)
    hi = min(max(alpha_max, alpha_min), best_alpha + (max(alpha_max, alpha_min) / 10) * 2)
    for _ in range(10):

        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        h1, _ = project_feasible(h0 + m1 * direction, n)
        s1 = evaluate_exact(h1)

        s2 = evaluate_exact(h2)
        if s1 < s2:
            hi = m2
        else:
            lo = m1
        # Track best

            h_best = h1
        if s2 < best_score:
            best_score = s2

        # Update best_alpha
        if s1 < s2:

            h_best = h2
