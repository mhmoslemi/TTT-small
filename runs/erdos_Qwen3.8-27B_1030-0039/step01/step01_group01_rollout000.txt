Actually wait - the complement h → 1-h: sum(1-h) = n - n/2 = n/2. So it's feasible! And:
C5(1-h) = max_d dx * Σ (1-h[i]) * h[i+d] = max_d dx * Σ (h[i+d] - h[i]*h[i+d])

Compare with C5(h) = max_d dx * Σ h[i]*(1-h[i+d]) = max_d dx * Σ (h[i] - h[i]*h[i+d])

So C5(1-h) = max_d dx * (Σ_{valid} h[i+d] - Σ_{valid} h[i]*h[i+d])
C5(h) = max_d dx * (Σ_{valid} h[i] - Σ_{valid} h[i]*h[i+d])

The Σ h[i] and Σ h[i+d] terms are the same (just different range of summation), so the difference is in the range. For d > 0:
- C5(h) at lag d: dx * (Σ_{i=0}^{n-1-d} h[i] - Σ_{i=0}^{n-1-d} h[i]*h[i+d])
- C5(1-h) at lag d: dx * (Σ_{i=0}^{n-1-d} h[i+d] - Σ_{i=0}^{n-1-d} h[i]*h[i+d])

These are different! The first uses Σ h[i] for i in [0, n-1-d], the second uses Σ h[i+d] for i in [0, n-1-d] (i.e., Σ h[j] for j in [d, n-1]).

So the complement is NOT the same objective. But it's a valid transformation that gives a different starting point.

import numpy as np
from scipy.optimize import brentq
import time

class TimeLimit(Exception):
    pass

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_max_lag(h):
    """Return the index (in full correlation array) of the maximum."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return int(np.argmax(corr))

def project_feasible(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=np.float64).copy()
    x = np.clip(x, 0.0, 1.0)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        t = t_lo
    elif fhi <= 0:
        t = t_hi
    else:
        t = brentq(F, t_lo, t_hi, xtol=1e-15, maxiter=100)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            out = out.copy()
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    p = e / Z
    val = (m + np.log(Z)) / beta
    grad = np.zeros(n)
    for j in range(len(c)):
        w = p[j]
        if w < 1e-15:
            continue
        d = j - (n - 1)
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[:n-d] += w * dx * (1.0 - h[d:])
            grad[d:] += w * dx * (-h[:n-d])
        else:
            k = -d
            grad[k:] += w * dx * (1.0 - h[:n-k])
            grad[:n-k] += w * dx * (-h[k:])
    return val, grad

def projected_gradient_optimize(h, beta, deadline, n_iter=200):
    """Projected gradient descent with the smooth objective."""
    n = len(h)
    target = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h_curr)
    
    # Adaptive step size
    alpha = 0.1
    for it in range(n_iter):
        if time.time() > deadline:
            break
        val, grad = smooth_obj_and_grad(h_curr, beta)
        h_try = project_feasible(h_curr - alpha * grad, target)
        score = evaluate_exact(h_try)
        if score < best_score - 1e-14:
            best_h = h_try.copy()
            best_score = score
            h_curr = h_try.copy()
            # Keep alpha or increase slightly
            alpha = min(alpha * 1.1, 0.3)
        else:
            alpha *= 0.8
            if alpha < 0.001:
                break
    return best_h, best_score

def golden_pair_search(h, i, j, deadline):
    """Golden section search for optimal transfer from j to i."""
    n = len(h)
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, evaluate_exact(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    h[i] += c; h[j] -= c
    fc = evaluate_exact(h)
    h[i] -= c; h[j] += c
    h[i] += d; h[j] -= d
    fd = evaluate_exact(h)
    h[i] -= d; h[j] += d
    
    for _ in range(25):
        if time.time() > deadline:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[i] += c; h[j] -= c
            fc = evaluate_exact(h)
            h[i] -= c; h[j] += c
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[i] += d; h[j] -= d
            fd = evaluate_exact(h)
            h[i] -= d; h[j] += d
    
    best_t = (a + b) / 2.0
    h[i] += best_t; h[j] -= best_t
    fs = evaluate_exact(h)
    h[i] -= best_t; h[j] += best_t
    return best_t, fs

def gradient_guided_pair_descent(h, deadline, beta=200.0, max_sweeps=5):
    """Pair descent using gradient to prioritize pairs."""
    n = len(h)
    h = h.copy()
    best_score = evaluate_exact(h)
    target = n / 2.0
    rng = np.random.RandomState(12345)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Compute gradient for prioritization
        if beta > 0:
            _, grad = smooth_obj_and_grad(h, beta)
            # Pairs where gradient suggests a move
            # We want to increase h[i] where grad[i] < 0 and decrease h[j] where grad[j] > 0
            # Priority: grad[i] - grad[j] (more negative = higher priority for i getting from j)
        else:
            grad = np.zeros(n)
        
        # Get all pairs with their priority
        pairs = []
        for i in range(n):
            for j in range(i+1, n):
                diff = grad[j] - grad[i]  # positive means we want to transfer from j to i
                if diff > 0.01:  # only consider pairs with clear gradient direction
                    pairs.append((diff, i, j))
        
        # Sort by priority (largest first)
        pairs.sort(key=lambda x: -x[0])
        
        # Also add some random pairs for exploration
        for _ in range(min(20, n*n//4)):
            i, j = rng.randint(0, n, 2)
            if i != j:
                i, j = min(i,j), max(i,j)
                pairs.append((0.0, i, j))
        
        improved = False
        for priority, i, j in pairs[:200]:  # limit number of pairs per sweep
            if time.time() > deadline:
                break
            # Determine direction from gradient
            if grad[j] > grad[i]:
                # transfer from j to i
                best_t, best_s = golden_pair_search(h, i, j, deadline)
                if best_t > 1e-10 and best_s < best_score - 1e-13:
                    h[i] += best_t
                    h[j] -= best_t
                    h = np.clip(h, 0.0, 1.0)
                    if abs(h.sum() - target) > 1e-10:
                        h = project_feasible(h, target)
                    best_score = best_s
                    improved = True
            else:
                # transfer from i to j
                best_t, best_s = golden_pair_search(h, j, i, deadline)
                if best_t > 1e-10 and best_s < best_score - 1e-13:
                    h[j] += best_t
                    h[i] -= best_t
                    h = np.clip(h, 0.0, 1.0)
                    if abs(h.sum() - target) > 1e-10:
                        h = project_feasible(h, target)
                    best_score = best_s
                    improved = True
        
        if not improved:
            break
    
    return h, best_score

def stochastic_refine(h, deadline, n_iters=50000):
    """Stochastic local search with simulated annealing."""
    n = len(h)
    target = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h_curr)
    rng = np.random.RandomState(67890)
    
    T_start = 0.02
    T_end = 0.0005
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / n_iters
        T = T_start * (T_end / T_start) ** frac
        
        # Pick random pair and random transfer
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        
        t_max = min(h_curr[i], 1.0 - h_curr[j])
        t_min = max(-h_curr[i], h_curr[j] - 1.0)
        if t_max <= t_min:
            continue
        t = rng.uniform(t_min, t_max)
        
        # Save state
        old_i = h_curr[i]
        old_j = h_curr[j]
        
        h_curr[i] = old_i + t
        h_curr[j] = old_j - t
        h_curr = np.clip(h_curr, 0.0, 1.0)
        if abs(h_curr.sum() - target) > 1e-10:
            h_curr = project_feasible(h_curr, target)
        
        score = evaluate_exact(h_curr)
        delta = score - best_score
        
        if score < best_score - 1e-14:
            best_h = h_curr.copy()
            best_score = score
        elif rng.random() < np.exp(-max(0, delta) / (T + 1e-10)):
            pass  # accept worse solution
        else:
            # Revert
            h_curr[i] = old_i
            h_curr[j] = old_j
            h_curr = np.clip(h_curr, 0.0, 1.0)
            if abs(h_curr.sum() - target) > 1e-10:
                h_curr = project_feasible(h_curr, target)
    
    return best_h, best_score
