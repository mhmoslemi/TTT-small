def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    if n == 1:
        return np.array([min(max(target, 0.0), 1.0)]), True
    def f_off(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    lo = -float(np.max(h)) - 0.5
    hi = 1.0 - float(np.min(h)) + 0.5
    flo = f_off(lo)
    fhi = f_off(hi)
    if flo > 0:
        return np.clip(h + lo, 0.0, 1.0), True
    if fhi < 0:
        return np.clip(h + hi, 0.0, 1.0), True
    try:
        t = brentq(f_off, lo, hi, xtol=1e-15, rtol=1e-15, maxiter=200)
    except Exception:
        try:
            t = brentq(f_off, lo, hi, xtol=1e-12, maxiter=300)
        except Exception:
            t = 0.5 * (lo + hi)
    result = np.clip(h + t, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result, True

        rng.shuffle(pairs)
        nsamp = 15
        for (i, j) in pairs[:40]:
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-13:
                continue
            deltas = np.linspace(lo, hi, nsamp)
            best_d = 0.0
            best_s = cur_score
            for d in deltas:
                if abs(d) < 1e-15:
                    continue
                h[i] += d
                h[j] -= d
                s = evaluate_exact(h)
                if s < best_s - tol:
                    best_s = s
                    best_d = d
            if best_d != 0.0:
                h[i] += best_d
                h[j] -= best_d
                cur_score = best_s
                improved = True
        if not improved:
            break
    return h, cur_score

def gradient_step(h, beta, top_k=None, steps=8):
    n = len(h)
    h = h.copy()
    cur_score = evaluate_exact(h)
    for step in range(steps):
        _, grad = smooth_obj_grad(h, beta, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step_sizes = [0.001, 0.003, 0.01, 0.03, 0.1]
        best_h = h.copy()
        best_s = cur_score
        for ss in step_sizes:
            direction = -grad / gnorm
            candidate = h + ss * direction
            candidate, ok = project_feasible(candidate, n)
            if not ok:
                continue
            s = evaluate_exact(candidate)
            if s < best_s - 1e-14:
                best_s = s
                best_h = candidate.copy()
        if best_s < cur_score - 1e-14:
            h = best_h
            cur_score = best_s
        else:
            break
    return h, cur_score

def make_fourier_candidate(n, a1=0.15, a2=0.1, phi1=0.0, phi2=0.5, rng=None):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + a1 * np.cos(np.pi * x + phi1) + a2 * np.cos(2.0 * np.pi * x + phi2)
    if rng is not None:
        h += rng.randn(n) * 0.01
    h, ok = project_feasible(h, n)
    return h

def sa_continuous(h_init, n, n_iters=5000, T_start=0.02, T_end=0.001, rng=None):
    if rng is None:
        rng = np.random.RandomState(77)
    h = h_init.copy()
    cur_score = evaluate_exact(h)
    best_h = h.copy()
    best_score = cur_score
    for it in range(n_iters):
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters - 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi - lo < 1e-12:
            continue
        d = rng.uniform(lo, hi) if rng.random() > 0.5 else -rng.uniform(-lo, -hi)
        h[i] += d
        h[j] -= d
        s_new = evaluate_exact(h)
        delta = s_new - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_score = s_new
            if s_new < best_score:
                best_score = s_new
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 10.0
    rng = np.random.RandomState(seed)

    def time_left():
        return deadline - time.time()

    try:
        h_init = np.array(initial_h_values, dtype=np.float64).flatten().copy()
        n_init = len(h_init)
    except Exception:
        n_init = 90
        x = np.linspace(0.0, 2.0, n_init)
        h_init = 0.5 + 0.2 * np.cos(np.pi * x) + 0.1 * np.cos(2.0 * np.pi * x + 0.3)
        h_init, _ = project_feasible(h_init, n_init)

    h_init, _ = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_score = evaluate_exact(best_h)
    best_n = n_init

    pool = []

    def add_to_pool(hc, nc):
        hc, ok = project_feasible(hc, nc)
        if not ok:
            return None
        if not np.all(np.isfinite(hc)):
            return None
        s = evaluate_exact(hc)
        pool.append((s, hc.copy(), nc))
        return s

    add_to_pool(h_init, n_init)
    h_refl = 1.0 - h_init[::-1].copy()
    add_to_pool(h_refl, n_init)

    for sig in [0.02, 0.04, 0.08]:
        add_to_pool(h_init + rng.randn(n_init) * sig, n_init)

    for a1 in [0.1, 0.2, 0.3]:
        for a2 in [0.05, 0.1, 0.2]:
            for ph in [0.0, 0.5, 1.0]:
                add_to_pool(make_fourier_candidate(n_init, a1, a2, ph, 0.3, None), n_init)

    resolutions = [n_init]
    for mult in [0.8, 1.2, 1.5]:
        nr = int(round(n_init * mult))
        nr = max(20, nr)
        if nr % 2 == 1:
            nr += 1
        if nr not in resolutions and 20 <= nr <= 160:
            resolutions.append(nr)
    resolutions.sort()

    for nr in resolutions:
        if nr != n_init:
            h_tr = transfer_to_resolution(h_init, nr)
            add_to_pool(h_tr, nr)

    pool.sort(key=lambda x: x[0])
    if len(pool) > 20:
        pool = pool[:20]

    processed = set()
    max_process = 50

    while time_left() > 50 and len(processed) < max_process:
        if not pool:
            h_pert = best_h + rng.randn(best_n) * 0.03
            add_to_pool(h_pert, best_n)
            if not pool:
                break
            pool.sort(key=lambda x: x[0])

        score_c, h_c, n_c = pool.pop(0)
        h_key = (n_c, tuple(np.round(h_c, 6)))
        if h_key in processed:
            continue
        processed.add(h_key)

        if score_c > best_score + 0.05:
            if len(pool) < 15:
                pool.append((score_c, h_c, n_c))
            continue

        h_work = h_c.copy()
        h_work, ok = project_feasible(h_work, n_c)
        if not ok:
            continue
        cur_score = evaluate_exact(h_work)
        n = n_c

        beta_schedule = [4.0, 12.0, 30.0, 80.0, 200.0, 500.0]
        maxiter_base = 100 if n <= 60 else (60 if n <= 120 else 40)

        for beta in beta_schedule:
            if time_left() < 20:
                break
            top_k = min(n, max(12, int(0.2 * (2 * n - 1))))
            try:
                def obj_func(x, _b=beta, _k=top_k, _n=n):
                    val, _ = smooth_obj_grad(x, _b, _k)
                    return val
                def grad_func(x, _b=beta, _k=top_k, _n=n):
                    _, g = smooth_obj_grad(x, _b, _k)
                    return g
                res = minimize(obj_func, h_work, jac=grad_func, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq',
                                             'fun': lambda x, _n=n: np.sum(x) - _n / 2.0,
                                             'jac': lambda x, _n=n: np.ones(_n)}],
                               options={'maxiter': maxiter_base, 'ftol': 1e-13, 'disp': False})
                if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
                    h_new, ok2 = project_feasible(res.x, n)
                    if ok2:
                        s_new = evaluate_exact(h_new)
                        if s_new < cur_score - 1e-13:
                            cur_score = s_new
                            h_work = h_new
                        else:
                            h_work = h_new
            except Exception:
                pass

        if time_left() > 25:
            h_work, _ = project_feasible(h_work, n)
            h_grad, s_grad = gradient_step(h_work, beta=20.0, top_k=max(12, n//3), steps=10)
            if s_grad < cur_score - 1e-13:
                cur_score = s_grad
                h_work = h_grad

        if time_left() > 20:
            h_cd, s_cd = targeted_cd(h_work, n_rounds=50, tol=1e-14, rng=rng)
            if s_cd < cur_score - 1e-14:
                cur_score = s_cd
                h_work = h_cd

        if time_left() > 15:
            h_cd2, s_cd2 = coordinate_descent(h_work, n_rounds=40, tol=1e-14, rng=rng)
            if s_cd2 < cur_score - 1e-14:
                cur_score = s_cd2
                h_work = h_cd2

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_h = h_work.copy()
            best_n = n_c

        if time_left() > 80:
            add_to_pool(h_work + rng.randn(n_c) * 0.015, n_c)
            add_to_pool(h_work + rng.randn(n_c) * 0.05, n_c)
            h_refl2 = 1.0 - h_work[::-1].copy()
            add_to_pool(h_refl2, n_c)
            for nr in resolutions:
                if nr != n_c:
                    h_tr2 = transfer_to_resolution(h_work, nr)
                    add_to_pool(h_tr2, nr)
            pool.sort(key=lambda x: x[0])
            if len(pool) > 18:
                pool = pool[:18]

    if time_left() > 10:
        h_fin, s_fin = targeted_cd(best_h, n_rounds=100, tol=1e-15, rng=rng)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    if time_left() > 8:
        h_fin2, s_fin2 = coordinate_descent(best_h, n_rounds=80, tol=1e-15, rng=rng)
        if s_fin2 < best_score:
            best_score = s_fin2
            best_h = h_fin2

    if time_left() > 6:
        pairs, _ = active_lag_pairs(best_h, top_lags=8)
        for (i, j) in pairs[:20]:
            if time_left() < 3:
                break
            best_h, best_score = fine_tune_pair(best_h, i, j, best_score)

    best_h, ok_final = project_feasible(best_h, best_n)
    if not ok_final:
        best_h = np.full(best_n, 0.5)
    final_score = evaluate_exact(best_h)
    if final_score < best_score + 1e-12:
        best_score = final_score

    return best_h, best_score, best_n
