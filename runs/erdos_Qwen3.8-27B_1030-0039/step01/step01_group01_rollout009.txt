import numpy as np
from scipy.optimize import minimize, brentq
import time
import sys

class TimeLimit(Exception):
    pass

def evaluate_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    """Compute all correlation values (not just the max)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(x, target_sum):
    n = len(x)
    def F(t):
        return np.sum(np.clip(x + t, 0.0, 1.0)) - target_sum
    t_lo = -x.max() - 0.01
    t_hi = 1.0 - x.min() + 0.01
    flo = F(t_lo)
    fhi = F(t_hi)
    if flo >= 0:
        return np.clip(x + t_lo, 0.0, 1.0)
    if fhi <= 0:
        return np.clip(x + t_hi, 0.0, 1.0)
    t = brentq(F, t_lo, t_hi, xtol=1e-15)
    out = np.clip(x + t, 0.0, 1.0)
    diff = target_sum - out.sum()
    if abs(diff) > 1e-13:
        mask = (out > 1e-10) & (out < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            out[mask] += diff / cnt
            out = np.clip(out, 0.0, 1.0)
    return out

def gradient_w_at_k(h, k):
    """Compute gradient of w(k) = sum_n h[n+k]*(1-h[n]) with respect to h.
    ∂w(k)/∂h[i] = (1-h[i-k]) * [0<=i-k<n] - h[i+k] * [0<=i+k<n]
    """
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # (1-h[i-k]) term: valid when i-k >= 0, i.e., i >= k
        indices = np.arange(k, n)
        grad[indices] += (1.0 - h[indices - k])
        # -h[i+k] term: valid when i+k < n, i.e., i < n-k
        indices = np.arange(0, n - k)
        grad[indices] -= h[indices + k]
    else:
        kk = -k
        # (1-h[i-k]) = (1-h[i+kk]) term: valid when i+kk < n, i.e., i < n-kk
        indices = np.arange(0, n - kk)
        grad[indices] += (1.0 - h[indices + kk])
        # -h[i+k] = -h[i-kk] term: valid when i-kk >= 0, i.e., i >= kk
        indices = np.arange(kk, n)
        grad[indices] -= h[indices - kk]
    return grad

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    bc = beta * c
    m = bc.max()
    e = np.exp(bc - m)
    Z = e.sum()
    p = e / Z
    val = (m + np.log(Z)) / beta
    grad = np.zeros(n)
    for j in range(len(c)):
        d = j - (n - 1)
        w = p[j] * dx
        if w < 1e-16:
            continue
        grad += w * gradient_w_at_k(h, d)
    return val, grad

def transfer_to_new_n(h_old, n_new):
    m = len(h_old)
    x_old = np.linspace(0.0, 2.0, m)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, n_new / 2.0)

def run_slsqp_stage(h, deadline, beta_list):
    n = len(h)
    target_sum = n / 2.0
    h_curr = h.copy()
    best_h = h.copy()
    best_score = evaluate_exact(h)
    for beta in beta_list:
        rem = deadline - time.time()
        if rem < 15:
            break
        maxiter = max(30, min(300, int(rem * 3)))
        def _f(x):
            if time.time() > deadline:
                raise TimeLimit()
            v, _ = smooth_obj_grad(x, beta)
            return v
        def _g(x):
            if time.time() > deadline:
                raise TimeLimit()
            _, g = smooth_obj_grad(x, beta)
            return g
        def _c(x):
            return x.sum() - target_sum
        bounds = list(zip(np.zeros(n), np.ones(n)))
        cons = [{'type': 'eq', 'fun': _c}]
        try:
            res = minimize(_f, h_curr, jac=_g, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
                sc = evaluate_exact(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                h_curr = h_new.copy()
        except TimeLimit:
            break
        except Exception:
            break
    return best_h, best_score

def binary_local_search(h, deadline, max_iter=500):
    """Local search on near-binary structure.
    Moves: flip a bit (with compensation via the fractional element), 
    move the fractional element, adjust fractional values.
    """
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_score = evaluate_exact(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        improved = False
        
        # Compute current correlations
        corrs = compute_all_corr(h)
        max_corr = corrs.max()
        max_idx = corrs.argmax()
        
        # Try flipping each position from 0 to 1 or 1 to 0
        # and adjusting the "compensating" position
        for i in range(n):
            if time.time() > deadline:
                return h, best_score
            
            # Try setting h[i] = 1 (if it's not already close to 1)
            if h[i] < 0.5:
                # Need to reduce sum elsewhere by (1 - h[i])
                delta = 1.0 - h[i]
                # Find the best position j to reduce by delta
                for j in range(n):
                    if j == i:
                        continue
                    if h[j] >= delta:
                        h_new = h.copy()
                        h_new[i] = 1.0
                        h_new[j] = h[j] - delta
                        if 0 <= h_new[j] <= 1:
                            sc = evaluate_exact(h_new)
                            if sc < best_score - 1e-12:
                                h = h_new
                                best_score = sc
                                improved = True
                                break
            # Try setting h[i] = 0 (if it's not already close to 0)
            elif h[i] > 0.5:
                delta = h[i]
                for j in range(n):
                    if j == i:
                        continue

                        h_new = h.copy()
