def smooth_grad(h, beta, top_k=None):

    Z = np.sum(w)

    # Gradient

def coordinate_descent(h, n_rounds=100, order='systematic', tol=1e-14):
    n = len(h)
    h = h.copy()
    best_score = evaluate_exact(h)
    
    for rnd in range(n_rounds):
