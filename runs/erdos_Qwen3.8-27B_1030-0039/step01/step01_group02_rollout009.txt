Note that the k=0 term is: (2/n) * Σ a_i(1-a_i) = (2/n)(n/2 - Σa_i²) = 1 - (2/n)Σa_i²

To minimize this, maximize Σa_i². For a_i ∈ [0,1] with Σa_i = n/2, the max of Σa_i² is achieved at the extremes: a_i ∈ {0,1} with exactly n/2 ones. Then Σa_i² = n/2 and k=0 overlap = 0.

So the optimal solution should be close to binary! The k=0 term pushes toward binary, while other terms create competing pressures.

Given this, the optimal is likely a "soft binary" - values very close to 0 or 1, with the pattern optimized to balance the other lags.

This suggests a two-phase approach:
1. Find a good binary pattern (which ones are 1)
2. Smooth it slightly to reduce peak lags

For the binary pattern, we want: minimize max_k |S \ (S-k)| where S is the set of positions with h=1, |S|=n/2.

This is equivalent to: maximize min_k |S ∩ (S-k)| for k≠0.

This is a combinatorial optimization problem. For n=90, we can use simulated annealing or local search on the binary pattern.

    """Log-sum-exp smoothing of the max correlation."""

def evaluate_score(h):
    n = len(h)
    if n < 1:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    if not np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.5)
    h = np.clip(h, -2.0, 3.0)

    def sum_clipped(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target

    lo = -h.max() - 0.5
    hi = 1.0 - h.min() + 0.5
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if sum_clipped(mid) > 0:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-14:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)

            cnt = mask.sum()
            if abs(diff) < cnt * 1e-8:

                result = np.clip(result, 0.0, 1.0)
    return result


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    y = beta * corr
    y_max = y.max()
    w = np.exp(y - y_max)
    Z = np.sum(w)
    psi = (y_max + np.log(Z)) / beta
    p = w / Z
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        weight = p[k] * dx
        if weight < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            upper = n - lag
            if upper > 0:
                grad[:upper] += weight * (-h[lag:n])
                grad[lag:] += weight * (1.0 - h[:upper])
        else:

            upper = n - pn
            if upper > 0:
                grad[:upper] += weight * (1.0 - h[pn:n])
                grad[pn:] += weight * (-h[:upper])
    return psi, grad


def golden_section_minimize(obj_func, a, b, max_iter=30):
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = obj_func(c)
    fd = obj_func(d)
    for _ in range(max_iter):
        if b - a < 1e-14:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / gr
            fc = obj_func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / gr
            fd = obj_func(d)
    best_x = (a + b) / 2.0
    return best_x, obj_func(best_x)
