```python
def find_direction_lp(h, active_lags, trust_radius):
    """
    Find direction d that reduces all active lags simultaneously.
    minimize max_lag grad_lag^T d
    subject to sum(d) = 0, ||d||_inf <= trust_radius, 
              and h + d stays in [0,1] (approximated by box on d)
    """
    n = len(h)
    K = len(active_lags)
    
    G = np.array([compute_grad_for_lag(h, li) for li in active_lags])  # K x n
    
    d = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        d >= -h + 1e-10,  # h + d >= 0
        d <= 1 - h - 1e-10,  # h + d <= 1
    ]
    
    # Trust region (optional, for stability)
    # constraints.append(cp.norm(d, 2) <= trust_radius)
    
    obj = cp.Minimize(t)
    for k in range(K):
        constraints.append(G[k] @ d <= t)
    
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000)
    except:
        try:
            prob.solve(solver=cp.SCS)
        except:
            return None
    
    if d.value is None:
        return None
    
    d_val = np.array(d.value).flatten()
    if not np.all(np.isfinite(d_val)):
        return None
    
    # Project to ensure feasibility
    d_val = np.clip(d_val, -h + 1e-12, 1 - h - 1e-12)
    d_val -= np.mean(d_val)  # ensure sum = 0
    # Re-clip
    d_val = np.clip(d_val, -h + 1e-12, 1 - h - 1e-12)
    # Fix sum approximately
    d_val -= np.mean(d_val)
    
    return d_val
```

            grad[:] = 1.0 - 2.0 * h
        else:
            if n - 1 - p >= 0:
                grad[:n - p] += (1.0 - h[p:])
            if p < n:
                grad[p:] -= h[:n - p]
    else:
        p = -lag
        if p < n:
            grad[p:] += (1.0 - h[:n - p])
        if n - 1 - p >= 0:
            grad[:n - p] -= h[p:]
    return grad * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h, True
    # Use bisection on offset
    def f_offset(t):
        return np.sum(np.clip(h + t, 0.0, 1.0)) - target
    lo = -1.0 - np.max(h)
    hi = 1.0 - np.min(h)
    try:
        t = brentq(f_offset, lo, hi, xtol=1e-15, maxiter=300)
    except Exception:
        # Fallback: distribute to interior points
        h = np.clip(h, 0.0, 1.0)
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            # All at bounds, just add uniformly
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h, True


def active_set_direction(h, K=20):
    """Find a direction that reduces the top K active lags."""
    n = len(h)
    n_lags = 2 * n - 1
    c = get_correlations(h)
    C = np.max(c)
    
    # Find top K lags
    indices = np.argsort(c)[::-1][:K]
    
    # Compute gradients for these lags
    G = np.array([compute_grad_lag(h, idx) for idx in indices])  # K x n
    
    if HAS_CVXPY:
        d = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(d) == 0,
            d >= -h + 1e-12,
            d <= 1.0 - h - 1e-12,
        ]
        for k in range(K):
            constraints.append(G[k] @ d <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                return None
            
        if d.value is None:
            return None
        
        d_val = np.array(d.value).flatten()
        t_val = float(t_var.value)
        
        if t_val >= -1e-10:
            return None  # No descent direction
        
        # Ensure feasibility
        d_val = np.clip(d_val, -h + 1e-12, 1.0 - h - 1e-12)
        d_val -= np.mean(d_val)
        # Re-check feasibility
        h_test = h + d_val
        if np.any(h_test < -1e-10) or np.any(h_test > 1.0 + 1e-10):
            scale = min(1.0, 
                       np.min(np.where(d_val < 0, -h / d_val, np.inf)),
                       np.min(np.where(d_val > 0, (1 - h) / d_val, np.inf)))
            d_val *= max(scale, 0.0)
        
        return d_val
    else:
        # Fallback: use negative of weighted gradient
        # Average gradient of active lags
        g_avg = np.mean(G, axis=0)
        # Project out the mean (to satisfy sum constraint)
        g_avg -= np.mean(g_avg)
        # Scale to respect box constraints
        max_pos = np.max(g_avg[g_avg > 0]) if np.any(g_avg > 0) else 0
        max_neg = np.min(g_avg[g_avg < 0]) if np.any(g_avg < 0) else 0
        scale = 1.0
        if max_pos > 0:
            scale = min(scale, (1 - h)[g_avg > 0].min() / max_pos) if np.any(g_avg > 0) else 1
        if max_neg < 0:
            scale = min(scale, h[g_avg < 0].min() / (-max_neg)) if np.any(g_avg < 0) else 1
        d_val = -g_avg * min(scale, 0.1)
        d_val -= np.mean(d_val)
        h_test = h + d_val
        if np.any(h_test < -1e-10) or np.any(h_test > 1.0 + 1e-10):
            return None
        return d_val
