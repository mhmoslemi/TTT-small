        F = sum_i h_i (1-h_i)
        grad[i] = 1 - 2*h_i
    
    For lag = s > 0 (k = N-1+s):
        F = sum_{i=0}^{N-s-1} h_{i+s} (1-h_i)
        grad[j]:
          if s <= j < N: grad[j] += (1 - h_{j-s})
          if 0 <= j < N-s: grad[j] += -h_{j+s}
    
    For lag = -s (s > 0) (k = N-1-s):
        F = sum_{i=0}^{N-s-1} h_i (1-h_{i+s})
        grad[j]:
          if 0 <= j < N-s: grad[j] += (1 - h_{j+s})
          if s <= j < N: grad[j] += -h_{j-s}
    """
    N = h.size
    g = np.zeros(N)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        m = N - s
        # h_{i+s}(1-h_i) for i=0..m-1
        # h_j appears as h_{i+s} when j = i+s, i = j-s, so j in [s, N-1]:
        g[s:] += (1.0 - h[:m])  # h[j]*(1-h[j-s]) → grad contributes (1-h[j-s])
        # h_j appears as h_i when j = i, so j in [0, m-1]:
        g[:m] += (-h[s:])  # grad contributes -h[j+s]
    else:
        s = -s
        m = N - s
        # h_i(1-h_{i+s}) for i=0..m-1
        # h_j appears as h_i when j = i, j in [0, m-1]:
        g[:m] += (1.0 - h[s:])  # grad contributes (1-h[j+s])
        # h_j appears as h_{i+s} when j = i+s, j in [s, N-1]:
        g[s:] += (-h[:m])  # grad contributes -h[j-s]
    return g

def all_gradients(h):
    """Compute gradients for all lags. Returns array of shape (2N-1, N)."""
    N = h.size
    grads = np.empty((2*N - 1, N))
    for k in range(2 * N - 1):
        s = k - (N - 1)
        grads[k] = compute_gradient(h, s)
    return grads

def tangent_lp(h, active_lags, near_lags, clip_vals, trust_r):
    """
    Solve the tangent LP to find a descent direction.
    
    h: current feasible point
    active_lags: list of lag indices (0 to 2N-2) that are active
    near_lags: list of lag indices that are near-active
    clip_vals: the overlap values (2N-1,)
    trust_r: trust radius
    
    Returns direction d (N,) or None if infeasible.
    """
    N = h.size
    target = 0.5 * N
    n_vars = N + 1  # d_0..d_{N-1}, delta
    
    # Compute gradients for active and near lags
    active_grads = np.array([compute_gradient(h, lag - (N-1)) for lag in active_lags])
    near_grads = np.array([compute_gradient(h, lag - (N-1)) for lag in near_lags])
    
    M = float(np.max(clip_vals))
    
    # Build constraints A_ub @ x <= b_ub
    # x = [d_0, ..., d_{N-1}, delta]
    
    # Active constraints: active_grads @ d + delta <= 0
    # In matrix form: [active_grads, 1] @ x <= 0
    A_active = np.hstack([active_grads, np.ones((len(active_lags), 1))])
    b_active = np.zeros(len(active_lags))
    
    # Near constraints: near_grads @ d <= M - clip_vals[near_lag]
    # In matrix form: [near_grads, 0] @ x <= slack
    slacks = M - clip_vals[near_lags]
    A_near = np.hstack([near_grads, np.zeros((len(near_lags), 1))])
    b_near = slacks
    
    # Combine
    if len(near_lags) > 0:
        A_ub = np.vstack([A_active, A_near])
        b_ub = np.concatenate([b_active, b_near])
    else:
        A_ub = A_active
        b_ub = b_active
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on d_i: [-min(r, h_i), min(r, 1-h_i)]
    # delta: [0, 1] (upper bound on delta to keep things reasonable)
    delta_ub = max(0.01, 5.0 / N)
    bounds = []
    for i in range(N):
        lo_i = -min(trust_r, h[i])
        hi_i = min(trust_r, 1.0 - h[i])
        bounds.append((lo_i, hi_i))
    bounds.append((0.0, delta_ub))
    
    # Objective: minimize -delta
    c_obj = np.zeros(n_vars)
    c_obj[N] = -1.0  # minimize -delta = maximize delta
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            d = res.x[:N].copy()
            delta = res.x[N]
            if delta > 1e-10:
                # Fix the sum to be exactly 0
                d_sum = float(np.sum(d))
                if abs(d_sum) > 1e-10:
                    d -= d_sum / N
                return d, delta
            else:
                return None, 0.0
        else:
            return None, 0.0
    except Exception:
        return None, 0.0

def line_search(h, d, current_c, t0, budget, reserve):
    """
    Exact line search along direction d from point h.
    Returns (best_h, best_c) found during the search.
    """
    N = h.size
    max_alpha = 1.0
    # Check box constraints for the maximum step
    for i in range(N):
        if d[i] > 1e-15:
            max_alpha = min(max_alpha, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            max_alpha = min(max_alpha, h[i] / (-d[i]))
    max_alpha = max(max_alpha, 0.0)
    
    alphas = [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01, 0.005]
    best_h = h.copy()
    best_c = current_c
    
    for a in alphas:
        alpha = a * max_alpha
        if alpha <= 1e-12:
            break
        if remaining(t0, budget) < reserve + 1.0:
            break
        cand = h + alpha * d
        cand = make_feasible(cand)
        c = exact_c5(cand)
        if np.isfinite(c) and c < best_c - 1e-13:
            best_c = c
            best_h = cand.copy()
    
    return best_h, best_c

def pair_local_search(h, rng, max_evals, t0, budget, reserve):
    """Coordinate pair local search."""
    x = make_feasible(h)
    best_h = x.copy()
    best_c = exact_c5(best_h)
    N = x.size
    evals = 0
    fractions = [0.5, 0.25, 0.1, 0.4, 0.15, 0.05]
    
    while evals < max_evals and remaining(t0, budget) > reserve + 5.0:
        i = int(rng.integers(N))
        j = int(rng.integers(N))
        if i == j:
            continue
        hi_val = float(x[i])
        hj_val = float(x[j])
        q = hi_val + hj_val
        for frac in fractions:
            # Try moving mass from j to i
            t_new = hi_val + frac * min(1.0 - hi_val, hj_val)
            if t_new != hi_val:
                cand = x.copy()
                cand[i] = t_new
                cand[j] = q - t_new
                cand = make_feasible(cand)
                c = exact_c5(cand)
                evals += 1
                if np.isfinite(c) and c < best_c - 1e-13:
                    best_c = c
                    best_h = cand.copy()
                    x = cand
                    break
            # Try moving mass from i to j
            t_new = hi_val - frac * min(hi_val, 1.0 - hj_val)
            if t_new != hi_val:
                cand = x.copy()
                cand[i] = t_new
                cand[j] = q - t_new
                cand = make_feasible(cand)
                c = exact_c5(cand)
                evals += 1
                if np.isfinite(c) and c < best_c - 1e-13:
                    best_c = c
                    best_h = cand.copy()
                    x = cand
                    break
            if evals >= max_evals:
                break
    
    return best_h, best_c

def transfer_length(h, new_N):
    h = np.asarray(h, dtype=float).reshape(-1)
    new_N = int(new_N)
    if h.size == new_N:
        return make_feasible(h)
    if h.size == 0 or new_N <= 0:
        return np.full(max(1, new_N), 0.5)
    if new_N % h.size == 0:
        factor = new_N // h.size
        y = np.repeat(h, factor)
        return make_feasible(y)
    if h.size % new_N == 0:
        factor = h.size // new_N
        y = h.reshape(new_N, factor).mean(axis=1)
        return make_feasible(y)
    src = (np.arange(h.size) + 0.5) * (2.0 / float(h.size))
    dst = (np.arange(new_N) + 0.5) * (2.0 / float(new_N))
    y = np.interp(dst, src, h, left=float(h[0]), right=float(h[-1]))
    return make_feasible(y)

def tangent_lp_descent(h, t0, budget, reserve, max_iter=50):
    """
    Main tangent LP descent loop.
    """
    N = h.size
    x = make_feasible(h)
    best_h = x.copy()
    best_c = exact_c5(best_h)
    
    # Trust radius (scales with problem)
    trust_r = 0.05  # initial trust radius
    
    for it in range(max_iter):
        if remaining(t0, budget) < reserve + 10.0:
            break
        
        # Compute overlaps
        corr = compute_overlaps(x)
        clip_vals = corr.copy()
        
        # Identify active and near lags
        M = float(np.max(clip_vals))
        active_tol = max(1e-7, 1e-5 * M)
        near_tol = max(10 * active_tol, 5e-4 * M)
        
        sorted_lags = np.argsort(clip_vals)[::-1]
        
        # Try different active set sizes
        found_improvement = False
        for n_active in [10, 5, 3, 1]:
            if remaining(t0, budget) < reserve + 5.0:
                break
            
            active_lags = sorted_lags[:n_active].tolist()
            # Near lags: next 15
            near_start = n_active
            near_end = min(n_active + 15, 2 * N - 1)
            near_lags = sorted_lags[near_start:near_end].tolist()
            
            d, delta = tangent_lp(x, active_lags, near_lags, clip_vals, trust_r)
            
            if d is not None and delta > 1e-10:
                # Line search
                h_new, c_new = line_search(x, d, best_c, t0, budget, reserve)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = h_new.copy()
                    x = h_new
                    found_improvement = True
                    break
        
        if found_improvement:
            trust_r = min(trust_r * 1.5, 0.2)
        else:
            # No improvement from LP, try smaller trust
            trust_r *= 0.5
            if trust_r < 1e-4:
                break
            
            # Also try without the LP: direct gradient of the max
            # Find the lag achieving the max
            max_lag = int(np.argmax(clip_vals))
            s = max_lag - (N - 1)
            g = compute_gradient(x, s)
            # Project to feasible direction (sum zero, stay in bounds)
            g_mean = float(np.mean(g))
            d_direct = -(g - g_mean)
            # Scale to trust region
            norm_d = float(np.max(np.abs(d_direct)))
            if norm_d > 1e-15:
                d_direct = d_direct / norm_d * trust_r
                h_new, c_new = line_search(x, d_direct, best_c, t0, budget, reserve)
                if c_new < best_c - 1e-13:
                    best_c = c_new
                    best_h = h_new.copy()
                    x = h_new
                else:
                    break
    
    return best_h, best_c
