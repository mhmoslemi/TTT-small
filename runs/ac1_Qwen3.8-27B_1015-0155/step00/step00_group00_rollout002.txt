```python
def _clean_parent(obj):
    """Return a finite, evaluator-consistent copy of the supplied parent."""
    if obj is None:

    """Build a compact nonnegative constant candidate of length n."""
    n = int(n)
    if n <= 0:

    return [float(value)] * n

    try:
        s = float(evaluate_sequence(sequence))
    except Exception:
        return float("inf")
    if not np.isfinite(s):

    return s

    best_score = _safe_score(best)

    # Keep the parent if it is valid and, for any roundoff reason, better.
    if parent is not None and time.monotonic() < deadline:
        cand = [float(x) for x in parent]
        val = _safe_score(cand)
        if val < best_score:
            best = cand
            best_score = val

    # A few constant scales are O(n) to build and can only differ by roundoff
    # in the authoritative score; keep the best exact-scoring candidate.
    for amp in np.array([0.25, 0.5, 1.0, 2.0, 10.0, 1000.0], dtype=np.float64):
        if time.monotonic() >= deadline:
            break
        a = float(max(amp, 0.01 / max(1, n)))
        a = float(min(1000.0, a))
        cand = _constant_sequence(n, a)
        val = _safe_score(cand)
        if val < best_score:
            best = cand
            best_score = val

    if not np.isfinite(best_score):
        best = _constant_sequence(max(1, n), 1.0)

    return [float(x) for x in best]
```
