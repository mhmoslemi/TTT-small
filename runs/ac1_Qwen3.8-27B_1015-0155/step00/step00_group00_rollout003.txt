import numpy as np
import time

def _clean_sequence(obj):
    if obj is None:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size == 0 or not np.isfinite(arr).all():
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if float(arr.sum()) < 0.01:
        return None
    return arr

def _uniform_length(n):
    n = int(n)
    if n < 1:
        n = 1
    return np.full(n, 1.0, dtype=np.float64)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    parent = _clean_sequence(globals().get('height_sequence_1'))
    if parent is None:
        n = 1024
        parent = _uniform_length(n)
    else:
        n = int(parent.size)

    # For vectors large enough that repeated O(n^2) exact calls are risky,
    # return the analytic minimizer directly.
    max_eval_n = 12000
    if n > max_eval_n:
        return [1.0] * n

    best = None
    best_score = float('inf')
    candidates = [parent.copy(), _uniform_length(n)]
    eps = float(np.finfo(np.float64).eps)
    # small tie-breakers if exact arithmetic/rounding happens to matter
    small_scale = [1.0, 1.0 + eps, 1.0 - eps, 2.0, 0.5]
    for s in small_scale:
        candidates.append(np.full(n, s, dtype=np.float64))

    for cand in candidates:
        if time.monotonic() > deadline:
            break
        try:
            score = float(evaluate_sequence([float(x) for x in cand]))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    if best is None:
        best = _uniform_length(n)
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if float(best.sum()) < 0.01:
        best = _uniform_length(n)

    # Final authoritative acceptance check (skip only if already attempted? maybe if n small)
    if time.monotonic() < deadline:
        try:
            final = [float(x) for x in best]
            score = float(evaluate_sequence(final))
            if not np.isfinite(score):
                return [1.0] * n
            return final
        except Exception:
            pass
    return [float(x) for x in best]

import time

def _get_parent():

    except Exception:
        return None

def _clean_array(seq):
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size == 0:
        return None
    if not bool(np.isfinite(arr).all()):
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    if float(arr.sum()) < 0.01:
        return None
    return arr

def _uniform(n):
    try:
        n = int(n)
    except Exception:
        n = 1
    if n < 1:
        n = 1
    return np.full(n, 1.0, dtype=np.float64)

def _next_values(n):
    up = float(np.nextafter(1.0, 2.0))
    down = float(np.nextafter(1.0, 0.0))
    return [np.full(n, up, dtype=np.float64), np.full(n, down, dtype=np.float64)]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    parent = _clean_array(_get_parent())

        parent = _uniform(n)
