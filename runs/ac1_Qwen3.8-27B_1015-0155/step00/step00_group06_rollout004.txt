def parent_vector(n):
    obj = globals().get("height_sequence_1")
    try:
        arr = np.asarray(obj, dtype=np.float64).reshape(-1)
    except Exception:
        arr = np.ones(n, dtype=np.float64)

    if arr.size < n:
        tmp = np.ones(n, dtype=np.float64)
        m = arr.size
        if m > 0:
            tmp[:m] = arr
        arr = tmp
    return arr

def block_perturbation(base, rng):
    cand = base.astype(np.float64, copy=True)
    n = cand.size
    if n <= 1:
        return cand
    width = int(max(1, min(n, int(round(n * 0.005)))))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.10)))
    if rng.random() < 0.5:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = runtime_length()
    uniform = np.ones(n, dtype=np.float64)
    uniform_list = uniform.tolist()
    uniform_score = score_list(uniform_list)
    if np.isfinite(uniform_score) and uniform_score <= 2.0 + OPT_TOL:
        return uniform_list
    # Fallback for an evaluator/numeric situation where the plain constant is
    # not already at the proven lower bound.
    best = parent_vector(n)
    best_list = best.tolist()
    best_score = score_list(best_list)
    if (not np.isfinite(best_score)) or (np.isfinite(uniform_score) and uniform_score < best_score):
        best = uniform
        best_list = uniform_list
        best_score = uniform_score
    if not np.isfinite(best_score):
        best_list = [1.0]
        best_score = score_list(best_list)
    if np.isfinite(best_score) and best_score > 2.0 + OPT_TOL and time.monotonic() < deadline:
        rng = np.random.default_rng(seed)
        while time.monotonic() < deadline:
            cand = block_perturbation(best, rng)
            cl = cand.tolist()
            cs = score_list(cl)
            if np.isfinite(cs) and cs < best_score:
                best = cand
                best_list = cl
                best_score = cs
                if best_score <= 2.0 + OPT_TOL:
                    break
    return [float(x) for x in best_list]

                n = int(len(obj))

def flatten_truncate(obj, n):
    if obj is None:
        return None
    try:
        if hasattr(obj, "size") and int(obj.size) > n:
            return obj.reshape(-1)[:n]
    except Exception:
        pass
    try:
        if len(obj) > n:
            return obj[:n]
    except Exception:
        pass
    return obj


def score_list(lst):
    fn = globals().get("evaluate_sequence")
    if not callable(fn):
        return float("inf")
    try:
        return float(fn(lst))
    except Exception:
        return float("inf")


def parent_vector(kwargs, n):

    try:
        arr = np.asarray(obj, dtype=np.float64).reshape(-1)
    except Exception:

def block_perturbation(base, rng):
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    if n <= 1:
        return cand

    width = int(max(1, min(n, int(round(n * 0.005)))))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.10)))

    if rng.random() < 0.5:
        cand = 0.5 * (cand + cand[::-1])

    return np.clip(cand, 0.0, 1000.0)
