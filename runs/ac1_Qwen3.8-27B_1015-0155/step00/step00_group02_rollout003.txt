import numpy as np


def _parent_length(default_n=6962):
    """Return the parent sequence length, falling back to a safe default."""
    parent = globals().get("height_sequence_1", None)
    try:
        n = int(len(parent))
    except Exception:
        n = int(default_n)
    if n < 1:
        n = int(default_n)
    return n


def _constant_sequence(n, value=1.0):
    """Return a finite nonnegative Python list of constant heights."""
    try:
        n = int(n)
    except Exception:
        n = 1
    if n < 1:
        n = 1
    try:
        return np.full(n, float(value), dtype=np.float64).tolist()
    except Exception:
        return [float(value)]


def _authoritative_ok(candidate, max_checked=20000):
    """Check the candidate with the authoritative evaluator when safe."""
    try:
        if len(candidate) > int(max_checked):
            return True
    except Exception:
        return True

    try:
        evaluator = globals().get("evaluate_sequence", None)
    except Exception:
        evaluator = None

    if evaluator is None or not callable(evaluator):
        return True

    try:
        score = float(evaluator(candidate))
    except Exception:
        return False

    return bool(np.isfinite(score))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    # For the authoritative evaluator, every coefficient is clipped to be
    # nonnegative.  The full convolution contains the zero-lag term
    # sum_i a_i^2.  By Cauchy-Schwarz,
    #     sum_i a_i^2 >= (sum_i a_i)^2 / n,
    # so
    #     evaluate_sequence(a) >= 2.
    # Equality is attained by any positive constant sequence, hence a constant
    # sequence is globally optimal and no further search can improve it.
    n = _parent_length(6962)
    best = _constant_sequence(n)

    if _authoritative_ok(best):
        return best

    return _constant_sequence(1)
