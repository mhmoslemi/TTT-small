def _parent_n():
    h = globals().get("height_sequence_1")
    if h is None: return 1

    except Exception:
        return None

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    n = _parent_n()

        if arr is not None and arr.size == n:

        n = max(1, n // 2)
    if arr is None:
        return [1.0]
    out = arr.tolist()
    try:
        if len(out) <= 20000 and 'evaluate_sequence' in globals():
            s = float(evaluate_sequence(out))
            if np.isfinite(s):
                return out
    except Exception:
        pass
