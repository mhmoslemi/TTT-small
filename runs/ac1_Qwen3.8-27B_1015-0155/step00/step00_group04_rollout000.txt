        lvl = np.floor(np.linspace(0.0, 2.999999, n)).astype(np.int64)
        return np.where(lvl == 0, 1.0, np.where(lvl == 1, 2.0, 0.5))
    out = np.ones(n, dtype=np.float64)
    i = 0
    while i < n:
        width = int(rng.integers(max(1, n // 500), max(2, n // 25)))
        out[i:min(n, i + width)] = float(rng.choice([0.5, 1.0, 2.0]))
        i += width
    return out


def _block_perturb(base, rng):
    n = int(base.size)
    cand = base.copy()
    if rng.random() < 0.5:
        width = int(rng.integers(max(1, n // 400), max(2, n // 8)))
        st = int(rng.integers(0, max(1, n - width + 1)))
        cand[st:st + width] *= float(np.exp(rng.normal(0.0, 0.2)))
    else:
        for _ in range(int(rng.integers(1, 5))):
            st = int(rng.integers(0, n))
            stop = min(n, st + int(rng.integers(1, max(2, n // 150))))
            cand[st:stop] *= float(rng.uniform(0.0, 2.0))
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(20.0, float(budget_s) - 6.0)

    parent = _as_valid(height_sequence_1)
    if parent is None:
        parent = np.ones(1024, dtype=np.float64)
    n = int(parent.size)

    best = _uniform(n, 1.0)
    t0 = time.monotonic()
    best_score = _exact_score(best)
    per_call = max(1e-3, time.monotonic() - t0)

    # Exact-evaluated candidate pool: proves the bound and hunts rounding.
    pool = [_uniform(n, v) for v in (0.5, 2.0, 4.0, 16.0)]
    pm = float(np.mean(parent))
    if np.isfinite(pm) and n * pm >= 0.01:
        pool.append(_uniform(n, pm))
    for k in range(4):
        pool.append(_structured(n, k, rng))
    pool.append(parent)

    for cand in pool:
        if time.monotonic() > deadline:
            break
        s = _exact_score(cand)
        if np.isfinite(s) and s < best_score:
            best, best_score = cand, s

    # Bounded search, active only while floor not reached.
    iters = 0
    while best_score > 2.0 + 1e-12 and time.monotonic() < deadline:
        cand = _block_perturb(best, rng)
        s = _exact_score(cand)
        if np.isfinite(s) and s < best_score:
            best, best_score = cand, s
        iters += 1

    out = np.asarray(best, dtype=np.float64)
    if out.size == 0 or not np.isfinite(out).all() or float(np.sum(out)) < 0.01:
        out = np.ones(max(n, 1), dtype=np.float64)
    out = np.clip(out, 0.0, 1000.0)

    # Final authoritative acceptance, with margin for the sandbox timeout.
    if time.monotonic() + 3.0 * per_call + 0.5 < deadline:
        s = _exact_score(out)
        if not np.isfinite(s):
            out = np.ones(max(n, 1), dtype=np.float64)

    return [float(x) for x in out]
