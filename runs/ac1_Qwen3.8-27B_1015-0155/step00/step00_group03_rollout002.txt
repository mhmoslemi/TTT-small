import numpy as np
import time

_MAX_RETURN_N = 20000

def _valid_score(candidate):
    try:
        value = float(evaluate_sequence(candidate.tolist()))
    except Exception:
        return np.inf

    # at lag 0 and Cauchy's inequality gives score >= 2, with equality for
    # constant height.  Use the parent length and a cheap scale search for

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0

    if parent is None:
        try:
            parent = height_sequence_1
        except NameError:
            parent = None

    if parent is not None:
        try:
            m = int(len(parent))
            if m > 0:
                n = m
        except Exception:
            n = 1024

        n = 1024
    if n > _MAX_RETURN_N:
        n = _MAX_RETURN_N

    try:
        best = np.full(n, 1.0, dtype=np.float64)
    except Exception:
        best = np.full(64, 1.0, dtype=np.float64)

    deadline = time.monotonic() + max(0.0, budget - 20.0)
    best_score = 2.0
    eval_works = False

    if time.monotonic() < deadline:
        score = _valid_score(best)
        if np.isfinite(score):
            best_score = float(score)
            eval_works = True

    if (not eval_works) and time.monotonic() < deadline:
        small = np.full(64, 1.0, dtype=np.float64)
        score = _valid_score(small)
        if np.isfinite(score):
            best = small
            best_score = float(score)
            eval_works = True

    if eval_works and time.monotonic() < deadline:
        size = int(best.size)
        target_ops = 500000000
        max_prov = 64
        try_count = max(0, min(max_prov, target_ops // max(1, size * size)))

        if try_count > 0:
            try:
                lo = 0.02 / max(1, size)
                hi = 1.0
                if lo > 0.0 and lo < hi:
                    grid = np.exp(np.linspace(np.log(lo), np.log(hi), num=try_count))
                else:
                    grid = np.empty(0, dtype=np.float64)
                extras = np.array([1.0, 0.5, 0.2, 0.05, lo], dtype=np.float64)
                scales = np.concatenate((extras, grid))

                scales = np.array([1.0, 0.5, 0.2, 0.05], dtype=np.float64)

                    continue

                    best_score = score
                    best = cand
