                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                cand_score = float("inf")
            if np.isfinite(cand_score) and cand_score < best_score:
                best = cand
                best_score = cand_score

    if best.size == 0 or not np.all(np.isfinite(best)):
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
