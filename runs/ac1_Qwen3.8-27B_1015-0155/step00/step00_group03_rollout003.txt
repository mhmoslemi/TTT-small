def _parent_array():
    raw = globals().get("height_sequence_1", None)
    try:
        arr = np.asarray(raw, dtype=np.float64)
    except Exception:
        return None

        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if float(np.sum(arr)) <= 0.0:
        return None
    return arr


def _parent_length(arr):
    if arr is not None:
        n = int(arr.size)
        if 1 <= n <= 4096:
            return n
    return 2535


def _constant_candidate(n, c):

def _add_amplitude_values(out, base, steps=3):
    x = float(base)
    if not np.isfinite(x) or x <= 0.0 or x > 1000.0:
        return
    out.append(x)
    y = x
    for _ in range(steps):
        y = float(np.nextafter(y, -np.inf))
        if not np.isfinite(y) or y <= 0.0 or y > 1000.0:
            break
        out.append(y)
    y = x

        y = float(np.nextafter(y, np.inf))
        if not np.isfinite(y) or y <= 0.0 or y > 1000.0:
            break
        out.append(y)


def _deterministic_amplitudes():
