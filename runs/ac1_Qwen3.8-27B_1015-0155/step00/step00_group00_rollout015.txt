import numpy as np
import time


def _power_law(n, p, scale=1000.0):
    """Generate a power-law ramp of length n with max value = scale."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return scale * (idx / n) ** p


def _block_perturb(arr, rng, n):
    """Return an O(n) block-perturbed copy of arr."""
    c = arr.copy()
    lo = max(1, n // 500)
    hi = max(lo + 1, n // 20)
    bw = int(rng.integers(lo, hi))
    st = int(rng.integers(0, n - bw + 1))
    f = float(np.exp(rng.normal(0.0, 0.04)))
    c[st:st + bw] *= f
    return np.clip(c, 0.0, 1000.0)
