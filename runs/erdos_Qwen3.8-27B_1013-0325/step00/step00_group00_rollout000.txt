        if 0 <= jk < n:
            g += (1.0 - h[jk])
        if 0 <= jp < n:
            g -= h[jp]
        grad[j] = g * dx
    return grad

def _slsqp_obj(x):
    return x[_CFG_N]

def _slsqp_obj_grad(x):
    g = np.zeros(_CFG_N + 1)
    g[_CFG_N] = 1.0
    return g

def _slsqp_ineq_fun(x):
    n = _CFG_N
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full') * _CFG_DX
    return t - corr

def _slsqp_ineq_jac(x):
    n = _CFG_N
    dx = _CFG_DX
    h = x[:n]
    nc = 2 * n - 1
    jac = np.ones((nc, n + 1))
    for i in range(nc):
        shift = i - (n - 1)
        for j in range(n):
            g = 0.0
            jk = j - shift
            jp = j + shift
            if 0 <= jk < n:
                g += (1.0 - h[jk])
            if 0 <= jp < n:
                g -= h[jp]
            jac[i, j] = -g * dx
    return jac

def _slsqp_eq_fun(x):
    return x[:_CFG_N].sum() - _CFG_TARGET

def _slsqp_eq_jac(x):
    g = np.zeros(_CFG_N + 1)
    g[:_CFG_N] = 1.0
    return g

def _slsqp_optimize(h_start, maxiter=150):
    n = len(h_start)
    _CFG_N = n
    _CFG_DX = 2.0 / n
    _CFG_TARGET = n / 2.0
    h0 = np.clip(h_start.copy(), 0.0, 1.0)
    h0 += (n / 2.0 - h0.sum()) / n
    h0 = np.clip(h0, 0.0, 1.0)
    c5_0 = _eval_c5(h0)
    x0 = np.concatenate([h0, [c5_0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_grad,
                   bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': _slsqp_ineq_fun, 'jac': _slsqp_ineq_jac},
                                {'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None:
        return np.clip(res.x[:n], 0.0, 1.0)
    return h0

def _subgrad_refine(h, n_iter=3000, lr0=1.0):
    n = len(h)
    target = n / 2.0
    h_avg = h.copy()
    h_cur = h.copy()
    best_local = h.copy()
    best_local_c5 = _eval_c5(h_cur)
    for t in range(1, n_iter + 1):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        imax = np.argmax(corr)
        shift = imax - (n - 1)
        grad = _corr_grad_single(h_cur, shift)
        lr = lr0 / np.sqrt(t)
        h_step = _project(h_cur - lr * grad, target)
        h_avg = (t * h_avg + h_step) / (t + 1)
        c5_step = _eval_c5(h_step)
        if c5_step < best_local_c5:
            best_local_c5 = c5_step
            best_local = h_step.copy()
        h_cur = h_step
    c5_avg = _eval_c5(_project(h_avg, target))
    if c5_avg < best_local_c5:
        best_local = _project(h_avg, target)
    return best_local

def _rand_binary(n, rng):
    target = n / 2.0
    h = np.zeros(n)
    whole = int(target)
    frac = target - whole
    idx = rng.choice(n, size=whole + (1 if frac > 0 else 0), replace=False)
    h[idx[:whole]] = 1.0
    if frac > 0:
        h[idx[whole]] = frac
    return h

def _rand_cont(n, rng):
    h = rng.uniform(0.0, 1.0, n)
    return _project(h, n / 2.0)

def _structured(n, rng):
    target = n / 2.0
    f = rng.uniform(0.5, 3.5)
    ph = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * f * x / n + ph)
    return _project(np.clip(h, 0, 1), target)

def _perturb(h, rng, sigma=0.15):
    n = len(h)
    return _project(h + rng.normal(0, sigma, n), n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global _CFG_N, _CFG_DX, _CFG_TARGET
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def _elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals():
        h_init = initial_h_values.astype(np.float64)
    else:
        h_init = None

    best_h = None
    best_c5 = float('inf')

    if h_init is not None and len(h_init) > 0:
        h_try = _project(np.clip(h_init, 0, 1), len(h_init) / 2.0)
        c5_try = _eval_c5(h_try)
        best_h = h_try
        best_c5 = c5_try

    n_pts = len(best_h) if best_h is not None else 45

    if best_h is not None and _elapsed() < budget_s - 30:
        h_opt = _slsqp_optimize(best_h, maxiter=150)
        c5_val = _eval_c5(h_opt)
        if c5_val < best_c5:
            best_h, best_c5 = h_opt, c5_val

    if best_h is not None and _elapsed() < budget_s - 20:
        h_sg = _subgrad_refine(best_h, n_iter=3000, lr0=0.8)
        c5_sg = _eval_c5(h_sg)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg
        if _elapsed() < budget_s - 15:
            h_opt2 = _slsqp_optimize(best_h, maxiter=100)
            c5_v2 = _eval_c5(h_opt2)
            if c5_v2 < best_c5:
                best_h, best_c5 = h_opt2, c5_v2

    # Multi-restart phase
    n_restarts = 80
    for trial in range(n_restarts):
        if _elapsed() > budget_s - 10:
            break
        r = rng.random()
        if r < 0.35:
            h_try = _rand_binary(n_pts, rng)
        elif r < 0.55:
            h_try = _rand_cont(n_pts, rng)
        elif r < 0.75 and best_h is not None:
            h_try = _perturb(best_h, rng, 0.12)
        elif r < 0.90:
            h_try = _structured(n_pts, rng)
        else:
            h_try = _rand_binary(n_pts, rng)
            flip = rng.choice(n_pts, size=rng.randint(3, 8), replace=False)
            h_try[flip] = 1.0 - h_try[flip]
            h_try = _project(h_try, n_pts / 2.0)

        if _elapsed() > budget_s - 15:
            break
        try:
            h_opt = _slsqp_optimize(h_try, maxiter=80)
            c5_opt = _eval_c5(h_opt)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        except Exception:
            pass

        if _elapsed() > budget_s - 20:
            break
        h_sg = _subgrad_refine(h_try if c5_opt >= best_c5 else best_h, n_iter=2000, lr0=0.6)
        c5_sg = _eval_c5(h_sg)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg
            h_opt2 = _slsqp_optimize(best_h, maxiter=60)
            c5_v2 = _eval_c5(h_opt2)
            if c5_v2 < best_c5:
                best_h, best_c5 = h_opt2, c5_v2

    # Try a few other n values
    for n_alt in [35, 50, 55, 40]:
        if _elapsed() > budget_s - 25:
            break
        for attempt in range(4):
            if _elapsed() > budget_s - 15:
                break
            h_try = _rand_binary(n_alt, rng)
            h_try = _perturb(h_try, rng, 0.1)
            try:
                h_opt = _slsqp_optimize(h_try, maxiter=80)
                c5_opt = _eval_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    h_sg = _subgrad_refine(best_h, n_iter=2000, lr0=0.5)
                    c5_sg = _eval_c5(h_sg)
                    if c5_sg < best_c5:
                        best_h, best_c5 = h_sg, c5_sg
            except Exception:
                pass

    # Final polish
    if best_h is not None and _elapsed() < budget_s - 8:
        h_sg = _subgrad_refine(best_h, n_iter=5000, lr0=0.3)
        c5_sg = _eval_c5(h_sg)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg
        if _elapsed() < budget_s - 5:
            h_op = _slsqp_optimize(best_h, maxiter=200)
            c5_op = _eval_c5(h_op)
            if c5_op < best_c5:
                best_h, best_c5 = h_op, c5_op

    # Ensure feasibility
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, n_final / 2.0)
    best_c5 = _eval_c5(best_h)

    return (best_h, best_c5, n_final)
