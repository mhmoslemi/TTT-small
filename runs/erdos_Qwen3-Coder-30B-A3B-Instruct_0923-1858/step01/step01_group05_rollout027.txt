import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Create feasible starting vector
        n_points = 140
        h = np.zeros(n_points)
        # Place most mass in center region
        center_start = n_points // 3
        center_end = 2 * n_points // 3
        h[center_start:center_end] = 1.0
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping with adaptive perturbations
    max_iter = 200
    iter_count = 0
    sigma = 0.05
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h with Gaussian noise
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local optimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6, 'eps': 1e-5})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        iter_count += 1
        # Reduce noise amplitude over time
        if iter_count % 50 == 0:
            sigma /= 2.0
        if iter_count % 10 == 0:
            # Occasionally do a larger stochastic perturbation
            h = np.clip(h + np.random.normal(0, sigma*2, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
    
    # Final local refinement of best solution
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Final validation of constraints
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2.0) > 1e-6:
        # Re-adjust if constraint violated
        best_h = best_h * (n_points / 2.0) / sum_h
    
    # Ensure bounds
    best_h = np.clip(best_h, 0, 1)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
