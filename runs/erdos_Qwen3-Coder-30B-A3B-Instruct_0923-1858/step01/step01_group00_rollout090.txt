import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points if needed
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Binary search bounds
    c_low = 0.0
    c_high = 0.38085
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s and c_high - c_low > 1e-5:
        c_mid = (c_low + c_high) / 2.0
        
        # Formulate LP
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints for all k
        for k in range(n_points):
            # Compute shift for the current k
            shifted = np.roll(h_var, k)
            # Compute overlap integral
            overlap_coeffs = h_var * (1 - shifted)
            overlap = cp.sum(overlap_coeffs) * dx
            constraints.append(overlap <= c_mid)
            
        objective = cp.Minimize(1)  # dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=CPLEX, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Solution exists
                h_new = h_var.value
                # Refine with gradient-based optimization
                h_refined = h_new.copy()
                
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec)
                
                # Bounds for h: [0, 1]
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Constraint for sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2
                
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                # Optimize
                try:
                    result = minimize(
                        objective,
                        h_refined,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=constraint,
                        options={'maxiter': 100},
                        tol=1e-6
                    )
                    if result.success:
                        current_c5 = evaluate_erdos_solution(result.x)
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = result.x.copy()
                        c_high = c_mid
                    else:
                        c_low = c_mid
                except:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid

    # Return final best solution
    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5

    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, actual_c5, n_points)
