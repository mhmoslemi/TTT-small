import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1):
    """Apply small random perturbations to h."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, 0.02)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    child = parent1.copy()
    for i in range(len(child)):
        if np.random.rand() < 0.5:
            child[i] = parent2[i]
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Select parents using tournament selection."""
    selected = []
    for _ in range(num_parents):
        tournament_size = 3
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def ga_optimize(initial_h, dx, n_points, budget_s):
    """Genetic algorithm optimization."""
    pop_size = 50
    generations = 1000
    mutation_rate = 0.1
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = mutate(initial_h, dx)
        population.append(h)
    
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    start_time = time.time()
    gen = 0
    
    while time.time() - start_time < budget_s - 20 and gen < generations:
        # Evaluate all solutions
        fitnesses = [1.0 / (evaluate(h, dx) + 1e-10) for h in population]
        
        # Update best
        current_best_idx = np.argmax(fitnesses)
        current_best_C = 1.0 / fitnesses[current_best_idx]
        if current_best_C < best_C:
            best_C = current_best_C
            best_h = population[current_best_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Create next generation
        new_population = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child1 = crossover(p1, p2)
            child2 = crossover(p1, p2)
            child1 = mutate(child1, dx, mutation_rate)
            child2 = mutate(child2, dx, mutation_rate)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        gen += 1
    
    return best_h, best_C
