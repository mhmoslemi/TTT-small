import numpy as np
import time
from scipy.signal import correlate

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    perturb_rate = 0.03  # 3% of points
    max_iter = 2000
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if abs(current_sum) > 1e-10:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def perturb_solution(h_vec, rate):
        h_new = h_vec.copy()
        indices = np.random.choice(n_points, size=int(n_points * rate), replace=False)
        for i in indices:
            # Modify the value while keeping it feasible
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
        return project_feasible(h_new)
    
    # Main tabu search loop
    iter_count = 0
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = perturb_solution(h, perturb_rate)
        
        # Check tabu list
        candidate_tuple = tuple(candidate_h.round(8))
        if candidate_tuple in tabu_list:
            iter_count += 1
            continue
        
        # Evaluate candidate
        candidate_C5 = compute_C5(candidate_h)
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate_h.copy()
            # Add to tabu list
            tabu_list.append(candidate_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        # Move to best so far solution
        h = candidate_h.copy()
        
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
