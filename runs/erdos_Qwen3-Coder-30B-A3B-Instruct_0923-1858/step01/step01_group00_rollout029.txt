import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better precision
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    omega = 0.7
    c1 = c2 = 1.5
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    global_best = h.copy()
    global_best_score = evaluate_erdos_solution(global_best)
    
    for _ in range(swarm_size):
        particle = h + np.random.normal(0, 0.01, n_points)
        particle = project_to_feasible(particle, n_points)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(particle)
        personal_best.append(particle.copy())
        personal_best_scores.append(score)
        if score < global_best_score:
            global_best = particle.copy()
            global_best_score = score
    
    start_time = time.time()
    
    # PSO main loop
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Low-pass filter the velocity (alpha = 0.9)
            velocities[i] = 0.9 * velocities[i] + 0.1 * np.random.normal(0, 0.005, n_points)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best if needed
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
        
        # Apply global perturbation on particles
        for i in range(swarm_size):
            particles[i] += np.random.normal(0, 0.01, n_points)
            particles[i] = project_to_feasible(particles[i], n_points)
        
        iteration += 1

    # Final projection and evaluation
    final_h = project_to_feasible(global_best, n_points)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, n_points)
