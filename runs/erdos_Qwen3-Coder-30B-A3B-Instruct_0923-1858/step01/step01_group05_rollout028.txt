import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to higher resolution
        if n_points < 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
        elif n_points > 140:
            h = h[:140]
            n_points = 140
    else:
        # Create an initial construction
        n_points = 140
        h = np.ones(n_points) * 0.5
        # Make a smooth distribution
        x = np.linspace(0, 2, n_points)
        h = np.exp(-((x - 1)**2) / 0.1) / np.sum(np.exp(-((x - 1)**2) / 0.1)) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Rescale to correct sum
    h = h * (n_points / 2.0) / np.sum(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_h_saved = best_h.copy()
    best_c5_saved = best_c5
    
    # Basin hopping search
    max_iter = 200
    sigma = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the solution
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    best_h_saved = best_h.copy()
                    best_c5_saved = best_c5
                    h = h_new.copy()
        except:
            pass
            
        # Decrease noise level
        if (iteration + 1) % 50 == 0:
            sigma /= 2
        
        # Occasionally do a larger exploration
        if (iteration + 1) % 20 == 0:
            h = h + np.random.normal(0, sigma * 2, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
    
    # Final local refinement of best solution
    try:
        res = minimize(objective, best_h_saved, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5_saved:
                best_h_saved = h_final.copy()
                best_c5_saved = c5_final
    except:
        pass
    
    # Force final feasibility check
    best_h_saved = np.clip(best_h_saved, 0, 1)
    best_h_saved = best_h_saved * (n_points / 2.0) / np.sum(best_h_saved)
    
    final_c5 = evaluate_erdos_solution(best_h_saved)
    
    return (best_h_saved.tolist(), final_c5, n_points)
