import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluates the Erdős constant C_5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:10] = 0.8
        h[-10:] = 0.2
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Create bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    # Optimization settings
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Adaptive perturbation schedule
    perturbation_magnitude = 0.1
    improvement_count = 0
    stagnation_count = 0
    
    # Random restart optimization
    counter = 0
    
    while time.time() - start_time < budget_s - 1:
        counter += 1
        
        try:
            # Perturb with adaptive size
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            x0 = np.clip(best_h + perturbation, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)  # Renormalize
            
            # Use L-BFGS-B which is efficient and handles bounds well
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                          constraints=cons, options={'maxiter': 100})
            
            if not res.success:
                continue
                
            current_c5 = evaluate_erdos_solution(res.x)
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5
                improvement_count += 1
                stagnation_count = 0
                # Reduce perturbation if improving
                if improvement_count > 5: 
                    perturbation_magnitude = max(0.01, perturbation_magnitude * 0.9)
            else:
                stagnation_count += 1
                # Increase perturbation if stagnating
                if stagnation_count > 10:
                    perturbation_magnitude = min(0.5, perturbation_magnitude * 1.2)
                    stagnation_count = 0
                    improvement_count = 0
                    
        except Exception as e:
            # In case of optimization issues, try with a slightly different perturbation
            pass
            
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final evaluation and trimming
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Make sure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
