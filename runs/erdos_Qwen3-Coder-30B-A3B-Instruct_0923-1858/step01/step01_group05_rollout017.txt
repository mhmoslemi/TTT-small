import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def perturb_and_repair(h, noise_scale=0.03):
    # Apply perturbation in the middle region where overlap tends to be high
    n = len(h)
    h_perturbed = h.copy()
    # Focus on middle region
    start_idx = int(0.3 * n)
    end_idx = int(0.7 * n)
    # Add Gaussian noise to the middle region
    noise = np.random.normal(0, noise_scale, end_idx - start_idx)
    h_perturbed[start_idx:end_idx] += noise
    # Clip and rescale
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = h_perturbed * (n / 2.0) / np.sum(h_perturbed)
    return h_perturbed

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular construction as a better initial guess
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx

    # Local optimization with gradient-based methods
    def optimize_step(h):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Try different local optimizers
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6, 'eps': 1e-6})
            if res.success:
                h_new = res.x
                return h_new
        except:
            pass
        # Fallback to simple perturbation if optimization fails
        return perturb_and_repair(h)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 3000
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Perturb the current solution
        h_trial = perturb_and_repair(h)
        
        # Optimize the perturbed solution
        h_new = optimize_step(h_trial)
        
        # Evaluate and update if better
        c5_new = objective(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
        
        # Update current solution
        h = h_new.copy()
        
        # Occasionally do a global exploration with random perturbations that are more aggressive
        if iter_count % 10 == 0:
            h = perturb_and_repair(h, noise_scale=0.05)
            h = optimize_step(h)
            
        iter_count += 1
        
        # Stop if we haven't improved in a while (early stopping)
        if iter_count - last_improvement > 100:
            break

    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
