import time

    elite_size = 5

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)

        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

        h_vec = np.array(h_vec)

        return np.max(corr) * (2.0 / n_points)

        h = np.random.rand(n_points)
        return project_feasible(h)

        new_individual = create_individual()

def tournament_selection(population, fitness, tournament_size):
    selected_indices = np.random.choice(len(population), tournament_size)
    selected_fitness = [fitness[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitness)]
    return winner_idx

def uniform_crossover(parent1, parent2):
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate_child(child, mutation_rate):
    for i in range(len(child)):
        if np.random.rand() < mutation_rate:
            # Add Gaussian noise
            delta = np.random.normal(0, 0.05)
            child[i] += delta
