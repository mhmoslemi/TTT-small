import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Try low resolution first (n=50) 
    if n_points > 50:
        h_low = h[:50].copy()

            if c5_trial < best_c5:

        # Upsample and refine 
        h = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, 50), best_h)
        h = h * (100 / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)

            return np.sum(h_vec) - n_points / 2

    return (list(best_h), final_c5, n_points)
