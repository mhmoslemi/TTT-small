import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

        h[0] = 0.0
        h[-1] = 0.0
    
    # Normalize
    h = h / np.sum(h) * (n_points / 2)
    
    # Gradient descent with projection
    alpha = 0.1
    stagnation_counter = 0
    prev_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute current value
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Update step
        corr = correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        k_max = np.argmax(overlaps)
        # Derivative approximation for gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = i - (n_points - 1) // 2 + k_max
            if 0 <= k < len(overlaps):
                grad[i] += overlaps[k] * (1 - 2 * h[i])
        # Slightly more precise gradient for max overlap term
        max_lag = np.argmax(overlaps)
        lag = max_lag - (n_points - 1) // 2
        for i in range(n_points):

            grad[i] += (1 - 2 * h[i]) * dx

        h = h_new
        stagnation_counter += 1
        
        # Adapt learning rate
        if current_c5 < prev_c5:
            stagnation_counter = 0
        else:
            stagnation_counter += 1

            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            stagnation_counter = 0
            alpha = min(alpha * 0.95, 1e-3)
