import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

            lambda_var = cp.Variable()

            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # Add overlap constraints
            for k in range(n_points):
                # Compute overlap for shift k, generating the appropriate indices
                # In the original code, the shift is from left to right, but for 
                # full correlation with wrap-around, we compute manually
                overlap_expr = 0
                if k == 0:
                    overlap_expr = cp.sum(cp.multiply(h_var, 1 - h_var))
                else:
                    # Handle wrap-around in correlation calculation manually
                    h_lags = []
                    one_minus_h_lags = []
                    for i in range(n_points - k):
                        h_lags.append(h_var[i])
                        one_minus_h_lags.append(1 - h_var[i + k])
                    for i in range(k):
                        h_lags.append(h_var[i])
                        one_minus_h_lags.append(1 - h_var[i + n_points - k])
                    overlap_expr = cp.sum(cp.multiply(h_lags, one_minus_h_lags)) 
                
                # Original overlap is approximated but modified for simplicity
                # This is a conservative convex approximation; exact corresponds to nonconvex problem
                constraints.append(overlap_expr >= lambda_var)
            
            # Objective
            objective = cp.Maximize(lambda_var)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == 'optimal':
                h = np.array(h_var.value).reshape(-1)
                # Project to feasible region
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            pass
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")

        eps = 1e-5

            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

            eta *= 0.95
        
        iter_count += 1

    # Final validation

    if final_C5 < best_C5:

    return (best_h.tolist(), best_C5, len(best_h))
