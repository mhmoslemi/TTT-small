import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint function for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Define constraints for differential evolution
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Start with a coarse resolution for faster exploration
    coarse_n = 50
    if n_points > coarse_n:
        # Create a coarse version
        coarse_h = np.interp(np.linspace(0, 1, coarse_n), 
                            np.linspace(0, 1, n_points), h)
        coarse_h = coarse_h * (coarse_n / 2) / np.sum(coarse_h)
        coarse_h = np.clip(coarse_h, 0, 1)
    else:
        coarse_h = h.copy()
        coarse_n = n_points
    
    # Differential Evolution for global exploration
    de_result = differential_evolution(
        objective,
        bounds,
        args=(),
        maxiter=200,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        atol=1e-6,
        tol=1e-6,
        constraints=constraints,
        seed=seed,
        workers=1,
        updating='deferred',
        disp=False
    )
    
    # Refine the best solution from DE
    if de_result.success:
        refined_h = np.clip(de_result.x, 0, 1)
        refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
        
        # Local SLSQP optimization
        try:
            res = minimize(
                objective,
                refined_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                final_h = np.clip(res.x, 0, 1)
                final_h = final_h * (n_points / 2) / np.sum(final_h)
                final_c5 = evaluate_erdos_solution(final_h)
            else:
                final_h = refined_h
                final_c5 = evaluate_erdos_solution(final_h)
        except Exception:
            final_h = refined_h
            final_c5 = evaluate_erdos_solution(final_h)
    else:
        final_h = h
        final_c5 = evaluate_erdos_solution(h)
    
    # Check if we want to enhance further
    best_c5 = final_c5
    best_h = final_h.copy()
    
    # Try to upsample and refine if needed and time allows
    if n_points < 100 and (time.time() - start_time) < budget_s - 5:
        # Upsample to higher resolution
        new_n = 100
        upsampled_h = np.interp(np.linspace(0, 1, new_n), 
                              np.linspace(0, 1, n_points), best_h)
        upsampled_h = upsampled_h * (new_n / 2) / np.sum(upsampled_h)
        upsampled_h = np.clip(upsampled_h, 0, 1)
        
        # Refine with SLSQP
        try:
            res = minimize(
                objective,
                upsampled_h,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(new_n)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - new_n / 2},
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                new_best_h = np.clip(res.x, 0, 1)
                new_best_h = new_best_h * (new_n / 2) / np.sum(new_best_h)
                new_c5 = evaluate_erdos_solution(new_best_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_best_h
                    n_points = new_n
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
