import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.9999
    min_temp = 1e-8
    perturb_scale = 0.05
    
    iteration = 0

        h_candidate = h.copy()
        
        # Choose two random indices to move mass between
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Amount to move (small random value)
        delta = np.random.uniform(-perturb_scale, perturb_scale)
        
        # Ensure we don't go out of bounds
        if h_candidate[i] + delta < 0:
            delta = -h_candidate[i]
        elif h_candidate[i] + delta > 1:
            delta = 1 - h_candidate[i]
            
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Normalize to ensure sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)

        h_candidate = np.clip(h_candidate, 0, 1)

        candidate_c5 = evaluate_erdos_solution(h_candidate)

        if candidate_c5 < best_c5:

            # Accept worse solution with probability
            prob = np.exp(-(candidate_c5 - best_c5) / T)
            if np.random.random() < prob:
                h = h_candidate.copy()
        
        # Cool down temperature
        T = max(T * cooling_rate, min_temp)
        
        # Early termination check
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
