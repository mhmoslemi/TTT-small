import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # High resolution for better accuracy
    n_points = 140
    dx = 2.0 / n_points
    
    # Initialize from provided construction or create a starting point
    if 'initial_h_values' in globals():
        # Interpolate to higher resolution
        h = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
    else:
        # Start with a reasonable construction
        h = np.ones(n_points) * 0.5
        # Add some clustering
        mid = n_points // 2
        h[mid-10:mid+10] = 1.0
    
    # Scale to correct sum
    h = h / np.sum(h) * (n_points / 2.0)
    h = np.clip(h, 0, 1)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1-h_vec, mode="full")
        return np.max(corr) * dx
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin hopping with adaptation
    iter_count = 0
    max_iter = 200
    noise_scale = 0.05
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Create perturbed version
        h_trial = h + np.random.normal(0, noise_scale, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local optimization with SLSQP
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        iter_count += 1
        if iter_count % 50 == 0:
            noise_scale *= 0.5  # Reduce noise over time
    
    # Final refinement of top candidates
    candidate_list = [best_h]
    for _ in range(5):
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
    
    # Ensure final solution is valid
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
