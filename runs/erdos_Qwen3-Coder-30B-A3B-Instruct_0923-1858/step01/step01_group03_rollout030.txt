import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    budget_s *= 0.95  # Leave margin
    
    # Get initial solution if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h0 = np.maximum(0, 1 - np.abs(x - 1))
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Genetic Algorithm parameters
    pop_size = 100
    num_generations = 500
    mutation_rate = 0.15
    tournament_size = 5
    elite_size = 2
    
    # Initialize population
    population = []
    population.append(h0)
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = h0.copy()
    best_C5 = float('inf')
    
    # Helper function to evaluate C5
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # Helper function to project feasible
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Main Evolution Loop
    for gen in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            fit = compute_C5(h)
            fitness_scores.append(fit)
            if fit < best_C5:
                best_C5 = fit
                best_h = h.copy()
        
        # Tournament Selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            best_idx = tournament[np.argmin([fitness_scores[i] for i in tournament])]
            selected.append(population[best_idx].copy())
        
        # Create offspring with crossover and mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate rest of population
        while len(new_population) < pop_size:
            parent1 = selected[np.random.randint(0, pop_size)]
            parent2 = selected[np.random.randint(0, pop_size)]
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(0, n_points)
                child[idx] += np.random.normal(0, 0.05)
            
            child = project_feasible(child)
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
