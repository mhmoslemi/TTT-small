import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points for better resolution
        if n_points != 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            h = np.clip(h, 0, 1)
            n_points = 140
    else:
        # Create a feasible random h vector
        n_points = 140
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Scale to sum to 70 (since 140 * 2/140 = 1)
    h = h / np.sum(h) * (n_points / 2.0)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)  
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 1000
    iter_count = 0
    perturb_strength = 0.05
    
    # Keep track of top solutions for local refinement
    top_solutions = []
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturbation phase - global exploration
        h_new = h + np.random.normal(0, perturb_strength, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2.0)
        
        c5_new = objective(h_new)
        
        # Store top solutions for later local refinement
        top_solutions.append((c5_new, h_new.copy()))
        top_solutions.sort(key=lambda x: x[0])
        top_solutions = top_solutions[:10]  # Keep top 10
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
            
        # Local refinement phase - refine top solutions
        for i in range(min(5, len(top_solutions))):  # Refine top 5 or all
            if time.time() - start_time > budget_s - 1:
                break
            h_trial = top_solutions[i][1]
            try:
                res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
                        h = h_refined.copy()
            except:
                pass
        
        iter_count += 1
        # Reduce perturbation strength over time for focused search
        if iter_count % 50 == 0:
            perturb_strength *= 0.5
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Clip h to [0,1] and verify sum constraint one last time
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2.0)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
