import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 75
    max_iter = 2000
    perturb_rate = 0.04  # 4% of points
    max_perturb = max(1, int(n_points * perturb_rate))
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def generate_neighbor(current_h, max_perturb):
        candidate = current_h.copy()
        num_perturb = np.random.randint(1, max_perturb + 1)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            candidate[idx] = np.random.uniform(0, 1)
        return project_feasible(candidate)
    
    def hash_solution(h_vec):
        return tuple(np.round(h_vec, 6))
    
    # Initialize
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5
    
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Generate neighbor
        neighbor = generate_neighbor(current_h, max_perturb)
        neighbor_C5 = compute_C5(neighbor)
        
        # Check if neighbor is better or not in tabu list
        neighbor_hash = hash_solution(neighbor)
        if neighbor_C5 < best_C5 or neighbor_hash not in tabu_list:
            current_h = neighbor
            current_C5 = neighbor_C5
            
            if neighbor_C5 < best_C5:
                best_h = neighbor.copy()
                best_C5 = neighbor_C5
            
            # Add to tabu list
            tabu_list.append(neighbor_hash)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
