import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():

            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Create a feasible h vector
        n_points = 140
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly 70 (since 2/140 * 70 = 1)
    h = h / np.sum(h) * 70.0
    
    # Define objective function for optimization
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Global exploration with perturbation
    num_perturbations = 50
    perturbation_strength_start = 0.05
    max_iter = 1000
    iter_count = 0
    
    # Main optimization loop
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Global perturbation step
        perturbed_solutions = []
        for i in range(num_perturbations):
            if time.time() - start_time > budget_s - 1:
                break
            noise = np.random.normal(0, perturbation_strength_start, n_points)
            h_trial = h + noise
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * 70.0  # Keep sum = 70
            c5_trial = objective(h_trial)
            perturbed_solutions.append((c5_trial, h_trial.copy()))
            
        # Sort by objective value (ascending)
        perturbed_solutions.sort(key=lambda x: x[0])
        
        # Refine top solutions using local optimization
        num_refinements = min(10, len(perturbed_solutions))
        for i in range(num_refinements):
            c5_val, h_trial = perturbed_solutions[i]
            
            # Define bounds and constraint
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - 70.0
                
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                # SLSQP local optimization
                res = minimize(objective, h_trial, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except Exception:
                pass
                
        # Update h with best solution from perturbations

            _, h = perturbed_solutions[0]
            # Ensure h is valid after update
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * 70.0

        # Reduce perturbation strength over time
        if iter_count % 50 == 0 and perturbation_strength_start > 1e-5:
            perturbation_strength_start *= 0.5
        
        # Occasionally restart with random exploration
        if iter_count % 100 == 0 and iter_count > 0:
            h = np.clip(h + np.random.normal(0, 0.03, n_points), 0, 1)
            h = h / np.sum(h) * 70.0
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
