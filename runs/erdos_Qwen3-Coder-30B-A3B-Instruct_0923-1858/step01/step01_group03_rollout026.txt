import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation approach - initialize with CVXOPT
    try:
        # Create the convex optimization problem to find a lower bound
        h_cvx = Variable(n_points)
        # Objective: maximize minimum overlap
        lambda_var = Variable()
        constraints = [h_cvx >= 0, h_cvx <= 1]
        
        # Add constraint for each overlap
        for k in range(n_points):
            # Compute indices for the convolution
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_cvx[i] * (1 - h_cvx[j])
            constraints.append(expr >= lambda_var)
        
        # Sum constraint
        constraints.append(sum(h_cvx) == n_points / 2)
        
        prob = Problem(Maximize(lambda_var), constraints)
        prob.solve(solver=ECOS, verbose=False)
        
        if prob.status == 'optimal':
            h = np.array(h_cvx.value).reshape(-1)
            # Project back to feasible set
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Main optimization loop
    iter_count = 0
    eta = 0.01
    patience = 0
    max_patience = 100
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
    
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1
    
    # Try random restarts
    for _ in range(5):
        h_new = np.random.rand(n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        current_C5 = compute_C5(h_new)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_new.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
