import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 140
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensuring bounds
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Constraint: sum = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Perturbation parameters
    perturbation_strength = 0.05
    perturbation_decay = 0.5
    max_iter = 1000
    
    # Global search loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturbation
        h_trial = h + np.random.normal(0, perturbation_strength, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Evaluate
        c5_trial = objective(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
            
        # Local refinement - 10 best candidates
        candidates = []
        for _ in range(10):
            h_trial = h + np.random.normal(0, perturbation_strength, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            candidates.append(h_trial)
        
        # Refine candidates with SLSQP
        for h_trial in candidates:
            try:
                res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass
                
        # Decay perturbation
        if i % 50 == 0 and i > 0:
            perturbation_strength *= perturbation_decay
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
