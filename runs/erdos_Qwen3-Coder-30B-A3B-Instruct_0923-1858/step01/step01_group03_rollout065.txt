import time

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    stagnation_limit = 50
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal best and global best
    pbest = [swarm[i].copy() for i in range(Pop_size)]
    pbest_fitness = [float('inf')] * Pop_size
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Compute initial fitness
    for i in range(Pop_size):
        overlaps = correlate(swarm[i], 1 - swarm[i], mode="full") * dx
        fitness = np.max(overlaps)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    # Track best overall
    if gbest_fitness < best_C5:
        best_C5 = gbest_fitness
        best_h = swarm[gbest_idx].copy()
    
    iter_count = 0
    stagnation = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Update particles
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Update personal best
            overlaps = correlate(swarm[i], 1 - swarm[i], mode="full") * dx
            fitness = np.max(overlaps)
            
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
                    stagnation = 0
                    
                    if fitness < best_C5:
                        best_C5 = fitness
                        best_h = swarm[gbest_idx].copy()

        stagnation += 1
        if stagnation > stagnation_limit:
            for i in range(Pop_size // 10):  # Reset 10% of particles
                idx = np.random.randint(Pop_size)
                swarm[idx] = np.random.rand(n_points)
                swarm[idx] = swarm[idx] * (n_points / 2) / np.sum(swarm[idx])

                pbest_fitness[idx] = float('inf')
            stagnation = 0
            
        w = 0.7 + (0.3 * iter_count / max_iter)  # linearly decrease w
        
        iter_count += 1
    
    # Final evaluation
    final_overlaps = correlate(best_h, 1 - best_h, mode="full") * dx
    final_C5 = np.max(final_overlaps)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
