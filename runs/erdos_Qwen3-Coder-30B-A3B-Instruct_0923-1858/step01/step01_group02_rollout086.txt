import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h onto feasible set: sum = n_points/2, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 80
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Target: gradually lower the bound
    target_c5 = 0.4
    
    # Enable adaptive adjustment for better performance
    step_size = 0.0001
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Increase target with convex optimization
        # Problem: minimize max_k integral over all shifts
        # We'll add constraints and iteratively tighten

        # Create decision variables
        h_var = cp.Variable(n_points)
        # Constraint: integral for each shift should be <= target_c5 / dx
        dx = 2.0 / n_points

        # Create constraint matrix for all shifts
        constraints = []
        # Ensure bounds and sum
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        constraints.append(cp.sum(h_var) == n_points / 2)

        # Add integral constraints for each valid shift
        for k in range(n_points):
            # We compute sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k])
            # Use indexing to handle circular shifts
            shifted_indices = np.arange(k, k + n_points) % n_points
            integral_expr = cp.sum(cp.multiply(h_var, 1 - h_var[shifted_indices]))
            constraints.append(integral_expr <= target_c5 / dx)

        # Objective: minimize target_c5
        objective = cp.Minimize(target_c5)

        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Pull the new solution
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Compute new C5 value
                test_c5 = evaluate_erdos_solution(h_new)
                if test_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = test_c5
                target_c5 = target_c5 - step_size
            else:
                # If infeasible, increase target
                target_c5 = target_c5 + step_size
                step_size = min(step_size * 2, 0.01)
        except Exception:
            target_c5 = target_c5 + step_size
            step_size = min(step_size * 2, 0.01)
        
        if iteration % 20 == 0:
            # Periodic reevaluation of best solution
            current_c5 = evaluate_erdos_solution(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                
        # Early exit if we're very close to target
        if target_c5 < 0.3807:
            break
            
        # Early termination based on budget
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
