import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    max_iter = 1000
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < max_iter:
        try:
            # Gradient-based optimization
            result = minimize(
                lambda x: evaluate_erdos_solution(x),
                h.copy(),
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 100},
                tol=1e-6
            )

            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()

                # Guide optimization with a slight perturbation
                perturbation = np.random.normal(0, 0.01, n_points)
                h = result.x + perturbation
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
            else:
                # Fallback if optimization fails
                h = np.random.rand(n_points) * 0.5
                h = h / np.sum(h) * (n_points / 2)

        except Exception as e:
            # Reinitialize on failure
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)

        iter_count += 1

    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except:
        final_c5 = best_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, final_c5, n_points)
