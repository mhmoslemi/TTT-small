import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize timing
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    # Helper function to compute C5
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Project solution back to feasible region
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Convex relaxation for initial solution
    def convex_relaxation():
        # Set up a convex relaxation of the problem
        h_vars = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add constraints for the correlation matrix
        # We approximate the maximum correlation by solving this relaxation
        # For simplicity, just ensure sum of h = n_points/2 and bounds
        objective = cp.Maximize(lambda_var)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL:
                return project_feasible(h_vars.value)
        except:
            pass
        # Fallback to random projection
        return project_feasible(np.random.rand(n_points))

    # Initialize with convex relaxation
    h = convex_relaxation()

    # Calculate initial C5
    current_C5 = compute_C5(h)
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = h.copy()

    # Optimization using scipy minimize with trust-constr
    def objective(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return -np.max(corr) * dx  # negative because we minimize

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])  # 0 <= h <= 1

    # Create bounds for the optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    initial_h = h.copy()

    # Optimization loop
    patience = 0
    max_patience = 100
    iter_count = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Optimize using trust-constr
        try:
            result = minimize(
                objective, 
                initial_h, 
                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'disp': False}
            )
            
            if result.success:
                h = project_feasible(result.x)
                current_C5 = compute_C5(h)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        except:
            patience += 1

        # Perturb if no improvement
        if patience > 0 and patience % 10 == 0:
            idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[idx] += np.random.normal(0, 0.05, len(idx))
            h = project_feasible(h)
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1

        # Early exit condition
        if patience > max_patience:
            break
            
        iter_count += 1
        if iter_count > 10000:
            break

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
