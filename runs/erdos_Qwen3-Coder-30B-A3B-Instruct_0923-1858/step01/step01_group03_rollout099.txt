import time
from scipy.signal import correlate
import cvxpy as cp

        n_points = 45

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Setup for convex relaxation
    def cvx_relaxation(h_start, n_points):
        # Create variables
        h_vars = cp.Variable(n_points)
        # Constraint: sum(h) = n_points / 2
        sum_constraint = cp.Equality(cp.sum(h_vars), n_points / 2)
        # Constraints: 0 <= h[i] <= 1
        bound_constraints = [cp.constraints.NonPos(1 - h_vars), 
                             cp.constraints.NonPos(h_vars)]
        
        # Relaxation: maximize a scalar lambda
        lambda_var = cp.Variable()
        
        # Create constraints for each shift k
        constraints = [lambda_var <= cp.sum(h_vars[i] * (1 - h_vars[i+k]) 
                                           for i in range(n_points - k)) 
                       for k in range(1, n_points)]
        constraints.extend([lambda_var <= cp.sum(h_vars[i] * (1 - h_vars[i-k]) 
                                                for i in range(k, n_points)) 
                           for k in range(1, n_points)])
        constraints.append(sum_constraint)
        constraints.extend(bound_constraints)
        
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return np.array(h_vars.value)
        except:
            pass
        return h_start
    
    # Main optimization loop
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    patience = 0
    max_patience = 100
    
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
        
        # Gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        max_k_idx = np.argmax(correlate(h, 1-h, mode="full"))
        max_k = max_k_idx - (n_points - 1)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        # Update using gradient
        lr = 0.01
        h = h - lr * grad
        h = project_feasible(h)
        
        iter_count += 1
        
        if iter_count % 200 == 0 and iter_count > 0:
            # Occasionally try convex relaxation for better start
            h = cvx_relaxation(h, n_points)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
