import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better precision
    n_points = 200
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        # Ensure boundary values are zero
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    perturb_std = 0.01
    alpha = 0.9  # low-pass filter parameter
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    global_best_idx = 0
    global_best_score = float('inf')
    
    for _ in range(swarm_size):
        # Random initialization within bounds
        particle = np.random.rand(n_points) * 0.5
        particle = particle / np.sum(particle) * (n_points / 2)
        particle = np.clip(particle, 0, 1)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best.append(particle.copy())
        personal_best_scores.append(float('inf'))
    
    # Initialize global best
    for i in range(swarm_size):
        score = evaluate_erdos_solution(particles[i])
        if score < global_best_score:
            global_best_score = score
            global_best_idx = i
    
    # Main PSO loop
    start_time = time.time()
    last_improvement = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                omega * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (personal_best[global_best_idx] - particles[i])
            )
            
            # Apply low-pass filter
            velocities[i] = alpha * velocities[i] + (1 - alpha) * np.roll(velocities[i], 1)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility projection
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
            
            # Evaluate objective
            score = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_idx = i
                last_improvement = iter_count
            
            # Apply perturbations to encourage exploration
            if iter_count % 50 == 0:
                for j in range(swarm_size):
                    particles[j] += np.random.normal(0, perturb_std, n_points)
                    particles[j] = np.clip(particles[j], 0, 1)
                    particles[j] = particles[j] / np.sum(particles[j]) * (n_points / 2)
        
        # Stop if no improvement in last 1000 iterations
        if iter_count - last_improvement > 1000:
            break
        if iter_count % 100 == 0:
            # Progress report
            pass
    
    # Final validation
    best_h = personal_best[global_best_idx]
    best_c5 = global_best_score
    
    # Final refinement with local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)
        
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 100},
            tol=1e-6
        )
        if result.success:
            refined_c5 = evaluate_erdos_solution(result.x)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = result.x.copy()
    except:
        pass

    return (best_h, best_c5, n_points)
