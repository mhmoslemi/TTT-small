import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex optimization approach
    def convex_initialization():
        # Convex surrogate: maximize lambda subject to overlap constraints
        h_var = cp.Variable(n_points)
        lambda_var = cp.Variable(1)
        
        # Constraints
        constraints = []
        # 0 <= h <= 1
        constraints += [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        # sum(h) = n_points / 2
        constraints += [cp.sum(h_var) == n_points / 2]
        
        # Overlap constraints
        for k in range(n_points):
            # The overlap O_k = sum_i h[i] * (1 - h[i+k])*dx
            # Wrap-around: for negative indices, use the corresponding positive index
            # Approximate by estimating correlation up to n_points-1
            # But it's easier to consider full correlation for all valid lags
            # Here, we consider up to the effective number of lags we can compute
            # We'll use a safe, large number of lags just to be thorough
            # Maximum lags we can consider with full overlap
            # Actually, this would be simpler using a full correlation approach
            # Let's think:
            # The shift k in [0, n_points-1] -> correlation of length 2*n_points-1
            # For lag j, we compute sum_{i} h[i] * (1 - h[i+j]) *
            # But since we're constrained to a finite interval, 
            # the convolution/dot product will catch each k in [0, n_points-1]
            # For finite discrete cases, 
            # O_k = \sum_i h[i] * (1 - h[i+k]) where if i+k overflows, wrap around
            # But to make it an infinite extent, we can pad and be careful with indexing
            
            # Inserting padding here
            h_padded = np.concatenate([h_var, np.zeros(n_points-1)])
            # Determine the required overlap at shift k
            lag_sum = 0
            for i in range(n_points):
                # Overlaps with wraparound allowed
                idx_k = (i + k) % n_points
                lag_sum += h_var[i] * (1 - h_var[idx_k])
            constraints += [lag_sum >= lambda_var]
        
        # Objective
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                print("Found by convex optimization")

        except Exception as e:

        init_time = time.time()

            h = convex_initialization()
        except:
            pass

            budget_s -= init_time  # small down-time

        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    patience = 0
    max_patience = 50
    iter_count = 0
    
    def get_gradient(h_vec):
        # Compute correlation
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Simplified: gradient for the max_k index only
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                if np.random.random() < 0.5:
                    # Random perturbation
                    idx = np.random.choice(n_points)
                    h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                    h = project_feasible(h)
                patience = 0
                
        # Update step with gradient
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Adaptive learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    # Random restart refinement
    for _ in range(10):
        if time.time() - start_time >= budget_s * 0.9:
            break
        
        h_restart = np.random.rand(n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        current_C5 = compute_C5(h_restart)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_restart.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
