import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_h(n_points):
    """Initialize h with a triadic waveform pattern."""
    h = np.zeros(n_points)
    for i in range(n_points):
        phase = np.abs(np.sin(np.pi * i / (n_points / 3))) ** 3
        h[i] = np.clip(2 * phase - 0.5, 0, 1)
    return project(h, 2.0 / n_points)

def gradient_estimate(h, dx, epsilon=1e-8):
    """Estimate gradient of C5 with respect to each h[i]."""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project(h_plus, dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        max_plus = np.max(corr_plus)

        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project(h_minus, dx)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        max_minus = np.max(corr_minus)

        grad[i] = (max_plus - max_minus) / (2 * epsilon)
    return grad

def optimize_h(h, dx, n_points, time_budget):
    """Optimize h using scipy's L-BFGS-B method."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    def objective(x):
        h_temp = project(x, dx)
        return evaluate(h_temp, dx)
    
    def gradient(x):
        h_temp = project(x, dx)
        return gradient_estimate(h_temp, dx)
    
    # Reshape h for scipy optimization
    initial_x = h.copy()
    
    # Create bounds (0, 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            objective,
            initial_x,
            method='L-BFGS-B',
            bounds=bounds,
            jac=gradient,
            options={'ftol': 1e-9, 'gtol': 1e-9, 'maxiter': 10000}
        )
        
        if result.success and time.time() - start_time < time_budget:
            final_h = project(result.x, dx)
            final_c5 = evaluate(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
                
    except Exception as e:
        pass
        
    # Perform additional local search
    h_current = best_h.copy()
    for i in range(10000):
        if time.time() - start_time > time_budget:
            break
            
        grad = gradient_estimate(h_current, dx)
        h_current = h_current - 0.01 * grad
        h_current = project(h_current, dx)
        
        current_c5 = evaluate(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Occasionally perturb to avoid local minima
        if i % 500 == 0:
            # Add small random noise
            noise = 0.001 * np.random.randn(n_points)
            h_current += noise
            h_current = project(h_current, dx)
    
    return best_h, best_c5
