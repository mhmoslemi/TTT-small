        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Genetic algorithm parameters
    pop_size = 100
    num_generations = 200
    mutation_rate = 0.15
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def create_individual():
        # Create a new individual with random values, normalized to sum to n_points / 2
        individual = np.random.rand(n_points)
        individual = individual * (n_points / 2) / np.sum(individual)
        return np.clip(individual, 0, 1)
    
    def project_feasible(individual):
        # Clip between 0 and 1
        individual = np.clip(individual, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (n_points / 2) / current_sum
        return individual
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Create initial population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        individual = create_individual()
        population.append(project_feasible(individual))
    
    # Evolution loop
    no_improvement_count = 0
    generation = 0
    
    while generation < num_generations:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate fitness for entire population
        fitness = []
        for ind in population:
            fit = compute_C5(ind)
            fitness.append(fit)
            if fit < best_C5:
                best_C5 = fit
                best_h = ind.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track the best solution
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                # Reset population if no improvement for 20 generations
                population = [population[0].copy()]
                for _ in range(pop_size - 1):
                    individual = create_individual()
                    population.append(project_feasible(individual))
                no_improvement_count = 0
        
        # Create new population
        new_population = population[:elite_size]  # Keep elite individuals
        
        # Tournament selection
        def tournament_selection():
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            return population[winner_index].copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection for parents
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Project to feasible region
            child1 = project_feasible(child1)
            child2 = project_feasible(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        generation += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
