import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 50
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try several optimization strategies
    iter_count = 0
    max_iter = 50
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Strategy 1: Local optimization with different random starts
        for _ in range(3):  # Multiple random starts
            if time.time() - start_time > budget_s - 2:
                break
            # Perturb slightly
            h_trial = best_h + np.random.normal(0, 0.05, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                continue
                
        # Strategy 2: Simple random search refinement
        if time.time() - start_time > budget_s - 2:
            break
        h_trial = best_h + np.random.normal(0, 0.02, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_candidate = evaluate_erdos_solution(h_trial)
        if c5_candidate < best_c5:
            best_h = h_trial
            best_c5 = c5_candidate
            
        iter_count += 1
        
        # Occasionally increase resolution
        if iter_count % 10 == 0 and n_points < 100:
            # Increase resolution if time allows
            if time.time() - start_time < budget_s - 5:
                n_points = min(100, n_points + 10)
                h_resized = np.interp(np.linspace(0, 1, n_points), 
                                    np.linspace(0, 1, len(best_h)), 
                                    best_h)
                best_h = h_resized
                best_h = best_h * (n_points / 2) / np.sum(best_h)
                best_h = np.clip(best_h, 0, 1)
                best_c5 = evaluate_erdos_solution(best_h)
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
