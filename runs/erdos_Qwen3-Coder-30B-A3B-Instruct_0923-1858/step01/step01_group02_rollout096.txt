import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initialize with simple construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Add a small bump to encourage exploration
        h[10] = 0.7
        h[11] = 0.3
        # Normalize to meet the constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    dx = 2.0 / n_points
    
    # Track number of iterations
    iteration = 0
    
    # Main cutting plane loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        current_overlaps = corr * dx  # Scale by dx
        max_overlap_idx = np.argmax(current_overlaps)
        max_overlap_val = current_overlaps[max_overlap_idx]
        
        # Create a new convex subproblem
        h_var = cp.Variable(n_points)
        C_var = cp.Variable(1)
        
        # Constraints
        constraints = []
        
        # Convex approximation using cutting planes
        # We add one cut at the maximum overlap, for simplicity
        # In principle, we should add cuts for all shifts to form a tighter convex relaxation
        for k in range(2 * n_points - 1):
            # Compute gradient for shift k
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                if j < n_points:
                    gradient[i] = dx * (1 - h[j])
                else:
                    gradient[i] = dx * (1 - h[j - n_points])
            
            # Add the supporting hyperplane constraint
            # O_k(h) <= O_k(h^t) + gradient^T (h - h^t)
            # Which gives: gradient^T h <= O_k(h^t) - gradient^T h^t + O_k(h^t)
            # Vectorized form
            cut_value = current_overlaps[k] - np.dot(gradient, h)
            constraints.append(cp.dot(gradient, h_var) <= cut_value)
        
        # Box constraints
        constraints.extend([h_var >= 0, h_var <= 1])
        
        # Sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Objective
        objective = cp.Minimize(C_var)
        
        # Add the approximation that C >= max overlap so far
        constraints.append(C_var >= max_overlap_val)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=100)  # CVXOPT should be available
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                c5_new = C_var.value
                
                # Ensure bounds
                h_new = np.clip(h_new, 0, 1)
                # Renormalize sum to equality constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate new solution
                c5_check = evaluate_erdos_solution(h_new)
                
                if c5_check < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_check
                    
                h = h_new.copy()
            else:
                # Fallback to just perturbing and using local search approach
                perturbation = np.random.normal(0, 0.02, n_points)
                h_pert = np.clip(h + perturbation, 0, 1)
                # Renormalize
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                c5_new = evaluate_erdos_solution(h_pert)
                if c5_new < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_new
                h = h_pert.copy()
        except Exception:
            # If the solver fails, fall back to simple perturbation
            perturbation = np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h + perturbation, 0, 1)
            # Renormalize
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            c5_new = evaluate_erdos_solution(h_pert)
            if c5_new < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_new
            h = h_pert.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
