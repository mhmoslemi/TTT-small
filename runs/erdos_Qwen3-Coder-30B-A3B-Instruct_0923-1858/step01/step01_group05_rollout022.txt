import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular function centered at x=1, scaled to meet constraint
        n_points = 70
        h = np.array([1.0 - abs(2.0 * i / n_points - 1.0) for i in range(n_points)])
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define bounds and constraint for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    # Initial evaluation
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Global exploration: perturb in critical region
        h_trial = h.copy()
        # Only perturb in the interval [0.3, 0.7] in terms of normalized position
        for i in range(n_points):
            x = i * 2.0 / n_points
            if 0.3 <= x <= 0.7:
                h_trial[i] += np.random.normal(0, 0.03)
        # Repair feasibility
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Evaluate trial point
        c5_trial = objective(h_trial)
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            
        # Local refinement using SLSQP
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                # Repair feasibility
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined / np.sum(h_refined) * (n_points / 2.0)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
        
        # Update current point to the best seen so far
        h = best_h.copy()
        
        iteration += 1
        # Prevent infinite loop
        if iteration > 10000:
            break
            
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
