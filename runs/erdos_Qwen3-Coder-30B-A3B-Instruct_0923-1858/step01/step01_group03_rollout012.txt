import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    perturb_rate = 0.04
    max_iter = 2000
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def generate_neighbor(current_h):
        neighbor = current_h.copy()
        num_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            # Perturb by small random step
            step = np.random.normal(0, 0.05)
            neighbor[idx] += step
        return project_feasible(neighbor)
    
    # Tabu search main loop
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate neighbor
        candidate = generate_neighbor(h)
        
        # Compute candidate C5
        candidate_C5 = compute_C5(candidate)
        
        # Evaluate if it's better than best or not tabu
        candidate_tuple = tuple(candidate.round(8))
        is_tabu = candidate_tuple in tabu_list
        
        if candidate_C5 < best_C5 or not is_tabu:
            if candidate_C5 < best_C5:
                best_C5 = candidate_C5
                best_h = candidate.copy()
            
            h = candidate.copy()
            
            # Add to tabu list
            tabu_list.append(candidate_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        iter_count += 1
        if iter_count >= max_iter:
            break
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
