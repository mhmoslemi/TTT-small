import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use a combination of strategies
    # Strategy 1: Gradient-based optimization with constraints
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Strategy 2: Search for the maximum overlap
    max_iter = 100
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try random perturbations
        pert_h = h.copy() + np.random.normal(0, 0.02, n_points)
        pert_h = np.clip(pert_h, 0, 1)
        pert_h = pert_h * (n_points / 2) / np.sum(pert_h)
        c5_pert = evaluate_erdos_solution(pert_h)
        if c5_pert < best_c5:
            best_h = pert_h
            best_c5 = c5_pert
            
        # Use SLSQP optimization from current best
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
            
        # Try a local adjustment based on largest overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n_points - 1)
        # Create adjustment based on this shift
        adjusted_h = best_h.copy()
        if shift < 0:
            for i in range(len(adjusted_h) + shift):
                adjusted_h[i] = max(0, min(1, adjusted_h[i] - 0.01 * (1 - adjusted_h[i + shift])))
        else:
            for i in range(shift, len(adjusted_h)):
                adjusted_h[i] = max(0, min(1, adjusted_h[i] - 0.01 * (1 - adjusted_h[i - shift])))
                
        adjusted_h = adjusted_h * (n_points / 2) / np.sum(adjusted_h)
        c5_adjusted = evaluate_erdos_solution(adjusted_h)
        if c5_adjusted < best_c5:
            best_h = adjusted_h
            best_c5 = c5_adjusted
            
        iter_count += 1
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
