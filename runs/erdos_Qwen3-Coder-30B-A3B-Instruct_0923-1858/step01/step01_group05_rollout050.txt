import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h_original = np.array(initial_h_values, dtype=float)
        n_points = len(h_original)
        # Interpolate to 140 points
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_orig, h_original)
        n_points = 140
    else:
        # Create a feasible random h vector
        n_points = 140
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping with global search
    max_iter = 200
    iter_count = 0
    sigma = 0.05
    
    # Store top candidates
    candidates = []
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local refinement
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                candidates.append((c5_new, h_new.copy()))
        except:
            pass
        
        iter_count += 1
        
        # Adjust sigma
        if iter_count % 50 == 0:
            sigma /= 2
            
        # Occasionally update the search space
        if iter_count % 10 == 0:
            # Update h using the best candidate found so far
            if candidates:
                candidates.sort(key=lambda x: x[0])
                h = candidates[0][1].copy()
                current_c5 = candidates[0][0]
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
    # Final local refinement of the best candidates
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:5]
    
    for c5_val, candidate_h in top_candidates:
        try:
            res = minimize(objective, candidate_h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            continue
            
    # Ensure final solution is valid
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2.0)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
