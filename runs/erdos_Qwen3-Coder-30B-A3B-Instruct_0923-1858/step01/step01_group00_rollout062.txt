import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for better resolution
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to desired resolution
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_old, h)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to sum(h) = n_points / 2
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling factor
    sigma = 0.1  # Perturbation scale
    max_no_improve = 100  # Shake if no improvement
    shake_prob = 1.0
    
    # Keep track of the best solution found
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    no_improve_count = 0
    
    while time.time() - start_time < budget_s:
        # Generate perturbation
        delta_h = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_new = h + delta_h
        
        # Project to feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        try:
            new_c5 = evaluate_erdos_solution(h_new)
            old_c5 = evaluate_erdos_solution(h)
            
            # Accept or reject based on Metropolis criterion
            if new_c5 < old_c5 or np.random.rand() < np.exp(-(new_c5 - old_c5) / T):
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                no_improve_count += 1
                
        except Exception:
            no_improve_count += 1

        # Cooling and shaking
        T = max(T * alpha, 1e-6)
        
        if no_improve_count >= max_no_improve:
            # Shake the solution
            h = h + np.random.normal(0, T * sigma, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            no_improve_count = 0

    return (best_h, best_c5, n_points)
