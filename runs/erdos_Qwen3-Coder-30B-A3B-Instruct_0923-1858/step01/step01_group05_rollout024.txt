import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from existing construction or create triangular function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        # Create triangular function centered at x=1
        h = 1 - np.abs(2 * np.arange(n_points) / n_points - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Evaluate initial solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 10000
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Perturb h in critical region (0.3 <= x <= 0.7) with moderate noise
        h_trial = h.copy()
        # Calculate indices for critical region
        dx = 2.0 / n_points
        start_idx = int(0.3 / dx)
        end_idx = int(0.7 / dx) + 1
        # Only perturb the critical region
        if start_idx < end_idx:
            noise = np.random.normal(0, 0.03, end_idx - start_idx)
            h_trial[start_idx:end_idx] += noise
            h_trial = np.clip(h_trial, 0, 1)
            # Scale to integrality constraint
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Try local optimization
        try:
            # Define bounds and constraint
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h_array):
                return np.sum(h_array) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h_array):
                corr = np.correlate(h_array, 1-h_array, mode="full")
                return np.max(corr) * (2.0 / n_points)
            
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
            else:
                # Fallback: just use the perturbed version if optimization fails
                c5_new = evaluate_erdos_solution(h_trial)
                if c5_new < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_new
                    h = h_trial.copy()
        except:
            # In case of error, keep current best but perturb and retry
            c5_new = evaluate_erdos_solution(h_trial)
            if c5_new < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_new
                h = h_trial.copy()
        
        iter_count += 1
        # Occasionally do a small random walk for further exploration
        if iter_count % 20 == 0:
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
    
    # Final validation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
