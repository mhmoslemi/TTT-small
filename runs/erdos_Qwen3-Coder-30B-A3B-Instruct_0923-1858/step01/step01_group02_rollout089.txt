import numpy as np
from cvxpy import Variable, Minimize, Problem, sum_expr, Norm
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h[:10] = 0.8
        h[10:20] = 0.2
        h[20:30] = 0.6
        h[30:40] = 0.4
        h[40:50] = 0.7
        h[50:60] = 0.3
        h[60:] = 0.5
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Initialize C5 to a loose upper bound
    C5_estimate = 0.4
    
    # Precompute the correlation structure
    # For efficiency, compute the full correlation
    dx = 2.0 / n_points

    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Use CVXPY to solve the constrained optimization
        # Variables
        h_vars = Variable(n_points)
        
        # Objective: minimize the current estimate of C5
        objective = Minimize(C5_estimate)
        constraints = []
        
        # Bounds
        constraints += [h_vars >= 0]
        constraints += [h_vars <= 1]
        
        # Sum constraint
        constraints += [sum_expr(h_vars) == n_points / 2]
        
        # Overlap constraints
        # For each shift k, we have the constraint:
        # integral(h(x)(1 - h(x+k)) dx) <= C5/dx
        for shift in range(n_points):
            if shift == 0:
                continue
            # Sum over valid indices for this shift
            if shift > 0:
                # Align h[i] with (1 - h[i+shift])
                # Need h[0:shift] for h[i] and (1-h[shift:]) for (1-h[i+shift])
                valid_len = n_points - shift
                sum_val = sum_expr(h_vars[:valid_len] * (1 - h_vars[shift:shift+valid_len]))
            else:
                # Align h[i+shift] with (1 - h[i])
                # Need h[-shift:] for h[i+shift] and (1-h[:shift]) for (1-h[i])
                valid_len = n_points + shift
                sum_val = sum_expr(h_vars[-shift:] * (1 - h_vars[:valid_len]))
            
            # Constraint: sum(h[i] * (1 - h[i+shift])) <= C5 / dx
            constraints.append(sum_val <= C5_estimate / dx)
        
        # Create and solve the problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)

                    best_c5 = current_c5
                    # Optionally adjust C5 estimate
                    C5_estimate = current_c5
                else:
                    # Increase C5 to relax constraints
                    C5_estimate += 0.0001
            else:

        except Exception:
            C5_estimate += 0.0001

    # Final re-evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
