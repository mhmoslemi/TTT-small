import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create an initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation approach
    def solve_convex_relaxation(h_init, n_points):
        # Create optimization variables
        h = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective: maximize lambda
        objective = cp.Maximize(lambda_var)
        
        # Overlap constraints for all lags
        for k in range(n_points):
            # Compute overlap for lag k
            # We consider zero-padding for simplicity
            corr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr += h[i] * (1 - h[j])
            constraints.append(corr >= lambda_var)
            
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return h_init
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Initial convex relaxation step
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.7:
        h = solve_convex_relaxation(h, n_points)
        h = project_feasible(h)
    
    # Gradient-based local search
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

    # Main loop for gradient refinement
    eta = 0.01
    patience = 0
    max_patience = 100
    iter_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
    
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    # Random restarts
    for i in range(10):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        h_retry = np.random.rand(n_points)
        h_retry = project_feasible(h_retry)
        c5_retry = compute_C5(h_retry)
        if c5_retry < best_C5:
            best_C5 = c5_retry
            best_h = h_retry.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
