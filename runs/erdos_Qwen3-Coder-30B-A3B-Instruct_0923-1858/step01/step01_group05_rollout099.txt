import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Time tracking
    max_iter = 5000  # Sufficient number of iterations within 600s
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create a perturbed version in 0.3 <= x <= 0.7 region
        h_trial = h.copy()
        
        # Select indices in the critical region (0.3 to 0.7)
        x_vals = np.linspace(0, 2, n_points)

        noise = np.random.normal(0, 0.03, len(idxs))
        h_trial[idxs] += noise
        
        # Enforce bounds and scaling
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)

        c5_trial = evaluate_erdos_solution(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

        # Local refinement with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}

            corr = np.correlate(h, 1-h, mode="full")
            return np.max(corr) * (2.0 / n_points)

        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds,

                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2.0) / np.sum(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass

        # Update h with best found so far
        h = best_h.copy()
        iter_count += 1

    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
