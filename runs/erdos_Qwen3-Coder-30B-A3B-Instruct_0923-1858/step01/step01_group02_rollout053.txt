import numpy as np
from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, norm, sum as cvxsum
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 70
        h = np.ones(n_points) * 0.5
        # Ensure it sums to n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1] and sum constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization parameters
    max_iter = 1000
    perturbation_magnitude = 0.1
    perturbation_decay = 0.95
    iter_since_improvement = 0
    max_stagnation = 50
    
    for iteration in range(max_iter):
        # Early termination
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Perturb the solution
        perturbation = np.random.normal(0, perturbation_magnitude, n_points)
        x0 = np.clip(best_h + perturbation, 0, 1)
        x0 = x0 * (n_points / 2) / np.sum(x0)
        
        # Define the objective for local optimization
        def objective(x):
            return evaluate_erdos_solution(x)
            
        # Perform optimization
        try:
            # Use L-BFGS-B for better behavior with bounds
            res = minimize(objective, x0, method='L-BFGS-B',
                           bounds=[(0, 1) for _ in range(n_points)],
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                current_c5 = evaluate_erdos_solution(res.x)
                if current_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = current_c5
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
            else:
                # If optimization fails, keep the previous best
                pass
                
        except Exception:
            # If optimization fails, keep the previous best
            pass
            
        # Adjust perturbation magnitude
        if iter_since_improvement > 20:
            perturbation_magnitude *= perturbation_decay
            iter_since_improvement = 0 
                
        if iter_since_improvement > max_stagnation:
            # Random restart
            perturbation_magnitude = 0.1
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            best_h = h.copy()
            best_c5 = evaluate_erdos_solution(best_h)
            iter_since_improvement = 0
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
