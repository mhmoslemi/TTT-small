import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Simple construction
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Start with a reasonable upper bound
    c5_bound = 0.4
    step_size = 0.0001
    max_iter = 1000
    iter_count = 0
    
    # CVXPY variables
    x = cp.Variable(n_points)
    
    # Constraints
    constraints = [x >= 0, x <= 1]
    
    # Sum constraint
    sum_constraint = cp.sum(x) == n_points / 2
    constraints.append(sum_constraint)
    
    # Objective: minimize C5 (which we will adjust iteratively)
    # Create a CP Problem with a placeholder objective
    problem = cp.Problem(cp.Minimize(0), constraints)
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute all correlations for the current h
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        max_corr = np.max(corr)
        
        # If we are already below the target, we stop
        if max_corr * dx <= 0.3807:
            break
            
        # Update the c5_bound as maximum overlap
        c5_bound = max_corr * dx
        
        # Solve optimization problem with current c5_bound
        # Set a constraint that all overlaps are at most c5_bound
        # We construct constraints for each shift k (from 0 to n_points-1)
        all_constraints = constraints[:]
        
        # Process double correlations, add constraint for each k
        for k in range(n_points):
            # Need correlation term ∫h(x)(1-h(x+k)) dx 
            # Compute lhs of constraint for this k
            # we will form the sum in matrix form
            if k == 0:
                # Just sum of h(x)(1-h(x)) = sum(h - h^2)
                expr = cp.sum(cp.multiply(x, 1 - x))
            elif k < n_points:
                # Use indexing as h[k:] from h[0:]  
                expr = cp.sum(cp.multiply(x[:-k], 1 - x[k:]))
            else:
                expr = 0
            # Add constraint
            all_constraints.append(expr <= c5_bound / dx)
        
        # Use a new problem
        objective = cp.Minimize(0)
        problem = cp.Problem(objective, all_constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = x.value
                if h_new is not None:
                    # Ensure bounds and correct sum
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    current_c5 = evaluate_erdos_solution(h_new)
                    if current_c5 < c5_bound:
                        h = h_new.copy()
                        c5_bound = current_c5
                    else:
                        # Perturb and solve again
                        eps = 0.0001
                        perturbation = np.random.normal(0, eps, n_points)
                        h_perturb = np.clip(h + perturbation, 0, 1)
                        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                        c5_perturb = evaluate_erdos_solution(h_perturb)
                        if c5_perturb < c5_bound:
                            h = h_perturb.copy()
                            c5_bound = c5_perturb
                else:
                    c5_bound += step_size * 2
            else:
                c5_bound += step_size * 2
        except:
            c5_bound += step_size * 2
        
        iter_count += 1
        
        # When we have some progress, do not let it get too wide
        if c5_bound > 0.4:
            c5_bound = 0.4
            
        # Occasionally reduce the step size to refine near target
        if iter_count % 50 == 0:
            step_size *= 0.9

    final_c5 = evaluate_erdos_solution(h)
    
    return (h.tolist(), float(final_c5), n_points)
