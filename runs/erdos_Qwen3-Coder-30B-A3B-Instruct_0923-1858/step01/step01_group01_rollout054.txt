import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_h(n_points):
    """Initialize h with a triadic pattern to reduce overlaps."""
    h = np.zeros(n_points)
    for i in range(n_points):
        phase = np.abs(np.sin(np.pi * i / (n_points / 3))) ** 3
        h[i] = np.clip(2 * phase - 0.5, 0, 1)
    return project(h, 2.0 / n_points)

def gradient_estimate(h, dx, epsilon=1e-8):
    """Estimate gradient of C5 w.r.t. h using finite differences."""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project(h_plus, dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        max_plus = np.max(corr_plus)

        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project(h_minus, dx)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        max_minus = np.max(corr_minus)

        grad[i] = (max_plus - max_minus) / (2 * epsilon)
    return grad

            temp_c5 = evaluate(temp_h, dx)

                best_c5 = temp_c5
