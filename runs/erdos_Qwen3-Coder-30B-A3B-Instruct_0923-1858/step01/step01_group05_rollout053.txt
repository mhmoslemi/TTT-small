import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points
        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_old, h)
        n_points = 140
    else:
        # Initialize with a simple construction
        n_points = 140
        h = np.zeros(n_points)
        # Cluster in middle of the interval
        center = n_points // 2
        width = n_points // 4
        h[center-width:center+width] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define objective function
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Multi-phase optimization
    max_iter = 200  # basin-hopping iterations
    iter_count = 0
    sigma = 0.05
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Basin hopping step
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local optimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        iter_count += 1
        # Adaptively decrease noise
        if iter_count % 50 == 0:
            sigma /= 2.0
    
    # More local refinement on the best solution
    refined_h = best_h.copy()
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        try:
            res = minimize(objective, refined_h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                refined_h = res.x
                refined_c5 = objective(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h.copy()
                    best_c5 = refined_c5
        except:
            break
    
    # Final validation: ensure constraints
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points / 2.0) > 1e-6:
        best_h = best_h * (n_points / 2.0) / final_sum
    
    # Ensure bounds
    best_h = np.clip(best_h, 0, 1)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
