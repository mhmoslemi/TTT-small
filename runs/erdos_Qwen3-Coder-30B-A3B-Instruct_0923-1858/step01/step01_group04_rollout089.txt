import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_values = np.array(initial_h_values)
        n_points = len(h_values)
    else:
        # Fallback: create a basic feasible solution
        n_points = 70
        h_values = np.ones(n_points) * 0.5  # Start with uniform
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
    
    # Ensure integral constraint
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)
    
    # Start with current best and perform successive optimization steps
    best_h = h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try several local optimizations from different random starts
    max_iter = 10
    iter_count = 0
    
    # Initial strategy: use a pattern-based local search
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try to improve the solution
        h_trial = best_h.copy()
        
        # Add small random perturbations
        perturbation = np.random.normal(0, 0.05, n_points)
        h_trial += perturbation
        h_trial = np.clip(h_trial, 0, 1)
        
        # Scale to meet integral constraint
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Try different optimization approaches in sequence
        try:
            # First optimize with trust-constr method - usually good
            res = minimize(
                lambda x: evaluate_erdos_solution(x),
                h_trial,
                method='trust-constr',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 300, 'disp': False}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
        
        iter_count += 1
        
        # Also try gradient-based optimization from multiple starting points
        if iter_count % 3 == 0:
            for _ in range(2):
                # Multi-start
                h_multi = np.random.rand(n_points) * 0.5
                h_multi = h_multi * (n_points / 2) / np.sum(h_multi)
                h_multi = np.clip(h_multi, 0, 1)
                
                try:
                    res = minimize(
                        lambda x: evaluate_erdos_solution(x),
                        h_multi,
                        method='L-BFGS-B',
                        bounds=[(0, 1) for _ in range(n_points)],
                        options={'maxiter': 200}
                    )
                    
                    if res.success:
                        h_candidate = np.clip(res.x, 0, 1)
                        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                        c5_candidate = evaluate_erdos_solution(h_candidate)
                        if c5_candidate < best_c5:
                            best_h = h_candidate
                            best_c5 = c5_candidate
                            
                except Exception:
                    continue
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
