import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the provided initial construction if available

        n_points = len(h)
    else:

        n_points = 70
        h = np.zeros(n_points)
        h[::2] = 1.0
        h[1::2] = 0.0
        # Normalize sum to be exactly n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Frank-Wolfe style trust region approach
    max_iter = 1000
    trust_radius = 0.1
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        current_c5 = evaluate_erdos_solution(best_h)
        
        # Compute gradient approximation by finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Frank-Wolfe subproblem: min s' * grad s.t. ||s|| <= trust_radius, 0 <= h + s <= 1, sum(s) = 0
        # Simplified: let s = -step_size * grad with trust region
        s = -step_size * grad
        s_norm = np.linalg.norm(s)
        if s_norm > trust_radius:
            s = s * trust_radius / s_norm
            
        # Ensure step respects bounds and sum constraint
        s_new = s.copy()
        h_new = best_h + s_new
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Backtracking line search
        alpha = 1.0
        while alpha > 1e-6:
            h_trial = best_h + alpha * s_new
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial)
            if c5_trial < current_c5:
                h_new = h_trial
                break
            alpha *= 0.5
        
        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Adjust trust radius
        if new_c5 < current_c5:
            trust_radius *= 1.1
        else:
            trust_radius *= 0.9
            
        trust_radius = max(trust_radius, 1e-5)
        
        # Occasionally perturb to escape stagnation
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.001, n_points)
            h_noisy = best_h + noise
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points / 2) / np.sum(h_noisy)
            c5_noisy = evaluate_erdos_solution(h_noisy)
            if c5_noisy < best_c5:
                best_h = h_noisy
                best_c5 = c5_noisy
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
