import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iter = 200
    iter_count = 0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Store the best result found so far
    best_result = (best_h.copy(), best_c5, n_points)
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Random perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        # Reset integral constraint
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Local optimization
        try:
            # Use SLSQP for optimization
            def objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                return evaluate_erdos_solution(h_vec)
            
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_integral}
            
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    best_result = (best_h.copy(), best_c5, n_points)
                    
        except Exception as e:
            pass
        
        # Occasionally try a higher resolution
        if iter_count % 20 == 0 and n_points == 50:
            # Increase resolution to 100
            h_upsampled = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, 50), best_h)
            h_upsampled = h_upsampled * 50.0 / np.sum(h_upsampled)  # scale to integral
            h_upsampled = np.clip(h_upsampled, 0, 1)
            n_points = 100
            best_h = h_upsampled
            best_c5 = evaluate_erdos_solution(best_h)
            best_result = (best_h.copy(), best_c5, n_points)
        
        iter_count += 1
    
    # Final refinement: try upsampling if low res was used
    if n_points == 50:
        h_upsampled = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, 50), best_h)
        h_upsampled = h_upsampled * 50.0 / np.sum(h_upsampled)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        final_c5 = evaluate_erdos_solution(h_upsampled)
        if final_c5 < best_result[1]:
            best_result = (h_upsampled, final_c5, 100)
    
    return best_result
