import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_old, h)
        n_points = 140
    else:
        # Create a feasible starting h vector
        n_points = 140
        h = np.ones(n_points) * 0.5  # Uniform initial guess
        h = h / np.sum(h) * (n_points / 2.0)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Global exploration with perturbation
    iter_count = 0
    perturbation_strength = 0.05
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Perturbation-based global search
        h_candidates = []
        for _ in range(50):  # Generate 50 perturbed candidates
            h_trial = h + np.random.normal(0, perturbation_strength, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            h_candidates.append(h_trial)
        
        # Evaluate and keep top 10
        candidates_eval = [(h_cand, evaluate_erdos_solution(h_cand)) for h_cand in h_candidates]
        candidates_eval.sort(key=lambda x: x[1])
        top_candidates = candidates_eval[:10]
        
        # Local refinement with SLSQP
        for h_trial, _ in top_candidates:
            try:
                res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass
        
        # Update h with best from current iteration
        h = best_h.copy()
        
        # Decrease perturbation strength
        if iter_count % 50 == 0 and iter_count > 0:
            perturbation_strength /= 2.0
        
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
