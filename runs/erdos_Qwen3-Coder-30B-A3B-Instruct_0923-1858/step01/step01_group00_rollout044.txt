import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import Variable, Minimize, Problem, sum as cvx_sum, norm
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Binary search parameters
    low = 0.0
    high = 0.38085
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    
    while high - low > tolerance and (time.time() - start_time) < budget_s * 0.9:
        C = (low + high) / 2.0
        # Create variables and constraints
        h_vars = Variable(n_points)
        constraints = [0 <= h_vars, h_vars <= 1]
        constraints.append(cvx_sum(h_vars) == n_points / 2)
        
        # Add the constraint on overlaps
        for shift in range(n_points):
            overlap_expr = 0.0
            for i in range(n_points):
                j = (i + shift) % n_points
                overlap_expr += h_vars[i] * (1 - h_vars[j])
            overlap_expr *= dx
            constraints.append(overlap_expr <= C)
        
        # Minimize the objective (dummy)
        objective = Minimize(0)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_result = np.array(h_vars.value).flatten()
                # Refine solution using gradient descent
                def objective(h):
                    return evaluate_erdos_solution(h)
                
                bounds = [(0, 1) for _ in range(n_points)]
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2
                
                constraint = {'type': 'eq', 'fun': constraint_sum}
                result = minimize(
                    objective,
                    h_result,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 500},
                    tol=1e-6
                )
                if result.success:
                    current_c5 = evaluate_erdos_solution(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()
                    if current_c5 < C:
                        high = C
                    else:
                        low = C
                else:
                    low = C
            else:
                low = C
        except Exception:
            low = C
    
    # Final refinement if no progress in binary search
    if best_c5 == float('inf'):
        # Try local optimization with initial solution
        def objective(h):
            return evaluate_erdos_solution(h)
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        constraint = {'type': 'eq', 'fun': constraint_sum}
        try:
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 500},
                tol=1e-6
            )
            if result.success:
                best_c5 = evaluate_erdos_solution(result.x)
                best_h = result.x.copy()
        except Exception:
            pass

    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
