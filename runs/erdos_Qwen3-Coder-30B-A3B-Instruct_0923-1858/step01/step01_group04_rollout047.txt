import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)

    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Try a few different approaches and keep the best
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    # First do a coarse optimization
    coarse_n = min(50, n_points)
    if coarse_n < n_points:
        # Interpolate coarse grid
        coarse_h = np.interp(np.linspace(0, 1, coarse_n), 
                           np.linspace(0, 1, n_points), h)
        coarse_h = coarse_h * (coarse_n / 2) / np.sum(coarse_h)

            if c5_de < best_c5:
                best_h = h_de

    except Exception:
        pass

    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
