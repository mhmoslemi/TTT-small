import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_old, h)
            n_points = 140
    else:
        # Create an initial vector
        n_points = 140
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Ensure h sums to 70 (since 2/140 * 70 = 1)
    h = h / np.sum(h) * 70.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation parameters
    perturbation_strength = 0.05
    perturbation_decay = 0.95
    perturbation_decay_interval = 50
    
    # Set up optimization parameters
    dx = 2.0 / n_points
    
    # Define constraint function
    def constraint_sum(h):
        return np.sum(h) - 70.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # The objective function
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    # Global search with perturbations
    iter_count = 0
    max_iter = 1000
    population_size = 50
    top_candidates = 10
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate population
        population = []
        for _ in range(population_size):
            # Add Gaussian noise
            h_perturbed = h + np.random.normal(0, perturbation_strength, n_points)
            # Clip to valid range and rescale
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * 70.0
            
            # Evaluate
            c5_val = evaluate_erdos_solution(h_perturbed)
            population.append((h_perturbed.copy(), c5_val))
            
        # Sort population by C5
        population.sort(key=lambda x: x[1])
        
        # Keep top candidates for local refinement
        top_solutions = population[:top_candidates]
        
        # Local refinement with SLSQP
        for h_candidate, _ in top_solutions:
            try:
                res = minimize(objective, h_candidate, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass
        
        # Choose the best from the current population as new starting point
        h = population[0][0].copy()
        
        # Adapt perturbation strength
        if iter_count % perturbation_decay_interval == 0 and iter_count > 0:
            perturbation_strength *= perturbation_decay
            
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
