import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def initialize_population(n_points, population_size=50):
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = initial_h_values.copy()
        else:
            h = np.linspace(0, 1, n_points//2, endpoint=False)
            h = np.concatenate([h, np.linspace(1, 0, n_points - n_points//2)])
        # Random perturbation
        if np.random.rand() < 0.5:
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.normal(0, 0.05)
        h = project(h, dx)
        population.append(h)
    return population

def evaluate_fitness(candidate, dx):
    corr = correlate(candidate, 1 - candidate, mode='full')
    max_overlap = np.max(corr) * dx
    # Complexity regularization
    diff = np.abs(np.diff(candidate))
    complexity = np.sum(diff > 0.1)
    return 1 / (max_overlap + 0.01 * complexity)

def tournament_selection(population, fitnesses, num_parents=10):
    tournament_size = 3
    selected = []
    for _ in range(num_parents):
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    # Uniform crossover
    mask = np.random.rand(len(parent1)) < 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(candidate, dx, mutation_rate=0.1):
    mutated = candidate.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return project(mutated, dx)

def evolve_population(population, dx, budget_s=600):
    start_time = time.time()
    best_fitness = 0
    best_h = None
    
    while time.time() - start_time < budget_s - 60:
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_fitness(c, dx) for c in population]
        
        # Track best solution
        current_best = population[np.argmax(fitnesses)]
        current_fitness = max(fitnesses)
        if current_fitness > best_fitness:
            best_fitness = current_fitness
            best_h = current_best.copy()
        
        # Select parents
        parents = tournament_selection(population, fitnesses)
        
        # Generate new population
        new_population = []
        while len(new_population) < len(population):
            # Perform crossover
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            # Apply mutation
            child = mutate(child, dx)
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        if len(population) == 0:
            break
    
    # Final evaluation of best solution found
    if best_h is not None:
        final_fitness = evaluate_fitness(best_h, dx)
        return best_h, 1 / final_fitness
    else:
        # If no solution found, return first population member
        return population[0], evaluate(population[0], dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points)
    
    # Evolve population
    best_h, best_C = evolve_population(population, dx, budget_s)
    
    return (best_h, best_C, n_points)
