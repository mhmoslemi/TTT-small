```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback to a construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Better optimization approach using sequential quadratic programming
    # with a more direct approach of optimizing overlaps
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Several optimization attempts with different starting points
    max_trials = 20
    trial_count = 0
    
    while time.time() - start_time < budget_s - 1 and trial_count < max_trials:
        # Random perturbation
        perturbation = np.random.normal(0, 0.1, n_points)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Local optimization using L-BFGS-B
        try:
            res = minimize(
                objective,
                h_trial,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-7}
            )
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
        
        # More aggressive local search using trust-constr
        try:
            res = minimize(
                objective,
                h_trial,
                method='trust-constr',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-7}
            )
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
        
        trial_count += 1
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
```
