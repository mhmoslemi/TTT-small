import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    max_iter = 100000
    
    try:
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h_initial = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h_initial[i] = max(0, 1 - abs(x[i] - 1))
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.995
    max_stagnation = 50
    mutation_rate = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add the initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal best and global best
    pbest = [swarm[i].copy() for i in range(Pop_size)]
    pbest_fitness = [float('inf')] * Pop_size
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Evaluate initial positions
    for i in range(Pop_size):
        corr = correlate(swarm[i], 1 - swarm[i], mode="full")
        fitness = np.max(corr) * (2.0 / n_points)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    iter_count = 0
    stagnation_count = 0
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # Main PSO loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        if iter_count >= max_iter:
            break
            
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate fitness
            corr = correlate(swarm[i], 1 - swarm[i], mode="full")
            fitness = np.max(corr) * (2.0 / n_points)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        # Apply inertia decay
        w = max(0.3, w * inertia_decay)
        
        # Reset some particles if stagnation
        if stagnation_count > max_stagnation:
            for i in range(int(Pop_size * mutation_rate)):
                idx = np.random.randint(0, Pop_size)
                swarm[idx] = np.random.rand(n_points)
                swarm[idx] = swarm[idx] * (n_points / 2) / np.sum(swarm[idx])
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                stagnation_count = 0
                
        iter_count += 1
    
    # Final evaluation with the best solution
    final_C5 = compute_C5(swarm[gbest_idx])
    
    return (swarm[gbest_idx].tolist(), final_C5, n_points)
