import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def optimize_step(h, dx, lr=0.01, noise_level=0.01):
    """Single optimization step with gradient estimation."""
    n = len(h)
    grad = np.zeros_like(h)
    
    # Estimate gradient via finite differences for each component
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += 1e-5
        h_plus = project(h_plus, dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        max_plus = np.max(corr_plus)
        
        h_minus = h.copy()
        h_minus[i] -= 1e-5
        h_minus = project(h_minus, dx)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        max_minus = np.max(corr_minus)
        
        grad[i] = (max_plus - max_minus) / (2e-5)
    
    # Update with gradient descent
    h = h - lr * grad
    h = project(h, dx)
    
    # Occasionally add noise for escaping local minima
    if np.random.rand() < 0.1:
        h += np.random.normal(0, noise_level, n)
        h = project(h, dx)
    
    return h
