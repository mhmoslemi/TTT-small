import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define the objective function
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraints
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Try a few optimizations
    iter_count = 0
    max_iter = 50
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Create a perturbed version
        h_trial = h.copy()
        # Add random noise
        noise = np.random.normal(0, 0.05, n_points)
        h_trial += noise
        h_trial = np.clip(h_trial, 0, 1)
        # Re-normalize
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Try optimization
        try:
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )

                h_candidate = np.clip(res.x, 0, 1)
