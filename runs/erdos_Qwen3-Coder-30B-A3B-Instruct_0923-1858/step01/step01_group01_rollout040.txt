import numpy as np
import time
from scipy.signal import correlate

def project_feasible(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def mutate_individual(h, dx, rate=0.1):
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, 0.05)
    return project_feasible(mutated, dx)

def crossover(parent1, parent2):
    child = parent1.copy()
    mask = np.random.rand(len(parent1)) < 0.5
    child[mask] = parent2[mask]
    return child

def select_tournament(population, fitnesses, k=3):
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
    return population[best_idx]

        initial_h = np.concatenate([initial_h, np.linspace(1, 0, n_points - n_points//2)])
        initial_h = project_feasible(initial_h, dx)
    
    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate_individual(initial_h, dx, rate=0.2)
        population.append(h_mut)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitnesses = [1.0 / (evaluate_c5(h, dx) + 1e-10) for h in population]
        
        # Find best in current population
        max_fitness_idx = np.argmax(fitnesses)
        if fitnesses[max_fitness_idx] > 1.0 / best_c5:
            best_c5 = evaluate_c5(population[max_fitness_idx], dx)
            best_h = population[max_fitness_idx].copy()
        
        # Selection (tournament)
        new_pop = []
        for _ in range(population_size // 2):
            parent1 = select_tournament(population, fitnesses)
            parent2 = select_tournament(population, fitnesses)
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            child1 = mutate_individual(child1, dx)
            child2 = mutate_individual(child2, dx)
            new_pop.extend([child1, child2])
        
        population = new_pop[:population_size]
