import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 100
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Initial guess for C5
    C5 = 0.4
    step_size = 0.0001
    max_iter = int(budget_s / 0.1) # Roughly 6000 iterations
    
    # Setup CVXPY problem
    h_var = cp.Variable(n_points)
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    
    # Optimization variables
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Update C5 using a smaller step size
        step_size = max(0.00001, step_size * 0.999)
        
        # Create a list to collect constraints
        new_constraints = constraints[:]
        
        # Add all overlap constraints with current C5
        dx = 2.0 / n_points
        for k in range(n_points):
            # Compute indices affected by shift k
            idx = np.arange(n_points)
            sh_idx = (idx + k) % n_points
            overlap_constraint = cp.sum(cp.multiply(h_var, 1 - h_var[sh_idx])) <= C5 / dx
            new_constraints.append(overlap_constraint)
            
        # Define objective (this is a dummy, we're doing feasibility check)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, new_constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            
            # If solution is feasible, reduce C5
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    # Project back to feasible set
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    
                    current_c5 = evaluate_erdos_solution(h_sol)
                    if current_c5 < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = current_c5
                        
                    # Adjust C5 based on how much we're short of feasibility
                    C5 -= step_size
                    if C5 < 0:
                        C5 = 0.0001
                        
            else:
                # Infeasible, so increase C5
                C5 += step_size
                
        except Exception:
            C5 += step_size
            pass
        
        # Occasionally try to escape local minimum by randomizing a bit
        if it % 50 == 0 and it > 0:
            # Perturb solution
            eps = np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(best_h + eps, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            current_c5 = evaluate_erdos_solution(h_pert)
            if current_c5 < best_c5:
                best_h = h_pert.copy()
                best_c5 = current_c5
            
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
