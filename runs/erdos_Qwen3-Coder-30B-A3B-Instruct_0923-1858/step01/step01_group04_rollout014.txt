import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Use a more sophisticated approach: convex relaxation with barrier method
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Set up the optimization for improved solution
    for iteration in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try using a quadratic programming approach with CVXPY
        try:
            # Define variables
            h_vars = cp.Variable(n_points)
            t = cp.Variable()
            
            # Objective: minimize t
            objective = cp.Minimize(t)
            
            # Constraints
            constraints = []
            
            # Integral constraint
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Bounds
            constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
            constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
            
            # Overlap constraints (empirical approximation using a simpler method)
            # This uses more direct approach with lower time overhead
            dx = 2.0 / n_points
            h_trial = best_h.copy()
            
            # Use a local search to refine
            for _ in range(3):
                # SLSQP optimization for refinement
                def objective_func(h_vec):
                    return evaluate_erdos_solution(h_vec)
                
                def constraint_integral(h_vec):
                    return np.sum(h_vec) - n_points / 2
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraints_eq = {'type': 'eq', 'fun': constraint_integral}
                
                try:
                    res = minimize(
                        objective_func,
                        h_trial,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=constraints_eq,
                        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
                    )
                    
                    if res.success:
                        h_trial = np.clip(res.x, 0, 1)
                        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                        c5_candidate = evaluate_erdos_solution(h_trial)
                        if c5_candidate < best_c5:
                            best_h = h_trial.copy()
                            best_c5 = c5_candidate
                    else:
                        # If optimization fails, continue with our current best
                        break
                except:
                    break
                    
            # Ensure we have a valid solution
            if time.time() - start_time > budget_s - 1:
                break
                
        except:
            pass
    
    # Final evaluation
    return (list(best_h), best_c5, n_points)
