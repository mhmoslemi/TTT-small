import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points for better resolution
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_orig, h)
        n_points = 140
    else:
        # Initialize with a smooth distribution
        n_points = 140
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.tanh(2 * (x - 1)))  # Smooth step-like function
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping approach with adaptive perturbation
    num_iterations = 200
    sigma = 0.05
    
    for i in range(num_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local optimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        # Adaptive sigma decay
        if i % 50 == 0 and i > 0:
            sigma /= 2
            
        # Random walk step
        if i % 7 == 0:
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
    
    # Local refinement of best solution
    h_refined = best_h.copy()
    for _ in range(3):
        try:
            res = minimize(objective, h_refined, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_temp = res.x
                c5_temp = objective(h_temp)
                if c5_temp < best_c5:
                    best_h = h_temp.copy()
                    best_c5 = c5_temp
                    h_refined = h_temp.copy()
        except:
            break
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure the final solution is feasible
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points/2.0) > 1e-6:
        best_h = best_h / final_sum * (n_points / 2.0)
    
    return (best_h.tolist(), final_c5, n_points)
