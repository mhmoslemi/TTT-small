import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Use higher resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)

    else:

        h = h / np.sum(h) * (n_points / 2)

    # PSO parameters
    swarm_size = 50
    max_iter = budget_s * 10  # Estimate iterations
    inertia_weight = 0.7
    c1 = c2 = 1.5
    perturb_std = 0.01
    alpha = 0.9  # Low-pass filter constant
    
    # Initialize swarm
    swarm = np.random.rand(swarm_size, n_points) * 0.5
    swarm = swarm / np.sum(swarm, axis=1, keepdims=True) * (n_points / 2)
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.1
    
    # Evaluate initial swarm
    fitness = np.array([evaluate_erdos_solution(swarm[i]) for i in range(swarm_size)])
    
    # Personal best
    pbest = swarm.copy()
    pbest_fitness = fitness.copy()
    
    # Global best
    gb_idx = np.argmin(fitness)
    gbest = swarm[gb_idx].copy()
    gbest_fitness = fitness[gb_idx]
    
    # PSO iteration
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        for i in range(swarm_size):
            # Update velocity (with low-pass filtered perturbation)
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia_weight * velocities[i] +
                c1 * r1 * (pbest[i] - swarm[i]) +
                c2 * r2 * (gbest - swarm[i])
            )
            # Apply low-pass filter for smoother update
            velocities[i] = alpha * velocities[i] + (1 - alpha) * np.random.normal(0, 0.01, n_points)
                
            # Update position
            swarm[i] += velocities[i]
            
            # Apply feasibility constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] / np.sum(swarm[i]) * (n_points / 2)
            
            # Evaluate fitness
            new_fitness = evaluate_erdos_solution(swarm[i])
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = new_fitness
                
            # Update global best
            if new_fitness < gbest_fitness:
                gbest = swarm[i].copy()
                gbest_fitness = new_fitness
                
        # Perturb entire swarm to escape local minima
        for i in range(swarm_size):
            swarm[i] += np.random.normal(0, perturb_std, n_points)
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] / np.sum(swarm[i]) * (n_points / 2)
    
    # Final evaluation
    final_fitness = evaluate_erdos_solution(gbest)
    
    return (gbest, final_fitness, n_points)
