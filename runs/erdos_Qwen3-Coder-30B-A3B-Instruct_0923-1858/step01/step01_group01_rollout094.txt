import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1):
    """Apply small random perturbations to h."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, 0.05)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Tournament selection."""
    indices = np.random.choice(len(population), size=2*num_parents, replace=True)
    tournaments = indices.reshape(-1, 2)
    selected = [population[np.argmax([fitnesses[i] for i in t])] for t in tournaments]
    return selected[:num_parents]

    for _ in range(population_size):
        if np.random.rand() < 0.5:
            h = mutate(h0, 2.0/n_points)
        else:
            h = h0.copy()
        population.append(h)
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points)
    best_C = float('inf')
    best_h = None
    
    t0 = time.time()
    gen = 0
    while time.time() - t0 < budget_s - 10:
        gen += 1
        # Evaluate fitness
        fitnesses = [1 / (evaluate(h, dx) + 1e-10) for h in population]
        
        # Track best
        current_best_idx = np.argmax(fitnesses)
        current_C = 1 / fitnesses[current_best_idx]
        if current_C < best_C:
            best_C = current_C
            best_h = population[current_best_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Reproduction
        new_population = []
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < 0.1:
                child = mutate(child, dx)
            new_population.append(child)
        
        population = new_population[:len(population)]
    
    return (best_h, best_C, n_points)
