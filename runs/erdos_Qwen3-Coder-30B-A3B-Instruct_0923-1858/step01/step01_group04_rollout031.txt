import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Low resolution optimization (n=50)
    n_low = 50
    if n_points >= n_low:
        h_low = h[:n_low].copy()
    else:
        h_low = np.interp(np.linspace(0, 1, n_low), np.linspace(0, 1, n_points), h)
    
    h_low = h_low * (n_low / 2) / np.sum(h_low)
    h_low = np.clip(h_low, 0, 1)
    
    def objective_low(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    constraints = {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_low / 2}
    bounds = [(0, 1) for _ in range(n_low)]
    
    try:
        # SLSQP optimization from this starting point
        res = minimize(
            objective_low,
            h_low,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        if res.success:
            h_candidate = np.clip(res.x, 0, 1)
            h_candidate = h_candidate * (n_low / 2) / np.sum(h_candidate)

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_high)
