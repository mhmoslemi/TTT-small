import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define the objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_rate = 0.995
    min_temperature = 1e-6
    iterations_per_temp = 50
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Local optimization with SLSQP
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
        
        # Simulated Annealing step
        for _ in range(iterations_per_temp):
            # Create neighbor solution
            neighbor = best_h + np.random.normal(0, 0.02, n_points)
            neighbor = np.clip(neighbor, 0, 1)
            neighbor = neighbor * (n_points / 2) / np.sum(neighbor)
            c5_neighbor = evaluate_erdos_solution(neighbor)
            
            # Accept or reject based on Metropolis criterion
            if c5_neighbor < best_c5:
                best_h = neighbor
                best_c5 = c5_neighbor
            else:
                # Accept with probability based on temperature
                delta = c5_neighbor - best_c5
                if np.random.rand() < np.exp(-delta / temperature):
                    best_h = neighbor
                    best_c5 = c5_neighbor
        
        # Cool down temperature
        temperature = max(min_temperature, temperature * cooling_rate)
        iter_count += 1
        if iter_count % 100 == 0:
            # Periodic validation
            current_c5 = evaluate_erdos_solution(best_h)
            if abs(current_c5 - best_c5) > 1e-10:

        if time.time() - start_time > budget_s * 0.8 and n_points < 100:

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
