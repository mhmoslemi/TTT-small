import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation initialization
    def convex_relaxation(h0, n_points):
        # Create variables for convex relaxation
        h = Variable(n_points)
        
        # Create correlation constraints
        # Want to maximize lambda such that sum(h[i] * (1 - h[i+k])) >= lambda for all k
        # Would be best done through quadratic programming but we'll simplify
        
        # For now, let's make a simpler convex relaxation
        # Set up constrained optimization
        prob = Problem(Maximize(sum(h)), [
            0 <= h,
            h <= 1,
            sum(h) == n_points / 2,
        ])
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                return np.array(h.value).flatten()
        except:
            pass
        return h0.copy()

    h = convex_relaxation(h, n_points)
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Trust region implementation
    def trust_region_sqp(h_vec, max_iter=500):
        # Implement trust region SQP
        # Approximate gradient via finite difference
        def get_sqp_gradient(h_vec, k):
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[k] - corr_minus[k]) / (2 * eps)
            return grad
        
        h_new = h_vec.copy()
        for iter in range(max_iter):
            elapsed = time.time() - start_time
            if elapsed >= budget_s * 0.95:
                break
            
            # Compute correlation and find the best k
            corr = correlate(h_new, 1 - h_new, mode="full")
            k = np.argmax(corr) - (len(h_new) - 1)
            
            # Get gradient for best k
            grad = get_sqp_gradient(h_new, k)
            
            # Update h with gradient descent
            step_size = 0.1
            h_new = h_new - step_size * grad
            h_new = project_feasible(h_new)
            
        return h_new
    
    # Perturbation and reinitialization
    patience = 0
    max_patience = 30
    
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Add noise to get out of local optima
                noise = np.random.normal(0, 0.02, n_points)
                h = h + noise
                h = project_feasible(h)
                patience = 0

        # Apply locally improved update
        h = trust_region_sqp(h, max_iter=10)
        h = project_feasible(h)
        
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
