import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular h vector centered at x=1
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(2 * x - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Objective function
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Main optimization loop
    iter_count = 0
    max_iter = 5000  # Reduced to allow more iterations within time budget
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make perturbation in the critical region (0.3, 0.7)
        h_trial = h.copy()
        # Focus perturbations on middle region (0.3 to 0.7)
        region_start = int(0.3 * n_points)
        region_end = int(0.7 * n_points)
        if region_end > region_start:
            noise = np.random.normal(0, 0.03, region_end - region_start)
            h_trial[region_start:region_end] += noise
        else:
            # If region is too small, just add noise to entire vector
            h_trial += np.random.normal(0, 0.03, n_points)
        
        # Repair feasibility
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        # Update h for next round with backup strategy
        if iter_count % 10 == 0:
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
            
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
