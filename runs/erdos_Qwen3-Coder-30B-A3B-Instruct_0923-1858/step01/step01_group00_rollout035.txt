import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 150
    dx = 2.0 / n_points
    T0 = 1.0
    alpha = 0.999
    sigma = 0.1
    shake_prob = 1.0
    max_no_improve = 100
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to the target size
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing
    T = T0
    no_improve_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Generate a random perturbation
        delta_h = np.random.uniform(-T * sigma, T * sigma, n_points)
        
        # Apply perturbation
        h_candidate = h + delta_h
        
        # Project to feasibility: [0,1] and sum(h) = n_points/2
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        try:
            # Evaluate current and candidate solutions
            current_c5 = evaluate_erdos_solution(h)
            candidate_c5 = evaluate_erdos_solution(h_candidate)
            
            # Accept or reject the candidate
            if candidate_c5 < current_c5:
                h = h_candidate
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = h_candidate.copy()
                no_improve_count = 0
            else:
                # Accept with probability based on temperature
                delta_E = candidate_c5 - current_c5
                acceptance_prob = np.exp(-delta_E / T)
                if np.random.rand() < acceptance_prob:
                    h = h_candidate
                no_improve_count += 1
                
        except Exception:
            no_improve_count += 1
        
        # Cooling
        T *= alpha
        
        # Optional shake to escape local minima
        if no_improve_count >= max_no_improve and np.random.rand() < shake_prob:
            h = h + np.random.normal(0, T * sigma, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            no_improve_count = 0
            
        # Occasionally re-evaluate best
        if no_improve_count % 100 == 0:
            try:
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            except:
                pass
    
    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5

    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, actual_c5, n_points)
