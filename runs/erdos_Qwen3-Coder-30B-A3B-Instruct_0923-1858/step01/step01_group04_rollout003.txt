import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

    start_time = time.time()

        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Optimize with sequential convex programming (trust-constr) with multiple restarts
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try multiple restarts with different strategies
    for restart in range(15):
        # Random perturbation
        perturbation = np.random.normal(0, 0.02, n_points)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Try to optimize with trust-constr method
        try:
            res = minimize(
                objective,
                h_trial,
                method='trust-constr',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'disp': False}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
        
        if time.time() - start_time > budget_s - 1:
            break
            
        # Occasionally refine resolution
        if restart % 5 == 0 and n_points < 100:
            n_points = min(100, n_points * 2)
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            best_h = h
            best_c5 = evaluate_erdos_solution(best_h)
            
    # Final validation and return
    final_c5 = evaluate_erdos_solution(best_h)
