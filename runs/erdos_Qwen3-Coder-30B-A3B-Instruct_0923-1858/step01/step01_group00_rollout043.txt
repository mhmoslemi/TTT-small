def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    
    # Initialize h with the provided initial construction if available

            h = np.interp(x_new, x_orig, h)
        # Normalize to satisfy sum(h) * dx = 1
        dx = 2.0 / n_points
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        # Make it more sparse to avoid local minima
        h[:20] = 0.2
        h[-20:] = 0.2
        h = h / np.sum(h) * (n_points / 2)

    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Binary search on C5 using a heuristic approach
    low, high = 0.0, 0.38085
    tolerance = 1e-5
    
    # Sample points for gradient descent
    while time.time() - start_time < budget_s and high - low > tolerance:
        # Try a midpoint
        C = (low + high) / 2.0
        
        # Create binary search objective via optimization
        def objective(h):
            return evaluate_erdos_solution(h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimizing with constraints and bounds
        try:
            result = minimize(

                constraints=constraint,

                if current_c5 <= C:
                    high = C
                else:
                    low = C
                
                # Perturb hint h for next iteration
                h = result.x.copy() + np.random.normal(0, 0.005, n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)

                h = np.random.rand(n_points) * 0.5
                h = h / np.sum(h) * (n_points / 2)

    # Final refinement
    try:
        refine_result = minimize(
            evaluate_erdos_solution,
