import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 70
        h = np.ones(n_points) * 0.5
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    resolutions = [50, 100]  # Start coarse, then fine
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate samples at this resolution
        if res == 50:
            # Use current h resampled to n=50
            n_points = 50
            h_downsampled = np.array([np.mean(h[i::(len(h)//n_points)]) for i in range(n_points)])
            h_trial = h_downsampled * (n_points / 2) / np.sum(h_downsampled)
            h_trial = np.clip(h_trial, 0, 1)
        else:
            # Upsample from 50 to 100 points
            n_points = 100
            h_upsampled = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, 50), h)
            h_trial = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
            h_trial = np.clip(h_trial, 0, 1)
        
        # Differential Evolution for global optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec)
            
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        # DE parameters
        de_kwargs = {
            'bounds': bounds,
            'constraints': constraints,
            'popsize': 40,
            'mutation': 0.8,
            'recombination': 0.9,
            'maxiter': 100,
            'tol': 1e-6,
            'workers': 1,
            'updating': 'sequential'
        }
        
        try:
            res_de = differential_evolution(
                objective,
                bounds,
                constraints=constraints,
                popsize=40,
                mutation=0.8,
                recombination=0.9,
                maxiter=100,
                tol=1e-6,
                workers=1,
                updating='sequential'
            )
            
            if res_de.success:
                # Local refinement with SLSQP
                h_refined = np.clip(res_de.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
                try:
                    result = minimize(
                        objective,
                        h_refined,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=constraints,
                        options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
                    )
                    
                    if result.success:
                        h_candidate = np.clip(result.x, 0, 1)
                        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                        c5_candidate = evaluate_erdos_solution(h_candidate)
                        
                        if c5_candidate < best_c5:
                            best_h = h_candidate
                            best_c5 = c5_candidate
                except:
                    pass
        except:
            pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, len(best_h))
