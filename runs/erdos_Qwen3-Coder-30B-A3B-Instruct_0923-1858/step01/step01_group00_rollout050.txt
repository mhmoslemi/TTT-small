import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and 0 <= h[i] <= 1"""
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = budget_s * 10  # Rough estimate for iterations
    inertia_weight = 0.7
    cognitive_weight = 1.5
    social_weight = 1.5
    perturbation_std = 0.01
    lowpass_alpha = 0.9
    
    # Initialize particles
    if 'initial_h_values' in globals():
        # Use initial construction and scale to current size
        old_h = np.array(initial_h_values)
        # Interpolate to new size
        old_x = np.linspace(0, 2, len(old_h))
        new_x = np.linspace(0, 2, n_points)
        h_interp = np.interp(new_x, old_x, old_h)
        h_particles = []
        for _ in range(swarm_size):
            # Add some noise
            noise = np.random.normal(0, 0.01, n_points)
            h_current = h_interp + noise
            h_current = project_to_feasible(h_current, n_points)
            h_particles.append(h_current)
    else:
        # Create random initial particles
        h_particles = []
        for _ in range(swarm_size):
            h_current = np.random.rand(n_points) * 0.5
            h_current = project_to_feasible(h_current, n_points)
            h_particles.append(h_current)
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    # Store personal bests
    pbest_positions = [h.copy() for h in h_particles]
    pbest_scores = [evaluate_erdos_solution(h) for h in h_particles]
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_scores)
    global_best_position = pbest_positions[global_best_idx].copy()
    global_best_score = pbest_scores[global_best_idx]
    
    # Time tracking
    start_time = time.time()
    iter_count = 0
    
    # PSO main loop
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        try:
            for i in range(swarm_size):
                # Update velocity
                r1, r2 = np.random.rand(2)
                cognitive_component = cognitive_weight * r1 * (pbest_positions[i] - h_particles[i])
                social_component = social_weight * r2 * (global_best_position - h_particles[i])
                velocities[i] = (inertia_weight * velocities[i] + 
                                cognitive_component + 
                                social_component)
                
                # Low-pass filter the velocity
                velocities[i] = lowpass_alpha * velocities[i] + (1 - lowpass_alpha) * np.random.normal(0, 0.05, n_points)
                
                # Update position
                h_particles[i] += velocities[i]
                
                # Apply feasibility
                h_particles[i] = project_to_feasible(h_particles[i], n_points)
                
                # Evaluate
                current_score = evaluate_erdos_solution(h_particles[i])
                
                # Update personal best
                if current_score < pbest_scores[i]:
                    pbest_scores[i] = current_score
                    pbest_positions[i] = h_particles[i].copy()
                
                # Update global best
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best_position = h_particles[i].copy()
                    
            # Apply perturbation to all particles
            for i in range(swarm_size):
                noise = np.random.normal(0, perturbation_std, n_points)
                h_particles[i] += noise
                h_particles[i] = project_to_feasible(h_particles[i], n_points)
                
        except Exception as e:
            # If there's an exception, continue with a potentially better state
            pass
    
    # Final check and return
    final_h = project_to_feasible(global_best_position, n_points)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, n_points)
