import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Create a piecewise constant shape
        h[:n_points//3] = 0.7
        h[n_points//3:2*n_points//3] = 0.3
        h[2*n_points//3:] = 0.7
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
        
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Set initial C_5 value
    C5 = 0.4
    best_C5 = C5
    best_h = h.copy()
    
    # Number of shifts to consider - full range
    shifts = list(range(0, n_points))  # 0 to n_points - 1
    
    # Iteration counter
    iteration = 0
    max_iterations = 1000
    step = 0.0001
    
    try:
        while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
            iteration += 1
            
            # Construct constraints
            # Create the CVXPY variables
            h_vars = Variable(n_points)
            
            # Sum constraint
            constraints = [sum(h_vars) == n_points / 2]
            
            # Bound constraints
            constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
            constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
            
            # Overlap constraints for all shifts
            for k in shifts:
                # Compute the sum for shift k
                # Index for shift k overlap
                # h[i] * (1 - h[i + k]) over all valid i
                # This needs to respect the mode="full" computation
                # The range of i is from 0 to n_points - k - 1 inclusive
                # if k > 0, for i from 0 to n_points - k - 1
                # if k = 0, for i from 0 to n_points - 1
                # The full correlation involves overlapping segments
                
                # We compute the constraint manually to simulate current h behavior
                # We might enumerate the valid indices for the convolution
                
                # Using a compact loop to reduce compile time
                valid_indices = list(range(min(n_points, n_points - k)))
                sum_term = 0
                for i in valid_indices:
                    if 0 <= i + k < n_points:
                        sum_term += h_vars[i] * (1 - h_vars[i + k])
                    else:
                        # Can happen when k >= n_points
                        pass
                        
                # Add constraint to ensure it doesn't exceed current estimate of C5
                constraints.append(sum_term <= C5 / (2.0 / n_points))
                
            # Define the objective as a placeholder; we'll just try to satisfy constraints
            # This isn't a real objective but will return a feasible solution
            # So we'll add a small mild minimization
            # But for maintaining feasibility, the only concern is satisfying constraints
            
            # Solve the problem
            try:
                prob = Problem(Maximize(0), constraints)
                prob.solve(solver=CP.SCS, verbose=False)
                
                # Extract solution
                if prob.status == OPTIMAL or prob.status == FEASIBLE:
                    new_h = np.array([float(h_vars[i].value) for i in range(n_points)])
                    new_C5 = evaluate_erdos_solution(new_h)
                    
                    if new_C5 < best_C5:
                        best_C5 = new_C5
                        best_h = new_h.copy()
                    
                    # Minimize C5 if successful
                    C5 = max(0.01, C5 - step)
                else:
                    # If problem fails to solve with current C5, increase it
                    C5 = min(1.0, C5 + step)
            except:
                # If CVXPY fails, adjust step size and retry
                C5 = min(1.0, C5 + step)
                
        # Final adjustment
        final_h = best_h.copy()
        final_C5 = evaluate_erdos_solution(final_h)
        return (final_h.tolist(), float(final_C5), n_points)
        
    except Exception as e:
        pass
        
    # If something goes wrong, return the most recently found
    final_C5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), float(final_C5), n_points)
