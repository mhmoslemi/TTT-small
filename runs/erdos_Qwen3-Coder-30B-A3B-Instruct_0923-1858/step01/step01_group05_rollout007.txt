import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):

    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2.0)
    
    # Initialize GA parameters
    population_size = 50
    mutation_rate = 0.03
    crossover_rate = 0.5

    population = []

        # Evaluate all individuals
        fitness_scores = []

            fitness_scores.append(c5)

def tournament_select(population, fitness_scores, tournament_size):
    """Select an individual using tournament selection"""
    selected_indices = np.random.choice(len(population), tournament_size)
    selected_fitness = [fitness_scores[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitness)]
    return winner_idx

def crossover(parent1, parent2):
    """Perform intermediate crossover between two parents"""
    alpha = random.random()
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, rate):
    """Mutate an individual with Gaussian noise"""
    for i in range(len(individual)):
        if random.random() < rate:
            individual[i] += np.random.normal(0, 0.03)
