import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Helper function to project h to satisfy integral constraint
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Objective function
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)
    
    # Constraints
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Try multiple random restarts with different strategies
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-start optimization
    n_starts = 20
    for start in range(n_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h_trial = best_h + perturb
        h_trial = project_h(h_trial)
        
        # Try different optimization methods
        methods = ['SLSQP', 'L-BFGS-B']
        for method in methods:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method=method,
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                if res.success:
                    h_candidate = project_h(res.x)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                continue
    
    # Local refinement with simulated annealing approach
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Simulated annealing loop
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        perturb = np.random.normal(0, 0.02, n_points)

        c5_candidate = evaluate_erdos_solution(h_candidate)

        if c5_candidate < current_c5:

            current_c5 = c5_candidate

                current_h = h_candidate
                current_c5 = c5_candidate
        
        # Cool down
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
        
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
