import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)

    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}

    # Use 50 points for initial exploration
    if n_points > 50:

        h_downsampled = np.array([np.mean(h[i:i*2]) for i in range(0, len(h), 2) if i*2 < len(h)])

        else:
            h_downsampled = h_downsampled[:50]
        
        # Use differential evolution for global search at low resolution
        de_bounds = [(0, 1) for _ in range(50)]
        de_result = differential_evolution(

    try:
