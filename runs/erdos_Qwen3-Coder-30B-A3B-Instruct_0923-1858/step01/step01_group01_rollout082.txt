import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, dx, eps=1e-8):
    """Compute gradient of C5 w.r. to each h[i] using central finite differences."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_feasible(h_plus, dx)
        h_minus = project_feasible(h_minus, dx)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad

def initialize_h(n_points):
    """Initialize h with a triadic waveform."""
    t = np.linspace(0, 1, n_points)
    h = np.abs(np.sin(2 * np.pi * t * (n_points / 3))) ** 2
    h = np.clip(2 * h - 0.5, 0, 1)
    return h

            result = minimize(
                objective,
                h_perturbed.flatten(),
                method='L-BFGS-B',
                bounds=bounds,
                jac=gradient,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-8}
            )
            
            if result.success:
                h_result = result.x.reshape(-1)
                h_result = project_feasible(h_result, dx)
                current_C = evaluate(h_result, dx)
                if current_C < best_C:
                    best_C = current_C
                    best_h = h_result.copy()
        except:
            pass
    
    # Final refinement with projected gradient descent
    h_refined = best_h.copy()
    lr = 0.01

        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient(h_refined, dx)
        h_refined = h_refined - lr * grad
        h_refined = project_feasible(h_refined, dx)
        
        # Projected gradient step with adaptive learning rate
        if i % 100 == 0:
            lr *= 0.95
    
    final_C = evaluate(h_refined, dx)
    if final_C < best_C:
        best_C = final_C
        best_h = h_refined
    
    return best_h, best_C
