import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Use a simple initial construction
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    h = h / np.sum(h) * (n_points / 2)

    # Keep track of best solution
    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Binary search on C5 bound
    low_c5 = 0.0
    high_c5 = 0.38085

    # Convex relaxation approach with LP formulations
    def is_feasible(h, C):
        try:
            # Formulate the LP problem to check feasibility
            h_var = cp.Variable(n_points)
            constraints = [h_var >= 0, h_var <= 1]
            constraints.append(cp.sum(h_var) == n_points / 2)
            
            # Generate constraints for each k
            for k in range(n_points):
                expr = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    expr += h_var[i] * (1 - h_var[j]) * dx
                constraints.append(expr <= C)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            return prob.status == cp.OPTIMAL
        except:
            return False

    def gradient_descent_step(h, c5):
        # Simple gradient descent for refinement
        # Exact gradient isn't possible due to complex structure,
        # use heuristic to perturb and improve
        step_size = 0.001
        iterations = 100
        for i in range(iterations):
            current_c5 = evaluate_erdos_solution(h)
            # Approximate gradient using finite differences
            grad = np.zeros_like(h)
            delta = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += delta
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
                c5_plus = evaluate_erdos_solution(h_plus)
                grad[j] = (c5_plus - current_c5) / delta
            
            h = h - step_size * grad
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
        return h

    # Binary search loop
    while time_left() > 20 and high_c5 - low_c5 > 1e-6:
        mid_c5 = (low_c5 + high_c5) / 2.0
        if is_feasible(h, mid_c5):
            high_c5 = mid_c5
            # Refine with gradient descent if possible
            try:
                refined_h = gradient_descent_step(h, mid_c5)
                refined_c5 = evaluate_erdos_solution(refined_h)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
            except:
                pass
        else:
            low_c5 = mid_c5
            
        # Update h with a slightly adjusted version (prevents stagnation)
        h = h + 0.01 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

    # Final refinement with gradient descent
    x0 = best_h.copy()
    try:
        result = minimize(evaluate_erdos_solution, x0, method='L-BFGS-B',
                          bounds=[(0, 1) for _ in range(n_points)],
                          options={'maxiter': 500})
        if result.success:
            final_c5 = evaluate_erdos_solution(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x
    except:
        pass

    return (best_h, best_c5, n_points)
