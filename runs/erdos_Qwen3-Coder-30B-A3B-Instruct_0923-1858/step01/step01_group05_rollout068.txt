import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a smooth triangular construction centered at x=1
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Start optimization loop with time budget
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Store current best
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Local optimization with random perturbation
        h_trial = h.copy()
        
        # Apply noise to middle region where overlap is concentrated
        mid_indices = np.where((np.linspace(0, 2, n_points) >= 0.3) & 
                               (np.linspace(0, 2, n_points) <= 0.7))[0]
        for i in mid_indices:
            h_trial[i] += np.random.normal(0, 0.03)
        
        # Repair h to stay in bounds and maintain sum constraint
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Optimize with constrained minimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5 or np.random.rand() < 0.1:  # Accept sometimes even if worse
                    h = h_new.copy()
        except:
            pass
            
    # Final validation with strict enforcement
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2.0)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
