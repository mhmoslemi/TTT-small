import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for better resolution
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        # Ensure it's in [0,1]
        h = np.clip(h, 0, 1)
    else:
        # Start with uniform
        h = np.ones(n_points) * 0.5
        # Zero endpoints to help with boundary conditions
        h[0] = 0.0
        h[-1] = 0.0

    # Normalize so that sum(h) * dx = 1, i.e., sum(h) = n_points/2
    h = h / np.sum(h) * (n_points / 2)
    
    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    cooling_rate = 0.999
    perturbation_scale = 0.1
    shake_threshold = 100
    shake_prob = 1.0
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Track recent improvements
    last_improvement = 0
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Generate perturbation
        delta_h = np.random.uniform(-T * perturbation_scale, T * perturbation_scale, n_points)
        h_prime = h + delta_h
        
        # Project to [0,1]
        h_prime = np.clip(h_prime, 0, 1)
        
        # Rescale to maintain sum(h) = n_points/2
        h_prime = h_prime / np.sum(h_prime) * (n_points / 2)
        
        # Evaluate
        c5_prime = evaluate_erdos_solution(h_prime)
        
        # Accept or reject
        delta_E = c5_prime - best_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_prime.copy()
            if c5_prime < best_c5:
                best_h = h_prime.copy()
                best_c5 = c5_prime
                last_improvement = 0
            else:
                last_improvement += 1
        else:
            last_improvement += 1
            
        # Cooling and shaking
        T *= cooling_rate
        if last_improvement > shake_threshold and np.random.rand() < shake_prob:
            # Add noise to escape local minima
            h += np.random.normal(0, T * perturbation_scale, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            last_improvement = 0
        
        # Occasionally check if we should break early
        if iteration % 100 == 0 and time.time() - start_time > budget_s * 0.95:
            break

    # Final check for feasibility and evaluation
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
