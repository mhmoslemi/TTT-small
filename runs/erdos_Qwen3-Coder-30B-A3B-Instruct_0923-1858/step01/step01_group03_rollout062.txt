import numpy as np
import time
from scipy.signal import correlate

        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 75
    max_iter = 2000
    perturb_rate = 0.04  # 4% of points
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Run tabu search
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate neighbor solution
        candidate_h = h.copy()
        n_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        for idx in indices:
            # perturb in [0,1]
            candidate_h[idx] = np.random.uniform(0, 1)
            
        # Project to feasible region
        candidate_h = project_feasible(candidate_h)
        
        # Check if in tabu list
        candidate_hash = tuple(np.round(candidate_h, decimals=6))
        if candidate_hash in tabu_list:
            continue
            
        # Evaluate
        candidate_C5 = compute_C5(candidate_h)
        
        # Update if better and not tabu
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate_h.copy()
            
        # Add to tabu list
        tabu_list.append(candidate_hash)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Move to candidate
        h = candidate_h.copy()
        iter_count += 1
        if iter_count >= max_iter:
            break
        
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
