import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization with existing construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation approach
    def solve_convex_relaxation(h_initial, n_points, dx):
        # Setting up the convex problem: max λ s.t. h[i]*(1-h[i+k]) >= λ for all k
        h = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2,
            lambda_var <= 0.5  # conservative upper bound
        ]
        
        # Generate overlap constraints
        # This is approximate but fast - use an initial solution to guide
        h_vec = h_initial.copy()
        for k in range(n_points):
            # Compute O_k = sum h[i] * (1 - h[i+k]) for lag k
            # Going to compute manually using indices
            lag_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                lag_sum += h_vec[i] * (1 - h_vec[j])
            constraints.append(lambda_var <= lag_sum * dx)
        
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                return np.array(h.value)
        except:
            pass
            
        # Fallback to initial if convex solve fails
        return h_initial.copy()
    
    # Convex relaxation phase - 70% of time
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.7:
        h = solve_convex_relaxation(h, n_points, dx)
    
    # Gradient descent refinement with projection
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Gradient descent parameters
    eta = 0.01
    max_patience = 50
    patience = 0
    
    # Main optimization loop - 20% of time
    elapsed = time.time() - start_time
    iter_count = 0
    
    while elapsed < budget_s * 0.9:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Compute gradient estimation
        def compute_grad(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_vec) - 1)

            grad = np.zeros_like(h_vec)
            eps = 1e-6
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            return grad
        
        # Update step
        grad = compute_grad(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Adaptive learning rate
        if iter_count % 100 == 0:
            eta *= 0.95
            
        iter_count += 1
        elapsed = time.time() - start_time

    # Random restart phase - 10% of time
    elapsed = time.time() - start_time
    num_restarts = 5
    for _ in range(num_restarts):
        if elapsed >= budget_s * 0.95:
            break
        h_restart = np.random.rand(n_points)
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)  # normalize
        h_restart = np.clip(h_restart, 0, 1)
        
        current_C5 = compute_C5(h_restart)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_restart.copy()
        elapsed = time.time() - start_time
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
