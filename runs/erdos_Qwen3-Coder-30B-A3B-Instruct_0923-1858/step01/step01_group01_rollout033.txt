import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, dx, eps=1e-8):
    """Compute gradient of C5 w.r. to each h[i] using central finite differences."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_feasible(h_plus, dx)
        h_minus = project_feasible(h_minus, dx)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad

def initialize_h(n_points):
    """Initialize h with atriadic waveform pattern."""
    t = np.linspace(0, 1, n_points)
    h = np.abs(np.sin(2 * np.pi * t * (n_points / 3))) ** 2
    h = np.clip(2 * h - 0.5, 0, 1)
    return h

def optimize(initial_h, dx, n_points, budget_s=600):
    """Optimize h using L-BFGS-B with structured initialization."""
    start_time = time.time()
    best_h = initial_h.copy()
    best_C = evaluate(initial_h, dx)
    
    # For the outer loop, we'll use a simpler scheme
    # with functional evaluations and adaptation
    lr = 0.01
    # Use simple gradient descent but with adaptive learning rate
    for i in range(100000):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(best_h, dx)
        
        # Update step with adaptive learning rate
        new_h = best_h - lr * grad
        new_h = project_feasible(new_h, dx)
        
        # Periodic perturbation
        if i % 100 == 0:
            perturb = 0.001 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 100)
            new_h = (new_h + perturb).clip(0, 1)
            new_h = project_feasible(new_h, dx)
            
        # Evaluate
        new_C = evaluate(new_h, dx)
        
        # Update learning rate based on improvement
        if new_C < best_C:
            best_h = new_h.copy()
            best_C = new_C
            lr *= 1.05  # Accelerate
        else:
            lr *= 0.95  # Slow down
            
        # Regularly reassess and keep close to feasible
        if i % 200 == 0:
            # Project back to feasible
            best_h = project_feasible(best_h, dx)
    
    return best_h, best_C
