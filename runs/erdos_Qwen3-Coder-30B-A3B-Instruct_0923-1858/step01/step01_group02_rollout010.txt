import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Start with C5 around 0.4
    C5 = 0.4
    step_size = 0.0001
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    while time.time() - start_time < budget_s - 1:
        # Initialize CVXPY problem
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        
        # Sum constraint
        constraints.append(sum(h_var) == n_points / 2)
        
        # Overlap constraints
        dx = 2.0 / n_points
        for k in range(n_points):
            # For correlation, we need to consider it correctly
            # The overlap for shift k is sum_i h[i] * (1 - h[i+k])
            # But in circular shift, we have to be careful
            # But in correlation, the indices match correctly
            corr_term = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_term += h_var[i] * (1 - h_var[j])
            constraints.append(corr_term <= C5 / dx)
        
        # Objective - minimize C5 in the constraints or just ensure constraints are met
        # Actually, we want to minimize C5, so we can just make this a feasibility problem
        # and iteratively decrease C5 until infeasible.
        
        # This is actually a bit complex. Let's switch approach:
        # Use bisection-like iteration on C5 instead of CVXPY for this incremental part.
        # But need to use CVXPY less in the optimization loop for better performance.
        
        # If current solution works, try to decrease C5
        # Try to solve with current C5
        
        # Simple gradient step approach: 
        # Project feasible solution onto space where max overlap <= C5 / dx
        try:
            # Simple iterative improvement with tight projections
            h_new = h.copy()
            
            # Instead of iteratively solving CVXPY, just adjust gradient-like update
            # This is a heuristic experimentation - not rigorous optimization
            # But to avoid CVXPY at each step, let's do incremental search
            
            # Assume we get to a feasible solution
            current_c5 = evaluate_erdos_solution(h_new)
            
            if current_c5 <= C5:
                # If feasible and better, use it
                if current_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = current_c5
                    # Try a smaller C5
                    C5 = max(0.0, C5 - step_size)
                else:
                    # Otherwise, just reduce step size and keep going
                    C5 = max(0.0, C5 - step_size)
            else:
                # If infeasible, increase C5
                C5 = min(1.0, C5 + step_size)
            
            # Random perturbation to avoid local minima
            if np.random.random() > 0.5:
                pert = np.random.normal(0, 0.001, n_points)
                h_new = np.clip(h_new + pert, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)  # Renormalize
                h = h_new
            else:
                h = best_h.copy()
                
        except Exception as e:
            # Recovery from error by using best_h and keep trying
            h = best_h.copy()
            C5 += step_size

        # Time check to avoid hard timeout
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final verification
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure bounds
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), float(final_c5), n_points)
