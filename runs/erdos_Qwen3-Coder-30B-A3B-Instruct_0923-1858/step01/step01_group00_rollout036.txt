import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for better resolution
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        # Interpolate to 150 points
        h_raw = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_raw))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h_raw)
        h = np.clip(h, 0, 1)
        # Normalize to sum(h) = n_points / 2
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    perturbation_scale = 0.1
    shake_threshold = 100
    shake_prob = 1.0
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Start timer
    start_time = time.time()
    
    iteration = 0
    no_improve_count = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Generate perturbation
        delta_h = np.random.uniform(-temp * perturbation_scale, temp * perturbation_scale, n_points)
        
        # Update candidate
        h_candidate = h + delta_h
        
        # Project to feasibility
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate candidate
        try:
            c5_candidate = evaluate_erdos_solution(h_candidate)
        except:
            no_improve_count += 1
            continue
        
        # Accept or reject
        delta_e = c5_candidate - best_c5
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / temp):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
        
        # Cool temperature
        temp *= cooling_rate
        
        # Shake periodically if no improvement
        if no_improve_count >= shake_threshold and np.random.rand() < shake_prob:
            h += np.random.normal(0, temp * perturbation_scale, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            no_improve_count = 0
        
        # Occasionally reinitialize if stuck
        if iteration % 500 == 0 and temp < 0.01:
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
            temp = 1.0  # Reset temperature
            no_improve_count = 0
    
    return (best_h, best_c5, n_points)
