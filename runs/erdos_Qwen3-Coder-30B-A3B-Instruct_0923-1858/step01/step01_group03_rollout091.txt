import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial solution if available
    try:
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    except NameError:
        # Use a triangular solution as fallback
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_initial = np.maximum(0, 1 - np.abs(x - 1))
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    Pop_size = 100
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    stagnation_limit = 50
    mutation_rate = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate initial swarm
    fitness = []
    pbest = []
    pbest_fitness = []
    
    for i in range(Pop_size):
        h = swarm[i]
        corr = correlate(h, 1 - h, mode="full")
        fit = np.max(corr) * (2.0 / n_points)
        fitness.append(fit)
        pbest.append(h.copy())
        pbest_fitness.append(fit)
    
    # Initialize global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    # PSO main loop
    iter_count = 0
    stagnation = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        w = w_start - (w_start - w_end) * (iter_count / (budget_s * 0.8))
        
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = (w * velocities[i] + 
                         c1 * r1 * (pbest[i] - swarm[i]) + 
                         c2 * r2 * (gbest - swarm[i]))
            
            velocities[i] = vel_update
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Evaluate fitness
            corr = correlate(swarm[i], 1 - swarm[i], mode="full")
            fit = np.max(corr) * (2.0 / n_points)
            
            # Update personal best
            if fit < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fit
            
            # Update global best
            if fit < gbest_fitness:
                gbest = swarm[i].copy()
                gbest_fitness = fit
                stagnation = 0
            else:
                stagnation += 1
        
        # Mutation to avoid local minima
        if stagnation > stagnation_limit:
            for i in range(int(Pop_size * mutation_rate)):
                idx = np.random.randint(0, Pop_size)
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                # Reevaluate
                corr = correlate(swarm[idx], 1 - swarm[idx], mode="full")
                fit = np.max(corr) * (2.0 / n_points)
                pbest[idx] = swarm[idx].copy()
                pbest_fitness[idx] = fit
                if fit < gbest_fitness:
                    gbest = swarm[idx].copy()
                    gbest_fitness = fit
            stagnation = 0
        
        iter_count += 1
    
    # Final evaluation
    corr = correlate(gbest, 1 - gbest, mode="full")
    final_C5 = np.max(corr) * (2.0 / n_points)
    
    return (gbest.tolist(), final_C5, n_points)
