import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint is satisfied
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define objective function: minimize C5
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Initial best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-start optimization with different perturbations
    max_iter = 20
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Generate perturbed starting point
        perturbation = np.random.normal(0, 0.03, n_points)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        # Renormalize to satisfy integral constraint
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Optimize from this point
        try:
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-7}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
            
        iter_count += 1
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
