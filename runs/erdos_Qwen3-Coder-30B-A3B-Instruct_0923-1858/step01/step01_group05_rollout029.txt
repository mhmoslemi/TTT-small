import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points for better resolution
        old_x = np.linspace(0, 2, n_points)
        new_x = np.linspace(0, 2, 140)
        h = np.interp(new_x, old_x, h)
        n_points = 140
    else:
        # Create a simple initial construction
        n_points = 140
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-10:mid+10] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Normalize to correct sum
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Ensure all values are within [0,1]
    h = np.clip(h, 0, 1)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping loop
    iteration = 0
    max_iter = 1000
    sigma = 0.05
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Perturb
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)
        
        try:
            # Local optimization
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        iteration += 1
        # Gradually decrease noise level
        if iteration % 50 == 0 and sigma > 0.001:
            sigma *= 0.5
            
    # Final refinement
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    h_copy = best_h.copy()
    try:
        res = minimize(objective, h_copy, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    except:
        pass
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
