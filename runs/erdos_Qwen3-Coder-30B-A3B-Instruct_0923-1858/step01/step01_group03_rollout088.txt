            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iterations = 2000
    perturb_rate = 0.05
    perturb_magnitude = 0.1
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def generate_neighbor(current_h):
        neighbor = current_h.copy()
        # Perturb a small percentage of entries
        n_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for idx in indices:
            delta = np.random.uniform(-perturb_magnitude, perturb_magnitude)
            neighbor[idx] += delta
            
        return project_feasible(neighbor)
    
    # Initialize best solution
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = current_h.copy()
    
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate neighbor
        neighbor = generate_neighbor(current_h)
        
        # Check if neighbor is in tabu list
        neighbor_tuple = tuple(neighbor)
        if neighbor_tuple in tabu_list:
            # Accept with low probability
            if np.random.random() > 0.2:
                iter_count += 1
                continue
        
        # Evaluate neighbor
        neighbor_C5 = compute_C5(neighbor)
        
        # Accept if better or satisfies tabu criteria
        if neighbor_C5 < current_C5 or (neighbor_C5 < best_C5 and neighbor_tuple not in tabu_list):

            tabu_list.append(neighbor_tuple)

        best_C5 = final_C5
        best_h = best_h.copy()
