import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create initial h as a triangular function
        n_points = 70
        h = np.array([1.0 - abs(2.0 * i / (n_points - 1) - 1.0) for i in range(n_points)])
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    # Track best solution
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Global exploration with local refinement
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h in the critical region [0.3, 0.7]
        h_trial = h.copy()
        # Identify critical region indices
        start_idx = int(0.3 * n_points)
        end_idx = int(0.7 * n_points)
        # Add moderate Gaussian noise
        noise = np.random.normal(0, 0.03, end_idx - start_idx)
        h_trial[start_idx:end_idx] += noise
        # Repair feasibility
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Try optimizing with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        if iter_count % 10 == 0:
            # Simple local search by small perturbations
            h_trial = h + np.random.normal(0, 0.005, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            
            c5_new = objective(h_trial)
            if c5_new < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_new
                h = h_trial.copy()
        
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
