import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_h_bilevel(initial_h, dx, n_points, budget_s=600):
    best_h = initial_h.copy()
    best_C = evaluate(initial_h, dx)
    
    # Parameters for optimization
    lr = 0.01
    max_iter = 10000
    t0 = time.time()
    max_time = budget_s - 10  # leave some buffer
    
    # Gradient estimation using finite differences
    for i in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        # Estimate gradient
        grad = np.zeros_like(best_h)
        epsilon = 1e-5
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += epsilon
            h_plus = project(h_plus, dx)
            
            h_minus = best_h.copy()
            h_minus[j] -= epsilon
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)
        
        # Update step
        new_h = best_h - lr * grad
        new_h = project(new_h, dx)
        
        # Perturbation to escape local optima
        if np.random.rand() < 0.05:
            perturb = np.random.normal(0, 0.01, size=n_points)
            new_h += perturb
            new_h = project(new_h, dx)
        
        # Evaluate new solution
        new_C = evaluate(new_h, dx)
        if new_C < best_C:
            best_C = new_C
            best_h = new_h.copy()
            lr = max(0.001, lr * 0.995)  # Adaptive decay
        else:
            lr *= 0.99  # decay learning rate
            
    return best_h, best_C
