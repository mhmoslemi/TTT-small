import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach: start with coarse resolution for faster exploration
    resolutions = [50, 100]  # Start with 50 points then 100
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create a population for DE
        popsize = 40
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Helper function for differential evolution
        def de_objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            integral_constraint = np.sum(h_vec) - n_points / 2
            penalty = 1e6 * integral_constraint**2  # Penalty for constraint violation
            return evaluate_erdos_solution(h_vec) + penalty
        
        # Differential evolution with bounds and population size
        try:
            res = differential_evolution(
                de_objective,
                bounds,
                popsize=popsize,
                maxiter=100,
                mutation=0.8,
                recombination=0.9,
                seed=seed,
                disp=False,
                tol=1e-6
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                # Adjust to integral constraint
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
            
        # Local SLSQP refinement for better precision
        if time.time() - start_time > budget_s - 5:
            break
            
        # Define constraint for integral = 1
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        try:
            res = minimize(
                evaluate_erdos_solution,
                best_h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)

                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
            
        # Periodic evaluation check to ensure we don't run out of time
        if time.time() - start_time > budget_s - 2:
            break
    
    # Final return
    return (list(best_h), best_c5, len(best_h))
