import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Set a target for C5
    target_c5 = 0.3807
    
    # Use a gradient-based method with constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    # Iteratively refine using local optimization
    iteration_max = 1000
    for it in range(iteration_max):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Simple optimization step with small perturbations
            x0 = best_h + np.random.normal(0, 0.005, n_points)
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)  # Renormalize
            
            result = minimize(objective, x0, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 100})
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_h = result.x.copy()
                    best_c5 = current_c5
                    
                    # Early stopping if we reach target
                    if best_c5 <= target_c5:
                        break
                        
        except Exception:
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
