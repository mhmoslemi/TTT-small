import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to our n_points
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h)), h)
        h = h_interp / np.sum(h_interp) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()

    # Start timer
    start_time = time.time()
    
    # Try different initial configurations to escape local minima
    initial_configs = [h.copy()]
    for _ in range(5):
        # Add some random perturbation to a clone of the best h
        perturbed = h.copy() + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
        initial_configs.append(perturbed.copy())
    
    # Try optimization from each initial configuration
    for i, h_start in enumerate(initial_configs):
        h_current = h_start.copy()
        iteration = 0
        while time.time() - start_time < budget_s and iteration < 1000:  # Prevent infinite loops
            try:
                # Objective function
                def obj(h):
                    return evaluate_erdos_solution(h)
                
                # Bounds for h: [0, 1]
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Constraint for sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2
                
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                # Run optimization
                result = minimize(
                    obj,
                    h_current.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 100},
                    tol=1e-6
                )
                
                if result.success:
                    current_c5 = evaluate_erdos_solution(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()
                    
                    # Use this result as starting point for next iteration
                    h_current = result.x.copy()
                else:
                    # If optimization fails, try a new initial guess
                    h_current = np.random.rand(n_points) * 0.5
                    h_current = h_current / np.sum(h_current) * (n_points / 2)
                
            except Exception as e:
                # On exception, try with a new initial guess  
                h_current = np.random.rand(n_points) * 0.5
                h_current = h_current / np.sum(h_current) * (n_points / 2)
            
            iteration += 1
            
            # Occasionally try a Markov chain step to escape saddle points
            if iteration % 50 == 0 and time.time() - start_time < budget_s:
                h_current = best_h.copy() + np.random.normal(0, 0.005, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current / np.sum(h_current) * (n_points / 2)
        
        # Early exit if time is almost up
        if time.time() - start_time > 0.9 * budget_s:
            break

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    
    try:
        actual_c5 = evaluate_erdos_solution(final_h)
    except:
        actual_c5 = best_c5

    return (final_h, actual_c5, n_points)
