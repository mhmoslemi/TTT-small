import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def compute_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def perturb_and_repair(h, strength=0.03):
    # Apply perturbation in the middle region
    n = len(h)
    h_new = h.copy()
    mid_start = int(0.3 * n)
    mid_end = int(0.7 * n)
    
    # Add noise to middle region only
    for i in range(mid_start, mid_end):
        h_new[i] += np.random.normal(0, strength)
    
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    # Rescale to sum = n/2
    h_new = h_new * (n / 2.0) / np.sum(h_new)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create symmetric triangular function
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * (n_points / 2.0)
    
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Perturb and repair
        h_trial = perturb_and_repair(h, strength=0.03)
        
        # Try local optimization with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_vec):
            corr = np.correlate(h_vec, 1-h_vec, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        # Simplified gradient-like step to explore
        if iter_count % 5 == 0:
            h = perturb_and_repair(h, strength=0.01)
            
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
