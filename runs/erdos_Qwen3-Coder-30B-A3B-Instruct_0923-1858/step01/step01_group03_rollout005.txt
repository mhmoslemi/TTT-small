import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Genetic Algorithm parameters
    population_size = 100
    mutation_rate = 0.2
    tournament_size = 5
    generations = 500
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def create_individual():
        # Create a random individual
        h = np.random.rand(n_points)
        h = project_feasible(h)
        return h
    
    def tournament_selection(population, fitnesses):
        selected_idx = np.random.choice(len(population), tournament_size)
        best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        # Single-point crossover
        crossover_point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(individual):
        # Gaussian mutation with adaptive rate
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.05)
        return project_feasible(mutated)
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [compute_C5(ind) for ind in population]
        
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_C5:
            best_C5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            # Reset population
            population = [h.copy()]
            for _ in range(population_size - 1):
                population.append(create_individual())
            no_improvement_count = 0
        
        # Selection, crossover, mutation
        new_population = [population[best_idx]]  # Elitism
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
