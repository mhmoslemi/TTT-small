import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random particles
    for _ in range(Pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        
    swarm = np.array(swarm)
    velocities = np.array(velocities)
    
    # Track personal bests
    pbest = swarm.copy()
    pbest_fitness = np.full(Pop_size, float('inf'))
    
    # Track global best
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    def compute_C5(h_vec):
        # Convert to correct size
        h_vec = np.array(h_vec)
        # Note: correlate with mode='full' to handle all shifts
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum over all shifts
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Main loop
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 50
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Evaluate swarm
        current_fitness = np.array([compute_C5(swarm[i]) for i in range(Pop_size)])
        
        # Update personal bests
        for i in range(Pop_size):
            if current_fitness[i] < pbest_fitness[i]:
                pbest_fitness[i] = current_fitness[i]
                pbest[i] = swarm[i].copy()
                
        # Update global best
        min_fitness_idx = np.argmin(current_fitness)
        if current_fitness[min_fitness_idx] < gbest_fitness:
            gbest_fitness = current_fitness[min_fitness_idx]
            gbest_idx = min_fitness_idx
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update best solution
        if gbest_fitness < best_C5:
            best_C5 = gbest_fitness
            best_h = swarm[gbest_idx].copy()
            
        # Reset if stagnation
        if stagnation_count > max_stagnation:
            # Reset 10% of particles
            reset_count = max(1, Pop_size // 10)
            reset_indices = np.random.choice(Pop_size, reset_count, replace=False)
            for idx in reset_indices:
                h_rand = np.random.rand(n_points)
                h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                swarm[idx] = h_rand
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0
            
        # Update velocities and positions
        for i in range(Pop_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
        # Decay inertia weight
        w = max(0.4, w * 0.999)
        
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
