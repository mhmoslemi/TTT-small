            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation to get a good starting point
    def convex_relaxation():
        # This computes a convex relaxation that can give us a good starting point
        h_init = np.zeros(n_points)
        h_init[10:35] = 1.0  # Start with a sparse solution
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        return h_init
    
    # Use convex relaxation for better initialization
    h = convex_relaxation()
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Trust region optimization with perturbations
    eta = 0.01
    max_iter = 1000
    patience = 0
    max_patience = 50
    
    def compute_gradient(h_vec, k_idx):
        # Compute gradient for a specific shift k_idx
        h_vec = np.array(h_vec)
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[k_idx] - corr_minus[k_idx]) / (2 * eps)
        return grad
    
    # Optimization loop
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb the solution
                idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                for i in idx:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
        
        # Determine the shift that gives the maximum
        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        
        # Compute gradient for that specific k
        grad = compute_gradient(h, max_k_idx)
        
        # Update with gradient
        h = h - eta * grad
        h = project_feasible(h)
        
        # Decay learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
