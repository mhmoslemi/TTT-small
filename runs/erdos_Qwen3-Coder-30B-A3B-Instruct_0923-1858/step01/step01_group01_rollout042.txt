import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, mutation_strength=0.1):
    """Apply Gaussian mutation to h."""
    mutated = h.copy()
    idx = np.random.randint(len(mutated))
    mutated[idx] += np.random.normal(0, mutation_strength)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Perform uniform crossover between parents."""
    child = parent1.copy()
    mask = np.random.rand(len(parent1)) < 0.5
    child[mask] = parent2[mask]
    return child

def tournament_selection(population, fitnesses, k=2):
    """Select parent using tournament selection."""
    selected_indices = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), k, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
        selected_indices.append(winner_idx)
    return [population[i] for i in selected_indices]
