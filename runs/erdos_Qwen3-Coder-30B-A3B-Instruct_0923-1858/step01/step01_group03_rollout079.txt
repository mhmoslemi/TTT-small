import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:

        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation to get good starting point
    if time.time() - start_time < budget_s * 0.7:
        try:
            # Set up convex problem
            h_cvx = cp.Variable(n_points)
            lambda_var = cp.Variable()
            
            # Constraints: sum(h) = n_points/2
            constraints = [cp.sum(h_cvx) == n_points / 2]
            
            # Constraints: 0 <= h <= 1
            constraints += [h_cvx >= 0, h_cvx <= 1]
            
            # Add overlap constraints
            for k in range(n_points):
                # Compute correlation at lag k
                lhs = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    lhs += h_cvx[i] * (1 - h_cvx[j])
                constraints.append(lhs >= lambda_var)
            
            prob = cp.Problem(cp.Maximize(lambda_var), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                relaxed_h = h_cvx.value
                h = np.clip(relaxed_h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            pass
    
    # Gradient descent with adaptivity
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Initial refinement with gradient descent
    eta = 0.01
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while time.time() - start_time < budget_s * 0.9:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                # Rescale
                h = project_feasible(h)
                patience = 0
    
        # Compute gradient and update
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Decay learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1
    
    # Random restarts
    restart_count = 0
    while time.time() - start_time < budget_s * 0.99:
        h_restart = np.random.rand(n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        # Gradient descent from this restart
        eta_restart = 0.01
        patience_restart = 0
        max_patience_restart = 50
        
        for _ in range(200):
            if time.time() - start_time >= budget_s * 0.99:
                break
                
            current_C5 = compute_C5(h_restart)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_restart.copy()
                patience_restart = 0
            else:
                patience_restart += 1
                if patience_restart > max_patience_restart:
                    break
            
            grad = get_gradient(h_restart)
            h_restart = h_restart - eta_restart * grad
            h_restart = project_feasible(h_restart)
            
            eta_restart *= 0.995
        
        restart_count += 1
    
    return (best_h.tolist(), best_C5, len(best_h))
