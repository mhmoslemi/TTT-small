import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import Variable, Minimize, Problem, norm, sum, constraints
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Create an alternating pattern to reduce overlaps
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 0.6
            else:
                h[i] = 0.4
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    # Optimization settings
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturbations
    perturbation_magnitude = 0.1
    
    # Optimization with adaptive perturbations
    counter = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 1:
        counter += 1
        
        try:
            # Perturb the current solution
            x0 = h + np.random.normal(0, perturbation_magnitude, n_points)
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)
            
            # Local optimization with L-BFGS-B
            res = minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 100})
            
            if not res.success:
                continue
                
            current_c5 = evaluate_erdos_solution(res.x)
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5
                last_improvement = 0
            else:
                last_improvement += 1
                
            # Adjust perturbation magnitude
            if last_improvement > 20:
                perturbation_magnitude = min(0.2, perturbation_magnitude * 1.1)
            elif last_improvement < 5:
                perturbation_magnitude = max(0.01, perturbation_magnitude * 0.9)
            
            if time.time() - start_time > budget_s - 0.5:
                break
                
        except Exception:
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
