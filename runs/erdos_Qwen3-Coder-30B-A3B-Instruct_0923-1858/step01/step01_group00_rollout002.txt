import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available

        h = h / np.sum(h) * (n_points / 2)
    else:

        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 600  # Assuming ~1 iter per second based on approximate time estimates
    
    # Initialize swarm
    swarm = np.random.rand(swarm_size, n_points) * 0.5
    swarm = swarm / np.sum(swarm, axis=1, keepdims=True) * (n_points / 2)
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    # Evaluate initial swarm
    fitness = np.array([evaluate_erdos_solution(particle) for particle in swarm])
    
    # Personal best
    pbest = swarm.copy()
    pbest_fitness = fitness.copy()
    
    # Global best
    gbest_idx = np.argmin(fitness)
    gbest = swarm[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    alpha = 0.9  # Low-pass filter alpha for velocity
    
    # Main loop
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (pbest[i] - swarm[i]) 
                           + c2 * r2 * (gbest - swarm[i]))
            
            # Apply low-pass filtering to velocity
            velocities[i] = alpha * velocities[i] + (1 - alpha) * np.random.normal(0, 0.001, n_points)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Apply feasibility constraints - clip to [0,1]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Project to sum constraint
            swarm[i] = swarm[i] / np.sum(swarm[i]) * (n_points / 2)
            
            # Evaluate fitness
            current_fitness = evaluate_erdos_solution(swarm[i])
            
            # Update personal best
            if current_fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = current_fitness
            
            # Update global best
            if current_fitness < gbest_fitness:
                gbest = swarm[i].copy()
                gbest_fitness = current_fitness
        
        # Apply perturbation to all particles to avoid local minima
        perturb = np.random.normal(0, 0.01, (swarm_size, n_points))
        swarm += perturb
        
        # Re-project all particles to feasible region
        for i in range(swarm_size):
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] / np.sum(swarm[i]) * (n_points / 2)
        
        iter_count += 1
    
    # Final evaluation of the best solution
    try:
        final_c5 = evaluate_erdos_solution(gbest)
    except:
        final_c5 = gbest_fitness
    
    return (gbest, final_c5, n_points)
