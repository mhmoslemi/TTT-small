import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a triangular solution
    try:
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_initial = np.maximum(0, 1 - np.abs(x - 1))
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 10000
    stagnation_limit = 50
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add the initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points) * 0.1)
    
    # Fill the rest with random feasible solutions
    for _ in range(Pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points) * 0.1)
    
    # Track personal and global bests
    pbest = [s.copy() for s in swarm]
    pbest_fitness = [float('inf')] * Pop_size
    
    global_best = swarm[0].copy()
    global_best_fitness = float('inf')
    
    # Evaluate initial fitness
    for i in range(Pop_size):
        h = swarm[i]
        overlaps = correlate(h, 1 - h, mode="full")
        fitness = np.max(overlaps) * (2.0 / n_points)
        pbest_fitness[i] = fitness
        if fitness < global_best_fitness:
            global_best_fitness = fitness
            global_best = h.copy()
    
    # PSO main loop
    iter_count = 0
    stagnation = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Evaluate fitness
            overlaps = correlate(swarm[i], 1 - swarm[i], mode="full")
            fitness = np.max(overlaps) * (2.0 / n_points)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best = swarm[i].copy()
                    stagnation = 0
        
        # Check for stagnation
        stagnation += 1
        if stagnation >= stagnation_limit:
            # Randomly reset some particles
            for i in range(int(Pop_size * 0.1)):
                idx = np.random.randint(0, Pop_size)
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points) * 0.1
                # Update pbest
                overlaps = correlate(swarm[idx], 1 - swarm[idx], mode="full")
                fitness = np.max(overlaps) * (2.0 / n_points)
                pbest_fitness[idx] = fitness
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best = swarm[idx].copy()
            stagnation = 0
            
        iter_count += 1
        # Decay inertia weight
        w = max(0.3, 0.7 - 0.4 * iter_count / max_iter)
    
    return (global_best.tolist(), global_best_fitness, len(global_best))
