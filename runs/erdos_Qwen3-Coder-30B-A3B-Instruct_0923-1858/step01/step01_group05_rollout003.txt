import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # interpolate to 140 points
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_orig, h)
        n_points = 140
    else:
        # Default construction
        n_points = 140
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-10:mid+10] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping with 200 iterations
    sigma = 0.05
    for iter_count in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the current solution
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local optimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        # Adaptively decrease sigma
        if iter_count % 50 == 0 and iter_count > 0:
            sigma = sigma / 2.0
    
    # Final validation with tighter tolerances
    h_final = best_h.copy()
    for _ in range(5):
        try:
            res = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
        except:
            break
    
    final_c5 = evaluate_erdos_solution(h_final)
    return (h_final.tolist(), final_c5, n_points)
