import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    """Evaluates the Erdős constant C5 for a given step function h."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_feasible(h, n_points):
    """Projects h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with better starting point if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h * (n_points / 2) / np.sum(h), n_points)
    else:
        # Construct a reasonable initial guess
        h = np.ones(n_points) * 0.5
        # Ensure boundary conditions
        h[0] = 0.0
        h[-1] = 0.0
        h = project_feasible(h, n_points)

    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia_weight = 0.7
    c1 = c2 = 1.5
    perturbation_std = 0.01
    alpha = 0.9  # low-pass filter alpha
    
    # Initialize swarm
    particles = np.array([h.copy() for _ in range(swarm_size)])
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    # Evaluate initial positions
    fitness = np.array([evaluate_erdos_solution(p) for p in particles])
    
    # Personal bests
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    
    # Global best
    best_idx = np.argmin(fitness)
    gbest = particles[best_idx].copy()
    gbest_fitness = fitness[best_idx]
    
    # For low-pass filtering
    prev_velocities = np.zeros_like(velocities)
    
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Apply low-pass filter to velocity
            velocities[i] = alpha * velocities[i] + (1 - alpha) * prev_velocities[i]
            prev_velocities[i] = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_weight * velocities[i] 
                           + c1 * r1 * (pbest[i] - particles[i]) 
                           + c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_feasible(particles[i], n_points)
            
            # Evaluate fitness
            new_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best if needed
                if new_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = new_fitness
        
        # Perturb all particles slightly to avoid local optima
        for i in range(swarm_size):
            noise = np.random.normal(0, perturbation_std, n_points)
            particles[i] += noise
            particles[i] = project_feasible(particles[i], n_points)
        
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(gbest)
    
    # Return final solution
    return (gbest, final_c5, n_points)
