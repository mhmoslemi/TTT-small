import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Initialize h with uniform density
        n_points = 140
        h = np.ones(n_points) * 0.5
        # Scale to sum 70
        h = h * 70 / np.sum(h)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define objective function 
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    # Define constraint function
    def constraint_sum(h):
        return np.sum(h) - 70.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Best solution found so far
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation parameters
    pert_std = 0.05
    pert_decay = 0.5
    pert_decay_interval = 50

    # Global exploration with perturbations
    for iter_count in range(1000):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb current solution
        h_trial = h + np.random.normal(0, pert_std, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        # Maintain sum = 70
        h_trial = h_trial * 70 / np.sum(h_trial)
        
        c5_trial = objective(h_trial)
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
            
        # Local refinement of top perturbations
        if iter_count % 10 == 0:
            # Evaluate a few candidates and refine
            candidates = []
            for _ in range(10):
                h_cand = h + np.random.normal(0, pert_std, n_points)
                h_cand = np.clip(h_cand, 0, 1)
                h_cand = h_cand * 70 / np.sum(h_cand)
                candidates.append((h_cand, objective(h_cand)))
                
            # Find best candidate
            candidates.sort(key=lambda x: x[1])
            for h_cand, c5_cand in candidates[:3]:  # Top 3
                try:
                    res = minimize(objective, h_cand, method='SLSQP', bounds=bounds, constraints=cons, 
                                  options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        h_refined = np.clip(h_refined, 0, 1)
                        h_refined = h_refined * 70 / np.sum(h_refined)
                        c5_new = objective(h_refined)
                        if c5_new < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_new
                            h = h_refined.copy()
                except:
                    pass

        # Adapt perturbation strength
        if iter_count > 0 and iter_count % pert_decay_interval == 0:
            pert_std *= pert_decay
                
    # Final validation 
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
