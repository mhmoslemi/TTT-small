import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            original_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, 140)
            h = np.interp(new_x, original_x, h)
            n_points = 140
    else:
        # Create a feasible h vector with 140 points
        n_points = 140
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Ensure h sums to 70
    h = h / np.sum(h) * 70.0
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - 70.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    def objective(h):
        return evaluate_erdos_solution(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 1000
    iter_count = 0
    perturb_sigma = 0.05
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Global exploration: perturb
        h_trial = h + np.random.normal(0, perturb_sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * 70.0
        
        c5_trial = evaluate_erdos_solution(h_trial)
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
        
        # Local refinement on top 10 solutions
        top_candidates = []
        for _ in range(10):
            h_cand = h + np.random.normal(0, perturb_sigma, n_points)
            h_cand = np.clip(h_cand, 0, 1)
            h_cand = h_cand / np.sum(h_cand) * 70.0
            c5_cand = evaluate_erdos_solution(h_cand)
            top_candidates.append((c5_cand, h_cand))
        
        top_candidates.sort(key=lambda x: x[0])
        for c5_cand, h_cand in top_candidates[:5]:  # Refine top 5
            try:
                # SLSQP optimization
                res = minimize(objective, h_cand, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_new = res.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new / np.sum(h_new) * 70.0
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass

        iter_count += 1
        # Adjust perturbation strength
        if iter_count % 50 == 0 and perturb_sigma > 1e-5:
            perturb_sigma /= 2
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
