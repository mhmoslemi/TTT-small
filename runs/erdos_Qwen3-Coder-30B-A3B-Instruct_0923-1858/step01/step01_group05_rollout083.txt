import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

    best_h = h.copy()

    pop_size = 50

    mutation_rate = 0.1
    elite_size = 5

            perturbed = h + np.random.normal(0, 0.02, n_points)

            individual = np.random.rand(n_points)

            population.append(individual)
    
    # GA loop
    count_no_improvement = 0
