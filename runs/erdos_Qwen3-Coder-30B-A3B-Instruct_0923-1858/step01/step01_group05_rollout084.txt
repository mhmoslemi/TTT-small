import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

    pop_size = 50
    generations = 1000
    crossover_rate = 0.5
    mutation_rate = 0.3
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i < pop_size // 2 and 'initial_h_values' in globals():
            # Perturb initial solution
            h_perturbed = h_base + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        else:
            # Random initialization
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2.0)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    no_improvement_count = 0
    max_no_improvement = 150
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness_scores.append(c5)
            
        # Track best solution
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_c5:
            best_c5 = fitness_scores[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Selection with tournament
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            best_in_tournament = tournament[np.argmin([fitness_scores[i] for i in tournament])]
            selected.append(population[best_in_tournament].copy())
            
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Crossover
            if np.random.rand() < crossover_rate and len(new_population) + 1 < pop_size:
                parent1_idx, parent2_idx = np.random.choice(pop_size, 2, replace=False)
                parent1, parent2 = selected[parent1_idx], selected[parent2_idx]
                
                # Uniform crossover with some block swapping
                child1, child2 = parent1.copy(), parent2.copy()
                mask = np.random.rand(n_points) < 0.5
                
                child1[mask] = parent2[mask]
                child2[mask] = parent1[mask]
                
                # Ensure constraint satisfaction
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2.0) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2.0) / np.sum(child2)
                
                new_population.append(child1)
                new_population.append(child2)
            else:
                # Mutation
                parent_idx = np.random.choice(pop_size)
                mutant = selected[parent_idx].copy()
                
                # Mutate 30% of the elements
                mutate_indices = np.random.choice(n_points, int(0.3 * n_points), replace=False)
                for i in mutate_indices:
                    mutant[i] += np.random.normal(0, 0.03)
                    
                # Ensure constraint satisfaction
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (n_points / 2.0) / np.sum(mutant)
                
                new_population.append(mutant)
                
        # Trim to population size
        population = new_population[:pop_size]
        
    if best_h is None:
        # Fallback to initial solution if no improvement
        best_h = h_base
        best_c5 = evaluate_erdos_solution(best_h)
        
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
