import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """Evaluates the Erdős constant C5 for a given step function h."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

    else:

        h[-1] = 0.0

    h = h / np.sum(h) * (n_points / 2)

    # Precompute correlation
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        return np.max(overlaps)

    # Gradient-based optimization with projected steps
    best_c5 = float('inf')
    best_h = h.copy()
    alpha = 0.1
    stagnation_count = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        if stagnation_count > 200:
            alpha = 0.1
            
        # Compute gradient approximation using finite differences
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute finite difference gradient
        eps = 1e-5
        grad = np.zeros(n_points)

            grad[i] = (c5_plus - current_c5) / eps
            
        # Take gradient step
        h_new = h - alpha * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)

        # Update current solution
        h = h_new
        
        # Adaptive decrease of learning rate
        alpha *= 0.995
        
        # Perturbation to escape local minima
        if stagnation_count > 100 and np.random.rand() < 0.1:
            h += np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
