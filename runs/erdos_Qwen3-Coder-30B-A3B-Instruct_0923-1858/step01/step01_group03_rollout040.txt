import time

        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_C5 = float('inf')
    
    # Particle Swarm Optimization parameters
    Pop_size = 100
    max_iter = 10000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    # Add initial solution
    swarm.append(h.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    pbest.append(h.copy())
    pbest_fitness.append(float('inf'))
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_rand.copy())
        pbest_fitness.append(float('inf'))
    
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Helper function to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Helper function to project feasible
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Main PSO loop
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 50
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Update particles
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) +
                           c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate fitness
            fitness = compute_C5(swarm[i])
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
            
            # Update global best
            if fitness < gbest_fitness:
                gbest_idx = i
                gbest_fitness = fitness
                stagnation_count = 0
                
        stagnation_count += 1
        
        # Check for stagnation
        if stagnation_count > max_stagnation:
            # Reset some particles
            for i in range(int(Pop_size * 0.1)):
                idx = np.random.randint(0, Pop_size)
                h_rand = np.random.rand(n_points)
                h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                swarm[idx] = h_rand
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[idx] = h_rand.copy()
                pbest_fitness[idx] = float('inf')
            stagnation_count = 0
            
        iter_count += 1
        
        # Update inertia weight
        w = 0.7 - (0.5 * iter_count / max_iter)
    
    # Return best solution
    final_h = swarm[gbest_idx]
    final_C5 = gbest_fitness
    
    return (final_h.tolist(), final_C5, len(final_h))
