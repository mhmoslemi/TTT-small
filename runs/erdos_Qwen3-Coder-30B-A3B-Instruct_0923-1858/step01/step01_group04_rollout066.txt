import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = np.clip(h, 0, 1)

    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)

    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Coarse optimization with multiple random starts
    max_iter_coarse = 30
    for i in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 2:
            break

        perturbation = np.random.normal(0, 0.1, n_points)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        try:
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
