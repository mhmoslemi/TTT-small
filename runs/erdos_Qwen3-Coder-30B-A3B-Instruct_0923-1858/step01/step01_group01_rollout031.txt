import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, dx, eps=1e-8):
    """Compute gradient of C5 w.r.t. h using central finite differences."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_feasible(h_plus, dx)
        h_minus = project_feasible(h_minus, dx)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad

def optimize_h(h, dx, n_points, budget_s):
    """Optimize h using L-BFGS-B with projections."""
    best_h = h.copy()
    best_C = evaluate(h, dx)
    start_time = time.time()
    
    # Objective function for scipy optimizer
    def obj_func(h_flat):
        h = h_flat.reshape(-1)
        h = project_feasible(h, dx)
        return evaluate(h, dx)
    
    # Gradient function for scipy optimizer
    def grad_func(h_flat):
        h = h_flat.reshape(-1)
        grad = compute_gradient(h, dx)
        return grad.flatten()
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with L-BFGS-B
    try:
        res = minimize(
            obj_func, 
            h.flatten(), 
            method='L-BFGS-B', 
            jac=grad_func,
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        if res.success:
            h_final = res.x.reshape(-1)
            h_final = project_feasible(h_final, dx)
            c5_final = evaluate(h_final, dx)
            if c5_final < best_C:
                best_C = c5_final
                best_h = h_final.copy()
    except Exception:
        pass
    
    # Fine-tune with local search
    for i in range(5000):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        grad = compute_gradient(h, dx)
        # Adaptive learning rate
        lr = 0.01 / (1 + i / 1000)
        h = h - lr * grad
        h = project_feasible(h, dx)
        
        # Periodic perturbations
        if i % 200 == 0:
            pert = 0.005 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 100)
            h = (h + pert).clip(0, 1)
            h = project_feasible(h, dx)
        
        # Check for improvement
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
    
    return best_h, best_C
