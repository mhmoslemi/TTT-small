import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Add some structure to start with
        h[:10] = 0.6
        h[10:20] = 0.4
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Optimization settings
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use a cutting-plane approach with CVXPY for better results
    def solve_subproblem(h_current):
        n = len(h_current)
        dx = 2.0 / n
        
        # Create CVXPY variables
        h_var = cp.Variable(n)
        C_var = cp.Variable(1)
        
        # Define constraints
        constraints = [
            cp.sum(h_var) == n / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Compute current overlaps
        current_overlaps = []
        for k in range(2 * n - 1):
            overlap = 0.0
            for i in range(n):
                j = (i + k) % n
                overlap += h_current[i] * (1 - h_current[j])
            current_overlaps.append(overlap * dx)
        
        # Add cutting planes
        for k in range(2 * n - 1):
            # Compute gradient for overlap function O_k at h_current
            grad = np.zeros(n)
            for i in range(n):
                j = (i + k) % n
                grad[i] = dx * (1 - h_current[j])
            
            # Add constraint: grad^T * h <= current_overlap + grad^T * (h_current - h_current)
            constraints.append(cp.sum(grad * h_var) <= current_overlaps[k])
        
        # Objective: minimize C
        objective = cp.Minimize(C_var)
        
        # Formulate and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_var.value, C_var.value
        except:
            pass
            
        return h_current, current_overlaps[np.argmax(current_overlaps)]
    
    # Run optimization
    iter_counter = 0
    while time.time() - start_time < budget_s - 1:
        iter_counter += 1
        try:
            h_new, c5_new = solve_subproblem(best_h)
            if isinstance(h_new, np.ndarray) and not np.isnan(h_new).any():
                h_new = np.clip(h_new, 0, 1)
                current_c5 = evaluate_erdos_solution(h_new)
                if current_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = current_c5
        except Exception as e:
            pass
            
        if time.time() - start_time > budget_s - 0.5:
            break
    
    # Final evaluation and trimming
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
