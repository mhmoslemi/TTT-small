import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def optimize_h(h, dx, n_points, max_time):
    """Optimize h using gradient-based method."""
    best_h = h.copy()
    best_C = evaluate(h, dx)
    lr = 0.05
    
    t0 = time.time()
    
    for i in range(50000):
        if time.time() - t0 > max_time:
            break
            
        if i % 200 == 0:
            lr *= 0.97
            
        grad = np.zeros_like(h)
        epsilon = 1e-5
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)

        h = h - lr * grad
        h = project(h, dx)
        
        if i % 50 == 0:
            current_C = evaluate(h, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h.copy()
                
            # Perturb occasionally to escape local minima
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, 0.01, n_points)
                h = project(h + noise, dx)
                
    return best_h, best_C
