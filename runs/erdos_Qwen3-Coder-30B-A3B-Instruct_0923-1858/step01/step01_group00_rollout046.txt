import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

    dx = 2.0 / n_points

        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    
    # Main optimization loop
    while time.time() - start_time < budget_s:
        # Use Sequential Least Squares Programming
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Use multiple random initializations to escape local optima
            for _ in range(5):
                # Random perturbation around current best
                h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
                
                result = minimize(
                    objective,
                    h_perturbed,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 100},
                    tol=1e-6
                )
                
                if result.success:
                    current_c5 = evaluate_erdos_solution(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()
                
                # Prevent timeout by checking periodically
                if time.time() - start_time > budget_s * 0.95:
                    break
        
        except Exception:
            pass
        
        # Early termination if we're close to target
        if best_c5 <= 0.3807:
            break
            
        # Perturb the best to escape local minima
        if time.time() - start_time < budget_s * 0.9:
            h = best_h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)

    try:
        actual_c5 = evaluate_erdos_solution(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)

    return (best_h, actual_c5, n_points)
