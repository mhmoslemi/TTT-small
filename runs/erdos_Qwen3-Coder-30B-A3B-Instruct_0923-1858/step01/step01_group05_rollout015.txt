import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a feasible random h vector
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        t = time.time()
        # Local optimization with random perturbation
        h_trial = h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Try different local optimizers
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
                    last_improvement = iter_count
        except:
            pass
            
        iter_count += 1
        
        # Occasionally perturb to escape local minima
        if iter_count % 50 == 0 and iter_count - last_improvement > 300:
            # Create a new random interpolated function with different properties
            # Add stronger noise to escape local minima
            h = h + np.random.normal(0, 0.03, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
            last_improvement = iter_count
            
        # Occasionally try amplitude adjustments in important regions
        if iter_count % 25 == 0 and iter_count - last_improvement > 100:
            # Focus on adjusting the center regions where overlaps happen
            center = n_points // 2
            region = max(5, n_points // 10)
            si = max(0, center - region)
            ei = min(n_points, center + region)
            # Random small changes in center region to encourage variation
            h[si:ei] += np.random.normal(0, 0.005, ei-si)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
