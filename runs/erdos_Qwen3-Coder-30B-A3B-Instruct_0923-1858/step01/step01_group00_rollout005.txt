import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    # Precompute correlation matrix
    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return corr * dx
    
    # Gradient-based optimization with projection
    def project(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Initialize gradient descent
    alpha = 0.1
    patience = 0
    last_c5 = float('inf')
    step = 0
    
    while time.time() - start_time < budget_s:
        # Compute current correlation and shift index of max
        corr = compute_correlation(h)
        max_k_idx = np.argmax(corr)
        max_val = corr[max_k_idx]
        
        # Compute gradient - Approximate from correlation derivative
        # We use the fact that the gradient w.r.t. h_i corresponds to the derivative of max_k
        # For the case of a single maximal lag k, the contribution to the gradient at position i
        # is -2* (1 - h[i - k]) if that term contributes to max_k
        # But we compute this more reliably using finite differences
        
        # As a proxy, compute gradient using finite differences
        # Create a finite difference approximation of gradient
        grad = np.zeros_like(h)
        finite_diff_step = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += finite_diff_step
            h_plus = project(h_plus)
            corr_plus = compute_correlation(h_plus)
            grad[i] = (corr_plus[max_k_idx] - max_val) / finite_diff_step
        
        # Update h
        h = h - alpha * grad
        h = project(h)
        
        # Evaluate new C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Adaptive learning rate
        if patience >= 100:
            alpha *= 0.95
            patience = 0
            
        # Perturb periodically
        if patience >= 100 and np.random.rand() < 0.1:
            # Add noise and smooth
            noise1 = np.random.uniform(-0.1, 0.1, n_points)
            noise2 = np.random.normal(0, 0.01, n_points)
            h += noise1 + noise2
            # Apply median filter
            h = np.convolve(h, np.ones(3)/3, mode='same')
            h = project(h)
        
        # Occasionally reset learning rate
        if patience >= 200:
            alpha = 0.1
        
        step += 1
        
    # Final correction
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except:
        final_c5 = best_c5
        
    return (best_h, final_c5, n_points)
