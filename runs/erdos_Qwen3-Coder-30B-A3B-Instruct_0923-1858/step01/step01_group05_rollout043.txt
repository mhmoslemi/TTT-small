import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def perturb_and_repair(h, sigma=0.03):
    n = len(h)
    # Perturb in the middle region where overlaps are likely to be large
    mask = np.linspace(0, 2, n) >= 0.3
    mask &= np.linspace(0, 2, n) <= 0.7
    h_pert = h.copy()
    h_pert[mask] += np.random.normal(0, sigma, np.sum(mask))
    h_pert = np.clip(h_pert, 0, 1)
    h_pert = h_pert * (35.0 / np.sum(h_pert))
    return h_pert

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create triangular function as a better starting point
        x = np.linspace(0, 2, 70)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * 35.0
        n_points = 70
    
    # Ensure initial is feasible
    h = np.clip(h, 0, 1)
    h = h * (35.0 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    for i in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local perturbation
        h_trial = perturb_and_repair(h)
        
        # SLSQP optimization on this point
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_var):
            corr = np.correlate(h_var, 1-h_var, mode="full")
            return np.max(corr) * (2.0 / n_points)
            
        def constraint_sum(h_var):
            return np.sum(h_var) - 35.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', 
                          bounds=bounds, constraints=cons, options={'maxiter': 50})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        # Occasionally do a broader perturbation
        if i % 50 == 0:
            h = perturb_and_repair(h, sigma=0.05)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
