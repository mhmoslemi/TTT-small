import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    # Start with rough resolution
    coarse_n = 50
    fine_n = 100
    
    # If we have spare time, let's try refining
    if n_points <= 50:
        resolution = [coarse_n]
        if budget_s > 100:
            resolution = [coarse_n, fine_n]
    else:
        resolution = [n_points]  # Already have a good resolution
    
    # Phase 1: coarse optimization
    for res in resolution:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Adjust resolution
        if res != n_points:
            # Upsample or downsample
            if res > n_points:
                # Upsample using linear interpolation
                h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h)
            else:
                # Downsample using averaging
                h_new = np.array([np.mean(h[i:i + n_points//res]) for i in range(0, n_points, n_points//res)])
                # Extend to res points by padding
                if len(h_new) < res:
                    h_new = np.pad(h_new, (0, res - len(h_new)), constant_values=h_new[-1])
                elif len(h_new) > res:
                    h_new = h_new[:res]
            h = h_new.copy()
            n_points = res
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
        # Mutation-based local search
        num_trials = 50
        for i in range(num_trials):
            if time.time() - start_time > budget_s - 2:
                break
                
            h_trial = h.copy()
            
            # Add small random perturbation
            perturb = np.random.normal(0, 0.02, n_points)
            h_trial += perturb
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # Local refinement with SLSQP
            def objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                return evaluate_erdos_solution(h_vec)
                
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_integral}
                
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except:
                pass
                
            # Perturbation-based greedy search for local improvements
            # Identify dominant shifts (high overlap)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr_idx = np.argmax(corr)
            if max_corr_idx >= len(corr) // 2:
                shift = max_corr_idx - len(corr) // 2
            else:
                shift = max_corr_idx - len(corr) // 2
                
            # Slowly adjust h along the worst shift
            h_new = h.copy()
            for idx in range(n_points):
                # Adjust h[idx] to reduce overlap at shift k
                # If h[idx] * (1 - h[idx + shift]) is significant, adjust
                if max_corr_idx < len(corr) // 2:
                    jdx = idx + shift
                else:
                    jdx = idx - shift
                    
                if 0 <= jdx < n_points:
                    # If this configuration contributes significantly to overlap
                    contribution = h[idx] * (1 - h[jdx])
                    if contribution > 0.05:
                        step = 0.005
                        # Gradient-like update (heuristic - adjust towards reducing overlap)
                        h_new[idx] = np.clip(h_new[idx] - step * h[jdx], 0, 1)
                        h_new[idx] = np.clip(h_new[idx] + step * h[jdx], 0, 1)
            
            # Renormalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h = h_new
            
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
