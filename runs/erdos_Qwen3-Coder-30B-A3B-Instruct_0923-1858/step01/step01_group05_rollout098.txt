import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create initial h as triangular function
        n_points = 70
        h = np.array([1.0 - abs(2.0 * i / (n_points - 1) - 1.0) for i in range(n_points)])
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 10000
    iter_count = 0
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Global exploration: perturb in regions where high overlaps are likely
        h_trial = h.copy()
        # Only perturb in region [0.3, 0.7] which corresponds to indices [0.3*70, 0.7*70] = [21, 49]
        perturbation_indices = np.arange(21, 49)
        noise = np.random.normal(0, 0.03, len(perturbation_indices))
        h_trial[perturbation_indices] += noise
        # Repair feasibility
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Evaluate the new trial
        c5_trial = evaluate_erdos_solution(h_trial)
        
        # Accept if better
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
        
        # Local refinement with SLSQP if we have some improvement
        if iter_count % 5 == 0:
            try:
                # Define bounds and constraints for SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                # Local optimization
                res = minimize(
                    evaluate_erdos_solution, 
                    h_trial, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons, 
                    options={'maxiter': 30, 'ftol': 1e-6}
                )
                if res.success:
                    h_refined = res.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined / np.sum(h_refined) * (n_points / 2.0)
                    c5_refined = evaluate_erdos_solution(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
            except:
                pass
                
        iter_count += 1
        
        # Occasionally restart with current best to avoid stagnation
        if iter_count % 100 == 0:
            h = best_h.copy()
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
