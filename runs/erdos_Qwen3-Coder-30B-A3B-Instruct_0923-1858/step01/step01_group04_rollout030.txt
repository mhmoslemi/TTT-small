import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    current_best_h = h.copy()
    current_best_c5 = evaluate_erdos_solution(current_best_h)
    
    # Try a sequence of optimizations at different resolutions
    resolutions = [50, 70, 100]

            old_x = np.linspace(0, 2, len(current_best_h))
            new_x = np.linspace(0, 2, n_points)
            h = np.interp(new_x, old_x, current_best_h)

        max_iter = 30
