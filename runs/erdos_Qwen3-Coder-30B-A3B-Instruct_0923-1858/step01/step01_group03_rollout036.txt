import time

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    perturb_rate = 0.05
    max_iter = 2000
    iter_count = 0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Generate candidate solutions
    def generate_candidate(current_h, perturb_rate):
        candidate = current_h.copy()
        num_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for idx in indices:
            # Small random perturbation
            delta = np.random.normal(0, 0.05)
            candidate[idx] += delta
        return project_feasible(candidate)
    
    # Main tabu search loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate new candidate
        candidate = generate_candidate(h, perturb_rate)
        
        # Evaluate
        candidate_C5 = compute_C5(candidate)
        
        # Check if it's better than current best
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate.copy()
        
        # Check tabu list
        candidate_tuple = tuple(candidate.round(6))
        if candidate_tuple not in tabu_list:
            # Accept this candidate
            h = candidate
            # Add to tabu list
            tabu_list.append(candidate_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        # Update perturbation rate
        if iter_count % 50 == 0 and iter_count > 0:
            perturb_rate = min(0.2, perturb_rate * 1.1)
        
        iter_count += 1

    if final_C5 < best_C5:
        best_C5 = final_C5
