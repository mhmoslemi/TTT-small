import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, dx):
    """Compute gradient of C5 via finite differences."""
    eps = 1e-8
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_feasible(h_plus, dx)
        h_minus = project_feasible(h_minus, dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad

def initialize_h(n_points):
    """Initialize h with a triadic waveform."""
    t = np.linspace(0, 1, n_points)
    h = np.abs(np.sin(2 * np.pi * t * (n_points / 3))) ** 2
    h = np.clip(2 * h - 0.5, 0, 1)
    return project_feasible(h, dx=2.0 / n_points)

def optimize(h, dx, n_points, budget_s=600, seed=42):
    """Optimize h using L-BFGS-B with structured perturbations."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    def objective(x):
        h_temp = project_feasible(x, dx)
        return evaluate(h_temp, dx)
    
    def gradient(x):
        h_temp = project_feasible(x, dx)
        return compute_gradient(h_temp, dx)
    
    # Use L-BFGS-B for bound-constrained optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, 
                      options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9})
    
    current_h = project_feasible(result.x, dx)
    current_C = evaluate(current_h, dx)
    if current_C < best_C:
        best_C = current_C
        best_h = current_h.copy()
    
    # Post-optimization with random structured perturbations
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Apply sine wave perturbation
        perturb = 0.01 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 100)
        h_perturbed = (current_h + perturb).clip(0, 1)
        h_perturbed = project_feasible(h_perturbed, dx)
        current_C_pert = evaluate(h_perturbed, dx)
        if current_C_pert < best_C:
            best_C = current_C_pert
            best_h = h_perturbed.copy()
        current_h = h_perturbed
    
    return best_h, best_C

def run(seed=42, budget_s=600, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with triadic waveform
    initial_h = initialize_h(n_points)
    
    # Optimize with L-BFGS-B
    best_h, best_C = optimize(initial_h, dx, n_points, budget_s=budget_s, seed=seed)
    
    return (best_h, best_C, n_points)
