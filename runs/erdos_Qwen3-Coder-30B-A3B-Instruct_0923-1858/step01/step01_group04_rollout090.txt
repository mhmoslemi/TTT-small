import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution optimization
    resolutions = [50, 70, 100]
    
    for resolution in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        if resolution > n_points:
            # Upsample
            h_upsampled = np.interp(np.linspace(0, 2, resolution), 
                                  np.linspace(0, 2, n_points), 
                                  h)
            h = h_upsampled
            n_points = resolution
        
        # Local optimization with SLSQP
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)
        
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        # Try several local optimizations from different random starts
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Random perturbation
            perturbation = np.random.normal(0, 0.1, n_points)
            h_trial = h + perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # SLSQP optimization from this starting point
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                        
            except Exception:
                continue
    
    # Final refinement steps
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Identify worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        # Get the shift k for maximum overlap
        k = max_idx - (len(best_h) - 1)
        
        # Create adjustment vector
        adjustment = np.zeros_like(best_h)
        
        # Make adjustments based on this shift's contribution
        for i in range(len(best_h)):
            j = i + k
            if 0 <= j < len(best_h):
                adjustment[i] -= best_h[j]  # Add to get negative effect
        
        # Apply adjustment to step function
        best_h += 0.01 * adjustment
        best_h = np.clip(best_h, 0, 1)
        # Normalize to maintain constraint
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        c5_candidate = evaluate_erdos_solution(best_h)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
    
    return (list(best_h), best_c5, n_points)
