import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    population_size = 100
    generations = 500
    tournament_size = 5
    mutation_rate = 0.2
    crossover_rate = 0.8
    
    # Use initial_h_values if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h0 = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize population
    population = []
    population.append(h0.copy())
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def tournament_selection(population, fitnesses, k):
        selected_indices = np.random.choice(len(population), k)
        selected_fitnesses = [fitnesses[i] for i in selected_indices]
        winner_idx = selected_indices[np.argmin(selected_fitnesses)]
        return population[winner_idx]
    
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h_vec):
        h_mut = h_vec.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                h_mut[i] = np.clip(h_mut[i] + np.random.normal(0, 0.05), 0, 1)
        return project_feasible(h_mut)
    
    best_h = h0.copy()
    best_C5 = compute_C5(h0)
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [compute_C5(h) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_C5:
            best_C5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection, crossover and mutation
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
