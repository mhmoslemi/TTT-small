import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Create a feasible h vector with 140 points
        n_points = 140
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Ensure h is in [0,1] and sum to 70 (since 2/140 * 70 = 1)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2.0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation loop
    max_iter = 1000
    iter_count = 0
    perturbation_strength = 0.05
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Multiple random perturbations
        num_perturb = 50
        candidates = []
        
        for _ in range(num_perturb):
            h_trial = h + np.random.normal(0, perturbation_strength, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            c5_trial = evaluate_erdos_solution(h_trial)
            candidates.append((c5_trial, h_trial.copy()))
            
        # Sort candidates and keep top 10
        candidates.sort(key=lambda x: x[0])
        top_candidates = candidates[:10]
        
        # Local refinement with SLSQP
        for c5_val, h_trial in top_candidates:
            if c5_val < best_c5:
                # Apply SLSQP with constraints for local refinement
                bounds = [(0, 1) for _ in range(n_points)]
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(
                        lambda x: evaluate_erdos_solution(x),
                        h_trial,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 50, 'ftol': 1e-6}
                    )
                    if res.success:
                        h_new = res.x
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                        c5_new = evaluate_erdos_solution(h_new)
                        if c5_new < best_c5:
                            best_h = h_new.copy()
                            best_c5 = c5_new
                            h = h_new.copy()
                except:
                    pass
                
                # If local optimization fails, just take candidate if better
                if c5_val < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_val
                    h = h_trial.copy()
        
        iter_count += 1
        # Decrease perturbation strength over time
        if iter_count % 50 == 0:
            perturbation_strength *= 0.5
            
        # Occasionally do large random perturbation
        if iter_count % 100 == 0:
            h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
