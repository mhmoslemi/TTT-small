import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project_feasible(h, dx):
    """Project to satisfy 0 <= h[i] <= 1 and sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, dx, eps=1e-8):
    """Compute gradient of C5 using central finite differences"""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_feasible(h_plus, dx)
        h_minus = project_feasible(h_minus, dx)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad

def initialize_h(n_points):
    """Initialize with a triadic waveform"""
    t = np.linspace(0, 1, n_points)
    h = np.abs(np.sin(2 * np.pi * t * (n_points / 3))) ** 2
    h = np.clip(2 * h - 0.5, 0, 1)
    return h

        h_test = project_feasible(x, dx)
        return evaluate(h_test, dx)

        h_test = project_feasible(x, dx)
        return compute_gradient(h_test, dx)
    
    # Optimize using scipy minimize
    result = minimize(
        objective,
        h,

    iteration = 0

            current_C = evaluate(h_pert, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h_pert.copy()
        # Structured perturbations
        elif iteration % 50 == 0:
            perturb = 0.005 * np.sin(2 * np.pi * np.arange(n_points) / n_points * 200)
            h_pert = best_h + perturb
            h_pert = project_feasible(h_pert, dx)
            current_C = evaluate(h_pert, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h_pert.copy()

    h_refine = best_h.copy()

        current_C = evaluate(h_refine, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_refine.copy()
        lr *= 0.99
