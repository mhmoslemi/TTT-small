from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def tournament_select(population, fitness, k):
    selected_indices = np.random.choice(len(population), size=k, replace=False)
    best_index = selected_indices[np.argmin([fitness[i] for i in selected_indices])]
    return population[best_index].copy()

def crossover(p1, p2):
    # Blend crossover
    alpha = np.random.rand()
    child1 = alpha * p1 + (1 - alpha) * p2
    child2 = (1 - alpha) * p1 + alpha * p2
    
    # Ensure constraints
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    
    # Rescale to sum requirement
    n = len(p1)
    child1 = child1 / np.sum(child1) * (n / 2.0)
    child2 = child2 / np.sum(child2) * (n / 2.0)
    
    return child1, child2

def mutate(individual, std):
    # Apply Gaussian noise
    mutation = np.random.normal(0, std, len(individual))
    mutated = individual + mutation
    
    # Clip values
    mutated = np.clip(mutated, 0, 1)
    
    # Rescale
    n = len(individual)
    mutated = mutated / np.sum(mutated) * (n / 2.0)
    
    return mutated
