import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution differential evolution with local refinement
    resolutions = [50, 100]
    popsize = 40
    mutation = 0.8
    recombination = 0.9
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Upsample if needed
        if res > n_points:
            h_upsampled = np.interp(np.linspace(0, 2, res), 
                                  np.linspace(0, 2, n_points), h)
            h_upsampled = h_upsampled * (res / 2) / np.sum(h_upsampled)
            h_upsampled = np.clip(h_upsampled, 0, 1)
        else:
            h_upsampled = h.copy()
            res = n_points
            
        # Define constraints
        def constraint_integral(h_vec):
            return np.sum(h_vec) - res / 2
        
        bounds = [(0, 1) for _ in range(res)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        # Differential evolution for global optimization
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)
            
        # Run DE
        de_result = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            maxiter=200,
            popsize=popsize,
            mutation=mutation,
            recombination=recombination,
            seed=seed,
            callback=None,
            tol=1e-6,
            init='latinhypercube'
        )
        
        if de_result.success:
            h_candidate = np.clip(de_result.x, 0, 1)
            h_candidate = h_candidate * (res / 2) / np.sum(h_candidate)
            c5_candidate = evaluate_erdos_solution(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
            
        # Local refinement with SLSQP from best DE candidate
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            res_ref = minimize(
                objective,
                h_candidate,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res_ref.success:
                h_local = np.clip(res_ref.x, 0, 1)
                h_local = h_local * (res / 2) / np.sum(h_local)
                c5_local = evaluate_erdos_solution(h_local)
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
        except Exception:
            pass

    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    n_points_final = len(best_h)
    
    return (list(best_h), final_c5, n_points_final)
