import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_feasible(h, n_points):
    """Project h onto feasible region: [0,1] and sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    inertia_w = 0.7
    c1 = c2 = 1.5
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    global_best = h.copy()
    global_best_val = evaluate_erdos_solution(global_best)
    
    for _ in range(swarm_size):
        # Start from a slightly perturbed version of the previous best or a random feasible solution
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.02, n_points)
        else:
            particle = np.random.rand(n_points) * 0.5
        particle = project_feasible(particle, n_points)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        val = evaluate_erdos_solution(particle)
        personal_best.append(particle.copy())
        personal_best_vals.append(val)
        if val < global_best_val:
            global_best = particle.copy()
            global_best_val = val
    
    # Low-pass filter parameters
    alpha = 0.9
    filtered_velocities = [np.zeros_like(v) for v in velocities]
    
    # Start timer
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    try:
        while time.time() - start_time < budget_s and iteration < max_iter:
            for i in range(swarm_size):
                # Update velocity with low-pass filtered version
                vel = velocities[i]
                filtered_vel = alpha * filtered_velocities[i] + (1 - alpha) * vel
                
                # PSO update
                r1, r2 = np.random.rand(), np.random.rand()
                cognitive = c1 * r1 * (personal_best[i] - particles[i])
                social = c2 * r2 * (global_best - particles[i])
                new_vel = inertia_w * filtered_vel + cognitive + social
                
                # Update position
                new_pos = particles[i] + new_vel
                
                # Apply feasibility projection
                new_pos = project_feasible(new_pos, n_points)
                
                # Evaluate new solution
                new_val = evaluate_erdos_solution(new_pos)
                
                # Update personal best
                if new_val < personal_best_vals[i]:
                    personal_best[i] = new_pos.copy()
                    personal_best_vals[i] = new_val
                    if new_val < global_best_val:
                        global_best = new_pos.copy()
                        global_best_val = new_val
                        last_improvement = iteration
                
                # Update particles
                particles[i] = new_pos
                velocities[i] = new_vel
                filtered_velocities[i] = filtered_vel

            if iteration % 50 == 0:

                    perturbation = np.random.normal(0, 0.01, n_points)
                    particles[i] += perturbation
                    particles[i] = project_feasible(particles[i], n_points)
            
            iteration += 1
            
            # Stop if no improvement for too long
            if iteration - last_improvement > 200:
                break
                
    except Exception:
        pass
    
    return (global_best, global_best_val, n_points)
