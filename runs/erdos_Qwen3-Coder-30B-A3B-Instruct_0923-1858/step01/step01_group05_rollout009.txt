import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = len(h0)
    else:
        n_points = 140
        h0 = np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2.0)

    # Compute bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Objective function
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Genetic Algorithm Implementation
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    max_generations = 300
    patience = 150
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i < pop_size // 2 and 'initial_h_values' in globals():
            h = h0 + np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2.0)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    patience_count = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = objective(h)
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Sorting
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Early stopping
        if gen > 0 and fitness[0] >= fitness[1]:
            patience_count += 1
        else:
            patience_count = 0
            
        if patience_count >= patience:
            break
            
        # Selection (top 20%)
        parents = population[:pop_size//5]
        
        # Crossover and mutation
        new_population = parents.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(len(parents), 2, replace=False)
            p1, p2 = parents[idx1], parents[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.3:
                        child[i] += np.random.normal(0, 0.03)
                        
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2.0)
            
            new_population.append(child)
            
        population = new_population[:pop_size]

    if best_h is None:
        # Fallback
        best_h = h0
        best_c5 = objective(best_h)
        
    return (best_h.tolist(), best_c5, n_points)
