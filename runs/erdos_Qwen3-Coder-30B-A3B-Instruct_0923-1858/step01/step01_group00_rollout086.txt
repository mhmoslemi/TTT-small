import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to the target resolution
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    sigma = 0.1
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    no_improvement_count = 0
    max_no_improvement = 100
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Generate a random perturbation
        delta_h = np.random.uniform(-T * sigma, T * sigma, n_points)
        
        # Propose new solution
        h_new = h + delta_h
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Accept or reject the move
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
            h = h_new.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Cool temperature
        T = alpha * T
        
        # Shake if no improvement for a while
        if no_improvement_count >= max_no_improvement:
            h = h + np.random.normal(0, T * sigma, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            no_improvement_count = 0
            
    return (best_h, best_c5, n_points)
