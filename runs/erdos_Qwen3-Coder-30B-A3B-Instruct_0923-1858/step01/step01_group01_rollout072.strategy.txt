def project_feasible(h, dx, tolerance=1e-6):
    # Clip to [0, 1] and enforce integral constraint
    h = np.clip(h, 0, 1)

#### 3. Inner Problem (Overlap Maximization)
```python
def compute_overlap_max(h, dx, m_samples=1000):
    # Discretize shifts k
    shift_candidates = np.linspace(0, 2, m_samples)
    
    # Compute overlaps for each k
    max_overlap = 0
    best_k = 0
    for k in shift_candidates:
        overlap = 0
        for i in range(len(h)):
            j = (i + int(k / dx)) % len(h)
            overlap += h[i] * (1 - h[j])
        overlap *= dx
        if overlap > max_overlap:
            max_overlap = overlap
            best_k = k
    return max_overlap, best_k
```
