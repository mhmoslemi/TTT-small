import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define the objective function
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    # Track the best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Genetic Algorithm parameters
    pop_size = 50

    mutation_rate = 0.3
    tournament_size = 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        # Seed with initial values
        population.append(h.copy())
        for i in range(pop_size-1):
            noise = np.random.normal(0, 0.02, n_points)
            h_trial = h + noise
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            population.append(h_trial.copy())
    else:
        # Create random initial population
        for _ in range(pop_size):
            h_trial = np.random.rand(n_points)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            h_trial = np.clip(h_trial, 0, 1)
            population.append(h_trial.copy())

        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        if fitness[0] < best_c5:

        population = new_population[:pop_size]
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
