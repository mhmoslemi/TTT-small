import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, dx, eps=1e-8):
    """Compute gradient using central finite differences."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_feasible(h_plus, dx)
        h_minus = project_feasible(h_minus, dx)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad

def initialize_h(n_points):
    """Create a structured initial function that's aperiodic."""
    t = np.linspace(0, 1, n_points)
    # Use a triadic-like waveform
    h = np.sin(2 * np.pi * t * (n_points / 3)) ** 2
    h = 2 * h - 0.5
    h = np.clip(h, 0, 1)
    return project_feasible(h, dx=2.0 / n_points)

    # Set bounds for L-BFGS-B

    # Optimization loop

        h_new = result.x.reshape(-1)
        h_new = project_feasible(h_new, dx)
        
        # Soft update
        h = h_new
        
        # Update best solution if improved
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
            
        iteration += 1
    
    return best_h, best_C
