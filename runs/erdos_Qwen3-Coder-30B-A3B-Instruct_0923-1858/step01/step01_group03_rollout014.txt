        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum over all shifts
        return np.max(corr) * dx

        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

            candidate = current_h.copy()
            
            # Select random indices to perturb
            num_perturb = np.random.randint(1, max_perturb + 1)

                candidate[idx] = np.clip(candidate[idx] + np.random.normal(0, 0.03), 0, 1)

            candidate = project_feasible(candidate)
            neighbors.append(candidate.copy())
            
        return neighbors
