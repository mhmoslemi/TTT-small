import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to required n_points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        # Ensure bounds and normalization
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated annealing
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling factor
    sigma = 0.1  # Perturbation scale
    stagnation_count = 0
    max_stagnation = 100

    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Generate perturbation
        delta = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_candidate = h + delta
        
        # Ensure h is in [0,1] and sum equals n_points/2
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5 or np.random.rand() < np.exp(-(c5_candidate - best_c5) / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Apply shaking if stagnation
        if stagnation_count >= max_stagnation:
            h += np.random.normal(0, T * sigma, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            stagnation_count = 0
        
        # Cool temperature
        T = alpha * T

    # Final evaluation to make sure the solution is valid
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
