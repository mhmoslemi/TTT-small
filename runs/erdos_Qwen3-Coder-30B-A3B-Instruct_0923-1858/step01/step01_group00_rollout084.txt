import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    
    h = h / np.sum(h) * (n_points / 2)
    
    # Precompute the correlation
    def compute_corr(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return corr * dx
    
    def gradient(h):
        corr = compute_corr(h)
        max_lag = np.argmax(corr)
        # 
        grad = np.zeros_like(h)
        # Compute gradient contribution
        for i in range(len(h)):
            for k in range(-len(h)+1, len(h)):
                if max_lag == k + len(h) - 1:
                    # When k is such that correlation is maximized
                    if i + k >= 0 and i + k < len(h):
                        grad[i] += 1 - 2 * h[i + k]
        return grad * dx
    
    # Gradient descent parameters
    alpha = 0.1
    patience = 0
    max_patience = 200
    last_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        try:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Gradient step
            grad = gradient(h)
            h = h - alpha * grad
            
            # Project to feasible set
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            # Adaptive learning rate
            if c5 > last_c5:
                alpha *= 0.95
            else:
                alpha = min(alpha * 1.05, 0.5)
            last_c5 = c5
            
            # Perturbation
            patience += 1
            if patience > 100:
                noise = np.random.uniform(low=-0.1, high=0.1, size=n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
                h = signal.medfilt(h, 3)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
                patience = 0
            
            # Reset learning rate if too slow
            if patience > max_patience:
                alpha = 0.1
                patience = 0
                
        except Exception as e:
            # Reset if any error occurs
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            else:
                h = np.ones(n_points) * 0.5
                h[0] = 0.0
                h[-1] = 0.0
            h = h / np.sum(h) * (n_points / 2)
            alpha = 0.1
    
    if best_h is None:
        best_h = h.copy()
    
    actual_c5 = evaluate_erdos_solution(best_h)
    return (best_h, actual_c5, n_points)
