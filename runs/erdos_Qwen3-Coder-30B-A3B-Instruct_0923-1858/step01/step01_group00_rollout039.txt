import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use n_points = 150 for more precision and control
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():

        # Interpolate to the target n_points
        h_original = h.copy()
        x_original = np.linspace(0, 2, len(h_original))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_original, h_original)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to sum(h) = n_points / 2
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling factor
    sigma = 0.1  # Scaling for perturbation
    shake_freq = 100  # Shake every shake_freq iterations
    max_stagnation = 100  # Max iterations without improvement before shake
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    stagnation_count = 0
    start_time = time.time()

    # Annealing loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Perturbation
        delta = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_candidate = h + delta
        # Project to feasible region [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        # Rescale to sum(h_candidate) = n_points / 2
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Accept or reject
        delta_E = c5_candidate - best_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Shake occasionally when stagnating
        if stagnation_count >= max_stagnation:
            h += np.random.normal(0, T * sigma, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            stagnation_count = 0
            # Optionally lower temperature to reduce noise
            T = max(T * alpha, 1e-6)
            
        # Cool temperature
        T = T * alpha
        iteration += 1
        
        # Occasionally check time
        if iteration % 100 == 0 and time.time() - start_time > budget_s - 1:
            break
    
    # Final check of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
