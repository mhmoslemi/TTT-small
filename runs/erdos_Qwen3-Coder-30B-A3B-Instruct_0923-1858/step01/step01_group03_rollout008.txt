import time
from scipy.signal import correlate

        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Parameters for genetic algorithm
    population_size = 100
    num_generations = 500
    mutation_rate = 0.15
    tournament_size = 5
    crossover_rate = 0.8
    
    # Initialize population
    def create_individual():
        individual = np.random.rand(n_points)
        # Clip to [0, 1]
        individual = np.clip(individual, 0, 1)
        # Rescale to satisfy constraint
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (n_points / 2) / current_sum
        return individual
    
    def project_feasible(individual):
        # Clip between 0 and 1
        individual = np.clip(individual, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (n_points / 2) / current_sum
        return individual
    
    def compute_C5(h_vec):
        # Convert to correct size
        h_vec = np.array(h_vec)
        # Note: correlate with mode='full' to handle all shifts
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum over all shifts
        return np.max(corr) * (2.0 / n_points)
    
    def evaluate_fitness(individual):
        return compute_C5(individual)
    
    # Create initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    # Initial best solution
    best_individual = h.copy()
    best_fitness = compute_C5(best_individual)
    
    # Evolutionary process
    generations_without_improvement = 0
    max_generations_without_improvement = 20
    
    for gen in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Evaluate fitness for entire population
        fitness_scores = [evaluate_fitness(ind) for ind in population]
        
        # Track best in current generation
        current_best_idx = np.argmin(fitness_scores)
        current_best = population[current_best_idx]
        current_best_fitness = fitness_scores[current_best_idx]
        
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = current_best.copy()
            generations_without_improvement = 0
        else:
            generations_without_improvement += 1
            
        if generations_without_improvement > max_generations_without_improvement:
            # Reset population if not improving
            population = [best_individual.copy()]
            for _ in range(population_size - 1):
                population.append(create_individual())
            generations_without_improvement = 0
            
        # Tournament selection
        def tournament_select():
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            return population[winner_idx].copy()
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        # Generate rest through crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = tournament_select()
                child2 = tournament_select()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                child1[idx] = np.clip(child1[idx] + np.random.normal(0, 0.05), 0, 1)
            
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                child2[idx] = np.clip(child2[idx] + np.random.normal(0, 0.05), 0, 1)
                
            new_population.append(project_feasible(child1))
            if len(new_population) < population_size:
                new_population.append(project_feasible(child2))
        
        population = new_population
    
    # Final evaluation
    final_C5 = compute_C5(best_individual)
    if final_C5 < best_fitness:
        best_fitness = final_C5
    
    return (best_individual.tolist(), best_fitness, n_points)
