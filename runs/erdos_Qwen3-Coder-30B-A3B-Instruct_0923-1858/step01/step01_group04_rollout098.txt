import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    resolutions = [50, 100]  # Low to high resolution
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate initial population using differential evolution
        def objective_de(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            # Reconstruct full vector of size n_points
            if res < n_points:
                # Upsample
                h_upsampled = np.interp(np.linspace(0, 1, n_points), 
                                       np.linspace(0, 1, res), h_vec)
                h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
                return evaluate_erdos_solution(h_upsampled)
            else:
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                return evaluate_erdos_solution(h_vec)
        
        # Bounds for the differential evolution
        bounds = [(0, 1) for _ in range(res)]
        
        # Differential Evolution
        de_result = differential_evolution(
            objective_de,
            bounds,
            maxiter=200,
            popsize=40,
            mutation=0.8,
            recombination=0.9,
            seed=seed,
            tol=1e-6,
            disp=False
        )
        
        if de_result.success:
            # Process the best found solution
            h_trial = de_result.x
            h_trial = np.clip(h_trial, 0, 1)
            
            # Upsample if needed
            if res < n_points:
                h_upsampled = np.interp(np.linspace(0, 1, n_points), 
                                       np.linspace(0, 1, res), h_trial)
                h_trial = h_upsampled
            else:
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
            # Local refinement using SLSQP
            def objective_slsqp(h_vec):
                return evaluate_erdos_solution(h_vec)
            
            # Constraint for integral = 1
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            # Bounds
            bounds_slsqp = [(0, 1) for _ in range(n_points)]
            constraints_slsqp = {'type': 'eq', 'fun': constraint_integral}
            
            try:
                res_slsqp = minimize(
                    objective_slsqp,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds_slsqp,
                    constraints=constraints_slsqp,
                    options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res_slsqp.success:
                    h_candidate = np.clip(res_slsqp.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                pass
            
            # Final evaluation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
