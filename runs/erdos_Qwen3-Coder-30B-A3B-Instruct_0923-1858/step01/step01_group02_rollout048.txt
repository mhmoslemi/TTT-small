import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    dx = 2.0 / n_points
    
    # Track previous solutions to avoid cycles
    prev_solutions = set()
    
    # Use cutting plane method
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlaps to find maximum
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Build convex optimization problem
        h_vars = Variable(n_points)
        C = Variable(1)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # Sum constraint
        constraints.append(sum(h_vars) == n_points / 2)
        
        # Cutting planes
        for k in range(len(corr)):
            # Compute gradient of correlation for current k
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] = dx * (1 - best_h[j])
            
            # Add cutting plane: grad^T * h <= C - grad^T * h_best + corr[k]*dx
            constraints.append(grad.T @ h_vars <= C - grad.T @ best_h + corr[k] * dx)
            # Ensure C >= current overlap
            constraints.append(C >= corr[k] * dx)
            
        #Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except:
            break
            
        if prob.status != cp.OPTIMAL:
            break
            
        h_new = h_vars.value.flatten()
        c5_new = C.value
        
        # Check if solution is valid and better
        if np.all(h_new >= 0) and np.all(h_new <= 1):
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)  # Renormalize
            current_c5 = evaluate_erdos_solution(h_new)
            
            if current_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = current_c5
                
        # Check for convergence
        sol_tuple = tuple(np.round(h_new, decimals=6))
        if sol_tuple in prev_solutions:
            break
        else:
            prev_solutions.add(sol_tuple)
            
        if time.time() - start_time > budget_s - 0.1:
            break

    final_c5 = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), float(final_c5), n_points)
