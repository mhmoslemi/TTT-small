import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    resolutions = [50, 100]
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Differential Evolution for global search
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            # Ensure the integral constraint
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            return evaluate_erdos_solution(h_vec)
        
        # DE optimization
        try:
            # Use a simpler strategy for DE
            de_res = differential_evolution(
                objective, 
                bounds, 
                maxiter=100, 
                popsize=40,
                mutation=0.8,
                recombination=0.9,
                seed=seed,
                strategy='best1bin'
            )
            
            if de_res.success:
                h_candidate = np.clip(de_res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            # If DE fails, keep existing
            pass
            
        # Local refinement with SLSQP if not on final resolution
        if n_points < resolutions[-1] and time.time() - start_time < budget_s - 5:
            try:
                # Upsample to higher resolution
                h_upsampled = np.interp(
                    np.linspace(0, 1, n_points), 
                    np.linspace(0, 1, len(best_h)), 
                    best_h
                )
                h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
                
                # Refine with SLSQP
                constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(
                    objective,
                    h_upsampled,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                        
            except Exception:
                pass
    
    # Final optimization with SLSQP on the best found
    if time.time() - start_time < budget_s - 1:
        try:
            # Refine with SLSQP from the final best
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - len(best_h) / 2}
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            res = minimize(
                lambda x: evaluate_erdos_solution(np.clip(x, 0, 1) * (len(best_h) / 2) / np.sum(x)),
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-7}
            )
            
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (len(best_h) / 2) / np.sum(h_final)
                c5_final = evaluate_erdos_solution(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
                    
        except Exception:
            pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, len(best_h))
