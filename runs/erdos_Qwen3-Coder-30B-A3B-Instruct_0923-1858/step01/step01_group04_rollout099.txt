import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function for differential evolution
    def de_objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Define constraints for differential evolution (only equality constraint)
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Run differential evolution for global search
    de_res = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=40,
        tol=1e-6,
        recombination=0.9,
        mutation=0.8,
        seed=seed,
        constraints=constraints,
        disp=False
    )
    
    best_h = de_res.x.copy()
    best_h = np.clip(best_h, 0, 1)
    # Ensure integral constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Now try local refinement with SLSQP
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    constraints_slsqp = {'type': 'eq', 'fun': constraint_integral}
    
    # Try several random starting points for SLSQP
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution slightly
        perturbation = np.random.normal(0, 0.02, n_points)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        try:
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds_slsqp,
                constraints=constraints_slsqp,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
