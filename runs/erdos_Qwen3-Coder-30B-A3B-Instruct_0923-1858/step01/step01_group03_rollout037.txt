import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_tenure = 50
    tabu_list = []
    max_iter = 10000
    perturb_rate = 0.03  # 3% of points
    max_perturb = max(1, int(n_points * perturb_rate))
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def perturb_solution(h_vec, max_perturb):
        new_h = h_vec.copy()
        num_perturb = np.random.randint(1, max_perturb + 1)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            new_h[idx] = np.random.uniform(0, 1)
        return project_feasible(new_h)
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = current_h.copy()
    
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        candidate_h = perturb_solution(current_h, max_perturb)
        candidate_C5 = compute_C5(candidate_h)
        
        # Check if candidate is better and not tabu
        is_better = candidate_C5 < best_C5
        is_tabu = any(np.allclose(candidate_h, h_prev, atol=1e-6) for h_prev in tabu_list)
        
        if is_better or not is_tabu:
            current_h = candidate_h
            current_C5 = candidate_C5
            
            if is_better:
                best_C5 = current_C5
                best_h = current_h.copy()
            
            # Add to tabu list
            tabu_list.append(current_h.copy())
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        
        iter_count += 1
        if iter_count > max_iter:
            break
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
