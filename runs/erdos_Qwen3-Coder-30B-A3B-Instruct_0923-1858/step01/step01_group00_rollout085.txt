import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for better resolution
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling factor
    sigma = 0.1  # Perturbation scale
    shake_prob = 1.0  # Probability of shaking when no improvement
    max_no_improve = 100  # Number of iterations without improvement to trigger shake
    no_improve_count = 0
    
    # Keep track of the best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start timer
    start_time = time.time()
    
    # SA loop
    while time.time() - start_time < budget_s:
        # Generate a perturbation
        delta = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_new = h + delta
        
        # Project to feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate both solutions
        current_c5 = evaluate_erdos_solution(h)
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Accept or reject the new solution
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
        
        # Cool the temperature
        T = alpha * T
        
        # Shake if no improvement
        if no_improve_count >= max_no_improve and np.random.rand() < shake_prob:
            h = np.clip(h + np.random.normal(0, T * sigma, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            no_improve_count = 0

    # Final evaluation of the best solution
    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5

    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, actual_c5, n_points)
