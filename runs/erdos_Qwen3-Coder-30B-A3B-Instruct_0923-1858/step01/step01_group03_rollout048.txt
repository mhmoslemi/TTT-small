            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    # Simplified gradient descent with better adaptive steps
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec):
        # Use finite differences to estimate gradient
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        h_base = np.array(h_vec)
        corr_base = correlate(h_base, 1 - h_base, mode="full")
        max_k_idx = np.argmax(corr_base)
        
        for i in range(len(h_vec)):
            h_plus = h_base.copy()
            h_minus = h_base.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-8:
            return eta / grad_norm
        return eta
    
    # Main loop with adaptive steps and restarts
    iter_count = 0
    patience = 0
    max_patience = 50
    eta = 0.01

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
