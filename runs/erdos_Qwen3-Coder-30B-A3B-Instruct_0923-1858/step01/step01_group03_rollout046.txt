import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Trust Region SQP approach
    def trust_region_sqp(h_start, max_iter=1000, tr_radius=0.5, tol=1e-4):
        h = h_start.copy()
        for _ in range(max_iter):
            if elapsed() >= budget_s * 0.95:
                break
                
            # Compute current gradient (simplified here)
            # In practice, this would be computed as gradient of the max correlation
            # For this implementation, we'll use finite differences
            eps = 1e-5
            grad = np.zeros_like(h)
            current_val = compute_C5(h)
            
            for i in range(len(h)):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                grad[i] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
            
            # Gradient descent step in trust region
            step = -grad
            step_norm = np.linalg.norm(step)
            if step_norm > tr_radius:
                step = step * tr_radius / step_norm
            
            new_h = h + step
            new_h = project_feasible(new_h)
            
            # Accept step if it improves objective
            new_C5 = compute_C5(new_h)
            if new_C5 < compute_C5(h):
                h = new_h
            else:
                # Reduce trust region radius
                tr_radius *= 0.5
            
            # Update best solution
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                
        return h
    
    # Simple optimization loop with restarts and perturbations
    patience = 0
    max_patience = 50
    perturb_prob = 0.1
    eta = 0.01
    
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        iter_count += 1
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Perturb randomly
        if np.random.random() < perturb_prob or patience > max_patience:
            idx = np.random.choice(n_points)
            h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
            # Rescale
            h = project_feasible(h)
            patience = 0
            
        # Simple gradient descent step
        grad = np.zeros_like(h)
        eps = 1e-5
        current_val = compute_C5(h)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            grad[i] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
            
        h = h - eta * grad
        h = project_feasible(h)
        
        # Decay learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
            
        # Occasionally restart with best solution found
        if iter_count % 1000 == 0:
            h = best_h.copy()
            
    # Ensure we have the best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
