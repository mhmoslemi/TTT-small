import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iterations = 2000
    perturb_rate = 0.03  # 3% of points
    max_no_improve = 100
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_perturbed_solution(current_h, rate):
        # Create a copy
        new_h = current_h.copy()
        num_perturb = max(1, int(rate * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            # Random perturbation
            new_h[i] = np.clip(new_h[i] + np.random.normal(0, 0.05), 0, 1)
        # Project to feasible set
        new_h = project_feasible(new_h)
        return new_h
    
    # Main tabu search loop
    iter_count = 0
    no_improve_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = get_perturbed_solution(h, perturb_rate)
        
        # Check if candidate is in tabu list
        candidate_tuple = tuple(candidate_h.round(6))
        if candidate_tuple in tabu_list:
            # Skip this candidate
            continue
            
        # Compute C5 for candidate
        candidate_C5 = compute_C5(candidate_h)
        
        # Update best solution if better
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate_h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                # Reset perturbation rate or add noise
                perturb_rate = min(0.1, perturb_rate * 1.1)
                no_improve_count = 0
        
        # Update tabu list
        tabu_list.append(candidate_tuple)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Update current solution
        h = candidate_h
        
        iter_count += 1
        # Occasionally reduce mutation rate to refine search
        if iter_count % 500 == 0:
            perturb_rate = max(0.01, perturb_rate * 0.95)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
