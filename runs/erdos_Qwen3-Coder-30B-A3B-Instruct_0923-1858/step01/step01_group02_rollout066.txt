import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

            else:
                h[i] = 0.3
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    cooling_rate = 0.9995
    min_temperature = 1e-6
    max_iter = 100000
    iteration = 0
    
    # Current best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Acceptance probability function
    def acceptance_probability(old_energy, new_energy, temperature):
        if new_energy < old_energy:
            return 1.0
        return np.exp(-(new_energy - old_energy) / temperature)
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and T > min_temperature:
        # Generate candidate solution by moving mass between random indices
        candidate_h = h.copy()
        i, j = np.random.randint(0, n_points, 2)
        
        # Move a small amount of mass from i to j
        delta = np.random.uniform(0.001, 0.01)

        current_energy = evaluate_erdos_solution(h)
