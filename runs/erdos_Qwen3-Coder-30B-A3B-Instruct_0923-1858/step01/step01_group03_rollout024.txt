    def solve_convex_relaxation():
        # Use CVXPY for convex relaxation
        h_var = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        # Constraints for convex relaxation
        constraints = [0 <= h_var, h_var <= 1]
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Add constraints for each shift
        for k in range(-n_points + 1, n_points):
            # Compute correlation indices efficiently
            corr_indices = []
            valid_idx = 0
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    corr_indices.append((i, j))
                    valid_idx += 1
            
            # Sum of h[i] * (1 - h[j]) for valid pairs
            if corr_indices:
                corr_expr = cp.sum([h_var[i] * (1 - h_var[j]) for i, j in corr_indices])
                # Add constraint that this sum >= lambda_var
                constraints.append(corr_expr >= lambda_var)
        
        # Objective: maximize lambda
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            return np.array(h_var.value).reshape(-1)
        except:
            return None

    try:
        relaxed_h = solve_convex_relaxation()
        if relaxed_h is not None:
            h = relaxed_h

        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)

            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

        grad = np.zeros_like(h_vec)

        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
