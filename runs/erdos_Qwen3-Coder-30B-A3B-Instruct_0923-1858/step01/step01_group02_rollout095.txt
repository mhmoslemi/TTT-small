import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        h = np.ones(n_points) * 0.5
        # Add some variation to avoid symmetry
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Time left for optimization
    remaining_time = budget_s - (time.time() - start_time)
    
    # Hyperparameters
    max_iterations = min(1000, int(remaining_time / 0.5)) if remaining_time > 0.5 else 100
    iter_count = 0
    
    while iter_count < max_iterations and (time.time() - start_time) < budget_s - 0.5:
        # Compute current overlaps
        dx = 2.0 / n_points
        n_corr = 2 * n_points - 1
        corr = np.correlate(h, 1 - h, mode="full")
        # Normalize correlation to get overlap integrals
        overlaps = corr * dx
        
        # Find the index of maximum overlap (this is C_5)
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        # Formulate the cutting plane approach
        # We'll solve a convex optimization problem with constraints based on current h
        hp = cp.Variable(n_points)
        C = cp.Variable()
        
        # Compute the gradient of overlaps with respect to h
        # We'll use the fact that ∂O_k/∂h[i] = dx * (1 - h[i+k]) when k is valid
        constraints = []
        
        # Feasibility constraints
        constraints.append(cp.sum(hp) == n_points / 2)
        constraints.extend([hp[i] >= 0 for i in range(n_points)])
        constraints.extend([hp[i] <= 1 for i in range(n_points)])
        
        # Cutting planes based on current h
        for k in range(n_points):
            # The contribution of shift k to overlap is sum_{i} h[i] * (1 - h[(i+k) mod n_points])
            # Gradient w.r.t. h[i] is dx*(1 - h[(i+k) mod n_points])
            # We estimate this with h and create a linear approximation at h
            gradient = np.zeros(n_points)
            for i in range(n_points):
                idx = (i + k) % n_points
                gradient[i] = dx * (1 - h[idx])
            
            # Cutting plane: gradient^T * h <= gradient^T * h + overlap_k - gradient^T * h
            # I.e.: gradient^T * h <= overlap_k
            # We are doing: 0 <= overlap_k - gradient^T * h + gradient^T * h  
            # This is equivalent to: gradient^T * h <= overlap_k
            constraints.append(gradient @ hp <= overlaps[k])

        # Add objective to minimize the C
        objective = cp.Minimize(C)
        
        # Setup problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = hp.value
                new_h = np.clip(new_h, 0, 1)
                
                # Renormalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                current_c5 = evaluate_erdos_solution(new_h)
                
                if current_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = current_c5
        except Exception:
            pass
            
        # Update h to make it more diverse
        h = best_h.copy()  # Latest known good solution
        
        iter_count += 1
        if (time.time() - start_time) > budget_s - 0.5:
            break
            
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
