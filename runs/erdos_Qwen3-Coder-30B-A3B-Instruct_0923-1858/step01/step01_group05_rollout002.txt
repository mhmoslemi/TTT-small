import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Resample to 140 points for better resolution
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_orig, h)
        n_points = 140
    else:
        # Create a feasible h vector
        n_points = 140
        h = np.zeros(n_points)
        # Place most mass in the center
        center_start = n_points // 3
        center_end = 2 * n_points // 3
        h[center_start:center_end] = 1.0
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping with multiple iterations
    n_iterations = 200
    noise_std = 0.05
    for i in range(n_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_trial = h + np.random.normal(0, noise_std, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Refine with SLSQP
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        # Decaying noise
        if i % 50 == 0 and i > 0:
            noise_std /= 2
    
    # Local refinement of top candidates
    top_candidates = []
    for _ in range(5):
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                top_candidates.append((res.x, objective(res.x)))
        except:
            pass
    
    if top_candidates:
        best_candidate = min(top_candidates, key=lambda x: x[1])
        if best_candidate[1] < best_c5:
            best_h = best_candidate[0]
            best_c5 = best_candidate[1]
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
