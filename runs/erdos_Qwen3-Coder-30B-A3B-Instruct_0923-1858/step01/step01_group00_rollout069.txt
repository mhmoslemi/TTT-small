import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define number of points
    n_points = 100
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Linearly interpolate to n_points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Start timer
    start_time = time.time()
    
    # Keep track of best solution
    best_c5 = float('inf')
    best_h = h.copy()

    # Use multiple optimization strategies
    strategies = [
        {'method': 'L-BFGS-B', 'options': {'maxiter': 200}},
        {'method': 'SLSQP', 'options': {'maxiter': 200}},
    ]
    
    # Try multiple restarts
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Random perturbation of initial solution
            h_pert = h + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert / np.sum(h_pert) * (n_points / 2)

            # Try multiple strategies
            for strategy in strategies:
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                try:
                    result = minimize(
                        evaluate_erdos_solution,
                        h_pert.copy(),
                        method=strategy['method'],
                        bounds=[(0, 1) for _ in range(n_points)],
                        options=strategy['options'],
                        tol=1e-6
                    )
                    
                    if result.success:
                        current_c5 = evaluate_erdos_solution(result.x)
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = result.x.copy()
                except:
                    continue
                    
            # Perturb for next restart
            h_pert = best_h.copy() + np.random.normal(0, 0.005, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert / np.sum(h_pert) * (n_points / 2)
            h = h_pert
            
        except:
            # Reset h if we encountered an error
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)

    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5

    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, actual_c5, n_points)
