            h[i] = max(0, 1 - abs(x[i] - 1))

        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.2
    stagnation_limit = 50
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add the initial solution
    swarm.append(h.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        candidate = np.random.rand(n_points)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        swarm.append(candidate)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal best and global best
    pbest = [swarm[i].copy() for i in range(Pop_size)]
    pbest_fitness = [float('inf')] * Pop_size
    gbest = swarm[0].copy()
    gbest_fitness = float('inf')
    
    # Evaluate initial solutions
    for i in range(Pop_size):
        corr = correlate(swarm[i], 1 - swarm[i], mode="full")
        fitness = np.max(corr) * (2.0 / n_points)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest = swarm[i].copy()
    
    # Main PSO loop
    iter_count = 0
    stagnant = 0
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave margin for finalization
            break
        
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +

                            c2 * r2 * (gbest - swarm[i]))

            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])

            corr = correlate(swarm[i], 1 - swarm[i], mode="full")
            fitness = np.max(corr) * (2.0 / n_points)

                pbest_fitness[i] = fitness
                pbest[i] = swarm[i].copy()

            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest = swarm[i].copy()
                stagnant = 0  # reset stagnation counter
        
        # Check for stagnation
        stagnant += 1

            for i in range(int(Pop_size * 0.1)):

                corr = correlate(swarm[idx], 1 - swarm[idx], mode="full")

                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest = swarm[idx].copy()
            stagnant = 0
        
        iter_count += 1
        # Decay inertia weight
        w = max(0.3, 0.7 * (1.0 - iter_count / 500))
    
    # Final evaluation
    final_C5 = compute_C5(gbest)
    if final_C5 < gbest_fitness:
        gbest_fitness = final_C5
    
    return (gbest.tolist(), gbest_fitness, len(gbest))
