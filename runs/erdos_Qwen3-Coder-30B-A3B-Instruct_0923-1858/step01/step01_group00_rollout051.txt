import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    # Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up PSO parameters
    n_points = 200
    swarm_size = 50
    max_iter = int(budget_s * 10)  # Rough estimate of iterations
    inertia = 0.7
    c1 = c2 = 1.5
    perturb_std = 0.01
    alpha = 0.9  # low-pass filter alpha
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = project_to_feasible(h_init, n_points)
        
    particles = [h_init.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    # Evaluate initial particles
    fitness = [evaluate_erdos_solution(p) for p in particles]
    
    # Personal bests
    personal_best = particles.copy()
    personal_best_fitness = fitness.copy()
    
    # Global best
    global_best_idx = np.argmin(fitness)
    global_best = particles[global_best_idx].copy()
    global_best_fitness = fitness[global_best_idx]
    
    # Main loop
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        for i in range(swarm_size):
            # Update velocity with low-pass filtered perturbation
            r1, r2 = np.random.rand(2)
            v_new = (inertia * velocities[i] + 
                    c1 * r1 * (personal_best[i] - particles[i]) + 
                    c2 * r2 * (global_best - particles[i]))
            
            # Apply low-pass filter
            velocities[i] = alpha * velocities[i] + (1 - alpha) * v_new
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            current_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = current_fitness
                
                # Update global best if needed
                if current_fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = current_fitness
        
        # Perturb all particles to avoid getting stuck
        for i in range(swarm_size):
            noise = np.random.normal(0, perturb_std, n_points)
            particles[i] = particles[i] + noise
            particles[i] = project_to_feasible(particles[i], n_points)
    
    # Final evaluation of the global best
    final_c5 = evaluate_erdos_solution(global_best)
    
    # Ensure final solution is valid
    final_h = project_to_feasible(global_best, n_points)
    
    return (final_h, final_c5, n_points)
