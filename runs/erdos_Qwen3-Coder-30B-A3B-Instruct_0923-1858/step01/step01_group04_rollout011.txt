import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback to basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure constraints are met
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Convert to CVXPY variables
    h_var = Variable(n_points)
    t = Variable()
    
    # Compute overlaps for all shifts
    dx = 2.0 / n_points
    constraints = []
    
    # Add integral constraint
    constraints.append(sum(h_var) == n_points / 2)
    
    # Add bounds
    constraints.extend([h_var[i] >= 0 for i in range(n_points)])
    constraints.extend([h_var[i] <= 1 for i in range(n_points)])
    
    # Add overlap constraints with barrier
    max_shift = n_points - 1
    for k in range(-max_shift, max_shift + 1):
        # Compute the correlation for shift k
        if k >= 0:
            corr = h_var[k:] * (1 - h_var[:n_points-k])
        else:
            corr = h_var[:n_points+k] * (1 - h_var[-k:])
        overlap = sum(corr)
        constraints.append(overlap <= t * dx)
    
    # Objective: minimize t
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    iter_count = 0
    
    # Iterative refinement with CVXPY
    while time.time() - start_time < budget_s - 1 and iter_count < 3:
        try:
            # Set the initial value
            h_var.value = best_h.copy()
            
            # Solve with CVXOPT solver
            prob.solve(solver=CVXOPT, verbose=False, max_iters=1000)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = np.array(h_var.value).flatten()
                new_h = np.clip(new_h, 0, 1)
                # Renormalize to sum to n_points / 2
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = evaluate_erdos_solution(new_h)
                
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    
        except:
            pass
            
        iter_count += 1
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
