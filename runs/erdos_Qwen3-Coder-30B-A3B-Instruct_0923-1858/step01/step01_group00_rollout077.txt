import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use larger discretization for better precision
    n_points = 200
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Use Particle Swarm Optimization
    # Parameters
    swarm_size = 50
    max_iter = 600
    inertia = 0.7
    c1 = c2 = 1.5
    perturb_std = 0.01
    alpha = 0.9  # Low-pass filter alpha for velocity
    
    # Initialize swarm
    swarm = np.random.rand(swarm_size, n_points) * 0.5
    swarm = swarm / np.sum(swarm, axis=1, keepdims=True) * (n_points / 2)
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    # Evaluate initial swarm
    costs = np.array([evaluate_erdos_solution(particle) for particle in swarm])
    
    # Personal best
    pbest = swarm.copy()
    pbest_costs = costs.copy()
    
    # Global best
    gbest_idx = np.argmin(costs)
    gbest = swarm[gbest_idx].copy()
    gbest_cost = costs[gbest_idx]
    
    # Keep track of best solution found
    best_c5 = gbest_cost
    best_h = gbest.copy()
    
    start_time = time.time()
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        for i in range(swarm_size):
            # Update velocity with low-pass filtered perturbation
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] 
                           + c1 * r1 * (pbest[i] - swarm[i]) 
                           + c2 * r2 * (gbest - swarm[i]))
            
            # Apply low-pass filter to velocity
            velocities[i] = alpha * velocities[i] + (1 - alpha) * np.random.normal(0, 0.01, n_points)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Apply feasibility constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] / np.sum(swarm[i]) * (n_points / 2)
            
            # Evaluate new position
            new_cost = evaluate_erdos_solution(swarm[i])
            
            # Update personal best
            if new_cost < pbest_costs[i]:
                pbest[i] = swarm[i].copy()
                pbest_costs[i] = new_cost
                
            # Update global best
            if new_cost < gbest_cost:
                gbest = swarm[i].copy()
                gbest_cost = new_cost
                
                if gbest_cost < best_c5:
                    best_c5 = gbest_cost
                    best_h = gbest.copy()
        
        # Apply global perturbation to all particles
        for i in range(swarm_size):
            swarm[i] += np.random.normal(0, perturb_std, n_points)
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] / np.sum(swarm[i]) * (n_points / 2)
    
    # Ensure final solution is feasible and recompute
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
