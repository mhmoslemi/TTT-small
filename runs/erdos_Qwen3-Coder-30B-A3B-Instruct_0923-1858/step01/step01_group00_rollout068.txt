import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Try to improve the result using an improved gradient-based approach
    best_c5 = float('inf')
    best_h = h.copy()

    # Binary search parameters
    low_bound = 0.0
    high_bound = 0.38085
    tolerance = 1e-5
    
    # Optimization parameters
    max_iter = int(budget_s / 10)  # Roughly 60 iterations
    start_time = time.time()

    # Start with a simple gradient-based step to speed things up
    def objective(h):
        return evaluate_erdos_solution(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Start optimization with good initialization
    try:
        result = minimize(
            objective,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 200},
            tol=1e-6
        )
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    except:
        pass

    # Use multiple random restarts
    for i in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random restart with scaling
        h_rand = np.random.rand(n_points) * 0.5
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        
        try:
            result = minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 100},
                tol=1e-6
            )
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        except:
            continue

    # Final refinement using BFGS with same bounds and constraint if needed
    try:
        result = minimize(
            objective,
            best_h.copy(),
            method='BFGS',
            options={'maxiter': 100, 'gtol': 1e-6}
        )
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    except:
        pass

    # Ensure best_h is in the correct range
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)

    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception as e:
        final_c5 = best_c5

    return (best_h, final_c5, n_points)
