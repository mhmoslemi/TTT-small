import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a feasible random h vector
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 2000
    iter_count = 0
    perturb_freq = 50
    
    # Initial triangular construction for potentially better starting point
    if 'initial_h_values' not in globals():
        h = np.zeros(n_points)
        x_vals = np.linspace(0, 2, n_points)
        # Create a triangular shaped function
        h = 1 - np.abs(x_vals - 1)  # Triangle centered at x=1
        h = h / np.sum(h) * (n_points / 2.0)
        h = np.clip(h, 0, 1)
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perform perturbation every few iterations
        if iter_count % perturb_freq == 0:
            h_perturbed = h.copy()
            # Perturb in the middle region where overlap typically happens
            mid_idx = n_points // 4  # In the center 50% region
            for i in range(mid_idx, 3*mid_idx):
                h_perturbed[i] += np.random.normal(0, 0.03)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2.0)
            h_trial = h_perturbed
        else:
            # Slightly perturb existing h
            h_trial = h + np.random.normal(0, 0.005, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Try different local optimizers
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
