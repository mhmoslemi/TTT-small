import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with existing construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation approach
    def convex_initialization():
        # Maximize a lower bound lambda such that all overlaps are >= lambda
        h_vars = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints
        for k in range(n_points):
            # For each lag k, sum h[i] * (1 - h[i+k]) >= lambda
            # wrap around k by using modulo
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr >= lambda_var)
        
        # Objective
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6, reltol=1e-6)
            if prob.status == cp.OPTIMAL:
                return np.array(h_vars.value).reshape(-1)
        except:
            pass
        return h.copy()  # fallback
    
    # Try convex initialization first
    if time.time() - start_time < budget_s * 0.4:
        h = convex_initialization()
    
    # Local refinement with constrained gradient descent
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Main iterative optimization
    eta = 0.02
    max_patience = 50
    patience = 0
    max_iter = 10000
    
    # Random restarts
    num_restarts = 5
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize from current best or random
        h = best_h.copy() if restart == 0 else np.random.rand(n_points) * 0.5
        h = project_feasible(h)
        
        iter_count = 0
        while iter_count < max_iter and time.time() - start_time < budget_s * 0.95:
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Perturb randomly
                    idx = np.random.choice(n_points)
                    h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                    h = project_feasible(h)
                    patience = 0

            # Gradient step
            grad = get_gradient(h)
            h = h - eta * grad
            h = project_feasible(h)
            
            # Adaptive learning rate
            if iter_count % 50 == 0 and iter_count > 0:
                eta *= 0.98
            
            iter_count += 1
            if iter_count % 100 == 0 and best_C5 < 0.3809:
                break  # Early exit if close to target
    
    # Final validation if needed
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
