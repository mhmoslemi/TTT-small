import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h[::2] = 0.7
        h[1::2] = 0.3
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-8
    iterations_per_temp = 100
    
    temp = initial_temp
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and temp > min_temp:
        for _ in range(iterations_per_temp):
            # Create a new candidate solution by perturbing h
            h_candidate = h.copy()
            
            # Select two random indices to transfer mass
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            
            # Amount of mass to transfer (small random amount)
            delta = np.random.uniform(0.001, 0.05)
            
            # Transfer mass from idx1 to idx2
            if h_candidate[idx1] >= delta:
                h_candidate[idx1] -= delta
                h_candidate[idx2] += delta
            else:
                delta = h_candidate[idx1]  # Transfer all remaining mass
                h_candidate[idx1] = 0
                h_candidate[idx2] += delta
            
            # Ensure the sum constraint remains satisfied
            current_sum = np.sum(h_candidate)
            h_candidate = h_candidate * (n_points / 2) / current_sum
            
            # Ensure values are within bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Calculate energy (C5 value) of the candidate
            current_c5 = evaluate_erdos_solution(h_candidate)
            
            # Accept or reject the candidate
            if current_c5 < best_c5:
                h = h_candidate.copy()
                best_c5 = current_c5
                if current_c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = current_c5
            else:
                # Metropolis criterion
                delta_e = current_c5 - best_c5
                if np.random.rand() < np.exp(-delta_e / temp):
                    h = h_candidate.copy()
            
            iteration += 1
            
        # Decrease temperature
        temp *= cooling_rate
        
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
