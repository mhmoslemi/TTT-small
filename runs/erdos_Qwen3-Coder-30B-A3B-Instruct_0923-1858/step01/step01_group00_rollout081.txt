import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):

    # Parameters
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 10000
    step_size = 0.1
    decay_rate = 0.95
    perturb_prob = 0.1
    smooth_window = 3
    patience = 100
    best_c5 = float('inf')

        h[0] = 0.0
        h[-1] = 0.0
    
    # Normalize
    h = h / np.sum(h) * (n_points / 2)
    
    # Optimization loop
    start_time = time.time()
    last_improvement = 0
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:

        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = i
        
        # Calculate gradient
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        grad = np.zeros(n_points)
        
        # Construct gradient based on the maximum lag
        for k in range(n_points):
            if k == max_k:
                idx = k - (n_points - 1) // 2
                if idx >= 0 and idx < n_points:
                    grad[idx] += 2.0 * (1.0 - 2.0 * h[idx])
        
        # Gradient step
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Perturbation
        if np.random.rand() < perturb_prob and i - last_improvement > patience:

        # Reduce step size
        if i % 100 == 0:
            step_size *= decay_rate
    
    # Final check
    if best_h is None:
        best_h = h.copy()
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
