import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize to sum(h) = n_points/2
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling factor
    sigma = 0.1  # Perturbation scale
    max_stagnation = 100
    stagnation_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Annealing loop
    while time.time() - start_time < budget_s:
        # Generate perturbation
        delta_h = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_trial = h + delta_h
        
        # Project to feasibility
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2)
        
        # Evaluate
        try:
            c5_trial = evaluate_erdos_solution(h_trial)
            c5_current = evaluate_erdos_solution(h)
            
            # Accept or reject
            if c5_trial < c5_current or np.random.rand() < np.exp(-(c5_trial - c5_current) / T):
                h = h_trial.copy()
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
            
            # Shake every max_stagnation steps if no improvement
            if stagnation_count >= max_stagnation:
                h = h + np.random.normal(0, T * sigma, n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
                stagnation_count = 0
                
        except:
            # On any error, do nothing (stay with current h)
            pass
        
        # Cool temperature
        T = T * alpha
        
        # Occasionally check time
        if (time.time() - start_time) > 0.1:
            if time.time() - start_time > budget_s:
                break
    
    # Final check
    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5
    
    return (best_h, actual_c5, n_points)
