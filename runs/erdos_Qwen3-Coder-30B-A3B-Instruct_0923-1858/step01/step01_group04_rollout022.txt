import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    # Step 1: coarse resolution
    n_coarse = min(50, n_points)
    if n_coarse < n_points:
        # Downsample by averaging
        h_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            start_idx = i * n_points // n_coarse
            end_idx = (i + 1) * n_points // n_coarse
            h_coarse[i] = np.mean(h[start_idx:end_idx])
        # Normalize to integral constraint
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        h_coarse = np.clip(h_coarse, 0, 1)
    else:
        h_coarse = h.copy()
    
    # Define objective for coarse resolution
    def objective_coarse(h_coarse_vec):
        # Interpolate to full resolution
        h_full = np.interp(np.linspace(0, n_coarse, n_points),

        h_full = np.clip(h_full, 0, 1)

    bounds_coarse = [(0, 1) for _ in range(n_coarse)]

        def objective_full(h_vec):
            return evaluate_erdos_solution(h_vec)

            return np.sum(h_vec) - n_points / 2

    final_c5 = evaluate_erdos_solution(best_h)
