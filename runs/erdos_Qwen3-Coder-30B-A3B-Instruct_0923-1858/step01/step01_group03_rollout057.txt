import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    n_points = 45
    pop_size = 100
    mutation_rate = 0.2
    tournament_size = 5
    max_patience = 50
    
    # Use initial_h_values if available
    try:
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    except NameError:
        # Create a triangular initial solution
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h_init[i] = max(0, 1 - abs(x[i] - 1))
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create initial population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def tournament_selection(pop, k):
        indices = np.random.choice(len(pop), size=k, replace=False)
        tournament = [pop[i] for i in indices]
        fitnesses = [compute_C5(h) for h in tournament]
        return tournament[np.argmin(fitnesses)]
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(h_vec, rate):
        h_new = h_vec.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                # Add small Gaussian noise
                h_new[i] += np.random.normal(0, 0.05)
        return project_feasible(h_new)
    
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    patience = 0
    
    # Evolution loop
    generation = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = compute_C5(h)
            fitnesses.append(c5)
            if c5 < best_C5:
                best_C5 = c5
                best_h = h.copy()
                patience = 0
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        
        # Check for improvement
        patience += 1
        if patience > max_patience:
            # Reset population with new random individuals
            population = [best_h.copy()]
            for _ in range(pop_size - 1):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
            patience = 0
        
        # Create new population
        new_population = [population[0]]  # Elitism
        
        # Tournament selection, crossover, and mutation
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, tournament_size)
            parent2 = tournament_selection(population, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    return (best_h.tolist(), best_C5, len(best_h))
