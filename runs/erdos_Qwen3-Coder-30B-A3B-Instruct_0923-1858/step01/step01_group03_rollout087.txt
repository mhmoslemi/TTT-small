        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 2000
    perturb_rate = 0.05  # 5% of points
    max_perturb_per_iter = max(1, int(n_points * perturb_rate))
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def generate_neighbor(current_h):
        neighbor = current_h.copy()
        # Randomly select indices to perturb
        num_perturb = np.random.randint(1, max_perturb_per_iter + 1)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        for idx in indices:
            # Perturb value within [0, 1]
            perturbation = np.random.normal(0, 0.05)
            neighbor[idx] = neighbor[idx] + perturbation
            # Clip and rescale to keep feasible
            neighbor[idx] = np.clip(neighbor[idx], 0, 1)
        
        # Rescale to maintain constraint
        neighbor = project_feasible(neighbor)
        return neighbor
    
    # Main tabu search loop
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate neighbor
        neighbor = generate_neighbor(h)
        neighbor_C5 = compute_C5(neighbor)
        
        # Check if neighbor is better than current best
        if neighbor_C5 < best_C5:
            best_C5 = neighbor_C5
            best_h = neighbor.copy()
        
        # Add current to tabu list
        h_tuple = tuple(h)
        if h_tuple in tabu_list:
            tabu_list.remove(h_tuple)
