import time

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Genetic Algorithm parameters
    population_size = 100
    mutation_rate = 0.2
    tournament_size = 5
    generations = 500
    
    def compute_C5(h_vec):
        # Convert to correct size
        h_vec = np.array(h_vec)
        # Note: correlate with mode='full' to handle all shifts
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum over all shifts
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h.copy())
        for _ in range(population_size - 1):
            # Random perturbation of initial solution with clipping and rescaling
            pert = h + np.random.normal(0, 0.05, n_points)
            population.append(project_feasible(pert))
    else:
        for _ in range(population_size):
            h_rand = np.random.rand(n_points)
            population.append(project_feasible(h_rand))

    patience = 0
    max_patience = 100

        fitnesses = []

            fit = compute_C5(ind)
            fitnesses.append(fit)
            if fit < best_C5:
                best_C5 = fit

            new_population.append(child)
            
        population = new_population
    
    # Final evaluation

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
