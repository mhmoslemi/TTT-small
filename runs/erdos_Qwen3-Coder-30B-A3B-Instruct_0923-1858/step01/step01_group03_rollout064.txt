import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    perturbation_rate = 0.03  # 3% of points
    max_iter = 2000
    
    def compute_C5(h_vec):
        # Convert to correct size
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_neighbors(h_vec, num_neighbors=5):
        neighbors = []
        for _ in range(num_neighbors):
            new_h = h_vec.copy()
            # Randomly select indices to modify
            num_modify = max(1, int(perturbation_rate * n_points))
            indices = np.random.choice(n_points, num_modify, replace=False)
            for i in indices:
                new_h[i] = np.random.uniform(0, 1)
            new_h = project_feasible(new_h)
            neighbors.append(new_h)
        return neighbors
    
    # Main tabu search loop
    iter_count = 0
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = current_h.copy()
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave margin
            break
            
        neighbors = get_neighbors(current_h)
        
        best_neighbor = None
        best_neighbor_C5 = float('inf')
        
        for neighbor in neighbors:
            neighbor_C5 = compute_C5(neighbor)
            
            # Check if this neighbor is tabu
            is_tabu = False
            for tabu_h in tabu_list:
                if np.allclose(neighbor, tabu_h, atol=1e-6):
                    is_tabu = True
                    break
            
            if neighbor_C5 < best_neighbor_C5 and not is_tabu:
                best_neighbor = neighbor
                best_neighbor_C5 = neighbor_C5
        
        if best_neighbor is not None:
            current_h = best_neighbor
            current_C5 = best_neighbor_C5
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
            
            # Add to tabu list
            tabu_list.append(current_h.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
            
            # Check for convergence
            if iter_count % 100 == 0 and iter_count > 0:
                # If no improvement in last 100 iterations, add more perturbations
                pass
        else:
            # No good neighbors found, so do random perturbation
            idx = np.random.choice(n_points)
            current_h[idx] = np.clip(current_h[idx] + np.random.normal(0, 0.05), 0, 1)
            current_h = project_feasible(current_h)
        
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
