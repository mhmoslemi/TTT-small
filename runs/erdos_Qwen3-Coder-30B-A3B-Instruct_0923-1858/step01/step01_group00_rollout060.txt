import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up the problem size
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        # Interpolate the initial construction to our n_points
        h_original = np.array(initial_h_values)
        # Create a linear interpolation function
        x_orig = np.linspace(0, 2, len(h_original))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h_original)
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a simple construction
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing Parameters
    temp = 1.0
    cooling_rate = 0.999
    perturb_scale = 0.1
    shake_threshold = 100
    shake_prob = 1.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s:
        # Perturb the current solution
        delta = np.random.uniform(-temp * perturb_scale, temp * perturb_scale, n_points)
        h_candidate = h + delta
        # Project to feasible region [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        # Rescale to maintain the integral constraint
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate objective
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Accept or reject the candidate
        if c5_candidate < best_c5 or np.random.rand() < np.exp(-(c5_candidate - best_c5) / temp):
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Apply shake if stagnation is detected
        if stagnation_count > shake_threshold and np.random.rand() < shake_prob:
            h += np.random.normal(0, temp * perturb_scale, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            stagnation_count = 0
            
        # Cool the temperature
        temp *= cooling_rate
        iteration += 1
        
        # Occasionally check if time is almost up
        if iteration % 100 == 0 and time.time() - start_time > budget_s - 5:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
