import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize PSO parameters
    swarm_size = 50
    max_iter = int(budget_s / 0.1) if budget_s > 0.1 else 1000
    omega = 0.7
    c1 = c2 = 1.5
    alpha = 0.9  # low-pass filter alpha for velocity
    
    # Initialize particles and velocities
    if 'initial_h_values' in globals():
        particles = np.array(initial_h_values)
        particles = particles / np.sum(particles) * (n_points / 2)
        particles = particles.reshape(-1, 1)
    else:
        particles = np.random.rand(n_points, swarm_size) * 0.5
        for i in range(swarm_size):
            particles[:, i] = project_to_feasible(particles[:, i], n_points)
    
    velocities = np.random.uniform(-0.1, 0.1, (n_points, swarm_size))
    
    # Personal and global bests
    personal_best = particles.copy()
    personal_best_fitness = np.array([evaluate_erdos_solution(p) for p in particles.T])
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[:, global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # Start timing
    start_time = time.time()
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        for i in range(swarm_size):
            # Update velocity with inertia and cognitive/social components
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = c1 * r1 * (personal_best[:, i] - particles[:, i])
            social = c2 * r2 * (global_best - particles[:, i])
            velocities[:, i] = omega * velocities[:, i] + cognitive + social
            
            # Apply low-pass filtering to velocity
            velocities[:, i] = alpha * velocities[:, i] + (1 - alpha) * np.roll(velocities[:, i], 1)
            
            # Update position
            particles[:, i] += velocities[:, i]
            
            # Project to feasibility
            particles[:, i] = project_to_feasible(particles[:, i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[:, i])
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[:, i] = particles[:, i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best = particles[:, i].copy()
                    global_best_fitness = fitness
        
        # Perturb all particles slightly to avoid stagnation
        perturbation = np.random.normal(0, 0.01, (n_points, swarm_size))
        particles += perturbation
        
        # Re-project all to feasibility
        for i in range(swarm_size):
            particles[:, i] = project_to_feasible(particles[:, i], n_points)
    
    # Final check with the best found solution
    final_h = project_to_feasible(global_best, n_points)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, n_points)
