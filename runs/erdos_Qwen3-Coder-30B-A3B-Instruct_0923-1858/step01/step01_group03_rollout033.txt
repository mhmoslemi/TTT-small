import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        # Create a triangular initial solution

        h_init = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Genetic Algorithm Parameters
    population_size = 100
    mutation_rate = 0.2
    tournament_size = 5
    num_generations = 500
    elite_size = 10
    
    # Initialize population
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # Create initial population
    population = []
    population.append(h_init.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h)
        population.append(h)
    
    best_h = h_init.copy()
    best_C5 = compute_C5(best_h)
    
    # Keep track of generation without improvement
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Generation loop
    for generation in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_C5(h)
            fitness.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C5:
            best_C5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Reset if no improvement for too long
        if no_improvement_count > max_no_improvement:
            # Reinitialize population with current best
            population = [best_h.copy()]
            for _ in range(population_size - 1):
                h = np.random.rand(n_points)
                h = project_feasible(h)
                population.append(h)
            no_improvement_count = 0
            continue
        
        # Selection (Tournament)
        new_population = []
        
        # Elite individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = tournament_selection(population, fitness, tournament_size)
            parent2_idx = tournament_selection(population, fitness, tournament_size)
            
            # Crossover
            child1, child2 = crossover(population[parent1_idx], population[parent2_idx])
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Project
            child1 = project_feasible(child1)
            child2 = project_feasible(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:population_size]
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))

def tournament_selection(population, fitness, tournament_size):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return winner_idx

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate):
    """Gaussian mutation"""
    mutated = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return mutated
