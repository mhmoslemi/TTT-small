        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 2000
    perturb_rate = 0.03  # 3% of points per iteration
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_neighbors(h_vec, num_neighbors=5):
        neighbors = []
        for _ in range(num_neighbors):
            # Create a new neighbor by perturbing some points
            new_h = h_vec.copy()
            num_perturb = max(1, int(perturb_rate * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                # Add a small random perturbation, ensuring feasibility
                delta = np.random.normal(0, 0.05)
                new_h[idx] += delta
            new_h = project_feasible(new_h)
            neighbors.append(new_h)
        return neighbors
    
    # Main tabu search loop
    current_h = h.copy()
    iter_count = 0
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
        
        neighbors = get_neighbors(current_h, 5)
        
        # Evaluate all neighbors
        best_neighbor = None
        best_neighbor_C5 = float('inf')
        
        for neighbor in neighbors:
            neighbor_C5 = compute_C5(neighbor)
            # Check if neighbor is not in tabu list
            neighbor_tuple = tuple(neighbor.round(6))
            if neighbor_tuple not in tabu_list:
                if neighbor_C5 < best_neighbor_C5:
                    best_neighbor = neighbor.copy()
                    best_neighbor_C5 = neighbor_C5
        
        # Update best solution
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_C5 = current_C5

        # Update tabu list
        current_tuple = tuple(current_h.round(6))
        if current_tuple not in tabu_list:
            tabu_list.append(current_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)

        if best_neighbor is not None:
            current_h = best_neighbor
        else:

            idx = np.random.choice(n_points)
            current_h[idx] = np.clip(current_h[idx] + np.random.normal(0, 0.03), 0, 1)
            current_h = project_feasible(current_h)
        
        iter_count += 1
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
