import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def mutate(h, dx, rate=0.1):
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, 0.05)
    return project(mutated, dx)

def crossover(parent1, parent2):
    child = parent1.copy()
    for i in range(len(child)):
        if np.random.rand() < 0.5:
            child[i] = parent2[i]
    return child

def select_parents(population, fitnesses, num_parents=10):
    indices = np.argsort(fitnesses)[-num_parents:]
    return [population[i] for i in indices]

def evolve_population(population, dx, budget_s=600):
    start_time = time.time()
    best_fitness = 0
    best_h = None
    population_size = len(population)
    
    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitnesses = [1 / (evaluate(ind, dx) + 1e-10) for ind in population]
        current_best = population[np.argmax(fitnesses)]
        current_fitness = max(fitnesses)
        if current_fitness > best_fitness:
            best_fitness = current_fitness
            best_h = current_best.copy()
        
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Create new population
        new_population = []
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < 0.2:
                child = mutate(child, dx)
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation of best solution found
    best_C = evaluate(best_h, dx)
    return best_h, best_C

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, population_size=50)
    
    # Evolve
    best_h, best_C = evolve_population(population, dx, budget_s)
    
    return (best_h, best_C, n_points)
