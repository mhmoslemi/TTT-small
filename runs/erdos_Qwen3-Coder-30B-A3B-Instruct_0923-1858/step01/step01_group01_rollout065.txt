import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1, strength=0.1):
    """Mutate h with small random changes."""
    new_h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < rate:
            new_h[i] += np.random.normal(0, strength)
    return project(new_h, dx)

def crossover(parent1, parent2):
    """Create child from two parents using uniform crossover."""
    child = np.where(np.random.rand(len(parent1)) < 0.5, parent1, parent2)
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Tournament selection."""
    selected = []
    for _ in range(num_parents):
        idx = np.random.choice(len(population), size=2, replace=False)
        if fitnesses[idx[0]] > fitnesses[idx[1]]:
            selected.append(population[idx[0]])
        else:
            selected.append(population[idx[1]])
    return selected

    dx = 2.0 / n_points
    
    population = initialize_population(n_points)
    best_h = None
    best_C = float('inf')
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = 1 / (evaluate(h, dx) + 1e-12)
            fitnesses.append(fitness)
        
        # Track best solution
        current_best_idx = np.argmax(fitnesses)
        current_C = evaluate(population[current_best_idx], dx)
        if current_C < best_C:
            best_C = current_C
            best_h = population[current_best_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses, num_parents=20)
        
        # Generate new population
        new_population = []
        # Elitism: keep best
        new_population.append(population[current_best_idx])
        
        # Create offspring
        for _ in range(len(population) - 1):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child, dx, rate=0.15, strength=0.05)
            new_population.append(child)
        
        population = new_population
    
    if best_h is None:
        # Fallback to the best individual from initial population
        test_fitnesses = [1 / (evaluate(h, dx) + 1e-12) for h in population]
        best_idx = np.argmax(test_fitnesses)
        best_h = population[best_idx]
        best_C = evaluate(best_h, dx)
    
    return (best_h, best_C, n_points)
