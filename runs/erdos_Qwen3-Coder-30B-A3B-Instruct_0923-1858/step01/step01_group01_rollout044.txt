import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1, strength=0.1):
    """Mutate h with given rate and strength."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, strength)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    child = np.zeros_like(parent1)
    for i in range(len(child)):
        if np.random.rand() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Tournament selection."""
    tournament_size = 3
    selected = []
    for _ in range(num_parents):
        tournament = np.random.choice(len(population), size=tournament_size, replace=False)
        winner = tournament[np.argmax([fitnesses[i] for i in tournament])]
        selected.append(population[winner])
    return selected

def initialize_population(n_points, population_size=50):
    """Initialize population using initial_h_values with mutations."""
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = initial_h_values.copy()
        else:
            # Create a triangular shape if no parent
            h = np.linspace(0, 1, n_points//2, endpoint=False)
            h = np.concatenate([h, np.linspace(1, 0, n_points - n_points//2)])
        # Apply mutation to get diversity
        if np.random.rand() < 0.3:
            h = mutate(h, 2.0/n_points, 0.1, 0.05)
        population.append(h)
    return population

def evolve_population(population, dx, budget_s=600):
    """Evolution process for the genetic algorithm."""
    start_time = time.time()
    best_fitness = 0
    best_h = None
    
    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitnesses = [1 / (evaluate(h, dx) + 1e-10) for h in population]
        
        # Track best solution
        current_best_idx = np.argmax(fitnesses)
        current_fitness = fitnesses[current_best_idx]
        if current_fitness > best_fitness:
            best_fitness = current_fitness
            best_h = population[current_best_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Generate new population
        new_population = []
        while len(new_population) < len(population):
            # Crossover
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            # Mutation
            if np.random.rand() < 0.2:
                child = mutate(child, dx)
            new_population.append(child)
        
        # Keep population size
        population = new_population[:len(population)]
        
        # Add a random individual every few iterations to maintain diversity
        if len(population) > 0 and np.random.rand() < 0.1:
            population.pop()
            if 'initial_h_values' in globals():
                h = initial_h_values.copy()
            else:
                h = np.linspace(0, 1, 45//2, endpoint=False)
                h = np.concatenate([h, np.linspace(1, 0, 45 - 45//2)])
            h = mutate(h, dx, 0.1, 0.03)
            population.append(h)
    
    return best_h, 1 / best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, 50)
    
    # Evolve population
    best_h, best_C = evolve_population(population, dx, budget_s)
    
    return (best_h, best_C, n_points)
