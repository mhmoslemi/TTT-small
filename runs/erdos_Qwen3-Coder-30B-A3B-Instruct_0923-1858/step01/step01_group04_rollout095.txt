import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define integral constraint
    def constraint_integral(x):
        return np.sum(x) - n_points / 2
    
    # Constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Objective function to minimize
    def objective(x):
        x = np.clip(x, 0, 1)
        return evaluate_erdos_solution(x)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    resolutions = [50, 100]
    for n_res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        if n_res == 50:
            # Coarse resolution for global search
            # Use differential evolution for better exploration
            de_bounds = [(0, 1) for _ in range(n_res)]
            
            def de_objective(x):
                # Rescale to make sum = n_res/2
                x = np.clip(x, 0, 1)
                x = x * (n_res / 2) / np.sum(x)
                return evaluate_erdos_solution(x)
            
            # Differential evolution with custom bounds
            res = differential_evolution(
                de_objective,
                de_bounds,
                seed=seed,
                maxiter=100,
                popsize=40,
                mutation=(0.5, 1.0),
                recombination=0.9,
                disp=False,
                polish=False
            )
            
            if res.success:
                h_coarse = np.clip(res.x, 0, 1)
                h_coarse = h_coarse * (n_res / 2) / np.sum(h_coarse)
                # Evaluate this coarse candidate
                c5_candidate = evaluate_erdos_solution(h_coarse)
                if c5_candidate < best_c5:
                    best_h = h_coarse
                    best_c5 = c5_candidate
        
        elif n_res == 100 and n_points < 100:
            # Upsample from 50 to 100 points
            h_upsampled = np.interp(
                np.linspace(0, 1, 100),
                np.linspace(0, 1, 50),
                best_h
            )
            # Rescale to sum to 50
            h_upsampled = h_upsampled * 50 / np.sum(h_upsampled)
            h_upsampled = np.clip(h_upsampled, 0, 1)  
            
            # Refine with SLSQP
            try:
                result = minimize(
                    objective,
                    h_upsampled,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200}
                )
                if result.success:
                    h_refined = np.clip(result.x, 0, 1)
                    h_refined = h_refined * (n_res / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final improvement with local search using SLSQP
    if time.time() - start_time < budget_s - 1:
        try:
            # Random restarts to avoid local optima
            for _ in range(5):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Perturb best solution
                perturbation = np.random.normal(0, 0.05, n_points)
                h_trial = best_h + perturbation
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                result = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100}
                )
                if result.success:
                    h_candidate = np.clip(result.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
        except Exception:
            pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
