import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    resolutions = [50, 70, 100]
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new h with selected resolution
        if n_points <= len(h):
            h_new = h[:n_points].copy()
        else:
            # Upsample by interpolation
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, x_old, h)
            # Adjust to meet integral
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_new = np.clip(h_new, 0, 1)
        
        # Try different starting points
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break

            perturbation = np.random.normal(0, 0.05, n_points)

            def objective(h_vec):

                return evaluate_erdos_solution(h_vec)

            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2

            constraints = {'type': 'eq', 'fun': constraint_integral}

                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,

                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
