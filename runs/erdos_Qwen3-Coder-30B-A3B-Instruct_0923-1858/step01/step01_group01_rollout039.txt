import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def initialize_h(n_points, initial_h):
    if initial_h is not None:
        return initial_h
    # Initialize with a smoother curve to decay to zero at edges
    t = np.linspace(0, 2, n_points)
    h = np.exp(-10 * (t - 1)**2) + 0.1 * np.sin(2 * np.pi * t)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * 2.0 / n_points))
    return h

def optimize_cvxpy(h, dx, n_points, budget_s=580):
    """Optimize h using convex optimization in frequency domain."""
    try:
        # Create variables in frequency domain  
        H = cp.Variable(n_points, complex=True)
        
        # Constraint: h must be real (Hermitian symmetry)
        # We do this by constraining real part to represent a valid real signal
        h_real = cp.real(H)
        
        # Constraint: integral = 1
        # The integral of h(x) dx = 1 means sum(h) * dx = 1
        # In frequency domain: sum(h) = 1/dx * H[0] (by DC component)
        constraint1 = cp.real(H[0]) == 1.0 / dx
        
        # The overlap integral is related to |H(k) * (1 - H*(-k))|
        # For maximization, we want to minimize the maximum of this
        # Create correlation in frequency domain (simplified approach)
        # We'll use a simpler dual approach using smoothness constraint
        
        # Optimization
        objective = cp.Minimize(cp.norm(H, 2) + 1e-8)  # Regularized minimization
        prob = cp.Problem(objective, [constraint1])
        start_time = time.time()
        prob.solve(solver=cp.SCS, verbose=False, time_limit=budget_s-10)
        
        end_time = time.time()
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            H_sol = H.value
            h_sol = np.real(np.fft.ifft(H_sol))
            h_sol = project(h_sol, dx)
            return h_sol, evaluate(h_sol, dx)
        else:
            # If SCS fails, fall back to gradient descent
            return h, evaluate(h, dx)
    except Exception as e:
        return h, evaluate(h, dx)

def optimize_hill_climb(h, dx, n_points, budget_s=580):
    """Optimize using hill-climbing with projected gradient."""
    t0 = time.time()
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    lr = 0.005
    decay = 0.999
    max_iter = 10000  # Typical number of steps
    
    for i in range(max_iter):
        if time.time() - t0 > budget_s:
            break
            
        # Estimate gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        h = h - lr * grad
        h = project(h, dx)
        
        # Project more frequently
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_h = h.copy()
            best_C = current_C
            lr = max(0.0001, lr * 1.01)  # Occasionally increase learning rate
            
        lr *= decay
        if i % 100 == 0:
            lr = max(lr, 0.0001)
            
    return best_h, best_C
