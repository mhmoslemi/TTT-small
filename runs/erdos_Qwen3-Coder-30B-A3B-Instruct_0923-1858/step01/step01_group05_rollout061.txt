import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Create a feasible initial h vector
        n_points = 140
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Ensure h is in [0,1] and sum to 70 (i.e. n_points/2)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2.0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Define the constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * (2.0 / n_points)
    
    # Global exploration with perturbation
    iter_count = 0
    max_iter = 1000
    sigma = 0.05
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the solution
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Evaluate trial solution
        c5_trial = objective(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
            
        # Local refinement: take the top 10 previously good solutions
        if iter_count % 10 == 0 and iter_count > 0:
            try:
                # Select top 10 by tolerance
                top_indices = np.argsort([evaluate_erdos_solution(h) for h in [best_h]])[:10]
                for i in range(min(10, len(top_indices))):
                    h_candidate = best_h + np.random.normal(0, 0.005, n_points)
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2.0)
                    
                    res = minimize(objective, h_candidate, method='SLSQP', bounds=bounds, constraints=cons, 
                                 options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                            h = h_refined.copy()
            except:
                pass
                
        iter_count += 1
        # Progessively reduce noise
        if iter_count % 50 == 0:
            sigma *= 0.5
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
