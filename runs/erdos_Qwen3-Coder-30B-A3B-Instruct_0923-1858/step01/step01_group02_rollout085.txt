import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        # Add some variation to avoid symmetry
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Initialize C5 to a high value
    c5_bound = 0.4
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Simple optimization approach using scipy
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(x):
            return evaluate_erdos_solution(x)
        
        # Do local optimization
        try:
            # Random perturbation to escape local minima
            perturbation = np.random.normal(0, 0.01, n_points)
            x0 = np.clip(best_h + perturbation, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)  # Renormalize
            
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
            
            if res.success:
                current_c5 = evaluate_erdos_solution(res.x)
                if current_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = current_c5
        except Exception:
            pass
            
        # Alternate approach: try small local changes to h
        try:
            h_trial = best_h.copy()
            # Make a small change to one element
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.05, 0.05)
            h_trial[idx] += delta
            h_trial = project_to_feasible(h_trial, n_points)
            
            current_c5 = evaluate_erdos_solution(h_trial)
            if current_c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = current_c5
        except Exception:
            pass
            
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
