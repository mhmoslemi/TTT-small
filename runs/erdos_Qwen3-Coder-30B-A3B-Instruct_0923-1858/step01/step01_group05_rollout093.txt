import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def perturb_and_repair(h, sigma=0.03):
    # Apply Gaussian noise only to the middle region where overlap is typically high
    n = len(h)
    h_new = h.copy()
    # Define middle region: x in [0.3, 0.7] corresponds to indices [0.3*70, 0.7*70] = [21, 49]
    start_idx = int(0.3 * n)
    end_idx = int(0.7 * n)
    
    # Add Gaussian noise to middle region
    noise = np.random.normal(0, sigma, end_idx - start_idx)
    h_new[start_idx:end_idx] += noise
    
    # Clip and rescale
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (35.0 / np.sum(h_new))
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create an initial triangle-like function for starting point
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure it's clipped and rescaled
    h = np.clip(h, 0, 1)
    h = h * (35.0 / np.sum(h))
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - 35.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h_vec):
        # Compute correlation with (1-h)
        corr = np.correlate(h_vec, 1-h_vec, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 5000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break

        # Apply perturbation in critical region
        h_trial = perturb_and_repair(h)
        
        # Try local optimization with gradient method
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        iter_count += 1
        # Occasionally do a full global perturbation
        if iter_count % 100 == 0:
            h = perturb_and_repair(h, sigma=0.05)
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
