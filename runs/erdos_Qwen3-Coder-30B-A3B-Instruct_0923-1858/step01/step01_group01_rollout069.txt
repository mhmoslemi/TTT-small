import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def initialize_population(n_points, pop_size=50, initial_h=None):
    population = []
    for i in range(pop_size):
        if initial_h is not None and i < len(initial_h):
            h = initial_h.copy()
        else:
            # Start with triangular waveform
            h = np.linspace(0, 1, n_points//2, endpoint=False)
            h = np.concatenate([h, np.linspace(1, 0, n_points - n_points//2)])
        # Add some random noise
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = project(h, 2.0/n_points)
        population.append(h)
    return population

def evaluate_fitness(h, dx):
    return 1.0 / (evaluate(h, dx) + 1e-12)

def select_parents(population, fitnesses, num_parents=10):
    tournament_size = 3
    selected = []
    for _ in range(num_parents):
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner])
    return selected

def crossover(parent1, parent2):
    # Uniform crossover
    mask = np.random.rand(len(parent1)) < 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, dx, mutation_rate=0.1, noise_std=0.1):
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, noise_std)
    return project(mutated, dx)

def evolve_population(population, dx, budget_s=600):
    start_time = time.time()
    best_fitness = 0
    best_h = None
    
    while time.time() - start_time < budget_s - 10:  # leave time for final check
        # Evaluate fitness
        fitnesses = [evaluate_fitness(h, dx) for h in population]
        
        # Track best solution
        current_best_idx = np.argmax(fitnesses)
        current_fitness = fitnesses[current_best_idx]
        if current_fitness > best_fitness:
            best_fitness = current_fitness
            best_h = population[current_best_idx].copy()
        
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Generate new population
        new_population = []
        while len(new_population) < len(population):
            # Perform crossover
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            # Apply mutation
            if np.random.rand() < 0.2:
                child = mutate(child, dx)
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Early stopping if no improvement in last 50 iterations
        # Not implemented here for simplicity, but can be added
        
    return best_h, 1.0 / best_fitness
