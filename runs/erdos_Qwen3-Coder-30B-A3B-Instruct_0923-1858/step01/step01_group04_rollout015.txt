import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Iteratively refine the solution
    max_iter = 300
    iter_count = 0
    remaining_time = budget_s - 1
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Step 1: Try gradient-based optimization with random perturbations
        # Create perturbed version
        h_trial = h.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.03, n_points)
        h_trial += noise
        h_trial = np.clip(h_trial, 0, 1)
        # Rescale to meet constraint
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Local optimization
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)
        
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
                
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        try:
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    # Use this as the new base for next iteration
                    h = h_candidate.copy()
            else:
                # If not successful, try another approach
                h = h_trial
        except Exception:
            h = h_trial
        
        # Step 2: Sometimes use a different random initialization strategy
        if iter_count % 5 == 0 and iter_count > 0:
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            h_rand = np.clip(h_rand, 0, 1)
            c5_rand = evaluate_erdos_solution(h_rand)
            if c5_rand < best_c5:
                best_h = h_rand
                best_c5 = c5_rand
                h = h_rand.copy()
        
        iter_count += 1
        
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
