import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points

        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_orig, h)
        n_points = 140
    else:
        n_points = 140
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum to 70 (since 2/140 * 70 = 1)
    h = h / np.sum(h) * 70.0
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Global exploration with perturbations
    iter_count = 0
    max_iter = 1000
    sigma = 0.05
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb h globally
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * 70.0  # normalize to sum 70
        
        # Evaluate
        c5_trial = evaluate_erdos_solution(h_trial)
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
            
        # Local refinement using SLSQP - take top 10 perturbations
        if iter_count % 20 == 0 and iter_count > 0:
            # Select top 10 candidates
            candidates = []
            for _ in range(50):
                hp = h + np.random.normal(0, sigma, n_points)
                hp = np.clip(hp, 0, 1)
                hp = hp / np.sum(hp) * 70.0
                c5_test = evaluate_erdos_solution(hp)
                candidates.append((c5_test, hp))
            
            candidates.sort(key=lambda x: x[0])
            
            # Refine top 10
            for i in range(min(10, len(candidates))):
                h_candidate = candidates[i][1]
                bounds = [(0, 1) for _ in range(n_points)]
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - 70.0
                
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h_vec):
                    corr = np.correlate(h_vec, 1-h_vec, mode="full")
                    return np.max(corr) * (2.0 / n_points)
                
                try:
                    res = minimize(objective, h_candidate, method='SLSQP', 
                                 bounds=bounds, constraints=cons, 
                                 options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        h_refined = np.clip(h_refined, 0, 1)
                        h_refined = h_refined / np.sum(h_refined) * 70.0
                        
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                except:
                    pass
                    
        iter_count += 1
        # Adaptive sigma
        if iter_count % 50 == 0:
            sigma = max(0.005, sigma * 0.5)
    
    # Final validation - crude check still needed
    final_c5 = evaluate_erdos_solution(best_h)
    # Ensure final sum matches exactly
    best_h = best_h / np.sum(best_h) * 70.0
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
