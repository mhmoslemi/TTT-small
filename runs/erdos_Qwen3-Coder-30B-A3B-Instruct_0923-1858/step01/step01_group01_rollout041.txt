import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate_population(population, dx, mutation_rate=0.1):
    """Apply structured mutation to a population."""
    mutated = []
    for h in population:
        if np.random.rand() < mutation_rate:
            # Random perturbation with projection
            idx = np.random.randint(len(h))
            h[idx] += np.random.normal(0, 0.05)
            h = project(h, dx)
        mutated.append(h)
    return mutated

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def select_tournament(population, fitnesses, tournament_size=3):
    """Tournament selection for parent selection."""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        fitnesses_selected = [fitnesses[i] for i in indices]
        winner_idx = indices[np.argmax(fitnesses_selected)]
        selected.append(population[winner_idx])
    return selected

def evolve_population(population, dx, budget_s):
    """Main evolution loop for genetic algorithm."""
    start_time = time.time()
    best_h = None
    best_C = float('inf')
    population_size = len(population)
    
    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness (inverse of C5)
        fitnesses = [1.0 / (evaluate(h, dx) + 1e-12) for h in population]
        
        # Track best solution so far
        current_best_idx = np.argmax(fitnesses)
        current_C = 1.0 / fitnesses[current_best_idx]
        if current_C < best_C:
            best_C = current_C
            best_h = population[current_best_idx].copy()
        
        # Selection
        parents = select_tournament(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            new_population.append(child)
        
        new_population = mutate_population(new_population, dx)
        population = new_population
        
    return best_h, best_C
