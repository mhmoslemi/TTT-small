import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize C5 bound
    c5_bound = 0.4
    
    # Iteratively decrease the bound using convex optimization
    iteration = 0
    max_iterations = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        try:
            # Define variables
            h_var = cp.Variable(n_points)
            
            # Objective is to minimize the maximum overlap
            # Set objective to minimize c5_bound (will be updated iteratively)
            objective = cp.Minimize(c5_bound)
            
            # Constraints
            constraints = []
            
            # Sum constraint
            constraints.append(cp.sum(h_var) == n_points / 2)
            
            # Bound constraints (for each shift k)
            dx = 2.0 / n_points
            for k in range(n_points):
                # Compute correlation constraint for shift k
                # This approximates ∫ h(x)(1-h(x+k))dx ≤ c5_bound * dx
                if k < n_points:
                    corr_constraint = cp.sum(cp.multiply(h_var[:n_points-k], (1 - h_var[k:]))) <= c5_bound / dx
                    constraints.append(corr_constraint)
                else:
                    corr_constraint = cp.sum(cp.multiply(h_var[:n_points-k], (1 - h_var[k:]))) <= c5_bound / dx
                    constraints.append(corr_constraint)
            
            # Bound constraints for h[i]
            constraints.extend([h_var[i] >= 0 for i in range(n_points)])
            constraints.extend([h_var[i] <= 1 for i in range(n_points)])
            
            # Formulate and solve the problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h_var.value
                c5_test = evaluate_erdos_solution(h_sol)
                if c5_test < c5_bound:
                    c5_bound = c5_test
                else:
                    # Try to decrease the bound further
                    c5_bound -= 0.0001
            else:
                # If infeasible, increase bound
                c5_bound += 0.0001
        except:
            c5_bound += 0.0001
        
        # Project solution back to feasible set
        h = np.clip(h_sol, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        iteration += 1
        
        # Early termination check
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h)
    
    return (h.tolist(), float(final_c5), n_points)
