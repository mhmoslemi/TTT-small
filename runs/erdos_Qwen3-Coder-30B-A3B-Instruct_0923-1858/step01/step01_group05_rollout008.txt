import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a feasible random h vector with more points for better resolution
        n_points = 140
        h = np.random.rand(n_points) * 0.5  # Start with smaller values
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 300
    crossover_rate = 0.5
    mutation_rate = 0.3
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i < population_size // 2 and 'initial_h_values' in globals():
            # Use initial values with small perturbation
            h_perturbed = h + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2.0)
            population.append(h_perturbed)
        else:
            # Random initialization
            h_rand = np.random.rand(n_points) * 0.5
            h_rand = h_rand / np.sum(h_rand) * (n_points / 2.0)
            h_rand = np.clip(h_rand, 0, 1)
            population.append(h_rand)
    
    def tournament_selection(population, fitness, k=tournament_size):
        indices = np.random.choice(len(population), size=k, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Blend crossover
        alpha = np.random.random()
        child1 = alpha * parent1 + (1 - alpha) * parent2
        child2 = (1 - alpha) * parent1 + alpha * parent2
        
        # Block swap for variation
        if np.random.random() > 0.5:
            block_size = max(1, n_points // 10)
            start1 = np.random.randint(0, n_points - block_size)
            start2 = np.random.randint(0, n_points - block_size)
            child1[start1:start1+block_size], child2[start2:start2+block_size] = \
                child2[start2:start2+block_size], child1[start1:start1+block_size]
        
        return child1, child2
    
    def mutate(individual, mutation_rate, sigma=0.03):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if np.random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, sigma)
        return mutated
    
    def repair(individual):
        individual = np.clip(individual, 0, 1)
        individual = individual / np.sum(individual) * (n_points / 2.0)
        return individual
    
    # Main GA loop
    no_improvement_count = 0
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        
        # Track best
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx]
        current_c5 = fitness[current_best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination if no improvement
        if no_improvement_count > 150:
            break
            
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            child1 = repair(child1)
            child2 = repair(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
