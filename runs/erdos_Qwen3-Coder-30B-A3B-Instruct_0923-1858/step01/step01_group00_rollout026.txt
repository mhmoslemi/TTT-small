import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    n_points = 200
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        # Rescale to match appropriate sum
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Use a particle swarm optimization approach with better exploration
    # Consistent with algorithmic planning guidance

    # Initialize swarm
    swarm_size = 50
    particles = []
    velocities = []
    
    # Initialize particles
    for _ in range(swarm_size):
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h_init = np.array(initial_h_values)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
            h_p = h_init + np.random.normal(0, 0.01, n_points)
        else:
            h_p = np.random.rand(n_points) * 0.5
            h_p = h_p / np.sum(h_p) * (n_points / 2)
        # Clip to [0,1]
        h_p = np.clip(h_p, 0, 1)
        # Project to sum constraint
        h_p = h_p / np.sum(h_p) * (n_points / 2)
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests and global best
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # PSO parameters
    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    alpha = 0.9  # Low-pass filter alpha
    perturbation_std = 0.01

    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        try:
            for i in range(swarm_size):
                # Update velocity with inertia and cognitive/social components
                r1, r2 = np.random.rand(2)
                personal_best = personal_bests[i]
                vel = velocities[i]
                
                # Apply low-pass filtering to velocity
                vel = alpha * vel + (1 - alpha) * np.random.normal(0, 0.02, n_points)
                
                vel = omega * vel + c1 * r1 * (personal_best - particles[i]) + \
                      c2 * r2 * (global_best - particles[i])
                
                # Update position
                new_pos = particles[i] + vel
                
                # Apply feasibility projection
                new_pos = np.clip(new_pos, 0, 1)
                new_pos = new_pos / np.sum(new_pos) * (n_points / 2)
                
                # Evaluate the new position
                new_score = evaluate_erdos_solution(new_pos)
                
                # Update personal best if improvement
                if new_score < personal_best_scores[i]:
                    personal_bests[i] = new_pos.copy()
                    personal_best_scores[i] = new_score
                    
                    # Update global best if needed
                    if new_score < global_best_score:
                        global_best = new_pos.copy()
                        global_best_score = new_score
                        
                # Store updated particle and velocity
                particles[i] = new_pos
                velocities[i] = vel
                
            # Apply gentle perturbation to all particles
            for i in range(swarm_size):
                noise = np.random.normal(0, perturbation_std, n_points)
                particles[i] += noise
                # Project to constraint
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
                
        except Exception:
            # Only continue with a fresh initialization if something breaks.
            if time.time() - start_time < budget_s - 5:
                particles = []
                velocities = []

                    if 'initial_h_values' in globals() and initial_h_values is not None:

                        h_init = h_init / np.sum(h_init) * (n_points / 2)

                    else:
                        h_p = np.random.rand(n_points) * 0.5
                        h_p = h_p / np.sum(h_p) * (n_points / 2)
                    h_p = np.clip(h_p, 0, 1)
                    h_p = h_p / np.sum(h_p) * (n_points / 2)
                    particles.append(h_p.copy())
                    velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    # Final evaluation and return best feasible result
    try:
        final_c5 = evaluate_erdos_solution(global_best)
    except:
        final_c5 = global_best_score
        
    return (global_best, final_c5, n_points)
