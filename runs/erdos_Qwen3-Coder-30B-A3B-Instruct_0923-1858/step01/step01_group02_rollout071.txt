import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initial construction using balanced piecewise constant function
        n_points = 70
        h = np.full(n_points, 0.5)
        # Create a more targeted construction with peaks
        h[:35] = 0.7
        h[35:] = 0.3
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    dx = 2.0 / n_points
    
    # Cutting plane method
    max_iter = 2000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for current solution
        overlaps = []
        # Compute the full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # The overlap integrals for all k shifts are in indices n_points-1 to 2*n_points-2
        for k in range(n_points):
            overlap_val = corr[n_points - 1 + k] * dx
            overlaps.append(overlap_val)
        
        max_overlap = np.max(overlaps)
        
        # Build the convex optimization problem
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = []
        for i in range(n_points):
            constraints.append(0 <= h_vars[i])
            constraints.append(h_vars[i] <= 1)
        
        constraints.append(cp.sum(h_vars) == n_points / 2)
        constraints.append(C >= max_overlap)
        
        # Add cutting planes
        for k_idx, overlap in enumerate(overlaps):
            # Gradient for shift k_idx - we compute the gradient of the overlap integral
            # O_k(h) = dx * sum(h[i] * (1 - h[i+k_idx]))
            # dO_k/dh[i] = dx * (1 - h[i + k_idx]) if i + k_idx < n_points else 0 
            grad = np.zeros(n_points)
            for i in range(n_points):
                next_i = i + k_idx 
                if next_i < n_points:
                    grad[i] = dx * (1 - h[next_i])
            
            # Constraint: grad^T * h <= grad^T * h_current + (overlap - grad^T * h_current)
            # => grad^T * h <= overlap
            constraints.append(grad @ h_vars <= overlap)
            
        # Objective
        objective = cp.Minimize(C)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                c5_new = C.value
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                # Improve with local gradient-based update
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to heuristics
                h = np.clip(h + 0.01 * np.random.randn(n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            h = np.clip(h + 0.01 * np.random.randn(n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), float(final_c5), n_points)
