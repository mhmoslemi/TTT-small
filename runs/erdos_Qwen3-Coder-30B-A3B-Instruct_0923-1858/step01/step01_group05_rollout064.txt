import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():

        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_orig, h)
        n_points = 140
    else:
        # Create a feasible initial h vector
        n_points = 140
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation and local refinement approach
    perturb_sigma = 0.05
    max_iter = 1000
    iter_count = 0
    population_size = 50
    
    # Track the top solutions
    top_solutions = []
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Global search through perturbations
        candidates = []
        for _ in range(population_size):
            if time.time() - start_time > budget_s - 1:
                break
            h_trial = h + np.random.normal(0, perturb_sigma, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            c5_trial = objective(h_trial)
            candidates.append((h_trial.copy(), c5_trial))
            
        # Sort candidates by objective value
        candidates.sort(key=lambda x: x[1])
        
        # Keep top 10 for local refinement
        top_candidates = candidates[:10] if len(candidates) >= 10 else candidates
        
        # Local refinement with SLSQP
        for h_trial, c5_trial in top_candidates:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass
            
        # Update the best solution from global search
        for h_trial, c5_trial in candidates:
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                h = h_trial.copy()
                
        # adaptive perturbation
        if iter_count % 50 == 0:
            perturb_sigma *= 0.5
        
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
