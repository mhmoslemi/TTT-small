def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = h * (n_points / 2) / np.sum(h)

    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization settings
    max_iter = 500
    perturbation_magnitude = 0.1
    stagnation_count = 0
    stagnation_threshold = 20

            x0 = x0 * (n_points / 2) / np.sum(x0)

            bounds = [(0, 1) for _ in range(n_points)]

            def objective(x):
                return evaluate_erdos_solution(x)
            
            # Use scipy optimize for local refinement
            opt_result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 30})
