import time

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 2000
    perturb_rate = 0.03  # 3% of points
    max_perturbations = int(n_points * perturb_rate)
    
    def compute_C5(h_vec):
        # Convert to correct size and compute correlation
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_neighbors(h_vec, num_neighbors=5):
        # Generate neighbors by perturbing a few points
        neighbors = []
        for _ in range(num_neighbors):
            new_h = h_vec.copy()
            # Pick random indices to perturb
            num_perturb = np.random.randint(1, max_perturbations + 1)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                # Add small random perturbation
                new_h[idx] += np.random.normal(0, 0.05)
            # Ensure feasibility
            new_h = project_feasible(new_h)
            neighbors.append(new_h)
        return neighbors
    
    # Main tabu search loop
    iter_count = 0
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Generate neighbors
        neighbors = get_neighbors(h, num_neighbors=5)
        
        best_neighbor = None
        best_neighbor_C5 = float('inf')
        
        # Evaluate neighbors
        for neighbor in neighbors:
            neighbor_C5 = compute_C5(neighbor)
            # Check if it's in tabu list
            neighbor_tuple = tuple(neighbor.round(10).tolist())  # Round for comparison
            if neighbor_tuple not in tabu_list:
                if neighbor_C5 < best_neighbor_C5:
                    best_neighbor = neighbor
                    best_neighbor_C5 = neighbor_C5
        
        # Apply best neighbor if found
        if best_neighbor is not None:
            # Update tabu list
            h_tuple = tuple(h.round(10).tolist())
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(h_tuple)
            
            h = best_neighbor
            
            # Update global best
            if best_neighbor_C5 < best_C5:
                best_C5 = best_neighbor_C5
                best_h = h.copy()
        else:
            # No improving neighbor found, perturb randomly
            idx = np.random.choice(n_points)
            h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
            h = project_feasible(h)
            
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
