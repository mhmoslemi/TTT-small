import numpy as np
import time
from scipy.signal import correlate

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate_h(h, dx, mutation_strength=0.1):
    """Apply small random mutations to h while maintaining feasibility."""
    mutated = h.copy()
    idx = np.random.randint(len(mutated))
    mutated[idx] += np.random.normal(0, mutation_strength)
    return project_feasible(mutated, dx)

def crossover_h(h1, h2):
    """Perform uniform crossover between two solutions."""
    child = np.where(np.random.rand(len(h1)) < 0.5, h1, h2)
    return child

def select_parents(population, fitnesses, num_parents=20):
    """Select parents using tournament selection."""
    selected = []
    for _ in range(num_parents):
        tournament_indices = np.random.choice(len(population), size=3, replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner])
    return selected

        h = mutate_h(initial_h, dx)
        population.append(h)

            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
            
        # Selection
        parents = select_parents(population, fitnesses, num_parents)
        
        # Create new population
        new_population = [best_h.copy()]
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover_h(p1, p2)
            if np.random.rand() < mutation_rate:
                child = mutate_h(child, dx)
            new_population.append(child)
            
        population = new_population
    
    return best_h, best_C
