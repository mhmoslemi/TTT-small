import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def initialize_h(n_points, initial_h):
    if initial_h is not None:
        return initial_h
    # Create a smooth initial waveform
    t = np.linspace(0, 2, n_points)
    h = 0.5 + 0.3 * np.sin(np.pi * t) * np.cos(0.5 * np.pi * t)
    h = np.clip(h, 0, 1)
    return project_feasible(h, dx=2.0 / n_points)
