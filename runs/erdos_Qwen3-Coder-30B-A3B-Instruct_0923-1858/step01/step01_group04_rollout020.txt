import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    # Step 1: Low resolution optimization (n=50)
    n_low = 50
    if n_points > n_low:
        # Downsample the initial solution
        h_low = np.interp(np.linspace(0, 1, n_low), np.linspace(0, 1, n_points), h)
    else:
        h_low = h[:n_low]
    
    # Normalize to sum = n_low/2
    h_low = h_low * (n_low / 2) / np.sum(h_low)
    h_low = np.clip(h_low, 0, 1)
    
    # Define the objective function to minimize
    def objective_low(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Upsample to full resolution
        h_full = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_low), h_vec)
        h_full = h_full * (n_points / 2) / np.sum(h_full)
        return evaluate_erdos_solution(h_full)
    
    # Create bounds for DE
    bounds = [(0, 1) for _ in range(n_low)]
    
    # Differential evolution at low resolution
    try:
        res = differential_evolution(
            objective_low,
            bounds,
            maxiter=100,
            popsize=40,
            mutation=0.8,
            recombination=0.9,
            seed=seed,
            strategy='best1bin'
        )
        
        if res.success:
            h_candidate = np.clip(res.x, 0, 1)
            # Upsample and refine
            h_upsampled = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_low), h_candidate)
            h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
            h_upsampled = np.clip(h_upsampled, 0, 1)
            
            # Local refinement with SLSQP
            def objective_refine(h_vec):
                return evaluate_erdos_solution(h_vec)
            
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            bounds_refine = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_integral}
            
            try:
                res_refine = minimize(
                    objective_refine,
                    h_upsampled,
                    method='SLSQP',
                    bounds=bounds_refine,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}

            except Exception:

        pass
    
    # Final validation and return
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) != n_points / 2:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
