import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Genetic Algorithm parameters
    pop_size = 50
    num_generations = 300
    crossover_rate = 0.5
    mutation_rate = 0.03
    tournament_size = 5
    
    # Create initial population
    population = []

            h_var = np.clip(h_var, 0, 1)
            h_var = h_var / np.sum(h_var) * (n_points / 2.0)
            population.append(h_var)
        # Fill the rest with random solutions
        for i in range(pop_size // 2, pop_size):
            h_rand = np.random.rand(n_points)

            population.append(h_rand)
    else:
        # All random solutions
        for i in range(pop_size):

            h_rand = np.clip(h_rand, 0, 1)
            population.append(h_rand)

        sorted_indices = np.argsort(fitness)

        if no_improvement_count > max_no_improvement:
            break

        new_population = [population[0].copy()]  # Keep the best individual (elitism)
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_select(population, fitness, tournament_size)
            parent2 = tournament_select(population, fitness, tournament_size)

                child1, child2 = crossover(parent1, parent2, n_points)

            child1 = mutate(child1, mutation_rate, n_points)

            # Add offspring to new population

    return (best_h.tolist(), final_c5, n_points)

def tournament_select(population, fitness, tournament_size):
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx]

def crossover(parent1, parent2, n_points):
    # Uniform crossover with slight modification
    mask = np.random.rand(n_points) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    
    # Ensure sum constraint by rescaling
    child1 = child1 / np.sum(child1) * (n_points / 2.0)
    child2 = child2 / np.sum(child2) * (n_points / 2.0)
    
    # Clip to [0,1] range
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    
    return child1, child2

def mutate(individual, mutation_rate, n_points):
    # Gaussian mutation
    mutated = individual.copy()
    num_mutation = int(n_points * mutation_rate)
    mutation_indices = np.random.choice(n_points, num_mutation, replace=False)
    
    for i in mutation_indices:
        mutated[i] += np.random.normal(0, 0.03)
    
    # Clip to [0,1] range
    mutated = np.clip(mutated, 0, 1)
    
    # Ensure sum constraint by rescaling
    mutated = mutated / np.sum(mutated) * (n_points / 2.0)
    
    return mutated
