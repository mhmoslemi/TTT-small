import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set initial points
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to target resolution if needed
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    # Gradient-based optimization with projected updates
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters
    max_iter = 10000
    alpha = 0.1
    decay = 0.95
    
    def project(h):
        """Project h to feasible region"""
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Estimate gradient direction more carefully
    def estimate_gradient(h):
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode="full")
        # Find max correlation index
        max_idx = np.argmax(corr)
        # Initialize gradient 
        grad = np.zeros_like(h)
        # Add gradient contribution from max overlap
        for i in range(len(h)):
            # This is the computation of gradient of the max corr
            # approximated via finite differences
            k = max_idx - len(h) + 1 + i
            if 0 <= k < len(h):
                grad[i] += (1 - 2 * h[i]) * dx  # Simplified gradient

        return grad
    
    start_time = time.time()
    iter_count = 0
    last_improvement = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count
            
        h_prev = h.copy()
        
        # Estimate gradient
        grad = estimate_gradient(h)
        
        # Update h
        h = h - alpha * grad
        h = project(h)
        # Perturb a little if no improvement recently
        if iter_count - last_improvement > 100:
            noise = np.random.uniform(-0.05, 0.05, n_points)
            h += noise
            h = project(h)
            no_improvement_count += 1
            if no_improvement_count > 200:
                alpha = 0.1  # Restart learning rate
                no_improvement_count = 0
        else:
            # Decaying learning rate
            alpha *= decay

    # Return the best result found
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
