import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    
    # Population-based genetic algorithm
    population_size = 100
    num_generations = 500
    mutation_rate = 0.2
    tournament_size = 5
    crossover_rate = 0.8
    
    # Use initial_h_values if available
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        initial_h = np.maximum(0, 1 - np.abs(x - 1))
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Create initial population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Helper functions
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def tournament_selection(population, fitnesses, tournament_size):
        selected_idx = np.random.choice(len(population), tournament_size)
        best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
        return population[best_idx].copy()
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = np.random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(h_vec, mutation_rate):
        h_vec = h_vec.copy()
        for i in range(len(h_vec)):
            if np.random.rand() < mutation_rate:
                # Add small Gaussian noise
                h_vec[i] += np.random.normal(0, 0.05)
        return h_vec
    
    # Main GA loop
    best_h = initial_h.copy()
    best_C5 = compute_C5(best_h)
    
    for generation in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate fitness for whole population
        fitnesses = [compute_C5(h) for h in population]
        
        # Track best individual in current population
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_C5:
            best_C5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(population[best_idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Project to feasible region
            child1 = project_feasible(child1)
            child2 = project_feasible(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
