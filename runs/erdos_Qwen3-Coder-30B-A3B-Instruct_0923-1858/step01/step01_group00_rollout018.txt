import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure it's length 100 by interpolation or padding/truncation
        h_new = np.zeros(n_points)
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h_new = np.interp(x_new, x_old, h, left=0, right=0)
        h = h_new
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Use a more sophisticated optimization approach
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Bounds for h: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()

    # Start timer
    start_time = time.time()
    
    # Try multiple strategies and initializations to escape local minima
    initial_strategies = [
        lambda: h.copy(),
        lambda: np.random.rand(n_points) * 0.5,
        lambda: np.ones(n_points) * 0.5,
        lambda: np.concatenate([np.linspace(0,1,n_points//2), np.linspace(1,0,n_points//2)])
    ]
    # Try a few initializations with more diversity
    for i, init_func in enumerate(initial_strategies):
        if time.time() - start_time > budget_s - 10:
            break
        h = init_func()
        h = h / np.sum(h) * (n_points / 2)
        # Perform local optimization with multiple solvers
        methods = ['SLSQP', 'L-BFGS-B']
        for method in methods:
            if time.time() - start_time > budget_s - 10:
                break
            try:
                result = minimize(
                    objective,
                    h.copy(),
                    method=method,
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 500},
                    tol=1e-6
                )
                if result.success:
                    current_c5 = evaluate_erdos_solution(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()
                        h = result.x.copy()
                else:
                    # Fallback to a new guess if optimization failed
                    h = np.random.rand(n_points) * 0.5
                    h = h / np.sum(h) * (n_points / 2)
            except Exception:
                # If anything goes wrong, try with a new initial guess
                h = np.random.rand(n_points) * 0.5
                h = h / np.sum(h) * (n_points / 2)
    
    # Final refinement using more iterations
    h_refine = best_h.copy()
    try:
        result = minimize(
            objective,
            h_refine.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 1000},
            tol=1e-8
        )
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    except Exception:
        pass
    
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    actual_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, actual_c5, n_points)
