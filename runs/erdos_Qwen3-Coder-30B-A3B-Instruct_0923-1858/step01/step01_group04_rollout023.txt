import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def upsample_and_rescale(h_coarse, n_fine):
    """Upsample coarse solution to fine resolution and rescale"""
    n_coarse = len(h_coarse)
    # Linear interpolation
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    # Rescale to meet integral constraint
    integral_fine = np.sum(h_fine) * 2.0 / n_fine
    h_fine = h_fine * (n_fine / 2) / integral_fine
    return np.clip(h_fine, 0, 1)

        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Initial best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use a multi-resolution approach
    # Level 1: coarse resolution for global exploration
    n_coarse = 50
    n_fine = 100
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_coarse)]
    
    def objective_coarse(h_coarse):
        h_fine = upsample_and_rescale(h_coarse, n_fine)
        return evaluate_erdos_solution(h_fine)
    
    # Differential evolution for global search
    de_result = differential_evolution(
        objective_coarse,
        bounds,
        maxiter=50,
        popsize=40,
        mutation=0.8,
        recombination=0.9,
        seed=seed,
        callback=lambda x, convergence: None,
        polish=False
    )
    
    # If successful, refine
    if de_result.success:
        h_candidate_coarse = de_result.x
        h_candidate_fine = upsample_and_rescale(h_candidate_coarse, n_fine)
        
        # Local refinement with SLSQP
        def objective_fine(h_fine):
            return evaluate_erdos_solution(h_fine)
        
        # Constraint for integral = 1
        def constraint_integral(h_fine):
            return np.sum(h_fine) - n_fine / 2
        
        bounds_fine = [(0, 1) for _ in range(n_fine)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        try:
            res = minimize(
                objective_fine,

                method='SLSQP',

            )
            
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                c5_refined = evaluate_erdos_solution(h_refined)
                if c5_refined < best_c5:

                    best_c5 = c5_refined
        except Exception:

    # Try another random start with local optimization
    if time.time() - start_time < budget_s - 1:
