import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_step(h, dx, learning_rate=0.01):
    """Perform a single optimization step."""
    n = len(h)
    grad = np.zeros(n)
    epsilon = 1e-5
    
    # Compute finite differences for gradient
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project(h_minus, dx)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)
    
    # Update
    h_new = h - learning_rate * grad
    h_new = project(h_new, dx)
    
    return h_new
