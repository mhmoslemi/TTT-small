import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()

        if n_points > len(h):

                end_idx = int((i + 1) * new_size / len(old_h))
