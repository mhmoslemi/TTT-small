import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        # Convert to correct size
        h_vec = np.array(h_vec)
        # Note: correlate with mode='full' to handle all shifts
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum over all shifts
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def objective(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return -np.max(corr) * dx  # Negative because we minimize
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def gradient(h_vec):
        # Finite difference approximation of gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
        
        return grad
    
    # Main optimization loop
    patience = 0
    max_patience = 100
    iter_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Optimization step using scipy
        constraints = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try different algorithms
        try:
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, 
                              options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
            if result.success:
                h = result.x
                h = project_feasible(h)
        except:
            pass
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Random perturbation
            # Perturb 10% of points
            indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for idx in indices:
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
            h = project_feasible(h)
            patience = 0
            
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
