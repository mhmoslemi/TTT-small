import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1):
    """Apply random mutation with given rate."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, 0.05)
    return project(mutated, dx)

def crossover(p1, p2):
    """Uniform crossover between two parents."""
    child = np.array(p1)
    for i in range(len(child)):
        if np.random.rand() < 0.5:
            child[i] = p2[i]
    return child

def select_parent(population, fitnesses):
    """Tournament selection."""
    k = 3
    indices = np.random.choice(len(population), k, replace=False)
    best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
    return population[best_idx]

def initialize_population(n_points, population_size=50):
    """Initialize population with variation around initial_h."""

    except NameError:
        # Fallback to triangular waveform
        base = np.linspace(0, 1, n_points//2, endpoint=False)
        base = np.concatenate([base, np.linspace(1, 0, n_points - n_points//2)])
    
    population = []
    for _ in range(population_size):
        h = base.copy()
        h = mutate(h, 2.0/n_points, rate=0.05)
        population.append(h)
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points)
    
    start_time = time.time()
    best_h = None
    best_C = float('inf')
    
    # Evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 5:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c = evaluate(h, dx)
            fitnesses.append(1.0 / (c + 1e-10))  # Avoid division by zero
            
        # Track best
        current_best_idx = np.argmax(fitnesses)
        if fitnesses[current_best_idx] > 1.0 / best_C:
            best_C = evaluate(population[current_best_idx], dx)
            best_h = population[current_best_idx].copy()
        
        # Selection, crossover, mutation
        new_population = []
        for _ in range(len(population) // 2):
            p1 = select_parent(population, fitnesses)
            p2 = select_parent(population, fitnesses)
            child1 = crossover(p1, p2)
            child2 = crossover(p1, p2)
            child1 = mutate(child1, dx, rate=0.1)
            child2 = mutate(child2, dx, rate=0.1)
            new_population.extend([child1, child2])
            
        population = new_population[:len(population)]
        generation += 1
        
    if best_h is None:
        # Fallback: select best from final generation
        final_fitnesses = [1.0 / (evaluate(h, dx) + 1e-10) for h in population]
        best_idx = np.argmax(final_fitnesses)
        best_h = population[best_idx].copy()
        best_C = evaluate(best_h, dx)
    
    return (best_h, best_C, n_points)
