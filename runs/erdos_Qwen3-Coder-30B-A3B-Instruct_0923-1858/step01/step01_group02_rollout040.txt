import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)

    current_energy = evaluate_erdos_solution(h)

    T = 1.0  # Initial temperature

    # Iteration counter
    max_iter = 100000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:

        # Perturb h: move mass between two random indices
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        
        # Ensure we don't violate bounds
        new_h = h.copy()
        if new_h[i1] + delta >= 0 and new_h[i1] + delta <= 1 and new_h[i2] - delta >= 0 and new_h[i2] - delta <= 1:

            new_h[i2] -= delta

            sum_h = np.sum(new_h)

            new_energy = evaluate_erdos_solution(new_h)
            
            # Accept or reject based on Metropolis criterion
            if new_energy < current_energy or np.random.rand() < np.exp((current_energy - new_energy) / T):

                current_energy = new_energy
                
                # Update best solution if applicable
                if current_energy < best_energy:
                    best_h = h.copy()
                    best_energy = current_energy
                    
        # Cooling schedule
        T *= cooling_rate
        if T < min_temp:
            T = min_temp
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
