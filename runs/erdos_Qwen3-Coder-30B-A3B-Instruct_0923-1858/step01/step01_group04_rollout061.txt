import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialization done
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use CVXPY with barrier method
    # Start with a simpler method approaching the optimal using constrained optimization
    
    max_iter = 5
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Use a second-order cone method to improve quality 
        # Formulate as QP with log barrier
        
        # Try to improve by local optimization using scipy with different initializations
        for _ in range(3):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Small randomization for diversity
            perturbation = np.random.normal(0, 0.05, n_points)
            h_trial = best_h + perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # Use SLSQP again with improved starting point
            def objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                return evaluate_erdos_solution(h_vec)
            
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_integral}
            
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                        
            except Exception:
                pass
        
        # Add a small perturbation to escape local minima
        perturbation = np.random.normal(0, 0.01, n_points)
        best_h += perturbation
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        iter_count += 1
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
