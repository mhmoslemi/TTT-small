import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_frequency_domain(h_init, dx, n_points, budget_s):
    """Optimize using convex optimization in frequency domain."""
    t0 = time.time()
    
    # Convert to frequency domain
    H = cp.Variable(n_points, complex=True)
    
    # Real-valued constraint: h is real, so H[0] is real, and H[-k] = conj(H[k])
    # For now, solve as unconstrained complex, and project later
    
    # Objective
    # Correlation in frequency domain: H * (1 - conj(H)) 
    # We want: max_k |H[k] * (1 - conj(H[k]))|
    # This is not directly convex in terms of H, so approach differently
    
    # Use a different convex approach: relax constraints
    # Minimize the maximum absolute value of correlation
    # For now, approximate with a simpler problem
    
    # Create a random initialization for H
    h0 = h_init.copy()
    H0 = np.fft.fft(h0)
    
    # Manually create a least-squares problem approach - 
    # Not perfectly convex but stable
    
    # Let's directly optimize a projected gradient approach, but with a better
    # idea: start with the initial h and apply a fast non-convex optimization
    
    # Use the dual of the max constraint
    # min_C s.t. max_k | int h(x) (1 - h(x+k)) dx | <= C
    # Can be treated as min_C s.t. each |int h(x) (1 - h(x+k)) dx| <= C
    
    # Instead, do it with alternating projection and gradient descent
    # But use 1D gradient descent in function space

    h_current = h_init.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current, dx)
    
    # Step size schedule and max steps
    max_iter = 10000
    lr = 0.01
    decay_rate = 0.995
    for i in range(max_iter):
        if time.time() - t0 > budget_s - 10:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_feasible(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project_feasible(h_minus, dx)
            
            c_plus = evaluate_c5(h_plus, dx)
            c_minus = evaluate_c5(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h_current = h_current - lr * grad
        h_current = project_feasible(h_current, dx)
        
        # Annealing
        lr *= decay_rate
        
        # Check for better solution
        current_c5 = evaluate_c5(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Occasionally perturb to escape local minima
        if i % 100 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h_current += pert
            h_current = project_feasible(h_current, dx)
    
    return best_h, best_c5
