import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

    start_time = time.time()

        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h[i] = 0.6
            else:
                h[i] = 0.4
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Initial energy
    current_energy = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_energy = current_energy
    
    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-8

    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and temp > min_temp:
        # Generate a candidate solution by perturbing h
        candidate_h = h.copy()
        # Select two random indices to exchange mass
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        # Amount to move
        move_amount = np.random.uniform(0.01, 0.1)
        
        # Ensure we do not violate bounds
        if candidate_h[idx1] >= move_amount and candidate_h[idx2] + move_amount <= 1.0:
            candidate_h[idx1] -= move_amount
            candidate_h[idx2] += move_amount
        elif candidate_h[idx2] >= move_amount and candidate_h[idx1] + move_amount <= 1.0:

        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        candidate_h = np.clip(candidate_h, 0, 1)

        candidate_energy = evaluate_erdos_solution(candidate_h)
        
        # Accept with Metropolis criterion
        if candidate_energy < current_energy or np.random.rand() < np.exp(-(candidate_energy - current_energy) / temp):
            h = candidate_h
            current_energy = candidate_energy
            if current_energy < best_energy:
                best_h = h.copy()
                best_energy = current_energy
        
        # Cooling schedule
        temp *= cooling_rate
        iter_count += 1
    
    # Final evaluation
    final_energy = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_energy), n_points)
