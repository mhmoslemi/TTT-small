import numpy as np
from scipy.optimize import minimize
import time
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Try several local optimizations from different random starts
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    max_iter = 200
    iter_count = 0
    
    # Use a more sophisticated approach: hybrid optimization
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Start with perturbed version of current best solution
        h_trial = best_h.copy()
        
        # Add noise for exploration
        noise_scale = 0.05
        perturbation = np.random.normal(0, noise_scale, n_points)
        h_trial = h_trial + perturbation
        
        # Ensure bounds and integral constraint
        h_trial = np.clip(h_trial, 0, 1)

            methods = ['SLSQP', 'L-BFGS-B']

                if time.time() - start_time >= budget_s - 1:

                    h_trial,
                    method=method,
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                        break  # Found an improvement, try next iteration with new best

            pass

        # Occasionally do a global search by restarting with a random feasible solution

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
