import numpy as np
from scipy.optimize import minimize
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Local optimization with different strategies
    strategies = ['SLSQP', 'L-BFGS-B', 'TNC']
    max_iter = 10
    iter_count = 0
    
    # Use a fixed number of local optimizations with different methods
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Random perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h_trial = best_h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Try different local optimization methods
        for method in strategies:
            if time.time() - start_time > budget_s - 1:
                break
            try:
                res = minimize(
                    lambda x: evaluate_erdos_solution(x),
                    h_trial,
                    method=method,
                    bounds=[(0, 1) for _ in range(n_points)],
                    options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                pass
                
        iter_count += 1
        
    # Phase 2: Global exploration with simulated annealing if time allows
    if time.time() - start_time < budget_s - 5:
        # Start from current best
        h_current = best_h.copy()
        c5_current = best_c5
        temp = 1.0
        cooling_rate = 0.995
        min_temp = 1e-6
        
        while time.time() - start_time < budget_s - 1 and temp > min_temp:
            # Generate neighbor by perturbing
            perturbation = np.random.normal(0, 0.05, n_points)
            h_neighbor = h_current + perturbation
            h_neighbor = np.clip(h_neighbor, 0, 1)
            h_neighbor = h_neighbor * (n_points / 2) / np.sum(h_neighbor)
            
            c5_neighbor = evaluate_erdos_solution(h_neighbor)
            
            # Accept or reject
            if c5_neighbor < c5_current or np.random.rand() < np.exp((c5_current - c5_neighbor) / temp):
                h_current = h_neighbor
                c5_current = c5_neighbor
                
            # Update best if improved
            if c5_current < best_c5:
                best_h = h_current
                best_c5 = c5_current
                
            temp *= cooling_rate
            
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
