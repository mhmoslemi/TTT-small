import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    cooling_rate = 0.9995
    min_T = 1e-6
    max_iter = 100000
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Temperature schedule
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and T > min_T:
        # Generate a new candidate by transferring mass between two points
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Amount of mass to move (small random value)
        delta = np.random.uniform(0.001, 0.05)
        delta = np.clip(delta, 0, min(h[i], 1 - h[j]))
        
        # Create candidate solution
        h_candidate = h.copy()
        h_candidate[i] -= delta
        h_candidate[j] += delta
        
        # Ensure constraints
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Renormalize to maintain sum = n_points / 2
        current_sum = np.sum(h_candidate)
        if abs(current_sum - n_points / 2) > 1e-10:
            scale_factor = (n_points / 2) / current_sum
            h_candidate *= scale_factor
        
        # Compute energy (C5 value) for candidate
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Accept or reject based on Metropolis criterion
        delta_E = c5_candidate - best_c5

            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
        
        # Decrease temperature
        T *= cooling_rate
        iter_count += 1
        
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final evaluation and trimming
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
