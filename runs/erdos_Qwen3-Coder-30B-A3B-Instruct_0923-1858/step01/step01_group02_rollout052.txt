import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, norm as cp_norm
import cvxpy as cp

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Create a more carefully constructed starting point
        h[:n_points//2] = 0.7
        h[n_points//2:] = 0.3
        # Normalize to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Set up bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints via CVXPY for feasibility checking
    def make_feasible(x):
        x = np.clip(x, 0, 1)
        # Normalize to sum = n_points / 2 using CVXPY
        h_var = Variable(n_points)
        prob = Problem(Minimize(cp_norm(h_var - x, 2)), 
                       [cp_sum(h_var) == n_points/2,
                        h_var >= 0,
                        h_var <= 1])
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return np.array(h_var.value).flatten()
        else:
            return x  # fallback
            
    # L-BFGS-B optimization
    def objective(x):
        return evaluate_erdos_solution(x)
    
    # Adaptive perturbation schedule
    perturbation_magnitude = 0.1
    no_improvement_count = 0
    improvement_threshold = 10  # after 10 iterations without improvement, decrease perturbation
    
    counter = 0
    while time.time() - start_time < budget_s - 1:
        counter += 1
        
        # Random perturbation of the solution
        perturbation = np.random.normal(0, perturbation_magnitude, n_points)
        x0 = best_h + perturbation
        x0 = np.clip(x0, 0, 1)
        # Ensure sum constraint is met
        x0 = x0 * (n_points / 2) / np.sum(x0)
        
        # Try local optimization with L-BFGS
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 100})
            if res.success:
                # Update if better solution found
                current_c5 = evaluate_erdos_solution(res.x)
                if current_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = current_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                # Fallback if optimization fails, use the feasible perturbation
                current_c5 = evaluate_erdos_solution(x0)
                if current_c5 < best_c5:
                    best_h = x0.copy()
                    best_c5 = current_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        except Exception:
            # Continuation in case of error
            current_c5 = evaluate_erdos_solution(x0)
            if current_c5 < best_c5:
                best_h = x0.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1

        # Adjust perturbation based on improvement:
        if no_improvement_count > improvement_threshold:
            perturbation_magnitude *= 0.9
            no_improvement_count = 0
        elif no_improvement_count < 5:
            perturbation_magnitude = min(perturbation_magnitude * 1.05, 0.2)  # Prevent over-perturbing
        
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
