import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Use a more sophisticated optimization approach: sequential quadratic programming with global search
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    max_iter = 15
    iter_count = 0
    
    # Keep track of time
    remaining_time = budget_s - 1
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try a few different local optimizations to avoid local minima
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Slightly perturb the solution
            perturb_factor = 0.1
            h_trial = h + np.random.normal(0, perturb_factor, n_points) * h
            h_trial = np.clip(h_trial, 0, 1)
            
            # Scale to maintain sum constraint
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # Define the objective function to minimize
            def objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                return evaluate_erdos_solution(h_vec)
            
            # Constraint for integral = 1
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            # Bounds for each variable
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Define constraints
            constraints = {'type': 'eq', 'fun': constraint_integral}
            
            try:
                # Use L-BFGS-B which is good for bound constrained problems
                res = minimize(
                    objective,
                    h_trial,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-7}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                        
            except Exception:
                pass
                
        iter_count += 1
        
        # Gradually decrease perturbation to fine-tune
        if iter_count % 3 == 0:
            perturb_factor *= 0.8
        
        # Early stopping if improvement is small
        if iter_count > 5 and best_c5 < 0.381:
            break

    # Final validation and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
