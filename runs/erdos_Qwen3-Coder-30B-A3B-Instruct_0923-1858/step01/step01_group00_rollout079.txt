import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better precision
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Rescale to n_points
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = project_feasible(h, n_points)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points)

    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    inertia_weight = 0.7
    cognitive_param = 1.5
    social_param = 1.5
    perturb_std = 0.01
    
    # Initialize swarm
    particles = np.array([h.copy() for _ in range(swarm_size)])
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    # Evaluate initial swarm
    fitness = np.array([evaluate_erdos_solution(p) for p in particles])
    
    # Personal best
    personal_best = particles.copy()
    personal_best_fitness = fitness.copy()
    
    # Global best
    global_best_idx = np.argmin(fitness)
    global_best = particles[global_best_idx].copy()
    global_best_fitness = fitness[global_best_idx]
    
    # Main PSO loop
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity with inertia and cognitive/social components
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = cognitive_param * r1 * (personal_best[i] - particles[i])
            social = social_param * r2 * (global_best - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Apply low-pass filter (with alpha=0.9)
            if iteration > 0:
                velocities[i] = 0.9 * velocities[i] + 0.1 * np.random.normal(0, 0.01, n_points)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply the projection to keep feasible
            particles[i] = project_feasible(particles[i], n_points)
            
            # Evaluate
            current_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = current_fitness
                
                # Update global best if needed
                if current_fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = current_fitness
        
        # Apply perturbation to all particles
        noise = np.random.normal(0, perturb_std, (swarm_size, n_points))
        particles += noise
        
        iteration += 1

    # Final evaluation of global best to ensure it's correct
    final_c5 = evaluate_erdos_solution(global_best)
    
    # Final projection in case of numerical errors
    final_h = project_feasible(global_best, n_points)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, n_points)
