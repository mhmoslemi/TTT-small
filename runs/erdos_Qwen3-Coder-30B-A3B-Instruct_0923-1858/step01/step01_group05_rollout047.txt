import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Generate triangular function
        n_points = 70
        h = 1 - np.abs(2 * np.arange(n_points) / n_points - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    max_iter = 5000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h with gaussian noise in critical region
        h_trial = h.copy()
        # Focus noise in the critical region where overlaps are most likely
        crit_indices = np.where((np.arange(n_points) / n_points >= 0.3) & 
                               (np.arange(n_points) / n_points <= 0.7))[0]
        noise = np.random.normal(0, 0.03, n_points)
        noise[crit_indices] = noise[crit_indices] * 2  # Scale up noise in critical region
        h_trial += noise
        h_trial = np.clip(h_trial, 0, 1)
        # Maintain constraint sum(h) = n_points / 2
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Try scipy optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            # Compute correlation with (1-h)
            corr = np.correlate(h, 1-h, mode="full")
            # Return maximum correlation scaled by dx
            return np.max(corr) * (2.0 / n_points)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        iter_count += 1
        
        # Global exploration step
        if iter_count % 100 == 0:
            # Take a step away from current best to escape local minima
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2.0)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new.copy()
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
