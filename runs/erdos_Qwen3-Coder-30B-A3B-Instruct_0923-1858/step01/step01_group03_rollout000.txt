import time
from scipy.signal import correlate

        h_vars = cp.Variable(n_points)
        
        # Define constraints
        # 0 <= h[i] <= 1
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # sum(h) = n_points / 2
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Helper function to compute overlap at shift k
        def compute_overlap(h_vec, k):
            if k >= 0:
                return cp.sum(cp.multiply(h_vec[:n_points-k], 1-h_vec[k:]))
            else:
                k = -k
                return cp.sum(cp.multiply(h_vec[k:], 1-h_vec[:n_points-k]))
        
        # For all shifts k, add constraint that overlap >= lambda
        # Just approximating few key shifts
        shifts = list(range(-n_points+1, n_points))
        lambda_var = cp.Variable()
        for shift in shifts[::2]:  # Use every 2nd shift for efficiency
            if shift >= 0:
                overlap = cp.sum(cp.multiply(h_vars[:n_points-shift], 1-h_vars[shift:]))
            else:
                shift_val = -shift
                overlap = cp.sum(cp.multiply(h_vars[shift_val:], 1-h_vars[:n_points-shift_val]))
            constraints.append(overlap >= lambda_var)
        
        # Objective: maximize lambda
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve convex problem
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZABLE:
                new_h = h_vars.value
                # Project to feasible region
                new_h = np.clip(new_h, 0, 1)
                current_sum = np.sum(new_h)
                if current_sum > 0:
                    new_h = new_h * (n_points / 2) / current_sum
                return new_h
            else:
                # If optimization fails, return initial
                return h_init
        except:
            return h_init

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)

            h_vec = h_vec * (n_points / 2) / current_sum

                h = project_feasible(h)
                patience = 0

        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h) - 1)
        
        # Compute gradient w.r.t. max k
        grad = compute_gradient(h, max_k)
        h = h - eta * grad

        # Decay learning rate

            best_h = h_restart.copy()
    
    # Final deterministic correction
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
