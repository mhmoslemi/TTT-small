import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        # Place values in a way that attempts to minimize overlap
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 0.2
            elif i % 3 == 1:
                h[i] = 0.8
            else:

        h = h * (n_points / 2) / np.sum(h)
    
    # Bounds and constraints

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simulated Annealing parameters
    T = 1.0  # Initial Temperature
    cooling_rate = 0.9995

    max_iter = max(1000, int(budget_s * 500))
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Perturb h: randomly select two indices and slightly move mass between them
        i1, i2 = np.random.randint(0, n_points, 2)
        
        # Ensure we don't pick the same index
        while i1 == i2:
            i2 = np.random.randint(0, n_points)
            
        # Amount of mass to transfer
        delta = np.random.uniform(0.001, 0.05)
        
        # Compute new h values
        h_new = h.copy()
        h_new[i1] = np.clip(h[i1] - delta, 0, 1)

        # Renormalize to ensure sum constraint (maintaining sum = n_points/2)
        current_sum = np.sum(h_new)
        scale_factor = (n_points / 2) / current_sum

        # Compute new energy (C5)
        current_c5 = evaluate_erdos_solution(h)
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Metropolis criterion for acceptance
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
        # Decrease temperature
        T = max(T * cooling_rate, min_T)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), float(final_c5), n_points)
