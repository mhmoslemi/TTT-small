import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            best_n = n_points
    
    # Multi-resolution approach
    resolutions = [50, 100]  # Start coarse, then refine
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Generate initial population for DE
        def constraint_integral(x):
            return np.sum(x) - n_points / 2
            
        # Use DE for global search
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            return evaluate_erdos_solution(h_vec)
            
        # Run for limited time
        try:
            # Use DE with smaller population for speed
            de_result = differential_evolution(
                objective,
                bounds,
                seed=seed,
                maxiter=200,
                popsize=20,
                mutation=(0.5, 1.0),
                recombination=0.7,
                atol=1e-6,
                tol=1e-6
            )
            
            if de_result.success:
                h_opt = np.clip(de_result.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                candidate_c5 = evaluate_erdos_solution(h_opt)
                
                if candidate_c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = candidate_c5
                    best_n = n_points
                    
        except Exception:
            pass
        
        # Local improvement with SLSQP if we have a good candidate
        if best_h is not None and time.time() - start_time < budget_s - 5:
            try:
                # Further local optimization
                def local_objective(h_vec):
                    h_vec = np.clip(h_vec, 0, 1)
                    return evaluate_erdos_solution(h_vec)
                    
                cons = {'type': 'eq', 'fun': constraint_integral}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Orthogonal perturbation for local search
                h_pert = best_h + np.random.normal(0, 0.01, n_points)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                
                res = minimize(
                    local_objective,
                    h_pert,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 300, 'ftol': 1e-8}
                )
                
                if res.success:
                    refined_h = np.clip(res.x, 0, 1)
                    refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                    refined_c5 = evaluate_erdos_solution(refined_h)
                    
                    if refined_c5 < best_c5:
                        best_h = refined_h.copy()
                        best_c5 = refined_c5
                        best_n = n_points

    # Final check

            h = np.clip(h, 0, 1)
            best_h = h

            # Fallback

            best_h = h
            best_c5 = evaluate_erdos_solution(best_h)
            best_n = n_points
    
    return (list(best_h), best_c5, best_n)
