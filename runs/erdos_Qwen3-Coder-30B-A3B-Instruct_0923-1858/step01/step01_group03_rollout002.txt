import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation approach
    def solve_convex_relaxation(h_start):
        # Matrix formulation for constraints
        # Each constraint is sum h[i] * (1 - h[i+k]) >= lambda
        # We assume equal weights for now and solve a simpler optimization
        h_var = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            lambda_var <= 0  # Will be maximized
        ]
        
        # Add a few representative constraints (not all)
        # Use a sampled subset for efficiency
        sample_k = np.linspace(0, n_points-1, min(20, n_points), dtype=int)
        for k in sample_k:
            # Compute indices for this k-shift
            h_shifted = np.roll(h_var, k)
            constraint_expr = cp.sum(cp.multiply(h_var, 1 - h_shifted))
            constraints.append(constraint_expr >= lambda_var)
        
        # Objective
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_var.value
        except:
            pass
        return h_start
    
    # Optimization with gradient descent & projection
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Step 1: Convex relaxation
    h_opt = solve_convex_relaxation(h)
    if h_opt is not None:
        h = h_opt
    
    # Step 2: Gradient-based refinement from convex solution
    eta = 0.01
    patience = 0
    max_patience = 100
    iter_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.8:  # Leave margin for final steps
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
    
        # Gradient update
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Decay learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1
    
    # Step 3: Random restarts for better global exploration
    for restart in range(10):
        if time.time() - start_time >= budget_s * 0.9:
            break
            
        # Random restart
        h_restart = np.random.rand(n_points)
        h_restart = project_feasible(h_restart)
        
        # Gradient-based refinement of this restart
        eta = 0.01
        patience = 0
        max_patience = 100
        
        for step in range(200):
            current_C5 = compute_C5(h_restart)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_restart.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
            
            grad = get_gradient(h_restart)
            h_restart = h_restart - eta * grad
            h_restart = project_feasible(h_restart)
            
            if step % 50 == 0:
                eta *= 0.95
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
