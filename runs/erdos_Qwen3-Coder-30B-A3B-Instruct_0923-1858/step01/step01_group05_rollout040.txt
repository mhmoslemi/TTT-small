import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Create a symmetric triangular function as initial guess
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 10000
    iter_count = 0
    idle_count = 0
    min_improvement = 1e-6
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create a perturbed version in critical region
        h_trial = h.copy()
        # Focus perturbations on the central region where overlap is likely highest
        critical_indices = np.where((np.linspace(0, 2, n_points) >= 0.3) & 
                                   (np.linspace(0, 2, n_points) <= 0.7))[0]
        
        # Add noise to critical region
        noise = np.random.normal(0, 0.03, len(critical_indices))
        h_trial[critical_indices] += noise
        
        # Clip and rescale
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Try local optimization with SLSQP
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
                    idle_count = 0
                else:
                    idle_count += 1
            else:
                idle_count += 1
        except:
            idle_count += 1
            
        # Periodic full mutation to escape local minima
        if iter_count % 100 == 0 and idle_count > 10:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
            idle_count = 0
            
        iter_count += 1

        if idle_count > 50:
            # Perform a more aggressive restart with better initialization
            x = np.linspace(0, 2, n_points)

            idle_count = 0
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
