from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, sum, power, 0, 1
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

    h = np.clip(h, 0, 1)
