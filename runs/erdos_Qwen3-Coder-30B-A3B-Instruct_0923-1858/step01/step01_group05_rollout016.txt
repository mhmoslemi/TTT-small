import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def compute_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def perturb_and_repair(h, sigma=0.03):
    # Add Gaussian noise in the middle region where overlap is typically high
    n = len(h)
    h_new = h.copy()
    x_vals = np.linspace(0, 2, n)
    mask = (x_vals >= 0.3) & (x_vals <= 0.7)
    h_new[mask] += np.random.normal(0, sigma, np.sum(mask))
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (35.0 / np.sum(h_new))
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a smooth symmetric triangular function
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1.0 - np.abs(x - 1.0)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    h = h * (35.0 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Perturb and repair
        h_trial = perturb_and_repair(h, sigma=0.03)
        
        # Local refinement using SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - 35.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            corr = np.correlate(h, 1-h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        # Occasionally perturb even without successful local optimization
        if iter_count % 10 == 0:
            h = perturb_and_repair(h, sigma=0.01)
            h = np.clip(h, 0, 1)
            h = h * (35.0 / np.sum(h))
        
        iter_count += 1
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
