import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution with 45 points
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Set up convex optimization problem
    def solve_convex_relaxation():
        # Variables
        h_vars = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # For each lag, add constraint sum h[i] * (1 - h[i+k]) >= lambda
        for k in range(n_points):
            # Compute the correlation with lag k (with wraparound)
            # For simplicity: we'll handle it manually
            lhs = 0
            for i in range(n_points):
                j = (i + k) % n_points
                lhs += h_vars[i] * (1 - h_vars[j])
            constraints.append(lhs >= lambda_var)
        
        # Objective
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_vars.value
        except:
            pass
        return None
        
    # Try convex relaxation
    if n_points <= 50:  # Only for smaller problems
        convex_solution = solve_convex_relaxation()
        if convex_solution is not None:
            h = np.clip(convex_solution, 0, 1)
            # Rescale to sum to n_points / 2
            h = h * (n_points / 2) / np.sum(h)
    
    # Refinement with gradient descent
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Gradient approximation (very crude but effective)
    def get_gradient(h_vec):
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        current_C5 = compute_C5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        return grad
    
    # Local optimization
    eta = 0.01
    patience = 0
    max_patience = 100
    
    # Refine for majority of the budget
    remaining_budget = budget_s * 0.7
    budget_end = time.time() + remaining_budget
    
    while time.time() < budget_end:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
    
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        eta = max(eta * 0.999, 1e-5)
        if patience < max_patience:
            patience += 1
    
    # Random restarts for final 10% of time
    remaining_budget = budget_s * 0.1
    budget_end = time.time() + remaining_budget
    
    for _ in range(50):
        if time.time() >= budget_end:
            break
        # Generate random feasible h
        h_restart = np.random.rand(n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        current_C5 = compute_C5(h_restart)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_restart.copy()
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
