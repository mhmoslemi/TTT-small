import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Ensure h satisfies the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Control iterations based on time budget
    max_iterations = 1000
    iteration = 0
    
    while iteration < max_iterations and (time.time() - start_time < budget_s - 1):
        iteration += 1
        # Compute current overlaps
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # Optimization approach using cutting plane method
        try:
            # Number of variables
            h_vars = Variable(n_points)
            C = Variable(1)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1]
            constraints.append(sum(h_vars) == n_points / 2)
            
            # Adding cutting planes
            for k in range(2*n_points - 1):
                # Compute gradient of correlation
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j = (i + k) % n_points
                    grad[i] = dx * (1 - h[j])
                
                # Add the cutting plane constraint: grad^T * h <= C - grad^T * h + O_k
                constraints.append(grad.T @ h_vars <= C - grad.T @ h + overlaps[k])
            
            # Add constraint that C is lower bound for overlaps 
            for k in range(2*n_points - 1):
                constraints.append(C >= overlaps[k])
            
            # Objective: minimize C
            objective = Minimize(C)
            prob = Problem(objective, constraints)
            
            # Solve
            prob.solve(solver=CPLEX, verbose=False, timelimit=budget_s/500)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = h_vars.value
                new_c5 = min(overlaps)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except Exception:
            pass
            
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
