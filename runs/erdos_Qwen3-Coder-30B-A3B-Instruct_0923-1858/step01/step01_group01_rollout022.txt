import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_with_gradient_descent(initial_h, dx, n_points, budget_s):
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Hyperparameters
    lr = 0.01
    decay_rate = 0.99
    max_time = budget_s * 0.95  # Leave some headroom
    start_time = time.time()
    
    for i in range(100000):
        if time.time() - start_time > max_time:
            break
            
        # Compute gradient estimate using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        h = h - lr * grad
        
        # Project to constraints
        h = project(h, dx)
        
        # Occasionally perturb to avoid local minima
        if i % 100 == 0 and np.random.rand() < 0.1:
            for _ in range(3):
                idx = np.random.randint(0, len(h))
                h[idx] += np.random.normal(0, 0.005)
            h = project(h, dx)
        
        # Check solution quality
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
            lr = max(0.001, lr * decay_rate)  # Adaptive learning rate
            
    return best_h, best_C
