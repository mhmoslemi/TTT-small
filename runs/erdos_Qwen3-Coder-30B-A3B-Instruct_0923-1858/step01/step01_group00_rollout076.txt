import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

        h = np.ones(n_points) * 0.5
        # Make it piecewise constant with some variation to avoid trivial solutions
        # Divide into 4 segments with alternating high/low values
        seg_len = n_points // 4
        for i in range(4):
            start, end = i * seg_len, (i + 1) * seg_len
            if i % 2 == 0:
                h[start:end] = 0.7
            else:
                h[start:end] = 0.3
        h = h / np.sum(h) * (n_points / 2)
        
    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    inertia = 0.7
    c1 = c2 = 1.5
    perturb_std = 0.01
    alpha = 0.9  # low-pass filter coefficient
    
    # Initialize swarm
    particles = np.array([h.copy() for _ in range(swarm_size)])
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    # Evaluate initial fitness
    fitness = np.array([evaluate_erdos_solution(p) for p in particles])
    
    # Personal best
    personal_best = particles.copy()
    personal_best_fitness = fitness.copy()
    
    # Global best
    gb_idx = np.argmin(fitness)
    global_best = particles[gb_idx].copy()
    global_best_fitness = fitness[gb_idx]
    
    # Main PSO loop
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        for i in range(swarm_size):
            # Update velocity with inertia, cognitive, and social terms
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = c1 * r1 * (personal_best[i] - particles[i])
            social = c2 * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Apply low-pass filter
            velocities[i] = alpha * velocities[i] + (1 - alpha) * np.roll(velocities[i], 1)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce feasibility
            particles[i] = np.clip(particles[i], 0, 1)
            # Re-project to correct sum
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
            
            # Evaluate new fitness
            new_fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if new_fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = new_fitness
            
            # Update global best
            if new_fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = new_fitness

        for i in range(swarm_size):

            particles[i] += noise
            particles[i] = np.clip(particles[i], 0, 1)

    # Final evaluation and return

    final_c5 = evaluate_erdos_solution(global_best)
    
    return (global_best, final_c5, n_points)
