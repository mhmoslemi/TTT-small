    # Genetic Algorithm Parameters
    population_size = 100

    generations = 500
    elite_size = 10
    
    # Initialize population
    def create_individual():
        # Create random individual
        individual = np.random.rand(n_points)
        # Clip to [0, 1]
        individual = np.clip(individual, 0, 1)
        # Rescale to meet constraint
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (n_points / 2) / current_sum
        return individual
    
    # Create initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def tournament_selection(pop, fitnesses, k):
        indices = np.random.choice(len(pop), size=k, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx]

        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child

        # Add Gaussian noise to selected genes
        mask = np.random.rand(n_points) < rate
        noise = np.random.normal(0, 0.05, n_points)
        individual = individual + mask * noise
        # Project back to feasible region
        individual = project_feasible(individual)
        return individual
