import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    sigma = 0.1
    stagnation_count = 0
    max_stagnation = 100
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        # Generate perturbation
        delta_h = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_new = h + delta_h
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate new solution
        try:
            c5_new = evaluate_erdos_solution(h_new)
        except:
            c5_new = float('inf')
        
        # Acceptance criterion
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            # Metropolis criterion
            if np.random.rand() < np.exp(-(c5_new - best_c5) / T):
                h = h_new
            else:
                # Shake occasionally to escape local minima
                if stagnation_count > max_stagnation:
                    h = best_h + np.random.normal(0, T * sigma, n_points)
                    h = np.clip(h, 0, 1)
                    h = h / np.sum(h) * (n_points / 2)
                    stagnation_count = 0
        
        # Cool down temperature
        T *= alpha
        
        # Occasionally reinitialize temperature if stagnation is too high
        if stagnation_count > 200:
            T = max(0.001, T * 0.9)
            stagnation_count = 0
    
    # Final evaluation
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except:
        final_c5 = best_c5
    
    return (best_h, final_c5, n_points)
