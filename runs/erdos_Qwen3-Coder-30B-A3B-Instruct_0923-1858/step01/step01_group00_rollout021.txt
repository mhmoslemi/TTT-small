import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def objective(h):
        return evaluate_erdos_solution(h)

    bounds = [(0, 1) for _ in range(n_points)]

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraint = {'type': 'eq', 'fun': constraint_sum}

    # Use a more robust multi-start strategy
    initial_hs = [h.copy()]
    for _ in range(5):
        perturbed = h + np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
        initial_hs.append(perturbed)

    # Try different optimizers and initial guesses
    optimizers = ['SLSQP', 'L-BFGS-B', 'TNC']
    for opt in optimizers:
        for i, h_start in enumerate(initial_hs):
            if time.time() - start_time > budget_s - 5:
                break
            try:
                result = minimize(
                    objective,
                    h_start,
                    method=opt,
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 300, 'ftol': 1e-6},
                    tol=1e-6
                )
                if result.success:
                    current_c5 = evaluate_erdos_solution(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()
            except Exception:
                continue

    # Local refinement with different optimizers
    if time.time() - start_time < budget_s - 5:
        for opt in ['Nelder-Mead', 'Powell']:
            try:
                result = minimize(
                    objective,
                    best_h.copy(),
                    method=opt,
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 200}
                )
                if result.success:
                    current_c5 = evaluate_erdos_solution(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()
            except Exception:
                continue

    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)

    return (best_h, actual_c5, n_points)
