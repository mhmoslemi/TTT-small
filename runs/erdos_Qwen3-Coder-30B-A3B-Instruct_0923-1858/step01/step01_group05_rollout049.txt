import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create triangular initial guess
        n_points = 70
        h = np.array([1.0 - abs(2.0 * i / n_points - 1.0) for i in range(n_points)])
        h = h / np.sum(h) * (n_points / 2.0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h in critical region (0.3 to 0.7)
        h_trial = h.copy()
        dx = 2.0 / n_points
        # Convert to index range: 0.3 to 0.7 maps to approximately 21 to 49
        start_idx = int(0.3 * n_points)
        end_idx = int(0.7 * n_points)
        noise = np.random.normal(0, 0.03, end_idx - start_idx)
        h_trial[start_idx:end_idx] += noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Try local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_vec):
            corr = np.correlate(h_vec, 1-h_vec, mode="full")
            return np.max(corr) * dx
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        iter_count += 1
        
        # Periodic update
        if iter_count % 20 == 0:
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
