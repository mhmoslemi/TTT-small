import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular initial guess: symmetric around center
        n_points = 70
        h = np.array([1.0 - abs(2.0*i/(n_points-1) - 1.0) for i in range(n_points)])
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define the objective function for optimization
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    iter_count = 0
    max_iter = 2000
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Global search: perturb with noise in critical region
        h_trial = h.copy()
        
        # Select indices in [0.3, 0.7) region of [0,2] interval
        # i from 0 to n_points-1 represents [0,2], so x = 2*i/n_points
        # We want 0.3 <= x <= 0.7 
        # i should satisfy 0.3 <= 2*i/n_points <= 0.7
        # So 0.15*n_points <= i <= 0.35*n_points
        
        start_idx = max(0, int(0.15 * n_points))
        end_idx = min(n_points, int(0.35 * n_points) + 1)
        
        # Add moderate Gaussian noise to those indices only
        if end_idx > start_idx:
            noise = np.random.normal(0, 0.03, end_idx - start_idx)
            h_trial[start_idx:end_idx] += noise
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            
        # Local optimization using SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_opt = res.x
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    h = h_opt.copy()
        except:
            pass
            
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
