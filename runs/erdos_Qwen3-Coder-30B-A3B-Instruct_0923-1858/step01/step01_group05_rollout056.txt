import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a random feasible h vector
        n_points = 140
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
        
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define objective function
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx

    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Genetic Algorithm Parameters
    population_size = 50
    num_parents = population_size // 5
    crossover_rate = 0.5
    mutation_rate = 0.1
    mutation_strength = 0.03
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_trial = h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        population.append(h_trial)
        
    # Function to enforce constraints
    def enforce_constraints(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2.0)
        return h
    
    # Genetic Algorithm Loop
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < 1500:
        # Evaluate fitness
        fitness = [objective(individual) for individual in population]
        sorted_indices = np.argsort(fitness)
        
        # Update best solution
        if fitness[sorted_indices[0]] < best_c5:
            best_c5 = fitness[sorted_indices[0]]
            best_h = population[sorted_indices[0]].copy()
            last_improvement = generation
        
        # Early stopping if no improvement for 150 iterations
        if generation - last_improvement > 150:
            break
            
        # Select parents (top 20%)
        parents = [population[i] for i in sorted_indices[:num_parents]]
        
        # Create offspring
        new_population = parents.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(parents) >= 2:
                # Select two parents
                p1, p2 = np.random.choice(parents, 2, replace=False)
                
                # Crossover: blend with uniform probability
                mask = np.random.rand(n_points) < 0.5
                child1 = p1.copy()
                child2 = p2.copy()
                child1[mask] = p2[mask]
                child2[mask] = p1[mask]
                
                # Mutate children
                if np.random.rand() < mutation_rate:
                    mut_indices = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
                    child1[mut_indices] += np.random.normal(0, mutation_strength, len(mut_indices))
                
                if np.random.rand() < mutation_rate:
                    mut_indices = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
                    child2[mut_indices] += np.random.normal(0, mutation_strength, len(mut_indices))
                
                # Enforce constraints
                new_population.append(enforce_constraints(child1))
                if len(new_population) < population_size:
                    new_population.append(enforce_constraints(child2))
            else:
                # Random mutation
                individual = np.random.choice(parents)
                mut_indices = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
                mutated = individual.copy()
                mutated[mut_indices] += np.random.normal(0, mutation_strength, len(mut_indices))
                new_population.append(enforce_constraints(mutated))
                
        population = new_population
        generation += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
