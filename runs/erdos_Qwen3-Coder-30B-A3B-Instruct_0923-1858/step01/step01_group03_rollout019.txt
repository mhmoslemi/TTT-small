import time

        # Normalize to sum to n_points / 2
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    pop_size = 100
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Add random solutions
    for i in range(pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Store personal bests
    pbest_positions = [swarm[i].copy() for i in range(pop_size)]
    pbest_fitness = [float('inf')] * pop_size
    
    # Global best
    gbest_idx = 0
    gbest_fitness = float('inf')
    gbest_position = None
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Main PSO loop
    iter_count = 0
    stagnation_counter = 0
    best_C5 = float('inf')
    best_h = h_initial.copy()
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Update global best
        for i in range(pop_size):
            fitness = compute_C5(swarm[i])
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest_positions[i] = swarm[i].copy()
            
            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest_position = swarm[i].copy()
                stagnation_counter = 0
        
        # Check if we found a better global best
        if gbest_fitness < best_C5:
            best_C5 = gbest_fitness
            best_h = gbest_position.copy()
        
        # Update velocities and positions
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(pop_size):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - swarm[i]) + 
                           c2 * r2 * (gbest_position - swarm[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
        
        iter_count += 1
        
        # Reset stagnation counter if improvement
        stagnation_counter += 1
        if stagnation_counter > 50:
            # Randomly reset some particles
            reset_count = max(1, pop_size // 10)
            for _ in range(reset_count):
                idx = np.random.randint(pop_size)
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.05, 0.05, n_points)
            stagnation_counter = 0
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, len(best_h))
