import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        # Create a triangular function as starting point
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * (n_points / 2.0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization with time tracking
    max_iter = 10000
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Perturb in critical region
        h_trial = h.copy()
        # Focus perturbation in [0.3, 0.7] region where most overlap concentrated
        idxs = np.where((np.linspace(0, 2, n_points) >= 0.3) & 
                        (np.linspace(0, 2, n_points) <= 0.7))[0]
        noise = np.random.normal(0, 0.03, len(idxs))
        h_trial[idxs] += noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Optimize with SLSQP
        try:
            res = minimize(evaluate_erdos_solution, h_trial, method='SLSQP', 
                          bounds=bounds, constraints=cons, options={'maxiter': 20})
            if res.success:
                h_new = res.x
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
                    last_improvement = iter_count
        except:
            pass
        
        iter_count += 1

        # Occasionally do more exploratory search in feasible region
        if iter_count % 25 == 0:
            h_trial = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            try:
                res = minimize(evaluate_erdos_solution, h_trial, method='SLSQP',
                              bounds=bounds, constraints=cons, options={'maxiter': 10})
                if res.success:
                    h_new = res.x
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
