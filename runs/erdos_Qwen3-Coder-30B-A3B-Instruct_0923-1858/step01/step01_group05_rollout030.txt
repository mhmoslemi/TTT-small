import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Genetic algorithm parameters
    pop_size = 50
    num_parents = pop_size // 5
    crossover_rate = 0.5
    mutation_std = 0.03
    max_stagnation = 150
    stagnation_count = 0
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i < pop_size // 2 and 'initial_h_values' in globals():
            h_ind = h + np.random.normal(0, 0.01, n_points)
        else:
            h_ind = np.random.rand(n_points)
        h_ind = np.clip(h_ind, 0, 1)
        h_ind = h_ind / np.sum(h_ind) * (n_points / 2.0)
        population.append(h_ind)
    
    best_history = []
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h_ind in population:
            c5 = evaluate_erdos_solution(h_ind)
            fitness.append(c5)
        
        # Track best in this generation
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        best_history.append(current_best_c5)
        
        if len(best_history) > max_stagnation:
            if all(x >= best_history[-1] for x in best_history[-max_stagnation:]):
                stagnation_count += 1
            else:
                stagnation_count = 0
        else:
            stagnation_count = 0
            
        if stagnation_count > max_stagnation:
            break
            
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5
            stagnation_count = 0
        
        # Selection: Tournament
        parents = []
        for _ in range(num_parents):
            idxs = np.random.choice(pop_size, size=5, replace=False)
            best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
            parents.append(population[best_idx])
        
        # Create new population
        new_population = parents.copy()
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(parents) >= 2:
                # Crossover
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
            else:
                # Mutation or copy
                parent = np.random.choice(parents)
                child = parent + np.random.normal(0, mutation_std, n_points)
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2.0)
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
