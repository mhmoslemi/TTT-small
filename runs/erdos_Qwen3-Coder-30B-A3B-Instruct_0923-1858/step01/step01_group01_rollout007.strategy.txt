### Gradient Computation
```python
def compute_gradient(h, dx):
    # Central finite difference with small epsilon
    eps = 1e-8
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        corr_plus = np.correlate(project_feasible(h_plus, dx), 1 - project_feasible(h_plus, dx), 'full')
        corr_minus = np.correlate(project_feasible(h_minus, dx), 1 - project_feasible(h_minus, dx), 'full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad
```

```python
def project_feasible(h, dx):
    h = np.clip(h, 0, 1)
    target_sum = 1.0 / dx  # ∑h = n_points / 2
    if np.isclose(np.sum(h), target_sum, atol=1e-6):
        return h
    simplex_scale = target_sum / np.sum(h)
    return h * simplex_scale
```

```python
def evaluate(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
```
