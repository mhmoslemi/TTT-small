import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        # Create a triangular initial solution
        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    perturb_rate = 0.04  # 4% of points
    max_iter = 2000
    
    def compute_C5(h_vec):
        # Convert to correct size
        h_vec = np.array(h_vec)
        # Note: correlate with mode='full' to handle all shifts
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum over all shifts
        return np.max(corr) * dx

    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def generate_candidate(h_vec, tabu_list):
        # Make a copy to avoid modifying original
        candidate = h_vec.copy()
        
        # Select indices to perturb
        n_perturb = max(1, int(perturb_rate * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Perturb values
        for idx in indices:
            # Add small random change
            change = np.random.normal(0, 0.05)
            candidate[idx] += change
            # Ensure it stays within bounds after change
            candidate[idx] = np.clip(candidate[idx], 0, 1)
        
        # Ensure feasibility
        candidate = project_feasible(candidate)
        
        # Check if this candidate is in the tabu list
        candidate_tuple = tuple(candidate.round(6))
        if candidate_tuple in tabu_list:
            # If it's tabu, try a different perturbation
            for _ in range(10):  # Try up to 10 times
                idx = np.random.randint(0, n_points)
                change = np.random.normal(0, 0.05)
                candidate[idx] += change
                candidate[idx] = np.clip(candidate[idx], 0, 1)
                candidate = project_feasible(candidate)
                candidate_tuple = tuple(candidate.round(6))
                if candidate_tuple not in tabu_list:
                    break
            
        return candidate
    
    # Main tabu search loop
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Generate candidate by perturbing current solution
        candidate = generate_candidate(h, tabu_list)
        
        # Compute C5 for the candidate
        candidate_C5 = compute_C5(candidate)
        
        # Check if it improves best solution
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate.copy()
            
        # Update current solution if candidate is better
        if candidate_C5 < compute_C5(h):
            h = candidate.copy()
        
        # Add to tabu list
        h_tuple = tuple(h.round(6))
        if h_tuple not in tabu_list:
            tabu_list.append(h_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)  # Remove oldest entry
        
        iter_count += 1
        if iter_count >= max_iter:
            break
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
