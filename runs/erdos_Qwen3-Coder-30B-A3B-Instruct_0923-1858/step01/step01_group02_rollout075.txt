import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Add some variation to decrease overlap
        h[0] = 0.75
        h[1] = 0.25
        h = h * (n_points / 2) / np.sum(h)
        
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Create bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimize with L-BFGS-B using scipy
    opt_kwargs = {
        'method': 'L-BFGS-B',
        'bounds': bounds,
        'options': {'maxiter': 100}
    }
    
    # Try multiple restarts to avoid local optima
    iter_count = 0
    last_improvement = 0
    perturbation_magnitude = 0.1
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        if iter_count % 100 == 0 and iter_count > 0:
            # Reduce perturbation if we're improving consistently
            if iter_count - last_improvement > 200:
                perturbation_magnitude *= 0.9
            else:
                last_improvement = iter_count
            if stagnation_count > 5:
                perturbation_magnitude *= 1.2  # Increase perturbation if stuck
                stagnation_count = 0
                
        # Slightly perturb current best to escape local minima
        perturbation = np.random.normal(0, perturbation_magnitude, n_points)
        x0 = np.clip(best_h + perturbation, 0, 1)
        # Normalize to correct sum
        x0 = x0 * (n_points / 2) / np.sum(x0)
        
        try:
            # Vanilla optimization
            res = minimize(objective, x0, **opt_kwargs)
            
            if res.success:
                current_c5 = evaluate_erdos_solution(res.x)
                if current_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = current_c5
                    last_improvement = iter_count
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        except Exception:
            stagnation_count += 1
            
        iter_count += 1
        
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final evaluation and trimming
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure final solution satisfies constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
