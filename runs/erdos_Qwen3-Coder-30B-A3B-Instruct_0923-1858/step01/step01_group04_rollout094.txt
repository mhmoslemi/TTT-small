import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Genetic Algorithm approach
    pop_size = 20
    max_generations = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    
    # Generate initial population
    population = []
    for _ in range(pop_size):
        h_trial = h + np.random.normal(0, 0.05, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        population.append(h_trial)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(ind) for ind in population]
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Create new population
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(pop_size//2, 2, replace=False)
            parent1 = elite[idx1]
            parent2 = elite[idx2] if np.random.random() < crossover_prob else elite[np.random.randint(0, len(elite))]
            
            # Crossover
            if np.random.random() < crossover_prob:
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutation_idx = np.random.randint(0, n_points)
                child[mutation_idx] += np.random.normal(0, 0.05)
                child[mutation_idx] = np.clip(child[mutation_idx], 0, 1)
            
            # Rescale to maintain integral constraint
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        fitness = [evaluate_erdos_solution(ind) for ind in population]
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    # Phase 2: Refinement with SLSQP
    # Try to refine the best solution
    h_refined = best_h.copy()
    
    # Define objective function
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)
    
    # Integral constraint
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # SLSQP optimization from the best solution found
    try:
        res = minimize(
            objective,
            h_refined,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        if res.success:
            h_candidate = np.clip(res.x, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c5_candidate = evaluate_erdos_solution(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Check if we should try increasing resolution
    if time.time() - start_time < budget_s - 5 and n_points == 70:
        # Try with higher resolution
        h_upsampled = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, 70), best_h)
        h_upsampled = h_upsampled * 50 / np.sum(h_upsampled)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        
        # Further refinement on the upsampled version
        try:
            res = minimize(
                objective,
                h_upsampled,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(100)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - 50},
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * 50 / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < final_c5:
                    final_c5 = c5_candidate
                    best_h = h_candidate
        except Exception:
            pass
    
    return (list(best_h), final_c5, len(best_h))
