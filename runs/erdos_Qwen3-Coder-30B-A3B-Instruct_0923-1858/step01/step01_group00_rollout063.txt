import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for better resolution
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Rescale to match n_points using linear interpolation
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_old, h)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    sigma = 0.1
    max_stagnation = 100
    stagnation_count = 0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Generate perturbation
        delta_h = np.random.uniform(-T * sigma, T * sigma, n_points)
        
        # Update candidate
        h_candidate = h + delta_h
        
        # Project to feasibility
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate
        try:
            c5_candidate = evaluate_erdos_solution(h_candidate)
        except:
            # If evaluation fails, skip this candidate
            continue
            
        # Accept or reject
        delta_E = c5_candidate - best_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Adaptive shaking if stuck
        if stagnation_count >= max_stagnation:
            h += np.random.normal(0, T * sigma, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            stagnation_count = 0
        
        # Cool temperature
        T *= alpha
        
        # Break early if time is almost up
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final evaluation of best
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except:
        final_c5 = best_c5
    
    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, final_c5, n_points)
