import numpy as np
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)

    h = np.clip(h, 0, 1)
    
    def accept_step(h_current, h_candidate, energy_current, energy_candidate, T):
        if energy_candidate < energy_current:
            return True
        else:
            p = np.exp(-(energy_candidate - energy_current) / T)
            return np.random.rand() < p

    def perturb_h(h, T):
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        # Transfer mass between indices
        d = np.random.rand() * T * 0.1
        d = min(d, h_new[i], 1 - h_new[j])
        h_new[i] -= d
        h_new[j] += d
        return h_new
    
    # Simulated Annealing
    T = 1.0  # Initial temperature
    cooling_rate = 0.999
    min_T = 1e-8
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    while time.time() - start_time < budget_s - 1:
        h_candidate = perturb_h(best_h, T)

        h_candidate = np.clip(h_candidate, 0, 1)

        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)

    final_c5 = evaluate_erdos_solution(best_h)
