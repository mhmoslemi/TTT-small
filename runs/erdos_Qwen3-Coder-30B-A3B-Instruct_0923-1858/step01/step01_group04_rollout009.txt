import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 50
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Low resolution optimization
    if n_points < 100:
        # Try to improve with slightly higher resolution
        n_points = 100
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(h)), h)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
        h = h_upsampled
    
    # Phase 2: Refinement with local optimization
    max_iter = 20
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # SLSQP optimization from current best
        # Define the objective function to minimize
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)
        
        # Constraint for integral = 1
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Define constraints
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
            
        # Random perturbation with some probability
        if np.random.random() < 0.3:
            perturbation = np.random.normal(0, 0.03, n_points)
            h_trial = best_h + perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial)
            
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
        
        iter_count += 1
    
    # Final validation and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
