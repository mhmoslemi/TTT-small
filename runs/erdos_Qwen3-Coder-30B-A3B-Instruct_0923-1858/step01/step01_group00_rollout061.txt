import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_old, h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize to sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    sigma = 0.1
    shake_threshold = 100
    shake_count = 0
    
    def project_feasible(h_vec):
        """Project h to feasible region: [0,1] and sum=1"""
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec / np.sum(h_vec) * (n_points / 2)
        return h_vec

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Start timer
    start_time = time.time()
    
    # Annealing loop
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Generate perturbation
        delta = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_candidate = h + delta
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate)
        
        # Evaluate
        try:
            c5_candidate = evaluate_erdos_solution(h_candidate)
        except:
            c5_candidate = np.inf
        
        c5_current = evaluate_erdos_solution(h)
        
        # Accept or reject
        if c5_candidate < c5_current or np.random.rand() < np.exp(-(c5_candidate - c5_current) / T):
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                shake_count = 0
            else:
                shake_count += 1
        else:
            shake_count += 1
            
        # Shake if no improvement for a while
        if shake_count > shake_threshold:
            h = project_feasible(h + np.random.normal(0, T * sigma, n_points))
            shake_count = 0
            
        # Cool down temperature
        T = max(1e-8, alpha * T)
    
    # Final evaluation
    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5
        
    return (best_h, actual_c5, n_points)
