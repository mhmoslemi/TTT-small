import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)

    best_c5 = evaluate_erdos_solution(best_h)

    n_stage1 = 50

            corr = np.correlate(h_stage1, 1 - h_stage1, mode='full')
            max_k = np.argmax(corr)
            # Find which shift is worst and try to improve it
            # max_k corresponds to shift from -n+1 to n-1
            shift = max_k - (n_stage1 - 1)
            
            # Now try changes to reduce this overlap
            # Create a small modification in the direction that reduces overlap
            h_trial = h_stage1.copy()
            if shift > 0:
                # shift positive: h[i] correlated with 1-h[i+shift]

                        delta = min(0.01, h_trial[i], 1 - h_trial[i + shift])
                        if delta > 0:
                            h_trial[i] -= delta
                            h_trial[i + shift] += delta
            else:

                for i in range(-shift, n_stage1):

    final_c5 = evaluate_erdos_solution(best_h)
