import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    n_points = 45
    
    # Use initial_h_values if available
    try:
        h_initial = np.array(initial_h_values)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    except NameError:
        # Create a triangular initial solution
        h_initial = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h_initial[i] = max(0, 1 - abs(x[i] - 1))
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best solutions
    pbest = [h.copy() for h in swarm]
    pbest_fitness = [float('inf')] * Pop_size
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Evaluate initial swarm
    for i in range(Pop_size):
        h = swarm[i]
        overlaps = correlate(h, 1 - h, mode="full")
        fitness = np.max(overlaps) * (2.0 / n_points)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    # Main PSO loop
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 50
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Update particles
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            inertia = w * velocities[i]
            cognitive = c1 * r1 * (pbest[i] - swarm[i])
            social = c2 * r2 * (swarm[gbest_idx] - swarm[i])
            velocities[i] = inertia + cognitive + social
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate fitness
            overlaps = correlate(swarm[i], 1 - swarm[i], mode="full")
            fitness = np.max(overlaps) * (2.0 / n_points)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
        # Periodic reset if no improvement
        if stagnation_count > max_stagnation:
            # Reset 10% of particles to random solutions
            reset_count = max(1, Pop_size // 10)
            for _ in range(reset_count):
                idx = np.random.randint(0, Pop_size)
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                overlaps = correlate(swarm[idx], 1 - swarm[idx], mode="full")
                fitness = np.max(overlaps) * (2.0 / n_points)
                pbest[idx] = swarm[idx].copy()
                pbest_fitness[idx] = fitness
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = idx
            stagnation_count = 0
            
        iter_count += 1
        
        # Decay inertia weight
        if iter_count % 10 == 0 and iter_count > 0:
            w = max(0.3, w * 0.99)
    
    # Return best solution found
    return (swarm[gbest_idx].tolist(), gbest_fitness, n_points)
