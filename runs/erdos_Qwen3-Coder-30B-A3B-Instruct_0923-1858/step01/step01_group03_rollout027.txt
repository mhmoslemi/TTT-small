import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Convex relaxation approach to get initial solution
    def solve_convex_relaxation():
        h_cvx = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        # Constraints for all lags k from -n+1 to n-1
        constraints = []
        
        # Add constraints: sum h[i] * (1 - h[i+k]) >= lambda for all k
        for k in range(-n_points + 1, n_points):
            # Ready indices for wrap-around
            if k >= 0:
                lhs = cp.sum(cp.multiply(h_cvx[:n_points-k], 1 - h_cvx[k:]))
            else:
                lhs = cp.sum(cp.multiply(h_cvx[-k:], 1 - h_cvx[:n_points+k]))
            
            constraints.append(lhs >= lambda_var)
        
        constraints.append(cp.sum(h_cvx) == n_points / 2)
        constraints.extend([h_cvx[i] >= 0 for i in range(n_points)])
        constraints.extend([h_cvx[i] <= 1 for i in range(n_points)])
        
        prob = cp.Problem(cp.Maximize(lambda_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.array(h_cvx.value)
        except:
            return None
    
    # Try convex relaxation first
    if budget_s > 200:
        conv_start = time.time()
        cvx_h = solve_convex_relaxation()
        if cvx_h is not None:
            h = project_feasible(cvx_h)
        conv_elapsed = time.time() - conv_start
        remaining = budget_s - conv_elapsed
        if remaining < 50:
            # Remove convex portion from remaining for local optimization
            pass
    else:
        remaining = budget_s
    
    # Local optimization with gradient descent
    eta = 0.01
    max_iter = int(remaining * 10) if remaining < 200 else 5000
    patience = 0
    max_patience = 50
    last_improvement = 0
    
    # Initialize local searches from different starting points
    def local_search(h0):
        h_local = h0.copy()
        for iter_count in range(max_iter):
            elapsed = time.time() - start_time
            if elapsed >= budget_s * 0.95:
                break
                
            # Compute gradient for max_k integral
            # Simplified: compute gradient numerically w.r.t. each h[i]
            grad = np.zeros(n_points)
            eps = 1e-5
            
            current_C5 = compute_C5(h_local)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_local.copy()
                last_improvement = iter_count
                
            # Compute estimated gradient of max_k term using finite differences
            corr = correlate(h_local, 1 - h_local, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (n_points - 1)
            
            for i in range(n_points):
                eps_vec = np.zeros(n_points)
                eps_vec[i] = eps
                h_plus = project_feasible(h_local + eps_vec)
                h_minus = project_feasible(h_local - eps_vec)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update step
            h_local = h_local - eta * grad
            h_local = project_feasible(h_local)
            
            # Adaptive learning rate
            if iter_count % 100 == 0 and iter_count > 0:
                eta *= 0.95
                
            # Early stopping
            if iter_count - last_improvement > max_patience:
                # Random perturbation
                idx = np.random.choice(n_points)
                h_local[idx] = np.clip(h_local[idx] + np.random.normal(0, 0.01), 0, 1)
                h_local = project_feasible(h_local)
                
                last_improvement = iter_count
                
        return h_local
    
    # Try various starting points
    for i in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_start = h.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.02, n_points)
        h_start += noise
        h_start = project_feasible(h_start)
        h_opt = local_search(h_start)
        C5_opt = compute_C5(h_opt)
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h_opt.copy()
    
    # Final run one more time with best result
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
