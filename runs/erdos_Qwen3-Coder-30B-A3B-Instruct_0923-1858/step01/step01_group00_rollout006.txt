import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters
    alpha = 0.1
    max_iter = 10000
    patience = 100
    no_improve_count = 0
    step = 0
    
    start_time = time.time()
    
    def project(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def get_gradient(h):
        # compute correlation of h with (1-h)
        corr = correlate(h, 1 - h, mode="full")
        # Find the maximum lag
        max_k = np.argmax(corr)
        # approximate gradient using the sense that C5 is at lag max_k
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # only considering one specific lag of correlative product
            if i + max_k < len(h):
                grad[i] += 1.0
            if max_k - i >= 0:
                grad[i] += -1.0
        return grad * (2.0 / n_points)
    
    # Optimize
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # gradient step
        grad = get_gradient(h)
        h = h - alpha * grad
        h = project(h)
        
        # perturbation if no improvement
        if no_improve_count > patience:
            # add noise
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = h + noise
            # median_smooth
            h = np.convolve(h, np.ones(3)/3, mode='same')
            h = project(h)
            no_improve_count = 0
            alpha *= 0.95
        
        step += 1
    
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except:
        final_c5 = best_c5
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
