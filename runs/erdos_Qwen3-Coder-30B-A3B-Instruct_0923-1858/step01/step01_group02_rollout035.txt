import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Simple construction with some variation
        h[0] = 0.75
        h[1] = 0.25
        h[2] = 0.6
        h[3] = 0.4
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Initialize C5 bound
    C5_bound = 0.4
    best_C5 = float('inf')
    best_h = h.copy()
    dx = 2.0 / n_points

    # Adaptive adjustment parameters
    step_size = 0.0001
    increase_factor = 2.0
    decrease_factor = 1.0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Build constraints for each k
        constraints = []
        # Set sum constraint: sum(h) * dx = 1 
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Build constraints for all shifts k
        # We just need to check valid overlaps from the correlation
        # For more efficient optimization, we take every 5th k to reduce size
        k_values = range(0, n_points*2 - 1, 5) if n_points > 20 else range(0, n_points*2 - 1)
        
        # Build correlation constraints
        for k in k_values:
            # Compute indices for h[i] and h[i+k] (wrap around)
            # In correlation notation, h[i] and 1-h[i+k] are multiplied
            
            # Left-aligned sums for correlation of h with 1-h
            # i from 0 to min(n_points-1, n_points-1-k)
            # We consider corrupted or shortened correlation for optimization
            sum_expr = 0
            for i in range(min(n_points, n_points-k)):
                sum_expr += h[i] * (1 - h[i+k])
            
            # Add constraint for this shift
            constraints.append(sum_expr <= C5_bound / dx)
            
        # Formulate optimization problem
        try:
            # Set up the problem
            problem = cp.Problem(cp.Minimize(0), constraints)
            
            # Try different solvers
            solver = cp.SCS
            problem.solve(solver=solver, verbose=False)
            
            if problem.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
                h_val = h.value
                if h_val is not None:
                    h_val = np.array(h_val)
                    h_val = np.clip(h_val, 0, 1)
                    h = h_val
                    
                    # Check current solution
                    temp_C5 = evaluate_erdos_solution(h)
                    if temp_C5 < best_C5:
                        best_C5 = temp_C5
                        best_h = h.copy()
                    
                    # Adjust C5 bound
                    C5_bound -= step_size * decrease_factor
                    if C5_bound < 0:
                        C5_bound = 0.0001  # Prevent negative bounds
                else:
                    # Fallback to simpler method if optimization fails
                    C5_bound += step_size * increase_factor
            else:
                # Failed to solve -> increase C5
                C5_bound += step_size * increase_factor
                
        except Exception:
            # Any other optimization error -> increase C5
            C5_bound += step_size * increase_factor
            
        # Projection step to keep h within bounds
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum

        # Early termination
        if time.time() - start_time > budget_s - 0.5:
            break

    # Final evaluation of best solution
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_C5), n_points)
