            h_initial[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 10000
    stagnation_limit = 50
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random particles
    for _ in range(Pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate initial fitness
    fitness = []
    for h in swarm:
        corr = correlate(h, 1 - h, mode="full")
        fit = np.max(corr) * (2.0 / n_points)
        fitness.append(fit)
    
    # Track personal best and global best
    pbest = swarm.copy()
    pbest_fitness = fitness.copy()
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Counter for stagnation
    stagnation_count = 0
    
    # Main PSO loop
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
        
        # Update each particle
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Evaluate fitness
            corr = correlate(swarm[i], 1 - swarm[i], mode="full")
            new_fitness = np.max(corr) * (2.0 / n_points)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = swarm[i].copy()
                    gbest_fitness = new_fitness
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        # Reset some particles if stagnation
        if stagnation_count > stagnation_limit:
            for _ in range(int(Pop_size * 0.1)):
                idx = np.random.randint(Pop_size)
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)

                fit = np.max(corr) * (2.0 / n_points)
                pbest[idx] = swarm[idx].copy()
                pbest_fitness[idx] = fit
            stagnation_count = 0
        
        iter_count += 1
        
        # Decay inertia weight
        w = max(0.3, w * 0.999)
    
    return (gbest.tolist(), gbest_fitness, len(gbest))
