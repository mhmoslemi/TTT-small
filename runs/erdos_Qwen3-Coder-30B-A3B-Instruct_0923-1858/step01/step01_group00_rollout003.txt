import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to sum to n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better precision
    n_points = 200
    dx = 2.0 / n_points

    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        # Scale initial solution to our resolution
        orig_h = np.array(initial_h_values)
        h_resampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(orig_h)), 
                               orig_h)
        h_resampled = h_resampled / np.sum(h_resampled) * (n_points / 2)
    else:
        # Initialize with a smooth ramp
        h_resampled = np.ones(n_points) * 0.5
        h_resampled[0] = 0.0
        h_resampled[-1] = 0.0
        h_resampled = h_resampled / np.sum(h_resampled) * (n_points / 2)
    
    # Initialize particles around this point
    for i in range(n_particles):
        particles[i] = h_resampled + np.random.normal(0, 0.02, n_points)
        particles[i] = project_to_feasible(particles[i], n_points)
        velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 1.5
    alpha = 0.9  # For low-pass filtering
    
    # Track best
    personal_best = particles.copy()
    personal_best_scores = np.array([evaluate_erdos_solution(p) for p in personal_best])
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Main loop
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        try:
            # Update particles
            for i in range(n_particles):
                # Update velocity with inertia and cognitive/social components
                r1, r2 = np.random.rand(2, n_points)
                velocities[i] = (omega * velocities[i] + 
                                c1 * r1 * (personal_best[i] - particles[i]) + 
                                c2 * r2 * (global_best - particles[i]))
                
                # Low-pass filter for smoother exploration
                velocities[i] = alpha * velocities[i] + (1 - alpha) * np.random.normal(0, 0.01, n_points)
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply feasibility constraints
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                score = evaluate_erdos_solution(particles[i])
                
                # Update personal best
                if score < personal_best_scores[i]:
                    personal_best_scores[i] = score
                    personal_best[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best = particles[i].copy()
                
            # Add perturbation to avoid local minima
            for i in range(n_particles):
                particles[i] += np.random.normal(0, 0.01, n_points)
                particles[i] = project_to_feasible(particles[i], n_points)
                
        except Exception:
            break
    
    # Final check
    final_h = project_to_feasible(global_best, n_points)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, n_points)
