import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 50
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Initial genetic algorithm-like search
    pop_size = 20
    population = []
    
    for _ in range(pop_size):
        h_trial = h + np.random.normal(0, 0.1, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        c5 = evaluate_erdos_solution(h_trial)
        population.append((h_trial, c5))
    
    # Sort by C5 value
    population.sort(key=lambda x: x[1])
    top_candidates = population[:5]
    
    # Phase 2: Refinement with SLSQP
    for h_start, _ in top_candidates:
        try:
            res = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
    
    # Phase 3: Shift-based refinement loop
    max_iter = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute all correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        
        # Find the worst shift (maximum overlap)
        max_overlap_idx = np.argmax(corr)
        max_overlap = np.max(corr)
        
        # The shifts are from -n+1 to n-1
        shift = max_overlap_idx - (n_points - 1)
        
        # Compute the contribution to overlap from each point
        # For a shift k, the correlation contribution at i is h[i] * (1 - h[i+k])
        # We want to reduce the total contribution
        
        # Create adjustment vector based on overlap contributions
        if shift >= 0:
            # h[i] * (1 - h[i+shift]) contributes to max_overlap
            # So we decrease h[i] to decrease overlap for positive shifts
            adjusted_h = best_h.copy()
            for i in range(n_points - shift):
                if best_h[i] > 0:  # Only adjust if h[i] > 0
                    # Reduce h[i] to reduce overlap, but ensure still feasible
                    # We reduce h[i] if h[i+shift] is relatively high
                    # and increase h[i] if h[i+shift] is low
                    adjustment = 0.005 * (1 - best_h[i+shift])
                    adjusted_h[i] = max(0, min(1, adjusted_h[i] - adjustment))
            
            # Rescale to maintain integral constraint
            adjusted_h = adjusted_h * (n_points / 2) / np.sum(adjusted_h)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(adjusted_h)
            if new_c5 < best_c5:
                best_h = adjusted_h
                best_c5 = new_c5
            
        else:
            # Negative shift case, similar logic applied in reverse
            adjusted_h = best_h.copy()
            for i in range(-shift, n_points):
                if best_h[i] > 0:
                    adjustment = 0.005 * (1 - best_h[i+shift])
                    adjusted_h[i] = max(0, min(1, adjusted_h[i] - adjustment))
            
            # Rescale to maintain integral constraint
            adjusted_h = adjusted_h * (n_points / 2) / np.sum(adjusted_h)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(adjusted_h)
            if new_c5 < best_c5:
                best_h = adjusted_h
                best_c5 = new_c5
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
