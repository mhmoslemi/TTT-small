    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_candidate(h_vec, perturb_rate):
        # Make a copy
        candidate = h_vec.copy()
        num_to_perturb = max(1, int(len(h_vec) * perturb_rate))
        indices = np.random.choice(len(h_vec), num_to_perturb, replace=False)
        
        for i in indices:
            # Small random perturbation
            perturbation = np.random.normal(0, 0.05)
            candidate[i] += perturbation
            # Ensure feasibility after perturbation
            candidate[i] = np.clip(candidate[i], 0, 1)
        
        # Rescale to maintain constraint
        candidate = project_feasible(candidate)
        return candidate
    
    def is_tabu(h_vec, tabu_list):
        # Compare with stored configurations (simple hash-based comparison)
        h_tuple = tuple(np.round(h_vec, decimals=6))
        return h_tuple in tabu_list
    
    def add_to_tabu(h_vec, tabu_list, tabu_size):
        h_tuple = tuple(np.round(h_vec, decimals=6))
        tabu_list.append(h_tuple)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        return tabu_list
