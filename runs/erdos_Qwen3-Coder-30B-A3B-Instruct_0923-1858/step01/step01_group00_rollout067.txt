import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluates the Erdős constant C5 for a given step function h."""
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

        h[0] = 0.0
        h[-1] = 0.0
    h = h / np.sum(h) * (n_points / 2)

    # Binary search on C5
    c_low = 0.0
    c_high = 0.38085
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s and c_high - c_low > 1e-5:
        c_mid = (c_low + c_high) / 2.0
        
        # Try to improve with gradient-based method
        h_trial = h.copy()
        
        def objective(h_trial):
            return evaluate_erdos_solution(h_trial)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_trial):
            return np.sum(h_trial) - n_points / 2
        
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 100},
                tol=1e-6
            )
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5

    retry_h = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(rh):
        return np.sum(rh) - n_points / 2
    
    constraint = {'type': 'eq', 'fun': constraint_sum}

        result = minimize(
            evaluate_erdos_solution,
