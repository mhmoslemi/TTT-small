import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular initial guess
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define the objective function
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    iter_count = 0
    perturb_freq = 50
    
    while time.time() - start_time < budget_s - 1:
        if iter_count % perturb_freq == 0:
            # Apply perturbation to the middle region where max overlap is likely
            h_trial = h.copy()
            x = np.linspace(0, 2, n_points)
            # Focus on region where overlap likely to be large
            mask = (x >= 0.3) & (x <= 0.7)
            noise = np.random.normal(0, 0.03, n_points)
            h_trial += noise * mask
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            
            # Optimize with SLSQP
            try:
                res = minimize(objective, h_trial, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
                else:
                    # Still try to improve by local gradient descent
                    h_trial = h + np.random.normal(0, 0.01, n_points)
                    h_trial = np.clip(h_trial, 0, 1)
                    h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
                    h = h_trial.copy()
            except:
                # Fallback to simple random perturbation
                h_trial = h + np.random.normal(0, 0.005, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
                h = h_trial.copy()
        else:
            # Simple gradient-like step with noise
            h_trial = h + np.random.normal(0, 0.002, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            h = h_trial.copy()
            
        iter_count += 1
        
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
