import time
from scipy.signal import correlate
import cvxpy as cp

        h = h * (n_points / 2) / np.sum(h)

    # Convex relaxation approach
    def solve_convex_relaxation(h_start, num_points):
        # We want to maximize the minimum overlap
        # Equivalent to: max λ subject to O_k >= λ for all k
        # Let's solve using CVXPY for a good starting point
        
        h = cp.Variable(num_points)
        lambda_var = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == num_points / 2)
        
        # Overlap constraints for all k
        for k in range(num_points):
            # Compute correlation terms
            left_sum = 0
            right_sum = 0
            for i in range(num_points):
                left_sum += h[i] * (1 - h[(i+k) % num_points])
            constraints.append(left_sum >= lambda_var)
        
        # Objective
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return h_start
    
    # Try convex relaxation for starting point
    try:
        if n_points <= 100:  # Only for reasonable size
            relaxed_h = solve_convex_relaxation(h, n_points)
            if relaxed_h is not None:
                h = np.clip(relaxed_h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Fast restart strategy for random exploration
    def fast_restart():
        # Generate several random restarts
        best_h_restart = h.copy()
        best_C5_restart = compute_C5(h)
        
        for _ in range(10):
            # Perturb more aggressively
            h_random = np.random.rand(n_points)
            h_random = h_random * 2  # Balance 0 to 2 range
            h_random = np.clip(h_random, 0, 1)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            
            trial_C5 = compute_C5(h_random)
            if trial_C5 < best_C5_restart:
                best_C5_restart = trial_C5
                best_h_restart = h_random.copy()
        
        return best_h_restart, best_C5_restart
    
    # Main algorithm
    eta = 0.01
    patience = 0
    max_patience = 100
    iter_count = 0
    
    while True:

        if elapsed >= budget_s * 0.95:
            break
            
        # Compute current C5
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random restart
                h, current_C5 = fast_restart()
                patience = 0
    
        # Gradient descent step with adaptive learning rate
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1
    
    # Final refinement if needed
    current_C5 = compute_C5(best_h)
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
