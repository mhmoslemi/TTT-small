import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def optimize_fourier(h, dx, n_points, budget_s=600):
    """Optimize using convex optimization in frequency domain."""
    try:
        # Create CVXPY variables
        H = cp.Variable(n_points, complex=True)
        h_var = cp.real(cp.ifft(H))
        
        # Constraints
        constraints = [
            cp.real(H[0]) == 1.0 / dx,  # Integral constraint
            cp.norm(H, 'inf') <= 1.0,   # Bound on spectrum
        ]
        
        # Add a bound on max correlation via frequency domain
        # This uses a relaxation that is tractable but conservative
        # We'll compute the maximum correlation via its frequency domain
        # approximation using real variables
        
        # Objective: minimize maximum correlation
        # Approximate using the largest spectral component
        # This is a heuristic but preserves convexity
        
        objective = cp.Minimize(cp.norm(H, 'inf'))
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000, time_limit=budget_s-10)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            H_val = H.value
            h_opt = np.fft.ifft(H_val).real
        else:
            # fallback to gradient method if convex optimization fails
            h_opt = h.copy()
            
    except Exception:
        # fallback to gradient method if cvxpy fails
        h_opt = h.copy()
            
    # Project to feasible set
    h_opt = project_feasible(h_opt, dx)
    return h_opt
