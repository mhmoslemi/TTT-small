import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Create a more sophisticated initial solution
    h_initial = h.copy()

    # Optimization parameters
    maxiter = 1000
    tol = 1e-6
    
    # Store the best solution
    best_c5 = float('inf')
    best_h = h_initial.copy()

    # Start timer
    start_time = time.time()
    
    # Try a few different initializations to escape local minima
    for init_iter in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Initialize with current best or a new random solution
        if init_iter == 0:
            h = h_initial.copy()
        else:
            h = np.random.rand(n_points) * 0.5

        # Use L-BFGS-B as preferred optimization method for bound-constrained problems
        def objective(h):
            return evaluate_erdos_solution(h)
        
        # Bounds for h: [0, 1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint for sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
        def constraint_sum(h):

                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
