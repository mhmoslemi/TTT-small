import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to desired resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h_interp
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()

    # Start timer
    start_time = time.time()
    steps_no_improvement = 0
    alpha = 0.1

    def compute_gradient(h, max_idx):
        # Compute the gradient w.r.t. h for the maximum overlap
        n = len(h)
        grad = np.zeros(n)
        # Contribution from the maximum lag
        # For correlation at max_idx, we consider contribution from indices i,j such that j-i = max_idx
        # So we compute d/dh_i of term h_j*(1-h_{j+max_idx}) where j = i + max_idx
        # This is -h_{i+max_idx} for i such that i+max_idx < n, and -h_{i-max_idx} for i max_idx <= i
        grad_full = np.zeros(n)
        if max_idx >= 0:
            grad_full[max_idx:] -= h[:-max_idx]
        else:
            grad_full[:max_idx] -= h[-max_idx:]
        grad_full += 1 - 2 * h
        # Averaging over the full correlation result (but we do not include the full one)
        grad = grad_full * 2 / n
        return grad

    def project(h):
        # Clip to feasible bounds
        h = np.clip(h, 0, 1)
        # Renormalize if needed
        h = h / np.sum(h) * (n_points / 2)
        return h

    while time.time() - start_time < budget_s:
        try:
            # Calculate C5
            corr = np.correlate(h, 1 - h, mode="full")
            overlaps = corr * dx
            c5 = np.max(overlaps)
            max_idx = np.argmax(overlaps)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                steps_no_improvement = 0
                alpha = 0.1  # Reset learning rate on improvement
            else:
                steps_no_improvement += 1

            # Compute gradient
            grad = compute_gradient(h, max_idx)
            # Update h
            h = h - alpha * grad
            h = project(h)

            # Perturb if no improvement for a while
            # This might involve more complex strategies for better exploration
            if steps_no_improvement >= 100:
                h = h + np.random.uniform(-0.1, 0.1, n_points)
                h = project(h)
                steps_no_improvement = 0
                alpha *= 0.95  # Decay learning rate
            
            if steps_no_improvement >= 200:
                # Reset for some further exploration
                alpha = 0.1
                h = np.array(best_h)  # Respawn from best

        except Exception as e:
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
            steps_no_improvement = 0
            alpha = 0.1
    
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = best_c5

    # Final projection to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, final_c5, n_points)
