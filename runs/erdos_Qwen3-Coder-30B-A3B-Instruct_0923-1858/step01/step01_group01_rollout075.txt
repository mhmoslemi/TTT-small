import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_h(n_points):
    """Initialize h with a triadic pattern."""
    h = np.zeros(n_points)
    for i in range(n_points):
        phase = np.abs(np.sin(np.pi * i / (n_points / 3))) ** 3
        h[i] = np.clip(2 * phase - 0.5, 0, 1)
    return project(h, 2.0 / n_points)
