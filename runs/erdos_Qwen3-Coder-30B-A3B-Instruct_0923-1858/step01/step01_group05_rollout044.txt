import numpy as np
from scipy.optimize import minimize
import time

def compute_overlap(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def perturb_and_repair(h, sigma=0.03, region=(0.3, 0.7)):
    # Add noise to a specific region of h and repair feasibility
    n = len(h)
    dx = 2.0 / n
    x = np.linspace(0, 2, n)
    
    # Identify indices in the specified region
    mask = (x >= region[0]) & (x <= region[1])
    
    # Add Gaussian noise to the region
    h_perturbed = h.copy()
    h_perturbed[mask] += np.random.normal(0, sigma, np.sum(mask))
    
    # Clip to [0,1] and scale to correct integral
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = h_perturbed * (n/2.0 / np.sum(h_perturbed))
    
    return h_perturbed

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a symmetric triangular function
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization loop
    max_iter = 5000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb and repair h
        h_trial = perturb_and_repair(h, sigma=0.03, region=(0.3, 0.7))
        
        # Local optimization with SLSQP
        try:
            res = minimize(
                compute_overlap, 
                h_trial, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if res.success:
                h_new = res.x
                c5_new = compute_overlap(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        iter_count += 1
        
        # Occasionally introduce broader perturbations
        if iter_count % 100 == 0:
            h = perturb_and_repair(h, sigma=0.05, region=(0.1, 0.9))
    
    # Final validation
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
