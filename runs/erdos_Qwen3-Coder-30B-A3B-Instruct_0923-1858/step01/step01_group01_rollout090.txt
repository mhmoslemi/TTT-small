import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1, strength=0.1):
    """Mutate the step function with small random changes."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, strength)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    child = parent1.copy()
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent2[i]
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Tournament selection."""
    tournament_size = 3
    selected = []
    for _ in range(num_parents):
        tournament_indices = np.random.choice(len(population), size=tournament_size, replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def calculate_fitness(h, dx):
    """Calculate fitness - inverse of C5 with complexity penalty."""
    c5 = evaluate(h, dx)
    # Complexity penalty based on number of significant changes
    diff = np.abs(np.diff(h))
    complexity = np.sum(diff > 0.05)
    return 1 / (c5 + 0.01 * complexity)

def initialize_population(n_points, pop_size=50, initial_h=None):
    """Initialize population with variants of initial solution."""
    population = []
    if initial_h is not None:
        population.append(initial_h)
        pop_size -= 1
    else:
        # Create a simple triangle shape
        initial_h = np.linspace(0, 1, n_points//2, endpoint=False)
        initial_h = np.concatenate([initial_h, np.linspace(1, 0, n_points - n_points//2)])
        population.append(initial_h)
        
    # Add mutated variants
    for _ in range(pop_size):
        mutated = mutate(initial_h, 1.0/n_points, rate=0.2, strength=0.1)
        population.append(mutated)
        
    return population

    best_fitness = 0
    best_h = None
    pop_size = len(population)

    fitnesses = [calculate_fitness(h, dx) for h in population]
    
    while time.time() - start_time < budget_s - 30:  # Leave buffer
        # Track best
        current_best_idx = np.argmax(fitnesses)
        if fitnesses[current_best_idx] > best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses, num_parents=pop_size//2)
        
        # Create new population
        new_population = []
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child, dx, rate=0.15, strength=0.08)
            new_population.append(child)
            
        population = new_population
        
        # Re-evaluate fitness
        fitnesses = [calculate_fitness(h, dx) for h in population]
        
        # Early stopping check
        if time.time() - start_time > budget_s - 60:
            break
    
    # Return final best
    current_best_idx = np.argmax(fitnesses)
    if fitnesses[current_best_idx] > best_fitness:
        best_h = population[current_best_idx]
        best_fitness = fitnesses[current_best_idx]
        
    return best_h, 1 / best_fitness
