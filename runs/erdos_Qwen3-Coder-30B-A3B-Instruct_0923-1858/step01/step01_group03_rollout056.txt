import time
from scipy.signal import correlate

        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 500
    mutation_rate = 0.1
    tournament_size = 5
    crossover_rate = 0.8
    
    # Initialize population
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def create_individual():
        # Create random individual
        h_new = np.random.rand(n_points)
        return project_feasible(h_new)
    
    # Create initial population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        population.append(create_individual())
    
    # Keep track of best solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    def tournament_selection(population, k):
        idx = np.random.choice(len(population), size=k)
        best_idx = idx[0]
        best_fitness = compute_C5(population[best_idx])
        for i in range(1, k):
            fitness = compute_C5(population[idx[i]])
            if fitness < best_fitness:
                best_fitness = fitness
                best_idx = idx[i]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
            
        # Single-point crossover
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h_vec):
        h_new = h_vec.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Add small Gaussian noise
                h_new[i] += np.random.normal(0, 0.02)
        return project_feasible(h_new)
    
    # Main evolution loop
    stagnation_counter = 0
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [compute_C5(ind) for ind in population]
        
        # Track best solution in this generation
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_C5:
            best_C5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Reset population if no improvement for a while
        if stagnation_counter > 30:
            population = [best_h.copy()]
            for _ in range(pop_size - 1):
                population.append(create_individual())
            stagnation_counter = 0
            continue
        
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_h.copy())
        
        # Generate rest of population
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, tournament_size)
            parent2 = tournament_selection(population, tournament_size)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final evaluation of best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
