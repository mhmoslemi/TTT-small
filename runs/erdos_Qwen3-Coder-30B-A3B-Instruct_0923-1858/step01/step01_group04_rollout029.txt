import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with given sum constraint"""
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Use a simpler construction
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    # Normalize initially
    h = project_to_feasible(h, n_points / 2)
    
    # Objective function
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Pre-computed bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    iter_count = 0
    max_iter = 200
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try random perturbations to escape local optima
        perturbation = np.random.normal(0, 0.1, n_points)
        h_trial = best_h + perturbation
        h_trial = project_to_feasible(h_trial, n_points / 2)
        
        # Local optimization using L-BFGS-B
        try:
            res = minimize(
                objective,
                h_trial,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = project_to_feasible(h_candidate, n_points / 2)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
            
        iter_count += 1
        
    # Try a different approach with higher resolution if time remains
    if time.time() - start_time < budget_s - 5 and n_points < 100:
        # Refine solution with higher resolution
        n_points = 100
        h_refined = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(best_h)), best_h)
        h_refined = project_to_feasible(h_refined, n_points / 2)
        
        # Optimize on refined mesh
        def objective_refined(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)
        
        try:
            bounds_refined = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                objective_refined,
                h_refined,
                method='L-BFGS-B',
                bounds=bounds_refined,
                options={'maxiter': 200, 'ftol': 1e-8}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = project_to_feasible(h_candidate, n_points / 2)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    
        except Exception:
            pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, len(best_h))
