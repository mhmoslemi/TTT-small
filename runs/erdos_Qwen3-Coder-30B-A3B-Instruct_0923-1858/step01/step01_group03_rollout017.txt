        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_limit = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution as first particle
    swarm.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random particles
    for _ in range(Pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal best
    pbest = [swarm[i].copy() for i in range(Pop_size)]
    pbest_fitness = [float('inf')] * Pop_size
    
    # Global best
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Evaluate initial swarm
    for i in range(Pop_size):
        h_eval = swarm[i]
        corr = correlate(h_eval, 1 - h_eval, mode="full")
        fitness = np.max(corr) * dx
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    # PSO main loop
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 50
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        # Update each particle
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]

            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Evaluate fitness
            corr = correlate(swarm[i], 1 - swarm[i], mode="full")
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Reset strategy if stagnation
        if stagnation_count > max_stagnation:
            for i in range(Pop_size):
                # Reset 10% of particles
                if np.random.rand() < 0.1:
                    swarm[i] = np.random.rand(n_points)
                    swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])

                    pbest[i] = swarm[i].copy()
                    corr = correlate(swarm[i], 1 - swarm[i], mode="full")
                    pbest_fitness[i] = np.max(corr) * dx
            
            stagnation_count = 0
            
        # Adaptive inertia weight
        w = max(0.3, 0.7 - (iter_count / 1000) * 0.4)
        
        iter_count += 1
    
    # Final best solution
    final_C5 = gbest_fitness
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = swarm[gbest_idx].copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
