import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Number of points
    n_points = 100
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Use L-BFGS for optimization with bounds and constraints
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Bounds for h: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()

    # Start timer
    start_time = time.time()
    
    # Optimize until timeout
    while time.time() - start_time < budget_s:
        try:
            # Run optimization
            result = minimize(
                objective,
                h.copy(),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 1000},
                tol=1e-6
            )
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
                # Perturb h to escape local minima
                h = best_h.copy() + np.random.normal(0, 0.005, n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
            else:
                # If optimization fails, try with a new initial guess
                h = np.random.rand(n_points) * 0.5
                h = h / np.sum(h) * (n_points / 2)
        except Exception:
            # If anything goes wrong, try with a new initial guess
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
    
    try:
        actual_c5 = evaluate_erdos_solution(best_h)
    except:
        actual_c5 = best_c5

    # Ensure final solution is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    return (best_h, actual_c5, n_points)
