import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    t = np.linspace(0, 2, n_points)
    h = np.maximum(0, 1 - np.abs(t - 1))
    return project_feasible(h, dx=2.0 / n_points)

                idx = np.random.randint(0, n_points)
                h_opt[idx] += np.random.normal(0, 0.005)
            h_opt = project_feasible(h_opt, dx)
            
    return best_h, best_C
