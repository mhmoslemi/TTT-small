import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for better resolution
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to new resolution
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize to sum(h) = n_points/2
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling factor
    sigma = 0.1  # Perturbation scale
    shake_freq = 100  # Shake every shake_freq iterations
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # For early stopping check
    last_improvement = 0
    stagnation_count = 0
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Generate perturbation
        delta_h = np.random.uniform(-T * sigma, T * sigma, n_points)
        
        # Apply perturbation
        h_candidate = h + delta_h
        
        # Project to feasibility: clip and rescale
        h_candidate = np.clip(h_candidate, 0, 1)
        if np.sum(h_candidate) > 0:
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        else:
            # Avoid division by zero
            h_candidate = np.ones(n_points) * 0.5
            h_candidate[0] = 0.0
            h_candidate[-1] = 0.0
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Accept or reject
        delta_E = c5_candidate - best_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                last_improvement = iteration
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
        
        # Adaptive cooling
        if iteration % 10 == 0:
            T *= alpha
        
        # Adaptive shaking
        if stagnation_count > shake_freq:
            # Add noise to escape local optima
            noise = np.random.normal(0, T * sigma, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            if np.sum(h) > 0:
                h = h / np.sum(h) * (n_points / 2)
            else:
                h = np.ones(n_points) * 0.5
                h[0] = 0.0
                h[-1] = 0.0
                h = h / np.sum(h) * (n_points / 2)
            stagnation_count = 0
    
    return (best_h, best_c5, n_points)
