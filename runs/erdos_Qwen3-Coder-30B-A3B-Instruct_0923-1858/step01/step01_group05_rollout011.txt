import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_old, h)
            n_points = 140
    else:
        # Use a simple uniform distribution with scaling
        n_points = 140
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
        
    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2.0)
    
    # Track the best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation parameters
    perturb_sigma = 0.05
    max_iter = 1000
    iter_count = 0
    
    # Global exploration and local refinement loop
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Global exploration: perturb solutions
        num_perturb = 50
        candidates = []
        
        for _ in range(num_perturb):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_perturb = h + np.random.normal(0, perturb_sigma, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2.0)
            
            c5_perturb = evaluate_erdos_solution(h_perturb)
            candidates.append((c5_perturb, h_perturb.copy()))
            
        # Sort candidates by C5 value
        candidates.sort(key=lambda x: x[0])
        
        # Local refinement on top 10 candidates
        num_refine = min(10, len(candidates))
        
        for i in range(num_refine):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_trial = candidates[i][1]
            
            # Define constraints
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                corr = np.correlate(h, 1-h, mode="full")
                return np.max(corr) * (2.0 / n_points)
            
            # Try SLSQP optimization
            try:
                res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass
                
        # Update global solution if needed
        if candidates[0][0] < best_c5:
            best_h = candidates[0][1].copy()
            best_c5 = candidates[0][0]
            
        # Issue a new global perturbation
        h = h + np.random.normal(0, perturb_sigma, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2.0)
        
        # Decrease perturbation strength periodically
        if iter_count % 50 == 0 and iter_count > 0:
            perturb_sigma *= 0.5
            
        iter_count += 1
        
        # Occasionally check time to ensure we finish on time
        if iter_count % 100 == 0:
            if time.time() - start_time > budget_s - 1:
                break
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
