import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach: start coarse then refine
    # Resolution levels: [50, 100]
    resolutions = [50, 100]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        if res == 50:
            # Coarse resolution for global search
            n_coarse = 50
            
            def objective_coarse(h_coarse):
                # Upsample to full resolution
                h_full = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, n_coarse), 
                                  h_coarse)
                h_full = np.clip(h_full, 0, 1)
                h_full = h_full * (n_points / 2) / np.sum(h_full)
                return evaluate_erdos_solution(h_full)
            
            # Bounds for coarse resolution
            bounds = [(0, 1) for _ in range(n_coarse)]
            
            # Differential evolution on coarse scale
            result = differential_evolution(
                objective_coarse,
                bounds,
                maxiter=100,
                popsize=40,
                mutation=0.8,
                recombination=0.9,
                seed=seed,
                atol=1e-6,
                tol=1e-6
            )
            
            if result.success:
                h_candidate = np.interp(np.linspace(0, 2, n_points), 
                                       np.linspace(0, 2, n_coarse), 
                                       result.x)
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    n_points = len(best_h)
        
        elif res == 100:
            # Fine resolution local search
            # Use SLSQP for local refinement
            def objective_fine(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                return evaluate_erdos_solution(h_vec)
            
            def constraint_fine(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_fine}
            
            try:
                res = minimize(
                    objective_fine,
                    best_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
