import time

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    perturb_rate = 0.03  # 3% of points
    max_iter = 2000
    iter_count = 0
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def perturb_solution(h_vec, rate):
        h_new = h_vec.copy()
        n_perturb = max(1, int(rate * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            # Add small random perturbation
            h_new[i] += np.random.normal(0, 0.05)
        return project_feasible(h_new)
    
    # Main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = perturb_solution(h, perturb_rate)
        
        # Compute C5 for candidate
        candidate_C5 = compute_C5(candidate_h)
        
        # Check if candidate is better than current best
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate_h.copy()
        
        # Add to tabu list (keep size limited)
        tabu_list.append(h.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Move to candidate
        h = candidate_h
        
        iter_count += 1
        if iter_count >= max_iter:
            break
    
    # Final evaluation of best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
