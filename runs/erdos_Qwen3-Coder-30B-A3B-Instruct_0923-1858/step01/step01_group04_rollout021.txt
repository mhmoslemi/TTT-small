import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Use initial construction if available

        n_points = len(h)
    else:

        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    # First, optimize at a lower resolution
    n_low = 50
    if n_points >= n_low:
        # Downsample to lower resolution
        h_low = h[::n_points//n_low]
        if len(h_low) < n_low:
            # Interpolate to make it exactly n_low
            x_orig = np.linspace(0, 1, n_points)
            x_new = np.linspace(0, 1, n_low)
            h_low = np.interp(x_new, x_orig, h)
        else:
            h_low = h_low[:n_low]
        
        # Normalize
        h_low = h_low * (n_low / 2) / np.sum(h_low)
        h_low = np.clip(h_low, 0, 1)
        
    else:
        # Create a new low-res version
        h_low = np.ones(n_low) * 0.5
        h_low = h_low * (n_low / 2) / np.sum(h_low)
        h_low = np.clip(h_low, 0, 1)
    
    # Differential evolution at low resolution
    def objective_low(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Interpolate to full resolution
        x_full = np.linspace(0, 1, n_points)
        x_low = np.linspace(0, 1, n_low)
        h_full = np.interp(x_full, x_low, h_vec)
        h_full = h_full * (n_points / 2) / np.sum(h_full)
        return evaluate_erdos_solution(h_full)
    
    bounds_low = [(0, 1) for _ in range(n_low)]
    
    de_result = differential_evolution(
        objective_low,
        bounds_low,
        maxiter=200,
        popsize=40,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    if de_result.success:
        h_candidate_low = de_result.x
        x_full = np.linspace(0, 1, n_points)
        x_low = np.linspace(0, 1, n_low)
        h_candidate = np.interp(x_full, x_low, h_candidate_low)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        c5_candidate = evaluate_erdos_solution(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    # Local refinement with SLSQP at high resolution
    def objective_high(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)

        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Try to refine best solution locally
    try:
        res = minimize(
            objective_high,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-8}

            h_refined = np.clip(res.x, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
