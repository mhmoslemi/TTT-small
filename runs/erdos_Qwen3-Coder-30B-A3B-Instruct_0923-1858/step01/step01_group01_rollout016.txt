import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1, strength=0.1):
    """Apply random mutations to h."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, strength)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    child = np.array(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent2[i]
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Tournament selection."""
    tournament_size = 3
    selected = []
    for _ in range(num_parents):
        tournament_indices = np.random.choice(len(population), 
                                            size=tournament_size, 
                                            replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner])
    return selected

def evaluate_fitness(h, dx):
    """Fitness is inverse of C5 with complexity penalty."""
    c5 = evaluate(h, dx)
    # Simpler representation counts changes
    complexity = np.sum(np.abs(np.diff(h)) > 0.05)
    return 1.0 / (c5 + 0.01 * complexity)

def initialize_population(n_points, population_size=50):
    """Initialize population with initial_h_values or random."""
    population = []
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create triangular waveform
        initial_h = np.linspace(0, 1, n_points//2, endpoint=False)
        initial_h = np.concatenate([initial_h, np.linspace(1, 0, n_points - n_points//2)])
    
    for _ in range(population_size):
        if np.random.rand() < 0.7:
            h = initial_h.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
        h = project(h, 2.0/n_points)
        population.append(h)
    return population

def evolve_population(population, dx, budget_s=600):
    """Evolve population using genetic algorithm."""
    start_time = time.time()
    best_fitness = 0
    best_h = None
    population_size = len(population)
    num_parents = max(5, population_size // 3)
    
    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitnesses = [evaluate_fitness(h, dx) for h in population]
        
        # Track best solution
        current_best_idx = np.argmax(fitnesses)
        if fitnesses[current_best_idx] > best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Select parents
        parents = select_parents(population, fitnesses, num_parents)
        
        # Generate new population
        new_population = []
        while len(new_population) < population_size:
            # Perform crossover
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            # Apply mutation
            if np.random.rand() < 0.3:
                child = mutate(child, dx, 0.15, 0.1)
            new_population.append(child)
        
        # Replace population
        population = new_population
        
    return best_h, 1.0 / best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points)
    
    # Evolve population
    best_h, best_C = evolve_population(population, dx, budget_s)
    
    return (best_h, best_C, n_points)
