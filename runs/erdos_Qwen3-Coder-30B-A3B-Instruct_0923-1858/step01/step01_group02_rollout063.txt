import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h onto feasible region: sum = n_points/2, bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[[0, 1, 2, 3, 4]] = 0.8
        h[[5, 6, 7, 8, 9]] = 0.2
        h = project_h(h, n_points)
    
    # Initialize C5 estimate
    C5 = 0.4
    step_size = 0.0001
    last_improvement = 0
    last_C5 = C5
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    iters = 0
    while time.time() - start_time < budget_s - 1:
        iters += 1
        
        # Try to improve by reducing C5
        new_C5 = C5 - step_size
        if new_C5 < 0:
            new_C5 = 0.0001
            
        feasible = False
        max_iter = 100
        
        # Create optimization problem with fixed C5
        h_var = cp.Variable(n_points)
        
        # Objective: minimize sum * dx (but we want to reduce C5, so use a dummy minimizing objective)
        # We'll instead add constraints and see if a feasible solution exists
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Create constraints for each k shift
        for k in range(n_points):
            # a = 0: k, b = n_points-n: k+n
            # But properly handle indexing
            if k == 0:
                # h[i] * (1 - h[i]) for shift 0
                continue
            else:
                if k < n_points:
                    idx_a = np.arange(n_points - k)
                    idx_b = np.arange(k, n_points)
                    # h[i] * (1 - h[i+k]) 
                    # sum(h[i] * (1 - h[i+k]))
                    top = np.sum(h_var[idx_a] * (1 - h_var[idx_b]))
                    constraints.append(top <= new_C5 / (2.0 / n_points))
                else:  # k >= n_points
                    constraints.append(cp.sum(h_var * (1 - h_var[:n_points-k])) <= new_C5 / (2.0 / n_points))
        
        # Solve the problem
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                feasible = True
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.array(h_sol)
                    h_sol = project_h(h_sol, n_points)
                    test_C5 = evaluate_erdos_solution(h_sol)
                    if test_C5 < best_C5:
                        best_C5 = test_C5
                        best_h = h_sol.copy()
                C5 = new_C5
                last_improvement = 0
            else:
                # Problem was infeasible, increase C5 and try again
                pass
        except:
            pass
            
        # If no improvement for a while, increase C5
        last_improvement += 1
        if last_improvement > 10 and C5 < last_C5 - 0.0005:
            C5 = last_C5 + step_size * 2
            step_size *= 2  # Increase step size
            last_improvement = 0

        if time.time() - start_time > budget_s - 0.1:
            break

    # Final check on best
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_C5), n_points)
