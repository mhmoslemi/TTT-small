from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

        return h_new

    def crossover(h1, h2, rate=0.5):
        mask = np.random.rand(len(h1)) < rate

        h_new = h_new / np.sum(h_new) * (n_points / 2.0)
