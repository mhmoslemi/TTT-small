import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    alpha = 0.1
    stagnation_count = 0
    max_stagnation = 200
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute current correlation and C5
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        # Find lag corresponding to max
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if max_lag_idx < n_points:
                j = max_lag_idx - i + n_points - 1
                if 0 <= j < n_points:
                    grad[i] = (1 - 2 * h[i]) * dx
            else:
                j = max_lag_idx - i + n_points - 1
                if 0 <= j < n_points:
                    grad[i] = (1 - 2 * h[i]) * dx
                    
        # Update
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        h = h_new.copy()
        
        if stagnation_count > max_stagnation:
            h = best_h.copy() + np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            stagnation_count = 0
            alpha = 0.1
            
        if stagnation_count % 100 == 0:
            alpha *= 0.95
            
    return (best_h, best_c5, n_points)
