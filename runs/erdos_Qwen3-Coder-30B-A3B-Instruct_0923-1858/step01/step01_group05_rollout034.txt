import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):

    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    population_size = 50
    num_generations = 1000
    mutation_rate = 0.03
    crossover_rate = 0.5
    elite_size = 5
    tournament_size = 5

    population = []
    if 'initial_h_values' in globals():

        for i in range(population_size):

                h_seed = np.clip(h_seed, 0, 1)
                h_seed = h_seed / np.sum(h_seed) * (n_points / 2.0)
                population.append(h_seed)
            else:
                # Random initialization
                h_rand = np.random.rand(n_points)
                h_rand = h_rand / np.sum(h_rand) * (n_points / 2.0)
                h_rand = np.clip(h_rand, 0, 1)
                population.append(h_rand)
    else:
        for _ in range(population_size):
            h_init = np.random.rand(n_points)

            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            last_improvement = generation
            
        # Check for stagnation
        if generation - last_improvement > patience:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i])

                parent = tournament_select(sorted_population, sorted_fitness, tournament_size)

def tournament_select(population, fitness, tournament_size):
    """Tournament selection for choosing parents"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index]

def blend_crossover(parent1, parent2):
    """Blend crossover between two parents"""
    alpha = np.random.rand()
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate):
    """Gaussian mutation"""
    mask = np.random.rand(len(individual)) < mutation_rate
    noise = np.random.normal(0, 0.03, len(individual))
    individual = individual + mask * noise
    # Ensure bounds
    individual = np.clip(individual, 0, 1)
    # Ensure sum constraint
    individual = individual / np.sum(individual) * (len(individual) / 2.0)
    return individual
