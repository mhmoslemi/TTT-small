import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Perturb to get a better starting point
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Cutting plane method
    max_iter = 1000
    iter_count = 0
    time_limit = budget_s - 1
    
    while time.time() - start_time < time_limit and iter_count < max_iter:
        # Compute overlaps for current h
        overlaps = []
        for k in range(2 * n_points - 1):
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps.append(overlap * dx)
        
        max_overlap = max(overlaps)
        
        # If we haven't improved, break
        if max_overlap >= best_c5:
            break

        best_c5 = max_overlap
        best_h = h.copy()

        # Formulate convex subproblem with cutting planes
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            C >= max_overlap
        ]
        
        # Add cutting planes
        max_k = np.argmax(overlaps)
        grad = np.zeros(n_points)
        # Compute gradient for shift k that caused max overlap
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad[i] = dx * (1 - h[j])
        
        # Add the cutting plane constraint
        cut_constraint = cp.sum(grad * h_var) <= np.dot(grad, h) - np.dot(grad, h) + overlaps[max_k]
        constraints.append(cut_constraint)
        
        # Minimize max overlap over all k in cutting plane constraints
        # We already have at least one such constraint from max_overlap
        prob = cp.Problem(cp.Minimize(C), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = np.array(h_var.value).flatten()
                h = np.clip(h, 0, 1)
        except Exception:
            # Fall back to random perturbation if optimization fails
            h = h.copy()
            # Add small random perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            # Renormalize
            h = h * (n_points / 2) / np.sum(h)
        
        iter_count += 1

        # Occasionally re-evaluate the best solution to make sure
        if iter_count % 5 == 0:
            current_c5 = evaluate_erdos_solution(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
