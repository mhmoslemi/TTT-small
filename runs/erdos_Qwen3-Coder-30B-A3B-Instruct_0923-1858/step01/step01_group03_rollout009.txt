import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Population-based approach with genetic algorithm
    POP_SIZE = 100
    MUTATION_RATE = 0.2
    TOURNAMENT_SIZE = 5
    MAX_NO_IMPROVEMENT = 50
    
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize population
    population = []
    population.append(h_init.copy())
    for _ in range(POP_SIZE - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def tournament_selection(pop, k=TOURNAMENT_SIZE):
        contestants = np.random.choice(len(pop), k)
        best = contestants[0]
        for i in range(1, k):
            if compute_C5(pop[contestants[i]]) < compute_C5(pop[best]):
                best = contestants[i]
        return pop[best]
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(h_vec):
        h_mut = h_vec.copy()
        for i in range(n_points):
            if np.random.rand() < MUTATION_RATE:
                h_mut[i] = np.clip(h_mut[i] + np.random.normal(0, 0.05), 0, 1)
        return project_to_feasible(h_mut)
    
    # Evolution loop
    best_solution = None
    best_C5 = float('inf')
    no_improvement = 0
    
    for gen in range(1000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate population
        fitnesses = [compute_C5(h) for h in population]
        current_best_idx = np.argmin(fitnesses)
        current_best = population[current_best_idx]
        current_C5 = fitnesses[current_best_idx]
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_solution = current_best.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > MAX_NO_IMPROVEMENT:
                # Restart with new random population
                population = []
                population.append(best_solution.copy())
                for _ in range(POP_SIZE - 1):
                    h = np.random.uniform(0, 1, n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    population.append(h)
                no_improvement = 0
                continue
        
        # Selection, crossover, mutation
        new_population = []
        # Keep best individual
        new_population.append(best_solution.copy())
        
        while len(new_population) < POP_SIZE:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
            
        population = new_population
    
    # Final evaluation
    final_C5 = compute_C5(best_solution)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_solution.tolist(), best_C5, n_points)
