import numpy as np
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    iter_count = 0
    last_improvement = 0
    perturb_mag = 0.1
    patience = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:

        # Perturb
        perturbation = np.random.normal(0, perturb_mag, n_points)

        try:
            # Use L-BFGS-B which handles bounds well
            res = minimize(
                lambda x: evaluate_erdos_solution(x),
                x0,
                method='L-BFGS-B',

                options={'maxiter': 50}
            )
            
            if res.success:
                current_c5 = evaluate_erdos_solution(res.x)
                if current_c5 < best_c5:

                    last_improvement = iter_count

            patience = 0

        if time.time() - start_time > budget_s - 0.1:

    return (best_h.tolist(), float(final_c5), n_points)
