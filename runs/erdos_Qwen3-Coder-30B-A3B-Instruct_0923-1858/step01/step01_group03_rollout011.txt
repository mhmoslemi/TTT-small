import time

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 75
    max_iter = 2000
    perturb_rate = 0.04  # 4% of points per iteration
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        """Compute maximum overlap integral"""
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        """Project to feasible region [0,1] with sum = n_points / 2"""
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def perturb_solution(h_vec, rate):
        """Create neighbor by perturbing a fraction of points"""
        h_new = h_vec.copy()
        n_perturb = max(1, int(rate * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            # Random perturbation with bounded support
            delta = np.random.normal(0, 0.1)
            h_new[idx] = h_new[idx] + delta
        return project_feasible(h_new)
    
    # Main tabu search loop
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Generate candidate by perturbing current solution
        candidate_h = perturb_solution(h, perturb_rate)
        
        # Check if candidate is in tabu list
        candidate_tuple = tuple(candidate_h.round(8))
        if candidate_tuple in tabu_list:
            # Accept with small probability to escape local minima
            if np.random.rand() > 0.8:
                pass
            else:
                iter_count += 1
                continue
        
        # Evaluate candidate
        candidate_C5 = compute_C5(candidate_h)
        
        # Update best solution if improved
        if candidate_C5 < best_C5:
            best_C5 = candidate_C5
            best_h = candidate_h.copy()
        
        # Update tabu list
        tabu_list.append(candidate_tuple)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Accept candidate as new current solution
        h = candidate_h
        
        iter_count += 1
    
    # Final best evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
