import time
from scipy.signal import correlate

        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_initial = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    Pop_size = 100
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 10000
    stagnation_limit = 50
    mutation_rate = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal best
    pbest = [swarm[i].copy() for i in range(Pop_size)]
    pbest_fitness = [float('inf')] * Pop_size
    
    # Track global best
    gbest_idx = 0
    gbest = swarm[0].copy()
    gbest_fitness = float('inf')
    
    # Evaluate initial swarm
    for i in range(Pop_size):
        h = swarm[i]
        overlaps = correlate(h, 1 - h, mode="full")
        fitness = np.max(overlaps) * (2.0 / n_points)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest = h.copy()
            gbest_idx = i
    
    # PSO main loop
    iter_count = 0
    stagnation_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # Leave a margin
            break
            
        if iter_count >= max_iter:
            break
            
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Evaluate fitness
            overlaps = correlate(swarm[i], 1 - swarm[i], mode="full")
            fitness = np.max(overlaps) * (2.0 / n_points)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = swarm[i].copy()
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest = swarm[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        # Mutation if stagnation
        if stagnation_count > stagnation_limit:
            for i in range(int(mutation_rate * Pop_size)):
                idx = np.random.randint(0, Pop_size)
                swarm[idx] = np.random.random(n_points)
                swarm[idx] = swarm[idx] * (n_points / 2) / np.sum(swarm[idx])
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0
            
        # Update inertia weight
        if iter_count % 100 == 0 and w > 0.3:
            w = max(0.3, w * 0.99)
            
        iter_count += 1
    
    # Final evaluation
    overlaps = correlate(gbest, 1 - gbest, mode="full")
    final_C5 = np.max(overlaps) * (2.0 / n_points)
    
    return (gbest.tolist(), final_C5, len(gbest))
