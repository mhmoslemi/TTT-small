import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution with midpoint values
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Normalize sum to meet constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Standardize dx
    dx = 2.0 / n_points

    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Iterative cutting plane method
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate current maximum overlap
        current_overlap = evaluate_erdos_solution(h)
        if current_overlap < best_c5:
            best_h = h.copy()
            best_c5 = current_overlap
            
        # Compute overlaps with all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        current_overlaps = corr[len(corr)//2:]  # Only shifts from 0 to n_points-1

        # Select the shift with highest overlap
        max_shift_idx = np.argmax(current_overlaps)
        max_shift = max_shift_idx

        # Build a convex subproblem with cutting plane for the max shift
        h_var = cp.Variable(n_points)
        C_var = cp.Variable(1)

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C_var >= 0
        ]

        # Add cutting plane for max shift
        # O_k(h) = dx * sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k) mod n_points])
        # Gradient of O_k with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_shift) % n_points
            grad[i] = dx * (1 - h[j])
            
        # Gradient point: ∇O_k(h)
        current_val = current_overlaps[max_shift_idx]
        # Cutting plane: ∇O_k(h^t)^T * h <= ∇O_k(h^t)^T * h^t + O_k(h^t) - ∇O_k(h^t)^T * h^t
        # Simplified as: ∇O_k(h^t)^T * h <= current_val + grad^T * (-h)
        # Substitute and rearrange to standard form
        # grad^T * h <= current_val - grad^T * h^t
        cut_plane_constraint = grad @ h_var <= current_val - grad @ h
        
        constraints.append(cut_plane_constraint)
        
        # Minimize C
        objective = cp.Minimize(C_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Ensure bounds and normalization
                new_h = np.clip(new_h, 0, 1)
                # Renormalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
            else:
                # Fall back to previous h if optimization fails
                pass
        except Exception:
            # Fall back to previous h if optimization fails
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
