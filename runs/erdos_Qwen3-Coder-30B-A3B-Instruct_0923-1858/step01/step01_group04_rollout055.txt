import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 50
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Phase 1: Optimize with lower resolution first
    if n_points < 100:
        n_points = 100
        # Upsample by interpolation
        h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h_new
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Optimize with SLSQP
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    max_iter = 20
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        try:
            # Random perturbation
            perturbation = np.random.normal(0, 0.05, n_points)
            h_trial = best_h + perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # SLSQP optimization
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
            
        # Periodic reset to best solution with small perturbation
        if iter_count % 5 == 0:
            try:
                perturbation = np.random.normal(0, 0.02, n_points)
                h_trial = best_h + perturbation
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                pass
                
        iter_count += 1
        
        # Sometimes try another optimization from the best current solution
        if iter_count % 7 == 0:
            try:
                h_trial = best_h.copy()
                # Add more significant noise
                perturbation = np.random.normal(0, 0.1, n_points)
                h_trial = h_trial + perturbation
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 400, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
