    population_size = 100
    mutation_rate = 0.2

    num_generations = 500
    
    # Prepare initial population
    population = [h.copy()]
    for _ in range(population_size - 1):

    best_h = h.copy()
    best_C5 = float('inf')
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def tournament_selection(population, fitnesses, k):
        selected_indices = np.random.choice(len(population), size=k)
        best_idx = selected_indices[np.argmax([fitnesses[i] for i in selected_indices])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(h_vec, rate):
        mutation_idx = np.random.choice(n_points, size=int(rate * n_points), replace=False)
        for i in mutation_idx:
            # Add small Gaussian noise
            h_vec[i] += np.random.normal(0, 0.05)
        return h_vec
    
    def evaluate_population(population):
        fitnesses = [compute_C5(individual) for individual in population]
        return fitnesses
    
    # Main GA loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50

        best_C5 = final_fitness
    
    return (best_h.tolist(), best_C5, len(best_h))
