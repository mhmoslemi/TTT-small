    # PSO parameters
    pop_size = 100

    c2 = 2.0
    max_iter = 10000
    stagnation_limit = 50

    swarm = []
    velocities = []
    
    # Add the initial solution to the swarm
    swarm.append(h.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points) * 0.1)
    
    # Add random particles to fill the swarm
    for _ in range(pop_size - 1):
        temp_h = np.random.rand(n_points)
        # Normalize to sum to n_points / 2
        temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
        swarm.append(temp_h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points) * 0.1)
    
    # Keep track of personal bests
    pbest = [swarm[i].copy() for i in range(pop_size)]
    pbest_fitness = [float('inf')] * pop_size
    
    # Initialize global best
    gbest = swarm[0].copy()
    gbest_fitness = float('inf')
    
    iter_count = 0
    stagnation = 0
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Clip between 0 and 1
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate all particles and update pbest and gbest
        for i in range(pop_size):
            particle = swarm[i]
            fitness = compute_C5(particle)
            
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = particle.copy()
            
            if fitness < gbest_fitness:

            best_C5 = gbest_fitness

            velocities[i] = (inertia_w * velocities[i] +

                           c2 * r2 * (gbest - swarm[i]))
            
            # Position update
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])

        inertia_w = 0.7 + (0.3 * iter_count / max_iter)

            stagnation = 0
        
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(gbest)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = gbest.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
