import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy sum(h) * dx = 1 with 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate maximum overlap integral C5."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_h(n_points):
    """Initialize h with a triadic pattern."""
    h = np.zeros(n_points)
    for i in range(n_points):
        phase = np.abs(np.sin(np.pi * i / (n_points / 3))) ** 3
        h[i] = np.clip(2 * phase - 0.5, 0, 1)
    return project(h, 2.0 / n_points)

def optimize_step(h, dx, n_points):
    """Compute gradient using finite differences and update h."""
    grad = np.zeros_like(h)
    eps = 1e-8
    h_orig = h.copy()
    
    for i in range(n_points):
        h_plus = h_orig.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        max_plus = np.max(corr_plus)

        h_minus = h_orig.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        max_minus = np.max(corr_minus)

        grad[i] = (max_plus - max_minus) / (2 * eps)
    
    # Update step
    lr = 0.01
    h = h_orig - lr * grad
    h = project(h, dx)
    
    return h
