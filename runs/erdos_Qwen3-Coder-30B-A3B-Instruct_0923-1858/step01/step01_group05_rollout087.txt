import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)

        if n_points != 140:

            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_old, h)
            n_points = 140
    else:
        # Create a feasible initial h vector
        n_points = 140
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Global exploration with perturbations
    perturb_strength = 0.05
    max_iter = 1000
    
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Perturb h
        h_trial = h + np.random.normal(0, perturb_strength, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        c5_trial = objective(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
        
        # Local refinement on top 10 candidates
        candidates = []
        for _ in range(10):
            h_trial = h + np.random.normal(0, perturb_strength, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
            c5_trial = objective(h_trial)
            candidates.append((c5_trial, h_trial.copy()))
        
        candidates.sort(key=lambda x: x[0])
        for c5_new, h_new in candidates[:5]:
            try:
                res = minimize(objective, h_new, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
                        h = h_refined.copy()
            except:
                pass
        
        iteration += 1
        # Decrease perturbation strength
        if iteration % 50 == 0:
            perturb_strength *= 0.5
            
        # Occasionally perform a simple gradient step to keep moving
        if iteration % 20 == 0:
            grad = np.zeros(n_points)
            # Approximate gradient using finite differences
            eps = 1e-4
            for i in range(n_points):
                h_eps = h.copy()

                h_eps = np.clip(h_eps, 0, 1)
                h_eps = h_eps / np.sum(h_eps) * (n_points / 2.0)
                c5_plus = objective(h_eps)
                
                h_eps[i] -= 2*eps
                h_eps = np.clip(h_eps, 0, 1)
                h_eps = h_eps / np.sum(h_eps) * (n_points / 2.0)
                c5_minus = objective(h_eps)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            h = h - 0.001 * grad
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
