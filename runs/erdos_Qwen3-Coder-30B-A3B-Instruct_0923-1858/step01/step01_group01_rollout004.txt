import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_h(h0, dx, max_iter=10000, time_budget=580):
    """Optimize h using L-BFGS-B with custom gradient."""
    best_h = h0.copy()
    best_c5 = evaluate(h0, dx)
    
    def objective(h):
        return evaluate(h, dx)
    
    def jac(h):
        grad = np.zeros_like(h)
        eps = 1e-8
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        return grad

    start_time = time.time()
    
    # Use scipy's L-BFGS-B method
    res = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        jac=jac,
        bounds=[(0, 1) for _ in range(len(h0))],
        options={'maxiter': max_iter, 'ftol': 1e-10},
        callback=lambda x: None
    )
    
    if not res.success:
        print(f"Optimization failed: {res.message}")
    
    if time.time() - start_time > time_budget:
        return best_h, best_c5
    
    h_opt = project(res.x, dx)
    c5_opt = evaluate(h_opt, dx)
    
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt
    
    return best_h, best_c5
