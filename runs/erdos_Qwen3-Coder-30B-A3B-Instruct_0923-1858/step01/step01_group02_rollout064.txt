import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Initialize with a loose upper bound
    c5_bound = 0.4
    tolerance = 1e-4
    step_size = 0.0001
    max_iter = 1000
    iter_count = 0
    
    h_best = h.copy()
    c5_best = evaluate_erdos_solution(h_best)
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Create optimization variables
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        dx = 2.0 / n_points
        for k in range(n_points):
            # Convolution constraint: sum h[i] * (1 - h[i+k]) <= c5_bound / dx
            if k == 0:
                # Direct constraint on h[i] * (1 - h[i])
                expr = sum(h_var[i] * (1 - h_var[i]) for i in range(n_points))
            else:
                # Convolution constraint
                expr = 0
                for i in range(n_points - k):
                    expr += h_var[i] * (1 - h_var[i + k])
                # Handle boundary conditions
                for i in range(n_points - k, n_points):
                    expr += h_var[i] * (1 - h_var[(i + k) % n_points])
            
            constraints.append(expr <= c5_bound / dx)
            
        # Optimization problem
        objective = Minimize(0)  # No actual objective, just feasibility
        prob = Problem(objective, constraints)
        
        try:
            # Solve the problem
            prob.solve(solver=ECOS, verbose=False)
            
            # Check if we found a feasible solution
            if prob.status in [OPTIMAL, FEASIBLE]:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                c5_new = evaluate_erdos_solution(h_new)
                
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new
                    # Reduce c5 to see if we can do better
                    c5_bound = max(c5_best - step_size, 0.0)
                else:
                    # If no improvement, gradually increase the bound
                    c5_bound += step_size
            else:
                # If infeasible, increase the bound
                c5_bound += step_size
        except:
            # If CVXPY fails, try a heuristic approach
            try:
                # Try a simple gradient descent approach with projection
                h_new = h_best.copy()
                # Simple random walk with gradient-like steps
                pert = np.random.normal(0, 0.01, n_points)
                h_new += pert
                h_new = np.clip(h_new, 0, 1)
                # Renormalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new
                    c5_bound = max(c5_best - step_size, 0.0)
                else:
                    c5_bound += step_size
            except:
                pass
                
        # Early exit if we're very close to target
        if abs(c5_best - 0.3807) < tolerance:
            break
            
    # Final validation
    final_c5 = evaluate_erdos_solution(h_best)
    
    return (h_best.tolist(), float(final_c5), n_points)
