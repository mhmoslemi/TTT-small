import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure the integral constraint is satisfied exactly
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Build the optimization problem
    # Variables
    h_vars = Variable(n_points)
    t = Variable()
    
    # Parameters
    dx = 2.0 / n_points
    h0 = h.copy()
    
    # Constraint: h_vars >= 0, h_vars <= 1
    constraints = [h_vars >= 0, h_vars <= 1]
    
    # Constraint: sum(h_vars) = n_points / 2
    constraints.append(sum(h_vars) == n_points / 2)
    
    # Constraint: t >= overlap_k for all k
    # Manually compute the overlap for all valid shifts
    # Only compute for shifts where correlation is defined
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # This corresponds to h[i] * (1 - h[i+k]) for i = 0,...,n_points-1-k
            # We sum over i from 0 to n_points-1-k
            corr_expr = sum(h_vars[i] * (1 - h_vars[i+k]) for i in range(n_points - k))
        else:
            # k < 0, so this is h[i+(-k)] * (1 - h[i]) for i = 0,...,n_points+k-1
            corr_expr = sum(h_vars[i+(-k)] * (1 - h_vars[i]) for i in range(n_points + k))
        constraints.append(corr_expr <= t * dx)
    
    # Objective: minimize t
    objective = Minimize(t)
    
    # Formulate the problem
    prob = Problem(objective, constraints)
    
    # Solver settings
    solver = ECOS
    max_iter = 3
    mu = 0.1
    
    # Iterative improvement
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        try:
            if time.time() - start_time > budget_s - 1:
                break
                
            # We'll do a few iterations of the QP solver, 
            # Solving with increasingly tight barrier parameter
            prob.solve(solver=solver, max_iters=500, verbose=False)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_vars.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                
                h = h_new.copy()
            else:
                break
        except:
            break
        
        # Update parameters for next iteration
        mu *= 0.5
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
