import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution
    n_points = 140
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values or create a reasonable starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Linearly interpolate to desired resolution
        old_x = np.linspace(0, 2, len(h))
        new_x = np.linspace(0, 2, n_points)
        h = np.interp(new_x, old_x, h)
    else:
        # Initialize with a simple piecewise constant function with 50% density
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping style global search
    max_iter = 200  # About 200 iterations to keep under budget
    current_h = h.copy()
    
    # Perturbation scaling
    sigma = 0.05
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        h_trial = current_h + np.random.normal(0, sigma, n_points)
        # Clip and rescale
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)
        
        # Local optimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    current_h = h_new.copy()
        except:
            pass
            
        # Gradually decrease perturbation
        if i % 50 == 0:
            sigma /= 2.0
    
    # Refine top candidates
    top_candidates = [best_h]
    for _ in range(5):
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
    
    # Ensure final feasibility
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2.0) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
