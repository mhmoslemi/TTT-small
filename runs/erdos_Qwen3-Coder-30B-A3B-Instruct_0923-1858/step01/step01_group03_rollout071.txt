import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Trust region optimization setup
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def trust_region_step(h_vec, max_iter=20):
        # Simple trust region approach with projected gradient descent
        h_current = h_vec.copy()
        radius = 0.3
        eps = 1e-6
        
        for _ in range(max_iter):
            # Compute gradient approximatly using finite differences
            grad = np.zeros_like(h_current)
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                max_k_plus = np.argmax(corr_plus)
                max_k_minus = np.argmax(corr_minus)
                
                grad[i] = (corr_plus[max_k_plus] - corr_minus[max_k_minus]) / (2 * eps)
            
            # Try a step
            h_new = h_current - 0.1 * grad
            h_new = project_feasible(h_new)
            
            # Check if step is acceptable
            if compute_C5(h_new) < compute_C5(h_current):
                h_current = h_new
            else:
                break
                
        return h_current
    
    # Main loop with perturbations and trust region steps
    max_patience = 50
    patience = 0
    iter_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
                
        # Trust region update
        h = trust_region_step(h, 10)
        
        iter_count += 1
        
        # Occasionally do a global perturbation
        if iter_count % 50 == 0:
            for _ in range(n_points // 5):
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
