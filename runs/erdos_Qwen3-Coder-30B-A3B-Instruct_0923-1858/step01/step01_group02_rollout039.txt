import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = 100
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    
    # Start with a high initial bound
    C5_bound = 0.4
    step_size = 0.0001

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Iterative tightening approach using CVXPY
    max_iterations = 1000
    iter_count = 0
    last_improvement = 0
    
    # Create variable for optimization
    h_var = cp.Variable(n_points)
    
    # Define constraints
    constraints = [h_var >= 0, h_var <= 1]
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    solver = cp.CLARABEL
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        last_improvement += 1
        iter_count += 1
        
        # Compute the required constraints
        # For each possible shift k, add a constraint on the overlap integral
        # We are actually solving the dual problem: minimize C5 such that
        # all overlap integrals are bounded by C5*dx
        
        current_constraints = constraints[:]
        
        # Create constraints for each shift k in the full correlation range
        # This is a conservative approach, but simpler to manage
        # We'll approximate with a finite set of shifts
        
        # For our purposes, discretize k in [0, 2] with sufficient density
        # As 2 dx = 2/n_points, we use shift indices from 0 to 2*n_points (full correlation)
        # But since our h is of length n_points, to cover all meaningful non-zero...
        shifted_indices = range(n_points)  # e.g., [0,1,2,...,n-1] for a relevant subset
        # We'll include constraints for a subset of possible shifts
        # For conservative approach, try first few shifts
        for shift in range(min(20, n_points)):  # Try a few important shifts
            if shift < n_points:
                # i -> i + k, so constraint becomes sum h[i] * (1 - h[i+k]) <= C5 * dx
                # The constraint is more complex for cvxpy due to the discrete nature
                expr = 0
                for i in range(n_points):
                    if i + shift < n_points:
                        expr += h_var[i] * (1 - h_var[i + shift])
                current_constraints.append(expr <= C5_bound / dx)
            else:
                break

        # Solve the optimization
        objective = cp.Minimize(cp.sum(h_var))  # dummy objective, we're just checking feasibility
        prob = cp.Problem(objective, current_constraints)
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # If we can find a feasible solution, try shrinking C5
                new_h = h_var.value
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    # Project back to correct L1 norm
                    new_h = new_h * (n_points / 2) / np.sum(new_h)
                    test_c5 = evaluate_erdos_solution(new_h)
                    if test_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = test_c5
                        last_improvement = 0
                    # If we're able to produce a feasible solution with current C5,
                    # try decreasing it
                    C5_bound = max(0.0001, C5_bound - step_size)
                else:
                    # Failed to solve or get feasible solution
                    C5_bound = min(0.5, C5_bound + step_size)  # relax
            else:
                # Problem is infeasible, we need to increase C5
                C5_bound = min(0.5, C5_bound + step_size)

            step_size = max(0.00001, step_size * 0.5)
            last_improvement = 0
        
        # Early termination if not making meaningful progress
        if C5_bound < 0.001:
            break

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
