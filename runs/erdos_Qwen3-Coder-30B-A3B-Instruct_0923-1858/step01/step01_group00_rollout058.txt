import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.1
    alpha_decay = 0.95
    patience = 100
    perturb_prob = 0.1
    
    # Load or construct initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    
    h = h / np.sum(h) * (n_points / 2)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    last_improvement = 0
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current C5 and get max lag
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        max_lag_idx = np.argmax(corr)
        
        # Compute gradient w.r.t. h  
        # ∇(max_k h[i] * (1 - h[i + k])) = (1 - 2*h[i]) for i = argmax
        grad = np.zeros(n_points)
        # Gradient contribution from the maximum correlation lag
        k = max_lag_idx - (n_points - 1)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] = 1 - 2 * h[j]
        
        # Clip gradient to avoid instability at boundaries
        grad = np.clip(grad, -1.0, 1.0)
        
        # Gradient step
        h_new = h - alpha * grad
        
        # Project onto feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        h = h_new
        
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = i
        
        # Perturb if no improvement recently
        if i - last_improvement > patience and np.random.random() < perturb_prob:
            # Add noise and smooth
            h = h + np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            # Apply median filter to reduce jitter
            h = signal.medfilt(h, kernel_size=3)
        
        # Adaptive learning rate
        if i - last_improvement > 200:
            alpha = 0.1
        elif i % 100 == 0:
            alpha *= alpha_decay
            
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
