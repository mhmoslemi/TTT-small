import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        n_points = 100
        h = np.ones(n_points) * 0.5
        # Adjust a few points to create a non-uniform distribution
        h[0] = 0.75
        h[1] = 0.25
        h = h * (n_points / 2) / np.sum(h)

    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    dx = 2.0 / n_points
    
    # Track the iteration counter and early termination
    iteration = 0

    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # Compute current overlaps and find max overlap
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_overlap_idx = np.argmax(corr)
        max_overlap = corr[max_overlap_idx]
        
        # Extract the maximum contributing shift
        max_k = max_overlap_idx - n_points + 1  # Shift corresponding to max overlap
        
        # Create a cutting plane: add constraint that O_k(h) <= max_overlap
        # For constructing the gradient at best_h for the given k:
        # We consider for i=0..n_points-1, the term h[i]*(1 - h[(i+k)%n_points])
        # Gradient is then: 
        #  d/dh_i O_k(h) = 1 - h[(i+k)%n_points] if k >= 0
        #  d/dh_i O_k(h) = 1 - h[(i+k+n_points)%n_points] if k < 0
        
        # Let's handle the gradient computation properly for general k:
        gradient = np.zeros(n_points)
        # For coefficient of h[i], the contribution comes from the term h[i] * (1 - h[(i + k) % n_points])  
        # So gradient[i] = 1 - h[(i + k) % n_points]
        for i in range(n_points):
            if (i + max_k) < n_points:
                gradient[i] = 1 - best_h[i + max_k]
            else:
                gradient[i] = 1 - best_h[i + max_k - n_points]

        # Now solve the convex subproblem minimized over C
        h_vars = cp.Variable(n_points)
        C = cp.Variable(1)
        
        # Objective: minimize C
        objective = cp.Minimize(C)
        
        # Constraints:
        # 1. Convex cutting plane constraint
        #   gradient.T @ h_vars <= C - gradient.T @ best_h + max_overlap
        cut_constraint = gradient.T @ h_vars <= C - gradient.T @ best_h + max_overlap
        
        # 2. Box constraints
        box_constraints = [0 <= h_vars, h_vars <= 1]
        
        # 3. Sum constraint
        sum_constraint = cp.sum(h_vars) == n_points / 2
        
        # 4. Constraint that C >= max_overlap
        c_constraint = C >= max_overlap
        
        constraints = [cut_constraint] + box_constraints + [sum_constraint] + [c_constraint]
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                C_val = C.value
                if C_val < best_c5:
                    best_h = h_new.copy()
                    best_c5 = C_val
        except:
            pass
        
        # Add some randomness occasionally
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.02, n_points)
            h_new = best_h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_test = evaluate_erdos_solution(h_new)
            if c5_test < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_test
                        
        # Early exit if we're stagnating
        if iteration > 1000 and abs(best_c5 - evaluate_erdos_solution(best_h)) < 1e-8:
            break

    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
