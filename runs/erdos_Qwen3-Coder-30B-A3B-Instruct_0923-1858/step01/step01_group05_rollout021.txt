import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular function centered at x = 1
        n_points = 70
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(2 * x - 1)
        h = h / np.sum(h) * (n_points / 2.0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation parameters
    perturbation_strength = 0.03
    critical_region = (0.3, 0.7)
    
    # Optimization loop with time budget
    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb in critical region
        h_trial = h.copy()
        indices = np.where((np.linspace(0, 2, n_points) >= critical_region[0]) & 
                           (np.linspace(0, 2, n_points) <= critical_region[1]))[0]
        
        # Apply Gaussian perturbation
        noise = np.random.normal(0, perturbation_strength, len(indices))
        h_trial[indices] += noise
        
        # Repair feasibility
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Local refinement with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            corr = np.correlate(h, 1-h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        # Simple gradient-like step for continued search
        if iter_count % 10 == 0:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
            
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
