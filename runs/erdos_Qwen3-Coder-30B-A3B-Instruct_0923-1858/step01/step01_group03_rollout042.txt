import time

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # PSO parameters
    Pop_size = 100
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_iter = 1000
    stagnation_limit = 50
    mutation_rate = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points) * 0.1)
    
    # Add random solutions
    for _ in range(Pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points) * 0.1)
    
    pbest = [swarm[i].copy() for i in range(Pop_size)]
    pbest_scores = [float('inf')] * Pop_size
    gbest_idx = 0
    gbest_score = float('inf')
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Evaluate initial swarm
    for i in range(Pop_size):
        swarm[i] = project_feasible(swarm[i])
        score = compute_C5(swarm[i])
        if score < pbest_scores[i]:
            pbest_scores[i] = score
            pbest[i] = swarm[i].copy()
        if score < gbest_score:
            gbest_score = score
            gbest_idx = i
    
    # Main PSO loop
    iter_count = 0
    stagnation = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Update inertial weight
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(Pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (c1 * r1 * (pbest[i] - swarm[i]) + 
                         c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            velocities[i] = w * velocities[i] + vel_update
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate fitness
            score = compute_C5(swarm[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = swarm[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest_idx = i
                    stagnation = 0
                    
        stagnation += 1
        if stagnation > stagnation_limit:
            # Reset some particles
            for _ in range(int(Pop_size * mutation_rate)):
                idx = np.random.randint(Pop_size)
                swarm[idx] = np.random.rand(n_points)
                swarm[idx] = swarm[idx] * (n_points / 2) / np.sum(swarm[idx])
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points) * 0.1
            stagnation = 0
            
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(swarm[gbest_idx])
    if final_C5 < gbest_score:
        gbest_score = final_C5
        best_h = swarm[gbest_idx].copy()
    
    return (best_h.tolist(), gbest_score, len(best_h))
