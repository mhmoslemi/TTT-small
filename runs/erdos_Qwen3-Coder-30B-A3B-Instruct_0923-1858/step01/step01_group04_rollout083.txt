import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure initial feasibility
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Use adaptive resolution strategy
    resolutions = [50, 100]
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization budget tracking
    time_budget = budget_s
    for res in resolutions:
        if time.time() - start_time > time_budget - 5:
            break
            
        # Create new h with desired resolution
        if res > n_points:
            # Upsample
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h)
        else:
            # Downsample or keep
            h_new = h[:res] if res < n_points else h.copy()
        
        # Ensure it's feasible
        h_new = h_new * (res / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Local optimization using different starting points
        local_best_h = h_new.copy()
        local_best_c5 = evaluate_erdos_solution(local_best_h)
        
        # Multiple local searches from different starting points
        max_iters = 10
        for iter_num in range(max_iters):
            if time.time() - start_time > time_budget - 1:
                break
                
            # Try perturbed version
            noise = np.random.normal(0, 0.05, res)
            h_trial = local_best_h + noise
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (res / 2) / np.sum(h_trial)
            
            # Optimization with bounds and constraints
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec)
                
            def constraint_integral(h_vec):
                return np.sum(h_vec) - res / 2
                
            bounds = [(0, 1) for _ in range(res)]
            constraints = {'type': 'eq', 'fun': constraint_integral}
            
            try:
                res_opt = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res_opt.success:
                    h_candidate = np.clip(res_opt.x, 0, 1)
                    h_candidate = h_candidate * (res / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < local_best_c5:
                        local_best_h = h_candidate
                        local_best_c5 = c5_candidate
                        
                        # Update global best if improved
                        if local_best_c5 < best_c5:
                            best_h = local_best_h.copy()
                            best_c5 = local_best_c5
            except Exception:
                pass

    # Final refinement
    if time.time() - start_time < time_budget - 5:
        # One final global refinement
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec)
            
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        # Try several random perturbations
        for _ in range(5):
            if time.time() - start_time > time_budget - 1:
                break
            noise = np.random.normal(0, 0.02, n_points)
            h_trial = best_h + noise
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            try:
                res_opt = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res_opt.success:
                    h_candidate = np.clip(res_opt.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                continue
    
    # Ensure final feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
