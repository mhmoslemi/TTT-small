import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h * (n_points / 2) / np.sum(h), n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
    
    # Initialize PSO variables
    swarm_size = 50
    particles = np.random.rand(swarm_size, n_points) * 0.5
    for i in range(swarm_size):
        particles[i] = project_to_feasible(particles[i], n_points)
    
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.2
    personal_best = particles.copy()
    personal_best_values = np.array([evaluate_erdos_solution(p) for p in personal_best])
    
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    alpha = 0.9  # Low-pass filter parameter
    
    start_time = time.time()
    last_improvement = start_time
    stagnation_count = 0
    
    while time.time() - start_time < budget_s:
        try:
            for i in range(swarm_size):
                # Update velocities with inertia and cognitive/social components
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best[i] - particles[i]) + 
                                c2 * r2 * (global_best - particles[i]))
                
                # Apply low-pass filter to velocity
                velocities[i] = alpha * velocities[i] + (1 - alpha) * np.roll(velocities[i], 1)
                
                # Update position
                particles[i] = particles[i] + velocities[i]
                
                # Project to feasible region
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                current_value = evaluate_erdos_solution(particles[i])
                if current_value < personal_best_values[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_values[i] = current_value
                    if current_value < global_best_value:
                        global_best = particles[i].copy()
                        global_best_value = current_value
                        last_improvement = time.time()
                        stagnation_count = 0
                        
            # Perturb particles slightly to prevent stagnation
            perturbation = np.random.normal(0, 0.01, (swarm_size, n_points))
            particles = np.clip(particles + perturbation, 0, 1)
            for i in range(swarm_size):
                particles[i] = project_to_feasible(particles[i], n_points)
                
            # Early stopping if no improvement for long time
            if time.time() - last_improvement > 30:
                stagnation_count += 1
                if stagnation_count > 10:
                    break
            
        except Exception:
            continue
            
    # Final evaluation
    try:
        final_c5 = evaluate_erdos_solution(global_best)
    except:
        final_c5 = global_best_value
        
    return (global_best, final_c5, n_points)
