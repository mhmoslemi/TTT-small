import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 50
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint and clip
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Multi-resolution search: first coarse, then fine
    resolutions = [50, 100]
    if n_points < 50:
        resolutions = [50]
    elif n_points > 100:
        resolutions = [100]
    else:
        resolutions = [n_points]
        
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate population of candidates using random perturbations
        population_size = 20
        population = []
        for i in range(population_size):
            h_trial = h + np.random.normal(0, 0.1, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_val = evaluate_erdos_solution(h_trial)
            population.append((h_trial, c5_val))
            
        # Sort and keep top candidates
        population.sort(key=lambda x: x[1])
        top_candidates = population[:5]
        
        # Refine using local optimization on top candidates
        for h_candidate, _ in top_candidates:
            if time.time() - start_time > budget_s - 1:
                break
            
            # Local optimization with SLSQP
            def objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                return evaluate_erdos_solution(h_vec)
                
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_integral}
            
            try:
                res = minimize(
                    objective,
                    h_candidate.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    refined_h = np.clip(res.x, 0, 1)
                    refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                    c5_candidate = evaluate_erdos_solution(refined_h)
                    if c5_candidate < best_c5:
                        best_h = refined_h
                        best_c5 = c5_candidate
                        
            except Exception:
                pass

    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)

    return (list(final_h), final_c5, n_points)
