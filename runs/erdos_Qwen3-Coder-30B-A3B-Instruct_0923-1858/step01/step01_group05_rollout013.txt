import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 140
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Initialize with a uniform distribution
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h / np.sum(h) * (n_points / 2.0)
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Global exploration with perturbations
    perturbation_strength = 0.05
    max_iter = 1000
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the current solution
        h_perturbed = h + np.random.normal(0, perturbation_strength, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2.0)
        
        # Evaluate this perturbation
        c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        # Update best if better
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
            h = h_perturbed.copy()
            
        # Local refinement on top solutions
        if iter_count % 20 == 0 and iter_count > 0:
            # Get top 10 candidates from recent perturbations
            candidates = []
            for _ in range(10):
                if time.time() - start_time > budget_s - 1:
                    break
                h_trial = h + np.random.normal(0, perturbation_strength, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
                c5_trial = evaluate_erdos_solution(h_trial)
                candidates.append((h_trial, c5_trial))
            
            # Optimize the best candidates
            for h_trial, c5_trial in sorted(candidates, key=lambda x: x[1])[:5]:
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Define bounds and constraint
                bounds = [(0, 1) for _ in range(n_points)]
                
                def constraint_sum(h_var):
                    return np.sum(h_var) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h_var):
                    corr = np.correlate(h_var, 1-h_var, mode="full")
                    return np.max(corr) * dx
                
                try:
                    res = minimize(objective, h_trial, method='SLSQP', 
                                 bounds=bounds, constraints=cons, 
                                 options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_new = res.x
                        c5_new = objective(h_new)
                        if c5_new < best_c5:
                            best_h = h_new.copy()
                            best_c5 = c5_new
                            h = h_new.copy()
                except:
                    pass
                    
        # Reduce perturbation strength occasionally
        if iter_count % 50 == 0 and iter_count > 0:
            perturbation_strength *= 0.5
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
