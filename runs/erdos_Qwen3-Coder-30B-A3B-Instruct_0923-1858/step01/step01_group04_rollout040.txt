import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Scale to fit budget
    max_iter = 300
    iter_count = 0
    
    # Optimization iterations
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try different strategies: random local search and greedy improvement
        
        # Strategy 1: Local search around current best
        h_trial = best_h + np.random.normal(0, 0.02, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # SLSQP optimization with modified constraints
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec)
            
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        try:
            res = minimize(
                objective,
                h_trial,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except:
            pass
        
        # Strategy 2: Greedy removal of worst overlaps
        # Compute overlaps for all shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        
        # Get the worst shift
        shift = max_idx - (len(best_h) - 1)
        
        # Try to adjust the function to reduce this overlap
        # Distribute adjustment to reduce the contribution of h[i] * (1 - h[i+shift])
        h_new = best_h.copy()
        if shift > 0:
            for i in range(len(best_h) - shift):
                # Reduce h[i] and increase h[i+shift] a bit to improve the overlap
                # In a weighted fashion
                if best_h[i] > 0.01 and best_h[i+shift] < 0.99:
                    adjust = min(best_h[i] * 0.01, (1 - best_h[i+shift]) * 0.01)
                    if adjust > 1e-6:
                        h_new[i] -= adjust
                        h_new[i+shift] += adjust
        else:
            for i in range(-shift, len(best_h)):
                if best_h[i] > 0.01 and best_h[i+shift] < 0.99:
                    adjust = min(best_h[i] * 0.01, (1 - best_h[i+shift]) * 0.01)

                        h_new[i] -= adjust
                        h_new[i+shift] += adjust

        h_new = np.clip(h_new, 0, 1)
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            
        iter_count += 1
    
    # Final check
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (list(final_h), final_c5, n_points)
