import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluates the Erdős constant C_5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 70
        h = np.ones(n_points) * 0.5
        # Slightly perturb to make it more useful
        h[0] = 0.7
        h[1] = 0.3
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    # Optimization settings
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Adaptive perturbation schedule
    perturbation_magnitude = 0.1
    improvement_count = 0
    stagnation_count = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        try:
            # Perturb the current solution
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            x0 = np.clip(best_h + perturbation, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)  # Renormalize
            
            # Use L-BFGS-B which benefits from gradient information
            res = minimize(objective, x0, method='L-BFGS-B', bounds=[(0,1)]*n_points, 
                           options={'maxiter': 100, 'gtol': 1e-6})
            
            if res.success:
                current_c5 = evaluate_erdos_solution(res.x)
                if current_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = current_c5
                    improvement_count += 1
                    stagnation_count = 0
                    # Reduce perturbation if there's consistent improvement
                    if improvement_count > 10:
                        perturbation_magnitude *= 0.95
                        improvement_count = 0
                else:
                    stagnation_count += 1
                    # Increase perturbation if no improvement
                    if stagnation_count > 10:
                        perturbation_magnitude *= 1.2
                        stagnation_count = 0
            else:
                # Fallback to previous best if optimization fails
                pass
                
        except Exception:
            # Fallback to previous best solution
            pass
            
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Occasionally just restart with a better random solution
        if iteration % 100 == 0:
            # Try a completely random solution
            new_h = np.random.rand(n_points)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
