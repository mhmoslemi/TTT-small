import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Number of points
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to required size
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        # Normalize to sum(h) = n_points/2
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Annealing parameters
    temperature = 1.0
    cooling_rate = 0.999
    perturbation_scale = 0.1
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # For early stopping
    stagnation_count = 0
    max_stagnation = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Perturb
        delta = np.random.uniform(-temperature * perturbation_scale, 
                                   temperature * perturbation_scale, n_points)
        h_new = h + delta
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject
        delta_c5 = c5_new - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
        
        # Apply cooling
        temperature *= cooling_rate
        
        # Shake occasionally if stagnating
        if stagnation_count >= max_stagnation:
            h += np.random.normal(0, temperature * perturbation_scale, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            stagnation_count = 0
    
    return (best_h, best_c5, n_points)
