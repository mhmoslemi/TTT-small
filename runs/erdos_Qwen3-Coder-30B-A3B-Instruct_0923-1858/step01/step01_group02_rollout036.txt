import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)

    # Initial relaxed upper bound
    C5 = 0.4
    dx = 2.0 / n_points
    step_size = 0.0001
    failed_iterations = 0
    max_failed_iterations = 100
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    while time.time() - start_time < budget_s - 1:
        # Try to find h that meets constraints
        # Build optimization problem
        h_var = Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(sum(h_var) == n_points / 2)
        
        # Apply the constraints for each possible k in the correlation
        corr_len = 2 * n_points - 1
        for k in range(corr_len):
            # indices in h of the valid overlapping positions
            if k < n_points:
                # k = 0 to n_points-1
                left_indices = slice(0, n_points-k)
                right_indices = slice(k, n_points)
            else:
                # k = n_points to 2*n_points-2
                left_indices = slice(k - n_points + 1, n_points)
                right_indices = slice(0, n_points - (k - n_points + 1))
            
            if k < n_points:
                overlap = sum(h_var[left_indices] * (1 - h_var[right_indices]))
            else:
                overlap = sum(h_var[left_indices] * (1 - h_var[right_indices]))
                
            constraints.append(overlap <= C5 / dx)
        
        # Solve optimization problem
        objective = Minimize(0)  # No actual objective, just satisfy constraints
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                h_sol = project_to_feasible(h_sol, n_points)
                current_C5 = evaluate_erdos_solution(h_sol)
                
                if current_C5 < best_C5:
                    best_h = h_sol.copy()
                    best_C5 = current_C5
                    
                # Update C5 based on success/failure
                C5 = max(C5 - step_size, 0)
                failed_iterations = 0
            else:
                C5 = C5 + step_size
                failed_iterations += 1
        except:
            C5 = C5 + step_size
            failed_iterations += 1
            
        if failed_iterations > max_failed_iterations:
            break
            
        if C5 < 0.0001:
            break  # Prevent extreme small values
            
    return (best_h.tolist(), float(best_C5), n_points)
