import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try different initialization strategies
    strategies = [
        lambda: np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5,
        lambda: np.clip(np.random.normal(0.5, 0.2, n_points), 0, 1),
        lambda: np.ones(n_points) * 0.5 + np.sin(np.linspace(0, 4*np.pi, n_points)) * 0.1,
    ]
    
    def project_h(h_vec):
        """Project h onto feasible set: sum(h) = n_points/2, 0 <= h <= 1"""
        h_vec = np.clip(h_vec, 0, 1)
        scale = (n_points / 2) / np.sum(h_vec)
        h_vec = h_vec * scale
        return np.clip(h_vec, 0, 1)
    
    for strategy in strategies:
        if time.time() - start_time > budget_s - 5:
            break
            
        h_start = strategy()
        h_start = project_h(h_start)
        
        # Local optimization using SLSQP
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec)
        
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        try:
            res = minimize(
                objective,
                h_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if res.success:
                h_candidate = project_h(res.x)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
    
    # Perform simulated annealing refinement on the best so far
    current_h = best_h.copy()
    current_c5 = best_c5
    temperature = 0.1
    cooling_rate = 0.995
    min_temperature = 1e-6
    max_iter = 5000
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        if temperature < min_temperature:
            break
            
        # Generate neighbor solution
        neighbor_h = current_h + np.random.normal(0, temperature * 0.1, n_points)
        neighbor_h = project_h(neighbor_h)
        
        # Compute overlap for neighbor
        neighbor_c5 = evaluate_erdos_solution(neighbor_h)
        
        # Accept or reject
        if neighbor_c5 < current_c5 or np.random.random() < np.exp(-(neighbor_c5 - current_c5) / temperature):
            current_h = neighbor_h
            current_c5 = neighbor_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        temperature *= cooling_rate
        iter_count += 1
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
