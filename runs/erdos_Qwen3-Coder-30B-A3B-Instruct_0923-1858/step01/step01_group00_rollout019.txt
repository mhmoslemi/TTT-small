import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initialize with the provided initial construction if available

    else:

        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Use L-BFGS-B for optimization with bounds but no constraint
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Bounds for h: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Try different initialization strategies and run optimization
    for attempt in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local perturbation starting_point
        if attempt == 0:
            h_start = h.copy()
        else:
            h_start = np.random.rand(n_points) * 0.5
            h_start = h_start / np.sum(h_start) * (n_points / 2)
            
        try:
            result = minimize(
                objective,
                h_start,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200},
                tol=1e-6
            )
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        except Exception:
            continue
            
        # More aggressive refinement step
        h_refine = best_h.copy()

        try:
            result = minimize(
                objective,
                h_refine,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100},
                tol=1e-6
            )
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        except Exception:
            continue

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure values are within constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)

    return (best_h, final_c5, n_points)
