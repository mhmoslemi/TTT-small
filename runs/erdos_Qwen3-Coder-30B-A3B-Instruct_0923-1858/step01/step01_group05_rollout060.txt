import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Start with a smooth construction
        n_points = 140
        x = np.linspace(0, 2, n_points)
        # Try a periodic-like shape to start with
        h = 0.5 + 0.3 * np.sin(np.pi * x) * np.cos(np.pi * x / 2)
        h = np.clip(h, 0, 1)
        # Normalize the sum to 70 (since 2/140 = 1/70)
        h = h * 70 / np.sum(h)

    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - 70.0  # n_points / 2.0 where n_points = 140
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Global exploration
    iter_count = 0
    perturbation_strength = 0.05
    while time.time() - start_time < budget_s - 1:
        # Perturb and look for better solutions
        h_perturbed = h + np.random.normal(0, perturbation_strength, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * 70.0
        
        # Try to refine this with local optimization
        try:
            # Use SLSQP for local improvement
            res = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        # Update h with the perturbed version
        h = h_perturbed.copy()
        if iter_count % 5 == 0:
            # Slightly reduce perturbation strength over time
            perturbation_strength *= 0.995
        
        iter_count += 1
    
    # Final validation 
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
