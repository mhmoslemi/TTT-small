import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def gradient_estimate(h, dx, epsilon=1e-8):
    """Estimate gradient of C5 w.r.t. each h[i]."""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project(h_minus, dx)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)
    return grad

def optimize_h(initial_h, dx, n_points, max_time=580):
    """Optimize h using L-BFGS-B with adaptive learning rate."""
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h, dx)
    
    start_time = time.time()
    lr = 0.01
    
    def objective(h):
        h = project(h, dx)
        return evaluate(h, dx)
    
    def grad(h):
        h = project(h, dx)
        return gradient_estimate(h, dx)
        
    # Use scipy's L-BFGS-B which handles bounds efficiently
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        jac=grad,
        options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 5000}
    )
    
    if result.success:
        final_h = project(result.x, dx)
        final_c5 = evaluate(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    # Fine-tune with additional iterations if time allows
    h = best_h.copy()
    for i in range(2000):
        if time.time() - start_time > max_time:
            break
        grad_h = gradient_estimate(h, dx)
        h = h - lr * grad_h
        h = project(h, dx)
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if i % 100 == 0:
            lr *= 0.95
            
    return best_h, best_c5

        phase = np.abs(np.sin(np.pi * i / (n_points / 3))) ** 3
        h[i] = np.clip(2 * phase - 0.5, 0, 1)
    return h
