import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Binary search between these values
    c_low = 0.0
    c_high = 0.38085
    
    # Define the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Read initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Rescale to match n_points
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(initial_h_values)), h)
        h = h_interp
    else:
        # Create uniform distribution with small perturbations
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        
    # Normalize to sum to 50
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Perform binary search
    while c_high - c_low > 1e-5 and (time.time() - start_time) < budget_s * 0.9:
        c_mid = (c_low + c_high) / 2.0
        
        # Gradient-based refinement if we're in a loose bound
        if c_mid > 0.37:
            # Create a modified objective that penalizes the max overlap
            def modified_objective(h_vec):
                # Ensure h is within bounds
                h_vec = np.clip(h_vec, 0, 1)
                # Normalize sum to be correct
                h_vec = h_vec / np.sum(h_vec) * (n_points / 2)
                return evaluate_erdos_solution(h_vec)
            
            # Use a simple refinement strategy
            # Start with best solution and search for better solution around it
            current_h = best_h.copy()
            # Perturb the current solution a bit
            perturb = np.random.normal(0, 0.002, n_points)
            current_h = current_h + perturb
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h / np.sum(current_h) * (n_points / 2)
            
            # Do a local optimization
            try:
                def objective_func(h_v):
                    return modified_objective(h_v)
                    
                result = minimize(objective_func, current_h.copy(), method='L-BFGS-B',
                                bounds=[(0, 1) for _ in range(n_points)],
                                options={'maxiter': 50}, tol=1e-6)
                
                if result.success:
                    current_c5 = evaluate_erdos_solution(result.x)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()
            except:
                pass
        
        # After refinement, do brute force search
        try:
            # Evaluate
            current_c5 = evaluate_erdos_solution(best_h)
            
            # Check if we can decrease c_high
            if current_c5 <= c_mid:
                c_high = c_mid
                # Keep current best
            else:
                c_low = c_mid
                
        except Exception as e:
            # If anything fails, pass
            c_low = c_mid
            
    # Final optimization pass
    best_solver_h = best_h.copy()
    best_solver_c5 = best_c5
    
    # Run a final optimization if time allows
    if (time.time() - start_time) < budget_s * 0.95:
        def ref_objective(h_new):
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            return evaluate_erdos_solution(h_new)
            
        try:
            # Only refine if we're close to target
            if best_solver_c5 > 0.37:
                result = minimize(ref_objective, best_solver_h.copy(), 
                                method='L-BFGS-B', 
                                bounds=[(0,1) for _ in range(n_points)],
                                options={'maxiter': 100}, tol=1e-6)
                
                if result.success:
                    final_c5 = evaluate_erdos_solution(result.x)
                    if final_c5 < best_solver_c5:
                        best_solver_c5 = final_c5
                        best_solver_h = result.x.copy()
        except:
            pass
                    
    return (best_solver_h, best_solver_c5, n_points)
