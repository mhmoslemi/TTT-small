import time

        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    dx = 2.0 / n_points
    best_h = h_init.copy()
    best_C5 = float('inf')
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.2
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Create initial population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        # Create random feasible individual
        h = np.random.rand(n_points)
        h = project_feasible(h)
        population.append(h)
    
    # Main GA loop
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Evaluate fitness of all individuals
        fitness_scores = []
        for h in population:
            fit = compute_C5(h)
            fitness_scores.append(fit)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_C5:
            best_C5 = fitness_scores[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:elite_size]  # Keep elite
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament_indices = np.random.choice(len(population), 
                                                    tournament_size, 
                                                    replace=False)
                tournament_fitness = [fitness_scores[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(population[winner_idx])
            
            # Crossover (uniform)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parents[0][i]
                else:
                    child[i] = parents[1][i]
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Add Gaussian noise and project to feasible region
                noise = np.random.normal(0, 0.03, n_points)
                child = child + noise
                child = project_feasible(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation of best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
