import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation to get better starting point
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        constraints = []
        # Constraint for each lag k
        for k in range(n_points):
            # For lag k, we compute sum_i h[i] * (1 - h[i+k])
            # Index wraparound handled manually
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr >= lambda_var)
            
        # Feasibility constraints
        constraints.append(cp.sum(h_var) == n_points / 2)
        for i in range(n_points):
            constraints.append(h_var[i] >= 0)
            constraints.append(h_var[i] <= 1)
            
        prob = cp.Problem(cp.Maximize(lambda_var), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                return h_var.value
        except:
            pass
        return None
    
    # Try convex relaxation
    if n_points <= 60:  # Only try convex relaxation for smaller problems
        h_convex = solve_convex_relaxation()
        if h_convex is not None:
            h = h_convex.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Gradient descent with better initialization
    eta = 0.01
    iter_count = 0
    patience = 0
    max_patience = 50
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad
    
    # Main optimization loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random restart
                h = np.random.rand(n_points) * 0.5
                h = project_feasible(h)
                patience = 0
    
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1
    
    # Random restarts for high quality
    num_restarts = 10
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
        h_restart = np.random.rand(n_points) * 0.8
        h_restart = project_feasible(h_restart)
        C5_restart = compute_C5(h_restart)
        if C5_restart < best_C5:
            best_C5 = C5_restart
            best_h = h_restart.copy()
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
