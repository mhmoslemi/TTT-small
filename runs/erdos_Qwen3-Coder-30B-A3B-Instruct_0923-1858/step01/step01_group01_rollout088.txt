import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    """Optimize using convex optimization in the frequency domain."""
    np.random.seed(seed)

    h_current = h.copy()

    try:

        H = cp.Variable(n_points, complex=True)

        # But we directly optimize the real part
        
        # Objective: minimize maximum correlation
        # Correlation in frequency domain: H * conj(H) 
        # But we want max_k |H[k] * (1 - conj(H[k]))|
        # For now, we'll handle constraint via TrustRegion
        # Relaxation may be needed for CVXPY feasibility
        
        # We'll use a simpler trust region approach directly in time domain
        # but start with fourier-based initialization

        # Directly solve a constrained optimization problem
        # Maximize: max_k (sum h[i] * (1 - h[i+k])) which is max_k correlate(h, 1-h)[k]
        # This is a non-convex problem, but we can use projections and local search
        # The key insight from the plan was to use CVXOPT, but direct optimization is often better
        # in this problem space for discrete, bounded variables.
        
        # We go for projected gradient on time domain, still using fourier for gap estimation
        
        best_h = h_current.copy()
        best_C = evaluate_erdos_solution(best_h, dx)
        
        # Optimization loop
        max_iter = 5000
        lr = 0.01
        decay_rate = 0.97
        patience = 100

        prev_C = best_C
        
        for i in range(max_iter):
            if time.time() - t0 > budget_s - 10:
                break
                
            # Estimate gradient numerically
            grad = np.zeros_like(best_h)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = best_h.copy()
                h_plus[j] += eps
                h_plus = project_feasible(h_plus, dx)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                C_plus = np.max(corr_plus) * dx
                
                h_minus = best_h.copy()
                h_minus[j] -= eps
                h_minus = project_feasible(h_minus, dx)
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                C_minus = np.max(corr_minus) * dx
                
                grad[j] = (C_plus - C_minus) / (2 * eps)
            
            # Update with gradient
            new_h = best_h - lr * grad
            
            # Project
            new_h = project_feasible(new_h, dx)
            
            # Evaluate
            new_C = evaluate_erdos_solution(new_h, dx)
            
            if new_C < best_C:
                best_h = new_h.copy()
                best_C = new_C
                patience_counter = 0
            else:
                patience_counter += 1
                if patience_counter > patience:
                    lr *= decay_rate
                    patience_counter = 0
        
        # Final projection
        best_h = project_feasible(best_h, dx)
        best_C = evaluate_erdos_solution(best_h, dx)
        
        return best_h, best_C
        
    except:
        # Fallback: just do projected gradient descent directly, not using cvxpy too much
        return h_current, evaluate_erdos_solution(h_current, dx)
