import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Use a specific pattern from previous constructions
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Start optimization
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Convert h to cvxpy variable
    h_var = cp.Variable(n_points)
    
    # Initial C5 value slightly above current approximation
    C5 = 0.4
    
    # Cross-validation for convergence
    prev_C5 = C5 + 1
    tolerance = 1e-4
    max_iterations = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Direct convex optimization approach
        # Objective: minimize C5
        # Constraint: for each k in [0, n_points-1], sum(h[i] * (1 - h[i+k])) <= C5 * dx
        # Also: sum(h) == n_points/2, 0 <= h[i] <= 1
        
        # Build the constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        dx = 2.0 / n_points
        
        # All shift constraints: prevent overlap larger than C5 * dx
        for k in range(n_points):
            # Compute the shifted correlation
            corr_k = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_k += h_var[i] * (1 - h_var[j])
            # Add constraint: corr_k <= C5 * dx
            constraints.append(corr_k <= C5 * dx)
            
        # Bounds on variables
        for i in range(n_points):
            constraints.append(h_var[i] >= 0)
            constraints.append(h_var[i] <= 1)
            
        # Construct problem
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve the problem
            prob.solve(solver=cp.ECOS, max_iters=100)
            
            # Check if optimal solution found
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h_var.value
                
                # Project solution onto valid set
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                
                # Evaluate new C5
                new_c5 = evaluate_erdos_solution(h_sol)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_sol.copy()
                # Adjust C5
                if new_c5 > C5:
                    C5 += 0.0001
                else:
                    C5 -= 0.0001
                prev_C5 = C5
            else:
                C5 += 0.0001
        except:
            # If optimization fails, increase C5
            C5 += 0.0001
            
        iteration += 1
        
        if abs(C5 - prev_C5) < tolerance:
            break
        
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
