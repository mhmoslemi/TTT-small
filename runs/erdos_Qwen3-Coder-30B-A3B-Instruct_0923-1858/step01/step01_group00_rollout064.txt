import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for better resolution
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        # Interpolate to the required resolution
        h = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        # Ensure it's within [0,1]
        h = np.clip(h, 0, 1)
        # Normalize to have sum(h) = n_points/2
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution with a specific pattern
        h = np.ones(n_points) * 0.5
        # Set boundary values to 0 to reduce edge effects
        h[0] = 0.0
        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    sigma = 0.1
    shake_threshold = 100
    no_improvement_count = 0
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Main annealing loop
    while time.time() - start_time < budget_s:
        # Generate perturbation
        delta = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_new = h + delta
        
        # Project to feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept with simulated annealing probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Apply shake if no improvement for a while
        if no_improvement_count >= shake_threshold:
            h += np.random.normal(0, T * sigma, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            no_improvement_count = 0
        
        # Cool temperature
        T *= alpha
    
    return (best_h, best_c5, n_points)
