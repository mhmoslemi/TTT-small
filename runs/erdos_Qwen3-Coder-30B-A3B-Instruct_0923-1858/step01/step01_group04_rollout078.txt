import numpy as np
from scipy.optimize import minimize
import time
from scipy.ndimage import gaussian_filter1d

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try multiple optimization strategies
    strategies = []
    
    # Strategy 1: Local optimization with perturbations
    if time.time() - start_time < budget_s - 2:
        strategies.append("local_optimization")
    
    # Strategy 2: Genetic algorithm approach (random mutations + selection)
    if time.time() - start_time < budget_s - 2:
        strategies.append("genetic")
    
    # Strategy 3: Gradient-based optimization with restarts
    if time.time() - start_time < budget_s - 2:
        strategies.append("gradient")
    
    # Strategy 4: Simulated Annealing
    if time.time() - start_time < budget_s - 2:
        strategies.append("annealing")
    
    # Run different strategies
    for strategy in strategies:
        if time.time() - start_time > budget_s - 1:
            break
            
        if strategy == "local_optimization":
            # Multiple local optimizations from different random starts
            for _ in range(30):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Random perturbation
                h_trial = h + np.random.normal(0, 0.02, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                # SLSQP optimization
                try:
                    res = minimize(
                        objective,
                        h_trial,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=constraints,
                        options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-5}
                    )
                    
                    if res.success:
                        h_candidate = np.clip(res.x, 0, 1)
                        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                        c5_candidate = evaluate_erdos_solution(h_candidate)
                        if c5_candidate < best_c5:
                            best_h = h_candidate
                            best_c5 = c5_candidate
                except Exception:
                    pass
                    
        elif strategy == "genetic":
            # Simple genetic algorithm approach
            population_size = 30
            generations = 20
            elite_size = 5
            
            pop = []
            for _ in range(population_size):
                h_ind = h + np.random.normal(0, 0.05, n_points)
                h_ind = np.clip(h_ind, 0, 1)
                h_ind = h_ind * (n_points / 2) / np.sum(h_ind)
                pop.append(h_ind)
            
            for gen in range(generations):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Evaluate population
                fitness = []
                for p in pop:
                    c5 = evaluate_erdos_solution(p)
                    fitness.append(c5)
                
                # Sort and select elite
                sorted_indices = np.argsort(fitness)
                elite = [pop[i] for i in sorted_indices[:elite_size]]
                
                # Create new population
                new_pop = elite.copy()
                while len(new_pop) < population_size:
                    parents = np.random.choice(elite_size, 2, replace=False)
                    parent1, parent2 = elite[parents[0]], elite[parents[1]]
                    
                    # Blend crossover
                    alpha = np.random.rand()
                    child = alpha * parent1 + (1 - alpha) * parent2
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                    
                    # Add random mutation
                    if np.random.rand() < 0.3:
                        mutation_strength = 0.1
                        child += np.random.normal(0, mutation_strength, n_points)
                        child = np.clip(child, 0, 1)
                        child = child * (n_points / 2) / np.sum(child)
                    
                    new_pop.append(child)
                
                pop = new_pop
                
            # Find best among final population
            final_fitness = [evaluate_erdos_solution(p) for p in pop]
            best_idx = np.argmin(final_fitness)
            if final_fitness[best_idx] < best_c5:
                best_h = pop[best_idx]
                best_c5 = final_fitness[best_idx]
                
        elif strategy == "gradient":
            # Gradient-based methods with restarts
            for i in range(10):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Perturbation with gradient direction approximation
                grad_est = np.zeros(n_points)
                epsilon = 1e-4
                
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += epsilon
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    h_minus = h.copy()
                    h_minus[j] -= epsilon
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    
                    grad_est[j] = (evaluate_erdos_solution(h_plus) - 
                                   evaluate_erdos_solution(h_minus)) / (2 * epsilon)
                
                # Move against gradient
                h_trial = h - 0.01 * grad_est
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                # Simple line search to make sure we're going downhill
                try:
                    res = minimize(
                        objective,
                        h_trial,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=constraints,
                        options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-5}
                    )
                    
                    if res.success:
                        h_candidate = np.clip(res.x, 0, 1)
                        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                        c5_candidate = evaluate_erdos_solution(h_candidate)
                        if c5_candidate < best_c5:
                            best_h = h_candidate
                            best_c5 = c5_candidate
                except Exception:
                    pass
                    
        elif strategy == "annealing":
            # Simulated Annealing
            temp = 1.0
            cooling_rate = 0.99
            iterations_per_temp = 100
            
            current_h = h.copy()
            current_c5 = evaluate_erdos_solution(current_h)
            
            for _ in range(1000):
                if time.time() - start_time > budget_s - 1:
                    break
                
                # Perturb current solution
                h_trial = current_h + np.random.normal(0, 0.02, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                trial_c5 = evaluate_erdos_solution(h_trial)
                
                # Accept or reject based on Metropolis criterion
                if trial_c5 < current_c5 or np.random.rand() < np.exp(- (trial_c5 - current_c5) / temp):
                    current_h = h_trial
                    current_c5 = trial_c5
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        
                temp *= cooling_rate
                
                if temp < 1e-6:
                    temp = 1e-6
                
    return (list(best_h), best_c5, n_points)
