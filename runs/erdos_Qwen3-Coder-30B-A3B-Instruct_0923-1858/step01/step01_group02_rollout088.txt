import numpy as np
from cvxpy import Variable, Minimize, Problem, sum_expr, constraints, norm, Min
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a basic feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use CVXPY for optimization
    # Iterative refinement of C₅
    c5_bound = 0.4
    step_size = 0.0001
    tolerance = 0.001
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create variable
        h_var = Variable(n_points)
        
        # Compute constraints for all shifts
        constraints_list = []
        
        # Sum constraint
        constraints_list.append(sum_expr(h_var) == n_points / 2)
        
        # Bound constraints
        for i in range(n_points):
            constraints_list.append(h_var[i] >= 0)
            constraints_list.append(h_var[i] <= 1)
        
        # Add overlap constraints
        dx = 2.0 / n_points
        # We'll compute this bounds adaptively
        max_k = n_points
        for k in range(max_k):
            # Compute correlation for shift k
            corr_sum = 0
            for i in range(n_points):
                if i + k < n_points:
                    corr_sum += h_var[i] * (1 - h_var[i + k])
            # Add constraint
            constraints_list.append(corr_sum <= c5_bound / dx)
        
        # Objective to minimize
        objective = Minimize(0)
        
        # Create and solve problem
        prob = Problem(objective, constraints_list)
        
        try:
            prob.solve(solver='ECOS', verbose=False)
            # Check if optimal
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array([float(h_var[i].value) for i in range(n_points)])
                # Project onto feasible set
                h_new = np.clip(h_new, 0, 1)
                # Renormalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Compute new C₅
                new_c5 = evaluate_erdos_solution(h_new)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
                
                # Adjust C₅ if needed
                if new_c5 <= c5_bound + tolerance:
                    c5_bound -= step_size
                    if c5_bound < 0:
                        c5_bound = 0
                else:
                    c5_bound += step_size
            else:
                # If infeasible, increase C₅
                c5_bound += step_size
                
        except Exception:
            c5_bound += step_size
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
