import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Bound constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Constraint dictionary
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # L-BFGS-B optimizer
    opt_kwargs = {
        'method': 'L-BFGS-B',
        'bounds': bounds,
        'options': {'maxiter': 100}
    }
    
    # Optimizer with decay control
    n_iter = 0
    perturbation_magnitude = 0.1
    last_improvement = 0
    stagnant_count = 0

        try:
            x0 = h.copy()

            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            x0 += perturbation

            x0 = np.clip(x0, 0, 1)

            res = minimize(objective, x0, **opt_kwargs)

                    if stagnant_count > 10:
                        # Increase perturbation to escape local minimum
                        perturbation_magnitude = min(0.5, perturbation_magnitude * 1.1)
                        stagnant_count = 0
            else:
                # If optimization failed, just use the perturbed step
                current_c5 = evaluate_erdos_solution(x0)
                if current_c5 < best_c5:
                    best_h = x0.copy()
