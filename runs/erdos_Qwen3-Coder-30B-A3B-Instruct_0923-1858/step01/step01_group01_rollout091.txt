import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, strength=0.1):
    """Apply small random mutation to h."""
    mutated = h.copy()
    idx = np.random.randint(len(mutated))
    mutated[idx] += np.random.normal(0, strength)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    child = parent1.copy()
    mask = np.random.rand(len(parent1)) < 0.5
    child[mask] = parent2[mask]
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Tournament selection."""
    tournament_size = 3
    selected = []
    for _ in range(num_parents):
        tournament_indices = np.random.choice(len(population), 
                                              size=tournament_size, 
                                              replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def calculate_complexity(h):
    """Basic measure of solution complexity."""
    diff = np.abs(np.diff(h))
    return np.sum(diff > 0.05)

def evolve_population(population, dx, budget_s=600):
    start_time = time.time()
    best_fitness = -np.inf
    best_h = None
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            corr = correlate(individual, 1 - individual, mode='full')
            max_overlap = np.max(corr) * dx
            complexity = calculate_complexity(individual)
            fitness = -max_overlap - 0.01 * complexity  # Negative for minimization
            fitnesses.append(fitness)
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)  # Minimize negative fitness
        current_fitness = fitnesses[current_best_idx]
        if current_fitness > best_fitness:
            best_fitness = current_fitness
            best_h = population[current_best_idx].copy()
        
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Generate new population
        new_population = []
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < 0.2:  # Mutation rate
                child = mutate(child, dx)
            new_population.append(child)
        
        population = new_population
        
        # Early stopping check
        if time.time() - start_time > budget_s - 60:
            break
    
    return best_h, -best_fitness  # Return positive C5
