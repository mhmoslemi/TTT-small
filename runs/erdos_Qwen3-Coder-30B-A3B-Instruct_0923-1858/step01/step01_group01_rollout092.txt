import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    except NameError:

        h_start = np.linspace(0, 1, n_points//2, endpoint=False)
        h_start = np.concatenate([h_start, np.linspace(1, 0, n_points - n_points//2)])
    
    # Use initial solution and add diverse mutations
    for _ in range(population_size):
        h = h_start.copy()
        # Add small random noise to diversify starting population
        if np.random.rand() < 0.5:
            noise_level = 0.05
            noise = np.random.normal(0, noise_level, n_points)
            h += noise
        h = project(h, 2.0/n_points)
        population.append(h)
    
    return population

def evaluate_fitness(h, dx):
    """Evaluate fitness: inverse of C5 with complexity regularization"""
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    # Complexity regularization using number of transitions
    transitions = np.sum(np.abs(np.diff(h)) > 0.05)
    complexity_penalty = 0.01 * transitions
    return 1.0 / (max_overlap + complexity_penalty)

def tournament_selection(population, fitnesses, k=2):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), size=k, replace=False)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    """Uniform crossover between two parents"""
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def mutate(h, dx, mutation_rate=0.1):
    """Mutate h by adding noise and projecting back to constraints"""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return project(mutated, dx)

def evolve_population(population, dx, budget_s=600):
    """Main evolution loop for optimizing the population"""
    start_time = time.time()
    best_fitness = 0
    best_h = None
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitnesses = [evaluate_fitness(h, dx) for h in population]
        
        # Track best solution
        current_best = population[np.argmax(fitnesses)]
        current_fitness = max(fitnesses)
        if current_fitness > best_fitness:
            best_fitness = current_fitness
            best_h = current_best.copy()
        
        # Select parents
        parents = tournament_selection(population, fitnesses)
        
        # Generate new population
        new_population = []
        for _ in range(len(population) // 2):
            # Perform crossover
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            # Apply mutation
            if np.random.rand() < 0.1:
                child = mutate(child, dx)
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Early stopping
        if time.time() - start_time > budget_s - 60:
            break
    
    return best_h, 1 / best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points)
    
    # Evolve population
    best_h, best_C = evolve_population(population, dx, budget_s)
    
    return (best_h, best_C, n_points)
