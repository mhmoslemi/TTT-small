import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with higher resolution
    n_points = 140
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Interpolate to 140 points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
    else:
        # Create initial guess
        h = np.ones(n_points) * 0.5
        h[30:110] = 1.0  # Cluster in middle
        h = h / np.sum(h) * (n_points / 2.0)
        
    h = np.clip(h, 0, 1)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    iter_count = 0
    max_iter = 200
    noise_std = 0.05
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Basin hopping
        h_trial = h + np.random.normal(0, noise_std, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        if iter_count % 50 == 0:
            noise_std /= 2.0
            
        iter_count += 1
        
        if iter_count % 5 == 0:
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)
    
    # Local refinement on best candidates
    candidates = [best_h]
    for _ in range(5):
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
    
    # Final validation and ensure exact constraint
    final_h = best_h.copy()
    final_h = final_h / np.sum(final_h) * (n_points / 2.0)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
