import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use more points for better resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    swarm = []
    velocities = []
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    # Initialize swarm with perturbed versions of h
    for _ in range(n_particles):
        particle = h + np.random.normal(0, 0.01, n_points)
        particle = project_to_feasible(particle, n_points)
        swarm.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize bests
    personal_best = [p.copy() for p in swarm]
    personal_best_scores = [evaluate_erdos_solution(p) for p in swarm]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    alpha = 0.9
    
    # Start timer
    start_time = time.time()
    last_improvement = start_time
    stagnation_count = 0
    
    while time.time() - start_time < budget_s:
        # Check for stagnation
        if time.time() - last_improvement > 30:
            stagnation_count += 1
            if stagnation_count > 5:
                # Perturb swarm to escape local minima
                for i in range(n_particles):
                    swarm[i] += np.random.normal(0, 0.02, n_points)
                    swarm[i] = project_to_feasible(swarm[i], n_points)
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                last_improvement = time.time()
                stagnation_count = 0
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity with inertia and cognitive/social components
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - swarm[i]) +
                c2 * r2 * (global_best - swarm[i])
            )
            
            # Apply low-pass filtering
            velocities[i] = alpha * velocities[i] + (1 - alpha) * vel_update
            
            # Update position
            swarm[i] += velocities[i]
            
            # Apply feasibility projection
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate objective
            score = evaluate_erdos_solution(swarm[i])
            
            # Update personal best if improvement
            if score < personal_best_scores[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_scores[i] = score
                if score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = score
                    last_improvement = time.time()
        
        # Add noise to particles and re-project
        for i in range(n_particles):
            swarm[i] += np.random.normal(0, 0.01, n_points)
            swarm[i] = project_to_feasible(swarm[i], n_points)
    
    # Final validation
    final_h = project_to_feasible(global_best.copy(), n_points)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, n_points)
