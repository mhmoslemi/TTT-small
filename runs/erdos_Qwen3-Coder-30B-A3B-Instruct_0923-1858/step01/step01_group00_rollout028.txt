import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_feasible(h, n_points):
    """Project h onto feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2) / n_points
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better precision
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h, n_points)
    else:
        # Initialize with a reasonable construction
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, n_points)

    # Initialize PSO parameters
    swarm_size = 50
    max_iter = budget_s // 0.05 if budget_s > 0.05 else 1000
    omega = 0.7
    c1 = c2 = 1.5
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    personal_bests = []
    personal_best_values = []
    global_best = h.copy()
    global_best_value = evaluate_erdos_solution(h)
    
    for _ in range(swarm_size):
        particle = h.copy() + np.random.normal(0, 0.01, n_points)
        particle = project_feasible(particle, n_points)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        value = evaluate_erdos_solution(particle)
        personal_bests.append(particle.copy())
        personal_best_values.append(value)
        if value < global_best_value:
            global_best = particle.copy()
            global_best_value = value
    
    # PSO main loop
    start_time = time.time()
    last_improvement = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity with inertia and cognitive/social components
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Apply low-pass filter to velocity (alpha=0.9)
            velocities[i] = 0.9 * velocities[i] + 0.1 * np.random.normal(0, 0.01, n_points)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility projection
            particles[i] = project_feasible(particles[i], n_points)
            
            # Evaluate objective
            current_value = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_bests[i] = particles[i].copy()
                personal_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
                    last_improvement = iter_count

        # Add perturbation to all particles to avoid sticking
        if iter_count % 5 == 0:  # Perturb every 5 iterations
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.01, n_points)
                particles[i] += noise
                particles[i] = project_feasible(particles[i], n_points)

    # Final evaluation of best solution
    best_h = project_feasible(global_best, n_points)
    c5_bound = evaluate_erdos_solution(best_h)
    
    return (best_h, c5_bound, n_points)
