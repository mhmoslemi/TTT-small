import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # High resolution for better precision
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # PSO parameters
    swarm_size = 50
    max_iter = 600
    w = 0.7
    c1 = c2 = 1.5
    alpha = 0.9  # Low-pass filter coefficient
    
    # Initialize swarm
    particles = np.array([h.copy() for _ in range(swarm_size)])
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    personal_best = particles.copy()
    personal_best_scores = np.array([evaluate_erdos_solution(p) for p in particles])
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        for i in range(swarm_size):
            # Update velocity with inertia and cognitive/social components
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Apply low-pass filter to velocity
            velocities[i] = alpha * velocities[i] + (1 - alpha) * np.roll(velocities[i], 1)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility projection
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
            
            # Evaluate objective
            score = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = particles[i].copy()
                global_best_score = score
        
        # Perturb all particles with noise to escape local optima
        noise = np.random.normal(0, 0.01, (swarm_size, n_points))
        particles += noise
        for i in range(swarm_size):
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
    
    # Return result
    final_c5 = evaluate_erdos_solution(global_best)
    return (global_best, final_c5, n_points)
