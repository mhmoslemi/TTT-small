import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):

    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize with provided initial construction if available

        h = h_interp
    else:
        # Default construction
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    
    # Normalize to meet constraint sum(h) = n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Project to feasible region
    def project(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Gradient approximation via finite differences
    def compute_gradient(h, eps=1e-6):
        current_c5 = evaluate_erdos_solution(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Optimization parameters
    alpha = 0.1
    patience = 0
    max_patience = 200
    step_size_decay = 0.95
    start_time = time.time()
    
    step = 0
    while time.time() - start_time < budget_s:
        # Compute gradient (this may not always be exact due to constraints)
        try:
            grad = compute_gradient(h)
        except:
            # Fallback
            grad = np.random.randn(n_points) * 0.01
            
        # Update step
        h_new = h - alpha * grad
        h_new = project(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Check for termination
        if patience > max_patience:
            alpha = 0.1
            # Perturb solution
            h = h + np.random.uniform(-0.1, 0.1, n_points)
            h = project(h)
            patience = 0
        else:
            h = h_new
        
        # Adjust learning rate
        if step % 100 == 0 and step > 0:
            alpha *= step_size_decay
            
        step += 1
        
        # Small check to avoid infinite loops
        if np.isnan(c5_new) or np.isinf(c5_new):
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
            patience = 0
            
    # Final evaluation of best solution
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except:
        final_c5 = best_c5
    
    # Final projection to ensure constraints
    best_h = project(best_h)
    
    return (best_h, final_c5, n_points)
