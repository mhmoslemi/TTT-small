import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

        h[1] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    # Define constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    opt_kwargs = {
        'method': 'L-BFGS-B',
        'bounds': bounds,
        'constraints': cons,
        'options': {'maxiter': 1000}
    }
    
    iteration = 0
    last_improvement = 0
    perturbation_std = 0.1
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        try:
            # Random perturbation to escape local minima
            perturbation = np.random.normal(0, perturbation_std, n_points)
            x0 = np.clip(best_h + perturbation, 0, 1)
            # Renormalize sum to 1
            x0 = x0 * (n_points / 2) / np.sum(x0)
            
            # Local optimization
            res = minimize(objective, x0, **opt_kwargs)
            
            if not res.success:
                continue

            current_c5 = evaluate_erdos_solution(res.x)
            if current_c5 < best_c5:

            else:
                perturbation_std = max(0.01, perturbation_std * 0.95)
                
        except Exception as e:
            pass
            
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
