        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.15
    tournament_size = 5
    crossover_rate = 0.8
    
    # Initialize population
    def create_individual():
        return np.random.rand(n_points)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_C5(h_vec):
        # Convert to correct size
        h_vec = np.array(h_vec)
        # Note: correlate with mode='full' to handle all shifts
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum over all shifts
        return np.max(corr) * dx
    
    def tournament_selection(population, fitnesses, k=tournament_size):
        idxs = np.random.choice(len(population), size=k)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        return population[best_idx].copy()
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Single-point crossover
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h_vec, rate=mutation_rate):
        for i in range(len(h_vec)):
            if np.random.rand() < rate:
                h_vec[i] += np.random.normal(0, 0.05)
        return h_vec

    population = [project_feasible(create_individual()) for _ in range(population_size)]

    fitnesses = [compute_C5(ind) for ind in population]

            patience = 0
        else:

    # Final validation

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
