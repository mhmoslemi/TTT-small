import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Successive refinement using L-BFGS-B with various initializations
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try multiple restarts with different strategies
    strategies = [
        # Original approach
        lambda: h.copy(),
        # Random perturbations
        lambda: np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1),
        # Piecewise constant construction
        lambda: np.clip(np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]) * 0.5 * n_points / 2, 0, 1),
        # Sine wave construction
        lambda: np.clip(0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points)), 0, 1),
    ]
    
    max_restarts = 10
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Try each strategy for this restart
        for strategy in strategies:
            if time.time() - start_time > budget_s - 1:
                break
                
            h_trial = strategy()
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            h_trial = np.clip(h_trial, 0, 1)
            
            # Optimize using L-BFGS-B
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec)
            
            def constraint_integral(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_integral}
            
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except Exception:
                pass
        
        restart_count += 1
    
    # Final validation and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
