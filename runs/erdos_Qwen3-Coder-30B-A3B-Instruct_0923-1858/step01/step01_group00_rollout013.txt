import numpy as np
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 150 points for higher resolution
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        # Ensure bounds and normalization
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a starting point with some structure
        h = np.ones(n_points) * 0.5
        h[0] = 0.0  # Ensure boundary condition
        h[-1] = 0.0
        # Normalize
        h = h / np.sum(h) * (n_points / 2)

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    sigma = 0.1
    min_T = 1e-6
    shake_threshold = 100
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    # Main optimization loop
    while time.time() - start_time < budget_s and T > min_T:
        # Generate perturbation
        delta = np.random.uniform(-T * sigma, T * sigma, n_points)
        h_new = h + delta
        
        # Project to feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1

        # Adaptive shaking
        if no_improvement_count >= shake_threshold:
            h = np.clip(h + np.random.normal(0, T * sigma, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            no_improvement_count = 0

        # Cool temperature
        T = max(min_T, T * alpha)

    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
