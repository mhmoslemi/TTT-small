import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_h(n_points):
    """Initialize h with a triadic waveform pattern."""
    h = np.zeros(n_points)
    for i in range(n_points):
        phase = np.abs(np.sin(np.pi * i / (n_points / 3))) ** 3
        h[i] = np.clip(2 * phase - 0.5, 0, 1)
    return project(h, 2.0 / n_points)

def compute_gradient(h, dx, eps=1e-8):
    """Compute gradient using finite differences."""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        max_plus = np.max(corr_plus)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        max_minus = np.max(corr_minus)
        
        grad[i] = (max_plus - max_minus) / (2 * eps)
    return grad

def optimize_h(h, dx, n_points, max_time=580):
    """Optimize h using L-BFGS-B with projections."""
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    t0 = time.time()
    
    def obj_and_grad(x):
        if time.time() - t0 > max_time:
            return np.inf, np.zeros_like(x)
        
        h_temp = project(x, dx)
        c5_val = evaluate(h_temp, dx)
        grad = compute_gradient(h_temp, dx)
        return c5_val, grad
    
    # Use L-BFGS-B for optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(
        fun=lambda x: obj_and_grad(x)[0],
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=lambda x: obj_and_grad(x)[1],
        options={'maxiter': 10000, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    if result.success or not np.isinf(result.fun):
        optimized_h = project(result.x, dx)
        final_c5 = evaluate(optimized_h, dx)
        if final_c5 < best_c5:
            best_h = optimized_h
            best_c5 = final_c5
    
    # Final check and projection
    final_h = project(result.x, dx) if not np.isinf(result.fun) else best_h
    final_c5 = evaluate(final_h, dx)
    
    return final_h, final_c5
