import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to meet constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    dx = 2.0 / n_points
    
    # Iterative cutting plane approach
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute current overlaps O_k(h) for all k
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        overlaps = corr * dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        
        # Build a convex model for this iteration
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            C >= 0
        ]
        
        # Add cutting planes: for each k, compute gradient and add constraint
        for k in range(len(overlaps)):
            # Compute gradient ∇O_k(h) for shift k
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] = dx * (1 - best_h[j])
            
            # Add cutting plane constraint: 
            # O_k(h) <= O_k(h^t) + grad^T (h - h^t)
            # => grad^T h <= O_k(h^t) - grad^T h^t + O_k(h^t)
            rhs = overlaps[k] - np.dot(grad, best_h) + overlaps[k]
            constraints.append(cp.sum(grad * h_var) <= rhs)
        
        # Objective: minimize C
        prob = cp.Problem(cp.Minimize(C), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                c_new = C.value
                
                # Ensure it's a feasible solution
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
        
        iter_count += 1
        
        # Early termination if time is short
        if time.time() - start_time > budget_s - 0.1:
            break
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
