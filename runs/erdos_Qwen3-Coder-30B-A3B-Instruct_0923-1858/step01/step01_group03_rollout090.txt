import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_initial = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    pop_size = 100
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.2
    stagnation_limit = 50
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random solutions
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests and global best
    pbest = [swarm[i].copy() for i in range(pop_size)]
    pbest_fitness = [float('inf')] * pop_size
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Compute initial fitness
    for i in range(pop_size):
        h = swarm[i]
        overlap = correlate(h, 1 - h, mode='full')
        fitness = np.max(overlap) * (2.0 / n_points)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    # PSO main loop
    iter_count = 0
    stagnation = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        # Update each particle
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = np.clip(swarm[i], 0, 1)
            # Rescale to satisfy constraint
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Evaluate fitness
            h = swarm[i]
            overlap = correlate(h, 1 - h, mode='full')
            fitness = np.max(overlap) * (2.0 / n_points)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                stagnation += 1
        
        # Reset stagnated particles
        if stagnation > stagnation_limit:
            for i in range(pop_size // 10):
                idx = np.random.randint(0, pop_size)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[idx] = h.copy()
                pbest_fitness[idx] = float('inf')
                stagnation = 0
        
        iter_count += 1
    
    # Final evaluation of global best
    final_h = swarm[gbest_idx]
    final_overlap = correlate(final_h, 1 - final_h, mode='full')
    final_C5 = np.max(final_overlap) * (2.0 / n_points)
    
    return (final_h.tolist(), final_C5, n_points)
