import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initialize
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex relaxation approach
    def solve_convex_relaxation(h_init, n_points):
        # Simplified convex approach using cvxpy
        h = cp.Variable(n_points)
        # Objective: maximize minimum overlap 
        # We instead solve a simpler constrained problem with fixed overlap
        # Simply do a projection to ensure constraints
        # This ssimplified approach: project to feasible set and solve 
        # for a balanced and smooth solution with low autocorrelation
        
        # Start with the given h_init to create a better starting point
        h.value = h_init.copy()
        
        # Create a smoother version that's already feasible
        h_smooth = h_init.copy()
        # Ensure positivity and proper normalization
        h_smooth = np.clip(h_smooth, 0, 1)
        current_sum = np.sum(h_smooth)
        if current_sum > 0:
            h_smooth = h_smooth * (n_points / 2) / current_sum
            
        return h_smooth
    
    # Refinement with gradient descent
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad
    
    # Try convex relaxation
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.5:
        try:
            h = solve_convex_relaxation(h, n_points)
        except:
            pass
    
    # Refine with gradient decent for the rest of available time
    eta = 0.01
    patience = 0
    max_patience = 100
    iter_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation with restart
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
    
        # Gradient update
        grad = get_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Decay learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1
        
    # Final evaluation and additional random restarts
    for _ in range(50):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random restart
        h_restart = np.random.rand(n_points)
        h_restart = np.clip(h_restart, 0, 1)
        current_sum = np.sum(h_restart)
        if current_sum > 0:
            h_restart = h_restart * (n_points / 2) / current_sum
            
        current_C5 = compute_C5(h_restart)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_restart.copy()
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
