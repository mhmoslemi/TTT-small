import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Make it more structured to escape early local minima
        h[:n_points//2] = 0.75
        h[n_points//2:] = 0.25
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
        
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # CVXPY variables for convex optimization
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints for sum = 1 and bounds
    constraints = [cp.sum(h_var) == n_points / 2,
                   h_var >= 0,
                   h_var <= 1,
                   C >= 0]
    
    # Initialize a list for cutting planes
    cut_planes = []
    
    # To track our progress
    iteration = 0
    max_iter = 10000  # Avoid infinite loop
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current overlaps
        current_overlap = np.correlate(best_h, 1 - best_h, mode="full")
        current_c5 = np.max(current_overlap) * (2.0 / n_points)
        
        # Add cutting plane for the shift with maximum overlap
        
        # Create a copy of the problem for this iteration
        temp_constraints = constraints.copy()
        
        # If we have cutting planes, use them
        for plane in cut_planes:
            temp_constraints.append(plane)
            
        # Set up the convex problem
        # We want to minimize C such that for all overlaps:
        # O_k(h) <= C
        # which is equivalent to adding:
        # - sum h[i] (1 - h[i+k]) <= C
        # For each k, we compute the gradient at current h
        max_k_idx = np.argmax(current_overlap)
        max_k = max_k_idx - (n_points - 1)  # shift amount
        
        # Let's construct the gradient w.r.t. h of O_k(h)
        # O_k(h) = dx * sum h[i] * (1 - h[i+k])
        # Gradient w.r.t. h[j] is: dx * sum_{i: i+k==j} (1 - h[i+k]) + dx * h[j] * delta_{j==i+k}
        # This is complex due to wraparound. Better approach is to explicitly compute it:
        # For i=0...n_points-1: dO_k/dh[i] = dx * (1 - h[(i+k)%n_points]) if i+k < n_points
        # or more broadly: we compute the linear approximation to each integral 
        # This is difficult to compute directly so let's make a smarter assumption
        
        # Or simply, we use a simpler approach.
        # We linearize at best_h to form a linear cut for one k (the one we're interested in)
        k_index = max_k + (n_points - 1)  # transition from shift index to full correl index
        
        # The expression O_k(h) = dx * sum h[i] (1 - h[i+k])
        # For the gradient of this term at h = h^t:
        # d/dh[i] O_k = dx * (1 - h[i+k]) if i+k < n_points
        # But we also have h[j] coefficient in h[i+k] term.
        # Just using a linear approximation at the current point:
        # O_k(h) <= O_k(h^t) + sum_i (dO_k/dh[i]) * (h[i] - h^t[i])
        # Where dO_k/dh[i] = dx * (1 - h^t[i+k])
        
        if len(cut_planes) == 0:
            # Add all cuts for first iteration to improve initial set
            for i in range(len(current_overlap)):
                k = i - (n_points - 1)
                # Compute at current point
                h_shift = np.roll(best_h, -k)
                term = best_h * (1 - h_shift)
                current_k_value = np.sum(term) * (2.0 / n_points)  # dx = 2/n_points, sum over all i
                
                # The gradient at best_h w.r.t. O_k
                grad = np.zeros(n_points)
                for j in range(n_points):
                    if 0 <= (j + k) < n_points:
                        grad[j] = (2.0 / n_points) * (1 - best_h[j + k])
                # Cut plane: sum_i grad[i] * h[i] <= current_k_value + sum_i grad[i] * (-best_h[i]) 
                # Equivalent to: sum_i grad[i] * h[i] <= current_k_value - sum_i grad[i]*best_h[i]
                cut_left = cp.sum(cp.multiply(grad, h_var))
                cut_right = current_k_value - np.dot(grad, best_h)
                temp_constraints.append(cut_left <= cut_right)
        else:
            # Just cut at max overlap one
            k_shift = np.argmax(current_overlap) - (n_points - 1)
            h_shift = np.roll(best_h, -k_shift)
            term = best_h * (1 - h_shift)
            current_k_value = np.sum(term) * (2.0 / n_points)
            
            # Gradient
            grad = np.zeros(n_points)
            for j in range(n_points):
                if 0 <= (j + k_shift) < n_points:
                    grad[j] = (2.0 / n_points) * (1 - best_h[j + k_shift])
                    
            # Add cutting plane: sum_i grad[i] * h[i] <= current_k_value - sum_i grad[i]*best_h[i]
            cut_left = cp.sum(cp.multiply(grad, h_var))
            cut_right = current_k_value - np.dot(grad, best_h)
            temp_constraints.append(cut_left <= cut_right)
        
        # Set up problem to minimize C
        prob = cp.Problem(cp.Minimize(C), temp_constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value
                new_c5 = C.value
                
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except Exception:
            pass
            
        iteration += 1
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
