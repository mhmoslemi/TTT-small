import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a triangular initial h
        n_points = 70
        h = np.array([1 - abs(2*i/n_points - 1) for i in range(n_points)])
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop with time budget
    max_iter = 10000
    iter_count = 0
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Global exploration with perturbation
        # Perturb in the central region where overlaps are most critical
        h_trial = h.copy()
        perturb_region = slice(int(0.3 * n_points), int(0.7 * n_points))
        noise = np.random.normal(0, 0.03, n_points)
        h_trial[perturb_region] += noise[perturb_region]
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # Evaluate the trial
        c5_trial = evaluate_erdos_solution(h_trial)
        
        # Accept if better
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h = h_trial.copy()
        
        # Local refinement
        if iter_count % 5 == 0:
            try:
                # Create bounds and constraint for SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h):
                    # Compute correlation with (1-h)
                    corr = np.correlate(h, 1-h, mode="full")
                    # Return maximum correlation scaled by dx
                    return np.max(corr) * (2.0 / n_points)
                
                res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new.copy()
            except:
                pass
        
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
