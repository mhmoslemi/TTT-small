import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better precision
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    swarm = np.random.rand(n_particles, n_points) * 0.5
    
    # If initial_h_values exists, use it as starting point
    if 'initial_h_values' in globals():

        # Interpolate to match n_points

            x_old = np.linspace(0, 2, len(initial))
            x_new = np.linspace(0, 2, n_points)
            initial = np.interp(x_new, x_old, initial)
        # Normalize to correct sum
        initial = initial / np.sum(initial) * (n_points / 2)
        # Fill first particle with initial and others randomly
        swarm[0] = initial
        for i in range(1, n_particles):
            swarm[i] = np.random.rand(n_points) * 0.5
    
    # Initialize velocities
    velocities = (np.random.rand(n_particles, n_points) - 0.5) * 0.1
    
    # Evaluate initial swarm
    fitness = np.array([evaluate_erdos_solution(project_to_feasible(p, n_points)) 
                       for p in swarm])
    
    # Personal best
    personal_best = swarm.copy()
    personal_best_fitness = fitness.copy()
    
    # Global best
    global_best_idx = np.argmin(fitness)
    global_best = swarm[global_best_idx].copy()
    global_best_fitness = fitness[global_best_idx]
    
    # Parameters
    omega = 0.7  # Inertia weight
    c1 = c2 = 1.5  # Cognitive and social coefficients
    
    # Low-pass filter for smoother velocity updates
    alpha = 0.9
    filtered_velocities = velocities.copy()
    
    # Begin optimization
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        # Update particles
        for i in range(n_particles):
            # Update velocity with inertia and cognitive/social components
            r1, r2 = np.random.rand(2, n_points)
            cognitive = c1 * r1 * (personal_best[i] - swarm[i])
            social = c2 * r2 * (global_best - swarm[i])
            velocities[i] = omega * velocities[i] + cognitive + social
            
            # Apply low-pass filter to velocity
            filtered_velocities[i] = alpha * filtered_velocities[i] + (1 - alpha) * velocities[i]
            
            # Update position
            swarm[i] += filtered_velocities[i]
            
            # Apply feasibility projection
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            new_fitness = evaluate_erdos_solution(swarm[i])
            
            # Update personal best
            if new_fitness < personal_best_fitness[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_fitness[i] = new_fitness
                
            # Update global best
            if new_fitness < global_best_fitness:
                global_best = swarm[i].copy()
                global_best_fitness = new_fitness
                
        # Add perturbation to escape local minima
        for i in range(n_particles):
            noise = np.random.normal(0, 0.01, n_points)
            swarm[i] += noise
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
    # Return best solution
    best_h = project_to_feasible(global_best, n_points)
    final_c5 = global_best_fitness
    
    return (best_h, final_c5, n_points)
