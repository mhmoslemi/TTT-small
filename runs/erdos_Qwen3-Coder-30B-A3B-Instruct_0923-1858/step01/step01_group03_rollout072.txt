import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Convex relaxation for initial solution
    def solve_convex_relaxation():
        # Variables
        h_vars = cp.Variable(n_points)
        lambda_var = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Add constraints from correlation
        # We'll add a few key constraints to get a good starting point
        for i in range(0, min(10, n_points)):
            k = i - (n_points - 1)  # shift index
            # For each shift k, we have: sum(h[i] * (1 - h[i+k])) >= lambda
            # This is approximated with a few constraints based on small slices
            if i < n_points - 1:
                # Simple pairwise constraint approximation
                constraints.append(h_vars[i] * (1 - h_vars[i+1]) >= lambda_var)
        
        # Objective
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return np.maximum(0, np.minimum(1, h_vars.value))
        except:
            pass
        
        # Fallback to simplex method
        return np.clip(h, 0, 1)
    
    # Run convex relaxation
    h = solve_convex_relaxation()
    h = project_feasible(h)
    
    # Trust region optimization
    def objective_and_gradient(h_vec):
        h_vec = np.array(h_vec)
        # This is max_k int h(x)(1 - h(x+k)) dx
        # We compute it rigorously...
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_val = corr[max_k_idx]
        
        # Estimate gradient (simplified)
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return -max_val * dx, -grad
    
    # Trust region method using scipy
    # We'll use a local optimizer with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    # Iterate with LBFGS
    patience = 0
    max_patience = 30
    etas = [0.01, 0.005, 0.001]
    eta_idx = 0
    
    for iteration in range(1000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Perturb if no improvement
        if patience > max_patience:
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)
            patience = 0
            eta_idx = min(eta_idx + 1, len(etas)-1)
        
        h = project_feasible(h)
        
        # Minimize with scipy
        try:
            res = minimize(
                lambda x: -compute_C5(x),
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 5}
            )
            if res.success:
                h = res.x.copy()
            else:
                # Fallback random perturbation
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        h[i] = np.clip(h[i] + np.random.normal(0, 0.01), 0, 1)
        except:
            # Fallback perturbation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.01), 0, 1)
        
        h = project_feasible(h)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
