import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Interpolate to 140 points to get better resolution
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, 140)
        h = np.interp(x_new, x_orig, h)
        n_points = 140
    else:
        n_points = 140
        h = np.ones(n_points) * 0.5  # Uniform initial guess
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Scale h to sum to 70 (since 2/140 = 1/70)
    h = h / np.sum(h) * 70
    
    # Define constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - 70.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective function
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation loop for global exploration
    n_perturb = 50
    iter_count = 0
    max_iter = 1000
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Perturb h
        h_perturbed = h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * 70
        
        # Compute C5 for the perturbed solution
        c5_perturbed = objective(h_perturbed)
        
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
        
        # Local refinement with SLSQP
        top_candidates = []
        for _ in range(n_perturb):
            h_trial = h + np.random.normal(0, 0.05, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * 70
            c5_trial = objective(h_trial)
            top_candidates.append((c5_trial, h_trial))
        
        # Sort candidates and refine top 10
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:10]
        
        for c5_trial, h_trial in top_candidates:
            try:
                res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except:
                pass
        
        iter_count += 1
        # Continue with small random steps to maintain dynamic exploration
        step_size = max(0.001, 0.05 * (1 - iter_count / max_iter))
        h = h + np.random.normal(0, step_size, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * 70
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
