import numpy as np
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.75
        h[1] = 0.25
        h = h * (n_points / 2) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Start optimization with projected gradient method
    max_iter = 1000
    c5_bound = 0.4
    step_size = 0.001
    tolerance = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve with CVXPY
        try:
            h_var = Variable(n_points)
            constraints = [
                h_var >= 0,
                h_var <= 1,
                sum(h_var) == n_points / 2
            ]
            
            # Add all overlap constraints
            dx = 2.0 / n_points
            for k in range(2 * n_points - 1):
                # Build the correlation constraint for shift k
                corr = 0
                for i in range(n_points):
                    j = i + k - (n_points - 1)
                    if 0 <= j < n_points:
                        corr += h_var[i] * (1 - h_var[j])
                constraints.append(corr <= c5_bound / dx)
            
            objective = Minimize(sum(h_var))  # Dummy objective, we just want feasible
            prob = Problem(objective, constraints)
            prob.solve(solver=ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = np.array(h_var.value).reshape(-1)
                new_h = np.clip(new_h, 0, 1)
                # Renormalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = evaluate_erdos_solution(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                c5_bound = best_c5 - step_size
            else:
                # If infeasible, try larger C5
                c5_bound += step_size
                
        except:
            # Fallback to simple optimization
            c5_bound += step_size
            
        # Safety: ensure we don't go too low
        c5_bound = max(c5_bound, 0.3)
        
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
