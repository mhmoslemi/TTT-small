import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def project_h(h, target_sum):
    """Project h to ensure sum equals target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h):
    """Compute gradient of C5 with respect to h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation manually for gradient
    corr = np.correlate(h, 1 - h, mode="full")
    max_lag = np.argmax(corr)
    # Create gradient based on maximum lag
    grad = np.zeros(n)
    for i in range(n):
        # Current contribution to max lag term
        if i + max_lag < n:
            grad[i] += (1 - 2 * h[i]) * dx
        if max_lag - i >= 0:
            grad[i] += (1 - 2 * h[max_lag - i]) * dx

    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use high resolution
    n_points = 150
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Initialize
    if 'initial_h_values' in globals():

        # Interpolate to n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = project_h(h, target_sum)
    else:
        # rough initial guess
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = project_h(h, target_sum)
    
    # Gradient descent parameters
    lr = 0.1
    decay = 0.95
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute current C5 and gradient
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adjust learning rate if needed
        if patience_counter > 200:
            lr = 0.1
            patience_counter = 0
        elif patience_counter > patience:
            lr *= decay
            
        # Take gradient step
        grad = compute_gradient(h)
        h = h - lr * grad
        h = project_h(h, target_sum)
        
        # Add some perturbations sometimes
        if patience_counter > 100:
            # Apply random noise
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            # Apply smoothing
            h = np.convolve(h, np.ones(3)/3, mode='same')
            h = project_h(h, target_sum)

    return (best_h, best_c5, n_points)
