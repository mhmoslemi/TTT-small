        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Genetic algorithm parameters
    population_size = 100
    generations = 100
    mutation_rate = 0.15
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def create_individual():
        # Create random individual
        individual = np.random.rand(n_points)
        # Normalize to sum = n_points / 2
        individual = individual * (n_points / 2) / np.sum(individual)
        # Clip to [0, 1]
        individual = np.clip(individual, 0, 1)
        return individual
    
    def project_feasible(individual):
        # Clip between 0 and 1
        individual = np.clip(individual, 0, 1)
        # Rescale so sum = n_points / 2
        current_sum = np.sum(individual)
        if current_sum > 0:
            individual = individual * (n_points / 2) / current_sum
        return individual
    
    def fitness(individual):
        # Compute correlation
        corr = correlate(individual, 1 - individual, mode="full")
        # Take maximum over all shifts
        return np.max(corr) * dx
    
    def tournament_selection(population, fitnesses):
        # Select tournament_size individuals
        tournament_indices = np.random.choice(len(population), 
                                             size=tournament_size, 
                                             replace=False)
        # Get fitness of the tournament
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        # Return index of best individual in tournament
        return tournament_indices[np.argmin(tournament_fitnesses)]
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        return child1, child2
    
    def mutate(individual):
        # Gaussian mutation
        mutation_mask = np.random.rand(n_points) < mutation_rate
        individual = np.copy(individual)
        individual[mutation_mask] += np.random.normal(0, 0.02, np.sum(mutation_mask))
        # Project to feasible region
        individual = project_feasible(individual)
        return individual
    
    # Create initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())

    patience = 0
    max_patience = 20
    for generation in range(generations):

        min_fitness_idx = np.argmin(fitnesses)

            best_C5 = fitnesses[min_fitness_idx]

            patience = 0

            patience += 1
            if patience > max_patience:
                # Reset population
                population = [h.copy()]
                for _ in range(population_size - 1):
                    population.append(create_individual())
                patience = 0
        
        # Create new population
        new_population = []
        # Keep elite
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())

        while len(new_population) < population_size:
            # Selection
            parent1_idx = tournament_selection(population, fitnesses)
            parent2_idx = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(population[parent1_idx], 
                                     population[parent2_idx])
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Keep population_size individuals
        population = new_population[:population_size]
    
    # Final evaluation
    final_C5 = fitness(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
