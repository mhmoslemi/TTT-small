import numpy as np
from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, norm
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Construct a feasible initial solution
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization settings
    perturbation_magnitude = 0.1
    min_perturbation = 1e-4
    max_perturbation = 0.5
    perturbation_decay = 0.95
    iterations_since_improvement = 0
    max_stagnation = 50
    
    def project_feasible(x):
        # Ensure bounds
        x = np.clip(x, 0, 1)
        # Normalize sum
        x = x * (n_points / 2) / np.sum(x)
        return x

    def objective(x):
        return evaluate_erdos_solution(x)

    # Main optimization loop
    counter = 0
    while time.time() - start_time < budget_s - 1:
        counter += 1
        try:
            # Random perturbation
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            x0 = project_feasible(best_h + perturbation)
            
            # Local optimization using L-BFGS-B
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 50})
            
            if res.success:
                current_h = project_feasible(res.x)
                current_c5 = evaluate_erdos_solution(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    iterations_since_improvement = 0
                    # Reduce perturbation magnitude if we're making progress
                    perturbation_magnitude = max(min_perturbation, perturbation_magnitude * perturbation_decay)
                else:
                    iterations_since_improvement += 1
                    # Increase perturbation if stagnating
                    if iterations_since_improvement > 20:
                        perturbation_magnitude = min(max_perturbation, perturbation_magnitude * 1.1)
            else:
                # Try again with smaller perturbation
                perturbation = np.random.normal(0, perturbation_magnitude / 2, n_points)
                x0 = project_feasible(best_h + perturbation)
                try:
                    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 20})
                    if res.success:
                        current_h = project_feasible(res.x)
                        current_c5 = evaluate_erdos_solution(current_h)
                        if current_c5 < best_c5:
                            best_h = current_h.copy()
                            best_c5 = current_c5
                            iterations_since_improvement = 0
                except:
                    pass

        except Exception as e:
            pass

        # Early termination
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
