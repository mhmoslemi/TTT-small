import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available

            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Create a feasible initial h vector
        n_points = 140
        h = np.zeros(n_points)
        # Place density in a central region
        mid = n_points // 2
        half_width = 20
        h[mid-half_width:mid+half_width] = 1.0
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Define constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define objective
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping with adaptive perturbation
    max_iter = 500
    iter_count = 0
    sigma = 0.05
    
    # Store best candidates to avoid local minima
    candidates = []
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break

        h_trial = h + np.random.normal(0, sigma, n_points)

        # Try to optimize this perturbed solution using SLSQP
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons,

            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                candidates.append((h_new, c5_new))
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
            
        iter_count += 1

        if iter_count % 50 == 0 and sigma > 0.005:
            sigma /= 2.0
            
        # Occasionally add some randomness to escape local minima
        if iter_count % 30 == 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h / np.sum(h) * (n_points / 2.0)

    if candidates:

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
