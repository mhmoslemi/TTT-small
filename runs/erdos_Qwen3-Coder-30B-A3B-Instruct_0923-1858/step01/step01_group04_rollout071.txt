import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a basic feasible solution
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach
    resolutions = [50, 100]  # Coarse to fine resolution
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create a population of candidate solutions
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)
            
        def constraint_integral(h_vec):
            return np.sum(h_vec) - res / 2
            
        bounds = [(0, 1) for _ in range(res)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        # Differential Evolution for global search
        de_bounds = [(0, 1) for _ in range(res)]
        
        def de_objective(x):
            # Reshape if needed and make feasible
            x = np.clip(x, 0, 1)
            # Normalize to integral = 1
            x = x * (res / 2) / np.sum(x)
            x = np.clip(x, 0, 1)
            return evaluate_erdos_solution(x)
        
        # Run DE for a fixed number of iterations
        try:
            # Use fewer iterations for faster search
            maxiter = min(30, max(10, int(500 * (budget_s - (time.time() - start_time)) / budget_s)))
            result = differential_evolution(
                de_objective, de_bounds, maxiter=maxiter, 
                popsize=40, mutation=(0.5, 1.0), recombination=0.9, 
                atol=1e-6, seed=seed, disp=False
            )
            
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_h = new_h * (res / 2) / np.sum(new_h)
                new_c5 = evaluate_erdos_solution(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
            
        except Exception:
            pass
            
        # Local refinement with SLSQP if we're at the final resolution
        if res == resolutions[-1] and time.time() - start_time < budget_s - 5:
            try:
                # Optimize the best found solution with refined method
                local_bounds = [(0, 1) for _ in range(res)]
                res_refine = minimize(
                    objective,
                    best_h,
                    method='SLSQP',
                    bounds=local_bounds,
                    constraints=constraints,
                    options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res_refine.success:
                    refined_h = np.clip(res_refine.x, 0, 1)
                    refined_h = refined_h * (res / 2) / np.sum(refined_h)
                    refined_c5 = evaluate_erdos_solution(refined_h)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
                        
            except Exception:
                pass

    # Final normalization to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, len(best_h))
