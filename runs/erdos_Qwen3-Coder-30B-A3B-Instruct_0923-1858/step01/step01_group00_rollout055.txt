import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use n=150 for better resolution
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points if needed
        if len(h) != n_points:
            h_orig = h.copy()
            x_orig = np.linspace(0, 2, len(h_orig))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct an initial feasible solution
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Projection function
    def project_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        return h_vals
    
    # Gradient estimation using finite differences
    def estimate_gradient(h_curr, eps=1e-5):
        grad = np.zeros_like(h_curr)
        base_c5 = evaluate_erdos_solution(h_curr)
        for i in range(len(h_curr)):
            h_plus = h_curr.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Main optimization loop
    best_c5 = float('inf')
    best_h = h_curr = h.copy()
    
    start_time = time.time()
    step_size = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s:
        # Estimate gradient
        try:
            grad = estimate_gradient(h_curr)
            # Update
            h_new = h_curr - step_size * grad
            # Project
            h_new = project_h(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            h_curr = h_new.copy()
            
            # Adaptive step size
            if stagnation_count > 10:
                step_size *= 0.95
                stagnation_count = 0
                
            # Perturb if stagnated
            if stagnation_count > max_stagnation:
                h_curr = best_h.copy() + np.random.uniform(-0.05, 0.05, n_points)
                h_curr = project_h(h_curr)
                stagnation_count = 0
                step_size = 0.1
                
        except Exception as e:
            # On error, fallback to best known
            if 'best_h' in locals():
                h_curr = best_h.copy()
            else:
                h_curr = np.ones(n_points) * 0.5
                h_curr[0] = 0.0
                h_curr[-1] = 0.0
                h_curr = h_curr / np.sum(h_curr) * (n_points / 2)
            step_size = 0.1
            stagnation_count = 0

    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
