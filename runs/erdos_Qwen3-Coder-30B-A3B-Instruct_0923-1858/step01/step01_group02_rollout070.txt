import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        h = np.ones(n_points) * 0.5
        # Add some variation
        h[0] = 0.7
        h[1] = 0.3
        h[2] = 0.6
        h[3] = 0.4
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Optimizer settings
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Cut-plane method
    n_iter = 0
    max_iter = 2000
    while time.time() - start_time < budget_s - 1 and n_iter < max_iter:
        n_iter += 1
        try:
            # Compute the overlap integrals
            n = len(h)
            dx = 2.0 / n
            corr = np.correlate(h, 1 - h, mode="full")
            overlaps = corr * dx
            max_k = np.argmax(overlaps)
            
            # Add a cutting plane
            # Try different strategies to vary the cutting planes
            if n_iter % 5 == 0:
                # Use the maximum overlap shift
                k = max_k
            else:
                # Use a random shift or use shift at middle
                k = n // 2
            
            # Compute gradient for shift k
            grad = np.zeros(n)
            for i in range(n):
                j = (i + k) % n
                grad[i] = dx * (1 - h[j])
            
            # Compute current value
            current_overlap = overlaps[max_k]
            
            # Create a relaxed constraint using the gradient
            h_new = h.copy()
            grad_step = 0.01
            h_new = np.clip(h_new + grad_step * grad, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Try multiple perturbations
            for _ in range(10):
                perturbation = np.random.normal(0, 0.005, n)
                x0 = np.clip(h_new + perturbation, 0, 1)
                x0 = x0 * (n_points / 2) / np.sum(x0)
                
                # Try to improve with local optimization
                try:
                    def objective(x):
                        return evaluate_erdos_solution(x)
                    
                    bounds = [(0, 1) for _ in range(n)]
                    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
                    
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
                    if res.success:
                        current_c5 = evaluate_erdos_solution(res.x)
                        if current_c5 < best_c5:
                            best_h = res.x.copy()
                            best_c5 = current_c5
                        h = res.x.copy()
                    else:
                        h = x0
                except:
                    h = x0
                    
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
        except Exception as e:
            pass
        
        # Occasionally resample
        if n_iter % 50 == 0:
            # Randomly intensify exploration
            h = best_h.copy()
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
