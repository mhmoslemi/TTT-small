import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Parameters for differential evolution
    de_params = {
        'bounds': [(0, 1) for _ in range(n_points)],
        'maxiter': 100,
        'popsize': 40,
        'mutation': (0.5, 1),
        'recombination': 0.7,
        'tol': 1e-6,
        'strategy': 'best1bin',
        'workers': 1
    }
    
    # Define objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_erdos_solution(h_vec)
    
    # Constraint for integral = 1
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Set up constraints
    constraints = {'type': 'eq', 'fun': constraint_integral}
    
    # Set up DE bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try DE for global optimization
    try:
        # Run differential evolution for global search
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=50,
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.7,
            tol=1e-6,
            seed=seed,
            workers=1
        )
        
        if de_result.success:
            h_de = np.clip(de_result.x, 0, 1)
            h_de = h_de * (n_points / 2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de)
            
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    
    except Exception as e:
        # Fallback to local optimization if DE fails
        pass

    # Local refinement using SLSQP
    try:
        # Run SLSQP for local refinement
        refined_result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        if refined_result.success:
            h_refined = np.clip(refined_result.x, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined)
            
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    
    except Exception as e:

            perturbation = np.random.normal(0, 0.05, n_points)
            h_trial = best_h + perturbation

            h_trial = np.clip(h_trial, 0, 1)

                    bounds=bounds,
                    constraints=constraints,

                pass

    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)
