import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a better starting point based on a sweep and refinement
        initial_h = np.linspace(0, 1, n_points//2, endpoint=False)
        initial_h = np.concatenate([initial_h, np.linspace(1, 0, n_points - n_points//2)])
        initial_h = project(initial_h, dx)
    
    # Begin time budget
    t0 = time.time()
    max_time = budget_s - 10  # Leave buffer for final processing
    
    # Best solution tracking
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    # Two phase optimization: gradient descent first to get close, then convex optimization
    h = initial_h.copy()
    
    # Phase 1: Gradient-based optimization
    lr = 0.01
    for i in range(5000):
        if time.time() - t0 > max_time:
            break
        if i % 100 == 0:
            lr *= 0.95
        # Stochastic gradient approximation
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        # Update step
        h = h - lr * grad
        h = project(h, dx)
        
        # Local search perturbations
        if i % 30 == 0 and np.random.rand() < 0.2:
            for _ in range(3):
                idx = np.random.randint(0, len(h))
                h[idx] += np.random.normal(0, 0.005)
            h = project(h, dx)
        
        # Check for improvement
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
    
    # Phase 2: Try convex relaxations (pony-up, we might not get here)
    try:
        # Use CVXPY for clean convex optimization if space permits
        H = cp.Variable(n_points, complex=True)
        # Symmetry constraint to ensure real values: H[0] is real, H[i] = conj(H[-i])
        # But for simplicity, we will use a simpler approach:
        h_real = cp.Variable(n_points, name="h_real")  
        h_real[0] = cp.real(H[0])
        # Optimization objective
        objective = cp.Minimize(cp.max(cp.abs(H * (1 - cp.conj(H)))))
        # Constraints
        constraints = [0 <= h_real, h_real <= 1,
                       cp.sum(h_real) * dx == 1]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, verbose=False, time_limit=max_time - (time.time() - t0))
        
        # Transform back
        h_opt = h_real.value
        if h_opt is not None:
            h_opt = project(h_opt, dx)
            C_opt = evaluate(h_opt, dx)
            if C_opt < best_C:
                best_C = C_opt
                best_h = h_opt
    except:
        pass  # Skip convex optimization if problems occur
    
    # Final check and return
    best_C = evaluate(best_h, dx)
    return (best_h, best_C, n_points)
