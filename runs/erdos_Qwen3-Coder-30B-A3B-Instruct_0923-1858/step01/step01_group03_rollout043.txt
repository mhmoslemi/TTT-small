import time
from scipy.signal import correlate

        # Normalize to sum to n_points / 2
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    pop_size = 100
    max_iter = 10000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_limit = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random particles
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    pbest = [swarm[i].copy() for i in range(pop_size)]
    pbest_fitness = [float('inf')] * pop_size
    
    # Initialize global best
    gbest_idx = 0
    gbest = swarm[gbest_idx].copy()
    gbest_fitness = float('inf')
    
    # Evaluate initial swarm
    for i in range(pop_size):
        h = swarm[i]
        corr = correlate(h, 1 - h, mode="full")
        fitness = np.max(corr) * (2.0 / n_points)
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest = h.copy()
            gbest_idx = i
    
    # Main PSO loop
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 50
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate fitness
            corr = correlate(swarm[i], 1 - swarm[i], mode="full")
            fitness = np.max(corr) * (2.0 / n_points)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest = swarm[i].copy()
                    stagnation_count = 0
            
        # Update inertia weight
        w = max(0.4, 0.7 - 0.3 * (iter_count / max_iter))
        
        # Check for stagnation
        stagnation_count += 1
        if stagnation_count > max_stagnation:
            # Reset some particles
            for _ in range(pop_size // 10):
                idx = np.random.randint(pop_size)
                swarm[idx] = np.random.rand(n_points)
                swarm[idx] = swarm[idx] * (n_points / 2) / np.sum(swarm[idx])
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                stagnation_count = 0
        
        iter_count += 1
    
    # Final evaluation
    final_corr = correlate(gbest, 1 - gbest, mode="full")
    final_fitness = np.max(final_corr) * (2.0 / n_points)
    
    return (gbest.tolist(), final_fitness, n_points)
