import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def mutate_individual(h, dx, mutation_strength=0.1):
    """Create a mutated version of h."""
    h_mut = h.copy()
    # Randomly perturb a few samples
    indices = np.random.choice(len(h), size=max(1, len(h) // 10), replace=False)
    for idx in indices:
        h_mut[idx] += np.random.normal(0, mutation_strength)
    return project(h_mut, dx)

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), size=k, replace=False)
    tournament_fitness = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available

        if len(initial_h) != n_points:

        initial_h = np.ones(n_points) * 0.5
        # Slightly perturb it for diversity
        initial_h = project(initial_h, dx)
    
    # Genetic Algorithm Parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        mutated = mutate_individual(initial_h, dx)
        population.append(mutated)
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate(individual, dx) for individual in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_C:
            best_C = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Select parents
        selected = []
        for _ in range(population_size):
            selected.append(tournament_selection(population, fitnesses))
        
        # Create new population via crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Crossover: blend two individuals
            alpha = np.random.rand()
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutate children
            if np.random.rand() < mutation_rate:
                child1 = mutate_individual(child1, dx)
            if np.random.rand() < mutation_rate:
                child2 = mutate_individual(child2, dx)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    return (best_h, best_C, n_points)
