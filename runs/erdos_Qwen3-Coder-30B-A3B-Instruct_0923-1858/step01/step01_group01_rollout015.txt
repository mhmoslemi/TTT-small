import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate(h, dx, rate=0.1, strength=0.1):
    """Mutate h with small random changes."""
    mutated = h.copy()
    for i in range(len(mutated)):
        if np.random.rand() < rate:
            mutated[i] += np.random.normal(0, strength)
    return project(mutated, dx)

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def select_parents(population, fitnesses, num_parents=10):
    """Tournament selection."""
    selected = []
    for _ in range(num_parents):
        idx = np.random.randint(0, len(population))
        best_idx = idx
        best_fitness = fitnesses[idx]
        for _ in range(4):
            idx = np.random.randint(0, len(population))
            if fitnesses[idx] > best_fitness:
                best_fitness = fitnesses[idx]
                best_idx = idx
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize
    population = initialize_population(n_points)
    best_h = None
    best_C = float('inf')
    
    t0 = time.time()
    generation = 0
    
    # GA iterations
    while time.time() - t0 < budget_s - 10:
        generation += 1
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c = evaluate(h, dx)
            fitnesses.append(1.0 / (c + 1e-12))
            if c < best_C:
                best_C = c
                best_h = h.copy()
                
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Generate new population
        new_population = []
        while len(new_population) < len(population):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child, dx, rate=0.2, strength=0.1)
            new_population.append(child)
        
        population = new_population[:len(population)]
        
    # Final check
    for h in population:
        c = evaluate(h, dx)
        if c < best_C:
            best_C = c
            best_h = h.copy()
    
    return (best_h, best_C, n_points)
