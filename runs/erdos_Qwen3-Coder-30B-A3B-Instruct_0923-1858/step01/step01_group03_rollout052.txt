import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Convex relaxation approach
    def solve_convex_relaxation():
        # Try to solve a relaxed version by setting up CVXOPT problem
        h_vars = cp.Variable(n_points)
        lambda_var = cp.Variable()
        constraints = [0 <= h_vars, h_vars <= 1]
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # For each lag k, compute the overlap constraint
        # Use finite differences to linearize constraints
        for k in range(n_points):
            # This is a simplified version. We'll execute the full constraint with most critical k
            # Just add constraint for some lag k to speed up
            if k < n_points // 2:
                # Approximate overlap for shift k
                # sum_i h[i] * (1 - h[i+k]) >= lambda_var
                expr = 0
                for i in range(n_points - k):
                    expr += h_vars[i] * (1 - h_vars[i + k])
                # Add some in-bounds handling
                constraints.append(expr >= lambda_var)
        
        objective = cp.Maximize(lambda_var)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=CP.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
                sol = h_vars.value
                return np.clip(sol, 0, 1)
        except:
            pass
        return None
    
    # Compute initial solution from convex relaxation if possible 
    # Use modest budget
    convex_time = 0.2 * budget_s
    if convex_time < 30:  # If enough time
        sub_start = time.time()
        h_candidate = solve_convex_relaxation()
        if h_candidate is not None:
            h = h_candidate
        convex_time_taken = time.time() - sub_start
    
    max_iter = int(100000)
    step_size = 0.01
    decay_rate = 0.999
    patience = 0
    max_patience = 500
    
    # Gradient descent with constraint handling
    def compute_gradient(h_vec, k_max):
        # Assume k_max is the shift that achieves the maximum
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[k_max + n_points - 1] - corr_minus[k_max + n_points - 1]) / (2 * eps)
        return grad
    
    # Random restart strategy with fine-tuning
    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Try random perturbation
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.02), 0, 1)
                h = project_feasible(h)
                patience = 0
        
        # Compute gradient
        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)
        
        grad = compute_gradient(h, max_k)
        h = h - step_size * grad
        h = project_feasible(h)
        
        # Update learning rate
        if iter % 100 == 0:
            step_size *= decay_rate
            
        # Occasionally sample an entirely new random solution
        if np.random.rand() < 0.01:
            h_new = np.random.rand(n_points)
            h_new = project_feasible(h_new)
            if compute_C5(h_new) < compute_C5(h):
                h = h_new
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
