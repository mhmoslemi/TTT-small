import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points if needed
        if n_points != 140:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_old, h)
            n_points = 140
    else:
        # Create a feasible starting h vector with 140 points
        n_points = 140
        # Start with a simple construction: uniform + small perturbation
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2.0)
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute correlation with (1-h)
        corr = np.correlate(h, 1-h, mode="full")
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Perturbation-based global search with adaptive sigma
    sigma = 0.05
    perturbation_count = 0
    while time.time() - start_time < budget_s - 1:
        # Generate perturbations
        num_perturbations = 50
        perturbations = []
        for _ in range(num_perturbations):
            h_pert = h + np.random.normal(0, sigma, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert / np.sum(h_pert) * (n_points / 2.0)
            c5 = objective(h_pert)
            perturbations.append((c5, h_pert))
            
        # Sort perturbations by objective value
        perturbations.sort(key=lambda x: x[0])
        
        # Take top 10 for local refinement
        top_perturbations = perturbations[:10]
        for c5, h_trial in top_perturbations:
            if c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5
                
        # Local refinement with SLSQP for top 10
        for c5, h_trial in top_perturbations[:5]:
            def con_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': con_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new / np.sum(h_new) * (n_points / 2.0)
                    c5_new = objective(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except:
                pass
        
        # Update h with best of perturbations
        if len(perturbations) > 0:
            h = perturbations[0][1].copy()
            
        # Adapt sigma
        perturbation_count += 1
        if perturbation_count % 50 == 0:
            sigma *= 0.5
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
