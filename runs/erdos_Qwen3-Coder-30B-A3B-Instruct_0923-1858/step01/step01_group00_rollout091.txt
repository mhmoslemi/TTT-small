import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    """
    Evaluates the Erdős constant C5 for a given step function h.
    """
    n = len(h_values)
    dx = 2.0 / n
    # Compute the overlap integral for all k values
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The overlap integral for each k
    overlaps = corr * dx
    # The maximum overlap for C5
    c5 = np.max(overlaps)
    return c5

        h = h / np.sum(h) * (n_points / 2)

        h[-1] = 0.0
        # Normalize to satisfy sum(h) * dx = 1
        h = h / np.sum(h) * (n_points / 2)

    # Use SLSQP for optimization with bounds and constraints
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Bounds for h: [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum(h) * dx = 1 (i.e., sum(h) = n_points/2)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Keep track of the best solution found
    best_c5 = float('inf')
    best_h = h.copy()

    # Start timer
    start_time = time.time()
    
    # Optimize until timeout
    while time.time() - start_time < budget_s:
        try:
            # Run optimization
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 500},
                tol=1e-6
            )
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:

    try:

            c5_now = evaluate_erdos_solution(h_tuned)
