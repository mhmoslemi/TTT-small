import numpy as np
import time
from collections import deque
from scipy.signal import correlate

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def generate_neighbor(h, n_points, perturb_rate=0.05):
    """Generate a neighbor by perturbing a small fraction of points"""
    neighbor = h.copy()
    n_perturb = max(1, int(n_points * perturb_rate))
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    for idx in indices:
        # Perturb by small random amount
        neighbor[idx] += np.random.normal(0, 0.05)
    return project_feasible(neighbor, n_points)
