import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Binary search bounds
    low = 0.0
    high = 0.38085
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()
    
    while time.time() - start_time < budget_s and high - low > 1e-5:
        mid = (low + high) / 2.0
        # Using a heuristic for feasibility check: try optimization
        # This is a simplified version of a more complex optimization
        def objective_trial(h_trial):
            return evaluate_erdos_solution(h_trial)

        # Perturbation for some variety
        h_perturbed = h.copy() + np.random.normal(0, 0.01, n_points) * (np.random.rand(n_points) > 0.7)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)

        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_trial):
            return np.sum(h_trial) - n_points / 2
        
        try:
            result = minimize(
                objective_trial,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 200}
            )
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
                
                if current_c5 <= mid:
                    high = mid
                    h = result.x.copy()
                else:
                    low = mid
            else:
                low = mid  # If optimization fails, move lower bound up
        except:
            low = mid

    # Final refinement using a local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        try:
            def objective_final(h_final):
                return evaluate_erdos_solution(h_final)
            
            result = minimize(
                objective_final,
                best_h.copy(),
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 100}
            )
            
            if result.success:
                current_c5 = evaluate_erdos_solution(result.x)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
        except:
            pass

    return (best_h, best_c5, n_points)
