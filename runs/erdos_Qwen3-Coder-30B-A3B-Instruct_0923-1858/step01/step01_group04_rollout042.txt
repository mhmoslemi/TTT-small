import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Returns the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Use initial construction if available, otherwise create feasible one
    if 'initial_h_values' in globals():

        n_points = len(h)

        h = np.ones(n_points) * 0.5  # Start with uniform
        # Adjust to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Multi-resolution approach: first coarse, then fine
    resolutions = [50, 100]
    for res in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # If not at target resolution, upsample
        if res > n_points:
            # Upsample by interpolation
            old_h = h.copy()
            new_h = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), old_h)
            # Adjust integral constraint
            new_h = new_h * (res / 2) / np.sum(new_h)
            h = np.clip(new_h, 0, 1)
            n_points = res
            
        # Local optimization phase using SLSQP for fine tuning
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec)
            
        def constraint_integral(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_integral}
        
        # Try multiple local optimizations
        local_optimizations = min(10, max(1, int(500 / (n_points // 10))))
        
        for _ in range(local_optimizations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb slightly
            perturbation = np.random.normal(0, 0.05, n_points)
            h_trial = h + perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            try:
                res = minimize(
                    objective,
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6}
                )
                
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    c5_candidate = evaluate_erdos_solution(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
                        
            except Exception:
                pass
                
        # Refinement loop to target worst overlaps
        if time.time() - start_time > budget_s - 1:
            break
            
        max_iter_refine = 20
        iter_count = 0
        while time.time() - start_time < budget_s - 1 and iter_count < max_iter_refine:
            # Identify the shift that gives maximum overlap
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            # Get shift index of maximum overlap (middle position is k=0)
            max_idx = np.argmax(corr)
            max_k = max_idx - (len(best_h) - 1)
            
            # Compute adjustments: adjust to reduce the problematic contribution
            # For shift k, we want to reduce the overlap h[i] * (1 - h[i+k])
            adjustment = np.zeros_like(best_h)
            for i in range(len(best_h)):
                j = i + max_k
                if 0 <= j < len(best_h):
                    adjustment[i] += (1 - best_h[j])  # larger adjustment if h[j] is small
                    
            # Apply adjustment
            adjusted = best_h - 0.2 * adjustment
            adjusted = np.clip(adjusted, 0, 1)
            
            # Rescale to maintain constraint
            adjusted = adjusted * (n_points / 2) / np.sum(adjusted)
            
            new_c5 = evaluate_erdos_solution(adjusted)
            if new_c5 < best_c5:
                best_h = adjusted
                best_c5 = new_c5
            iter_count += 1
            
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    return (list(best_h), final_c5, n_points)
