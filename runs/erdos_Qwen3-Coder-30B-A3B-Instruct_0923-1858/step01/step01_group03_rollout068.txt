import time
from scipy.signal import correlate

        # Create a triangular initial solution

        h_initial = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # PSO parameters
    pop_size = 100
    max_iter = 10000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_limit = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random particles
    for _ in range(pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests and global best
    pbest = [swarm[i].copy() for i in range(pop_size)]
    pbest_scores = [float('inf')] * pop_size
    gbest_idx = 0
    gbest_score = float('inf')
    
    # Initialize population scores
    for i in range(pop_size):
        h = swarm[i]
        corr = correlate(h, 1 - h, mode="full")
        score = np.max(corr) * (2.0 / n_points)
        pbest_scores[i] = score
        if score < gbest_score:
            gbest_score = score
            gbest_idx = i
    
    # PSO main loop
    stagnation_count = 0
    iter_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        iter_count += 1
        if iter_count > max_iter:
            break
            
        # Update each particle
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (swarm[gbest_idx] - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            if current_sum > 0:
                swarm[i] = swarm[i] * (n_points / 2) / current_sum
            
            # Evaluate fitness
            corr = correlate(swarm[i], 1 - swarm[i], mode="full")
            score = np.max(corr) * (2.0 / n_points)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = swarm[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest_idx = i
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        # Reset some particles if stagnation
        if stagnation_count > 50:
            for _ in range(pop_size // 10):
                idx = np.random.randint(pop_size)
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest_scores[idx] = float('inf')
            stagnation_count = 0
            
        # Decay inertia weight
        if iter_count % 100 == 0:
            w = max(0.3, w * 0.99)
    
    # Return best solution
    return (swarm[gbest_idx].tolist(), gbest_score, n_points)
