import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős constant C_5 for a given step function h.
    """
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals are stored in the center of the correlation
    # We want the maximum overlap over all k values
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        # Make it more structured
        h[0:20] = 0.7
        h[20:40] = 0.2
        h[40:60] = 0.8
        h[60:] = 0.3
        # Normalize to meet constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds [0, 1]
    h = np.clip(h, 0, 1)
    
    # Create bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sum constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    def objective(x):
        return evaluate_erdos_solution(x)
    
    # Optimization settings
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Switch to a hybrid strategy: gradient based optimization
    # and perturbation to escape local minima.
    counter = 0
    while time.time() - start_time < budget_s - 1:
        counter += 1
        try:
            # Try different optimization methods
            opt_methods = ['SLSQP', 'L-BFGS-B']
            
            for method in opt_methods:
                # Random perturbation to escape local minima
                perturbation = np.random.normal(0, 0.01, n_points)
                x0 = np.clip(best_h + perturbation, 0, 1)
                
                # Renormalize
                x0 = x0 * (n_points / 2) / np.sum(x0)
                
                opt_kwargs = {
                    'method': method,
                    'bounds': bounds,
                    'constraints': cons,
                    'options': {'maxiter': 100}
                }
                
                res = minimize(objective, x0, **opt_kwargs)
                
                if not res.success:
                    continue
                
                current_c5 = evaluate_erdos_solution(res.x)
                if current_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = current_c5
                    break
            
            # Try aggressive relaxation if necessary
            if counter % 10 == 0:
                for _ in range(5):
                    perturbation = np.random.normal(0, 0.05, n_points)
                    x0 = np.clip(best_h + perturbation, 0, 1)
                    x0 = x0 * (n_points / 2) / np.sum(x0)
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons)
                    if res.success:
                        current_c5 = evaluate_erdos_solution(res.x)
                        if current_c5 < best_c5:
                            best_h = res.x.copy()
                            best_c5 = current_c5
                            
        except Exception:
            pass
            
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 0.1:
            break

    # Final evaluation and trimming
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
