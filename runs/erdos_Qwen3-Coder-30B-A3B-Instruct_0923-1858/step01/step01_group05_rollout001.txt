import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    # Compute the overlap integral for given h
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # The maximum overlap term is the maximum value in corr
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points
        if n_points < 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
        elif n_points > 140:
            # Downsample appropriately
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
            n_points = 140
    else:
        # Create initial h
        n_points = 140
        h = np.zeros(n_points)
        # Place a few peaks to satisfy target quality
        # gradually increasing density towards center
        center = n_points // 2
        spread = n_points // 3
        for i in range(n_points):
            # create a density profile similar to the initial values but more refined
            dist = abs(i - center)
            if dist < spread:
                h[i] = 1.0 - (dist / spread) * 0.5
            else:
                h[i] = 0.0
        h = h / np.sum(h) * (n_points / 2.0)
    
    h = np.clip(h, 0, 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin hopping with reuse of previous good points
    # 200 global steps, plus refinement of top 5 best results
    max_iter = 200
    iter_count = 0
    best_candidates = []
    
    # Set adaptive perturbation scale for basin hopping
    sigma = 0.05
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Add noise to current h
        h_trial = h + np.random.normal(0, sigma, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial / np.sum(h_trial) * (n_points / 2.0)
        
        # SLSQP local optimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                best_candidates.append((h_new.copy(), c5_new))
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        iter_count += 1
        if iter_count % 50 == 0 and sigma > 1e-4:
            sigma /= 2  # decrease noise as we get better
    
    # Sort candidates
    best_candidates.sort(key=lambda x: x[1])
    best_candidates = best_candidates[:5]
    
    # Refine top candidates more precisely
    for h_refine, _ in best_candidates:
        try:
            res = minimize(objective, h_refine, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
    
    # Final validation that constraints are satisfied
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points / 2.0) > 1e-6:
        # Reproject to correct sum
        best_h = best_h / final_sum * (n_points / 2.0)
    
    # Ensure final h is in [0, 1]
    best_h = np.clip(best_h, 0, 1)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
