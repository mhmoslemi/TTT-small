import numpy as np
import time
from scipy.optimize import minimize

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

            h[i] = 0.5 * (1 + np.sin(2 * np.pi * phase / (n_points / 3) - np.pi/2))
        elif phase < 2 * n_points / 3:
            h[i] = 0.5 * (1 + np.sin(2 * np.pi * (phase - n_points/3) / (n_points / 3) - np.pi/2))
        else:
            h[i] = 0.5 * (1 + np.sin(2 * np.pi * (phase - 2*n_points/3) / (n_points / 3) - np.pi/2))
    
    # Ensure it's in range [0,1] and sum is correct
    h = project(h, 2.0 / n_points)
    return h

            h = project(res.x.copy(), dx)
