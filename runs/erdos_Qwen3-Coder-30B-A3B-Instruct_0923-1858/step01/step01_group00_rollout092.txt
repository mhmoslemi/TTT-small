import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    overlaps = corr * dx
    c5 = np.max(overlaps)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Simple piecewise constant construction
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    # Keep track of the best solution
    best_c5 = float('inf')
    best_h = h.copy()

    # Start the timer
    start_time = time.time()

    # Binary search on C5
    c_low = 0.0
    c_high = 0.38085

    while time.time() - start_time < budget_s * 0.9 and (c_high - c_low) > 1e-5:
        c_mid = (c_low + c_high) / 2.0
        # Check feasibility with simple heuristic: try optimization with c_mid constraint
        # In practice, we'll do gradient-based optimization and adjust search bounds
        
        # Try optimization with a fixed target
        # Modifying objective to have a soft constraint
        def objective(h):
            c5 = evaluate_erdos_solution(h)
            return c5
        
        # SLSQP optimization with box constraints and sum constraint
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 100},
                tol=1e-6
            )

                # Adjust binary search bounds
                if current_c5 <= c_mid:
                    c_high = c_mid
                else:
                    c_low = c_mid
                    
                # Perturb for escape
                h = result.x.copy() + np.random.normal(0, 0.01, n_points) * 0.1
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
            else:
                # If failed, reverse search or new restart
                h = np.random.rand(n_points) * 0.5
                h = h / np.sum(h) * (n_points / 2)
                
        except Exception:
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
            
    # Refinement: optimize with best known C5
    if time.time() - start_time < budget_s * 0.95:
        # Attempt one more strong optimization
        def objective(h):
            c5 = evaluate_erdos_solution(h)
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]

            return np.sum(h) - n_points / 2

                        best_h = result.x.copy()
            except:
                pass
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
