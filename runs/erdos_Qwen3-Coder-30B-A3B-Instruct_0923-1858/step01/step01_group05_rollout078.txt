import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
        # Interpolate to 140 points for higher resolution
        if n_points != 140:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 140)
            h = np.interp(x_new, x_orig, h)
        n_points = 140
    else:
        # Create a starting vector
        n_points = 140
        h = np.ones(n_points) * 0.5
        # Add some clustering to center
        center_start = n_points // 3
        center_end = 2 * n_points // 3
        h[center_start:center_end] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Define constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function
    dx = 2.0 / n_points
    def objective(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx

    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Basin-hopping with 200 iterations
    perturb_std = 0.05
    for hop in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb with Gaussian noise
        h_trial = h + np.random.normal(0, perturb_std, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2.0) / np.sum(h_trial)
        
        # Local optimization
        try:
            res = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_new = res.x
                c5_new = objective(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new.copy()
        except:
            pass
        
        # Reduce perturbation over time
        if hop % 50 == 0 and hop > 0:
            perturb_std /= 2
            
    # Final refinement step with tightened tolerance
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h.copy()
                best_c5 = final_c5
    except:
        pass
    
    # Final validate
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
