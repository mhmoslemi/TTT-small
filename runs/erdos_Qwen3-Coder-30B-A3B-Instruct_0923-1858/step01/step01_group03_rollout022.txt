    def convex_relaxation():

        h_vars = cp.Variable(n_points)
        lambda_var = cp.Variable()

        # Objective: maximize lambda
        objective = cp.Maximize(lambda_var)

        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Create correlation constraints (approximate for now)
        # We need to model h ⋆ (1 - h) >= lambda for all shifts
        # This is complex for general case, so use a simpler approximation
        # by making one of the maximum possible terms dominate
        
        # For now, we'll focus on use some known structure
        # A better approach is to use more sophisticated relaxation
        # but we'll proceed with simple trust region method
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                return np.maximum(0, np.minimum(1, h_vars.value))
        except:
            pass
        return None
    
    # Try convex relaxation
    relaxed_h = convex_relaxation()
    if relaxed_h is not None:
        h = relaxed_h
    
    # Gradient descent with trust region approach
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # SQP-like trust region update
    def trust_region_step(h_vec, eta=0.01, max_iter=10):
        h_current = h_vec.copy()
        C5_current = compute_C5(h_current)
        
        for _ in range(max_iter):
            # Compute the maximum overlap index
            corr = correlate(h_current, 1 - h_current, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_current) - 1)

                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update step
            h_new = h_current - eta * grad
            h_new = project_feasible(h_new)

                C5_current = C5_new

                patience = 0

                h = trust_region_step(h, 0.005)

        h = trust_region_step(h, 0.01)
        
        iter_count += 1
        
        # Cooling schedule
        if iter_count % 50 == 0:

    # Final evaluation

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
