import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.9
    inertia_decay = 0.995
    cognitive = 2.0
    social = 2.0
    velocity_clamp = 0.1
    
    # Initialize swarm
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    p_best = np.zeros((swarm_size, n_points))
    p_best_fitness = np.full(swarm_size, float('inf'))
    
    # Initialize particles around the initial solution
    for i in range(swarm_size):
        if i == 0:
            particles[i] = h.copy()
        else:
            # Add perturbations
            perturbation = np.random.normal(0, 0.05, n_points)
            particles[i] = h + perturbation
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        p_best[i] = particles[i].copy()
        
    # Compute initial fitness
    for i in range(swarm_size):
        corr = correlate(particles[i], 1 - particles[i], mode="full")
        fit = np.max(corr) * dx
        p_best_fitness[i] = fit
    
    # Global best
    global_best_idx = np.argmin(p_best_fitness)
    global_best = p_best[global_best_idx].copy()
    global_best_fitness = p_best_fitness[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        # Update inertia
        inertia = max(0.4, inertia * inertia_decay)
        
        # Update each particle
        for i in range(swarm_size):
            # Compute velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            cognitive_component = cognitive * r1 * (p_best[i] - particles[i])
            social_component = social * r2 * (global_best - particles[i])
            
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            corr = correlate(particles[i], 1 - particles[i], mode="full")
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best_fitness[i] = fitness
                p_best[i] = particles[i].copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = particles[i].copy()
        
        # Check for stagnation
        if iteration % 30 == 0 and iteration > 0:
            # Perturb particles
            for i in range(swarm_size):
                if i == 0:
                    # Keep the global best unchanged
                    continue
                perturbation = np.random.normal(0, 0.03, n_points)
                particles[i] = global_best + perturbation
                particles[i] = np.clip(particles[i], 0, 1)
                # Rescale
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Update best solution if needed
        if global_best_fitness < best_C5:
            best_C5 = global_best_fitness
            best_h = global_best.copy()
                
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode="full")
    final_C5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_C5, len(best_h))
