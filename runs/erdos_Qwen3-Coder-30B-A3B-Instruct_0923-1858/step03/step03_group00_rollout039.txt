import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def crossover(parent1, parent2):
    """Perform one-point crossover between two parents."""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate, generation, max_generations):
    """Mutate an individual with decreasing mutation scale."""
    n = len(individual)
    mutation_scale = 0.1 * (1 - generation / max_generations)
    for i in range(n):
        if np.random.random() < mutation_rate:
            individual[i] += np.random.normal(0, mutation_scale)
    return individual

    mutation_rate = 0.05

    # Sort population by fitness
