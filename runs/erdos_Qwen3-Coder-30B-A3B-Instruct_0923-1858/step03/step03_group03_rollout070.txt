import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Use a sin-based construction
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.sin(np.pi * x) / 2 + 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Enhanced optimization with learning rate scheduling
    def adaptive_gradient_descent(h_vec, max_iter=50, initial_lr=0.02):
        h_new = h_vec.copy()
        lr = initial_lr
        for i in range(max_iter):
            # Compute correlation and gradient at max overlap
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_new) - 1)
            
            # Approximate gradient via finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-5
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
            
            # Adaptive learning rate decay
            if i % 20 == 0:
                lr *= 0.95
        return h_new

    # Perturbation function for escaping local optima
    def perturb_solution(h_vec, magnitude=0.08):
        h_new = h_vec.copy()
        # Change random 10% of values
        idxs = np.random.choice(len(h_vec), size=max(1, len(h_vec) // 10), replace=False)
        for idx in idxs:
            h_new[idx] += np.random.normal(0, magnitude)
        return project_feasible(h_new)

    patience = 0
    max_patience = 200
    iter_count = 0
    lr = 0.02

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = perturb_solution(h, 0.1)
                patience = 0

        # Optimize using enhanced gradient steps
        h = adaptive_gradient_descent(h, max_iter=40, initial_lr=lr)
        
        # Adjust learning rate and noise level
        if iter_count % 100 == 0 and iter_count > 0:
            lr *= 0.92

        iter_count += 1

    # Final refinement
    final_h = adaptive_gradient_descent(best_h, max_iter=50, initial_lr=0.01)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
