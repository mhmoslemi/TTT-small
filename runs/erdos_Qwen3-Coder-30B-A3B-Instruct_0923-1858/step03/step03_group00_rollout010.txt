import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def mutate_individual(h, mutation_rate, mutation_scale):
    """Mutate an individual with Gaussian noise."""
    mask = np.random.rand(len(h)) < mutation_rate
    if not np.any(mask):
        # Ensure at least one mutation
        mask[np.random.randint(len(h))] = True
    h_new = h.copy()
    h_new[mask] += np.random.normal(0, mutation_scale, np.sum(mask))
    h_new = project(h_new, 1.0 / len(h))  # Assume dx = 1/n_points for mutation
    return h_new

def crossover(parent1, parent2):
    """Perform one-point crossover."""
    n = len(parent1)
    if n <= 1:
        return parent1.copy()
    crossover_point = np.random.randint(1, n)
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def tournament_selection(population, fitnesses, tournament_size=3):
    """Perform tournament selection."""
    selected_indices = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_idx)
    return [population[i] for i in selected_indices]
