import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # minimize sum h[i]^2 subject to sum h[i] = n_points/2
        # to get a "spread-out" solution
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    def check_feasible(C, h_vec):
        # Create a more accurate check by evaluating all shifts up to a limit
        # For efficiency, we'll just use a subset of key shifts
        shifts = np.arange(-n_points+1, n_points)
        max_shift = min(20, len(shifts) // 2)
        sample_shifts = shifts[::max(1, len(shifts) // (2 * max_shift))]
        h_vec = np.array(h_vec)
        
        # Check a subset of shifts for feasibility
        ll = len(sample_shifts)
        for k in sample_shifts[-max_shift:]:
            k_idx = k + n_points - 1
            if k_idx < 0 or k_idx >= 2 * n_points - 1:
                continue
            corr = np.sum(h_vec * np.roll(1 - h_vec, k))
            if corr < 1 - C:
                return False
        return True
    
    def trust_region_update(h_vec, eta=0.01, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Simple finite difference gradient
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            base_corr = compute_C5(h_new)
            
            for j in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[j] += eps
                h_plus = project_feasible(h_plus)
                corr_plus = compute_C5(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[j] -= eps
                h_minus = project_feasible(h_minus)
                corr_minus = compute_C5(h_minus)
                
                grad[j] = (corr_plus - corr_minus) / (2 * eps)
            
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    def perturb_solution(h_vec, prob=0.15):
        h_new = h_vec.copy()
        idxs = np.random.choice(n_points, size=int(prob * n_points), replace=False)
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.04), 0, 1)
        return project_feasible(h_new)
    
    # Try convex relaxation first to get a good warm start
    if elapsed() < budget_s * 0.6:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass

    # Bisection loop to find the minimal C5
    C_low = 0.0
    C_high = 1.0
    epsilon = 1e-5
    max_iter = 0
    
    while (C_high - C_low > epsilon) and (elapsed() < budget_s * 0.95):
        C_mid = (C_low + C_high) / 2
        
        # Try to find a feasible solution for this C
        h_try = h.copy()
        for _ in range(30):
            trial_C5 = compute_C5(h_try)
            if trial_C5 <= C_mid + 1e-4:
                # Found a good enough solution
                if trial_C5 < best_C5:
                    best_C5 = trial_C5
                    best_h = h_try.copy()
                break
            # gradient descent to reduce overlap
            h_try = trust_region_update(h_try, eta=0.005)
            h_try = project_feasible(h_try)
            
        # Then try to do a local search
        if compute_C5(h_try) <= C_mid + 1e-4:
            # feasible
            C_low = C_mid
            h = h_try.copy()
        else:
            C_high = C_mid

        max_iter += 1
        
        # Occasionally perturb to escape local minima
        if max_iter % 10 == 0 and elapsed() < budget_s * 0.9:
            h = perturb_solution(h)
    
    # Final check of best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
