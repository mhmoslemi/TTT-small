import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter_pso = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize PSO swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.02, n_points)
        else:
            particle = np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_points) - 1)) + np.random.normal(0, 0.02, n_points)
        particle = project_feasible(particle)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(compute_C5(particle))
    
    # Global best
    gb_idx = np.argmin(pbest_vals)
    gbest = pbest[gb_idx].copy()
    gbest_val = pbest_vals[gb_idx]
    
    # PSO optimization
    for iteration in range(max_iter_pso):
        if elapsed() > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            curr_val = compute_C5(particles[i])
            
            # Update personal best
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
                
                # Update global best
                if curr_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = curr_val
        
        # Early stopping condition
        if gbest_val < best_C5:
            best_C5 = gbest_val
            best_h = gbest.copy()

    # Refinement with L-BFGS-B
    def lbfgs_objective(h_vec):
        return compute_C5(h_vec)
    
    # Create a better initial guess for L-BFGS
    refined_h = project_feasible(gbest.copy())
    
    # Use scipy.optimize.minimize with L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        result = minimize(
            lbfgs_objective,
            refined_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6},
            callback=lambda x: None
        )
        if result.success:
            refined_h = project_feasible(result.x)
        final_C5 = compute_C5(refined_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = refined_h.copy()
    except:
        pass
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
