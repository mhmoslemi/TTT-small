import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def mutate_individual(h, mutation_rate, mutation_scale):
    """Mutate an individual by adding Gaussian noise."""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, mutation_scale, len(h))
    h_mutated = h + noise * mask
    return project(h_mutated, 2.0/len(h))

def crossover(parent1, parent2):
    """Single-point crossover between two parents."""
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def selection(population, fitnesses, num_select):
    """Tournament selection."""
    selected = []
    for _ in range(num_select):
        idx = np.random.choice(len(population), size=3, replace=False)
        best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    pop_size = 20
    generations = 5000
    mutation_rate = 0.05
    mutation_scale_start = 0.1
    
    # Initialize population
    population = []
    try:
        h_init = initial_h_values
        if len(h_init) != n_points:

        population.append(h_init.copy())
    except NameError:

        h_init = 0.3 + 0.2 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        h_init = project(h_init, dx)
        population.append(h_init)
    
    # Fill rest of population with diverse individuals
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = project(h, dx)
        population.append(h)
    
    best_h = population[0].copy()
    best_C = evaluate(best_h, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    
    for gen in range(generations):
        if time.time() - t0 > max_time:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate(ind, dx) for ind in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_C:
            best_C = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection
        selected = selection(population, fitnesses, pop_size // 2)
        
        # Crossover and mutation
        new_population = [population[min_fitness_idx].copy()]  # Elitism
        new_population.append(best_h.copy())
        
        # Replace half the population
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(selected, size=2, replace=False)
            if np.random.random() < 0.8:  # Crossover probability
                c1, c2 = crossover(p1, p2)
            else:
                c1, c2 = p1.copy(), p2.copy()
            
            # Mutate children
            mutation_scale = mutation_scale_start * (1 - gen / generations)
            c1 = mutate_individual(c1, mutation_rate, mutation_scale)
            c2 = mutate_individual(c2, mutation_rate, mutation_scale)
            
            new_population.extend([c1, c2])
        
        population = new_population[:pop_size]
    
    # Final validation
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
