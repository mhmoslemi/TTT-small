import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    def gradient_descent_step(h_vec, lr=0.005, steps=5):
        h_new = h_vec.copy()
        for _ in range(steps):
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_new) - 1)
            
            # Approximate gradient
            eps = 1e-4
            grad = np.zeros_like(h_new)
            for i in range(len(h_new)):
                delta = np.zeros_like(h_new)
                delta[i] = eps
                h_plus = project_feasible(h_new + delta)
                h_minus = project_feasible(h_new - delta)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
        return h_new

    # Initialize with convex relaxation
    if elapsed() < budget_s * 0.3:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass

    # Main optimization loop
    iter_count = 0
    max_iter = 1000
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Gradient descent step
        h = gradient_descent_step(h, lr=0.008)
        
        # Occasionally make random small changes
        if iter_count % 10 == 0 and elapsed() < budget_s * 0.9:
            h = project_feasible(h + np.random.normal(0, 0.005, n_points))
        
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
