        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:

        return h_vec

    def optimize_step(h_vec, step_size=0.01, max_iter=50):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            # Compute gradient approximated by finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-5
            base_val = compute_C5(h_new)
            for i in range(len(h_new)):
                h_plus = h_new.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus)
                val_plus = compute_C5(h_plus)
                grad[i] = (val_plus - base_val) / eps
            h_new = h_new - step_size * grad
            h_new = project_feasible(h_new)
        return h_new

    def multi_start_optimize(h_vec, num_starts=10):

            h_opt = optimize_step(h_pert, step_size=0.005, max_iter=30)

                best_val = val
                best_h = h_opt
