import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Simple initial construction: step function
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[:n_points//3] = 1.0
        h[2*n_points//3:] = 0.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Feasible projection
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Adaptive differential evolution approach
    def adaptive_de(h_vec, max_iter=100):
        # Simple: use random mutations with line search
        best_h_local = h_vec.copy()
        best_C5_local = compute_C5(h_vec)
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Mutate
            h_mut = best_h_local.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.05)
            h_mut[idx] += delta
            
            # Project and evaluate
            h_mut = project_feasible(h_mut)
            C5_mut = compute_C5(h_mut)
            
            if C5_mut < best_C5_local:
                best_C5_local = C5_mut
                best_h_local = h_mut.copy()
        
        return best_h_local, best_C5_local

    # Gradient descent update
    def gradient_descent_step(h_vec, max_iter=20, learning_rate=0.02):
        h_new = h_vec.copy()
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Numerical gradient estimate
            eps = 1e-5
            grad = np.zeros_like(h_new)
            base_val = compute_C5(h_new)
            
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_plus[j] += eps
                h_plus = project_feasible(h_plus)
                
                val_plus = compute_C5(h_plus)
                grad[j] = (val_plus - base_val) / eps
            
            # Step
            h_new = h_new - learning_rate * grad
            h_new = project_feasible(h_new)
            
        return h_new

    # Main optimization loop
    patience = 0
    max_patience = 50
    
    while elapsed() < budget_s * 0.95:
        # Gradient descent with restarts
        h_new = gradient_descent_step(h, max_iter=50, learning_rate=0.02)        
        h_new, c5_new = adaptive_de(h_new, max_iter=10)
        
        if c5_new < best_C5:
            best_C5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                h = np.clip(h + np.random.normal(0, 0.03, n_points), 0, 1)
                h = project_feasible(h)
                patience = 0
            else:
                h = h_new
                
        h = h_new

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
