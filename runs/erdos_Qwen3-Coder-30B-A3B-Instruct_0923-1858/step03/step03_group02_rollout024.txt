import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.9
    cognitive = 2.0
    social = 2.0
    velocity_clamp = 0.1
    particle_dim = n_points
    
    # Initialize swarm
    swarm = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    # Base particle from initial solution
    swarm.append(h.copy())
    velocities.append(np.zeros_like(h))
    p_best.append(h.copy())
    p_best_fitness.append(float('inf'))
    
    # Generate perturbed particles
    for i in range(1, swarm_size):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Rescale to integral constraint
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        swarm.append(perturbed)
        velocities.append(np.random.uniform(-velocity_clamp, velocity_clamp, n_points))
        p_best.append(perturbed.copy())
        p_best_fitness.append(float('inf'))
    
    # Global best tracking
    s_best = h.copy()
    s_best_fitness = float('inf')
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Iteration loop
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 30
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        if iter_count % 50 == 0:
            # Decay inertia
            inertia = max(0.4, inertia * 0.995)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive_component = cognitive * r1 * (p_best[i] - swarm[i])
            social_component = social * r2 * (s_best - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project into feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate fitness
            fitness = compute_C5(swarm[i])
            
            # Update particle best
            if fitness < p_best_fitness[i]:
                p_best[i] = swarm[i].copy()
                p_best_fitness[i] = fitness
            
            # Update swarm best
            if fitness < s_best_fitness:
                s_best = swarm[i].copy()
                s_best_fitness = fitness
                no_improvement_count = 0
            else:
                no_improvement_count += 1

        # Perturb if no improvement
        if no_improvement_count > max_no_improvement:
            for i in range(swarm_size):
                swarm[i] += np.random.normal(0, 0.03, n_points)
                swarm[i] = project_feasible(swarm[i])
            no_improvement_count = 0

    # Final validation
    final_fitness = compute_C5(s_best)
    if final_fitness < s_best_fitness:
        s_best_fitness = final_fitness
        
    return (s_best.tolist(), s_best_fitness, n_points)
