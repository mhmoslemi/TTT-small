import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def evaluate_gradients(h_vec, eps=1e-4):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            delta = np.zeros_like(h_vec)
            delta[i] = eps
            
            h_plus = project_feasible(h_vec + delta)
            h_minus = project_feasible(h_vec - delta)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad

    def gradient_step(h_vec, lr=0.01):
        grad = evaluate_gradients(h_vec)
        h_new = h_vec - lr * grad
        return project_feasible(h_new)

    def apply_turbulence(h_vec, amplitude=0.1):
        # Apply sine-based turbulence
        turbulence = amplitude * np.sin(np.linspace(0, np.pi, len(h_vec))) * np.random.normal(0, 1, len(h_vec))
        h_turb = h_vec + turbulence
        return project_feasible(h_turb)

    def get_random_perturbation(h_vec, magnitude=0.05):
        delta = np.random.normal(0, magnitude, len(h_vec))
        h_new = h_vec + delta
        return project_feasible(h_new)

    # Initial convex relaxation to get a good starting point
    if elapsed() < budget_s * 0.3:
        try:
            h_var = cp.Variable(n_points)
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            objective = cp.Minimize(cp.sum_squares(h_var))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            h = np.maximum(0, np.minimum(1, h_var.value))
        except:
            pass

    # Main optimization loop
    max_iter = 10000
    iter_count = 0
    min_C5 = float('inf')
    patience = 0
    max_patience = 50

    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        # Apply turbulence occasionally
        if iter_count % 10 == 0:
            h = apply_turbulence(h, amplitude=0.02)
        
        # Gradient-based refinement
        for _ in range(5):
            if elapsed() > budget_s * 0.95:
                break
            h = gradient_step(h, lr=0.005)

        # Random perturbation to escape local optima
        if iter_count % 20 == 0:
            h = get_random_perturbation(h, magnitude=0.01)
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # Adaptive patience-based restart
        if patience > max_patience:
            if elapsed() < budget_s * 0.8:
                h = apply_turbulence(h, amplitude=0.05)
                patience = 0
            else:
                # Final gradient descent refinement only
                for _ in range(20):
                    if elapsed() > budget_s * 0.95:
                        break
                    h = gradient_step(h, lr=0.001)

        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
