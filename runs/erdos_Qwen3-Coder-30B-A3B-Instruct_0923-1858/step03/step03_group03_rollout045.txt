import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    try:

        n_points = len(h)
    except NameError:
        # Create a smooth initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.exp(-((x - 1) ** 2) / 0.1)
        h = h / np.sum(h) * (n_points / 2)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def gradient_descent_step(h_vec, lr=0.02):
        h_vec = project_feasible(h_vec)
        # Compute gradient numerically
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_C5 = compute_C5(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            
            new_C5 = compute_C5(h_plus)
            grad[i] = (new_C5 - base_C5) / eps
            
        h_new = h_vec - lr * grad
        h_new = project_feasible(h_new)
        return h_new

    def perturb_solution(h_vec, strength=0.05):
        h_new = h_vec.copy()
        # Randomly perturb a few points
        num_perturb = max(1, int(0.1 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_perturb, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    # Main optimization loop
    iter_count = 0
    lr = 0.02
    max_patience = 50
    patience = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            lr *= 0.95
            
        # Optimization step
        h = gradient_descent_step(h, lr)
        
        # Perturbation if not improving
        if patience > max_patience:
            h = perturb_solution(h, 0.02)
            patience = 0
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
