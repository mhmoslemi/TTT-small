        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simplified gradient-free approach
    def local_search(h_init, max_iter=100, step_size=0.02):
        h = h_init.copy()
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            # Perturb the function
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            h_new[idx] += np.random.normal(0, step_size)
            h_new = project_feasible(h_new)
            
            if compute_C5(h_new) < compute_C5(h):
                h = h_new
                if compute_C5(h) < best_C5:
                    best_C5 = compute_C5(h)
                    best_h = h.copy()
        return h

    # Multi-start strategy combining local search and perturbations
    convergence_patience = 0
    max_patience = 100
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            convergence_patience = 0
        else:
            convergence_patience += 1
            if convergence_patience > max_patience:
                # Perturb if no improvement seen
                h = np.random.rand(n_points) * 0.5  # Random starting point
                h = project_feasible(h)
                convergence_patience = 0

        # Local search improvement
        h = local_search(h, max_iter=50, step_size=0.01)
        
        # Fine-tune with small local steps
        for _ in range(10):
            if elapsed() > budget_s * 0.95:

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
