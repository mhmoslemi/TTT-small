import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 1000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_clamp = 0.1

    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    # Initialize with the provided solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    p_best.append(h.copy())
    p_best_fitness.append(compute_C5(h))

    # Add perturbed versions
    for i in range(1, n_particles):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        perturbed = project_feasible(perturbed)
        particles.append(perturbed)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        p_best.append(perturbed.copy())
        p_best_fitness.append(compute_C5(perturbed))

    # Global best
    g_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[g_best_idx].copy()
    g_best_fitness = p_best_fitness[g_best_idx]

    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 30

    # Main PSO loop
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            new_velocity = (
                inertia * velocities[i] +
                cognitive_weight * r1 * (p_best[i] - particles[i]) +
                social_weight * r2 * (g_best - particles[i])
            )
            
            # Clamp velocity
            new_velocity = np.clip(new_velocity, -velocity_clamp, velocity_clamp)
            
            # Update position
            new_position = particles[i] + new_velocity
            
            # Apply feasibility projection
            new_position = project_feasible(new_position)
            
            # Evaluate new position
            new_fitness = compute_C5(new_position)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = new_position.copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best if needed
                if new_fitness < g_best_fitness:
                    g_best = new_position.copy()
                    g_best_fitness = new_fitness
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1

            # Store updated position and velocity
            particles[i] = new_position
            velocities[i] = new_velocity

        # Perturb particles if no improvement
        if no_improvement_count > max_no_improvement:
            for i in range(n_particles):
                perturbation = np.random.normal(0, 0.03, n_points)
                particles[i] = project_feasible(particles[i] + perturbation)
                velocities[i] = np.random.uniform(-0.05, 0.05, n_points)
            no_improvement_count = 0

        iter_count += 1

    # Final check
    final_fitness = compute_C5(g_best)
    if final_fitness < g_best_fitness:
        g_best_fitness = final_fitness
        
    return (g_best.tolist(), g_best_fitness, n_points)
