            h[i] = max(0, 1 - abs(x[i] - 1))

        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Improved gradient estimator with finite differences
    def get_gradient(h_vec, num_samples=10):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        # Get the actual shift that causes maximum correlation
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Finite difference with multiple perturbations and averaging
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        
        for _ in range(num_samples):
            # Sample random perturbation
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    # Turbulence mechanism to escape local minima
    def apply_turbulence(h_vec, amplitude=0.1):
        # Create a periodic perturbation
        turbulence = amplitude * np.sin(np.pi * np.arange(len(h_vec)) / len(h_vec)) * np.random.normal(0, 1, len(h_vec))
        h_turb = h_vec + turbulence
        return project_feasible(h_turb)

    # Local refinement with gradient descent
    def local_refinement(h_vec, steps=20, eta=0.01):
        h_new = h_vec.copy()
        for _ in range(steps):
            if elapsed() > budget_s * 0.95:
                break
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)

            last_best_update = iter_count

            patience += 1

        # Apply turbulence periodically
        if iter_count % turbulence_freq == 0:

            # Occasionally refine after turbulence
            if iter_count % (turbulence_freq * 2) == 0:
                h = local_refinement(h, steps=10, eta=eta)
        else:
            # Local refinement direction
            h = local_refinement(h, steps=1, eta=eta)
        
        # Adaptive learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.96

            patience = 0
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
