import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution with 45 points
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Use a more structured approach with trust region
    def optimize_step(h_vec, max_iter=100):
        # Define constraints
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
            {'type': 'ineq', 'fun': lambda x: x},
            {'type': 'ineq', 'fun': lambda x: 1 - x}
        ]
        
        # Objective function for scipy minimize
        def obj(h_vec):
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return -np.max(corr) * dx  # negative for minimization
        
        # Optimize
        try:
            result = minimize(
                obj,
                h_vec,
                method='SLSQP',
                constraints=constraints,
                bounds=[(0, 1) for _ in range(len(h_vec))],
                options={'maxiter': max_iter, 'disp': False}
            )
            if result.success:
                return result.x
        except:
            pass
        return h_vec
    
    # Mutation function with different strategies
    def perturb_solution(h_vec, mutation_strength=0.03):
        h_new = h_vec.copy()
        # Mix of different perturbations
        num_mutations = max(1, int(0.1 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_mutations, replace=False)
        
        # Add random noise
        for idx in indices:
            h_new[idx] += np.random.normal(0, mutation_strength)
            
        return project_feasible(h_new)

    iter_count = 0
    patience = 0
    max_patience = 50
    eta = 0.01
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb and reset patience
                h = perturb_solution(h, 0.02)
                patience = 0

        # Optimize with trust region steps
        h_opt = optimize_step(h, max_iter=30)
        if compute_C5(h_opt) < compute_C5(h):
            h = h_opt
        else:
            # Perform local perturbations
            h = perturb_solution(h, 0.01)
            
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count % 100 == 0 and eta > 1e-6:
            eta *= 0.95

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
