import time
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        # Compute the maximum overlap integral
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        # Ensure h is in [0,1] and sum equals n_points/2
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def objective(h_vec):
        # Minimize C5
        return compute_C5(h_vec)

    def constraint_sum(h_vec):
        # Sum constraint
        return np.sum(h_vec) - n_points / 2

    # Create bounds (0 to 1 for each h[i])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for optimization
    constraints = {'type': 'eq', 'fun': constraint_sum}

    # Phase 1: Differential Evolution for global search
    de_result = differential_evolution(
        objective, 
        bounds, 
        constraints=constraints,
        maxiter=int(budget_s * 0.6 / 5),  # Roughly 5 iterations per second
        popsize=15,
        seed=seed,
        init='latinhypercube'
    )
    
    if elapsed() < budget_s * 0.7:
        final_h = project_feasible(de_result.x)
        final_C5 = compute_C5(final_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = final_h.copy()

    # Phase 2: Local refinement with L-BFGS-B
    if elapsed() < budget_s * 0.95:
        # Use the best solution so far as starting point for local optimization
        x0 = best_h.copy()
        
        # Define bounds for L-BFGS-B
        lb = [0.0] * n_points
        ub = [1.0] * n_points
        
        # Optimization parameters
        options = {'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-5}
        
        # Define constraint for sum
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        try:
            # Use L-BFGS-B for refinement
            result = minimize(
                objective, 
                x0, 
                method='L-BFGS-B', 
                bounds=list(zip(lb, ub)), 
                constraints=cons,
                options=options,
                callback=lambda x: None if elapsed() < budget_s else None
            )
            
            if result.success and compute_C5(result.x) < best_C5:
                best_h = project_feasible(result.x)
                best_C5 = compute_C5(best_h)
        except Exception as e:
            pass

    # Final check on best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
