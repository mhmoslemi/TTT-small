    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, num_shifts=20):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute gradient direction through finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    def refine_solution(h_vec, max_iter=30, learning_rate=0.01):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new)
            h_new = h_new - learning_rate * grad
            h_new = project_feasible(h_new)
        return h_new

    def perturb_solution(h_vec, mutation_strength=0.05):
        h_new = h_vec.copy()
        num_mutations = max(1, int(0.1 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_mutations, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, mutation_strength)
        return project_feasible(h_new)

    convergence_patience = 0
    max_patience = 100
    iter_count = 0
    eta = 0.01  # initial learning rate
    decay_rate = 0.97

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)

            convergence_patience = 0

            if convergence_patience > max_patience:
                h = perturb_solution(h, 0.05)
                convergence_patience = 0

        h = refine_solution(h, max_iter=30, learning_rate=eta)
        
        if iter_count % 30 == 0 and iter_count > 0:
            eta *= decay_rate
            
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
