import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 30
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def perturb_h(h_vec, strength=0.1):
        h_new = h_vec.copy()
        idx = np.random.randint(len(h_vec))
        h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iter_count = 0
    refinement_interval = 1000
    max_refinement = 5
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        h_new = perturb_h(h, 0.1)
        C5_old = compute_C5(h)
        C5_new = compute_C5(h_new)
        
        delta = C5_new - C5_old
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        
        T *= alpha
        iter_count += 1
        
        # Periodically refine resolution
        if iter_count % refinement_interval == 0 and n_points < 500:
            # Double the grid size if under limit
            n_points *= 2
            old_h = h.copy()
            h = np.zeros(n_points)
            x_old = np.linspace(0, 2, len(old_h))
            x_new = np.linspace(0, 2, n_points)
            
            # Interpolate
            for i in range(n_points):
                x = x_new[i]
                idx = np.searchsorted(x_old, x)
                if idx == 0:
                    h[i] = old_h[0]
                elif idx == len(x_old):
                    h[i] = old_h[-1]
                else:
                    x0, x1 = x_old[idx-1], x_old[idx]
                    y0, y1 = old_h[idx-1], old_h[idx]
                    h[i] = y0 + (y1 - y0) * (x - x0) / (x1 - x0)
            
            h = project_feasible(h)
            if compute_C5(h) < best_C5:
                best_C5 = compute_C5(h)
                best_h = h.copy()

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
