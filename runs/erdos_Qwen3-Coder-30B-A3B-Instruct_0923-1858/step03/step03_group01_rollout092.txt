import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def is_valid_binary(h, n_points):
    """Check if h is a valid binary vector with exactly n_points/2 ones."""
    return np.all(np.logical_or(h == 0, h == 1)) and np.sum(h) == n_points / 2

def initialize_population(n_points, pop_size):
    """Initialize a population of binary vectors with exactly n_points/2 ones."""
    population = []
    for _ in range(pop_size):
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, int(n_points / 2), replace=False)
        h[ones_indices] = 1
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
    return population[best_idx]

def crossover(parent1, parent2):
    """Perform single-point crossover on binary vectors."""
    n_points = len(parent1)
    point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Fix the count of ones to be exactly n_points/2
    ones1 = np.sum(child1)
    ones2 = np.sum(child2)
    target = int(n_points / 2)
    
    if ones1 != target:
        # Find indices to flip
        zero_indices = np.where(child1 == 0)[0]
        one_indices = np.where(child1 == 1)[0]
        n_flip = abs(ones1 - target)
        
        if ones1 > target:
            # Too many ones, flip some zeros to ones
            flip_indices = np.random.choice(zero_indices, size=min(n_flip, len(zero_indices)), replace=False)
            child1[flip_indices] = 1
        else:
            # Not enough ones, flip some ones to zeros
            flip_indices = np.random.choice(one_indices, size=min(n_flip, len(one_indices)), replace=False)
            child1[flip_indices] = 0
    
    if ones2 != target:
        # Find indices to flip
        zero_indices = np.where(child2 == 0)[0]
        one_indices = np.where(child2 == 1)[0]
        n_flip = abs(ones2 - target)
        
        if ones2 > target:
            # Too many ones, flip some zeros to ones
            flip_indices = np.random.choice(zero_indices, size=min(n_flip, len(zero_indices)), replace=False)
            child2[flip_indices] = 1
        else:
            # Not enough ones, flip some ones to zeros
            flip_indices = np.random.choice(one_indices, size=min(n_flip, len(one_indices)), replace=False)
            child2[flip_indices] = 0
    
    return child1, child2

def mutate(h, mutation_rate):
    """Mutate a binary vector."""
    n_points = len(h)
    num_mutations = int(n_points * mutation_rate)
    indices_to_flip = np.random.choice(n_points, num_mutations, replace=False)
    h[indices_to_flip] = 1 - h[indices_to_flip]
    
    # Ensure exactly n_points/2 ones
    ones_count = np.sum(h)
    target = int(n_points / 2)
    if ones_count != target:
        # Flip the excess or deficit
        if ones_count > target:
            # Too many ones, flip some to zeros
            ones_indices = np.where(h == 1)[0]
            n_flip = ones_count - target
            flip_indices = np.random.choice(ones_indices, size=n_flip, replace=False)
            h[flip_indices] = 0
        else:
            # Not enough ones, flip some zeros to ones  
            zero_indices = np.where(h == 0)[0]
            n_flip = target - ones_count
            flip_indices = np.random.choice(zero_indices, size=n_flip, replace=False)
            h[flip_indices] = 1
    
    return h
