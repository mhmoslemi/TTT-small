import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a modified triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_gradients(h_vec, k_max):
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Forward difference approximation of gradient
            eps = 1e-5
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[k_max + len(h_vec) - 1] - corr_minus[k_max + len(h_vec) - 1]) / (2 * eps)
        return grad
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Unroll QP solver with CVXPY
    max_patience = 30
    patience = 0
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        # Compute all overlaps and find the most constraining one
        h_current = project_feasible(best_h)
        corr = correlate(h_current, 1 - h_current, mode="full")
        k_max = np.argmax(corr) - (len(h_current) - 1)
        
        # Create CVXPY variables
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints for h_new
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0  # Ensure t is not negative
        ]
        
        # Linearization of the most constraining constraint:
        # t >= sum h[i] * (1 - h[i + k_max])
        grad_k = compute_gradients(h_current, k_max)
        
        # Ensure k_max is valid for indexing
        if k_max >= 0:
            # For elements with valid indices forward
            linear_constraint = []
            for i in range(len(h_current)):
                if i + k_max >= 0 and i + k_max < len(h_current):
                    linear_constraint.append(i)
        else:
            linear_constraint = []
            for i in range(len(h_current)):
                if i + k_max >= 0 and i + k_max < len(h_current):
                    linear_constraint.append(i)
        
        # Approximate trend of the constraint using the gradient and current h
        linearized_constraint = t - corr[k_max + len(h_current) - 1] >= grad_k @ (h_new - h_current)
        constraints.append(linearized_constraint)
        
        # Try to solve the QP
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, tol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new_sol = h_new.value
                t_sol = t.value
                
                if t_sol < best_C5 and np.all(h_new_sol >= 0) and np.all(h_new_sol <= 1):
                    best_C5 = t_sol
                    best_h = h_new_sol.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
        except:
            patience += 1
        
        # Random perturbation if needed
        if patience > max_patience:
            for i in range(n_points):
                if np.random.random() < 0.1:
                    h_new_sol = best_h.copy()
                    h_new_sol[i] = np.clip(best_h[i] + np.random.normal(0, 0.05), 0, 1)
            h_new_sol = project_feasible(h_new_sol)
            patience = 0
            # Evaluate and update best
            current_C5 = compute_C5(h_new_sol)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_new_sol.copy()
        
        iter_count += 1
    
    # Make sure final solution is feasible
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, len(final_h))
