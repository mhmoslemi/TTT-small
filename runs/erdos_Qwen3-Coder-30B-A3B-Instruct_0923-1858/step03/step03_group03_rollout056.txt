import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_relaxed(h_vec, C_temp):
        """Solve a relaxed version to check feasibility for C_temp"""
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1]
        
        # Add normalization constraint
        constraints.append(sum(h) == n_points / 2)
        
        # Linearization of correlation constraints: use outer approximation
        # Approximate each h[i]*(1-h[i+k]) with linear terms around current h_vec
        shifted_h = np.roll(h_vec, 1)
        for k in range(n_points):
            # Linear approximation
            h_shifted = np.roll(h_vec, k)
            # Use a line through h_vec[i] and h_vec[i+k] to approximate the product
            # h[i]*(1-h[i+k]) ~ h[i] - h[i]*h[i+k] ~ h[i] - h_vec[i] * h_vec[i+k] + (h_vec[i] - h[i])*(h_vec[i+k] - h[i+k])
            # But simplifying, assume h[i] ~ h_vec[i], so h[i]*(1-h[i+k]) ~ h_vec[i]*(1-h_vec[i+k])
            # Quadratic penalty just to keep constraints simple, not exact
            h_shift = np.roll(h, k)
            constraints.append(sum(h * (1 - h_shift)) <= C_temp / dx)
        
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return True, h.value
        except:
            pass
        return False, None

    def local_refine(h_vec, max_iter=100, lr=0.01):
        """Local gradient-based refinement"""
        h_new = h_vec.copy()
        h_new = project_feasible(h_new)
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.9:
                break
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_new) - 1)
            
            # Approximate gradient
            grad = np.zeros_like(h_new)
            for i in range(len(h_new)):
                eps = 1e-4
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
            lr *= 0.995
        return h_new

    # Binary search on the bound C_5
    C_low = 0.0
    C_high = 0.38085
    C_best = 1.0
    h_best = h.copy()
    
    iter_count = 0
    while elapsed() < budget_s * 0.95 and C_high - C_low > 1e-4:
        C_mid = (C_low + C_high) / 2
        # This step can be simplified with a convex optimization approach.
        # For now, use a simplified heuristic.
        
        # Use clone of original heuristic but with better line search
        h_refined = local_refine(h, max_iter=30)
        C_test = compute_C5(h_refined)
        
        if C_test <= C_mid:
            C_high = C_mid
            h_best = h_refined.copy()
            C_best = C_test
        else:
            C_low = C_mid
        
        # Occasionally do more local refinement
        if iter_count % 5 == 0:
            h_refined = local_refine(h, max_iter=10)
            C_test = compute_C5(h_refined)
            if C_test < C_best:
                C_best = C_test
                h_best = h_refined.copy()
            
        h = h_refined
        
        iter_count += 1
        
        if C_best < best_C5:
            best_C5 = C_best
            best_h = h_best.copy()

    # Final local refinement
    h_final = local_refine(h_best, max_iter=50)
    final_C5 = compute_C5(h_final)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = h_final.copy()

    return (best_h.tolist(), best_C5, len(best_h))
