import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - 1.5 * np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    eta = 0.01
    patience = 0
    max_patience = 30
    iter_count = 0
    max_iter = 1000
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb the solution
                noise = np.random.normal(0, 0.05, n_points)
                h = np.clip(h + noise, 0, 1)
                h = project_feasible(h)
                patience = 0
        
        # Find the most constraining shift
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr) - (len(h) - 1)
        
        # Compute linearized gradient for this shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k_max >= 0 and i + k_max < len(h):
                grad[i] = -(1 - h[i + k_max])
            if i - k_max >= 0 and i - k_max < len(h):
                grad[i] += h[i - k_max]
        
        # Solve QP: minimize t such that t >= C_k_max + grad_k^T * (h_new - h)
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= corr[np.argmax(corr)] * dx + grad @ (h_new - h)
        ]
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, tol=1e-7)
            if prob.status == cp.OPTIMAL:
                h_new_val = h_new.value
                # Ensure the solution is within bounds
                h_new_val = np.clip(h_new_val, 0, 1)
                # Re-scale to maintain sum constraint
                h_new_val = h_new_val * (n_points / 2) / np.sum(h_new_val)
                h = h_new_val
        except:
            pass  # Fall back to simple update if optimization fails
        
        iter_count += 1
    
    # Final refinement
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
