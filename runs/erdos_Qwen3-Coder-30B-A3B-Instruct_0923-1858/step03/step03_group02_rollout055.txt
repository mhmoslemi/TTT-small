import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution with steeper decay
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_overlaps(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Only keep valid correlations (non-zero lags)
        return corr[len(h_vec)-1:]

    def compute_gradient(h_vec, k):
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        # For shift k, compute gradient of correlation
        for i in range(len(h_vec)):
            # Handle boundary conditions with zero padding
            if i + k < len(h_vec) and i - k >= 0:
                grad[i] = (1 - h_vec[i + k]) - h_vec[i - k]
            elif i + k < len(h_vec):
                grad[i] = (1 - h_vec[i + k])  # Only right neighbor contributing
            elif i - k >= 0:
                grad[i] = -h_vec[i - k]  # Only left neighbor contributing
            else:
                grad[i] = 0  # No contribution from neighbors
        return grad
    
    max_patience = 30
    patience = 0
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation to escape local minima
                mask = np.random.rand(n_points) < 0.1
                h[mask] += np.random.normal(0, 0.05, np.sum(mask))
                h = project_feasible(h)
                patience = 0

        # Compute overlaps and find most constraining shift
        overlaps = compute_overlaps(h)
        k_max = np.argmax(overlaps)
        max_shift = k_max - (len(h) - 1)

        # Linearize constraint around current point
        grad = compute_gradient(h, max_shift)
        h_new = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Define QP problem
        objective = cp.Minimize(t)
        constraints = [
            cp.sum(h_new) == n_points / 2,
            h_new >= 0,
            h_new <= 1,
            t >= cp.sum(cp.multiply(grad, (h_new - h))) + overlaps[k_max]
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_new.value
                if h_sol is not None:
                    h = project_feasible(h_sol)
        except:
            # Fallback to a simple gradient step
            grad = compute_gradient(h, max_shift)
            h_new = h - 0.01 * grad
            h = project_feasible(h_new)
            # Also make sure our constraint is respected by projection
            h = project_feasible(h)

        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Final projection to maintain constraints
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
