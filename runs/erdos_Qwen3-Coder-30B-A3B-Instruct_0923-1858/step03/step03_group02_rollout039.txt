            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def local_improve(h_vec, max_iter=20):
        h_current = h_vec.copy()
        for _ in range(max_iter):
            improved = False
            # Perturb each point slightly
            for i in range(n_points):
                h_trial = h_current.copy()
                # Small random perturbation
                delta = np.random.normal(0, 0.02)

                h_trial = project_feasible(h_trial)

                    improved = True
