    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def gradient_descent_step(h_vec, learning_rate=0.01):
        h_vec = project_feasible(h_vec)
        # Empirical gradient estimated via finite differences
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        base_C5 = compute_C5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            grad[i] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
        return project_feasible(h_vec - learning_rate * grad)

    def random_perturbation(h_vec, magnitude=0.03):
        h_new = h_vec + np.random.normal(0, magnitude, len(h_vec))
