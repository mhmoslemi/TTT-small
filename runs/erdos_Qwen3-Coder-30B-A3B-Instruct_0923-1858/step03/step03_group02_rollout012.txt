import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # More sophisticated optimization approach
    def optimize_block(h_vec, block_size=5):
        # Break into chunks and optimize each
        new_h = h_vec.copy()
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for i in range(0, n_points, block_size):
            block_indices = indices[i:i+block_size]
            if len(block_indices) == 0:
                continue
                
            # Optimize this block
            def obj_block(block_params):
                # Reconstruct h with optimized block
                h_temp = new_h.copy()
                for j, idx in enumerate(block_indices):
                    if j < len(block_params):
                        h_temp[idx] = block_params[j]
                h_temp = project_feasible(h_temp)
                return compute_C5(h_temp)
            
            # Use a simple bounded optimization
            bounds = [(0, 1) for _ in block_indices]
            initial = [new_h[i] for i in block_indices]
            
            try:
                result = minimize(obj_block, initial, bounds=bounds, method='L-BFGS-B', 
                                options={'ftol': 1e-8, 'maxiter': 50})
                if result.success:
                    for j, idx in enumerate(block_indices):
                        if j < len(result.x):
                            new_h[idx] = result.x[j]
            except:
                pass  # Skip if optimization fails
            
        return project_feasible(new_h)

    # Initialize with the best result
    current_h = h.copy()
    
    # Annealing schedule
    max_iter = 10000
    current_iter = 0
    temperature = 1.0
    min_temperature = 1e-6
    decay_rate = 0.9995
    
    # Try different strategies
    strategies = [
        'gradient', 'block', 'mutation', 'combined'
    ]
    
    def gradient_update(h_vec, learning_rate=0.01):
        """Simple gradient-based update"""
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Simple gradient estimation
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] = min(1.0, h_vec[i] + eps)
            h_minus[i] = max(0.0, h_vec[i] - eps)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        h_new = h_vec - learning_rate * grad
        return project_feasible(h_new)

    # Main optimization loop
    last_improvement = 0

        strategy = strategies[current_iter % len(strategies)]
        
        if strategy == 'gradient':
            current_h = gradient_update(current_h, learning_rate=0.01)
        elif strategy == 'block':
            current_h = optimize_block(current_h)
        elif strategy == 'mutation':
            # Random perturbation with gradually decreasing temperature
            current_h = project_feasible(current_h + np.random.normal(0, 0.01 * temperature, n_points))
            current_h = project_feasible(current_h)
        else:  # combined
            # Try multi-step approach
            current_h = gradient_update(current_h, learning_rate=0.005)
            current_h = optimize_block(current_h)
            current_h = project_feasible(current_h)
            
        # Check for improvement
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
            last_improvement = current_iter
        
        # Temperature decay
        if temperature > min_temperature:
            temperature *= decay_rate
            
        if current_iter - last_improvement > 500:
            # Reset with a better randomization
            reset_h = current_h.copy()
            for i in range(n_points):
                if np.random.random() < 0.1:
                    reset_h[i] = np.clip(reset_h[i] + np.random.normal(0, 0.05), 0, 1)
            reset_h = project_feasible(reset_h)
            if compute_C5(reset_h) < current_C5:
                current_h = reset_h
                
        current_iter += 1

    # Final refinement of best solution
    final_C5 = compute_C5(best_h)
    
    # Run one final optimization pass on best solution
    try:
        def refinement_objective(x):
            return compute_C5(project_feasible(x))
        
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        result = minimize(refinement_objective, best_h, method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 1000, 'ftol': 1e-12})
        if result.success:
            final_h = project_feasible(result.x)
            final_C5 = compute_C5(final_h)
            if final_C5 < best_C5:
                best_C5 = final_C5
                best_h = final_h
    except:
        pass
        
    return (best_h.tolist(), final_C5, len(best_h))
