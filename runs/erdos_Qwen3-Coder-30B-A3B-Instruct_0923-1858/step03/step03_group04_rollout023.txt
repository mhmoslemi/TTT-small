import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a better initial solution: smooth bell-shaped function
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.exp(-((x - 1) ** 2) / 0.1)
        # Normalize to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use binary search to find the minimal C5
    def search_for_C5():
        low, high = 0.0, 1.0
        epsilon = 1e-5
        best_feasible_h = None
        best_C5_found = float('inf')
        
        max_iterations = 30
        iteration = 0
        
        while low < high and iteration < max_iterations and elapsed() < budget_s * 0.95:
            mid = (low + high) / 2.0
            # Feasibility check using qp approach
            # Minimize (1 - h)^T * h - mid * (1 - h)^T * h both involving overlapping
            # Same as maximizing sum over all k: \int h(x) (1 - h(x+k)) dx
            # This is actually maximizing the minimum correlation structure
            
            # Instead of solving a QP every time, we'll just evaluate and use local search dirtily
            
            if iteration > 0:
                # Use current best_h as warm start
                current_h = best_h.copy()
            else:
                current_h = h.copy()
                
            # -- Local search around this point
            step_size = 0.005
            for _ in range(100):
                if elapsed() > budget_s * 0.9:
                    break
                # Compute gradient-like
                corr_vals = correlate(current_h, 1 - current_h, mode="full")
                max_k_idx = np.argmax(corr_vals)
                shift = max_k_idx - (n_points - 1)
                
                # Simple subspace step in direction of largest gradient
                direction = np.zeros_like(current_h)
                # The idea is to shift values to reduce the largest overlap
                # For simplicity, we adjust a few points
                for i in range(n_points):
                    # Simple heuristic gradient approximation
                    h_plus = current_h.copy()
                    h_minus = current_h.copy()
                    
                    h_plus[i] = min(1.0, h_plus[i] + step_size)
                    h_minus[i] = max(0.0, h_minus[i] - step_size)
                    
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    
                    # Since shifts potentially are not exact, just sample a few key indices
                    grad = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * step_size)
                    direction[i] = grad
                
                # Apply update
                updated_h = current_h - 0.001 * direction
                # Project
                updated_h = project_feasible(updated_h)
                current_h = updated_h
            
            # Compute final C5 for this mid point
            current_C5 = compute_C5(current_h)
            
            if current_C5 < best_C5_found:
                best_C5_found = current_C5
                best_feasible_h = current_h.copy()
            
            # Now construct another version using a different pattern
            h_v2 = np.zeros_like(current_h)
            # Create piecewise constant
            for i in range(n_points):
                h_v2[i] = max(0, min(1, 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)))
            h_v2 = project_feasible(h_v2)
            c5_v2 = compute_C5(h_v2)
            if c5_v2 < best_C5_found:
                best_C5_found = c5_v2
                best_feasible_h = h_v2.copy()

            # Even better, try direct CVXPy feasibility check
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    h_var >= 0,
                    h_var <= 1,
                    cp.sum(h_var) == n_points / 2
                ]
                
                # Sample a subset of constraints
                # We'll evaluate a few key shifts based on peak correlation
                corr = correlate(current_h, 1 - current_h, mode="full")
                max_k_idx = np.argmax(corr)
                max_k = max_k_idx - (n_points - 1)
                
                # Sample some additional qualitative k values that are likely important
                shift_samples = [max_k, -max_k, 0, max(-max_k, -10), min(max_k, 10)]
                shift_samples = [int(s) for s in shift_samples if -n_points < s < n_points]
                
                available_shifts = set(shift_samples)
                # Make sure we sample a few more based on the support -- collapse shifts near zero
                if max_k < 0:
                    available_shifts.update([-1, -3, -(n_points//2), -(n_points//3)])
                else:
                    available_shifts.update([1, 3, n_points//2, n_points//3])
                
                # Limit number of shifts due to budget
                available_shifts = list(available_shifts)[:max(10, n_points//6)]
                
                # Add feasibility constraints
                for k in available_shifts:
                    # Create cross-spectrum approximation
                    corr_k = correlate(current_h, 1 - current_h, mode="full")
                    corr_k_idx = k + (n_points - 1)
                    if 0 <= corr_k_idx < len(corr_k):
                        expr = cp.sum(cp.multiply(h_var, cp.roll(1 - h_var, k)))
                        constraints.append(expr >= 1 - mid)
                
                # Add at least some raw constraints
                expr = cp.sum(cp.multiply(h_var, cp.roll(1 - h_var, max_k)))
                constraints.append(expr >= 1 - mid)
                
                # Solve even just a minor feasibility
                objective = cp.Minimize(cp.sum_squares(h_var))
                prob = cp.Problem(objective, constraints)
                try:
                    prob.solve(solver=cp.ECOS, verbose=False)
                    if prob.status in ["optimal", "optimal_inaccurate"]:
                        sol_h = np.maximum(0, np.minimum(1, h_var.value))
                        sol_C5 = compute_C5(sol_h)
                        if sol_C5 < best_C5_found:
                            best_C5_found = sol_C5
                            best_feasible_h = sol_h.copy()
                except:
                    pass
                
            except:
                pass
                
            # Important: always update best
            if best_C5_found < best_C5:
                best_C5 = best_C5_found
                best_h = best_feasible_h.copy()
            
            # Binary search update based on compute result
            if current_C5 <= mid:
                low = mid
            else:
                high = mid
                
            iteration += 1
            
        return best_C5, best_h

    # Run the binary search
    final_C5, final_h = search_for_C5()
    
    # Final refinement using trust region type local optimization
    if elapsed() < budget_s * 0.95:
        def get_gradient(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_vec) - 1)
            
            eps = 1e-3
            grad = np.zeros_like(h_vec)
            num_samples = 10
            
            for _ in range(num_samples):
                delta = np.random.normal(0, eps, len(h_vec))
                h_plus = h_vec + delta
                h_minus = h_vec - delta
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
                
            return grad / num_samples
            
        def trust_region_update(h_vec, eta=0.01, max_iter=20):
            h_new = h_vec.copy()
            for i in range(max_iter):
                grad = get_gradient(h_new)
                h_new = h_new - eta * grad
                h_new = project_feasible(h_new)
            return h_new
            
        current_h = final_h.copy()
        for i in range(50):
            if elapsed() > budget_s * 0.95:
                break
            current_h = trust_region_update(current_h)
            
        current_C5 = compute_C5(current_h)
        if current_C5 < final_C5:
            final_C5 = current_C5
            final_h = current_h.copy()
    
    return (final_h.tolist(), final_C5, len(final_h))
