import numpy as np
import time
import math
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    T_start = 0.1
    T = T_start

    scale_decay = 0.99
    
    t0 = time.time()
    max_time = budget_s - 2
    
    iterations = 0
    while time.time() - t0 < max_time:
        if iterations % cooling_interval == 0 and iterations > 0:
            T *= alpha
            scale *= scale_decay
            
        # Generate candidate
        delta = np.random.normal(0, scale, n_points)
        h_candidate = h_current + delta
        h_candidate = project(h_candidate, dx)
        
        # Evaluate
        current_C = evaluate(h_candidate, dx)
        
        # Accept or reject
        if current_C < best_C:
            best_C = current_C
            best_h = h_candidate.copy()

        else:
            # Accept with probability
            if np.random.rand() < math.exp(-(current_C - best_C) / T):
                h_current = h_candidate.copy()

        # Occasionally refine with gradient descent on the best solution
        if iterations % 1000 == 0:
            # Simple gradient step with finite differences
            lr = 0.001
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = project(h_plus, dx)
                
                h_minus = h_current.copy()
                h_minus[i] -= eps
                h_minus = project(h_minus, dx)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                
                grad = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
                
                h_current[i] = h_current[i] - lr * grad

                best_h = h_current.copy()

    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
