import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Start with clean gradient estimator
    def get_gradient(h_vec, num_samples=5):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Use finite differencing directly
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        
        # Sample small perturbations
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    # Trust region step
    def trust_region_update(h_vec, eta=0.01, steps=10):
        h_new = h_vec.copy()
        for _ in range(steps):
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Reinforcement-based turbulence to maintain exploration
    def apply_turbulence(h_vec, amplitude=0.1):
        # Apply sinusoidal turbulence with random phase and noise
        turbulence = amplitude * np.sin(np.pi * np.arange(len(h_vec)) / len(h_vec)) * np.random.randn(len(h_vec))
        h_new = h_vec + turbulence
        h_new = project_feasible(h_new)
        return h_new

    # A globally robust heuristic optimizer
    def robust_optimize():
        nonlocal h, best_C5, best_h
        h = project_feasible(h)
        
        # Local micro-search
        for _ in range(20):
            if elapsed() > budget_s * 0.95:
                break
            h = trust_region_update(h, 0.01, 10)
            h = project_feasible(h)
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        # Occasionally apply turbulence for escape
        if np.random.random() < 0.3:
            h = apply_turbulence(h, 0.05)
            
        return h

    # Main loop
    iter_count = 0
    last_improvement = 0
    
    while elapsed() < budget_s * 0.95:
        # Every 20 iterations, also run a thorough gradient descent update
        if iter_count % 20 == 0:
            h = robust_optimize()
        else:
            h = apply_turbulence(h, 0.05)
            h = project_feasible(h)
            
        # Evaluate
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count
            
        # Early termination if not improving recently
        if iter_count - last_improvement > 100:
            # Run one warm restart
            try:
                relaxed_h = solve_convex_relaxation()
                if compute_C5(relaxed_h) < compute_C5(h):
                    h = relaxed_h.copy()
                    last_improvement = iter_count
            except:
                pass
                
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))

def solve_convex_relaxation():
    n_points = len(best_h)  # Use global n_points
    h_var = cp.Variable(n_points)
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Try to spread amplitudes with a quadratic objective
    objective = cp.Minimize(cp.sum_squares(h_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        return np.maximum(0, np.minimum(1, h_var.value))
    except:
        # Fallback to simple linear spread
        return np.ones(n_points) * 0.5
