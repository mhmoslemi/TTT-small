import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated annealing with grid refinement
    T = 0.1
    alpha = 0.999
    max_iter = 0
    patience = 0
    max_patience = 500
    
    while elapsed() < budget_s * 0.95:
        max_iter += 1
        
        # Perturb one point
        idx = np.random.randint(0, n_points)
        h_new = h.copy()
        
        # Add perturbation
        pert = np.random.normal(0, 0.1)
        h_new[idx] += pert
        
        # Project to feasibility
        h_new = project_feasible(h_new)
        
        # Evaluate
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            patience = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_C5 - best_C5) / T):
                h = h_new
            patience += 1
        
        # Temperature decay
        T *= alpha
        
        # Grid refinement
        if max_iter % 1000 == 0 and n_points < 500:
            # Double the resolution
            old_h = h.copy()
            n_points = min(500, n_points * 2)
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(old_h)), old_h)
            h = project_feasible(h)
        
        # Early termination if no improvement for a long time
        if patience > max_patience:
            break
    
    # Final optimization step
    def optimize_local(h_vec, max_iter=50):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            # Gradient-like update using finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-4
            
            for i in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                max_k_idx = np.argmax(corr_plus)
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - 0.02 * grad
            h_new = project_feasible(h_new)

    # Refine final solution if time allows

        h = optimize_local(h, max_iter=100)
        final_C5 = compute_C5(h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
