        return np.max(corr) * dx

        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def local_optimize(h_vec, max_iter=30):
        h_new = h_vec.copy()
        for i in range(max_iter):
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)

                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
