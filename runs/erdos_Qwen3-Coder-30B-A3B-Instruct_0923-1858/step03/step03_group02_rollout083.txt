import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available; otherwise create a modified triangular function
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - 1.5 * np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = float('inf')
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_gradients(h_vec, k_max):
        grad = np.zeros_like(h_vec)
        # Compute gradient of overlap at shift k_max
        k = k_max
        for i in range(len(h_vec)):
            if 0 <= i + k < len(h_vec) and 0 <= i - k < len(h_vec):
                grad[i] = (1 - h_vec[i + k]) - h_vec[i - k]
            elif 0 <= i + k < len(h_vec):
                grad[i] = (1 - h_vec[i + k])
            elif 0 <= i - k < len(h_vec):
                grad[i] = -h_vec[i - k]
        return grad

    # Main optimization loop
    max_patience = 30
    patience = 0
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Add small random noise and reproject
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = project_feasible(h)
                patience = 0

        # Determine k_max that gives the maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        k_max = max_k_idx - (len(h) - 1)
        
        # Formulate and solve QP for this constraint
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0
        ]
        
        # Linearized constraint for the most constraining overlap
        grad = compute_gradients(h, k_max)
        # The linearized constraint is: t >= C_k + grad @ (h_new - h)
        # where C_k = sum of h[i] * (1 - h[i + k]) over valid indices
        # But we can calculate this directly using corr vector
        constraints.append(
            t >= corr[max_k_idx] * dx + grad @ (h_new - h)
        )
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new_vals = h_new.value
                if np.all(h_new_vals >= 0) and np.all(h_new_vals <= 1):
                    h = h_new_vals
        except:
            # fallback to simple gradient method if QP fails
            grad = compute_gradients(h, k_max)
            h = h - 0.01 * grad
            h = project_feasible(h)

        iter_count += 1

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)

    return (best_h.tolist(), final_C5, len(best_h))
