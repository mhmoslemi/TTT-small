import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 20
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 200
    vel_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_C5 = []
    
    # Initialize particles
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_p = h + np.random.normal(0, 0.02, n_points)
        else:
            h_p = np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_points) - 1)) + np.random.normal(0, 0.02, n_points)
        h_p = project_feasible(h_p)
        swarm.append(h_p.copy())
        velocities.append(np.random.uniform(-vel_max, vel_max, n_points))
        personal_best.append(h_p.copy())
        personal_best_C5.append(compute_C5(h_p))

    # Global best
    best_idx = np.argmin(personal_best_C5)
    global_best = personal_best[best_idx].copy()
    global_best_C5 = personal_best_C5[best_idx]

    # PSO iterations
    iter_count = 0
    while elapsed() < budget_s * 0.6 and iter_count < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            current_C5 = compute_C5(swarm[i])
            
            # Update personal best
            if current_C5 < personal_best_C5[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_C5[i] = current_C5
                
                # Update global best
                if current_C5 < global_best_C5:
                    global_best = swarm[i].copy()
                    global_best_C5 = current_C5

        iter_count += 1

    # L-BFGS-B refinement
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def jac(h_vec):
        # Approximate gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    # Refine with L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    refined_result = minimize(
        objective, 
        global_best, 
        method='L-BFGS-B', 
        jac=jac,
        bounds=bounds,
        options={'maxiter': 300, 'ftol': 1e-6, 'gtol': 1e-5}
    )
    
    if refined_result.success:
        final_h = refined_result.x.copy()
    else:
        final_h = global_best.copy()
    
    final_h = project_feasible(final_h)
    final_C5 = compute_C5(final_h)
    
    # Final comparison
    if final_C5 < global_best_C5:
        final_h = project_feasible(final_h)
        final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
