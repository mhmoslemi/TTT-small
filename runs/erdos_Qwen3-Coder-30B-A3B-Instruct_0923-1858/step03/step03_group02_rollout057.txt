import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a modified triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_overlap_for_shift(h_vec, shift):
        # Compute overlap for a given shift
        h_vec = np.array(h_vec)
        if shift >= 0:
            # h[i] * (1 - h[i + shift])
            valid_len = len(h_vec) - shift
            if valid_len <= 0:
                return 0
            return np.sum(h_vec[:valid_len] * (1 - h_vec[shift:shift+valid_len])) * dx
        else:
            # h[i + shift] * (1 - h[i])
            valid_len = len(h_vec) + shift
            if valid_len <= 0:
                return 0
            return np.sum(h_vec[-shift:] * (1 - h_vec[:-shift])) * dx

    def compute_all_overlaps(h_vec):
        overlaps = []
        h_vec = np.array(h_vec)
        for k in range(-n_points + 1, n_points):
            overlap = compute_overlap_for_shift(h_vec, k)
            overlaps.append(overlap)
        return np.array(overlaps)

    def compute_gradients(h_vec, shift):
        # Compute the gradient of overlap at shift k
        grad = np.zeros_like(h_vec)
        h_vec = np.array(h_vec)
        if shift >= 0:
            # For shift k: overlap = sum(h[i] * (1 - h[i+k]))
            valid_len = len(h_vec) - shift
            if valid_len <= 0:
                return grad
            # Gradient: d/dh[j] = (1-h[j+k]) if j+k < len and j < len
            for j in range(valid_len):
                if j + shift < len(h_vec):
                    grad[j] += (1.0 - h_vec[j + shift])
                if j < len(h_vec):
                    grad[j] -= h_vec[j]
            grad *= dx
        else:
            # For negative shift (-k): overlap = sum(h[j+k] * (1 - h[j]))
            valid_len = len(h_vec) + shift
            if valid_len <= 0:
                return grad
            # Gradient: d/dh[j] = (1-h[j]) * (j >= -k)
            for j in range(valid_len):
                if j < len(h_vec):
                    grad[j] += (1.0 - h_vec[j])
                if j - shift < len(h_vec) and j - shift >= 0:
                    grad[j] -= h_vec[j - shift]
            grad *= dx
        return grad

    # Main optimization loop using QP approach
    max_patience = 30
    patience = 0
    iter_count = 0
    eta = 0.01
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                noise = np.random.normal(0, 0.05, len(h))
                h = np.clip(h + noise, 0, 1)
                h = project_feasible(h)
                patience = 0

        # Compute the most constraining overlap
        overlaps = compute_all_overlaps(h)
        k_max = np.argmax(overlaps)
        shift = k_max - (n_points - 1)

        # Compute gradient for this shift
        grad = compute_gradients(h, shift)
        
        # Optimization using QP: minimize t such that t >= overlap for given k_max
        # Variables
        h_vars = cp.Variable(n_points)
        t_var = cp.Variable(1)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Affine approximation of overlap constraint
        # t >= overlap + gradᵀ(h - h_current)
        overlap_val = compute_overlap_for_shift(h, shift)
        approx_constraint = t_var >= overlap_val + cp.sum(grad * (h_vars - h))
        constraints.append(approx_constraint)
        
        # Objective
        objective = cp.Minimize(t_var)
        
        # Solve the QP
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_vars.value
                if h_new is not None:
                    h = h_new
                else:
                    # Fallback to gradient descent
                    h = h - eta * grad
                    h = project_feasible(h)
            else:
                h = h - eta * grad
                h = project_feasible(h)
        except:
            h = h - eta * grad
            h = project_feasible(h)

        # Periodic learning rate decay and check feasibility
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.95
            h = project_feasible(h)

        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Final projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
