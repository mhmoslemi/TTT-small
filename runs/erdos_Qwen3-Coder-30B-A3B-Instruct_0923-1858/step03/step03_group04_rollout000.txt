import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Two-phase optimization
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Phase 1: PSO-inspired global search
    if elapsed() < budget_s * 0.6:
        # PSO parameters
        n_particles = 20
        max_iter = 500
        inertia = 0.7
        cognitive = 1.5
        social = 1.5
        velocity_bounds = 0.1
        
        # Initialize particles
        particles = []
        velocities = []
        personal_best = []
        personal_best_vals = []
        
        for _ in range(n_particles):
            h_init = project_feasible(h + np.random.normal(0, 0.05, n_points))
            particles.append(h_init.copy())
            velocities.append(np.zeros(n_points))
            personal_best.append(h_init.copy())
            personal_best_vals.append(compute_C5(h_init))
        
        global_best = best_h.copy()
        global_best_val = best_C5
        
        for iteration in range(max_iter):
            if elapsed() >= budget_s * 0.6:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia * velocities[i] +

                velocities[i] = np.clip(velocities[i], -velocity_bounds, velocity_bounds)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Update personal best
                current_val = compute_C5(particles[i])
                if current_val < personal_best_vals[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_vals[i] = current_val
                    
                # Update global best
                if current_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = current_val

                best_C5 = global_best_val
                best_h = global_best.copy()

        # Create bounds for the constraints
        bounds = [(0, 1) for _ in range(n_points)]

            return compute_C5(h_vec)
        
        # Precompute for constraints
        def constraint_fn(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_fn}
        
        # Use global best as initial guess
        initial_guess = project_feasible(best_h)
        
        # L-BFGS-B optimization
        try:
            result = minimize(objective, initial_guess, method='L-BFGS-B', 
                            bounds=bounds, constraints=cons, 
                            options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 300})
            
            if result.success:
                refined_h = project_feasible(result.x)
                refined_C5 = compute_C5(refined_h)

                    best_C5 = refined_C5
                    best_h = refined_h.copy()
        except:
            pass

        def get_gradient(h_vec, eps=1e-5):

            base_val = compute_C5(h_vec)
            
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                h_minus = project_feasible(h_minus)
                
                grad[i] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
            
            return grad
        
        # Simple gradient descent with adaptive step
        h_current = best_h.copy()
        step_size = 0.001
        patience = 0
        max_patience = 100
        
        for iteration in range(1000):
            if elapsed() >= budget_s * 0.95:
                break
                
            grad = get_gradient(h_current)
            h_new = h_current - step_size * grad
            h_new = project_feasible(h_new)
            
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    step_size *= 0.9
                
            h_current = h_new
            
    return (best_h.tolist(), best_C5, len(best_h))
