import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simple convex relaxation approach
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    # Gradient estimation with finite differences
    def estimate_gradient(h_vec, num_samples=10):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    # Trust region step
    def trust_region_update(h_vec, eta=0.01, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = estimate_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Perturbation operator with sinusoidal turbulence
    def apply_turbulence(h_vec, amplitude=0.1, frequency=1.0):
        noise = amplitude * np.sin(np.pi * np.arange(n_points) * frequency / n_points) * np.random.normal(0, 1, n_points)
        h_new = h_vec + noise
        return project_feasible(h_new)

    # Try convex relaxation first to get a good warm start
    if elapsed() < budget_s * 0.4:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass

    # Main optimization loop
    iteration = 0
    last_improvement = 0
    patience = 0
    max_patience = 30
    eta = 0.01
    turbulence_amplitude = 0.1
    turbulence_frequency = 1.0

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_improvement = iteration
        else:
            patience += 1

        # Check if we're stuck and should perturb or restart
        if patience > max_patience or (iteration - last_improvement > 50):
            h = apply_turbulence(h, turbulence_amplitude, turbulence_frequency)
            patience = 0
            if iteration - last_improvement > 100:
                # More thorough strategy restart
                try:
                    relaxed_h = solve_convex_relaxation()
                    if compute_C5(relaxed_h) < compute_C5(h):
                        h = relaxed_h.copy()
                except:
                    pass

        # Apply turbulence for exploration
        if iteration % 5 == 0:
            h = apply_turbulence(h, turbulence_amplitude, turbulence_frequency)

        # Local optimization with trust region
        h = trust_region_update(h, eta)
        
        # Adaptive learning rate decay
        if iteration % 20 == 0 and iteration > 0:
            eta *= 0.98
        
        # Occasionally reduce turbulence
        if iteration % 100 == 0:
            turbulence_amplitude = max(0.005, turbulence_amplitude * 0.95)
        
        iteration += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
