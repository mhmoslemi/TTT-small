import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter = 1000
    inertia_w = 0.7
    cognitive_w = 1.5
    social_w = 1.5
    vel_max = 0.1
    
    # Initialize PSO swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Generate initial swarm
    for _ in range(swarm_size):
        # Random feasible solution
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        swarm.append(h_init.copy())
        velocities.append(np.random.uniform(-vel_max, vel_max, n_points))
        pbest.append(h_init.copy())
        pbest_vals.append(compute_C5(h_init))
    
    # Global best
    gb_idx = np.argmin(pbest_vals)
    gbest = pbest[gb_idx].copy()
    gbest_val = pbest_vals[gb_idx]
    
    # PSO iterations
    for iter_num in range(max_iter):
        if elapsed() > budget_s * 0.7:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           cognitive_w * r1 * (pbest[i] - swarm[i]) + 
                           social_w * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair infeasibility
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            current_val = compute_C5(swarm[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
                    
                    if gbest_val < best_C5:
                        best_C5 = gbest_val
                        best_h = gbest.copy()
    
    # Refinement with L-BFGS-B
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use best from PSO as starting point
    h_start = gbest.copy()
    
    # Refine with L-BFGS-B
    max_lbfgs_iter = 300
    try:
        result = minimize(objective, h_start, method='L-BFGS-B', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': max_lbfgs_iter, 'ftol': 1e-9, 'gtol': 1e-6})
        if result.success and compute_C5(result.x) < best_C5:
            best_h = result.x.copy()
            best_C5 = compute_C5(best_h)
    except:
        pass
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
