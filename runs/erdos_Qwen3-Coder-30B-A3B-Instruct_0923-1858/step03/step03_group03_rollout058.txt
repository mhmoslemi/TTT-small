import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_feasibility(h_init, C_temp):
        # Create variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        
        # For each possible shift k, we'll add a constraint
        # To make it computationally feasible, let's consider a subset of shifts
        num_shifts = min(20, n_points)
        shifts = np.linspace(0, n_points - 1, num_shifts, dtype=int)
        
        for k in shifts:
            # Approximate sum_{x} h[x]*(1-h[(x+k)%n])  
            # by linearizing around current h
            # Use a simple approximation: h[x]*(1 - h[(x+k)%n]) <= h[x] - h[x]*h[(x+k)%n]
            # But we linearize around current point
            if k == 0:
                expr = cp.sum(cp.multiply(h, 1 - h))
            else:
                h_shifted = cp.hstack([h[k:], h[:k]])
                expr = cp.sum(cp.multiply(h, 1 - h_shifted))
            constraints.append(expr <= C_temp / dx)
        
        # No objective - just feasibility check
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            pass
        
        return None

    def local_refine(h_init, max_iter=30):
        h_new = h_init.copy()
        for i in range(max_iter):
            current_C5 = compute_C5(h_new)
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute gradient of the maximum overlap term
            # For binary search approach, we're not doing full gradient
            # But we'll perform simple perturbations
            # Apply small random noise to possible directions
            noise_magnitude = 0.02
            h_plus = project_feasible(h_new + np.random.normal(0, noise_magnitude, n_points))
            h_minus = project_feasible(h_new - np.random.normal(0, noise_magnitude, n_points))
            
            if compute_C5(h_plus) < compute_C5(h_new):
                h_new = h_plus
            elif compute_C5(h_minus) < compute_C5(h_new):
                h_new = h_minus
            
        return h_new

    # Binary search on C5
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-4
    best_feasible = None
    best_feasible_C5 = float('inf')

    while C_high - C_low > tolerance and elapsed() < budget_s * 0.95:
        C_mid = (C_low + C_high) / 2.0

        # Try to find feasible h for this C_mid
        h_feasible = solve_convex_feasibility(h, C_mid)
        
        if h_feasible is not None:
            # Project to feasible region  
            h_feasible = project_feasible(h_feasible)
            c5_val = compute_C5(h_feasible)
            
            if c5_val < best_feasible_C5:
                best_feasible = h_feasible
                best_feasible_C5 = c5_val
                
            C_high = C_mid
        else:
            C_low = C_mid

    # Local search around best_feasible if found
    if best_feasible is not None:
        final_h = local_refine(best_feasible, max_iter=50)
        final_C5 = compute_C5(final_h)
        if final_C5 < best_feasible_C5:
            best_feasible_C5 = final_C5
            best_feasible = final_h.copy()


    # If we didn't find a good solution in the search, just do local refinement on the input
    if best_feasible_C5 == float('inf'):
        final_h = local_refine(h, max_iter=100)
        final_C5 = compute_C5(final_h)
        best_feasible_C5 = final_C5
        best_feasible = final_h.copy()

    return (best_feasible.tolist(), best_feasible_C5, len(best_feasible))
