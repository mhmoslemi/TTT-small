import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    population_size = 100
    generations = 1000
    crossover_prob = 0.8
    mutation_rate = 0.01

    population = []
    for _ in range(population_size):
        h = initial_h.copy()

    best_h = initial_h.copy()

        fitness = []

            fitness.append(1 / (C + 1e-12))  # Avoid division by zero

        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50%

        while len(new_population) < population_size:

        for h in population:
            C = evaluate(h, dx)
            if C < best_C:

def tournament_selection(population, fitness, k):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmax([fitness[i] for i in indices])]
    return population[best_idx]

def crossover(parent1, parent2):
    """Perform single-point crossover on binary vectors."""
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Ensure correct number of ones
    ones_count = n // 2
    child1 = enforce_ones_count(child1, ones_count)
    child2 = enforce_ones_count(child2, ones_count)
    
    return child1, child2

def mutate(h, mutation_rate):
    """Mutate a binary vector."""
    h = h.copy()
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            h[i] = 1 - h[i]  # Flip bit
    
    # Enforce correct number of ones
    ones_count = len(h) // 2
    h = enforce_ones_count(h, ones_count)
    
    return h

def enforce_ones_count(h, target_ones):
    """Ensure h has exactly target_ones ones."""
    current_ones = np.sum(h)
    if current_ones == target_ones:
        return h
    
    # Determine how many to flip
    diff = int(current_ones - target_ones)
    
    if diff > 0:  # Too many ones
        zeros_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(zeros_indices, size=abs(diff), replace=False)
        h[flip_indices] = 1
        ones_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(ones_indices, size=abs(diff), replace=False)
        h[flip_indices] = 0
    else:  # Not enough ones
        ones_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(ones_indices, size=abs(diff), replace=False)
        h[flip_indices] = 0
        zeros_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(zeros_indices, size=abs(diff), replace=False)
        h[flip_indices] = 1
        
    return h
