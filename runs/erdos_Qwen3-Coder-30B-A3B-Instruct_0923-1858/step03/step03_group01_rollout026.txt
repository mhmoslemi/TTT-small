import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    h_current = initial_h.copy()
    h_current = project(h_current, dx)
    
    t0 = time.time()
    best_C = evaluate(h_current, dx)
    best_h = h_current.copy()
    
    # Metaheuristic: Simulated Annealing with local search
    T0 = 1.0
    T_min = 1e-6
    alpha = 0.99
    max_iter = 50000
    
    T = T0
    current_h = h_current.copy()
    current_C = best_C
    
    for i in range(max_iter):
        if time.time() - t0 > budget_s - 1:

        # Make a neighbor solution
        neighbor_h = current_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        neighbor_h = project(neighbor_h, dx)
        
        # Evaluate neighbor
        neighbor_C = evaluate(neighbor_h, dx)
        
        # Accept or reject
        if neighbor_C < current_C or np.random.rand() < np.exp(-(neighbor_C - current_C) / T):
            current_h = neighbor_h
            current_C = neighbor_C
            
            if current_C < best_C:
                best_C = current_C
                best_h = current_h.copy()
        
        # Cooling schedule
        T = max(T_min, T * alpha)
    
    # Try a local optimization with gradient descent on the best solution
    local_max_iter = 5000
    for i in range(local_max_iter):
        if time.time() - t0 > budget_s - 1:
            break
            
        # Gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        lr = 0.05 * (1.0 - i / local_max_iter)
        current_h = current_h - lr * grad
        current_h = project(current_h, dx)
        
        # Check improvement
        current_C = evaluate(current_h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
    
    return (best_h, best_C, n_points)
