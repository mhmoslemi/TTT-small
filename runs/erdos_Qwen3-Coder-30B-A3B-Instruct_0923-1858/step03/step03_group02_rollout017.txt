import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        # Create symmetric initial solution: triangular with center peak
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Main optimization loop with adaptive perturbations and dual gradient descent
    iter_count = 0
    patience = 0
    max_patience = 50
    eta = 0.01
    current_h = h.copy()
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation focusing on the center of mass
                for i in range(n_points):
                    if np.random.random() < 0.05:
                        pert = np.random.normal(0, 0.03)
                        current_h[i] = np.clip(current_h[i] + pert, 0, 1)
                current_h = project_feasible(current_h)
                patience = 0

        # Compute gradients via finite differences and a simple gradient ascent
        def compute_overlap_gradients(h_vec, k_target):
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (corr_plus[k_target + n_points - 1] - corr_minus[k_target + n_points - 1]) / (2 * eps)
            return grad

        # Identify current maximum overlap k
        corr = correlate(current_h, 1 - current_h, mode="full")
        k_max = np.argmax(corr) - (n_points - 1)
        k_max = max(1, min(k_max, n_points - 1))  # clamp k to valid range

        # Compute gradient w.r.t. that specific overlap
        grad = compute_overlap_gradients(current_h, k_max)
        current_h = current_h - eta * grad
        current_h = project_feasible(current_h)

        # Iteratively improve using multiple steps
        for _ in range(3):
            corr = correlate(current_h, 1 - current_h, mode="full")
            k_max = np.argmax(corr) - (n_points - 1)
            k_max = max(1, min(k_max, n_points - 1))
            grad = compute_overlap_gradients(current_h, k_max)
            current_h = current_h - 0.9 * eta * grad
            current_h = project_feasible(current_h)

        # Adaptively adjust learning rate
        if iter_count % 30 == 0 and iter_count > 0:
            eta *= 0.95
            if eta < 1e-6:
                eta = 1e-6
                
        iter_count += 1

    if compute_C5(best_h) < best_C5:
        best_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), best_C5, len(best_h))
