import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from sobol_seq import i4_sobol_generate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 90
    dx = 2.0 / n_points
    
    # Initialize h using provided initial construction or Sobol sequence

        initial_h[perturb_mask] = np.random.rand(np.sum(perturb_mask)) * 0.5
    except NameError:
        # Use Sobol sequence for initialization
        t = np.linspace(0, 2, n_points)
        initial_support = np.zeros(n_points)
        sobol_samples = i4_sobol_generate(1, n_points // 2)
        support_indices = np.sort(np.random.choice(n_points, size=n_points // 2, replace=False))

        initial_h = initial_support * (1.0 / np.mean(initial_support))

    h_current = initial_h.copy()
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    lr = 0.01
    decay_factor = 0.98
    decay_interval = 100
    it = 0
    
    while time.time() - t0 < max_time:
        if it % decay_interval == 0:
            lr *= decay_factor
            
        # Estimate gradient by finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Gradient step
        h_new = h_current - lr * grad
        h_new = project(h_new, dx)
        
        current_C = evaluate(h_new, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_new.copy()
            
        h_current = h_new.copy()
        it += 1
        
        # Early stopping
        if it > 1000 and it % 100 == 0:
            if np.abs(best_C - evaluate(best_h, dx)) < 1e-4:
                break
                
    # Final projection step
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    # Return result
    return (final_h, final_C, n_points)
