import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        n_points = 45
        # Create a smooth initial solution
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # More robust approach: Sequential Quadratic Programming
    def optimize_step(h_vec):
        # Project current solution
        h_vec = project_feasible(h_vec)
        
        # Create bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective: minimize C5 using scipy.optimize
        def objective(h_vals):
            h_vals = np.array(h_vals)
            corr = correlate(h_vals, 1 - h_vals, mode="full")
            return -np.max(corr) * dx  # Negative because we minimize
        
        def constraint_func(h_vals):
            return np.sum(h_vals) - n_points / 2
        
        constraints = {'type': 'eq', 'fun': constraint_func}
        
        # Use SLSQP solver with bounds and constraints
        try:
            result = minimize(objective, h_vec, method='SLSQP', bounds=bounds, 
                              constraints=constraints, options={'maxiter': 200})
            if result.success:
                return project_feasible(result.x)
        except:
            pass
        return h_vec
    
    # Hybrid search with improved perturbations
    def perturb_solution(h_vec, mutation_strength=0.05):
        h_new = h_vec.copy()
        # Ensure we preserve the integral constraint
        indices = np.random.choice(len(h_vec), max(1, int(0.1 * len(h_vec))), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, mutation_strength)
        return project_feasible(h_new)
    
    # Main optimization loop
    iter_count = 0
    max_patience = 100
    patience = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Early stopping if no improvement for too long
        if patience > max_patience:
            h = perturb_solution(h, 0.05)
            patience = 0

        # Optimize step
        h = optimize_step(h)
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
