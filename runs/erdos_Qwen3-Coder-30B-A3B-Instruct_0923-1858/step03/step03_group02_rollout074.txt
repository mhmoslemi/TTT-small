import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_start = np.maximum(0, 1 - np.abs(x - 1))
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_clamp = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    # Create initial swarm
    for i in range(swarm_size):
        if i == 0:
            particle_h = h_start.copy()
        else:
            # Perturb with Gaussian noise
            noise = np.random.normal(0, 0.05, n_points)
            particle_h = h_start + noise
            particle_h = project_feasible(particle_h)
        
        swarm.append(particle_h)
        velocities.append(np.zeros(n_points))
        p_best.append(particle_h.copy())
        p_best_fitness.append(compute_C5(particle_h))
    
    # Find global best
    global_best_idx = np.argmin(p_best_fitness)
    s_best = p_best[global_best_idx].copy()
    s_best_fitness = p_best_fitness[global_best_idx]
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_num / max_iter)
        
        for i in range(swarm_size):
            particle_h = swarm[i]
            vel = velocities[i]
            particle_pbest = p_best[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (particle_pbest - particle_h)
            social = social_weight * r2 * (s_best - particle_h)
            new_vel = inertia * vel + cognitive + social
            
            # Clamp velocity
            new_vel = np.clip(new_vel, -velocity_clamp, velocity_clamp)
            
            # Update position
            new_h = particle_h + new_vel
            new_h = project_feasible(new_h)
            
            # Evaluate new particle
            new_fitness = compute_C5(new_h)
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = new_h.copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < s_best_fitness:
                    s_best = new_h.copy()
                    s_best_fitness = new_fitness
        
        # Update swarm positions
        for i in range(swarm_size):
            swarm[i] = project_feasible(swarm[i] + velocities[i])
            velocities[i] = new_vel
            
        # Periodic restart if no improvement
        if iter_num % 30 == 0 and iter_num > 0:
            if iter_num % 100 == 0:
                # Random perturbations
                for i in range(swarm_size):
                    if np.random.random() < 0.5:
                        noise = np.random.normal(0, 0.03, n_points)
                        swarm[i] += noise
                        swarm[i] = project_feasible(swarm[i])
            
    # Return final solution
    final_C5 = compute_C5(s_best)
    
    return (s_best.tolist(), final_C5, len(s_best))
