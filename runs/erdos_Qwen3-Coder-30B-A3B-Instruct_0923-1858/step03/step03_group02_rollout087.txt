import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simple gradient-based optimization with restarts
    def optimize_step(h_vec, max_iter=50):
        bounds = Bounds(0, 1, keep_feasible=True)
        
        def objective(h_new):
            # Softened objective for avoiding local minima
            return compute_C5(h_new)
        
        def constraint_sum(h_new):
            return np.sum(h_new) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            objective,
            h_vec,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-8},
            callback=lambda x: None
        )
        
        if result.success:
            return project_feasible(result.x)
        else:
            return project_feasible(h_vec)

    # Enhanced sampling and mollification scheme
    def mollify(h_vec, sigma=0.1):
        # Smooth the function using a Gaussian kernel
        h_smooth = np.copy(h_vec)
        for i in range(len(h_vec)):
            for j in range(len(h_vec)):
                diff = abs(i - j)
                weight = np.exp(-0.5 * (diff / (sigma * len(h_vec)))**2)
                h_smooth[i] += weight * (h_vec[j] - h_vec[i])
        h_smooth = np.clip(h_smooth, 0, 1)
        return project_feasible(h_smooth)
    
    # Main optimization loop with multiple strategies
    iter_count = 0
    patience = 0
    max_patience = 20
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0

                h = project_feasible(h)

        # Strategy 1: Standard optimization
        h_new = optimize_step(h, max_iter=30)
        
        # Strategy 2: Mollification + optimization
        h_mollified = mollify(h, sigma=0.05)
        h_mollified = optimize_step(h_mollified, max_iter=25)
        
        # Strategy 3: Random exploration
        if np.random.random() < 0.2:
            h_random = np.random.rand(n_points) * 0.5 + 0.25  # between 0.25 and 0.75
            h_random = project_feasible(h_random)
            if compute_C5(h_random) < compute_C5(h_new):
                h_new = h_random

        # Select best of the strategies
        candidates = [h_new, h_mollified]
        candidate_C5s = [compute_C5(c) for c in candidates]
        best_candidate_idx = np.argmin(candidate_C5s)
        h = candidates[best_candidate_idx]

        # Periodic reevaluation
        if iter_count % 30 == 0 and iter_count > 0:
            h = project_feasible(h)
        
        iter_count += 1

    # Final evaluation

        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Final check and projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
