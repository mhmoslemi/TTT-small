    best_h = h.copy()
    best_C5 = float('inf')

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Quasi-Newton method with adaptive learning rate
    def optimize_step(h_vec, learning_rate=0.01, max_iter=30):
        h_new = h_vec.copy()
        
        for _ in range(max_iter):
            # Compute current overlap
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_new) - 1)
            
            # Approximate gradient through finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-4
            for i in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update with gradient descent step
            h_new = h_new - learning_rate * grad
            h_new = project_feasible(h_new)
            
        return h_new

        h_new = h_vec.copy()

        best_C5 = final_C5
        best_h = final_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
