import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    # Gradient approximation with explicit shift selection
    def get_gradient(h_vec, num_samples=15):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    # Trust region update with adaptive learning
    def trust_region_update(h_vec, eta=0.01):
        h_new = h_vec.copy()
        for i in range(5):
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Large step perturbation
    def perturb_solution(h_vec, step_size=0.1):
        h_new = h_vec.copy()
        # Perturb a fraction of indices
        idxs = np.random.choice(n_points, size=max(1, int(0.2 * n_points)), replace=False)
        for i in idxs:
            if np.random.rand() < 0.7:
                h_new[i] = np.clip(h_new[i] + np.random.normal(0, step_size), 0, 1)
        return project_feasible(h_new)

    # Convex optimization approach
    def optimize_with_cvxpy():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        # Try a more structured approach
        objective = cp.Minimize(cp.sum_squares(h_var - 0.5))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    # Early exploration with convex constraints for upper bound
    if elapsed() < budget_s * 0.3:
        try:
            relaxed_h = optimize_with_cvxpy()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h
        except:
            pass

    # Main Loop: Trust Region with adaptive bounds
    eta = 0.01
    patience = 0
    max_patience = 20
    iter_count = 0
    last_best_update = 0
    adaptive_eta = 0.01

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        if patience > max_patience or (iter_count - last_best_update > 50):
            h = perturb_solution(h)
            patience = 0

        if iter_count % 5 == 0:
            h = trust_region_update(h, adaptive_eta)

        if iter_count % 10 == 0 and iter_count > 0:
            adaptive_eta *= 0.97

        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    # Ensure final feasibility and return
    return (best_h.tolist(), best_C5, len(best_h))
