import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Gradient-based optimization with adaptive learning rate and momentum
    def optimize_step(h_vec, learning_rate=0.05, momentum=0.9, epsilon=1e-8):
        h_vec = np.array(h_vec)
        
        # Find max overlap shift
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute gradient via finite differences for the max shift only
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        # Apply momentum and update
        if not hasattr(optimize_step, 'velocity'):
            optimize_step.velocity = np.zeros_like(h_vec)
        
        optimize_step.velocity = momentum * optimize_step.velocity - learning_rate * grad
        h_new = h_vec + optimize_step.velocity
        
        # Project back to feasible set
        h_new = project_feasible(h_new)
        return h_new

    # Perturbation strategy to escape local minima
    def perturb_solution(h_vec, strength=0.1):
        h_new = h_vec.copy()
        # Add Gaussian noise to a random subset of points
        num_perturb = max(1, int(0.15 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_perturb, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    # A variant with deterministic annealing approach
    def simulated_annealing(h_vec, temp=1.0):
        h_new = h_vec.copy()
        h_new = perturb_solution(h_new, strength=temp)
        return h_new

    # Main optimization loop
    iter_count = 0
    patience = 0
    max_patience = 50
    learning_rate = 0.05
    temperature = 1.0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Restart with perturbation or simulated annealing
                h = perturb_solution(h, 0.05)
                patience = 0

        # Optimization step
        if iter_count < 200:
            h = optimize_step(h, learning_rate)
        else:
            # Lower learning rate and increase perturbation
            h = optimize_step(h, learning_rate * 0.5)
            if iter_count % 10 == 0:
                temperature = max(0.01, temperature * 0.98)
                h = simulated_annealing(h, temperature)

        iter_count += 1
    
    # Final refinement
    refine_count = 0
    while elapsed() < budget_s * 0.99 and refine_count < 20:
        h_refined = optimize_step(h, 0.01)
        if compute_C5(h_refined) < compute_C5(h):
            h = h_refined
        refine_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
