import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization Phase
    def pso_optimization():
        swarm_size = 20
        n_dims = n_points
        max_iter = 1000
        
        # Initialize swarm
        swarm = []
        velocities = []
        personal_best = []
        personal_best_scores = []
        
        for _ in range(swarm_size):
            h_init = project_feasible(np.random.rand(n_dims) * 0.5 + 0.25)  # Random feasible solution
            swarm.append(h_init.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_dims))
            personal_best.append(h_init.copy())
            personal_best_scores.append(compute_C5(h_init))
        
        global_best_idx = np.argmin(personal_best_scores)
        global_best = personal_best[global_best_idx].copy()
        global_best_score = personal_best_scores[global_best_idx]
        
        w = 0.7  # inertia weight
        c1 = 1.5  # cognitive weight
        c2 = 1.5  # social weight
        
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.5:
                break
                
            for i in range(swarm_size):
                # Update velocity
                r1, r2 = np.random.rand(n_dims), np.random.rand(n_dims)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best[i] - swarm[i]) + 
                                c2 * r2 * (global_best - swarm[i]))
                
                # Limit velocity
                velocities[i] = np.clip(velocities[i], -0.1, 0.1)
                
                # Update position
                swarm[i] += velocities[i]
                
                # Repair infeasibility
                swarm[i] = project_feasible(swarm[i])
                
                # Evaluate
                score = compute_C5(swarm[i])
                
                # Update personal best
                if score < personal_best_scores[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_scores[i] = score
                    
                    # Update global best
                    if score < global_best_score:
                        global_best = swarm[i].copy()
                        global_best_score = score
        
        return global_best

    def lbfgsb_refinement(h_start):
        def objective(h_vec):
            return compute_C5(h_vec)
            
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        constraints = {'type': 'eq', 'fun': constraint_func}
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(objective, h_start, method='L-BFGS-B', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6})
            return result.x
        except:
            return h_start

    # Run PSO phase
    try:
        if elapsed() < budget_s * 0.5:
            pso_solution = pso_optimization()
            if compute_C5(pso_solution) < compute_C5(best_h):
                best_h = pso_solution.copy()
    except:
        pass

    # Refinement phase using L-BFGS-B
    if elapsed() < budget_s * 0.95:
        try:
            refined_solution = lbfgsb_refinement(best_h)
            refined_C5 = compute_C5(refined_solution)
            if refined_C5 < best_C5:
                best_h = refined_solution.copy()
                best_C5 = refined_C5
        except:
            pass

    # Final optimization with local search
    patience = 0
    max_patience = 100
    iter_count = 0
    last_best_update = 0
    
    h = best_h.copy()
    current_C5 = compute_C5(h)
    
    while elapsed() < budget_s * 0.95:
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        if patience > max_patience or (iter_count - last_best_update > 50):
            # Random restart to escape local minimum
            h = project_feasible(np.random.rand(n_points) * 0.5 + 0.25)
            patience = 0
            
        # Local perturbation with small random steps
        perturbation = np.random.normal(0, 0.02, n_points)
        h_new = h + perturbation
        h_new = project_feasible(h_new)
        
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            h = h_new.copy()
            current_C5 = new_C5
        
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
