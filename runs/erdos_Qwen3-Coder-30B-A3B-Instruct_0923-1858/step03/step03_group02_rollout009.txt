import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution with steeper decay
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # New optimization approach using QP with constraint linearization
    def solve_qp_with_constraint(h_vec, k_max):
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0
        ]
        
        # Linearize the most constraining constraint
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Forward difference approximation for gradient
            if i + k_max < n_points:
                grad[i] = (1 - h_vec[i + k_max]) 
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max]
        
        # Linearized constraint: t >= C_k_max + grad_k @ (h_new - h_vec)
        # C_k_max = sum(h[i] * (1 - h[i + k_max]) for valid i)
        c_val = 0.0
        for i in range(n_points):
            if i + k_max < n_points:
                c_val += h_vec[i] * (1 - h_vec[i + k_max])
        
        linear_constraint = t >= c_val + np.dot(grad, h_new - h_vec)
        constraints.append(linear_constraint)
        
        # Objective
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iter=100, feastol=1e-7, reltol=1e-7)
            if prob.status == cp.OPTIMAL:
                return h_new.value
        except:
            pass
        
        return h_vec

    # Main optimization loop
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                noise = np.random.normal(0, 0.05, n_points)
                h = np.clip(h + noise, 0, 1)
                h = project_feasible(h)
                patience = 0

        # Identify most constraining shift
        corr = correlate(h, 1 - h, mode="full")
        k_max_idx = np.argmax(corr)
        k_max = k_max_idx - (n_points - 1)
        
        # Solve QP to improve
        h_new = solve_qp_with_constraint(h, k_max)
        
        if h_new is not None:
            # Evaluate if this improves
            if compute_C5(h_new) < compute_C5(h):
                h = h_new
            else:
                h = project_feasible(h)
        else:
            h = project_feasible(h)
        
        # Periodic reevaluation
        if iter_count % 20 == 0 and iter_count > 0:
            h = project_feasible(h)
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
