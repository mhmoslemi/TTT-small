import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO-based optimization
    def pso_optimization():
        # PSO parameters
        swarm_size = 20
        max_iter = 500
        inertia = 0.7
        c1 = c2 = 1.5
        max_vel = 0.1
        
        # Initialize swarm
        swarm = []
        velocities = []
        personal_best = []
        personal_best_vals = []
        global_best = None
        global_best_val = float('inf')
        
        for _ in range(swarm_size):
            # Random feasible solution
            h_temp = np.random.rand(n_points)
            h_temp = project_feasible(h_temp)
            swarm.append(h_temp.copy())
            velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
            val = compute_C5(h_temp)
            personal_best.append(h_temp.copy())
            personal_best_vals.append(val)
            if val < global_best_val:
                global_best_val = val
                global_best = h_temp.copy()
        
        # PSO iterations
        for iter_count in range(max_iter):
            if elapsed() > budget_s * 0.8:
                break
                
            for i in range(swarm_size):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia * velocities[i] + 
                                c1 * r1 * (personal_best[i] - swarm[i]) + 
                                c2 * r2 * (global_best - swarm[i]))
                velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
                
                # Update position
                swarm[i] += velocities[i]
                swarm[i] = project_feasible(swarm[i])
                
                # Evaluate
                val = compute_C5(swarm[i])
                
                # Update personal best
                if val < personal_best_vals[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_vals[i] = val
                
                # Update global best
                if val < global_best_val:
                    global_best_val = val
                    global_best = swarm[i].copy()
        
        return global_best, global_best_val

    # Gradient-based refinement with L-BFGS
    def refine_with_lbfgs(h_init):
        def objective(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return -np.max(corr) * dx  # Negative because we minimize

        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2

        def constraint_bounds(i):
            def c(h_vec):
                return h_vec[i] - 1  # h[i] <= 1
            return c
        
        # Define constraints
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        for i in range(n_points):
            cons.append({'type': 'ineq', 'fun': lambda h_vec, i=i: 1 - h_vec[i]})
        
        # Optimize with L-BFGS-B
        try:
            res = minimize(
                objective,
                h_init,
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if res.success:
                return project_feasible(res.x)
        except:
            pass
        return h_init

    # Try PSO optimization first
    if elapsed() < budget_s * 0.7:
        try:
            pso_h, pso_C5 = pso_optimization()
            if pso_C5 < best_C5:
                best_C5 = pso_C5
                best_h = pso_h.copy()
        except:
            pass

    # Refine with L-BFGS if time allows
    if elapsed() < budget_s * 0.9:
        try:
            refined_h = refine_with_lbfgs(best_h)
            refined_C5 = compute_C5(refined_h)
            if refined_C5 < best_C5:
                best_C5 = refined_C5
                best_h = refined_h.copy()
        except:
            pass
            
    # Final checks and refinement
    while elapsed() < budget_s * 0.99:
        # Sometimes smaller perturbations work better
        h_new = best_h + np.random.normal(0, 0.01, n_points)
        h_new = project_feasible(h_new)
        new_C5 = compute_C5(h_new)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
        else:
            # Occasionally try bigger perturbations
            h_new = best_h + np.random.normal(0, 0.03, n_points)
            h_new = project_feasible(h_new)
            new_C5 = compute_C5(h_new)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_new.copy()

    return (best_h.tolist(), best_C5, len(best_h))
