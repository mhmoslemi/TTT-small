import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h_base)

        # Create base solution
        n_points = 45

        h_base = np.maximum(0, 1 - np.abs(x - 1))
        h_base = h_base * (n_points / 2) / np.sum(h_base)

    dx = 2.0 / n_points
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.9
    inertia_decay = 0.995
    cognitive = 2.0
    social = 2.0
    velocity_clamp = 0.1
    perturbation_scale = 0.03
    patience_threshold = 30

    # Initialize swarm
    particles = []
    velocities = []
    
    # Base particle (initial solution)
    particles.append(h_base.copy())
    velocities.append(np.zeros_like(h_base))
    
    # Perturbed particles
    for _ in range(swarm_size - 1):
        h_pert = h_base + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
        velocities.append(np.zeros_like(h_pert))
    
    # Track personal bests and global best
    p_best = [p.copy() for p in particles]
    p_best_fitness = [float('inf')] * swarm_size
    s_best = particles[0].copy()
    s_best_fitness = float('inf')
    
    # Initialize fitnesses
    for i in range(swarm_size):
        h = particles[i]
        corr = correlate(h, 1 - h, mode="full")
        fitness = np.max(corr) * dx
        p_best_fitness[i] = fitness
        if fitness < s_best_fitness:
            s_best = h.copy()
            s_best_fitness = fitness

    patience = 0
    iter_count = 0
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        inertia = max(0.4, inertia * inertia_decay)
        
        # Update each particle
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Generate random numbers for PSO update
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            
            # Update velocity
            cognitive_component = cognitive * r1 * (p_best[i] - h)
            social_component = social * r2 * (s_best - h)
            v = inertia * v + cognitive_component + social_component
            
            # Clamp velocity
            v = np.clip(v, -velocity_clamp, velocity_clamp)
            
            # Update position
            h_new = h + v
            
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new fitness
            corr = correlate(h_new, 1 - h_new, mode="full")
            new_fitness = np.max(corr) * dx
            
            # Update personal best
            if new_fitness < p_best_fitness[i]:
                p_best[i] = h_new.copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < s_best_fitness:
                    s_best = h_new.copy()
                    s_best_fitness = new_fitness
                    patience = 0
                else:
                    patience += 1
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
        
        # Check if we need to perturb particles
        if patience > patience_threshold:
            for i in range(swarm_size):
                particles[i] = np.clip(particles[i] + np.random.normal(0, perturbation_scale, n_points), 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            patience = 0
            
        iter_count += 1
        if iter_count % 50 == 0:

    # Final evaluation
    corr = correlate(s_best, 1 - s_best, mode="full")
    final_fitness = np.max(corr) * dx
    
    # Ensure final best is correct
    return (s_best.tolist(), final_fitness, len(s_best))
