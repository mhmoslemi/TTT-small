import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    # Simulated Annealing
    h_current = initial_h.copy()
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)

    T = 0.1

    max_time = budget_s - 2

        # Decrease temperature and scale
        if it % decay_interval == 0 and it > 0:

        # Generate candidate
        delta = np.random.normal(0, scale, n_points)
        h_candidate = h_current + delta
        h_candidate = project(h_candidate, dx)
        
        # Evaluate candidate
        current_C = evaluate(h_candidate, dx)
        
        # Accept or reject
        if current_C < best_C:

    h_refined = best_h.copy()

            break
        grad = np.zeros(n_points)

        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_plus_max = np.max(corr_plus)
            
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            corr_minus_max = np.max(corr_minus)

    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
