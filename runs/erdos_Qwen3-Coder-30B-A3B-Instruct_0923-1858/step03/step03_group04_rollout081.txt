import time

from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization
    def pso_optimization():
        # PSO parameters
        n_particles = 20
        max_iter = 200
        w = 0.7
        c1 = c2 = 1.5
        v_max = 0.1
        
        # Initialize swarm
        particles = []
        velocities = []
        pbest = []
        pbest_vals = []
        
        # Initialize with best known as one particle
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(compute_C5(h))
        
        # Generate others using noise
        for _ in range(n_particles - 1):
            h_new = h + np.random.normal(0, 0.05, n_points)
            h_new = project_feasible(h_new)
            particles.append(h_new.copy())
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            pbest.append(h_new.copy())
            pbest_vals.append(compute_C5(h_new))
        
        # Find global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        iter_count = 0
        
        while iter_count < max_iter and elapsed() < budget_s * 0.6:
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Clip velocities
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Evaluate
                new_val = compute_C5(particles[i])
                
                # Update personal best
                if new_val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = new_val
                    
                    # Update global best
                    if new_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = new_val
            
            iter_count += 1
            
        return gbest

    # Local refinement using L-BFGS-B
    def local_refinement(h_start):
        def obj_func(h_vec):
            return compute_C5(h_vec)
        
        # Create bounds
        bounds = Bounds([0.0] * n_points, [1.0] * n_points)
        
        # Create constraint
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Solve
        result = minimize(obj_func, h_start, method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 300, 'ftol': 1e-5, 'gtol': 1e-3})
        
        if result.success:
            h_refined = project_feasible(result.x)
            return h_refined
        else:
            return project_feasible(h_start)

    # Try PSO first
    h_pso = pso_optimization()
    
    # Refine with local optimizer
    h_final = local_refinement(h_pso)
    
    # Final evaluation
    final_C5 = compute_C5(h_final)
    
    return (h_final.tolist(), final_C5, len(h_final))
