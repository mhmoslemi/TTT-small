import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available, otherwise create a triangular solution
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # QP solving approach
    def solve_qp(h_vec, k_max):
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0
        ]
        
        # Linearized constraint for the most constraining k_max
        shift = k_max - (n_points - 1)
        # For each i in valid range
        grad = np.zeros(n_points)
        C_k = 0.0
        valid_idx = []
        for i in range(n_points):
            if 0 <= i + shift < n_points:
                grad[i] = (1 - h_vec[i + shift])
                C_k += h_vec[i] * (1 - h_vec[i + shift])
            else:
                grad[i] = 0
        constraints.append(t >= C_k + grad @ (h_new - h_vec))
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_new.value
        except:
            pass
        return h_vec

    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb solution
                perturb_mask = np.random.random(n_points) < 0.1
                h[perturb_mask] += np.random.normal(0, 0.05, np.sum(perturb_mask))
                h = project_feasible(h)
                patience = 0

        # Find maximum overlap shift
        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        k_max = max_k_idx - (len(h) - 1)
        
        # Solve QP to improve with linearization
        h_new = solve_qp(h, k_max)
        if h_new is not None:
            h = project_feasible(h_new)
        
        iter_count += 1
        
        # Occasionally reproject to maintain sum constraint
        if iter_count % 20 == 0:
            h = project_feasible(h)

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
