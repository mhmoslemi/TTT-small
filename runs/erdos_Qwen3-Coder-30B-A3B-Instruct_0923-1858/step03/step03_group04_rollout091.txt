import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial triangular solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Apply sinusoidal turbulence to maintain exploration
    def apply_turbulence(h_vec, amplitude=0.1):
        h_new = h_vec.copy()
        # Add sinusoidal noise with random phase
        phases = np.random.rand() * 2 * np.pi
        noise = amplitude * np.sin(np.linspace(0, 2*np.pi, n_points) + phases)
        h_new += noise
        h_new = project_feasible(h_new)
        return h_new

    # Compute gradient using finite differences
    def compute_gradient(h_vec, eps=1e-4):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        grad = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    # Gradient descent update
    def gradient_step(h_vec, lr=0.01):
        grad = compute_gradient(h_vec)
        h_new = h_vec - lr * grad
        return project_feasible(h_new)

    # Optimization loop
    iter_count = 0
    patience = 0
    max_patience = 30
    last_improvement = 0
    restart_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_improvement = iter_count
        else:
            patience += 1
            
        # Apply turbulence periodically or when stuck
        if iter_count % 2 == 0 or patience > 10:
            h = apply_turbulence(h, amplitude=0.05 + 0.02 * (patience / 10))
        else:
            # Do gradient descent
            h = gradient_step(h, lr=0.01)
            
        # Occasionally restart with convex relaxation
        if iter_count % 100 == 0 and elapsed() < budget_s * 0.8:
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    h_var >= 0,
                    h_var <= 1,
                    cp.sum(h_var) == n_points / 2
                ]
                # Minimize sum of squares for well spread solution
                objective = cp.Minimize(cp.sum_squares(h_var))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                h_new = np.maximum(0, np.minimum(1, h_var.value))
                if compute_C5(h_new) < compute_C5(h):
                    h = h_new.copy()
            except:
                pass

        # Break early if no improvement for too long
        if iter_count - last_improvement > 100:
            if patience > 50:
                break
                
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
