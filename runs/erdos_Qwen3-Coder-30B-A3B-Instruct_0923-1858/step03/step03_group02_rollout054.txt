import time
from scipy.signal import correlate

        # Create a piecewise linear function that is 0 at edges, peaks at center
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1.0))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_analytical_gradient(h_vec, k_max):
        n = len(h_vec)
        grad = np.zeros(n)
        # k_max is offset in full correlation, from 0 to 2*n-2
        # the maximum k index maps to k = k_max - (n-1) for actual lag
        # Let's compute for one specific k_max
        # The gradient with respect to h[i] is affected by terms where:
        # C_k = sum_j h[j] * (1 - h[j+k]) for fixed k and all valid j
        # The partial derivative wrt h[i] = (1-h[i+k]) + (1-h[i-k])
        # where the terms are valid (≠ -1, ≠ n-1)  
        for i in range(n):
            # contributes (1-h[i+k]) and -(h[i+k]) if valid
            # contributes -(h[i-k]) and (1-h[i-k]) if valid
            if i + k_max < n:
                grad[i] += 1 - h_vec[i + k_max]
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max]
        return grad

    def trust_region_update(h_vec, eta=0.01, max_iter=5):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Find the maximum overlap index
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            # Compute gradient
            grad = compute_analytical_gradient(h_new, max_k_idx)
            # Project gradient direction to feasible region
            h_candidate = h_new - eta * grad
            h_candidate = project_feasible(h_candidate)
            
            # Simple backtracking line search just to make sure we don't get worse
            while compute_C5(h_candidate) > compute_C5(h_new) and eta > 1e-6:
                eta *= 0.5
                h_candidate = h_new - eta * grad
                h_candidate = project_feasible(h_candidate)
            
            h_new = h_candidate
            
        return h_new

    # Main optimization loop with adaptive gradients
    eta = 0.01
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb with small random change
                perturbation = np.random.normal(0, 0.01, n_points)
                h += perturbation
                h = project_feasible(h)
                patience = 0

        # Perform update with analytical gradients
        h = trust_region_update(h, eta=eta)
        
        # Adjustment and projection every 20 iterations
        if iter_count % 20 == 0:
            eta *= 0.97
            h = project_feasible(h)
        
        iter_count += 1

    # Final refinement
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Ensure final solution is feasible
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
