import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with initial_h_values if available, otherwise create a triangular function
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 30
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        # Ensure h[i] is in [0,1] and sum to n_points / 2
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iter_count = 0
    patience = 0
    max_patience = 100

    # Refinement control
    refine_interval = 1000
    refine_factor = 2

    while elapsed() < budget_s * 0.95 and T > min_T:
        # Perturb a random element
        i = np.random.randint(n_points)
        old_h = h[i]
        delta = np.random.uniform(-0.1, 0.1)
        h[i] = old_h + delta
        
        # Project to feasible set
        h = project_feasible(h)
        
        # Compute new C5
        new_C5 = compute_C5(h)
        
        # Accept or reject
        delta_C5 = new_C5 - compute_C5(best_h if iter_count == 0 else best_h)
        
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            h[i] = old_h  # revert change

        # Temperature update
        T *= alpha
        iter_count += 1
        
        # Grid refinement
        if iter_count % refine_interval == 0 and n_points < 1000:
            # Interpolate to higher resolution
            n_points *= refine_factor
            dx = 2.0 / n_points
            h_new = np.zeros(n_points)
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)
            for i in range(n_points):
                if x_new[i] <= x_old[0] or x_new[i] >= x_old[-1]:
                    h_new[i] = 0
                else:
                    # Simple interpolation
                    idx = np.searchsorted(x_old, x_new[i])
                    if idx == 0:
                        h_new[i] = best_h[0]
                    elif idx >= len(x_old):
                        h_new[i] = best_h[-1]
                    else:
                        frac = (x_new[i] - x_old[idx-1]) / (x_old[idx] - x_old[idx-1])
                        h_new[i] = best_h[idx-1] * (1-frac) + best_h[idx] * frac
            h = h_new
            # Normalize to sum to n_points / 2
            h = h * (n_points / 2) / np.sum(h)

        # Check for stagnation
        if patience > max_patience:
            T *= 1.1  # Increase temperature to escape local minima
            patience = 0

    # Final evaluation of best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
