import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Frequency domain optimization approach
    def optimize_frequency_domain(max_iter=100):
        h_current = h.copy()
        
        # Initialize frequency domain representation
        H = fft(h_current)
        
        # Learning rate and smoothing
        lr = 0.01
        gamma = 0.9
        prev_grad = None
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Transform h_current to frequency domain
            H = fft(h_current)
            
            # Compute overlap in frequency domain
            # Overlap integral is conv(h, 1-h)
            # In frequency domain: fft(h) * conj(fft(1-h)) = H * conj(-H + 1)
            # Which is H * conj(H) - H * conj(1) = |H|^2 - H * conj(1)
            # But we're actually computing: fft(h) * conj(fft(1-h)) = H * conj(1 - H)
            # Which expands to: H * conj(1) - H * conj(H) = conj(1)*H - |H|^2
            
            H_conj = np.conj(H)
            H_conj_1 = np.conj(1 - H)
            overlap_fft = H * H_conj_1  # This is going to produce complex thus need magnitude
            
            # Compute the real part of the overlap (magnitude spectrum)
            overlap_spectrum = np.abs(overlap_fft)
            
            # Optimizer: try to minimize the real part
            grad = -np.real(overlap_spectrum) # Approximate gradient is -real part! 
            
            # Apply smoothing if previous gradient exists
            if prev_grad is not None:
                grad = gamma * grad + (1 - gamma) * prev_grad
                
            # Update in frequency domain (use gradient ascent on |H|^2 part)
            H_new = H + lr * grad
            # Project to stable region in frequency domain?
            # Just project back to real numbers in time domain for now
            
            # Transform back to time domain
            h_new = np.real(ifft(H_new))
            
            # Project to feasible domain
            h_new = project_feasible(h_new)
            
            # Update for next iteration
            h_current = h_new
            prev_grad = grad
            
            current_C5 = compute_C5(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
        
        return h_current

    # Start with convex relaxation for better warm start
    def solve_convex_relaxation():
        # Heuristic: try to minimize sum of squares for spread out solution
        from scipy.optimize import minimize
        def obj(h): return np.sum(h**2)
        def cons(h): return np.sum(h) - n_points / 2
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons})
        if res.success:
            return project_feasible(res.x)
        return h.copy()
    
    # Try convex relaxation to warm start
    if elapsed() < budget_s * 0.3:
        relaxed_h = solve_convex_relaxation()
        if compute_C5(relaxed_h) < compute_C5(h):
            h = relaxed_h.copy()

    try:
        h_freq = optimize_frequency_domain(50)
        if compute_C5(h_freq) < compute_C5(h):
            h = h_freq.copy()
    except Exception as e:
        pass

    # Run gradient-based refinement if any time left
    if elapsed() < budget_s * 0.8:
        last_C5 = float('inf')
        for it in range(100):
            if elapsed() > budget_s * 0.95:
                break
                
            h_old = h.copy()
            
            # Estimate gradient of max correlation with finite differences
            corr_val = correlate(h, 1 - h, mode="full")
            max_k_idx = np.argmax(corr_val)
            max_k = max_k_idx - (len(h) - 1)
            
            # Compute finite difference gradient using small perturbations
            eps = 1e-4
            gradients = np.zeros_like(h)

            for i in range(len(h)):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")[max_k_idx]
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")[max_k_idx]
                
                gradients[i] = (corr_plus - corr_minus) / (2 * eps)

            # Steepest descent step
            h = h - 0.005 * gradients
            h = project_feasible(h)
            
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
            elif abs(current_C5 - last_C5) < 1e-8:
                break
            last_C5 = current_C5

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
