import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def initial_population(n_points, pop_size):
    """Generate initial population of binary vectors."""
    population = []
    for _ in range(pop_size):
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h[ones_indices] = 1
        population.append(h)
    return population

def fitness(h, dx):
    """Calculate fitness as inverse of C5 value."""
    c5 = evaluate(h, dx)
    return 1.0 / (c5 + 1e-12)

def select(population, fitnesses, num_selected):
    """Tournament selection."""
    selected = []
    for _ in range(num_selected):
        tournament_indices = np.random.choice(len(population), size=5)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    """Single-point crossover with constraint preservation."""
    point = np.random.randint(1, len(parent1)-1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Ensure correct number of ones
    ones1 = np.sum(child1)
    ones2 = np.sum(child2)
    desired = len(parent1) // 2
    if ones1 != desired:
        # Fix child1
        zeros_indices1 = np.where(child1 == 0)[0]
        ones_indices1 = np.where(child1 == 1)[0]
        if ones1 > desired:
            to_zero = ones1 - desired
            flip_indices = np.random.choice(ones_indices1, to_zero, replace=False)
            child1[flip_indices] = 0
        else:
            to_one = desired - ones1
            flip_indices = np.random.choice(zeros_indices1, to_one, replace=False)
            child1[flip_indices] = 1
    
    if ones2 != desired:
        # Fix child2
        zeros_indices2 = np.where(child2 == 0)[0]
        ones_indices2 = np.where(child2 == 1)[0]
        if ones2 > desired:
            to_zero = ones2 - desired
            flip_indices = np.random.choice(ones_indices2, to_zero, replace=False)
            child2[flip_indices] = 0
        else:
            to_one = desired - ones2
            flip_indices = np.random.choice(zeros_indices2, to_one, replace=False)
            child2[flip_indices] = 1
    
    return child1, child2

def mutate(h, mutation_rate):
    """Flip bits with given rate and preserve constraint."""
    h = h.copy()
    num_mutations = int(len(h) * mutation_rate)
    indices = np.random.choice(len(h), num_mutations, replace=False)
    for i in indices:
        h[i] = 1 - h[i]
    
    # Ensure correct number of ones
    ones = np.sum(h)
    desired = len(h) // 2
    if ones != desired:
        zeros_indices = np.where(h == 0)[0]
        ones_indices = np.where(h == 1)[0]
        if ones > desired:
            to_zero = ones - desired
            flip_indices = np.random.choice(ones_indices, to_zero, replace=False)
            h[flip_indices] = 0
        else:
            to_one = desired - ones
            flip_indices = np.random.choice(zeros_indices, to_one, replace=False)
            h[flip_indices] = 1
    
    return h
