import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        # Compute all correlations to identify the critical lag
        corr_full = correlate(h_current, 1 - h_current, mode='full')
        # Get correlation for each lag k (positive and negative)
        corr_lags = []
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                corr_lags.append(corr_full[n_points - 1 + k])
            else:
                corr_lags.append(corr_full[n_points - 1 + k])
        
        # Find the lag with minimum correlation
        min_idx = np.argmin(corr_lags)
        min_lag = min_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h for this lag
        grad = np.zeros_like(h_current)
        for j in range(n_points):
            if min_lag >= 0:
                if j + min_lag < n_points:
                    grad[j] += 1.0  # Differentiation of h[i] * h[i+lag]
            else:
                if j - min_lag < n_points:
                    grad[j] += 1.0  # Differentiation of h[i] * h[i+lag] where lag < 0
        
        # Update step
        lr = 0.005 * (1.0 - i / max_iter)  # Decaying learning rate
        h_new = h_current - lr * grad
        h_new = project(h_new, dx)

        current_C = evaluate(h_new, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_new.copy()
            
        h_current = h_new
        
    return (best_h, best_C, n_points)
