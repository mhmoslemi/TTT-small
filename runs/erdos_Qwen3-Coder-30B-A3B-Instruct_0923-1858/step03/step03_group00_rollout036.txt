import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def crossover(parent1, parent2):
    """Perform one-point crossover between two parents."""
    n = len(parent1)
    if n < 2:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate, mutation_scale):
    """Mutate an individual with Gaussian noise."""
    n = len(individual)
    mask = np.random.random(n) < mutation_rate
    if not np.any(mask):
        # Ensure at least one mutation
        mask[np.random.randint(0, n)] = True
    individual = individual.copy()
    individual[mask] += np.random.normal(0, mutation_scale, np.sum(mask))
    individual = np.clip(individual, 0, 1)
    return individual

    dx = 2.0 / n_points

        initial_h = np.clip(initial_h, 0, 1)

    # Genetic algorithm params
    mutation_rate = 0.05
    max_generations = 5000
    elitism_count = 2
    tournament_size = 5
    best_C = float('inf')
    best_h = None
    
    t0 = time.time()
    max_time = budget_s - 2  # Leave some buffer
    
    for generation in range(max_generations):
        if time.time() - t0 > max_time:
            break

        # Evaluate fitness
        fitness_scores = []
        for h in population:
            C = evaluate(h, dx)
            fitness_scores.append(C)
            if C < best_C:
                best_C = C
                best_h = h.copy()
                
        # Sort by fitness (ascending order)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Elitism: keep top individuals
        new_population = [population[i].copy() for i in range(elitism_count)]
        
        # Generate new population
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.choice(range(elitism_count, len(population)),

    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
