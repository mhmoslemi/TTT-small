import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_individual(n_points, initial_h=None):
    if initial_h is not None:
        h = initial_h.copy()
        h = project(h, 2.0 / n_points)
        return h
    else:
        t = np.linspace(0, 2, n_points)
        h = 0.3 + 0.2 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        h = project(h, 2.0 / n_points)
        return h

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = n // 2
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate, mutation_scale):
    n = len(h)
    mask = np.random.random(n) < mutation_rate
    h = h + np.random.normal(0, mutation_scale, n) * mask
    h = np.clip(h, 0, 1)
    return h

    pop_size = 20

    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2
    
    for generation in range(generations):

        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]

        # Elitism
        new_population = population[:elite_size]
        
        # Tournament selection
        def tournament_selection(k=3):
            idxs = np.random.choice(pop_size, k)
            return population[np.argmin([fitness[i] for i in idxs])]
            
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = tournament_selection(), tournament_selection()
            
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
                child1 = project(child1, dx)
                child2 = project(child2, dx)
                new_population.extend([child1, child2])
            else:
                new_population.extend([parent1, parent2])
                
        # Trim population to exact size
        population = new_population[:pop_size]
        
        # Mutation
        mutation_scale = 0.05 * (1 - generation / generations)
        for i in range(elite_size, pop_size):
            population[i] = mutate(population[i], mutation_rate, mutation_scale)
            
        # Evaluate new population
        fitness = [evaluate(ind, dx) for ind in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()
            
    # Final projection to ensure constraints
    best_individual = project(best_individual, dx)
    final_fitness = evaluate(best_individual, dx)
    
    return best_individual, final_fitness, n_points

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
