    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Quasi-Newton method with trust region
    def optimize_step(h_vec, learning_rate=0.01, trust_radius=0.1):
        h_vec = np.array(h_vec)
        # Compute correlation and find max shift
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        # Quasi-Newton update
        h_new = h_vec - learning_rate * grad
        h_new = project_feasible(h_new)
        
        # Trust region
        diff = h_new - h_vec
        if np.linalg.norm(diff) > trust_radius:
            h_new = h_vec + trust_radius * diff / np.linalg.norm(diff)
            h_new = project_feasible(h_new)
            
        return h_new

    # Final evaluation

        best_h = best_h.copy()
