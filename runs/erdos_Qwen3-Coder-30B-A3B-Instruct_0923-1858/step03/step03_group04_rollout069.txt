import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with pre-existing construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def apply_turbulence(h_vec, amp=0.1, freq=1):
        """Apply sinusoidal turbulence to solution."""
        turbulence = amp * np.sin(np.pi * np.arange(n_points) * freq / n_points) * np.random.normal(0, 1, n_points)
        h_turb = h_vec + turbulence
        return project_feasible(h_turb)

    def evaluate_and_update(h_vec, update_best=True):
        """Evaluate and update best solution if improved."""
        C5_val = compute_C5(h_vec)
        if C5_val < best_C5:
            if update_best:
                best_C5 = C5_val
                best_h[:] = h_vec
        return C5_val

    # Main optimization loop
    iter_count = 0
    turbulence_freq = 50
    patience = 0
    max_patience = 100
    last_improvement = 0

    while elapsed() < budget_s * 0.95:
        # Apply turbulence
        h_turb = apply_turbulence(h, amp=0.1, freq=1)
        
        # Evaluate and update
        C5_turb = evaluate_and_update(h_turb)
        
        if C5_turb < compute_C5(h):
            h = h_turb.copy()
            last_improvement = iter_count
            patience = 0
        else:
            patience += 1

        # Gradient-like updates every few iterations
        if iter_count % 20 == 0 and patience < max_patience and elapsed() < budget_s * 0.9:
            # Simple gradient ascent direction via random perturbations
            noisy_h = h + np.random.normal(0, 0.01, n_points)
            noisy_h = project_feasible(noisy_h)
            if compute_C5(noisy_h) < compute_C5(h):
                h = noisy_h.copy()
            # Occasionally explore with full perturbation
            if iter_count % 100 == 0:
                try:
                    h = apply_turbulence(h, amp=0.2, freq=2)
                except:
                    pass

        # Regularly check time and possibly restart if no progress
        if patience > max_patience or (iter_count - last_improvement > 100):
            # Restart with slightly different turbulence
            h = apply_turbulence(h, amp=0.05, freq=0.5)
            patience = 0
            last_improvement = iter_count

        if iter_count % 100 == 0:
            # Slow periodic stabilizing step
            h = project_feasible(h)

        iter_count += 1

    # Final evaluation to make sure best solution is set
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
