    best_h = h.copy()
    best_C5 = float('inf')

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def trust_region_update(h_vec, max_iter=50, trust_radius=0.1):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Evaluate at current point
            corr_current = correlate(h_new, 1 - h_new, mode="full")
            current_max_k_idx = np.argmax(corr_current)
            current_max_k = current_max_k_idx - (len(h_new) - 1)
            
            # Compute gradient numerically
            grad = np.zeros_like(h_new)
            eps = 1e-4
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (corr_plus[current_max_k_idx] - corr_minus[current_max_k_idx]) / (2 * eps)
            
            # Apply trust region update
            update = -grad
            update_norm = np.linalg.norm(update)
            if update_norm > trust_radius:
                update = update * trust_radius / update_norm
            
            h_new = h_new + update
            h_new = project_feasible(h_new)
        return h_new
