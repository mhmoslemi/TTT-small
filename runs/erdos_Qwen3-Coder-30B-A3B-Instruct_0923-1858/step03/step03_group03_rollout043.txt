import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Start with 30 points on a coarse grid
        n_points = 30
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iteration = 0
    patience = 0
    max_patience = 500
    
    # Refinement parameters
    refine_interval = 1000
    refine_factor = 2
    
    # For refinement
    last_refine_iter = 0
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        # Perturb one element
        new_h = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        new_h[idx] += delta
        
        # Ensure feasibility
        new_h = project_feasible(new_h)
        
        # Compute change in C5
        current_C5 = compute_C5(h)
        new_C5 = compute_C5(new_h)
        delta_C5 = new_C5 - current_C5
        
        # Accept or reject
        if delta_C5 < 0 or np.random.random() < np.exp(-delta_C5 / T):
            h = new_h
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Cooling and refinement
        T *= alpha
        iteration += 1
        
        # Check for refinement
        if iteration - last_refine_iter > refine_interval:
            if n_points < 500:  # Avoid too many points
                # Double the grid
                old_h = h.copy()
                old_n = n_points
                n_points *= refine_factor
                new_x = np.linspace(0, 2, n_points)
                old_x = np.linspace(0, 2, old_n)
                h = np.interp(new_x, old_x, old_h)
                # Normalize again
                h = h * (n_points / 2) / np.sum(h)
                # Update dx
                dx = 2.0 / n_points
                last_refine_iter = iteration
                # Reset temperature and patience
                T = 0.1
                patience = 0
    
    # Final refinement
    current_C5 = compute_C5(h)
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
