import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

            winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
            selected.append(population[winner_idx])
        return selected

        new_h1 = np.concatenate([h1[:point], h2[point:]])
        new_h2 = np.concatenate([h2[:point], h1[point:]])
