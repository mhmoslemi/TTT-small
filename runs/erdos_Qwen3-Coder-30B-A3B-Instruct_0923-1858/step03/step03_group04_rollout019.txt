import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Enhanced heuristic solving using convex relaxation
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Minimize a quadratic form to spread the solution more
        # This tries to avoid overly concentrated solutions
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    # Improved gradient estimation with better finite differences and sampling
    def get_gradient(h_vec, num_samples=15):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute gradient using two-sided finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    # Trust region update with proper curvature estimation
    def trust_region_update(h_vec, eta=0.01, max_iter=15):
        h_new = h_vec.copy()
        try:
            for _ in range(max_iter):
                grad = get_gradient(h_new)
                h_new = h_new - eta * grad
                h_new = project_feasible(h_new)
                # Small learning rate decay for stability
                eta = max(1e-4, eta * 0.97)
        except:
            pass
        return h_new

    # Enhanced perturbation
    def perturb_solution(h_vec, prob=0.2, magnitude=0.06):
        h_new = h_vec.copy()
        idxs = np.random.choice(n_points, size=int(prob * n_points), replace=False)
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, magnitude), 0, 1)
        return project_feasible(h_new)

    # Phase 1: Use convex relaxation to get a better starting point
    if elapsed() < budget_s * 0.3:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass

    # Phase 2: Main optimization loop with adaptive strategies
    eta = 0.01
    patience = 0
    max_patience = 25
    iter_count = 0
    last_best_update = 0
    turbulence_freq = 35  # Every this many iterations, do turbulence
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Apply turbulence periodically
        if iter_count % turbulence_freq == 0 and iter_count > 0:
            h = perturb_solution(h, prob=0.22, magnitude=0.055)

        # Occasionally run trust region optimization
        if iter_count % 20 == 0:
            h = trust_region_update(h, eta=eta)
        
        # Healing: if stuck, apply more aggressive perturbation
        if patience > max_patience or (iter_count - last_best_update > 60):
            h = perturb_solution(h, prob=0.3, magnitude=0.08)
            patience = 0
            if iter_count - last_best_update > 120:
                # Full restart
                try:
                    relaxed_h = solve_convex_relaxation()
                    if compute_C5(relaxed_h) < compute_C5(h):
                        h = relaxed_h.copy()
                except:
                    pass
        
        # Adaptive learning rate decay
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.93
        
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
