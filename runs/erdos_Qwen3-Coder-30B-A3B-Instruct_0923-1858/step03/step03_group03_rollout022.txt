import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with provided construction or create a simple one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a smooth initial solution
        n_points = 40
        x = np.linspace(0, 2, n_points)
        h = np.exp(-10 * (x - 1)**2)  # Gaussian bump centered at 1
        h = h / np.sum(h) * (n_points / 2)  # Normalize to sum to n_points / 2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Optimizer using scipy.optimize with constraints
    def optimize_step(h_vec, max_iter=20):
        def objective(h_flat):
            h = h_flat.reshape(-1)
            h = project_feasible(h)
            return compute_C5(h)
        
        def constraint_sum(h_flat):
            h = h_flat.reshape(-1)
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(len(h_vec))]
        res = minimize(objective, h_vec, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': max_iter, 'ftol': 1e-8}, tol=1e-8)
        return project_feasible(res.x).reshape(-1) if res.success else h_vec

    # Simple perturbation for exploration
    def perturb(h_vec, strength=0.1):
        h_new = h_vec.copy()
        # Randomly modify a few points
        indices = np.random.choice(len(h_vec), max(1, len(h_vec) // 10), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    # Keep track of best solution
    convergence_patience = 0
    max_patience = 100
    iter_count = 0

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            convergence_patience = 0
        else:
            convergence_patience += 1
            if convergence_patience > max_patience:
                h = perturb(h, 0.03)
                convergence_patience = 0

        # Optimization step
        h = optimize_step(h, max_iter=30 if iter_count % 10 == 0 else 10)
        
        # Occasionally perturb to avoid local minimum
        if iter_count % 15 == 0:
            h = perturb(h, 0.03)
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
