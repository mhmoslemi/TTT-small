import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        h_current = project(h_current, dx)
    
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # Adaptive targeting with QP
    it = 0

        # Compute overlap for all lags
        corr = correlate(h_current, 1 - h_current, mode='full')

        lag = max_idx - (n_points - 1)

        h = cp.Variable(n_points)
        # Minimize the overlap at the identified lag
        obj = cp.Minimize(cp.sum(cp.multiply(h, 1 - h[lag:]) if lag >= 0 else cp.multiply(h[-lag:], 1 - h[:n_points+lag])))
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == 1.0 / dx
        ]
        
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status == 'optimal':
                h_new = h.value
                h_new = project(h_new, dx)
                current_C = evaluate(h_new, dx)

                    best_C = current_C
                    best_h = h_new.copy()
                h_current = h_new.copy()

        if it % 10 == 0:
            # Random perturbation to avoid local minima
            perturb = np.random.normal(0, 0.01, n_points)
            h_perturb = h_current + perturb
            h_perturb = project(h_perturb, dx)
            current_C = evaluate(h_perturb, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h_perturb.copy()
                h_current = h_perturb.copy()
    
    # Final projection and evaluation
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
