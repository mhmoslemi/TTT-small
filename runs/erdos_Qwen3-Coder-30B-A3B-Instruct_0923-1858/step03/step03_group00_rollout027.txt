import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    np.random.seed(seed)

        h_current = 0.3 + 0.25 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t) + 0.1 * np.sin(5 * np.pi * t)
        h_current = np.clip(h_current, 0, 1)
        # Adjust to sum to 45
        h_current = h_current * (45.0 / np.sum(h_current))
    
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    lr = 0.01
    decay_interval = 1000
    decay_factor = 0.99
    
    it = 0
    while time.time() - t0 < max_time:
        if it % decay_interval == 0 and it > 0:
            lr *= decay_factor
            
        # Estimate gradient with finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update and project
        h_new = h_current - lr * grad
        h_new = project(h_new, dx)
        
        current_C = evaluate(h_new, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_new.copy()
            
        h_current = h_new.copy()
        it += 1
        
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
