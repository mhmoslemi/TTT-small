import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Frequency domain optimization approach
    def frequency_domain_optimization():
        h_vec = h.copy()
        H = np.fft.fft(h_vec)
        max_iter = 1000
        learning_rate = 0.01
        decay = 0.999
        patience = 0
        max_patience = 50
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute the correlation in frequency domain
            H_conj = np.conj(H)
            correlation_freq = H * H_conj
            correlation_time = np.fft.ifft(correlation_freq).real
            
            # Find the index of maximum overlap
            max_idx = np.argmax(correlation_time)
            max_val = correlation_time[max_idx]
            
            # Compute gradient in frequency domain
            # ∇_H |IFFT(H * conj(H))|_max
            grad_freq = np.zeros_like(H)
            grad_freq[max_idx] = 1.0
            
            # Inverse transform and compute gradient 
            grad_time = np.fft.ifft(grad_freq).real
            # Approximate gradient of the maximum in time domain
            grad_h = np.zeros_like(h_vec)
            for k in range(len(h_vec)):
                grad_h[k] = grad_time[k] + grad_time[k - len(h_vec)]

            # Update in frequency domain
            H = H - learning_rate * grad_h
            H = H * (n_points / 2) / np.sum(np.fft.ifft(H).real)
            
            # Project back to feasible region
            h_new = np.fft.ifft(H).real
            h_new = project_feasible(h_new)
            
            # Re-transform to frequency domain
            H = np.fft.fft(h_new)
            
            # Check if improved
            current_C5 = compute_C5(h_new)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
                
            # Adaptive learning rate
            if i % 10 == 0:
                learning_rate *= decay
                
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                for i in idxs:
                    h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.02), 0, 1)
                h_new = project_feasible(h_new)
                H = np.fft.fft(h_new)
                patience = 0
                
        return best_h

    # Try frequency domain optimization
    try:
        optimized_h = frequency_domain_optimization()
        final_C5 = compute_C5(optimized_h)
        if final_C5 < best_C5:
            best_h = optimized_h
            best_C5 = final_C5
    except:
        pass

    # Final refinement using gradient ascent with trust region
    def trust_region_update(h_vec, num_samples=5):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        # Compute gradient numerically
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        h_new = h_vec - 0.005 * grad
        h_new = project_feasible(h_new)
        return h_new

    # Perform a few iterations of final optimization
    final_h = best_h.copy()
    for i in range(500):
        if elapsed() > budget_s * 0.95:
            break
        final_h = trust_region_update(final_h)
        current_C5 = compute_C5(final_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = final_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
