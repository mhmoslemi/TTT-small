        h = h * 1.1

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_analytical_gradients(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Analytical gradient for the maximum correlation term
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Gradient contribution from 1st term of overlap sum
            if i + max_k < len(h_vec):
                grad[i] += (1 - h_vec[i + max_k])
            # Gradient contribution from 2nd term of overlap sum
            if i - max_k >= 0:
                grad[i] += -h_vec[i - max_k]
        return grad

    def trust_region_update(h_vec, eta=0.01, max_iter=5):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = compute_analytical_gradients(h_new)
            h_candidate = h_new - eta * grad
            h_candidate = project_feasible(h_candidate)
            
            # Backtracking line search for sufficient decrease
            while compute_C5(h_candidate) > compute_C5(h_new) and eta > 1e-6:
                eta *= 0.5
                h_candidate = h_new - eta * grad
                h_candidate = project_feasible(h_candidate)
            
            h_new = h_candidate
        return h_new

    eta = 0.01
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)

                perturb = np.random.normal(0, 0.01, n_points)
                h = h + perturb
                h = project_feasible(h)
                patience = 0

        # Trust region update with analytical gradients
        h = trust_region_update(h, eta=eta)

    final_C5 = compute_C5(best_h)

        best_h = best_h.copy()
        
    # Final check and projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
