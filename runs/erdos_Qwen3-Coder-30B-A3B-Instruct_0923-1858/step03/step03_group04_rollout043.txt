import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial triangular solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def turbulence(h_vec, alpha=0.1, freq=1):
        # Apply sinusoidal perturbations
        noise = np.random.normal(0, 1, len(h_vec))
        perturbation = alpha * np.sin(freq * np.pi * np.arange(len(h_vec)) / len(h_vec)) * noise
        h_turb = h_vec + perturbation
        h_turb = project_feasible(h_turb)
        return h_turb

    def gradient_step(h_vec, eta=0.02):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        # Compute gradient of the max correlation w.r.t. h
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # finite difference approximation
            eps = 1e-5
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        h_new = h_vec - eta * grad
        return project_feasible(h_new)

    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    patience = 0
    max_patience = 100
    turbulence_freq = 50

    while elapsed() < budget_s * 0.95:
        # Apply turbulence periodically
        if iter_count % turbulence_freq == 0:
            h = turbulence(h, alpha=0.1, freq=1)
        else:
            # Regular gradient step
            h = gradient_step(h, eta=0.01)

        # Evaluate current solution
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count
            patience = 0
        else:
            patience += 1

        # Check for stagnation and restart if needed
        if patience > max_patience:
            # Restart with a better starting point
            try:
                # Try more complex initial construction
                h = np.zeros(n_points)
                x = np.linspace(0, 2, n_points)
                for i in range(n_points):
                    h[i] = max(0, 1 - abs(x[i] - 1)) * 0.5 + 0.25
                h = h * (n_points / 2) / np.sum(h)
                patience = 0
            except:
                pass

        iter_count += 1

    # Final refinement with gradient descent
    final_h = best_h.copy()
    for _ in range(100):
        if elapsed() > budget_s * 0.98:
            break
        final_h = gradient_step(final_h, eta=0.005)
        current_C5 = compute_C5(final_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = final_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
