import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize h from initial_h_values if available, otherwise create
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - 1.5 * np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    # Normalize h to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_overlap_gradients(h_vec, k_shift):
        # Compute gradient of overlap at shift k_shift
        n = len(h_vec)
        grad = np.zeros(n)
        for i in range(n):
            j = i + k_shift
            if 0 <= j < n:
                grad[i] += (1 - h_vec[j])
            j = i - k_shift
            if 0 <= j < n:
                grad[i] -= h_vec[j]
        return grad

    # Try to solve QP for constraint with most constraining shift
    def solve_qp_constraint(h_vec, t_var, current_max_k):
        h_new = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0
        ]
        
        # Add constraint from most constraining overlap
        grad = compute_overlap_gradients(h_vec, current_max_k)
        # C(k) = sum h[i] * (1 - h[i+k]) for k = current_max_k
        # Linearize at current h
        c_val = np.sum(h_vec * (1 - np.roll(h_vec, current_max_k)))
        constraints.append(
            t >= c_val + np.dot(grad, (h_new - h_vec))
        )
        
        # Create the problem and solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, verbose=False, tol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new_sol = h_new.value
                t_sol = t.value
                return h_new_sol, t_sol
        except:
            pass
        return h_vec.copy(), compute_C5(h_vec)

    max_patience = 30
    patience = 0
    iter_count = 0
    t = compute_C5(h)

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                for i in range(n_points):
                    if np.random.random() < 0.1:
                        h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Find most constraining shift
        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)
        
        # Solve QP with this constraint
        h_new, new_t = solve_qp_constraint(h, t, max_k)
        if new_t < t or np.allclose(h_new, h):
            h = project_feasible(h_new)
            t = new_t
        else:
            h = project_feasible(h_new)
            t = compute_C5(h)

        iter_count += 1

    # Final check
    final_C5 = compute_C5(project_feasible(best_h))
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = project_feasible(best_h.copy())

    return (best_h.tolist(), best_C5, n_points)
