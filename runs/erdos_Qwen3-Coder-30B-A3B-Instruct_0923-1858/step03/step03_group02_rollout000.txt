        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum

    def compute_analytical_gradients(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Compute the derivative of C(max_k) w.r.t. h[i]
            # C(k) = sum h[j] * (1 - h[j+k]) for valid j
            # For the specific k that maximizes C(k), we look at contributions
            # to the gradient:
            if i - max_k >= 0:
                grad[i] += (1 - h_vec[i - max_k])  # term from h[i] * (1 - h[i-k])
            if i + max_k < len(h_vec):
                grad[i] -= h_vec[i + max_k]  # term from h[i+k] * (1 - h[i])
                
        return grad

    def trust_region_update(h_vec, eta=0.01, max_iter=10):

            h_candidate = h_new - eta * grad
