import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)

        # Create a triangular initial solution (coarse grid)
        n_points = 30
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing with grid refinement
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iter_count = 0
    patience = 0
    max_patience = 500
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = current_h.copy()

    # Look for best local region to avoid overfitting
    local_best_C5 = float('inf')
    local_best_h = current_h.copy()
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        # Perturb a random index
        i = np.random.randint(0, n_points)
        old_val = current_h[i]
        
        # Try a small random perturbation
        delta = np.random.normal(0, 0.1)
        new_val = current_h[i] + delta
        
        # Store old state for restoration if needed
        old_h = current_h.copy()
        
        # Update and project
        current_h[i] = new_val
        current_h = project_feasible(current_h)
        
        new_C5 = compute_C5(current_h)
        
        # Accept or reject
        if new_C5 < current_C5 or np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
                patience = 0
        else:
            # Restore
            current_h = old_h
            patience += 1
        
        # Grid refinement - double resolution every 1000 iterations
        if iter_count % 1000 == 0 and iter_count > 0:
            old_n = n_points
            n_points *= 2
            dx = 2.0 / n_points
            
            # Interpolate to new grid
            old_x = np.linspace(0, 2, old_n)
            new_x = np.linspace(0, 2, n_points)
            current_h = np.interp(new_x, old_x, current_h)
            current_h = project_feasible(current_h)
            # Need to keep it normalized
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            current_C5 = compute_C5(current_h)
        
        # Cool down
        T *= alpha
        iter_count += 1
        
        # Occasionally update local best
        if new_C5 < local_best_C5:
            local_best_C5 = new_C5
            local_best_h = current_h.copy()
        elif patience > max_patience and iter_count > 1000:
            # Occasionally restart from local best
            current_h = local_best_h.copy()
            current_C5 = local_best_C5
            patience = 0

    # Final check for best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
