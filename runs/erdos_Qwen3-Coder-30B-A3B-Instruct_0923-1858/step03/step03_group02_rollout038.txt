import time
from scipy.signal import correlate

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Alternative: gradient-free optimization with Nelder-Mead
    def nelder_mead_optimize(h_vec, max_iter=1000):
        # Use a slightly different starting point if too close to local minima
        result = minimize(
            lambda x: compute_C5(x),
            h_vec,
            method='Nelder-Mead',
            options={'maxiter': max_iter, 'xtol': 1e-6, 'ftol': 1e-6}
        )
        if result.success:
            return project_feasible(result.x)
        return h_vec

    # Alternative: Randomized search with local refinement
    def randomized_search():
        best_local = h.copy()
        best_local_C5 = compute_C5(best_local)
        
        for _ in range(300):
            # Random perturbations
            h_candidate = best_local.copy()
            for i in range(n_points):
                if np.random.random() < 0.15:
                    h_candidate[i] += np.random.normal(0, 0.01)
            h_candidate = project_feasible(h_candidate)
            
            # Local refinement with Nelder-Mead
            h_refined = nelder_mead_optimize(h_candidate, max_iter=100)
            candidate_C5 = compute_C5(h_refined)
            
            if candidate_C5 < best_local_C5:
                best_local = h_refined
                best_local_C5 = candidate_C5
                
            if elapsed() > budget_s * 0.95:
                break
        
        return best_local, best_local_C5
    
    # Hybrid approach: alternating between local search and global search
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        # Try randomized search every few iterations
        if iter_count % 20 == 0:
            h_local, C5_local = randomized_search()

    # Final refinement

    final_C5 = compute_C5(h_final)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = h_final.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
