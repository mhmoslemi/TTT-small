import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a modified triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_overlap_gradients(h_vec, shift_k):
        n = len(h_vec)
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + shift_k < n:
                grad[i] = (1 - h_vec[i + shift_k])
            if 0 <= i - shift_k < n:
                grad[i] -= h_vec[i - shift_k]
        return grad

    # QP solver with CVXPY
    def solve_qp_constraint(h_old, t_old, k_max, grad_k):
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= t_old  # Basic feasibility
        ]
        
        # Add constraint for the most constraining overlap
        linearized_constraint = t >= cp.sum(cp.multiply(grad_k, (h_new - h_old))) + \
                               np.sum(h_old * (1 - np.roll(h_old, k_max))) * dx
        constraints.append(linearized_constraint)
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_new.value, t.value
        except:
            pass
        return h_old, t_old

    # Iterative improvement
    iter_count = 0
    patience = 0
    max_patience = 30
    current_h = h.copy()
    current_t = compute_C5(current_h)
    best_C5 = current_t

    while elapsed() < budget_s * 0.95:
        iter_count += 1
        
        # Compute all overlaps
        corr = correlate(current_h, 1 - current_h, mode="full")
        # For compatibility with numpy behavior
        max_k_idx = np.argmax(corr)
        k_max = max_k_idx - (len(current_h) - 1)
        
        # Compute gradient for this shift
        grad_k = compute_overlap_gradients(current_h, k_max)
        
        # Solve QP to get new h and t
        new_h, new_t = solve_qp_constraint(current_h, current_t, k_max, grad_k)
        
        # Update if improves
        if new_t < current_t:
            current_t = new_t
            current_h = new_h
            patience = 0
        else:
            patience += 1
        
        # Keep track of best
        if current_t < best_C5:
            best_C5 = current_t
            best_h = current_h.copy()
        
        # Restart if stuck
        if patience > max_patience:
            # Add small random noise to break symmetry and avoid local minima
            noise = np.random.normal(0, 0.05, n_points)
            current_h = current_h + noise
            current_h = project_feasible(current_h)
            patience = 0
            
        # Ensure feasibility after each iteration
        current_h = project_feasible(current_h)

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()   
        
    # Final projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
