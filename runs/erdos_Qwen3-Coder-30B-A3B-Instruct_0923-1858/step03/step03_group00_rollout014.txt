import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def mutate_individual(h, mutation_rate, mutation_scale):
    """Mutate an individual with Gaussian noise."""
    h_mut = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * mutation_rate), replace=False)
    h_mut[indices] += np.random.normal(0, mutation_scale, len(indices))
    return project(h_mut, 2.0 / len(h))

def crossover(parent1, parent2):
    """Perform one-point crossover between two individuals."""
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), size=tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

    best_fitness = float('inf')

    # Parameters
    mutation_rate = 0.05
    mutation_scale = 0.1
    generations = 0
    stagnation_count = 0

        # Evaluate fitness
        fitnesses = [evaluate(ind, dx) for ind in population]
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:

            best_individual = population[np.argmin(fitnesses)].copy()
            stagnation_count = 0
        else:

        # Elitism: keep the top 2
        sorted_indices = np.argsort(fitnesses)
