import time
from scipy.signal import correlate

    np.random.seed(seed)

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated annealing with grid refinement
    T = 0.1
    alpha = 0.999
    iter_count = 0
    patience = 0
    max_patience = 200
    
    while elapsed() < budget_s * 0.95 and T > 1e-6:
        # Create a candidate solution by perturbing h
        h_candidate = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        h_candidate[idx] += delta
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate)
        
        # Compute C5 for both solutions
        current_C5 = compute_C5(h)
        candidate_C5 = compute_C5(h_candidate)
        
        # Accept or reject the candidate
        if candidate_C5 < current_C5:
            h = h_candidate
            if candidate_C5 < best_C5:
                best_C5 = candidate_C5
                best_h = h_candidate.copy()
            patience = 0
        else:
            # Probability of accepting worse solution
            if np.random.random() < np.exp(-(candidate_C5 - current_C5) / T):
                h = h_candidate
                patience = 0
            else:
                patience += 1
        
        # Temperature decay
        T *= alpha
        
        # Grid refinement every 1000 iterations
        if iter_count % 1000 == 0 and iter_count > 0:
            if n_points < 1000:  # Prevent too large grids
                n_points *= 2
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, np.linspace(0, 2, len(h)), h)
                h = project_feasible(h)
                dx = 2.0 / n_points
                
        iter_count += 1
        
        # Early stopping if no improvement for a while
        if patience > max_patience:
            break
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
