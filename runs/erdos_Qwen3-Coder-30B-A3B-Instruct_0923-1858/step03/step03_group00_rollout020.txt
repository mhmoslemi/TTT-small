import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        corr = correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        
        # Target the worst overlap (maximum correlation)
        target_lag = max_idx - (len(h_current) - 1)
        
        # Create the QP to minimize overlap at this specific lag
        h = cp.Variable(n_points)

        # Add lag-specific constraints for the target lag

        # Create terms for correlation at target_lag
        target_corr = 0

                target_corr += h[i + lag] * (1 - h[i])

        constraints = [

            h <= 1,
            cp.sum(h) == 1.0 / dx,
        ]

        problem = cp.Problem(objective, constraints)
