import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        # Get shift that causes maximum correlation
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Gradient approximation using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    def turbulence(h_vec, alpha=0.1, frequency=1):
        # Apply sinusoidal turbulence
        turbulence_signal = alpha * np.sin(np.pi * np.arange(len(h_vec)) * frequency / len(h_vec))
        noise = np.random.normal(0, 1, len(h_vec))
        h_turb = h_vec + turbulence_signal * noise
        return project_feasible(h_turb)

    def optimize_step(h_vec, steps=50):
        h_new = h_vec.copy()
        lr = 0.01
        for _ in range(steps):
            if elapsed() > budget_s * 0.95:
                break
            grad = get_gradient(h_new)
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
            lr *= 0.995  # Decay learning rate
        return h_new

    # Main loop
    iter_count = 0
    last_best_update = 0
    
    while elapsed() < budget_s * 0.95:
        if iter_count % 50 == 0:
            # Apply turbulence for exploration
            h_new = turbulence(h, alpha=0.1, frequency=1.0)
            h_new = project_feasible(h_new)
        else:
            h_new = h.copy()
        
        # Local refinement
        h_new = optimize_step(h_new, steps=20)
        
        current_C5 = compute_C5(h_new)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_new.copy()
            last_best_update = iter_count
        
        # Check for stagnation and restart if needed
        if iter_count - last_best_update > 50:
            # Try random restart
            h = turbulence(h, alpha=0.15, frequency=0.5)
        else:
            h = h_new
            
        iter_count += 1
    
    # Final refinement
    final_h = optimize_step(best_h, steps=100)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
