import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 30
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing with adaptive perturbation
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 500
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                # Occasionally perturb to escape local minima
                idx = np.random.randint(0, n_points)
                h[idx] = np.random.rand()
                h = project_feasible(h)
                no_improve_count = 0

        # Make a perturbation
        idx = np.random.randint(0, n_points)
        delta = (np.random.rand() - 0.5) * 0.2 * T
        h_perturbed = h.copy()
        h_perturbed[idx] += delta
        h_perturbed = project_feasible(h_perturbed)
        
        new_C5 = compute_C5(h_perturbed)
        delta_C5 = new_C5 - current_C5
        
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_perturbed
        
        T *= alpha
        iter_count += 1
        
        # Grid refinement every 1000 iterations
        if iter_count % 1000 == 0 and n_points < 1000:
            # Interpolate to finer grid
            n_points_new = min(1000, n_points * 2)
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_new = np.interp(x_new, x_old, h)
            h = h_new
            n_points = n_points_new
            dx = 2.0 / n_points

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
