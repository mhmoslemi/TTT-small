import time
from scipy.signal import correlate

        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_qp_warm_start(h_vec):
        bounds = Bounds(0, 1, keep_feasible=True)
        # Simple bound-constrained optimization
        result = minimize(
            lambda x: np.sum((x - h_vec)**2),
            h_vec,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x if result.success else h_vec

    # Improved trust region with better gradient estimation
    def compute_gradients(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)

            h_minus = h_vec.copy()

            h_plus[i] = np.clip(h_plus[i] + eps, 0, 1)
            h_minus[i] = np.clip(h_minus[i] - eps, 0, 1)
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)

            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")

            h_candidate = h_new - eta * grad

                h_candidate = h_new - eta * grad
                h_candidate = project_feasible(h_candidate)
            
            h_new = h_candidate

            h = project_feasible(h)

    # Final evaluation
    final_C5 = compute_C5(best_h)

        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Final check and projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
