import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    np.random.seed(seed)
    n_points = 90
    dx = 2.0 / n_points

        initial_h = np.ones(n_points) * 0.5

        initial_h += 0.3 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)

    iteration = 0
    
    while time.time() - t0 < max_time:
        if iteration % decay_interval == 0 and iteration > 0:
            lr *= lr_decay
            
        # Step 1: Gradient estimate for current h
        grad = np.zeros_like(h_current)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)

            h_minus[j] -= eps

        h_new = h_current - lr * grad
        h_new = project(h_new, dx)

        if current_C < best_C:
            best_C = current_C
            best_h = h_new.copy()
            
        h_current = h_new.copy()
        iteration += 1
        
        # Occasionally reset for escape from local minima
        if iteration % 2000 == 0:
            h_current = initial_h.copy()
            lr *= 0.8
            
    # Final verification
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
