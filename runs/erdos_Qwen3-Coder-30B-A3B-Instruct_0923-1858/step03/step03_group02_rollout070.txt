import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h0 = np.maximum(0, 1 - np.abs(x - 1))
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # PSO parameters
    n_particles = 50
    max_iter = 2000
    inertia = 0.9
    inertia_decay = 0.995
    cognitive = 2.0
    social = 2.0
    velocity_clamp = 0.1
    perturb_rate = 0.1
    perturb_magnitude = 0.03
    
    # Initialize particle swarm
    particles = []
    velocities = []
    
    # Create initial particles
    particles.append(h0.copy())
    velocities.append(np.zeros_like(h0))
    
    for i in range(1, n_particles):
        h_pert = h0 + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Evaluate initial particles
    fitnesses = []
    for i in range(n_particles):
        corr = correlate(particles[i], 1 - particles[i], mode="full")
        fitness = np.max(corr) * (2.0 / n_points)
        fitnesses.append(fitness)
    
    # Track personal bests
    pbest = particles.copy()
    pbest_fitness = fitnesses.copy()
    
    # Track global best
    global_best_idx = np.argmin(fitnesses)
    gbest = particles[global_best_idx].copy()
    gbest_fitness = fitnesses[global_best_idx]
    
    # PSO main loop
    last_improvement = 0
    iteration = 0
    
    while elapsed() < budget_s * 0.95 and iteration < max_iter:
        if iteration % 50 == 0 and iteration > 0:
            inertia = max(0.4, inertia * inertia_decay)
        
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            cognitive_component = cognitive * r1 * (pbest[i] - particles[i])
            social_component = social * r2 * (gbest - particles[i])
            
            velocities[i] = (inertia * velocities[i] + 
                           cognitive_component + 
                           social_component)
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], 
                                  -velocity_clamp, 
                                  velocity_clamp)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            corr = correlate(particles[i], 1 - particles[i], mode="full")
            new_fitness = np.max(corr) * (2.0 / n_points)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = new_fitness
                    last_improvement = iteration
        
        # Perturb particles if no improvement recently
        if iteration - last_improvement > 30:
            for i in range(n_particles):
                if np.random.random() < perturb_rate:
                    perturbation = np.random.normal(0, perturb_magnitude, n_points)
                    particles[i] = particles[i] + perturbation
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    
                    # Recompute fitness
                    corr = correlate(particles[i], 1 - particles[i], mode="full")
                    new_fitness = np.max(corr) * (2.0 / n_points)
                    pbest_fitness[i] = new_fitness  # in case it's now better
                    if new_fitness < gbest_fitness:
                        gbest = particles[i].copy()
                        gbest_fitness = new_fitness
                        last_improvement = iteration
        
        iteration += 1
    
    # Final evaluation of global best
    final_corr = correlate(gbest, 1 - gbest, mode="full")
    final_fitness = np.max(final_corr) * (2.0 / n_points)
    
    return (gbest.tolist(), final_fitness, n_points)
