import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Bisection search for optimal C5
    def bisect_C5():
        low, high = 0.0, 1.0
        eps = 1e-4
        max_iter = 100
        iter_count = 0
        best_feasible_C = 0.0
        
        # Helper to test feasibility for a given C
        def is_feasible(C):
            # Define the optimization variables
            h_var = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # Sample shifts to approximate the constraint set
            # Only test about 30 shifts
            n_shifts = min(30, n_points * 2 - 1)
            shifts = np.arange(-n_points + 1, n_points)
            shifts = shifts[shifts != 0]
            if len(shifts) > n_shifts:
                shifts = np.random.choice(shifts, n_shifts, replace=False)
            
            # Add cross-correlation constraints
            for shift in shifts:
                # For shift k, we want integral of h(x) * (1 - h(x + k)) to be at least C
                # This can be  written as sum over i of h[i] * (1 - h[i+k]) for valid k
                # Note the correlation is computed such that k increases from lhs to rhs
                # correlation has size 2*n_points -1
                # For cross correlation h(x) * (1-h(x+k)), we want to find the correct index
                full_corr_idx = shift + n_points - 1
                
                # Sum over pairs (i, i+k) for valid k where i+k < n_points
                # Create a constraint using dual formulation
                corr_term = 0
                for i in range(n_points):
                    j = i + shift
                    if 0 <= j < n_points:
                        corr_term += h_var[i] * (1 - h_var[j])
                    else:
                        continue
                constraints.append(corr_term >= (1 - C) * dx * n_points)  # Normalize by dx
            
            # Objective (minimize something to enforce feasibility)
            objective = cp.Minimize(0)
            
            # Solve the problem
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    return True
            except:
                pass
            return False
        
        while high - low > eps and iter_count < max_iter and elapsed() < budget_s * 0.9:
            mid = (low + high) / 2.0
            if is_feasible(mid):
                low = mid
                best_feasible_C = mid
            else:
                high = mid
            iter_count += 1
        
        # Return the highest feasible C
        return best_feasible_C
    
    # Optimize over samples along a few gradient directions
    def local_optimize(h_vec, max_iter=50):
        h_vec = np.array(h_vec)
        old_C5 = compute_C5(h_vec)
        h_current = h_vec.copy()
        step_size = 0.01
        
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            # Random sample to estimate gradient
            grad = np.zeros_like(h_current)
            eps = 1e-4
            num_perturbations = 5
            
            for _ in range(num_perturbations):
                delta = np.random.normal(0, eps, n_points)
                h_perturbed_plus = project_feasible(h_current + delta)
                h_perturbed_minus = project_feasible(h_current - delta)
                
                corr_plus = correlate(h_perturbed_plus, 1 - h_perturbed_plus, mode="full")
                corr_minus = correlate(h_perturbed_minus, 1 - h_perturbed_minus, mode="full")
                
                max_k_idx = np.argmax(corr_plus)
                grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
            grad /= num_perturbations
            h_current = h_current - step_size * grad
            h_current = project_feasible(h_current)
            
            new_C5 = compute_C5(h_current)
            if new_C5 < old_C5:
                old_C5 = new_C5
            else:
                step_size *= 0.95  # Reduce step size if no improvement
        
        return h_current
    
    # Main optimization loop
    iter_count = 0
    while elapsed() < budget_s * 0.9:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Optimization step
        h = local_optimize(h)
        
        # Occasionally try bisection search from current solution
        if iter_count % 20 == 0 and iter_count > 0:
            try:
                # Approximate CVX solution to get a better direction
                exact_C = bisect_C5()
                if exact_C > best_C5:
                    # Reoptimize below this new bound
                    pass  # Could do more grained search
            except:
                pass
        
        iter_count += 1
    
    # Final phase: local refinement
    final_h = best_h.copy()
    for _ in range(20):
        if elapsed() > budget_s * 0.99:
            break
        final_h = local_optimize(final_h, max_iter=10)
    
    final_C5 = compute_C5(final_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
