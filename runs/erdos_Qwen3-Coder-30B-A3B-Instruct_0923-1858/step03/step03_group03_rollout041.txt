import time

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Compute C5 efficiently
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Use correlation from scipy for performance
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Project to feasible region
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iteration = 0
    max_iter = 10000
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        # Backup current solution
        current_h = h.copy()
        current_C5 = compute_C5(h)
        
        # Perturb a random point
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        h[idx] += delta
        
        # Project to feasible region
        h = project_feasible(h)
        
        # Compute new C5
        new_C5 = compute_C5(h)
        
        # Accept or reject based on SA
        if new_C5 < current_C5:
            # Always accept improvement
            accepted = True
        else:
            # Accept with probability based on temperature
            delta_C5 = new_C5 - current_C5
            prob = np.exp(-delta_C5 / T)
            accepted = np.random.rand() < prob
        
        if not accepted:
            # Restore
            h = current_h.copy()
        
        # Update best solution
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
        
        # Update temperature
        T *= alpha
        
        # Grid refinement
        if iteration % 1000 == 0 and iteration > 0:

                new_n = min(n_points * 2, 1000)
                x_old = np.linspace(0, 2, n_points)

    # Final computation of best C5

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
