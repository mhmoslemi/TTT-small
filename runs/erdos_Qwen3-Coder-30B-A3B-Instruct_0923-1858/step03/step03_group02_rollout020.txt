import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 50
    max_iter = 2000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_clamp = 0.1

    # Initialize swarm with particles
    swarm = []
    velocities = []
    p_best = []
    p_best_fitness = []

    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # Perturb initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_feasible(particle)
        swarm.append(particle.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_fitness.append(compute_C5(particle))

    # Global best
    global_best_idx = np.argmin(p_best_fitness)
    s_best = p_best[global_best_idx].copy()
    s_best_fitness = p_best_fitness[global_best_idx]

    iter_count = 0
    patience = 0
    max_patience = 30

    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(swarm_size):
            particle = swarm[i]
            velocity = velocities[i]
            current_fitness = p_best_fitness[i]

            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (p_best[i] - particle)
            social = social_weight * r2 * (s_best - particle)
            velocity = inertia * velocity + cognitive + social

            # Clamp velocity
            velocity = np.clip(velocity, -velocity_clamp, velocity_clamp)

            # Update position
            new_position = particle + velocity
            new_position = project_feasible(new_position)

            # Evaluate
            new_fitness = compute_C5(new_position)

            # Update personal best
            if new_fitness < current_fitness:
                p_best[i] = new_position.copy()
                p_best_fitness[i] = new_fitness

            # Update global best
            if new_fitness < s_best_fitness:
                s_best = new_position.copy()
                s_best_fitness = new_fitness
                patience = 0
            else:
                patience += 1

            # Store updated particle and velocity
            swarm[i] = new_position
            velocities[i] = velocity
            
        # Perturb particles if no improvement for a while
        if patience > max_patience:
            for i in range(swarm_size):
                swarm[i] += np.random.normal(0, 0.03, n_points)
                swarm[i] = project_feasible(swarm[i])
                # Re-evaluate and update p_best
                fitness = compute_C5(swarm[i])
                if fitness < p_best_fitness[i]:
                    p_best[i] = swarm[i].copy()
                    p_best_fitness[i] = fitness
            patience = 0

        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(s_best)
    if final_C5 < s_best_fitness:
        s_best_fitness = final_C5
        
    return (s_best.tolist(), s_best_fitness, n_points)
