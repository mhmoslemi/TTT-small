        return h_vec

    def local_improvement(h_vec, max_iter=20):
        h_new = h_vec.copy()

            # Random small perturbation
            idx = np.random.randint(0, len(h_vec))
            step = np.random.normal(0, 0.02)
            h_new[idx] = np.clip(h_new[idx] + step, 0, 1)
            
            # Project to feasible region
            h_new = project_feasible(h_new)
            
            # Accept if better
            if compute_C5(h_new) < compute_C5(h_vec):
                h_vec = h_new.copy()

                diff = compute_C5(h_new) - compute_C5(h_vec)
