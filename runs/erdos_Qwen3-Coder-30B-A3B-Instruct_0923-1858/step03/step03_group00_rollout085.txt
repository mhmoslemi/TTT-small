import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def mutate_individual(h, mutation_rate, mutation_scale):
    """Mutate an individual with Gaussian noise."""
    # Mutate with Gaussian noise
    mutation = np.random.normal(0, mutation_scale, len(h))
    h_mut = h + mutation
    h_mut = project(h_mut, 1.0/len(h))
    return h_mut

def crossover(parent1, parent2):
    """Perform one-point crossover between two parents."""
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    winner_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 20
    population = []
    
    try:

        if len(individual) != n_points:

        individual = np.ones(n_points) * 0.5
        t = np.linspace(0, 2, n_points)
        individual = 0.3 + 0.4 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        individual = project(individual, dx)
        population.append(individual)
    
    # Generate additional individuals
    for _ in range(population_size - 1):
        # Start with a structure, then add some noise
        individual = np.ones(n_points) * 0.5
        t = np.linspace(0, 2, n_points)
        individual = 0.3 + 0.4 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        individual = project(individual, dx)
        
        # Add some mutation noise
        individual = mutate_individual(individual, 0.05, 0.1)
        population.append(individual)
    
    # GA parameters
    max_generations = 5000
    mutation_rate = 0.05
    crossover_rate = 0.8
    elite_size = 2
    mutation_scale_start = 0.2
    mutation_scale_end = 0.01
    
    # Evolution loop
    t0 = time.time()
    best_solution = None
    best_fitness = float('inf')
    
    for generation in range(max_generations):
        if time.time() - t0 > budget_s - 2:
            break
            
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            fitness = evaluate(individual, dx)
            fitnesses.append(fitness)
            
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_solution = population[min_fitness_idx].copy()
            
        # Elitism: keep top individuals
        sorted_indices = np.argsort(fitnesses)
        new_population = [population[i] for i in sorted_indices[:elite_size]]
        
        # Tournament selection and reproduction
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover

            # Mutation

    # Final validation and return
