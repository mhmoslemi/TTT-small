import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO Parameters
    n_particles = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Use initial solution as first particle
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h.copy())
    pbest_values.append(compute_C5(h))
    
    # Add random particles
    for i in range(n_particles - 1):
        h_new = project_feasible(h + np.random.normal(0, 0.05, n_points))
        particles.append(h_new.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h_new.copy())
        pbest_values.append(compute_C5(h_new))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO optimization
    for iteration in range(max_iter):
        if elapsed() >= budget_s * 0.6:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            current_value = compute_C5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
        
        # Periodic perturbation to avoid stagnation
        if iteration % 10 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.2:
                    particles[i] = project_feasible(particles[i] + np.random.normal(0, 0.01, n_points))

    # L-BFGS-B refinement
    if elapsed() < budget_s * 0.95:
        # Create bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            return compute_C5(h_vec)
        
        # Refine best solution from PSO
        result = minimize(objective, gbest, method='L-BFGS-B', bounds=bounds, 
                         options={'maxiter': 300, 'ftol': 1e-6, 'gtol': 1e-5})
        
        if result.success:
            refined_h = project_feasible(result.x)
            refined_C5 = compute_C5(refined_h)
            
            if refined_C5 < best_C5:
                best_h = refined_h.copy()
                best_C5 = refined_C5

    # Final evaluation and return
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
