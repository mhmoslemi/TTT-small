import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def pso_optimization():
        # PSO parameters
        swarm_size = 20
        max_iter = 200
        w = 0.7
        c1 = 1.5
        c2 = 1.5
        v_max = 0.1
        
        # Initialize swarm
        swarm = []
        velocities = []
        personal_best = []
        personal_best_values = []
        global_best = None
        global_best_value = float('inf')
        
        for _ in range(swarm_size):
            # Random initial positions
            h_init = np.random.rand(n_points)
            h_init = project_feasible(h_init)
            swarm.append(h_init)
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            personal_best.append(h_init.copy())
            value = compute_C5(h_init)
            personal_best_values.append(value)
            if value < global_best_value:
                global_best = h_init.copy()
                global_best_value = value
        
        # PSO iterations
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.7:
                break
                
            for i in range(swarm_size):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (w * velocities[i] + 
                               c1 * r1 * (personal_best[i] - swarm[i]) + 
                               c2 * r2 * (global_best - swarm[i]))
                
                # Clamp velocity
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                swarm[i] += velocities[i]
                swarm[i] = project_feasible(swarm[i])
                
                # Evaluate fitness
                fitness = compute_C5(swarm[i])
                
                # Update personal best
                if fitness < personal_best_values[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_values[i] = fitness
                    
                    # Update global best
                    if fitness < global_best_value:
                        global_best = swarm[i].copy()
                        global_best_value = fitness
        
        return global_best

    def lbfgsb_refinement(h_start):
        # L-BFGS-B optimization with constraints
        def objective(h_vec):
            return compute_C5(h_vec)
        
        def constraint_function(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_function}
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # minimize with L-BFGS-B
        result = minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                          options={'maxiter': 300, 'ftol': 1e-5, 'gtol': 1e-5})
        
        return result.x

    # Begin optimization
    # Phase 1: PSO optimization
    if elapsed() < budget_s * 0.7:
        h_pso = pso_optimization()
        c5_pso = compute_C5(h_pso)
        if c5_pso < best_C5:
            best_C5 = c5_pso
            best_h = h_pso.copy()
    
    # Phase 2: L-BFGS-B refinement on best so far
    if elapsed() < budget_s * 0.95:
        h_refined = lbfgsb_refinement(best_h)
        c5_refined = compute_C5(h_refined)
        if c5_refined < best_C5:
            best_C5 = c5_refined
            best_h = h_refined.copy()

    # Final iteration with more specialized refinement
    if elapsed() < budget_s:
        # Try to improve with random perturbations if time allows
        for _ in range(300):
            if elapsed() > budget_s * 0.99:
                break
            # Try small perturbations to escape local minima
            h_temp = best_h.copy()
            perturbation = np.random.normal(0, 0.01, n_points)
            h_temp += perturbation
            h_temp = project_feasible(h_temp)
            c5_temp = compute_C5(h_temp)
            if c5_temp < best_C5:
                best_C5 = c5_temp
                best_h = h_temp.copy()

    return (best_h.tolist(), best_C5, len(best_h))
