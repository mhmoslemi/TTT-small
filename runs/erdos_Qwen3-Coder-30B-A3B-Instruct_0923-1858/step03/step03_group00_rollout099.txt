import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    h_current = initial_h.copy()

    # Adaptive targeting: solve QPs for worst overlaps
    t0 = time.time()
    max_time = budget_s - 2

        corr = correlate(h_current, 1 - h_current, mode='full')

        objective_expr = 0

                if k == lag and i < n_points and j < n_points:
                    objective_expr += h[i] * (1 - h[j])

        prob = cp.Problem(cp.Minimize(objective_expr),
