import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available, otherwise create one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Triangular construction with steeper decay
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_overlap_and_grad(h_vec, k):
        h_vec = np.array(h_vec)
        n = len(h_vec)
        # Compute overlap for shift k
        if k >= 0:
            overlap = np.sum(h_vec[:n-k] * (1 - h_vec[k:]))
        else:
            overlap = np.sum(h_vec[-k:] * (1 - h_vec[:-k]))
        # Compute gradient
        grad = np.zeros(n)
        if k >= 0:
            for i in range(n-k):
                grad[i] += (1 - h_vec[i + k])
                grad[i + k] -= h_vec[i]
        else:
            for i in range(-k, n):
                grad[i] += (1 - h_vec[i - k])
                grad[i - k] -= h_vec[i]
        return overlap, grad
    
    def update_h_via_qp(h_vec, t_var, k_max):
        # Construct QP with current most constraining constraint
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [0 <= h_new, h_new <= 1, cp.sum(h_new) == n_points / 2, t >= 0]
        
        # Linearized constraint for the most constraining shift k_max
        overlap_val, grad = compute_overlap_and_grad(h_vec, k_max)
        # t >= overlap_val + grad @ (h_new - h_vec)
        constraints.append(t >= overlap_val + grad @ (h_new - h_vec))
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status == cp.OPTIMAL:
                h_updated = h_new.value
                t_final = t.value
                return h_updated, t_final, True
        except:
            pass
        return h_vec, t_var, False
    
    # Main optimization loop
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation step
                for i in range(n_points):
                    if np.random.random() < 0.1:
                        h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Find the most constraining overlap
        max_overlap = -1
        best_k = 0
        for k in range(1 - n_points, n_points):
            overlap_val, _ = compute_overlap_and_grad(h, k)
            if overlap_val > max_overlap:
                max_overlap = overlap_val
                best_k = k

        # Try updating via QP
        h_new, t_new, success = update_h_via_qp(h, current_C5, best_k)
            
        if success:
            h = h_new
            t_new = compute_C5(h)
            if t_new < current_C5:
                h = project_feasible(h)
                patience = 0
        else:
            h = project_feasible(h)

        # Periodic feasibility check
        if iter_count % 10 == 0:
            h = project_feasible(h)
        
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
