    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use frequency-domain approach
    def frequency_domain_update(h_vec, lr=0.01, smoothing=0.9):
        h_vec = np.array(h_vec)
        H = fft(h_vec)
        
        # Compute correlation in frequency domain
        G = H * np.conj(H)
        corr_freq = ifft(G)
        
        # Approximate gradient in frequency domain
        grad_freq = 2 * np.real(H) * np.conj(H)  # Simplified gradient approximation
        
        # Update in frequency domain
        H_new = H - lr * grad_freq
        
        # Convert back to spatial domain
        h_new = np.real(ifft(H_new))
        
        # Project back to feasible space
        h_new = project_feasible(h_new)
        
        return h_new
