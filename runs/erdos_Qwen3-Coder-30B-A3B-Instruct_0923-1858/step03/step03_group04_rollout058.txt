        h = h * (n_points / 2) / np.sum(h)

    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO Parameters
    swarm_size = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Create initial swarm around best solution or triangular
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # Add noise to initial solution
            noise = np.random.normal(0, 0.03, n_points)
            particle = h + noise
        particle = project_feasible(particle)
        swarm.append(particle)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Track personal bests
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_scores = [compute_C5(swarm[i]) for i in range(swarm_size)]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.6:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) +
                           c2 * r2 * (global_best - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            score = compute_C5(swarm[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = swarm[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = swarm[i].copy()

    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_fun(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Set bounds and constraints for L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_fun}

    # Refine the global best
    try:
        result = minimize(objective, 
                         global_best, 
                         method='L-BFGS-B',
                         bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 300, 'ftol': 1e-8})
        if result.success:
            refined_h = project_feasible(result.x)
            refined_C5 = compute_C5(refined_h)
            if refined_C5 < global_best_score:
                global_best = refined_h
                global_best_score = refined_C5
    except:
        pass
    
    final_C5 = compute_C5(global_best)
    
    return (global_best.tolist(), final_C5, len(global_best))
