import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        h_current = initial_h_values.copy()
        if len(h_current) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create initial guess
        t = np.linspace(0, 2, n_points)
        h_current = 0.3 + 0.4 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        h_current = project(h_current, dx)
    
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    num_qps = 0
    max_qps = 100
    
    while time.time() - t0 < max_time and num_qps < max_qps:
        # Find the maximum overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h_current) - 1)
        
        # Create QP to minimize correlation at max_lag
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == 1.0 / dx
        ]
        
        # Build the correlation term for the specific lag
        # correlation at lag k = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
        # This is a quadratic constraint with a linear objective
        # We want to minimize: sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
        # which is sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h[i+k]
        
        lag = max_lag
        if lag >= 0:
            # h[i] * (1 - h[i+lag]) for i in [0, n-lag-1]
            objective = 0
            for i in range(n_points - lag):
                objective += h[i] * (1 - h[i + lag])
        else:
            # h[i+lag] * (1 - h[i]) for i in [0, n+lag-1]
            objective = 0
            for i in range(n_points + lag):
                objective += h[i + (-lag)] * (1 - h[i])
        
        # minimize objective
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h.value
                # Make sure we respect constraints even from numerical inaccuracy
                h_new = project(h_new, dx)
                current_C = evaluate(h_new, dx)
                if current_C < best_C:
                    best_C = current_C
                    best_h = h_new.copy()
                h_current = h_new.copy()
                num_qps += 1
            else:
                # fallback to gradient descent if QP fails
                break
        except:
            # fallback to gradient descent
            break
    
    # Final gradient descent refinement
    lr = 0.005
    decay_interval = 500
    decay_factor = 0.95
    it = 0
    while time.time() - t0 < max_time:
        if it % decay_interval == 0:
            lr *= decay_factor
            
        grad = np.zeros_like(h_current)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        h_new = h_current - lr * grad
        h_new = project(h_new, dx)
        
        current_C = evaluate(h_new, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_new.copy()
            
        h_current = h_new.copy()
        it += 1

    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
