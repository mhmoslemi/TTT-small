from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def init_population(n_points, pop_size):
    """Initialize a population of binary vectors with exactly n_points/2 ones."""
    population = []
    target_ones = n_points // 2
    for _ in range(pop_size):
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, target_ones, replace=False)
        h[ones_indices] = 1
        population.append(h)
    return population

def get_fitness(population, dx):
    """Compute fitness (inverse of C) for population."""
    fitnesses = []
    for h in population:
        c = evaluate(h, dx)
        fitnesses.append(1 / c if c > 0 else float('inf'))
    return np.array(fitnesses)

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select individuals using tournament selection."""
    selected = []
    n_pop = len(population)
    for _ in range(n_pop):
        tournament_indices = np.random.choice(n_pop, tournament_size)
        tournament_fitnesses = fitnesses[tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
        selected.append(population[winner_idx].copy())
    return selected

    offspring2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Restore constraint: exactly n/2 ones
    ones1 = np.sum(offspring1)
    ones2 = np.sum(offspring2)
    target = n // 2
    
    if ones1 != target:
        # Find indices to flip
        if ones1 > target:
            flip_indices = np.where(offspring1 == 1)[0]

            flip_indices = np.where(offspring2 == 1)[0]
            flip_indices = np.random.choice(flip_indices, ones2 - target, replace=False)
            offspring2[flip_indices] = 0
        else:
            flip_indices = np.where(offspring2 == 0)[0]
            flip_indices = np.random.choice(flip_indices, target - ones2, replace=False)
            offspring2[flip_indices] = 1
    
    return offspring1, offspring2

    # Flip bits with mutation rate
    flip_indices = np.random.choice(n, int(mutation_rate * n), replace=False)
    mutated[flip_indices] = 1 - mutated[flip_indices]
    
    # Rebalance to have exactly target ones
    current_ones = np.sum(mutated)

        flip_indices = np.where(mutated == 1)[0]

        flip_indices = np.where(mutated == 0)[0]
        flip_indices = np.random.choice(flip_indices, target - current_ones, replace=False)
        mutated[flip_indices] = 1
    
    return mutated
