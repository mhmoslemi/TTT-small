import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        # Adapt scale based on temperature
        current_scale = max(min_scale, scale_factor * T)

        delta = np.random.normal(0, current_scale, n_points)
        h_candidate = h_current + delta
        h_candidate = project(h_candidate, dx)
        
        # Evaluate the candidate
        candidate_C = evaluate(h_candidate, dx)
        
        # Accept or reject based on simulated annealing
        if candidate_C < best_C or np.random.rand() < np.exp(-(candidate_C - best_C) / T):
