import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # More accurate gradient estimation using finite differences
    def get_gradient(h_vec, eps=1e-5):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

    # Improved trust region method with adaptive steps
    def refine_solution(h_vec, max_iter=100, learning_rate=0.02):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new)
            h_new = h_new - learning_rate * grad
            h_new = project_feasible(h_new)
        return h_new

    # Perturbation with structure preserving noise
    def perturb_solution(h_vec, mutation_strength=0.05):
        h_new = h_vec.copy()
        # Randomly select indices to modify, with more diversity
        indices = np.random.choice(len(h_vec), max(1, len(h_vec)//10), replace=False)
        for idx in indices:
            # Add noise respecting bounds
            h_new[idx] += np.random.normal(0, mutation_strength)
            # Bound check and project
            h_new[idx] = max(0, min(1, h_new[idx]))
        # Re-normalize
        current_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / current_sum
        return h_new

    # Alternative: Direct optimization approach with constraints
    def optimize_with_constraints(h0):
        # Constraint: sum h * dx = 1 -> sum h = n_points / 2
        def constraint_sum(x):
            return np.sum(x) - n_points / 2

        def objective(x):
            # Minimize max overlap
            x = np.array(x)
            corr = correlate(x, 1 - x, mode="full")
            return -np.max(corr) * dx  # Negative because we minimize

        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 200, 'ftol': 1e-8}, callback=None)
        return res.x
        
    # Main optimization loop
    convergence_patience = 0
    max_patience = 100
    iter_count = 0
    eta = 0.02
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            convergence_patience = 0
        else:
            convergence_patience += 1
            if convergence_patience > max_patience:
                # Perturb slightly
                h = perturb_solution(h, mutation_strength=0.05)
                convergence_patience = 0

        # Try different optimization strategies
        if iter_count % 10 == 0 and iter_count > 0:
            # Try constrained optimization
            try:
                h_opt = optimize_with_constraints(h)
                h = h_opt
            except:
                # Fall back to gradient refinement
                h = refine_solution(h, max_iter=20, learning_rate=eta)
        else:
            h = refine_solution(h, max_iter=20, learning_rate=eta)
            
        # Adjust learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
