import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = 45

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Quasi-Newton method with L-BFGS-like updates
    def lbfgs_update(h_vec, grad, s_list, y_list, rho_list, H0=1.0):
        # L-BFGS two-loop recursion
        m = len(s_list)
        if m == 0:
            return -grad * H0
        
        # Forward loop
        alpha_list = []
        q = grad.copy()
        for i in range(m-1, -1, -1):
            alpha = rho_list[i] * np.dot(s_list[i], q)
            alpha_list.append(alpha)
            q = q - alpha * y_list[i]
        
        # Initialize r = H0 * q
        r = q * H0
        
        # Backward loop
        for i in range(m):
            beta = rho_list[i] * np.dot(y_list[i], r)
            r = r + s_list[i] * (alpha_list[m-1-i] - beta)
            
        return -r
    
    # Function to get gradient using finite differences
    def get_gradient(h_vec, eps=1e-5):
        h_vec = np.array(h_vec)
        # Compute the current max correlation and its index
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad

    # Store recent updates for L-BFGS
    s_list = []
    y_list = []
    rho_list = []
    max_history = 10
    
    # Main optimization
    patience = 0
    max_patience = 30
    eta = 0.01

        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Small random perturbation if stagnation occurs
                noise = np.random.normal(0, 0.01, len(h))
                h = h + noise
                h = project_feasible(h)
                patience = 0

        grad = get_gradient(h)

        d = lbfgs_update(h, grad, s_list, y_list, rho_list, H0)

        alpha = eta
        h_new = h + alpha * d
        h_new = project_feasible(h_new)
        
        # Ensure we're improving
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            # Update for next iteration
            h = h_new
            s = h_new - h
            grad_new = get_gradient(h_new)
            y = grad_new - grad
            rho = 1.0 / (np.dot(s, y) + 1e-10)
            
            # Maintain history
            if len(s_list) >= max_history:
                s_list.pop(0)
                y_list.pop(0)
                rho_list.pop(0)
            
            s_list.append(s)
            y_list.append(y)
            rho_list.append(rho)
        else:
            # Reduce step size if no improvement
            eta *= 0.5
            if eta < 1e-6:
                # If step size too small, perform minor perturbation
                h = h + np.random.normal(0, 0.005, len(h))
                h = project_feasible(h)
        
        iter_count += 1
        
        # Occasionally reset for escaping local minima
        if iter_count % 500 == 0:
            h = project_feasible(np.random.rand(len(h)) * 0.5)
            s_list = []
            y_list = []
            rho_list = []
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
