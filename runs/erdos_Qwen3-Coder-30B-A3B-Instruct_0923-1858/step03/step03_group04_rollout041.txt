import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize solution
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Turbulence function to help escape local minima
    def apply_turbulence(h_vec, alpha=0.1):
        # Apply sinusoidal noise of varying frequency
        turbulence = alpha * np.sin(np.pi * np.arange(n_points) / n_points) * np.random.normal(0, 1, n_points)
        h_turb = h_vec + turbulence
        return project_feasible(h_turb)

    # Improved gradient estimation
    def estimate_gradient(h_vec, eps=1e-4, samples=8):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        grad = np.zeros_like(h_vec)
        
        for _ in range(samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = project_feasible(h_vec + delta)
            h_minus = project_feasible(h_vec - delta)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
        
        return grad / samples

    # Gradient descent step with projection
    def gradient_step(h_vec, lr=0.01):
        grad = estimate_gradient(h_vec)
        h_new = h_vec - lr * grad
        return project_feasible(h_new)

    # Main optimization loop with turbulence and gradient refinement
    iter_count = 0
    last_improvement = 0
    max_stagnation = 50
    
    while elapsed() < budget_s * 0.95:
        # Apply turbulence periodically
        if iter_count % 25 == 0:
            h = apply_turbulence(h)
            
        # Gradient descent steps
        for _ in range(10):
            if elapsed() >= budget_s * 0.95:
                break
            h = gradient_step(h, lr=0.01)
            
        # Evaluate current solution
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count
            
        # Early termination if no improvement for too long
        if iter_count - last_improvement > max_stagnation:
            # Apply larger turbulence to escape local minimum
            h = apply_turbulence(h, alpha=0.15)
            last_improvement = iter_count            
        
        iter_count += 1
        
        # Occasionally restart from a better solution or convex relaxation
        if iter_count % 100 == 0:
            try:
                # Try convex relaxation approach - minimize sum of squares
                h_var = cp.Variable(n_points)
                constraints = [
                    h_var >= 0,
                    h_var <= 1,
                    cp.sum(h_var) == n_points / 2
                ]
                objective = cp.Minimize(cp.sum_squares(h_var))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                relaxed_h = np.maximum(0, np.minimum(1, h_var.value))
                relaxed_C5 = compute_C5(relaxed_h)
                if relaxed_C5 < best_C5:
                    best_C5 = relaxed_C5
                    best_h = relaxed_h.copy()
                    h = relaxed_h.copy()
            except:
                pass

    # Final refinement with more gradient steps
    for _ in range(50):
        if elapsed() >= budget_s * 0.95:
            break
        h = gradient_step(h, lr=0.005)
        
    final_C5 = compute_C5(h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
