import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a modified triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Utility to compute the gradient of overlap with shift k
    def compute_shift_gradient(h_vec, k):
        grad = np.zeros_like(h_vec)
        if k >= 0:
            # grad[i] = (1 - h[i+k]) - h[i-k]   for valid indices
            # This handles both terms correctly
            for i in range(len(h_vec)):
                if i + k < len(h_vec):
                    grad[i] += (1 - h_vec[i + k])
                if i - k >= 0:
                    grad[i] -= h_vec[i - k]
        else:
            # k < 0
            k_abs = -k
            for i in range(len(h_vec)):
                if i + k_abs < len(h_vec):
                    grad[i] += (1 - h_vec[i + k_abs])
                if i - k_abs >= 0:
                    grad[i] -= h_vec[i - k_abs]
        return grad
    
    # Main optimization loop with QP
    max_patience = 30
    patience = 0
    iter_count = 0
    
    # Track best solution found
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    while elapsed() < budget_s * 0.95:
        # Compute overlaps and find most constraining shift
        corr = correlate(h, 1 - h, mode="full")
        k_vals = np.arange(-len(h) + 1, len(h))
        max_k_idx = np.argmax(corr)
        max_k = k_vals[max_k_idx]
        
        # Compute gradient at max_k
        grad = compute_shift_gradient(h, max_k)
        
        # Define QP problem using CVXPY
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constr = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= corr[max_k_idx] + grad.T @ (h_new - h)
        ]
        
        # QR solve approach
        prob = cp.Problem(cp.Minimize(t), constr)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_try = h_new.value
                if h_try is not None:
                    h_try = project_feasible(h_try)
                    new_C5 = compute_C5(h_try)
                    if new_C5 < best_C5:
                        best_C5 = new_C5
                        best_h = h_try.copy()
                        patience = 0
                    h = h_try
                else:
                    # Fallback: no solution found, try small perturbation
                    h = project_feasible(h)
                    patience += 1
            else:
                # QP failed, perturb and restart
                patience += 1
        except Exception:
            # If optimization fails, do perturbation
            patience += 1

        # Periodic restart if stuck
        if patience > max_patience:
            h = project_feasible(h + np.random.normal(0, 0.05, n_points))
            patience = 0

        iter_count += 1

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    # Final projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
