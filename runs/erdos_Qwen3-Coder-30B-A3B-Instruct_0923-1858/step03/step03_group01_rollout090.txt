import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        new_population = []

            parent1 = selected[i]

            child2 = fix_child(child2)

    # Final check
    if best_h is None:
        # Fallback to initial_h
        best_h = initial_h.copy()
        best_C = evaluate(best_h, dx)
        
    return (best_h, best_C, n_points)
