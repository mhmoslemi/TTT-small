import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Project to feasible region
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Better optimization using scipy minimize with bounds
    def optimize_step(h_vec, max_iter=50):
        def objective(h_new):
            return compute_C5(h_new)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            objective,
            h_vec,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-8},
            tol=1e-8
        )
        
        if result.success:
            return project_feasible(result.x)
        return h_vec

    # Evolve solution with mutation and selection
    def evolve_population(population, fitnesses, mutation_rate=0.1):
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        best_indices = sorted_indices[:len(population)//2]
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(len(population)):
            parent = population[best_indices[i % len(best_indices)]]
            # Mutation
            child = parent.copy()
            for j in range(len(child)):
                if np.random.random() < mutation_rate:
                    delta = np.random.normal(0, 0.02)
                    child[j] = np.clip(child[j] + delta, 0, 1)
            
            # Project to feasible region
            child = project_feasible(child)
            new_population.append(child)
        
        return new_population

    # Main optimization loop
    population_size = 8
    population = [project_feasible(h + np.random.normal(0, 0.01, n_points)) 
                  for _ in range(population_size)]
    
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        # Evaluate fitness
        fitnesses = [compute_C5(h_vec) for h_vec in population]
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_C5:
            best_C5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Reset with new random solutions
                population = [project_feasible(h + np.random.normal(0, 0.05, n_points)) 
                             for _ in range(population_size)]
                patience = 0
        
        # Evolve population
        population = evolve_population(population, fitnesses)
        
        # Apply local optimization to top individuals
        for i in range(len(population) // 2):
            population[i] = optimize_step(population[i])
        
        iter_count += 1

    # Final evaluation
    final_fitness = compute_C5(best_h)
    if final_fitness < best_C5:
        best_C5 = final_fitness
        
    return (best_h.tolist(), best_C5, len(best_h))
