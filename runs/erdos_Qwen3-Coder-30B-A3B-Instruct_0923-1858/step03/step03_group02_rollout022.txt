        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    best_C5 = float('inf')

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization
    n_particles = 50
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_w = 2.0
    social_w = 2.0
    velocity_clamp = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # Base particle
    particles.append(h.copy())
    velocities.append(np.zeros_like(h))
    
    # Perturbed particles
    for i in range(1, n_particles):
        perturbation = np.random.normal(0, 0.05, n_points)
        h_pert = h + perturbation
        h_pert = project_feasible(h_pert)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Track personal bests
    p_best = [p.copy() for p in particles]
    p_best_fitness = [compute_C5(p) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(p_best_fitness)
    s_best = p_best[global_best_idx].copy()
    s_best_fitness = p_best_fitness[global_best_idx]
    
    # Update loop
    max_patience = 30
    patience = 0
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        # Update inertia linearly
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / 1000)
        if iter_count > 1000:
            inertia = inertia_end
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)

                    cognitive_w * r1 * (p_best[i] - particles[i]) + 
                    social_w * r2 * (s_best - particles[i]))

            v_new = np.clip(v_new, -velocity_clamp, velocity_clamp)
            
            # Update position
            h_new = particles[i] + v_new
            
            # Project into feasible region
            h_new = project_feasible(h_new)
            
            # Evaluate fitness
            fitness = compute_C5(h_new)

                p_best[i] = h_new.copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < s_best_fitness:
                    s_best = h_new.copy()
                    s_best_fitness = fitness
                    patience = 0
                    
            # Store for next iteration
            particles[i] = h_new
            velocities[i] = v_new
            
        # Check for stagnation
        patience += 1
        if patience > max_patience:
            for i in range(n_particles):
                # Add small perturbation to escape local minimum
                perturbation = np.random.normal(0, 0.03, n_points)
                particles[i] += perturbation
                particles[i] = project_feasible(particles[i])
            patience = 0

        # Periodic adjustments

            # Occasionally sample the best result
            current_fitness = compute_C5(s_best)
            if current_fitness < s_best_fitness:
                s_best_fitness = current_fitness

    # Final check
    final_C5 = compute_C5(s_best)
    if final_C5 < s_best_fitness:
        s_best_fitness = final_C5
    
    return (s_best.tolist(), s_best_fitness, len(s_best))
