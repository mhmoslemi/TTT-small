    best_h = h.copy()
    best_C5 = float('inf')

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_analytical_gradients(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        # Calculate gradient for max_k lag
        # Each term is (1 - h[i + max_k]) + (h[i - max_k]) for valid indices
        # but we have to be careful with indexing due to convolution behavior
        
        # For i in 0..n-1 and (i + max_k) within valid range
        for i in range(len(h_vec)):
            # First term: derivative of ((1-h_i) * h_j) where j = i + max_k
            if i + max_k >= 0 and i + max_k < len(h_vec):
                grad[i] += (1 - h_vec[i + max_k])
            
            # Second term: derivative of (h_i * (1-h_j)) where j = i - max_k
            if i - max_k >= 0 and i - max_k < len(h_vec):
                grad[i] += -h_vec[i - max_k]
                
        return grad

    # Trust region update with analytical gradients
    def trust_region_update(h_vec, eta=0.01, max_iter=10):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            grad = compute_analytical_gradients(h_new)
            h_candidate = h_new - eta * grad
            h_candidate = project_feasible(h_candidate)
            
            # Simple backtracking for decrease in C5
            temp_h = h_new.copy()
            while compute_C5(h_candidate) > compute_C5(temp_h) and eta > 1e-6:
                eta *= 0.5
                h_candidate = temp_h - eta * grad
                h_candidate = project_feasible(h_candidate)
            
            h_new = h_candidate
        return h_new

    # Main optimization loop

    patience = 0
    max_patience = 50

                    if np.random.random() < 0.1:

                h = project_feasible(h)

        # Trust region update with analytical gradients
        h = trust_region_update(h, eta=eta)

        if iter_count % 50 == 0 and iter_count > 0:

            # Re-project to maintain feasibility
            h = project_feasible(h)
        
        iter_count += 1

    # Final evaluation

    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
