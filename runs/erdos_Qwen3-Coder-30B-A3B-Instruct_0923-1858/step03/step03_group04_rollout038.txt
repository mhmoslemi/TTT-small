import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Simple convex relaxation approach - minimize sum of squares
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    def frequency_domain_update(h_vec, learning_rate=0.01):
        # FFT to get frequency domain representation
        H = np.fft.fft(h_vec)
        H_conj = np.conj(H)
        # Compute correlation in frequency domain
        corr_freq = H * H_conj
        # IFFT to get back to spatial domain
        corr_spatial = np.fft.ifft(corr_freq).real
        
        # Find maximum shift index
        max_k_idx = np.argmax(corr_spatial)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute the gradient direction for simplification
        # Approximate by using finite differences of the main correlation
        # This is an estimation of how to adapt the frequency domain representation
        
        # Simple frequency-domain update: decrease concentration around max_k
        if max_k != 0:
            freq_shift = np.exp(-2j * np.pi * np.arange(len(h_vec)) * max_k / len(h_vec))
            H_update = H * freq_shift
        else:
            H_update = H
        
        # Apply update and inverse FFT to get new spatial domain
        H_new = H - learning_rate * H_update
        h_new = np.fft.ifft(H_new).real
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        return h_new

    def trust_region_update(h_vec, eta=0.01, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            h_new = frequency_domain_update(h_new, eta)
        return h_new

    # Try convex relaxation first to get a good warm start
    if elapsed() < budget_s * 0.3:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass
    
    # Main optimization loop
    eta = 0.01
    patience = 0
    max_patience = 50
    iter_count = 0
    last_best_update = 0
    prev_C5 = float('inf')

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Check if we're stuck
        if patience > max_patience or (iter_count - last_best_update > 100):
            # Try a more drastic change
            h = project_feasible(h + np.random.normal(0, 0.05, len(h)))
            patience = 0

        # Update with frequency domain approach
        h = trust_region_update(h, eta)
        
        # Adaptive shrinkage of learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            eta = max(0.001, eta * 0.95)
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
