import time

def run(seed=42, budget_s=600, **kwargs):

        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

        h_vec = np.array(h_vec)

        corr = correlate(h_vec, 1 - h_vec, mode="full")

        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Actual function value and gradient in one pass

        # Gradient computation - simplified finite difference
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        h_plus = h_vec.copy()
        h_minus = h_vec.copy()
