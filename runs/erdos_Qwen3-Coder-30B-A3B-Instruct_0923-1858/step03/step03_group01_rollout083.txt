    best_C = float('inf')

    # Use a combination of gradient-based optimization with projection
    # and discrete search to jump out of bad local minima

    for i in range(max_iter):
        if time.time() - t0 > budget_s - 1:
            break
            
        # Compute overlap using full correlation
        corr = correlate(h_current, 1 - h_current, mode='full')
        current_C = np.max(corr) * dx
        
        if current_C < best_C:
            best_C = current_C
            best_h = h_current.copy()
            
        # Simple gradient estimation for update
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (1.0 / (h_plus_sum * dx))
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus_sum = np.sum(h_minus)
            if h_minus_sum > 0:
                h_minus = h_minus * (1.0 / (h_minus_sum * dx))
                
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Adaptive learning rate
        lr = 0.01 * (1.0 - i / max_iter) * (1 + 0.1 * np.random.random())
        h_current = h_current - lr * grad

        h_sum = np.sum(h_current)
        if h_sum > 0:
            h_current = h_current * (1.0 / (h_sum * dx))
    
    # Final evaluation to make sure we have the correct value
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_C = np.max(corr) * dx
    
    return (best_h, final_C, n_points)
