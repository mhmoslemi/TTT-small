import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        # Create a piecewise linear function with a local minimum at center
        for i in range(n_points):
            if x[i] < 0.5:
                h[i] = 2 * x[i]
            elif x[i] > 1.5:
                h[i] = 2 * (2 - x[i])
            else:
                h[i] = 1
        # Normalize to integral = 1
        integral = np.trapz(h, x)
        h = h / integral * 1.0

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec) * dx
        if current_sum > 0:
            h_vec = h_vec * (1.0 / current_sum)
        return h_vec

    def compute_gradients(h_vec, eps=1e-5):
        current_C5 = compute_C5(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            grad[i] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
        return grad

    # Optimization using scipy minimize
    def optimize_step(h_vec, iterations=100):
        def objective(x):
            return compute_C5(x)
        
        def constraint_func(x):
            return np.sum(x) * dx - 1.0
        
        constraints = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(len(h_vec))]
        
        result = minimize(objective, h_vec, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': iterations})
        return project_feasible(result.x) if result.success else h_vec

    def local_search(h_vec, num_iters=50):
        h_new = h_vec.copy()
        for _ in range(num_iters):
            grad = compute_gradients(h_new, eps=1e-4)
            h_new = h_new - 0.01 * grad
            h_new = project_feasible(h_new)
        return h_new

    # Multi-start optimization with structured perturbations
    max_patience = 100
    patience = 0
    iter_count = 0
    eta = 0.02

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        if patience > max_patience:
            # Perturb with noise
            noise_magnitude = 0.05
            h = np.clip(h + np.random.normal(0, noise_magnitude, n_points), 0, 1)
            h = project_feasible(h)
            patience = 0

        # Local optimization step
        h_opt = optimize_step(h, iterations=20)
        h = local_search(h_opt, num_iters=30)
        
        # Adaptive learning rate
        if iter_count % 100 == 0:
            eta *= 0.99
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
