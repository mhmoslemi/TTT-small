import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def optimize_in_frequency_domain():
        h_freq = fft(h)
        learning_rate = 0.01
        max_iter = 500
        patience = 0
        last_improvement = 0
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            # Convert back to time domain
            h_time = ifft(h_freq).real
            # Project to feasible region
            h_time = project_feasible(h_time)
            # Transform back
            h_freq = fft(h_time)
            
            # Compute gradient in frequency domain (approximately)
            # This is a simplification - compute gradient based on correlation
            corr = correlate(h_time, 1 - h_time, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_time) - 1)
            
            # Adjust in frequency domain based on correlation peak
            # Approximate gradient direction
            grad = np.zeros_like(h_freq)
            eps = 1e-5
            
            # Perturb each freq component slightly
            for j in range(len(h_freq)//2):
                h_freq_pert = h_freq.copy()
                h_freq_pert[j] += eps
                h_time_pert = ifft(h_freq_pert).real
                h_time_pert = project_feasible(h_time_pert)
                corr_pert = correlate(h_time_pert, 1 - h_time_pert, mode="full")
                grad[j] += (corr_pert[max_k_idx] - corr[max_k_idx]) / eps
                
            # Update frequency domain
            h_freq = h_freq - learning_rate * grad
            
            # Reconstruct and check
            h_time = ifft(h_freq).real
            h_time = project_feasible(h_time)
            current_C5 = compute_C5(h_time)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_time.copy()
                patience = 0
                last_improvement = i
            else:
                patience += 1
                if patience > 30:
                    # Backtrack and reduce learning rate
                    learning_rate *= 0.9
                    patience = 0
            if i - last_improvement > 100:
                break

    # Run the main optimization
    optimize_in_frequency_domain()
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
