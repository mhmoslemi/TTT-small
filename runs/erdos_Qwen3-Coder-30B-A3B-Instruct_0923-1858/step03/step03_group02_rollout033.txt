import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - 1.5 * np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # QP-based refinement
    def refine_with_qp(h_vec):
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        constraints = [
            0 <= h_new, h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0
        ]
        
        # Find the most constraining overlap
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        k_max = np.argmax(corr) - (len(h_vec) - 1)
        
        # Linearize the constraint at current h
        h_current = np.array(h_vec)
        if k_max >= 0:
            grad = np.array([1 - h_current[i + k_max] - h_current[i - k_max] 
                             if i >= k_max and i + k_max < len(h_current) 
                             else 0 for i in range(len(h_current))])
            # Adjust for proper indices when k_max < 0
            if k_max < 0:
                grad = np.array([1 - h_current[i + k_max] - h_current[i - k_max] 
                                 if i + k_max >= 0 and i - k_max < len(h_current) 
                                 else 0 for i in range(len(h_current))])
        else:
            grad = np.array([1 - h_current[i + k_max] - h_current[i - k_max] 
                             if i + k_max >= 0 and i - k_max < len(h_current) 
                             else 0 for i in range(len(h_current))])
        
        # Add the linearized constraint
        constraint_expr = t >= np.dot(grad, h_new - h_current)
        constraints.append(constraint_expr)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                sol = h_new.value
                if sol is not None:
                    return project_feasible(sol)
        except:
            pass
        return h_vec

    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = project_feasible(h + np.random.normal(0, 0.05, n_points) * 0.1)
                patience = 0

        h_new = refine_with_qp(h)
        if compute_C5(h_new) < compute_C5(h):
            h = h_new
        else:
            h = project_feasible(h)

        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
