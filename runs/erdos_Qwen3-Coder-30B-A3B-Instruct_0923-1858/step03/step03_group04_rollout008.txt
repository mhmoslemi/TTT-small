import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1

    # Initialize particles
    particles = [h.copy() for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = [h.copy() for _ in range(n_particles)]
    pbest_values = [compute_C5(h) for h in particles]
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    # PSO main loop
    iter_count = 0
    while elapsed() < budget_s * 0.6 and iter_count < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            current_value = compute_C5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value

        iter_count += 1

    # Local refinement with L-BFGS-B
    def objective(h_vec):
        return compute_C5(h_vec)

    def jac(h_vec):
        # Gradient approximation using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad

    # L-BFGS-B optimization
    bounds = [(0, 1) for _ in range(n_points)]
    refined_result = minimize(objective, gbest, method='L-BFGS-B',
                             bounds=bounds, jac=jac, 
                             options={'maxiter': int(0.4 * budget_s / 10), 
                                      'ftol': 1e-8, 'gtol': 1e-6})

    # Refine with projected solution
    refined_h = project_feasible(refined_result.x)
    refined_C5 = compute_C5(refined_h)

    # Final comparison
    if refined_C5 < gbest_value:
        final_h = refined_h
        final_C5 = refined_C5
    else:
        final_h = gbest
        final_C5 = gbest_value

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
