import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # PSO parameters
    n_particles = 50
    max_iter = 2000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Generate other particles with noise
    for _ in range(n_particles - 1):
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = h + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Track personal bests and global best
    p_best_positions = [p.copy() for p in particles]
    p_best_fitness = [float('inf')] * n_particles
    g_best_position = best_h.copy()
    g_best_fitness = best_C5
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Main PSO loop
    iter_count = 0
    no_improvement_count = 0
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (p_best_positions[i] - particles[i])
            social = social_weight * r2 * (g_best_position - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project into feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best_fitness[i] = fitness
                p_best_positions[i] = particles[i].copy()
            
            # Update global best
            if fitness < g_best_fitness:
                g_best_fitness = fitness
                g_best_position = particles[i].copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Perturb particles if no improvement for too long
        if no_improvement_count > 30:
            for i in range(n_particles):
                pert = np.random.normal(0, 0.03, n_points)
                particles[i] = particles[i] + pert
                particles[i] = project_feasible(particles[i])
            no_improvement_count = 0
            
        iter_count += 1
    
    # Final best
    final_C5 = g_best_fitness
    best_h = g_best_position.copy()
    
    return (best_h.tolist(), final_C5, len(best_h))
