import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    dx = 2.0 / n_points

        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.mean(initial_h) * (n_points / 2)
    
    # Simulated annealing parameters
    T_start = 0.1
    alpha = 0.95
    T_min = 1e-6
    current_T = T_start
    
    # Initial solution
    h_current = initial_h.copy()
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # Iteration counter
    it = 0
    while time.time() - t0 < max_time:
        # Decrease temperature
        if it % 500 == 0 and current_T > T_min:
            current_T *= alpha
            
        # Calculate current scale adaptation - decreases with temperature
        scale = max(0.01, current_T * 0.5)
        
        # Generate neighbor: Gaussian perturbation
        delta = np.random.normal(0, scale, n_points)
        h_candidate = h_current + delta
        
        # Project to feasible set
        h_candidate = project(h_candidate, dx)
        
        # Compute C5 for candidate
        current_C = evaluate(h_candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        if current_C < best_C:
            # Accept improvement
            h_current = h_candidate.copy()
            if current_C < best_C:
                best_C = current_C
                best_h = h_candidate.copy()
        else:
            # Accept with probability based on difference and temperature
            delta_C = current_C - best_C
            prob = np.exp(-delta_C / current_T)
            if np.random.rand() < prob:
                h_current = h_candidate.copy()
                
        it += 1

    # Final projection and evaluation
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    # Refinement with gradient descent
    t_refine_start = time.time()
    lr = 0.01
    decay_interval = 1000
    decay_factor = 0.95
    
    h_refined = final_h.copy()
    best_refined_C = final_C
    
    while time.time() - t_refine_start < 30 and time.time() - t0 < max_time - 10:
        # Compute gradient
        grad = np.zeros_like(h_refined)
        eps = 1e-6

        # Gradient step
        h_new = h_refined - lr * grad
        h_new = project(h_new, dx)

        h_refined = h_new.copy()
        
        # Decay learning rate
        if it % decay_interval == 0:
            lr *= decay_factor
            
        it += 1
    
    return (h_refined, best_refined_C, n_points)
