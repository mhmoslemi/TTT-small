import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        for h in population:
            C = evaluate(h, dx)

        sorted_indices = np.argsort(fitnesses)[::-1][:population_size//2]
        selected = [population[i] for i in sorted_indices]
        
        # Generate new population
        new_population = selected.copy()
        while len(new_population) < population_size:
            # Tournament selection
            a_idx = np.random.choice(len(selected))
            b_idx = np.random.choice(len(selected))
            parent_a = selected[a_idx]
            parent_b = selected[b_idx]

                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent_a[:crossover_point], parent_b[crossover_point:]])
                
                # Ensure it has the right number of ones
                ones_count = np.sum(child)
                if ones_count != target_ones:
                    # Fix the count by flipping some bits
                    flips_needed = target_ones - ones_count
                    if flips_needed > 0:
                        zero_indices = np.where(child == 0)[0]
                        if len(zero_indices) >= flips_needed:
                            flip_indices = np.random.choice(zero_indices, flips_needed, replace=False)
                            child[flip_indices] = 1
                    else:
                        one_indices = np.where(child == 1)[0]
                        if len(one_indices) >= -flips_needed:
                            flip_indices = np.random.choice(one_indices, -flips_needed, replace=False)

            else:

                child = parent_a.copy()

            for i in range(n_points):

            # Ensure correct count
            ones_count = np.sum(child)

                    # Remove excess ones
                    one_indices = np.where(child == 1)[0]
                    to_flip = ones_count - target_ones

                    to_flip = target_ones - ones_count

                best_h = h.copy()
                stagnant_count = 0

                stagnant_count += 1
                if stagnant_count > 100:
                    break
    
    return (best_h, best_C, n_points)
