import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Start with a modified Frank-Wolfe approach  
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Minimize sum of squared values to get spread out solution
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    # Improving gradient estimation
    def compute_gradient(h_vec, num_samples=10):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        # Take small perturbations based on random sampling
        grad = np.zeros_like(h_vec)
        eps = 1e-3
        
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    # Use gradient descent with momentum
    def optimize_with_momentum(h_vec, lr=0.02, momentum=0.9, max_iter=100):
        h_new = h_vec.copy()
        velocity = np.zeros_like(h_vec)
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            grad = compute_gradient(h_new)
            velocity = momentum * velocity - lr * grad
            h_update = h_new + velocity
            h_new = project_feasible(h_update)
        return h_new

    # Apply trust region style update
    def trust_region_update(h_vec, eta=0.01, max_iter=5):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = compute_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Utility to get a better starting point
    if elapsed() < budget_s * 0.1:
        try:
            relaxed_h = solve_convex_relaxation()
            h = relaxed_h.copy()
        except:
            pass

    # Main loop with broader exploration
    iter_count = 0
    patience = 0
    max_patience = 30
    last_improvement = 0
    eta = 0.02
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_improvement = iter_count
        else:
            patience += 1

        # Perturb and restart when stuck
        if patience > max_patience or (iter_count - last_improvement > 30):
            # Random perturbation of existing solution
            pert_h = h.copy() + np.random.normal(0, 0.02, n_points)
            h = project_feasible(pert_h)
            patience = 0
            if iter_count - last_improvement > 100:
                # Full restart
                try:
                    relaxed_h = solve_convex_relaxation()
                    if compute_C5(relaxed_h) < compute_C5(h):
                        h = relaxed_h.copy()
                except:
                    pass

        # Adaptive learning rate decay
        if iter_count % 20 == 0 and iter_count > 0:
            eta = max(0.001, eta * 0.95)

        h = optimize_with_momentum(h, lr=eta, momentum=0.9, max_iter=3)
        iter_count += 1

    # Final refinement
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
