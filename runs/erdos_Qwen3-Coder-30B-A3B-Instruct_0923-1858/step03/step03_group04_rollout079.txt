import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_constraint_violation(h_vec):
        # Return the absolute violation of the constraint sum(h) = n_points/2
        return abs(np.sum(h_vec) - n_points / 2)

    # Particle Swarm Optimization for global exploration
    def pso_optimize():
        n_particles = 20
        max_iter = 500
        w = 0.7  # inertia weight
        c1 = c2 = 1.5  # cognitive and social parameters
        max_vel = 0.1
        
        # Initialize particles
        particles = []
        velocities = []
        best_positions = []
        best_scores = []
        
        for _ in range(n_particles):
            # Random feasible particle
            h_init = np.random.rand(n_points)
            h_init = project_feasible(h_init)
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
            best_positions.append(h_init.copy())
            best_scores.append(compute_C5(h_init))
        
        global_best = best_positions[np.argmin(best_scores)].copy()
        global_best_score = min(best_scores)
        
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.5:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (best_positions[i] - particles[i]) +
                                c2 * r2 * (global_best - particles[i]))
                
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
                
                # Update position
                particles[i] += velocities[i]
                
                # Repair feasibility
                particles[i] = project_feasible(particles[i])
                
                # Evaluate fitness
                score = compute_C5(particles[i])
                
                # Update personal best
                if score < best_scores[i]:
                    best_scores[i] = score
                    best_positions[i] = particles[i].copy()
                    
                    # Update global best
                    if score < global_best_score:
                        global_best_score = score
                        global_best = particles[i].copy()
        
        return global_best, global_best_score

    # Local refinement with L-BFGS-B
    def refine_with_lbfgsb(initial_h):
        def objective(h_vec):
            return compute_C5(h_vec)
            
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points/2
            
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a small subset of points for the optimization for speed
        # but do it on the full vector
        result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, 
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-5})
        
        return result.x if result.success else initial_h

    # Run optimization phases
    # Phase 1: PSO for global search
    try:
        pso_h, pso_C5 = pso_optimize()
        if pso_C5 < best_C5:
            best_C5 = pso_C5
            best_h = pso_h.copy()
    except:
        pass

    # Phase 2: Local refinement
    try:
        refined_h = refine_with_lbfgsb(best_h)
        refined_C5 = compute_C5(refined_h)
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
    except:
        pass
    
    # A few local search steps to fine-tune
    for _ in range(100):
        if elapsed() > budget_s * 0.98:
            break
            
        # Random perturb and project
        h_perturbed = best_h + np.random.normal(0, 1e-3, n_points)
        h_perturbed = project_feasible(h_perturbed)
        
        new_C5 = compute_C5(h_perturbed)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_perturbed.copy()

    return (best_h.tolist(), best_C5, len(best_h))
