import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Initialize with triangular function
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    inertia_decay = 0.99

    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize with current solution
    swarm.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h.copy())
    pbest_vals.append(compute_C5(h))
    
    # Add random variations
    for _ in range(swarm_size - 1):
        pert_h = h + np.random.normal(0, 0.02, n_points)
        pert_h = project_feasible(pert_h)
        swarm.append(pert_h)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(pert_h.copy())
        pbest_vals.append(compute_C5(pert_h))

    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]

    # PSO main loop
    iter_count = 0
    stagnation_count = 0
    best_improvement = 1e10
    
    while elapsed() < budget_s * 0.6 and iter_count < max_iter:
        if stagnation_count > 20:
            # Perturb particles to escape local minima
            for i in range(swarm_size):
                if np.random.rand() < 0.3:
                    swarm[i] += np.random.normal(0, 0.03, n_points)
                    swarm[i] = project_feasible(swarm[i])
                    new_val = compute_C5(swarm[i])
                    if new_val < pbest_vals[i]:
                        pbest[i] = swarm[i].copy()
                        pbest_vals[i] = new_val
                        if new_val < gbest_val:
                            gbest = swarm[i].copy()
                            gbest_val = new_val
                            best_improvement = 0
            stagnation_count = 0
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            new_val = compute_C5(swarm[i])
            if new_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = new_val
                
                if new_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = new_val
                    best_improvement = 0
                    stagnation_count = 0
            else:
                stagnation_count += 1

        if best_improvement < 1e-5:
            stagnation_count = 0
        best_improvement = min(best_improvement, gbest_val)
        w *= inertia_decay
        iter_count += 1

    # Local refinement using L-BFGS
    def lbfgs_objective(h_vec):
        return compute_C5(h_vec)

    def lbfgs_constraint(h_vec):
        return np.sum(h_vec) - n_points/2

    # Prepare bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lbfgs_constraint}

    # Local search with L-BFGS-B on the best PSO solution
    try:
        result = minimize(
            lbfgs_objective,
            gbest,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-6, 'gtol': 1e-5, 'maxiter': 300}
        )
        
        if result.success:
            refined_sol = result.x
        else:
            refined_sol = gbest.copy()
    except:
        refined_sol = gbest.copy()

    # Final evaluation
    final_C5 = compute_C5(refined_sol)
    
    if final_C5 < gbest_val:
        gbest_val = final_C5
        gbest = refined_sol.copy()

    return (gbest.tolist(), gbest_val, n_points)
