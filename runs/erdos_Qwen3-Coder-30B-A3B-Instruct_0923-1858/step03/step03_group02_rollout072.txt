import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    elapsed = lambda: time.time() - start_time

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Create initial particles
    swarm.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add perturbed particles
    for i in range(1, swarm_size):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        particle = project_feasible(perturbed)
        swarm.append(particle)
        velocities.append(np.random.uniform(-0.02, 0.02, n_points))
    
    # Track personal bests
    p_best = [swarm[i].copy() for i in range(swarm_size)]
    p_best_fitness = [compute_C5(swarm[i]) for i in range(swarm_size)]
    
    # Track global best
    global_best_idx = np.argmin(p_best_fitness)
    s_best = p_best[global_best_idx].copy()
    s_best_fitness = p_best_fitness[global_best_idx]
    
    # PSO iteration
    for iter_count in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (p_best[i] - swarm[i])
            social = social_weight * r2 * (s_best - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            new_h = swarm[i] + velocities[i]
            
            # Project to feasible region
            new_h = project_feasible(new_h)
            
            # Evaluate new fitness
            new_fitness = compute_C5(new_h)
            
            # Update personal best if better
            if new_fitness < p_best_fitness[i]:
                p_best[i] = new_h.copy()
                p_best_fitness[i] = new_fitness
                
                # Update global best if better
                if new_fitness < s_best_fitness:
                    s_best = new_h.copy()
                    s_best_fitness = new_fitness
                    
            # Update particle position
            swarm[i] = new_h
            
        # Every 30 iterations, perturb particles if no improvement
        if iter_count % 30 == 0 and iter_count > 0:
            for i in range(1, swarm_size):  # Skip first particle
                if np.random.random() < 0.3:
                    perturbed = swarm[i] + np.random.normal(0, 0.03, n_points)
                    swarm[i] = project_feasible(perturbed)
                    # Update personal best if improved
                    new_fitness = compute_C5(swarm[i])
                    if new_fitness < p_best_fitness[i]:
                        p_best[i] = swarm[i].copy()
                        p_best_fitness[i] = new_fitness
                        if new_fitness < s_best_fitness:
                            s_best = swarm[i].copy()
                            s_best_fitness = new_fitness

    # Final return with best solution
    return (s_best.tolist(), s_best_fitness, len(s_best))
