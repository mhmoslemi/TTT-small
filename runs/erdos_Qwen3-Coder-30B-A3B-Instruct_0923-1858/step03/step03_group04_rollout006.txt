import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1

    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_C5 = []
    
    # Use initial solution as global best
    global_best = h.copy()
    global_best_C5 = compute_C5(global_best)
    
    for i in range(swarm_size):
        if i == 0:
            h_p = h.copy()
        else:
            # Add small noise to initial solution
            h_p = h + np.random.normal(0, 0.02, n_points)
        h_p = project_feasible(h_p)
        particles.append(h_p.copy())
        velocities.append(np.zeros(n_points))
        
        # Evaluate personal best
        p_C5 = compute_C5(h_p)
        personal_best.append(h_p.copy())
        personal_best_C5.append(p_C5)
        
        # Update global best if needed
        if p_C5 < global_best_C5:
            global_best = h_p.copy()
            global_best_C5 = p_C5

    # PSO main loop
    for iter_idx in range(max_iter):
        if elapsed() > budget_s * 0.6:
            break

        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate current particle
            current_C5 = compute_C5(particles[i])
            
            # Update personal best
            if current_C5 < personal_best_C5[i]:
                personal_best_C5[i] = current_C5
                personal_best[i] = particles[i].copy()
                
                # Update global best if needed
                if current_C5 < global_best_C5:
                    global_best = particles[i].copy()
                    global_best_C5 = current_C5

    # Local refinement with L-BFGS-B
    def objective(h_vec):
        return compute_C5(h_vec)

    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Use global best as starting point for local refinement
    h_refined = global_best.copy()
    
    # Define bounds and constraints for L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Try L-BFGS-B refinement with hot start
    if elapsed() < budget_s * 0.95:
        try:
            result = minimize(
                objective,
                h_refined,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-5, 'gtol': 1e-5}
            )
            if result.success:
                h_refined = project_feasible(result.x)
            else:
                h_refined = project_feasible(h_refined)
        except:
            h_refined = project_feasible(h_refined)

    final_C5 = compute_C5(h_refined)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = h_refined.copy()

    # Final validation
    best_h = project_feasible(best_h)

    return (best_h.tolist(), best_C5, len(best_h))
