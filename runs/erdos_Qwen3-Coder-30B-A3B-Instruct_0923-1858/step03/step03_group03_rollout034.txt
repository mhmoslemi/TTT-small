import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use a trust region like approach
    def trust_region_update(h_old, max_iter=30):
        def objective(h_vec):
            h_vec = project_feasible(h_vec)
            return compute_C5(h_vec)

        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2

        cons = [{'type': 'eq', 'fun': constraint_sum}]

        # Use scipy's minimize with trust-constr
        result = minimize(objective, h_old, method='trust-constr', 
                          constraints=cons, bounds=[(0, 1) for _ in range(n_points)], 
                          options={'maxiter': max_iter})
        
        return project_feasible(result.x)

    # Simple mutation approach
    def perturb_solution(h_vec, strength=0.05):
        h_new = h_vec.copy()
        num_mutations = max(1, int(0.1 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_mutations, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    # Main optimization loop
    convergence_patience = 0
    max_patience = 100
    iter_count = 0
    learning_rate = 0.02
    eta = learning_rate

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            convergence_patience = 0
        else:
            convergence_patience += 1
            if convergence_patience > max_patience:
                h = perturb_solution(h, 0.02)
                convergence_patience = 0

        # Use trust region with adaptive steps
        h = trust_region_update(h, max_iter=20)
        
        # Adapt learning rate
        if iter_count % 30 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1

    # Final refinement
    final_h = trust_region_update(best_h, max_iter=50)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
