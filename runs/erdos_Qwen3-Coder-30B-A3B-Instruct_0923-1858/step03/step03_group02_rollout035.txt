import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Optimize using scipy with bounds
    def optimize_step(h_init):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            return compute_C5(h_vec)
        
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        result = minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            return project_feasible(result.x)
        return project_feasible(h_init)
    
    # Local search with random perturbations
    def local_search(h_vec, num_trials=10):
        best_h = h_vec.copy()
        best_C5 = compute_C5(h_vec)
        
        for _ in range(num_trials):
            h_trial = h_vec.copy()
            # Perturb a few points
            for i in range(n_points):
                if np.random.random() < 0.15:
                    h_trial[i] += np.random.normal(0, 0.02)
            
            h_trial = project_feasible(h_trial)
            trial_C5 = compute_C5(h_trial)
            
            if trial_C5 < best_C5:
                best_C5 = trial_C5
                best_h = h_trial
        
        return best_h, best_C5
    
    # Main optimization loop
    patience = 0
    max_patience = 50
    iter_count = 0

        if current_C5 < best_C5:

            best_h = h.copy()
            patience = 0

                h = np.random.uniform(0, 1, n_points)

            h = project_feasible(h)
        
        iter_count += 1

    # Final check

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
