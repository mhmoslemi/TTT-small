import time

        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Limited-memory BFGS-like update with adaptive learning rate
    def update_h(h_vec, grad, eta=0.02):
        h_new = h_vec - eta * grad
        h_new = project_feasible(h_new)
        return h_new

    # Compute gradient using finite differences
    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        h_ref = h_vec.copy()
        current_corr = correlate(h_ref, 1 - h_ref, mode="full")
        max_k_idx = np.argmax(current_corr)
        max_k = max_k_idx - (len(h_ref) - 1)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    def trust_region_update(h_vec, grad, trust_radius=0.1, max_iter=50):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            try:
                h_candidate = h_new - grad
                h_candidate = project_feasible(h_candidate)
                
                # Ensure step size not too large for trust region
                dist_to_new = np.linalg.norm(h_candidate - h_vec)
                if dist_to_new > trust_radius:
                    h_candidate = h_vec + (h_candidate - h_vec) * trust_radius / dist_to_new

                if compute_C5(h_candidate) < compute_C5(h_new):

            except:

        iter_count += 1

    # Final evaluation

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
