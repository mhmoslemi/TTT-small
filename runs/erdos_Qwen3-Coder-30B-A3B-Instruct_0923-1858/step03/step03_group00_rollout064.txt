import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def crossover(parent1, parent2):
    """Perform one-point crossover between two parents."""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05, mutation_scale=0.1):
    """Mutate the individual with Gaussian noise."""
    n = len(h)
    mask = np.random.rand(n) < mutation_rate
    if not np.any(mask):
        idx = np.random.randint(0, n)
        mask[idx] = True
    
    h_mut = h.copy()
    h_mut[mask] += np.random.normal(0, mutation_scale, np.sum(mask))
    h_mut = np.clip(h_mut, 0, 1)
    
    return h_mut

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]
