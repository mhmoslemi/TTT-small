import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive = 2.0
    social = 2.0
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_fitness.append(compute_C5(particle))
    
    # Global best
    s_best_idx = np.argmin(p_best_fitness)
    s_best = p_best[s_best_idx].copy()
    s_best_fitness = p_best_fitness[s_best_idx]
    
    if s_best_fitness < best_C5:
        best_C5 = s_best_fitness
        best_h = s_best.copy()
    
    # PSO main loop
    max_iter = 1000
    no_improvement_count = 0
    max_no_improvement = 30
    
    for iter_num in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        # Update inertia linearly
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive_component = cognitive * r1 * (p_best[i] - particles[i])
            social_component = social * r2 * (s_best - particles[i])
            velocities[i] = (inertia * velocities[i] + 
                           cognitive_component + 
                           social_component)
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < s_best_fitness:
                    s_best = particles[i].copy()
                    s_best_fitness = fitness
                    no_improvement_count = 0
                    
                    if s_best_fitness < best_C5:
                        best_C5 = s_best_fitness
                        best_h = s_best.copy()
            else:
                no_improvement_count += 1
        
        # Perturb particles if no improvement
        if no_improvement_count > max_no_improvement:
            for i in range(n_particles):
                perturbation = np.random.normal(0, 0.03, n_points)
                particles[i] = particles[i] + perturbation
                particles[i] = project_feasible(particles[i])
                fitness = compute_C5(particles[i])
                if fitness < p_best_fitness[i]:
                    p_best[i] = particles[i].copy()
                    p_best_fitness[i] = fitness
                    if fitness < s_best_fitness:
                        s_best = particles[i].copy()
                        s_best_fitness = fitness
            no_improvement_count = 0
            
        if iter_num % 50 == 0:
            # Periodic check of global best
            s_best_idx = np.argmin(p_best_fitness)
            if p_best_fitness[s_best_idx] < s_best_fitness:
                s_best = p_best[s_best_idx].copy()
                s_best_fitness = p_best_fitness[s_best_idx]
                
                if s_best_fitness < best_C5:
                    best_C5 = s_best_fitness
                    best_h = s_best.copy()

    # Final check
    final_fitness = compute_C5(best_h)
    if final_fitness < best_C5:
        best_C5 = final_fitness
        
    # Final projection
    best_h = project_feasible(best_h)
    final_fitness = compute_C5(best_h)
    
    return (best_h.tolist(), final_fitness, len(best_h))
