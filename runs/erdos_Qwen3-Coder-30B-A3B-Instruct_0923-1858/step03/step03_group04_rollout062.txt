        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Transform h to frequency domain: H = FFT(h)
    # The overlap integral becomes: ifft(H * conj(H)) 
    # We minimize the maximum of this in the time domain
    
    def compute_frequency_domain_gradient(h_vec, H):
        # Compute gradient in frequency domain
        # Use the fact that correlation is convolution and in Fourier domain it's product
        # So we want to minimize ||fft(h * (1-h))||_inf
        # Using a smooth approximation for numerical stability
        # gradients based on log barrier or similar
        # For now, we do it by dual ascent (simple)
        return np.zeros_like(h_vec)

    eta = 0.01
    max_iter = 200
    patience = 0
    max_patience = 50

        if elapsed() > budget_s * 0.95:
            break
        
        # Update in frequency domain
        h_freq = fft(h)
        
        # The overlap integral is equivalent to conv(h, 1-h)

            h_plus = project_feasible(h + pert)
            h_minus = project_feasible(h - pert)

            patience = 0
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
