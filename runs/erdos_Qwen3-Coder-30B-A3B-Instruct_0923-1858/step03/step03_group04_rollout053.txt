import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization
    def pso_optimization():
        # PSO parameters
        n_particles = 20
        max_iter = 1000
        w = 0.7  # inertia weight
        c1 = c2 = 1.5  # cognitive and social parameters
        v_max = 0.1  # max velocity
        
        # Initialize particles
        particles = []
        velocities = []
        personal_best = []
        personal_best_vals = []
        global_best = None
        global_best_val = float('inf')
        
        for i in range(n_particles):
            # Initialize particle with small random perturbations
            h_init = project_feasible(h + np.random.normal(0, 0.02, n_points))
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            p_best = h_init.copy()
            p_best_val = compute_C5(h_init)
            personal_best.append(p_best)
            personal_best_vals.append(p_best_val)
            if p_best_val < global_best_val:
                global_best = p_best.copy()
                global_best_val = p_best_val

        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.6:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best[i] - particles[i]) + 
                                c2 * r2 * (global_best - particles[i]))
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                particles[i] += velocities[i]
                
                # Repair infeasibility
                particles[i] = project_feasible(particles[i])
                
                # Evaluate fitness
                fitness = compute_C5(particles[i])
                
                # Update personal best
                if fitness < personal_best_vals[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_vals[i] = fitness
                    
                    # Update global best
                    if fitness < global_best_val:
                        global_best = particles[i].copy()
                        global_best_val = fitness
                        
        return global_best, global_best_val

    # Use L-BFGS-B for local refinement
    def refine_with_lbfgs(particle):
        def objective(h_vec):
            return compute_C5(h_vec)
            
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        constraints = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, particle, method='L-BFGS-B', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 300, 'gtol': 1e-5})
        
        return project_feasible(result.x)

    # PSO Phase
    try:
        pso_h, pso_C5 = pso_optimization()
        if pso_C5 < best_C5:
            best_C5 = pso_C5
            best_h = pso_h.copy()
    except:
        pass

    # Refinement Phase
    try:
        refined_h = refine_with_lbfgs(best_h)
        refined_C5 = compute_C5(refined_h)
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
    except Exception:
        pass

    # Final trust region refinement
    def trust_region_update(h_vec, eta=0.005, max_iter=50):
        h_new = h_vec.copy()
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            # Simple gradient estimator with finite differences
            eps = 1e-4
            grad = np.zeros_like(h_vec)
            base_c = compute_C5(h_new)
            
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_plus[j] = min(1, h_new[j] + eps)
                h_minus = h_new.copy()
                h_minus[j] = max(0, h_new[j] - eps)
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                c_plus = compute_C5(h_plus)
                c_minus = compute_C5(h_minus)
                
                grad[j] = (c_plus - c_minus) / (2 * eps)
            
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        
        return h_new

    # Try final trust region refinement
    try:
        final_h = trust_region_update(best_h)
        final_C5 = compute_C5(final_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = final_h.copy()
    except Exception:
        pass

    return (best_h.tolist(), best_C5, len(best_h))
