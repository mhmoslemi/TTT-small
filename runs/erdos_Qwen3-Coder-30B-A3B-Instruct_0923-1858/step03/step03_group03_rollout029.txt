import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Base gradient computation that's more stable
    def compute_gradient(h_vec, delta=1e-5):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += delta
            h_minus[i] -= delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * delta)
            
        return grad
    
    # Accelerated proximal gradient approach
    def prox_grad_descent(h_vec, max_iter=100, step_size=0.01):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = compute_gradient(h_new)
            h_candidate = h_new - step_size * grad
            h_new = project_feasible(h_candidate)
        return h_new

    # Enhanced perturbation with local search
    def enhance_solution(h_vec, strength=0.03):
        h_new = h_vec.copy()
        # Try to improve by local modification
        indices = np.random.choice(len(h_vec), size=max(1, len(h_vec)//10), replace=False)
        for idx in indices:
            old_val = h_new[idx]
            h_new[idx] = np.clip(old_val + np.random.normal(0, strength), 0, 1)
        return project_feasible(h_new)

    # Multi-scale gradient descent with local search
    convergence_patience = 0
    max_patience = 50
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            convergence_patience = 0
        else:
            convergence_patience += 1
            if convergence_patience > max_patience:
                h = enhance_solution(h, 0.05)
                convergence_patience = 0

        # Apply proximal gradient descent
        h = prox_grad_descent(h, max_iter=100, step_size=0.01)
        
        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
