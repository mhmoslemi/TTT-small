import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        h = h * (n_points / 2) / np.sum(h)

    best_C5 = float('inf')

        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_gradient(h_vec, k_max):
        h_vec = np.array(h_vec)
        # Compute correlation at the maximal shift
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = k_max + (len(h_vec) - 1) if k_max >= 0 else k_max + len(h_vec)
        
        # Numerical gradient via finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)

    def refine_solution(h_vec, max_iter=20, learning_rate=0.05):
        h_new = h_vec.copy()

            # Find the max shift
            corr = correlate(h_new, 1 - h_new, mode="full")
