import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Parse initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
        
    def evaluate_particle(h_vec):
        h_feasible = project_feasible(h_vec)
        return compute_C5(h_feasible)
    
    def perturbation(h_vec, scale=0.03):
        h_new = h_vec + np.random.normal(0, scale, len(h_vec))
        return project_feasible(h_new)
        
    # PSO Parameters
    swarm_size = 20
    max_iter_pso = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    gbest = h.copy()
    gbest_score = compute_C5(gbest)
    
    for i in range(swarm_size):
        if i == 0:
            h_init = h.copy()
        else:
            h_init = perturbation(h, scale=0.05)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        score = evaluate_particle(h_init)
        personal_best.append(h_init.copy())
        personal_best_scores.append(score)
        if score < gbest_score:
            gbest = h_init.copy()
            gbest_score = score
    
    # PSO iterations
    iter_count = 0
    for _ in range(max_iter_pso):
        if elapsed() > budget_s * 0.85:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            score = evaluate_particle(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
        
        iter_count += 1
    
    # Refine with L-BFGS-B
    def lbfgsb_objective(h_vec):
        return compute_C5(h_vec)
    
    def lbfgsb_constraint(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lbfgsb_constraint}
    
    try:
        result = minimize(
            lbfgsb_objective,
            gbest,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-5, 'maxiter': 300},
            callback=lambda x: None if elapsed() > budget_s * 0.99 else None
        )
        
        if result.success:
            final_h = project_feasible(result.x)
            final_C5 = compute_C5(final_h)
            
            if final_C5 < gbest_score:
                gbest = final_h.copy()
                gbest_score = final_C5
    except:
        pass
    
    # Final polishing
    best_C5 = gbest_score
    best_h = gbest.copy()
    
    # Return solution
    return (best_h.tolist(), best_C5, n_points)
