import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization approach
    def run_pso(max_iter=500):
        # Initialize particles with random perturbations of h
        n_particles = 20
        particles = []
        velocities = []
        personal_best = []
        personal_best_values = []
        global_best = h.copy()
        global_best_value = compute_C5(h)
        
        # Initialize particles
        for _ in range(n_particles):
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_feasible(particle)
            particles.append(particle.copy())
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
            pbest = particle.copy()
            pbest_val = compute_C5(particle)
            personal_best.append(pbest)
            personal_best_values.append(pbest_val)
            if pbest_val < global_best_value:
                global_best = pbest.copy()
                global_best_value = pbest_val

        w, c1, c2 = 0.7, 1.5, 1.5  # PSO parameters
                
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best[i] - particles[i]) + 
                                c2 * r2 * (global_best - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Evaluate
                current_value = compute_C5(particles[i])
                
                # Update personal best
                if current_value < personal_best_values[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_values[i] = current_value
                    
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value

        return global_best, global_best_value

    # Gradient-based approach with lbfgs
    def run_lbfgs():
        def objective(h_vec):
            return compute_C5(h_vec)
        
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                         options={'maxiter': 100, 'ftol': 1e-10})
        
        if result.success:
            return project_feasible(result.x), result.fun
        return h, compute_C5(h)

    # Simulated Annealing approach
    def run_simulated_annealing():
        current_h = h.copy()
        current_value = compute_C5(current_h)
        best_h = current_h.copy()
        best_value = current_value
        
        T = 1.0
        T_min = 1e-5
        alpha = 0.99
        max_iter = 1000
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Generate neighbor
            neighbor = current_h.copy()
            idx = np.random.randint(0, n_points)
            neighbor[idx] += np.random.normal(0, 0.01)
            neighbor = project_feasible(neighbor)
            
            neighbor_value = compute_C5(neighbor)
            delta = neighbor_value - current_value
            
            if delta < 0 or np.random.random() < np.exp(-delta / T):
                current_h = neighbor
                current_value = neighbor_value
                
                if current_value < best_value:
                    best_h = current_h.copy()
                    best_value = current_value
                    
            T = max(T_min, T * alpha)
            
        return best_h, best_value

    # Hybrid approach: start with PSO, then refine with LBFGS and SA
    # PSO
    try:
        pso_h, pso_value = run_pso(max_iter=100)
        if pso_value < best_C5:
            best_C5 = pso_value
            best_h = pso_h.copy()
    except:
        pass

    # Local refinement with L-BFGS
    try:
        lbfgs_h, lbfgs_value = run_lbfgs()
        if lbfgs_value < best_C5:
            best_C5 = lbfgs_value
            best_h = lbfgs_h.copy()
    except:
        pass

    # Local refinement with Simulated Annealing
    try:
        sa_h, sa_value = run_simulated_annealing()
        if sa_value < best_C5:
            best_C5 = sa_value
            best_h = sa_h.copy()
    except:
        pass

    # Final local search with gradient descent
    def gradient_descent_step(h_vec, lr=0.01):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute the gradient of correlation w.r.t. h_i
        eps = 1e-6
        grad = np.zeros(len(h_vec))
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
            
        return h_vec - lr * grad

    # Refinement step
    ref_iter = 0
    while ref_iter < 100 and elapsed() < budget_s * 0.95:
        old_value = compute_C5(best_h)
        better_h = gradient_descent_step(best_h, lr=0.005)
        better_h = project_feasible(better_h)
        new_value = compute_C5(better_h)
        
        if new_value < old_value:
            best_h = better_h
            if new_value < best_C5:
                best_C5 = new_value
        else:
            break
        ref_iter += 1

    return (best_h.tolist(), best_C5, len(best_h))
