import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

        # Create a triangular initial solution
        n_points = 45

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Improved trust region method with multi-scale search
    def refine_solution(h_vec, max_iter=30, learning_rate=0.01):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Compute gradient numerically
            grad = np.zeros_like(h_vec)
            eps = 1e-4
            current_C5 = compute_C5(h_new)
            for j in range(len(h_vec)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                max_k_idx = np.argmax(corr_plus)
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - learning_rate * grad
            h_new = project_feasible(h_new)
            
        return h_new

        h_new = h_vec.copy()
        # Randomly select a few indices to modify
        num_mutations = max(1, int(0.08 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_mutations, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, mutation_strength)
        return project_feasible(h_new)

    convergence_patience = 0

            if convergence_patience > max_patience:

                h = perturb_solution(h, 0.03)
                convergence_patience = 0

        h = refine_solution(h, max_iter=20, learning_rate=learning_rate)
        
        # Adaptive learning rate

            learning_rate *= 0.93

        iter_count += 1
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
