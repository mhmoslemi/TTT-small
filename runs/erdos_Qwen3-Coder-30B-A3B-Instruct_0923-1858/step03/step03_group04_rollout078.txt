import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO Parameters
    n_particles = 20
    max_iter = 300
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    vel_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        particle = project_feasible(np.random.rand(n_points) * 0.5)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-vel_max, vel_max, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(compute_C5(particle))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO loop
    for iter_ in range(max_iter):
        if elapsed() > budget_s * 0.5:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (pbest[i] - particles[i]) + 
                           social * r2 * (gbest - particles[i]))
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            curr_val = compute_C5(particles[i])
            
            # Update particle best
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
            
            # Update global best
            if curr_val < gbest_val:
                gbest = particles[i].copy()
                gbest_val = curr_val

        # Track best solution
        if gbest_val < best_C5:
            best_C5 = gbest_val
            best_h = gbest.copy()

    # L-BFGS-B Refinement using the best solution from PSO
    def obj_func(h_vec):
        return compute_C5(h_vec)
    
    def grad_func(h_vec):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            grad[i] = (obj_func(h_plus) - obj_func(h_minus)) / (2 * eps)
        return grad

    # Refinement with L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start from the best solution found by PSO
    h_refined = best_h.copy()
    result = minimize(obj_func, h_refined, method='L-BFGS-B', 
                      bounds=bounds, jac=grad_func, options={'maxiter': 300})
    
    if result.success:
        h_refined = result.x
    h_refined = project_feasible(h_refined)
    refined_C5 = compute_C5(h_refined)
    
    if refined_C5 < best_C5:
        best_C5 = refined_C5
        best_h = h_refined.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
