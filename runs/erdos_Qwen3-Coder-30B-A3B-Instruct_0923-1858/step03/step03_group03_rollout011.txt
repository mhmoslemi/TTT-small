import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Quasi-Newton trust region method with line search
    def update_trust_region(h_vec, max_iter=50, eta=0.01, trust_radius=0.1):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Compute correlation and find max overlap index
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_new) - 1)
            
            # Compute gradient by finite differences in directions
            grad = np.zeros_like(h_new)
            eps = 1e-5
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update with trust region step
            step = -eta * grad
            # Constrain step by trust region
            step_norm = np.linalg.norm(step)
            if step_norm > trust_radius:
                step = step * trust_radius / step_norm
            
            h_candidate = h_new + step
            h_candidate = project_feasible(h_candidate)
            
            # Accept if it improves the objective
            new_C5 = compute_C5(h_candidate)
            if new_C5 < compute_C5(h_new):
                h_new = h_candidate
                eta = min(eta * 1.1, 0.1)
            else:
                eta = max(eta * 0.5, 1e-5)
                trust_radius = max(trust_radius * 0.5, 1e-4)
        return h_new

    # Perturbation operator
    def perturb_solution(h_vec, strength=0.1):
        h_new = h_vec.copy()
        num_perturb = max(1, int(0.15 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_perturb, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    convergence_patience = 0
    max_patience = 100
    iter_count = 0

            best_C5 = current_C5
            best_h = h.copy()
            convergence_patience = 0
        else:
            convergence_patience += 1

        if convergence_patience > max_patience:
            h = perturb_solution(h, 0.05)
            convergence_patience = 0
        
        # Update with quasi-Newton trust region
        h = update_trust_region(h, max_iter=50, eta=0.01)
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
