            h[i] = max(0, 1 - abs(x[i] - 1))

    best_h = h.copy()

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

            if k >= 0:
                # h[x] * (1 - h[x+k]) 
                # approx = h_initial[x] * (1 - h_initial[(x+k)%n]) - h_initial[x] * (h[x] - h_initial[x])

                #    - h_initial[x] * h[x] + h_initial[x]^2

                # For negative k, we need h[x]* (1 - h[x-k])
                linear = h_initial[x] * (1 - h_initial[(x+k)%n]) - h_initial[x] * h_initial[(x+k)%n] + \
                        (1 - h_initial[(x+k)%n] - h_initial[x]) * h[x]
            constraints.append(linear <= C_temp / dx)
        
        # Objective is to minimize 0
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)

        h_vec = np.array(h_vec)

        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)

        h_plus = h_vec + eps

        h_plus = project_feasible(h_plus)
        h_minus = project_feasible(h_minus)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
        
        grad = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

            if elapsed() > budget_s * 0.98:
                break
            grad = compute_gradient(h)
            # Try small steps in gradient direction
            step = 0.01
            h_new = h - step * grad
            h_new = project_feasible(h_new)
            h = h_new
        
        return h

    C_low, C_high = 0.0, 0.4  # reasonable range for binary search

    while elapsed() < budget_s * 0.95:

        C_mid = (C_low + C_high) / 2
        
        # Try to solve the feasible problem for this C_mid
        # We use a certain number of random restarts to avoid local minimums

                    break

                best_C5 = new_C

        else:
            C_low = C_mid
            
    # Final polishing
    refined_h = local_search(best_h, 100)
    refined_C = compute_C5(refined_h)
    if refined_C < best_C5:
        best_C5 = refined_C
        best_h = refined_h.copy()
    
    if best_C5 == float('inf'):
        pass  # fallback to initial h
        
    return (best_h.tolist(), best_C5, len(best_h))
