import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def crossover(parent1, parent2):
    """Perform one-point crossover between two parents."""
    n = len(parent1)
    if n < 2:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate, mutation_scale):
    """Mutate a candidate solution."""
    h_mut = h.copy()
    mask = np.random.rand(len(h)) < mutation_rate
    if np.any(mask):
        h_mut += np.random.normal(0, mutation_scale, len(h)) * mask
        h_mut = np.clip(h_mut, 0, 1)
        # Normalize to maintain sum constraint
        dx = 2.0 / len(h)
        h_mut = project(h_mut, dx)
    return h_mut

    n_points = 45
    dx = 2.0 / n_points

    population = []
    
    # Try to use the initial construction if available

        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.mean(initial_h) * (n_points / 2)
    
    # Add initial solution to population
    population.append(initial_h.copy())
    
    # Fill the rest of population with variations
    for _ in range(population_size - 1):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project(h, dx)
        population.append(h)
    
    # Genetic Algorithm loop
    t0 = time.time()
    max_time = budget_s - 2
    generation = 0
    stagnation_count = 0
    best_fitness = float('inf')
    best_individual = initial_h.copy()
    
    while time.time() - t0 < max_time:
        generation += 1
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            fit = evaluate(h, dx)
            fitness.append(fit)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_fitness:
            best_fitness = fitness[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # If no improvement in a while, break
        if stagnation_count > 1000:
            break
            
        # Selection (tournament selection of top 50%)
        selected_indices = []
        tournament_size = 3
        for _ in range(population_size // 2):
            tournament_indices = np.random.choice(len(population),

            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner = tournament_indices[np.argmin(tournament_fitness)]
            selected_indices.append(winner)
            
        # Create new population via crossover and mutation
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:2]
        for idx in elite_indices:
            new_population.append(population[idx])
            
        # Generate offspring
        while len(new_population) < population_size:
            if len(selected_indices) < 2:
                # Randomly select two parents
                p1_idx = np.random.randint(0, len(population))
                p2_idx = np.random.randint(0, len(population))

            child2 = mutate(child2, mutation_rate, mutation_scale)
            
            new_population.extend([child1, child2])
            
        # Trim population
        population = new_population[:population_size]
    
    # Final validation
    final_h = project(best_individual, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
