import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available (the parent construction)
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Gradient estimation using finite differences
    def estimate_gradient(h_vec, eps=1e-4, samples=10):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        grad = np.zeros_like(h_vec)
        for _ in range(samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = project_feasible(h_vec + delta)
            h_minus = project_feasible(h_vec - delta)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
        
        return grad / samples

    # Perturbation function for escaping local minima
    def perturb_solution(h_vec, strength=0.05):
        h_new = h_vec.copy()
        idxs = np.random.choice(n_points, size=max(1, int(0.15 * n_points)), replace=False)
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, strength), 0, 1)
        return project_feasible(h_new)

    # Trust region step with adaptive learning rate
    def trust_region_update(h_vec, eta=0.01, max_iter=20):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = estimate_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Simple convex relaxation solver as warm-start
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Minimize the sum of squares to encourage spread-out solutions
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()  # fallback to current solution

    # Early optimization loop with turbulence and gradient updates
    iter_count = 0
    patience = 0
    max_patience = 30
    last_best_update = 0
    
    # Ensure we start with relaxed solution if possible
    if elapsed() < budget_s * 0.2:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass

    # Main loop with turbulence and refinement
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Apply turbulence to escape local minima
        if iter_count % 10 == 0:  # Turbulence every 10 steps
            # Apply sinusoidal random turbulences
            noise = np.random.normal(0, 0.01, n_points)
            h_turb = h + 0.1 * noise
            h = project_feasible(h_turb)
        
        # Occasionally refine with gradient step
        if iter_count % 30 == 0:
            # Refine with trust region
            h = trust_region_update(h, eta=0.01, max_iter=20)
        
        # Periodic perturbation if no progress
        if patience > max_patience or (iter_count - last_best_update > 100):
            h = perturb_solution(h)
            patience = 0
            # Restart with relaxation if it helps
            if iter_count - last_best_update > 200:
                try:
                    relaxed_h = solve_convex_relaxation()
                    if compute_C5(relaxed_h) < compute_C5(h):
                        h = relaxed_h.copy()
                except:
                    pass
                    
        iter_count += 1

    # Final refinement
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
