        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_analytical_gradient(h_vec, k_max):
        h_vec = np.array(h_vec)
        n = len(h_vec)
        grad = np.zeros_like(h_vec)

        # The derivative is essentially the sum over shifts
        # We compute an approximate analytical gradient of the max overlap
        for i in range(n):
            # If we increased h[i], the effect on the overlap

            grad[i] = 0

            if i + k_max < n:
                grad[i] += (1 - h_vec[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max]
                
        return grad

            grad = compute_analytical_gradient(h_new, k_max)
