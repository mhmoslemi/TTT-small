import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Phase 1: Particle Swarm Optimization (PSO)
    def pso_optimization():
        # PSO parameters
        swarm_size = 20
        max_iter = 200
        inertia = 0.7
        cognitive = 1.5
        social = 1.5
        vel_max = 0.1
        
        # Initialize swarm
        swarm = []
        velocities = []
        personal_best = []
        personal_best_scores = []
        
        for _ in range(swarm_size):
            h_init = project_feasible(h + np.random.normal(0, 0.05, n_points))
            swarm.append(h_init.copy())
            velocities.append(np.random.uniform(-vel_max, vel_max, n_points))
            score = compute_C5(h_init)
            personal_best.append(h_init.copy())
            personal_best_scores.append(score)
            
        global_best_idx = np.argmin(personal_best_scores)
        global_best = personal_best[global_best_idx].copy()
        global_best_score = personal_best_scores[global_best_idx]
        
        for iter_count in range(max_iter):
            if elapsed() > budget_s * 0.5:
                break
                
            for i in range(swarm_size):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia * velocities[i] + 
                                cognitive * r1 * (personal_best[i] - swarm[i]) + 
                                social * r2 * (global_best - swarm[i]))
                
                # Clamp velocity
                velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
                
                # Update position
                swarm[i] += velocities[i]
                
                # Repair feasibility
                swarm[i] = project_feasible(swarm[i])
                
                # Evaluate
                score = compute_C5(swarm[i])
                
                # Update personal best
                if score < personal_best_scores[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_scores[i] = score
                    
                    # Update global best
                    if score < global_best_score:
                        global_best = swarm[i].copy()
                        global_best_score = score
                        
        return global_best, global_best_score

    # Phase 2: L-BFGS-B refinement
    def lbfgsb_refinement(initial_h):
        # Define constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        # Objective function for optimization
        def objective(x):
            corr = correlate(x, 1 - x, mode="full")
            return -np.max(corr) * dx  # negative because we minimize
            
        # Run optimization
        result = minimize(objective, 
                         initial_h, 
                         method='L-BFGS-B', 
                         bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6})
        
        refined_h = result.x
        refined_h = project_feasible(refined_h)
        return refined_h, -result.fun
    
    # Run the optimization
    try:
        # Phase 1: PSO
        if elapsed() < budget_s * 0.5:
            pso_h, pso_C5 = pso_optimization()
            if pso_C5 < best_C5:
                best_C5 = pso_C5
                best_h = pso_h.copy()
                
        # Phase 2: L-BFGS-B refinement
        if elapsed() < budget_s * 0.9:
            refined_h, refined_C5 = lbfgsb_refinement(best_h)
            if refined_C5 < best_C5:
                best_C5 = refined_C5
                best_h = refined_h.copy()
                
        # Final evaluation
        final_C5 = compute_C5(best_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = best_h.copy()
    except:
        pass

    return (best_h.tolist(), best_C5, len(best_h))
