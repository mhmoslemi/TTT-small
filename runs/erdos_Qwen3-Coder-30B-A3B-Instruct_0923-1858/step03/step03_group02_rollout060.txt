        return h_vec

    def gradient_projection_update(h_vec, step_size=0.01, max_iter=50):
        h_new = h_vec.copy()

            # Compute correlation and find max index
            corr = correlate(h_new, 1 - h_new, mode="full")

            eps = 1e-5

                h_minus = h_new.copy()
