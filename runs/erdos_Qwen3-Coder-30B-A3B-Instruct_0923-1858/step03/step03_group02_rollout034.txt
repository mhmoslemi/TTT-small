import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a modified triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Quadratic programming approach
    def solve_qp_constraint(h_vec, k_max):
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= cp.sum(cp.multiply(h_new, 1 - h_new[k_max:])),
        ]
        
        # Add linear approximation constraint at current point
        # Compute the gradient w.r.t. h for shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = 1 - h_vec[i + k_max]
            else:
                grad[i] = 0
                
        # Add the linearized constraint
        linear_constraint = t >= (
            np.sum(h_vec * (1 - h_vec[k_max:])) * dx +
            np.sum(grad * (h_new - h_vec)) * dx
        )
        constraints.append(linear_constraint)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, tol=1e-7)
            if prob.status == cp.OPTIMAL:
                return h_new.value
        except:
            pass
        return h_vec
    
    # Main optimization loop
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb the solution
                perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                for i in perturb_indices:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
                
        # Evaluate all overlaps to find the most constraining one
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        shift_k = k_max - (n_points - 1)
        
        # Solve QP with this constraint
        h_new = solve_qp_constraint(h, shift_k)
        if h_new is not None:
            h = h_new
            
        # Ensure feasibility
        h = project_feasible(h)
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Final check and projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
