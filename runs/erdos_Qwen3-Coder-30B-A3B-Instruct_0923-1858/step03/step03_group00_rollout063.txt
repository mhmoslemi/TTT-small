import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def initialize_population(n_points, population_size, initial_h=None):
    """Initialize a population of candidate solutions."""
    population = []
    
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Fill the rest of the population with random variations
    for i in range(population_size - len(population)):
        # Start with a structured pattern
        t = np.linspace(0, 2, n_points)
        h = 0.3 + 0.2 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        h = np.clip(h, 0, 1)
        
        # Add small random noise
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = project(h, 2.0 / n_points)
        population.append(h)
    
    return population

def crossover(parent1, parent2):
    """Perform one-point crossover."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
    child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
    
    return child1, child2

def mutate(h, mutation_rate, mutation_scale):
    """Mutate a solution."""
    n_points = len(h)
    mask = np.random.random(n_points) < mutation_rate
    if np.sum(mask) > 0:
        noise = np.random.normal(0, mutation_scale, n_points)
        h = h + noise * mask
        h = np.clip(h, 0, 1)
        # Normalize to satisfy sum constraint
        dx = 2.0 / n_points
        h = project(h, dx)
    return h

def select_parents(population, fitnesses, tournament_size=3):
    """Tournament selection."""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected

    n_points = 45
    dx = 2.0 / n_points
    
    # Population and GA parameters
    population_size = 20
    mutation_rate = 0.05
    max_generations = 5000
    elite_size = 2

        initial_h = np.clip(initial_h, 0, 1)
        initial_h = project(initial_h, dx)
    
    population = initialize_population(n_points, population_size, initial_h)
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    t0 = time.time()
    generation = 0
    
    while time.time() - t0 < budget_s - 2 and generation < max_generations:
        # Evaluate fitness of population
        fitnesses = [evaluate(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_C:
            best_C = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select two parents
            parent1_idx, parent2_idx = np.random.choice(len(parents), 2, replace=False)
            parent1, parent2 = parents[parent1_idx], parents[parent2_idx]
            
            # Crossover
            if np.random.random() < 0.8:  # Crossover rate
                child1, child2 = crossover(parent1, parent2)
                
                # Mutation
                child1 = mutate(child1, mutation_rate, 0.01)
                child2 = mutate(child2, mutation_rate, 0.01)
                
                new_population.extend([child1, child2])
            else:
                # Copy parents without crossover
                child1 = parent1.copy()
                child2 = parent2.copy()
                
                # Mutation
                child1 = mutate(child1, mutation_rate, 0.01)
                child2 = mutate(child2, mutation_rate, 0.01)
                
                new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        generation += 1
    
    # Final check on best solution
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
