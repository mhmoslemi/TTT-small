import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Frequency domain optimization approach
    def frequency_domain_update(h_vec, learning_rate=0.01):
        # Convert to frequency domain
        H = fft(h_vec)
        # Compute the conjugate of (1-h) in frequency domain
        H_conj = np.conj(H)
        # Compute correlation in frequency domain
        corr_freq = H * H_conj
        # Get inverse transform in real domain
        corr_real = np.real(ifft(corr_freq))
        
        # Find index of maximum correlation
        max_idx = np.argmax(corr_real)
        # Compute gradient in frequency domain
        grad_h = np.real(ifft(-2 * H_conj * (corr_real[max_idx] - corr_real)))
        
        # Update in frequency domain
        H_new = H - learning_rate * grad_h
        h_new = np.real(ifft(H_new))
        
        return project_feasible(h_new)

    # Main optimization loop with frequency domain updates
    iter_count = 0
    eta = 0.01
    patience = 0
    max_patience = 20
    last_best_update = 0
    max_iter = 500

    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Check for stagnation
        if patience > max_patience or (iter_count - last_best_update > 30):
            # Add small random perturbations
            h = project_feasible(h + np.random.normal(0, 0.01, n_points))
            patience = 0

        # Frequency domain update
        h = frequency_domain_update(h, eta)
        
        # Adaptive learning rate
        if iter_count % 10 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
