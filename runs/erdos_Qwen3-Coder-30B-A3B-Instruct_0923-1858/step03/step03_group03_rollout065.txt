import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Coarse initial grid
        n_points = 30
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    refinement_threshold = 1000
    
    iter_count = 0
    while elapsed() < budget_s * 0.95 and T > min_T:
        # Perturbation
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        h_new[idx] += delta
        
        # Project to feasible set
        h_new = project_feasible(h_new)
        
        # Compute C5
        new_C5 = compute_C5(h_new)
        old_C5 = compute_C5(h)
        
        # Accept or reject
        if new_C5 < old_C5 or np.random.rand() < np.exp(-(new_C5 - old_C5) / T):
            h = h_new
            current_C5 = new_C5
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Refinement
        if iter_count > 0 and iter_count % refinement_threshold == 0:
            # Double grid resolution
            n_points = min(n_points * 2, 900)  # Cap at reasonable limit
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, np.linspace(0, 2, len(h)), h)
            h_new = project_feasible(h_new)
            h = h_new
            dx = 2.0 / n_points
        
        T *= alpha
        iter_count += 1
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
