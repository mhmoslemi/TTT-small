import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Swarm parameters
    num_particles = 50
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_clamp = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    # Create initial particles
    particles.append(h.copy())
    velocities.append(np.zeros_like(h))
    pbest.append(h.copy())
    pbest_fitness.append(float('inf'))
    
    for i in range(1, num_particles):
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = h + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(h_pert.copy())
        pbest_fitness.append(float('inf'))
    
    s_best = h.copy()
    s_best_fitness = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    iter_count = 0
    patience = 0
    max_patience = 30
    
    while elapsed() < budget_s * 0.95:
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / 1000)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (pbest[i] - particles[i])
            social = social_weight * r2 * (s_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < s_best_fitness:
                    s_best = particles[i].copy()
                    s_best_fitness = fitness
                    patience = 0
                    if s_best_fitness < best_C5:
                        best_C5 = s_best_fitness
                        best_h = s_best.copy()
            else:
                patience += 1
        
        # Perturbation if stuck
        if patience > max_patience:
            for i in range(1, num_particles):
                pert = np.random.normal(0, 0.03, n_points)
                particles[i] = particles[i] + pert
                particles[i] = project_feasible(particles[i])
                # Re-evaluate and update pbest if better
                fitness = compute_C5(particles[i])
                if fitness < pbest_fitness[i]:
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = fitness
                    if fitness < s_best_fitness:
                        s_best = particles[i].copy()
                        s_best_fitness = fitness
                        patience = 0
                        if s_best_fitness < best_C5:
                            best_C5 = s_best_fitness
                            best_h = s_best.copy()
            patience = 0
        
        iter_count += 1
        
        if iter_count % 50 == 0:
            # Periodic check of current best
            current_s_best_fitness = compute_C5(s_best)
            if current_s_best_fitness < s_best_fitness:
                s_best_fitness = current_s_best_fitness

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
