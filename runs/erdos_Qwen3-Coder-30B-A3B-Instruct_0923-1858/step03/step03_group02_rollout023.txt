import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 50
    max_iter = 2000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_w = 2.0
    social_w = 2.0
    velocity_clamp = 0.1
    
    # Initialize swarm
    swarm_h = np.zeros((swarm_size, n_points))
    swarm_vel = np.zeros((swarm_size, n_points))
    
    for i in range(swarm_size):
        if i == 0:
            swarm_h[i] = h_init.copy()
        else:
            pert = np.random.normal(0, 0.05, n_points)
            swarm_h[i] = h_init + pert
            swarm_h[i] = project_feasible(swarm_h[i])
    
    # Track personal bests
    pbest_h = swarm_h.copy()
    pbest_C5 = np.array([compute_C5(h) for h in pbest_h])
    
    # Global best
    gbest_idx = np.argmin(pbest_C5)
    gbest_h = pbest_h[gbest_idx].copy()
    gbest_C5 = pbest_C5[gbest_idx]
    
    iter_count = 0
    
    # Main PSO loop
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        # Update particles
        for i in range(swarm_size):
            # Generate random weights
            r1, r2 = np.random.random(2)
            
            # Update velocity
            swarm_vel[i] = (inertia * swarm_vel[i] + 
                           cognitive_w * r1 * (pbest_h[i] - swarm_h[i]) + 
                           social_w * r2 * (gbest_h - swarm_h[i]))
            
            # Clamp velocity
            swarm_vel[i] = np.clip(swarm_vel[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            swarm_h[i] += swarm_vel[i]
            
            # Project to feasible region
            swarm_h[i] = project_feasible(swarm_h[i])
            
            # Evaluate new position
            new_C5 = compute_C5(swarm_h[i])
            
            # Update personal best
            if new_C5 < pbest_C5[i]:
                pbest_C5[i] = new_C5
                pbest_h[i] = swarm_h[i].copy()
                
                # Update global best
                if new_C5 < gbest_C5:
                    gbest_C5 = new_C5
                    gbest_h = swarm_h[i].copy()
        
        # Check for stagnation
        if iter_count % 50 == 0 and iter_count > 0:
            # Perturb particles if no improvement
            improvement = False
            for i in range(swarm_size):
                if pbest_C5[i] < gbest_C5:
                    improvement = True
                    break

            if not improvement:
                for i in range(swarm_size):
                    perturbation = np.random.normal(0, 0.03, n_points)
                    swarm_h[i] += perturbation
                    swarm_h[i] = project_feasible(swarm_h[i])
                    # Re-evaluate
                    new_C5 = compute_C5(swarm_h[i])
                    if new_C5 < pbest_C5[i]:
                        pbest_C5[i] = new_C5
                        pbest_h[i] = swarm_h[i].copy()
                        if new_C5 < gbest_C5:
                            gbest_C5 = new_C5
                            gbest_h = swarm_h[i].copy()

        iter_count += 1

    # Final check
    final_C5 = compute_C5(gbest_h)
    
    return (gbest_h.tolist(), final_C5, len(gbest_h))
