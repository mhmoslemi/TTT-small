import time
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    # Initialize with feasible solutions
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            particle = project_feasible(h + np.random.normal(0, 0.02, n_points))
        else:
            particle = project_feasible(np.random.rand(n_points) * 0.5)
        swarm.append(particle.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        p_best = particle.copy()
        p_best_val = compute_C5(p_best)
        personal_best.append(p_best)
        personal_best_values.append(p_best_val)
    
    # Global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    iter_count = 0
    last_improvement = 0
    
    while elapsed() < budget_s * 0.6 and iter_count < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_best[i] - swarm[i])
                           + c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            current_value = compute_C5(swarm[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = swarm[i].copy()
                global_best_value = current_value
                last_improvement = iter_count

        # Check termination
        if iter_count - last_improvement > 50:
            break
        iter_count += 1

    # Local refinement with L-BFGS-B
    def lbfgs_objective(h_vec):
        return compute_C5(h_vec)

    def lbfgs_constraint(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Local refinement
    if elapsed() < budget_s * 0.95:
        # We'll refine using scipy.optimize.minimize with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lbfgs_constraint}
        
        result = minimize(
            lbfgs_objective,
            global_best,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'gtol': 1e-6, 'maxiter': 200},
            callback=lambda x: None
        )
        
        if result.success:
            refined_h = result.x
            refined_C5 = compute_C5(refined_h)
            if refined_C5 < global_best_value:
                global_best = refined_h
                global_best_value = refined_C5

    return (global_best.tolist(), global_best_value, n_points)
