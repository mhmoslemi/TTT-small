import time
from scipy.signal import correlate
from scipy.optimize import minimize

    start_time = time.time()

        # Create a piecewise linear function with peak at center
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def objective(h_vec):
        h_vec = project_feasible(h_vec)
        return compute_C5(h_vec)

    # Use scipy.optimize.minimize with bounds
    def optimize_step(h_vec, max_iter=100):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        result = minimize(objective, h_vec, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-6})
        return result.x

    # Local refinement with gradient-based approach
    def local_refine(h_vec, steps=50):
        for _ in range(steps):
            h_vec = optimize_step(h_vec, max_iter=20)
        return h_vec

    # Perturbation function to escape local minima
    def perturb_solution(h_vec, strength=0.05):
        h_new = h_vec.copy() + np.random.normal(0, strength, n_points) 
        return project_feasible(h_new)

    # Main optimization loop
    max_patience = 50
    patience = 0
    iter_count = 0
    eta = 0.05  # learning rate

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = perturb_solution(h, 0.08)
                patience = 0

        # Refine with optimization
        h = local_refine(h, 20)

        if iter_count % 10 == 0:
            h = perturb_solution(h, 0.03)

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
