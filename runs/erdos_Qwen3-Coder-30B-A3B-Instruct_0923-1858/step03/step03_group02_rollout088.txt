import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Gradient computation using finite differences
    def compute_gradients(h_vec):
        h_vec = np.array(h_vec)
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        
        # Find index of maximum correlation
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    # Quasi-Newton optimization with L-BFGS
    def optimize_step(h_vec, max_iter=10):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        def objective(h):
            return compute_C5(h)
            
        def jacobian(h):
            return compute_gradients(h)
            
        try:
            result = minimize(
                objective,
                h_vec,
                method='L-BFGS-B',
                bounds=bounds,
                jac=jacobian,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-8}
            )
            return result.x if result.success else h_vec
        except:
            return h_vec

    # Local neighborhood search
    def local_search(h_vec, max_iter=50):
        h_best = h_vec.copy()
        best_val = compute_C5(h_best)
        
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            
            # Random perturbations
            h_candidate = h_best.copy()
            for i in range(n_points):
                if np.random.random() < 0.1:
                    h_candidate[i] += np.random.normal(0, 0.01)
                    
            h_candidate = project_feasible(h_candidate)
            val = compute_C5(h_candidate)
            
            if val < best_val:
                h_best = h_candidate
                best_val = val
                
        return h_best

    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    
    while elapsed() < budget_s * 0.95:
        if iter_count % 10 == 0:
            # Periodic global refinement
            h = optimize_step(h, max_iter=3)
            
        # Local search
        h = local_search(h, max_iter=5)
        
        # Apply projection
        h = project_feasible(h)
        
        # Update best solution
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count
            
        iter_count += 1
        
        # Early stopping if no improvement in a while
        if iter_count - last_improvement > 100:
            break
            
    # Final refinement
    final_h = optimize_step(best_h, max_iter=5)
    final_h = project_feasible(final_h)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h
    
    # Final local search
    final_h = local_search(best_h, max_iter=10)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h
        
    return (best_h.tolist(), best_C5, len(best_h))
