import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 1000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_clamp = 0.1

    # Initialize swarm
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    for i in range(n_particles):
        if i == 0:
            particle_h = h.copy()
        else:
            particle_h = h + np.random.normal(0, 0.05, n_points)
            particle_h = np.clip(particle_h, 0, 1)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
        
        particles.append(particle_h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(particle_h.copy())
        p_best_fitness.append(compute_C5(particle_h))

    s_best = p_best[np.argmin(p_best_fitness)].copy()
    s_best_fitness = min(p_best_fitness)

    # PSO iteration
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (p_best[i] - particles[i])
            social = social_weight * r2 * (s_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            current_fitness = compute_C5(particles[i])
            
            # Update personal best
            if current_fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = current_fitness
                
                # Update global best
                if current_fitness < s_best_fitness:
                    s_best = particles[i].copy()
                    s_best_fitness = current_fitness

        # Occasionally perturb particles if no improvement
        if iteration % 30 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.3:
                    particles[i] += np.random.normal(0, 0.03, n_points)
                    particles[i] = project_feasible(particles[i])

    # Final evaluation
    final_C5 = compute_C5(s_best)
    if final_C5 < best_C5:
        best_h = s_best.copy()
        best_C5 = final_C5
    
    # Ensure we return the best solution found
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, len(best_h))
