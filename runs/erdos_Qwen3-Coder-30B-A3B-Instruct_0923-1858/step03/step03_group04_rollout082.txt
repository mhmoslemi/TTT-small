import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO parameters
    n_particles = 20
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    max_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    # Initialize with best solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    personal_best.append(h.copy())
    personal_best_vals.append(compute_C5(h))
    
    # Initialize rest of particles with noisy versions
    for i in range(1, n_particles):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        particles.append(project_feasible(perturbed))
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        personal_best.append(particles[i].copy())
        personal_best_vals.append(compute_C5(particles[i]))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO main loop
    for iter_count in range(max_iter):
        if elapsed() > budget_s * 0.6:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate new position
            new_val = compute_C5(particles[i])
            
            # Update personal best
            if new_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = new_val
                
                # Update global best if needed
                if new_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = new_val
        
        # Perturb stagnating particles
        if iter_count % 10 == 0:
            for i in range(n_particles):
                if personal_best_vals[i] == global_best_val:
                    perturb = np.random.normal(0, 0.03, n_points)
                    particles[i] = project_feasible(particles[i] + perturb)
                    new_val = compute_C5(particles[i])
                    if new_val < personal_best_vals[i]:
                        personal_best[i] = particles[i].copy()
                        personal_best_vals[i] = new_val
                        if new_val < global_best_val:
                            global_best = particles[i].copy()
                            global_best_val = new_val
    
    # L-BFGS-B refinement of the best PSO solution
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine using L-BFGS-B if time allows
    if elapsed() < budget_s * 0.9:
        try:
            result = minimize(objective, global_best, method='L-BFGS-B', 
                            bounds=bounds, constraints=constraints,
                            options={'ftol': 1e-8, 'gtol': 1e-5},
                            callback=None)
            if result.success:
                refined_h = project_feasible(result.x)
                refined_C5 = compute_C5(refined_h)
                if refined_C5 < global_best_val:
                    global_best = refined_h
                    global_best_val = refined_C5
        except:
            pass
    
    # Final evaluation
    final_C5 = compute_C5(global_best)
    if final_C5 < global_best_val:
        global_best_val = final_C5
    
    return (global_best.tolist(), global_best_val, n_points)
