import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)

        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter = 500
    inertia_w = 0.7
    cognitive_w = 1.5
    social_w = 1.5
    velocity_max = 0.1
    
    # Initialize PSO swarm
    swarm = []
    velocities = []
    
    # Initialize particles with feasible solutions
    for _ in range(swarm_size):
        particle = project_feasible(np.random.rand(n_points) * 2)
        swarm.append(particle.copy())
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Initialize personal bests
    personal_bests = [p.copy() for p in swarm]
    personal_best_scores = [compute_C5(p) for p in swarm]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                           cognitive_w * r1 * (personal_bests[i] - swarm[i]) +
                           social_w * r2 * (global_best - swarm[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate fitness
            fitness = compute_C5(swarm[i])
            
            # Update personal best
            if fitness < personal_best_scores[i]:
                personal_best_scores[i] = fitness
                personal_bests[i] = swarm[i].copy()
                
                # Update global best
                if fitness < global_best_score:
                    global_best_score = fitness
                    global_best = swarm[i].copy()
        
        # Keep track of best solution so far
        if global_best_score < best_C5:
            best_C5 = global_best_score
            best_h = global_best.copy()
    
    # Refinement phase with L-BFGS-B
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try refining with L-BFGS-B
    try:
        refined_start = global_best.copy()
        result = minimize(objective, refined_start, method='L-BFGS-B',
                         bounds=bounds, constraints=constraints,
                         tol=1e-6, options={'maxiter': 300})
        
        if result.success:
            refined_h = project_feasible(result.x)
            refined_C5 = compute_C5(refined_h)
            
            if refined_C5 < best_C5:
                best_C5 = refined_C5
                best_h = refined_h.copy()
    except:
        pass
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, n_points)
