import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        initial_h = project(initial_h, dx)
        
    # Simulated Annealing
    h_current = initial_h.copy()
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # Annealing parameters
    T = 0.1
    alpha = 0.95
    cooling_interval = 500
    
    # Start with a larger scale and decrease it
    current_scale = 0.1
    
    iteration = 0
    while time.time() - t0 < max_time:
        if iteration % cooling_interval == 0:
            T *= alpha
            current_scale = max(0.01, current_scale * 0.99)
            
        # Generate a candidate
        delta = np.random.normal(0, current_scale, n_points)
        h_candidate = h_current + delta
        h_candidate = project(h_candidate, dx)
        
        # Evaluate candidate
        C_candidate = evaluate(h_candidate, dx)
        
        # Accept or reject
        if C_candidate < best_C:
            best_C = C_candidate
            best_h = h_candidate.copy()
            
        if C_candidate < best_C or np.random.rand() < np.exp(-(C_candidate - best_C) / T):

        iteration += 1
    
    # Final project and evaluation
    final_h = project(best_h, dx)
