import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    best_C = evaluate(h_current, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.95

    scale = 0.1

    while time.time() - t0 < max_time and iteration < max_iter:
        # Generate candidate solution
        delta = np.random.normal(0, scale, n_points)
        h_candidate = h_current + delta
        h_candidate = project(h_candidate, dx)
        
        # Evaluate candidate
        C_candidate = evaluate(h_candidate, dx)
        
        # Accept or reject
        if C_candidate < best_C:
            best_C = C_candidate
            best_h = h_candidate.copy()
            h_current = h_candidate.copy()

            # Accept with probability
            if np.random.rand() < np.exp(-(C_candidate - best_C) / T):
                h_current = h_candidate.copy()
        
        # Update temperature and scale
        if iteration % 500 == 0:
            T = max(min_T, T * cooling_rate)
            # Gradually decrease scale
            scale = max(0.01, scale * 0.99)
        
        iteration += 1
    
    # Final projection and evaluation
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
