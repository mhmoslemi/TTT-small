import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize h from initial_h_values or create a triangular construction
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO parameters
    swarm_size = 20
    max_iter = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_bounds = 0.1
    
    # Initialize PSO
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for _ in range(swarm_size):
        particle = project_feasible(np.random.rand(n_points))
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))
        personal_best.append(particle.copy())
        personal_best_scores.append(compute_C5(particle))
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main iteration
    iter_count = 0
    patience = 0
    max_patience = 50
    
    while elapsed() < budget_s * 0.9 and iter_count < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                           cognitive * r1 * (personal_best[i] - particles[i]) +
                           social * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_bounds, velocity_bounds)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair infeasibility
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            score = compute_C5(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    patience = 0
                else:
                    patience += 1
            
            # Early stopping
            if patience > max_patience:
                break
                
        iter_count += 1
        if patience > max_patience:
            break
    
    # Refine using L-BFGS-B
    if elapsed() < budget_s * 0.95:
        def objective(h_vec):
            return compute_C5(h_vec)
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        constraint = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize with L-BFGS-B using the best solution from PSO as starting point
        result = minimize(objective, global_best, method='L-BFGS-B', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 300, 'ftol': 1e-6})
        
        if result.success:
            refined_h = project_feasible(result.x)
            refined_C5 = compute_C5(refined_h)
            if refined_C5 < global_best_score:
                global_best = refined_h
                global_best_score = refined_C5
                
    final_C5 = compute_C5(global_best)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = global_best.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
