import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, num_samples=20):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        # Get the actual shift that causes maximum correlation
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Finite difference with perturbations along diagonal
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    def trust_region_update(h_vec, eta=0.01, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Apply structured turbulence to escape local minima
    def apply_turbulence(h_vec, amplitude=0.05):
        # Apply sinusoidal perturbation in a structured way
        pert = amplitude * np.sin(np.pi * np.arange(len(h_vec)) / len(h_vec)) * np.random.normal(0, 1, len(h_vec))
        h_new = h_vec + pert
        return project_feasible(h_new)

    # Main optimization loop
    eta = 0.01
    patience = 0
    max_patience = 30
    iter_count = 0
    last_best_update = 0
    
    # Phase 1: Turbulent exploration (most of the time)
    while elapsed() < budget_s * 0.95:
        # Apply turbulence periodically
        if iter_count % 10 == 0:
            h = apply_turbulence(h, amplitude=0.03)
            
        # Local refinement with trust region
        h = trust_region_update(h, eta)
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Random restart if stuck
        if patience > max_patience or (iter_count - last_best_update > 50):
            # Reinitialize with a random perturbation of current best
            h = np.clip(best_h + np.random.normal(0, 0.02, len(best_h)), 0, 1)
            h = project_feasible(h)
            patience = 0
            if iter_count - last_best_update > 100:
                # Much more aggressive restart
                h = project_feasible(np.random.rand(len(h)) * 0.5)
                h = h * (n_points / 2) / np.sum(h)

        # Adaptive learning rate decay
        if iter_count % 20 == 0:
            eta = max(0.001, eta * 0.97)
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
