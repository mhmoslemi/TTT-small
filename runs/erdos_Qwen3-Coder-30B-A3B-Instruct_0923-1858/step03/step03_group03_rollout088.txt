import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()

        # Create a triangular initial solution
        n_points = 45

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def update_step(h_vec, learning_rate=0.02):
        # Compute gradient using finite differences
        eps = 1e-5
        h_vec = np.array(h_vec)
        current_corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(current_corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        # Update with gradient
        h_new = h_vec - learning_rate * grad
        h_new = project_feasible(h_new)
        return h_new

    # Adaptive learning rate and trust region
    max_iter = 1000
    eta = 0.02
    
    for i in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Adaptive update
        h_new = update_step(h, eta)
        
        # Check if new solution is better
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            h = h_new
            eta = min(eta * 1.1, 0.1)  # Increase step size
        else:
            eta = max(eta * 0.9, 1e-5)  # Decrease step size
            # Perturb to escape local minima
            h = project_feasible(h + np.random.normal(0, 0.005, len(h)))

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
