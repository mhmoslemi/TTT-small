import time
from scipy.fftpack import fft, ifft
from scipy.signal import correlate

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def optimize_frequency_domain():
        h_freq = fft(h)
        # Frequency domain optimization to minimize max correlation
        learning_rate = 0.01
        momentum = 0.9
        velocity = np.zeros_like(h_freq)
        
        # Eigenvalue estimation for adaptation
        max_iter = 1000
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute IFFT of h * conj(h) product in frequency domain
            h_conj = np.conj(h_freq)
            product = h_freq * h_conj
            corr_ifft = ifft(product).real
            max_idx = np.argmax(np.abs(corr_ifft))
            max_val = corr_ifft[max_idx]
            
            # Gradient in frequency domain for max correlation
            grad = np.zeros_like(h_freq)

                grad = np.zeros_like(h_freq)

    # Final evaluation

    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
