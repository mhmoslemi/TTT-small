        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlation manually to ensure correctness
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # The maximum correlation is at the center
        max_k_idx = len(h_vec) - 1
        return corr[max_k_idx] * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing with adaptive temperature
    T = 0.1
    alpha = 0.999
    iter_count = 0
    patience = 0
    max_patience = 500
    
    # Grid refinement parameters
    refinement_interval = 1000
    refinement_factor = 2

    while elapsed() < budget_s * 0.95:
        iter_count += 1
        
        # Perturb solution
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        h_new[idx] += delta
        h_new = project_feasible(h_new)
        
        # Compute C5 for new solution
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            patience = 0
        else:
            # Accept with probability based on temperature
            delta_C5 = new_C5 - best_C5
            if np.random.rand() < np.exp(-delta_C5 / T):
                h = h_new
            patience += 1

        # Decrease temperature
        T *= alpha
        
        # Grid refinement
        if iter_count % refinement_interval == 0 and iter_count > 0:
