import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_gradient(h_vec, eps=1e-4):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        grad = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            delta = np.zeros_like(h_vec)
            delta[i] = eps
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad

    def update_with_gradient(h_vec, eta=0.01, steps=10):
        h_new = h_vec.copy()
        for _ in range(steps):
            grad = compute_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    def apply_turbulence(h_vec, amplitude=0.1):
        h_new = h_vec.copy()
        turbulence = amplitude * np.sin(np.linspace(0, np.pi, len(h_vec))) * np.random.normal(0, 1, len(h_vec))
        h_new = h_new + turbulence
        h_new = project_feasible(h_new)
        return h_new

    # Main optimization loop
    iter_count = 0
    turbulence_frequency = 50
    max_patience = 50
    patience = 0
    last_best_update = 0
    
    while elapsed() < budget_s * 0.95:
        # Apply turbulence with periodic gradient refinement
        if iter_count % turbulence_frequency == 0:
            h = apply_turbulence(h, amplitude=0.05)
            h = update_with_gradient(h, eta=0.005, steps=10)
        else:
            h = apply_turbulence(h, amplitude=0.02)
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Restart the solution if stuck
        if patience > max_patience:
            try:
                # Use convex relaxation as random restart
                relaxed_h = np.random.rand(n_points)
                relaxed_h = project_feasible(relaxed_h)
                relaxed_h = update_with_gradient(relaxed_h, eta=0.005, steps=10)
                if compute_C5(relaxed_h) < compute_C5(h):
                    h = relaxed_h.copy()
            except:
                pass
            patience = 0

        iter_count += 1

    # Final refinement
    final_h = update_with_gradient(best_h, eta=0.001, steps=50)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
