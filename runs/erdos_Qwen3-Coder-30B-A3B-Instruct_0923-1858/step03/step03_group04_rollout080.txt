import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)

        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use global optimization for better exploration
    def objective(h_vec):
        return compute_C5(h_vec)
    
    # Handle constraints in a way that maintains feasibility
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=200,
        popsize=15,
        tol=1e-6,
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None,  # dummy callback to avoid warnings
        disp=False
    )
    
    if not de_result.success:
        h = h.copy()
    else:
        h = project_feasible(de_result.x)
    
    # Use L-BFGS-B for local refinement if we have time
    if elapsed() < budget_s * 0.9:
        # Refinement with local search
        def obj_and_grad(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k = np.argmax(corr)
            max_val = corr[max_k] * dx
            
            # Simple gradient estimator via finite differences
            grad = np.zeros_like(h_vec)
            eps = 1e-5
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (corr_plus[max_k] - corr_minus[max_k]) / (2*eps) * dx
            
            return max_val, grad

        # Local refine
        try:
            res = minimize(
                lambda h_vec: obj_and_grad(h_vec)[0],
                h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                jac=lambda h_vec: obj_and_grad(h_vec)[1],
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-5}
            )
            h = project_feasible(res.x)
        except:
            pass

    # Final evaluation
    final_C5 = compute_C5(h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
