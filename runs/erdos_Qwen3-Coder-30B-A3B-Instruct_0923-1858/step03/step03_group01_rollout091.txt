import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

        initial_h = np.round(initial_h).astype(int)

        initial_h = np.zeros(n_points, dtype=int)
        initial_h[:n_points//2] = 1
        np.random.shuffle(initial_h)

    # Population-based approach
    population_size = 100
    population = []
    fitness = []
    
    # Initialize population with random binary vectors satisfying constraints
    for _ in range(population_size):
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        population.append(h)
    
    # Add the initial solution
    population.append(initial_h)
    population = population[:population_size]
    
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    t0 = time.time()
    no_improvement_count = 0
    prev_best_C = best_C
    
    for iteration in range(1000):
        if time.time() - t0 > budget_s - 1:
            break
            
        # Compute fitness for current population
        fitness = []
        for h in population:
            C = evaluate(h, dx)
            fitness.append(1.0 / (C + 1e-10))  # Avoid division by zero
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)[::-1]
        selected_indices = sorted_indices[:population_size//2]
        selected_population = [population[i] for i in selected_indices]
        
        # Crossover: generate new population
        new_population = selected_population.copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected_population, 2, replace=False)
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Ensure exactly n_points//2 ones
            def fix_binary_vector(h):
                ones_count = np.sum(h)
                if ones_count > n_points//2:
                    # Zero out some ones
                    zero_indices = np.where(h == 1)[0]
                    np.random.shuffle(zero_indices)
                    for i in range(ones_count - n_points//2):
                        h[zero_indices[i]] = 0
                elif ones_count < n_points//2:
                    # One some zeros
                    one_indices = np.where(h == 0)[0]
                    np.random.shuffle(one_indices)
                    for i in range(n_points//2 - ones_count):
                        h[one_indices[i]] = 1
                return h
            
            child1 = fix_binary_vector(child1)
            child2 = fix_binary_vector(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        
        # Mutation: flip 1% of bits for each individual
        for h in population:
            for i in range(n_points):
                if np.random.rand() < 0.01:
                    h[i] = 1 - h[i]
            # Ensure it still satisfies constraint
            h = fix_binary_vector(h)
        
        # Update best solution
        for h in population:
            C = evaluate(h, dx)
            if C < best_C:
                best_C = C
                best_h = h.copy()
                
        # Check for convergence
        if abs(prev_best_C - best_C) < 1e-5:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        prev_best_C = best_C
        
        # Early stopping for stagnation
        if no_improvement_count > 100:
            break
    
    # Final evaluation
    final_C = evaluate(best_h, dx)
    return (best_h, final_C, n_points)
