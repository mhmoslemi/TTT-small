import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

            h[i] = max(0, 1 - abs(x[i] - 1))

        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Bisection-based approach
    C_low = 0.0
    C_high = 1.0
    epsilon = 1e-4
    max_iter = 500  # Bisections
    
    # Sample shifts for constraint discretization
    sample_shifts = np.linspace(-2, 2, 50)  # Sample 50 shifts
    sample_shifts = sample_shifts[sample_shifts != 0]  # Exclude zero

    def solve_feasibility(C_val):

        h_vars = cp.Variable(n_points)
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add the cross-correlation constraints
        for shift in sample_shifts:
            # Approximate the cross-correlation constraint using linearization
            # For small shifts, we can approximate using finite differences or 
            # precomputed values from coherence minimization (simpler for now)
            # We simplify by adding a direct constraint that approximates this
            # A proper constraint would be nonlinear and harder to solve            
            pass  # No explicit constraints added for now to keep this convex
        
        # This is a coarse approach: Using a small subset of shift constraints
        # Directly minimizing C would be better but requires more adaptive number of shifts
        # So we simply focus on a sufficient set of cross-corr values
        
        try:
            # Solve relaxed problem - the main constraint is that h is in feasible range
            # and the sum is correct
            object = cp.Minimize(cp.sum_squares(h_vars - 0.5))  # Encourage spread
            prob = cp.Problem(object, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)

        C_mid = (C_low + C_high) / 2.0
        
        # Solve for feasibility at C_mid
        h_test = solve_feasibility(C_mid)
        
        # Evaluate result
        test_C5 = compute_C5(h_test)
        
        if test_C5 <= C_mid:
            # Feasible: Try to decrease C
            C_high = C_mid
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = h_test.copy()
        else:
            # Infeasible: Need to increase C
            C_low = C_mid
            
        iter_count += 1

    # Final refinement using gradient-based approach if time allows
    if elapsed() < budget_s * 0.98:
        # Start from the best found and refine it
        h_refined = best_h.copy()

                h_plus = h_new + delta
                h_minus = h_new - delta
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_corr_idx] - corr_minus[max_corr_idx]) / (2 * eps)
            
            h_new = h_new - step_size * grad
            h_new = project_feasible(h_new)
            
            new_C5 = compute_C5(h_new)

                best_C5_refined = new_C5
                h_refined = h_new.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10:
                step_size *= 0.9
            
            if abs(prev_C - new_C5) < 1e-6:
                break
            prev_C = new_C5
            
        if best_C5_refined < best_C5:
            best_C5 = best_C5_refined
            best_h = h_refined.copy()
            
    return (best_h.tolist(), best_C5, len(best_h))
