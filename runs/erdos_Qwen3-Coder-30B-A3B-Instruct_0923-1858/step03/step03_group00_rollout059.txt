import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        # Evaluate
        current_C = evaluate(h_candidate, dx)

        if current_C < best_C or np.random.rand() < np.exp(-(current_C - best_C) / T):
            h_current = h_candidate
            if current_C < best_C:
                best_C = current_C

        if iter_count % 5000 == 0 and time.time() - t0 < max_time - 10:

                    h_plus[j] += eps
                    h_plus = project(h_plus, dx)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    h_minus = best_h.copy()
                    h_minus[j] -= eps
                    h_minus = project(h_minus, dx)
                    corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                    grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)

    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
