import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Initialize particle swarm
    num_particles = 50
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    
    # Set first particle to initial_h_values
    particles[0] = h.copy()
    
    # Initialize other particles with small Gaussian perturbations
    for i in range(1, num_particles):
        pert = np.random.normal(0, 0.05, n_points)
        particles[i] = h + pert
        particles[i] = project_feasible(particles[i])
    
    # Track personal bests
    p_best = particles.copy()
    p_best_fitness = np.array([compute_C5(p) for p in particles])
    
    # Global best
    best_idx = np.argmin(p_best_fitness)
    s_best = p_best[best_idx].copy()
    s_best_fitness = p_best_fitness[best_idx]
    
    # PSO parameters
    inertia = 0.9
    cognitive_weight = 2.0
    social_weight = 2.0
    max_velocity = 0.1
    
    # Evolution parameters
    max_iter = 2000
    patience = 0
    max_patience = 30
    decay_rate = 0.999
    
    # Main loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        # Update inertia
        if iteration > 0:
            inertia = max(0.4, inertia * decay_rate)
            
        # Update each particle
        for i in range(num_particles):
            # Compute velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (p_best[i] - particles[i])
            social = social_weight * r2 * (s_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < s_best_fitness:
                    s_best = particles[i].copy()
                    s_best_fitness = fitness
                    patience = 0
            else:
                patience += 1
                
        # Check for stagnation
        if patience > max_patience:
            # Perturb all particles
            for i in range(num_particles):
                pert = np.random.normal(0, 0.03, n_points)
                particles[i] = particles[i] + pert
                particles[i] = project_feasible(particles[i])
            patience = 0

    # Final check
    final_fitness = compute_C5(s_best)
    if final_fitness < s_best_fitness:
        s_best_fitness = final_fitness

    return (s_best.tolist(), s_best_fitness, n_points)
