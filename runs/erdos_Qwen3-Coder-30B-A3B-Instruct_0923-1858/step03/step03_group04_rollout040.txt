import time
from scipy.signal import correlate

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Virtual perturbation
    def perturb_solution(h_vec, prob=0.2):
        h_new = h_vec.copy()
        idxs = np.random.choice(n_points, size=int(prob * n_points), replace=False)
        for i in idxs:
            h_new[i] += np.random.normal(0, 0.03)
        return project_feasible(h_new)

    # Simulated Annealing Inspired wandering
    def wander(h_vec, temp=0.1):
        h_new = h_vec.copy()
        # Apply sinusoidal shape with random noise to create exploration
        for i in range(n_points):
            # Dynamic adjustment with sine pattern
            amp = temp * (0.5 + 0.5 * np.sin(np.pi * i / (n_points - 1)))
            h_new[i] += np.random.normal(0, amp)
        return project_feasible(h_new)

    # Gradient estimation based on finite differences
    def estimate_gradient(h_vec, eps=1e-3, samples=5):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        for _ in range(samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = project_feasible(h_vec + delta)
            h_minus = project_feasible(h_vec - delta)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / samples

        grad = estimate_gradient(h_vec)
        h_new = h_vec - learning_rate * grad
