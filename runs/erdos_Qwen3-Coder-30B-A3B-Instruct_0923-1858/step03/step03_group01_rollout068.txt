    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def tournament_selection(population, fitnesses, k):
    """Tournament selection with k participants."""
    indices = np.random.choice(len(population), k, replace=False)
    best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Single-point crossover that maintains exactly n_points//2 ones."""
    size = len(parent1)
    crossover_point = np.random.randint(1, size)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Ensure constraint of exactly n_points//2 ones
    if np.sum(child1) != size // 2:
        ones_to_flip = np.sum(child1) - size // 2
        if ones_to_flip > 0:
            # Remove ones
            zero_indices = np.where(child1 == 0)[0]
            flip_indices = np.random.choice(zero_indices, int(ones_to_flip), replace=False)
            child1[flip_indices] = 1
        else:
            # Add ones
            one_indices = np.where(child1 == 1)[0]
            flip_indices = np.random.choice(one_indices, int(-ones_to_flip), replace=False)
            child1[flip_indices] = 0
            
    if np.sum(child2) != size // 2:
        ones_to_flip = np.sum(child2) - size // 2
        if ones_to_flip > 0:
            zero_indices = np.where(child2 == 0)[0]
            flip_indices = np.random.choice(zero_indices, int(ones_to_flip), replace=False)
            child2[flip_indices] = 1
        else:
            one_indices = np.where(child2 == 1)[0]
            flip_indices = np.random.choice(one_indices, int(-ones_to_flip), replace=False)
            child2[flip_indices] = 0
    
    return child1, child2

def mutate(individual, mutation_rate, n_points):
    """Flip bits with mutation_rate while maintaining constraint."""
    individual = individual.copy()
    num_mutations = int(mutation_rate * n_points)
    
    # Identify 0s and 1s
    zeros = np.where(individual == 0)[0]
    ones = np.where(individual == 1)[0]
    
    # Flip some 0s to 1s and some 1s to 0s
    if len(zeros) > 0 and num_mutations > 0:
        flip_zeros = np.random.choice(zeros, min(num_mutations // 2, len(zeros)), replace=False)
        individual[flip_zeros] = 1
    
    if len(ones) > 0 and num_mutations > 0:
        flip_ones = np.random.choice(ones, min(num_mutations - num_mutations // 2, len(ones)), replace=False)
        individual[flip_ones] = 0
    
    # Ensure exactly n_points//2 ones by redistributing
    current_ones = np.sum(individual)
    needed = n_points // 2 - current_ones
    
    if needed > 0:
        # Add needed ones
        zero_indices = np.where(individual == 0)[0]
        add_indices = np.random.choice(zero_indices, needed, replace=False)
        individual[add_indices] = 1
    elif needed < 0:
        # Remove excess ones
        one_indices = np.where(individual == 1)[0]
        remove_indices = np.random.choice(one_indices, -needed, replace=False)
        individual[remove_indices] = 0
        
    return individual
