import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def crossover(parent1, parent2):
    """Perform one-point crossover between two parents."""
    n = len(parent1)
    point = np.random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(individual, mutation_rate, generation, max_generations):
    """Mutate an individual with decreasing mutation scale."""
    n = len(individual)
    mutation_scale = 0.1 * (1 - generation / max_generations)
    for i in range(n):
        if np.random.random() < mutation_rate:
            individual[i] += np.random.normal(0, mutation_scale)
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 20
    mutation_rate = 0.05
    max_generations = 5000
    
    try:

        if len(initial_h) != n_points:

    except NameError:

        initial_h = 0.3 + 0.2 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.mean(initial_h)
    
    # Create initial population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = project(h, dx)
        population.append(h)
    
    # GA parameters
    elite_size = 2
    tournament_size = 5
    
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    t0 = time.time()
    max_time = budget_s - 2
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - t0 < max_time and no_improvement_count < max_no_improvement:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            fitness = evaluate(h, dx)
            fitness_scores.append(fitness)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best
        if fitness_scores[0] < best_C:
            best_C = fitness_scores[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Create new population
        new_population = population[:elite_size]
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_index].copy()
            parent2 = population[np.random.choice(population_size)].copy()
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_rate, generation, max_generations)
            child2 = mutate(child2, mutation_rate, generation, max_generations)
            
            # Project to constraints
            child1 = project(child1, dx)
            child2 = project(child2, dx)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        generation += 1
    
    # Final verification
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
