import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def binary_evaluate(h, dx):
    """Evaluate C5 for binary h."""
    # Convert to float for correlation
    h_float = h.astype(float)
    corr = correlate(h_float, 1 - h_float, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
