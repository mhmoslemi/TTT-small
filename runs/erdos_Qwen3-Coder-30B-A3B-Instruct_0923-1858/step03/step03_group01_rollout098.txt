import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        corr = correlate(h_current, 1 - h_current, mode='full')

        # So we find the minimum value among all non-trivial correlations
        # Isolate non-trivial correlations (those with the center spike)
        # In full mode, this will be at middle indices
        corr_center = corr[len(corr)//2:]
        k_min = np.argmin(corr_center)
        # Convert k_min to the actual lag index for correlated sequence
        k = k_min - len(corr)//2
        
        # Gradient approximation for the minimum correlation
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)

            h_minus[j] -= eps

            # Compute the contribution to minimum correlation
            grad[j] = (corr_plus[len(corr_plus)//2 + k] - corr_minus[len(corr_minus)//2 + k]) / (2 * eps)
        
        # Update step with adaptive learning rate
        lr = 0.02 * (1.0 - i / max_iter)
        h_current = h_current - lr * grad
        h_current = project(h_current, dx)
        
        # Check improvement
        current_C = evaluate(h_current, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_current.copy()
    
    return (best_h, best_C, n_points)
