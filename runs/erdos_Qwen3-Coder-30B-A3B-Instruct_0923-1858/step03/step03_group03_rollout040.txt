import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 30
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing with grid refinement
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iter_count = 0
    patience = 0
    max_patience = 500
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        # Perturb solution
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_new[idx] += delta
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Compute C5
        C5_new = compute_C5(h_new)
        C5_old = compute_C5(h)
        
        # Accept or reject
        if C5_new < C5_old or np.random.rand() < np.exp(-(C5_new - C5_old) / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Grid refinement
        if iter_count > 0 and iter_count % 1000 == 0:
            if n_points < 80:  # Avoid too many points
                # Double grid resolution
                n_points_new = min(80, n_points * 2)
                x_old = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, n_points_new)
                h_new_interp = np.interp(x_new, x_old, h)
                h = project_feasible(h_new_interp)
                n_points = n_points_new
                dx = 2.0 / n_points
        
        # Update temperature
        T *= alpha
        iter_count += 1
        
        if patience > max_patience:
            break

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
