import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Frequency-domain approach
    def optimize_frequency_domain():
        h_freq = fft(h)
        # Use gradient descent in the frequency domain
        learning_rate = 0.1
        max_iter = 500
        patience = 0
        max_patience = 20
        last_best_C5 = float('inf')
        for i in range(max_iter):
            if elapsed() >= budget_s * 0.95:
                break
            # Update in frequency domain to minimize max correlation
            # Approximate gradient as negative of current correlation
            corr = correlate(h, 1-h, mode='full')
            max_idx = np.argmax(corr)
            # Construct gradient: negative of correlation gradient
            # This is a heuristic, direct gradient is complex for non-convex
            if max_idx < len(h):
                # Shift in positive direction
                grad = np.zeros_like(h)
                grad[:-max_idx] += corr[max_idx+1:]
                grad[-max_idx:] += corr[:max_idx]
            else:
                # Shift in negative direction
                shift = max_idx - len(h) + 1
                grad = np.zeros_like(h)
                grad[shift:] += corr[:len(h)-shift]
                grad[:shift] += corr[len(h)-shift:]
            
            # Apply pseudo-gradient update with frequency domain regularization
            grad_freq = fft(grad)
            h_freq = h_freq - learning_rate * grad_freq 
            # Inverse transform to get h vector
            h = ifft(h_freq).real
            h = project_feasible(h)
            C5 = compute_C5(h)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
            # Early stopping if no improvement
            if patience > max_patience:
                break
            # Adaptive learning rate

                learning_rate *= 0.95

    if elapsed() < budget_s * 0.8:
        try:
            h_opt = optimize_frequency_domain()
            if compute_C5(h_opt) < compute_C5(h):
                h = h_opt.copy()
        except Exception:
            pass

    # Apply small stochastic perturbations to escape local minima
    if elapsed() < budget_s * 0.95:

            num_pert = max(1, int(0.1 * n_points))

            for i in indices:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.01), 0, 1)
            
            h = project_feasible(h)
            C5 = compute_C5(h)
            if C5 < best_C5:
                best_C5 = C5

            if pert_count % 50 == 0 and elapsed() < budget_s * 0.9:
                try:
                    h = optimize_frequency_domain()
                except Exception:
                    pass

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, len(best_h))
