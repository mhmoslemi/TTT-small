import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO parameters
    swarm_size = 20
    max_iter = 500
    inertia_w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Generate swarm
    for _ in range(swarm_size):
        # Random feasible solution
        h_p = h + np.random.normal(0, 0.05, n_points)
        h_p = project_feasible(h_p)
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(h_p.copy())
        pbest_vals.append(compute_C5(h_p))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO Optimization
    iter_count = 0
    while elapsed() < budget_s * 0.5 and iter_count < max_iter:
        for i in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) +
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Position update
            particles[i] += velocities[i]
            
            # Repair feasibility
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            curr_val = compute_C5(particles[i])
            
            # Update personal best
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
            
            # Update global best
            if curr_val < gbest_val:
                gbest = particles[i].copy()
                gbest_val = curr_val
                
        iter_count += 1
    
    # Refinement with L-BFGS-B if time allows
    if elapsed() < budget_s * 0.9:
        # Use the best PSO solution as initial point
        x0 = gbest.copy()
        
        # Define constraints
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Minimize (we want to minimize the max overlap, so we use negative)
        def obj_func(x):
            # Return the maximum overlap
            corr = correlate(x, 1 - x, mode="full")
            return -np.max(corr) * dx
        
        # L-BFGS-B optimization
        try:
            res = minimize(obj_func, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                          options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-5})
            if res.success:
                refined_h = project_feasible(res.x)
                refined_C5 = compute_C5(refined_h)
                if refined_C5 < gbest_val:
                    gbest = refined_h
                    gbest_val = refined_C5
        except:
            pass
    
    # Final computation and return
    final_C5 = compute_C5(gbest)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = gbest.copy()
    
    return (best_h.tolist(), best_C5, n_points)
