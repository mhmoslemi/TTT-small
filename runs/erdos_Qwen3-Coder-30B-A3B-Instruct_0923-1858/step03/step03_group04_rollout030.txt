import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)

        # Create initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_min, v_max = -0.1, 0.1

    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    pbest_values = []

    # Initialize swarm with diverse solutions
    for _ in range(swarm_size):
        if _ == 0:
            init_h = h.copy()
        else:
            # Add some noise to the initial solution
            init_h = h + np.random.normal(0, 0.03, n_points)
        init_h = project_feasible(init_h)
        swarm.append(init_h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(init_h.copy())
        pbest_values.append(compute_C5(init_h))

    # Global best
    gb_idx = np.argmin(pbest_values)
    global_best = personal_best[gb_idx].copy()
    global_best_value = pbest_values[gb_idx]

    # PSO main loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], v_min, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            current_value = compute_C5(swarm[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                personal_best[i] = swarm[i].copy()
            
            # Update global best
            if current_value < global_best_value:
                global_best_value = current_value
                global_best = swarm[i].copy()

    # L-BFGS-B refinement with the global best from PSO
    def objective(h_vec):
        return compute_C5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Refinement
    result = minimize(
        objective,
        global_best,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'maxiter': 300}
    )

    refined_h = project_feasible(result.x)
    final_C5 = compute_C5(refined_h)

    return (refined_h.tolist(), final_C5, n_points)
