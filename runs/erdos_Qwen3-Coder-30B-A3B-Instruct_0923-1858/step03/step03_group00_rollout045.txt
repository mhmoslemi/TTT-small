import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        worst_lag = max_idx - (len(h) - 1)
        
        # Formulate QP to minimize overlap at the worst lag
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == 1.0 / dx
        ]
        
        # Compute correlation for the specific lag
        corr_expr = 0
        if worst_lag >= 0:
            for i in range(len(h) - worst_lag):
                corr_expr += h_var[i] * (1 - h_var[i + worst_lag])
        else:
            for i in range(-worst_lag, len(h)):
                corr_expr += h_var[i] * (1 - h_var[i + worst_lag])
        
        # Minimize the overlap at this lag
        objective = cp.Minimize(corr_expr)

                        best_h = h_new.copy()
                    h = h_new.copy()
        except:
            pass
    
    # Final validation and return
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
