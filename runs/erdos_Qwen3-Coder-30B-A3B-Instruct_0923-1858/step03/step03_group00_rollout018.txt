import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 90
    dx = 2.0 / n_points

        initial_h = 0.3 + 0.4 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)

    while time.time() - t0 < max_time:
        # Gradient estimation using finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps

            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Gradient step with learning rate decay and clipping

        if current_C < best_C:
            best_C = current_C
            best_h = h_new.copy()
        
        h_current = h_new.copy()
        
        # Learning rate decay
        if iteration % 100 == 0:
            lr *= 0.95
        
        iteration += 1
  
    # Final projection
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
