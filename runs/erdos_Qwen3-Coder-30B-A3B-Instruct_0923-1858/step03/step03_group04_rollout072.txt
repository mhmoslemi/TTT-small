import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use convex optimization to find a better starting point
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Minimize sum of squares to get a "spread-out" solution
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    # Gradient estimation with better finite differences
    def get_gradient(h_vec, num_samples=15):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, len(h_vec))
            h_plus = h_vec + delta
            h_minus = h_vec - delta
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
        return grad / num_samples

    # Trust region update with adaptive step size
    def trust_region_update(h_vec, eta=0.01, max_iter=15):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Perturbation to escape local minima
    def perturb_solution(h_vec, prob=0.2):
        h_new = h_vec.copy()
        idxs = np.random.choice(n_points, size=int(prob * n_points), replace=False)
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.05), 0, 1)
        return project_feasible(h_new)

    # Try convex relaxation first for warm start
    if elapsed() < budget_s * 0.6:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass
    
    # Main optimization loop
    eta = 0.01
    patience = 0
    max_patience = 25
    iter_count = 0
    last_best_update = 0

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Check if we're stuck
        if patience > max_patience or (iter_count - last_best_update > 40):
            h = perturb_solution(h)
            patience = 0
            if iter_count - last_best_update > 80:
                # Restart with convex relaxation
                try:
                    relaxed_h = solve_convex_relaxation()
                    if compute_C5(relaxed_h) < compute_C5(h):
                        h = relaxed_h.copy()
                except:
                    pass

        # Optimize using trust region step
        h = trust_region_update(h, eta)
        
        # Adaptive learning rate decay
        if iter_count % 20 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
