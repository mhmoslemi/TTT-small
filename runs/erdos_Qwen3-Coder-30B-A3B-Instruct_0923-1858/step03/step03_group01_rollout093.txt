import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        max_idx = np.argmax(fitness_scores)

            final_solution = temp_solution.copy()
            final_fitness_value = temp_fitness
    
    final_C = evaluate(final_solution, dx)
    
    return (final_solution.astype(float), final_C, n_points)
