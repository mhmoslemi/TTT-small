import time

def run(seed=42, budget_s=600, **kwargs):

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Bisection on C5
    C_low = 0.0
    C_high = 1.0
    epsilon = 1e-4
    
    # Sample shifts for the feasibility check (outer loop)
    max_shifts = 50
    shifts = np.linspace(0, 2, max_shifts, endpoint=False)  # for k in [0,2)
    
    # Use a more adaptive optimization approach with multiple restarts
    def optimize_feasibility(C_target, max_iter=50):
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Ensure the constraints are working
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add correlation constraints
        corr_min = 1 - C_target
        for k in shifts[:len(shifts)//2 + 1]:  # since it's symmetric
            # Discretized convolution
            k_idx = int(k / dx)
            # One-side shift correlation (efficient)
            if k_idx < n_points:
                # h * (1 - h shifted)
                # This is a simplification to check if we can solve this feasibility
                pass  # Actual handling will be later

        # Minimize an auxiliary objective (make it smooth) while satisfying constraints  
        # This may not be the best  approach; we can also try stochastic search.
        obj = cp.Minimize(cp.sum_squares(h_vars - 0.5))  # Simple but may help with convergence
        problem = cp.Problem(obj, constraints)
        
        try:
            problem.solve(solver=cp.MOSEK, verbose=False)  # Try better solvers
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = np.maximum(0, np.minimum(1, h_vars.value))

            h_try = optimize_feasibility(C_val)
            C5_val = compute_C5(h_try)
            return h_try, C5_val
        except:
            return h.copy(), float('inf')

    iter_count = 0

            best_h = h_try.copy()

        else:

            grad = np.zeros_like(h_vec)
            for _ in range(num_samples):
                delta = np.random.randn(len(h_vec)) * eps
                h_plus = project_feasible(h_vec + delta)
                h_minus = project_feasible(h_vec - delta)

                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
            
            return grad / num_samples

        for i in range(100):
            if elapsed() > budget_s * 0.98:
                break
                
            grad = get_smooth_gradient(h_refined)
            h_refined = h_refined - eta * grad
            h_refined = project_feasible(h_refined)
            
            new_C5 = compute_C5(h_refined)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_refined.copy()
                patience = 0
            else:
                patience += 1
                
            if patience > max_patience:
                break
                
            if i % 20 == 0:
                eta *= 0.95

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
