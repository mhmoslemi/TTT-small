import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set the discretization
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with the provided initial construction if available
    try:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Reconstruct with correct size using linear interpolation
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, initial_h_values)
    except NameError:
        # Create a symmetric triangular initial solution
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Initialize best solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # PSO parameters
    swarm_size = 20
    max_iter_pso = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Add initial solution as first particle
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h.copy())
    pbest_vals.append(compute_C5(h))
    
    # Generate additional random particles
    for _ in range(swarm_size - 1):
        # Random perturbations
        pert = np.random.normal(0, 0.05, n_points)
        tmp_h = h + pert
        tmp_h = project_feasible(tmp_h)
        particles.append(tmp_h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(tmp_h.copy())
        pbest_vals.append(compute_C5(tmp_h))
    
    # Find global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    # Track last improvements
    last_improvement = 0
    patience = 0
    max_patience = 50
    
    # PSO iteration
    for iter_pso in range(max_iter_pso):
        if elapsed() > budget_s * 0.6:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            current_val = compute_C5(particles[i])
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    last_improvement = iter_pso
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
                
            # Perturb stagnated particles
            if patience > max_patience:
                particles[i] = project_feasible(h + np.random.normal(0, 0.03, n_points))
                pbest[i] = particles[i].copy()
                pbest_vals[i] = compute_C5(particles[i])
                patience = 0
                
        # Early termination if no improvement in a while
        if iter_pso - last_improvement > 150:
            break
    
    # Refinement with L-BFGS-B
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Use the best PSO solution as starting point for L-BFGS
    h_refine = gbest.copy()
    
    # Create bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Refinement loop with early exit
    try:
        # Use scipy.optimize.minimize with L-BFGS-B
        result = minimize(objective, h_refine, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6})
        h_refine = result.x
        
        # Ensure feasibility of refined solution
        h_refine = project_feasible(h_refine)
        refined_C5 = compute_C5(h_refine)
        
        # Update if better
        if refined_C5 < best_C5:
            best_h = h_refine.copy()
            best_C5 = refined_C5
            
    except Exception:
        pass
    
    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, n_points)
