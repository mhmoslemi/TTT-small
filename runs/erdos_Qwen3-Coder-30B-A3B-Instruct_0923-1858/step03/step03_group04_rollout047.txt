def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Try to improve solution
    def maximize_minimum_overlap():
        # Bisection approach: find maximum C such that there exists h with all correlations >= 1-C
        # But we're trying to minimize max correlation, so we search for minimum C
        # Let's instead formulate this as a constrained optimization
        # For a given h, we want min_k int h(x)(1-h(x+k)) dx to be minimized.
        # So let's define a slightly different approach, inspired by duality:
        # If we're in a point (h, C) such that max_k int h(x)(1-h(x+k)) dx <= C,
        # then it's a constraint that makes sense to optimize.
        
        # Try to find a better h that minimizes the worst overlap.
        # Go with a two-phase approach: gradient-based with local search + tiny mutations

        max_iter = 5000
        eta = 0.005
        patience = 0
        max_patience = 100
        last_best_update = 0

        current_h = h.copy()
        current_C5 = compute_C5(current_h)
        best_h = current_h.copy()
        best_C5 = current_C5

        # Mutual function for gradient
        def eval_gradient(h_vec, samples=10):
            h_vec = np.array(h_vec)
            # Use finite differences
            eps = 1e-4
            grad = np.zeros_like(h_vec)
            # Determine which k gives max correlation for this h
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            k_max = np.argmax(corr) - (len(h_vec) - 1)  # shift corresponding to max
            for _ in range(samples):
                delta = np.random.normal(0, eps, len(h_vec))
                h_plus = project_feasible(h_vec + delta)
                h_minus = project_feasible(h_vec - delta)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad += (corr_plus[np.argmax(corr)] - corr_minus[np.argmax(corr)]) / (2 * eps) * delta
            return grad / samples

        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break

            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
                patience = 0
                last_best_update = iteration
            else:
                patience += 1

            if patience > max_patience or iteration > last_best_update + 200:
                # Perturb
                current_h = project_feasible(current_h + np.random.normal(0, 0.01, len(current_h)))
                patience = 0
                if iteration > last_best_update + 500:
                    # Reset with triangle
                    x = np.linspace(0, 2, n_points)
                    h = np.zeros(n_points)
                    for i in range(n_points):
                        h[i] = max(0, 1 - abs(x[i] - 1))
                    h = h * (n_points / 2) / np.sum(h)
                    current_h = h.copy()
                    patience = 0
            else:
                gradients = eval_gradient(current_h, 8)  
                current_h = current_h - eta * gradients
                current_h = project_feasible(current_h)
            # Decrease learning rate
            if iteration % 100 == 0:
                eta *= 0.95

        return best_h, best_C5

    # Run optimization
    h_opt, C5_value = maximize_minimum_overlap()
    return (h_opt.tolist(), C5_value, len(h_opt))
