import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available, otherwise create a modified triangular solution
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - 1.5 * np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Helper to compute overlap integrals
    def compute_overlaps(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Only valid lags k in [-n+1, n-1]
        valid_corr = corr[len(h_vec)-1:]
        return valid_corr * dx
    
    # Helper to compute C5 bound
    def compute_C5(h_vec):
        overlaps = compute_overlaps(h_vec)
        return np.max(overlaps)
    
    # Project h to feasible region
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Solve a quadratic program to refine optimal shift k
    def solve_qp_refinement(h_vec, k_max):
        h_new = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Constraints
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0
        ]
        
        # Linearized constraint for the most constraining overlap
        h_current = np.array(h_vec)
        grad_k = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max >= 0 and i + k_max < n_points:
                grad_k[i] = (1 - h_current[i + k_max])
            if i - k_max >= 0 and i - k_max < n_points:
                grad_k[i] -= h_current[i - k_max]
        
        # Linear approximation of overlap at current h
        current_overlap = np.sum(h_current * (1 - np.roll(h_current, k_max)))
        constraints.append(t >= current_overlap + np.dot(grad_k, h_new - h_current))
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_new.value
        except:
            pass
        return h_vec
    
    # Main optimization loop
    patience = 0
    max_patience = 30
    iter_count = 0
    eta = 0.01
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb solution
                perturb_mask = np.random.random(n_points) < 0.1
                h += np.random.normal(0, 0.05, n_points) * perturb_mask
                h = project_feasible(h)
                patience = 0

        # Find most constraining shift
        overlaps = compute_overlaps(h)
        k_max = np.argmax(overlaps)
        
        # Solve QP for this shift
        h_new = solve_qp_refinement(h, k_max)
        
        # Accept improvement
        h_new = project_feasible(h_new)
        if compute_C5(h_new) < compute_C5(h):
            h = h_new
        else:
            # Allow small degradation to escape local minima
            if np.random.random() < 0.1:
                h = project_feasible(h + np.random.normal(0, 0.02, n_points))
                
        iter_count += 1

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
