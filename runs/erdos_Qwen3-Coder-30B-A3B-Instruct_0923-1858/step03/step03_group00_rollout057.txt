import numpy as np
import time
import math

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    alpha = 0.95
    min_T = 1e-6
    cooling_interval = 500

            T *= alpha
            scale = max(0.01, scale * alpha)

        delta = np.random.normal(0, scale, n_points)
        h_candidate = h_current + delta
        h_candidate = project(h_candidate, dx)
        
        # Evaluate
        current_C = evaluate(h_candidate, dx)

        if current_C < best_C:

            if np.random.rand() < math.exp(-(current_C - best_C) / T):

    t_refine = time.time()

            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
