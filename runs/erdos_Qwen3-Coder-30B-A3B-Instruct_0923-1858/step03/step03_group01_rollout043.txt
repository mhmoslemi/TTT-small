import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def is_valid_binary(h, n_points):
    """Check if h is a valid binary vector with correct sum."""
    return np.all((h == 0) | (h == 1)) and np.sum(h) == n_points / 2

def initialize_population(n_points, pop_size):
    """Initialize a population of binary vectors."""
    population = []
    target_ones = int(n_points / 2)
    for _ in range(pop_size):
        h = np.zeros(n_points, dtype=int)
        ones_indices = np.random.choice(n_points, target_ones, replace=False)
        h[ones_indices] = 1
        population.append(h)
    return population

def selection(population, fitnesses, num_select):
    """Tournament selection."""
    selected = []
    for _ in range(num_select):
        tournament_indices = np.random.choice(len(population), 5, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2):
    """Single-point crossover with feasibility check."""
    n_points = len(parent1)
    point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Fix if the children do not have correct number of ones
    target_ones = n_points // 2
    if np.sum(child1) != target_ones:
        ones_indices = np.where(child1 == 1)[0]
        zeros_indices = np.where(child1 == 0)[0]
        diff = target_ones - np.sum(child1)
        if diff > 0:
            # Need to add ones
            np.random.shuffle(zeros_indices)
            child1[zeros_indices[:diff]] = 1
        else:
            # Need to remove ones
            np.random.shuffle(ones_indices)
            child1[ones_indices[:abs(diff)]] = 0
    
    if np.sum(child2) != target_ones:
        ones_indices = np.where(child2 == 1)[0]
        zeros_indices = np.where(child2 == 0)[0]
        diff = target_ones - np.sum(child2)
        if diff > 0:
            np.random.shuffle(zeros_indices)
            child2[zeros_indices[:diff]] = 1
        else:
            np.random.shuffle(ones_indices)
            child2[ones_indices[:abs(diff)]] = 0
    
    return child1, child2

def mutate(h, mutation_rate):
    """Flip bits with given mutation rate."""
    n_points = len(h)
    num_mutations = int(n_points * mutation_rate)
    mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
    for idx in mutation_indices:
        h[idx] = 1 - h[idx]
    
    # Ensure we still have exactly target_ones ones
    target_ones = n_points // 2
    current_ones = np.sum(h)
    if current_ones != target_ones:
        if current_ones > target_ones:
            # Remove extra ones
            ones_indices = np.where(h == 1)[0]
            np.random.shuffle(ones_indices)
            for idx in ones_indices[:int(current_ones - target_ones)]:
                h[idx] = 0
        else:
            # Add missing ones
            zeros_indices = np.where(h == 0)[0]
            np.random.shuffle(zeros_indices)
            for idx in zeros_indices[:int(target_ones - current_ones)]:
                h[idx] = 1
    
    return h
