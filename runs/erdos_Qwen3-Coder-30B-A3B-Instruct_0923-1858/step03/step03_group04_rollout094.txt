import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Gradient for correlation sum
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if i + max_k >= 0 and i + max_k < len(h_vec):
                grad[i] += (1 - h_vec[i + max_k]) * dx
            if i - max_k >= 0 and i - max_k < len(h_vec):
                grad[i] += h_vec[i - max_k] * dx
        
        return grad

    def add_turbulence(h_vec, amplitude=0.05):
        # Apply a sinusoidal turbulence
        turbulence = amplitude * np.sin(np.pi * np.arange(len(h_vec)) / len(h_vec)) * np.random.randn(len(h_vec))
        h_turb = h_vec + turbulence
        return project_feasible(h_turb)

    # Main optimization loop
    iter_count = 0
    patience = 0
    max_patience = 50
    last_best_update = 0

    while elapsed() < budget_s * 0.95:
        # Apply turbulence to introduce exploration
        h_turb = add_turbulence(h, amplitude=0.07)
        current_C5 = compute_C5(h_turb)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_turb.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Gradient-based refinement
        if iter_count % 10 == 0:
            num_grad_steps = min(5, max(1, int(100 * (1 - elapsed() / budget_s))))
            h_refined = h_turb.copy()
            for _ in range(num_grad_steps):
                grad = get_gradient(h_refined)
                h_refined = h_refined - 0.005 * grad
                h_refined = project_feasible(h_refined)
            refined_C5 = compute_C5(h_refined)
            if refined_C5 < best_C5:
                best_C5 = refined_C5
                best_h = h_refined.copy()
                patience = 0
                last_best_update = iter_count

        # Restart with new turbulence when too many iterations without improvement
        if patience > max_patience:
            # Perturb more aggressively
            h = add_turbulence(h, amplitude=0.15)
            patience = 0
        else:
            h = h_turb

        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
