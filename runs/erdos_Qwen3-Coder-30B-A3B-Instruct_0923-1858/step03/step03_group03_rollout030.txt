import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    patience = 0
    max_patience = 50

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def adaptive_local_search(h_vec, max_iter=200):
        h_current = h_vec.copy()
        
        def objective(h_candidate):
            h_candidate = project_feasible(h_candidate)
            return compute_C5(h_candidate)
        
        def constraint_sum(h_candidate):
            return np.sum(h_candidate) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, h_current, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': max_iter, 'ftol': 1e-6, 'gtol': 1e-6})
        
        if result.success:
            return project_feasible(result.x)
        return h_current

    # Precompute shifts for faster correlation
    shifts = list(range(1, n_points))  # Exclude 0 shift (no offset)

    # Heuristic: refine solution with both adaptation and mutation
    last_C5 = compute_C5(h)
    improved = True
    iter_count = 0
    
    while elapsed() < budget_s * 0.95 and improved:
        improved = False
        
        # Local optimization
        h_opt = adaptive_local_search(h, max_iter=50)
        C5_opt = compute_C5(h_opt)
        
        if C5_opt < last_C5:
            h = h_opt
            last_C5 = C5_opt
            improved = True
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Perturb
            h = project_feasible(h + np.random.normal(0, 0.02, n_points))
            patience = 0
        
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h.copy()
            
        iter_count += 1
    
    # Final refinement
    if elapsed() < budget_s * 0.98:
        final_h = adaptive_local_search(best_h, max_iter=100)
        final_C5 = compute_C5(final_h)
        if final_C5 < best_C5:
            best_h = final_h
            best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
