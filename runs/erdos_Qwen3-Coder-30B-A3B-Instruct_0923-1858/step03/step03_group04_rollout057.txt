import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO parameters
    swarm_size = 20
    max_iterations = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize particles
    for _ in range(swarm_size):
        if elapsed() > budget_s * 0.6:
            break
        particle = project_feasible(h + np.random.normal(0, 0.02, n_points))
        swarm.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_vals.append(compute_C5(particle))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    if gbest_val < best_C5:
        best_C5 = gbest_val
        best_h = gbest.copy()
    
    # PSO main loop
    for iteration in range(max_iterations):
        if elapsed() > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            current_val = compute_C5(swarm[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
                    
                    if gbest_val < best_C5:
                        best_C5 = gbest_val
                        best_h = gbest.copy()
        
        # Occasionally add noise to particles to avoid getting stuck
        if iteration % 20 == 0:
            for i in range(swarm_size):
                if np.random.random() < 0.3:
                    swarm[i] = project_feasible(swarm[i] + np.random.normal(0, 0.01, n_points))
    
    # L-BFGS-B refinement
    def lbfgsb_refinement(initial_h):
        if elapsed() > budget_s * 0.98:
            return initial_h
            
        def objective(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return -np.max(corr) * dx  # Negative because we want to minimize
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(objective, initial_h, method='L-BFGS-B', 
                            bounds=bounds, constraints=cons, 
                            options={'ftol': 1e-6, 'maxiter': 300})
            if result.success:
                refined_h = project_feasible(result.x)
                return refined_h
        except:
            pass
        return initial_h
    
    # Refine the global best with L-BFGS-B
    if elapsed() < budget_s * 0.98:
        refined_h = lbfgsb_refinement(gbest)
        refined_C5 = compute_C5(refined_h)
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
