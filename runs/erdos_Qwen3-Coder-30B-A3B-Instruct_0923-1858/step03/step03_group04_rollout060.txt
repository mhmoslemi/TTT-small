import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Frequency domain optimization approach
    def optimize_frequency_domain():
        # Initialize h
        h_current = h.copy()
        h_freq = np.fft.fft(h_current)
        
        # Frequency domain optimization parameters
        learning_rate = 0.01
        max_iter = 200
        decay_factor = 0.995
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute correlation in frequency domain
            # Overlap is IFFT(H * conj(H))
            corr_freq = h_freq * np.conj(h_freq)
            corr_time = np.fft.ifft(corr_freq).real
            
            # Find maximum correlation and its index
            max_idx = np.argmax(corr_time)
            max_corr = corr_time[max_idx]
            
            # Compute gradient in frequency domain
            # We approximate it as -2 * H * conj(H) * sign(dH/dt) 
            # but simpler: adjust H to reduce the largest correlate
            gradient_freq = np.zeros_like(h_freq)
            gradient_freq[max_idx - (n_points - 1)] = -np.sign(max_corr)
            h_freq = h_freq - learning_rate * gradient_freq
            
            # Check for convergence
            if i % 20 == 0 and i > 0:
                learning_rate *= decay_factor
                
            # Convert back to time domain
            h_current = np.fft.ifft(h_freq).real
            h_current = project_feasible(h_current)
            
            # Update frequency domain representation
            h_freq = np.fft.fft(h_current)
            
            # Evaluate current solution
            current_C5 = compute_C5(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
        
        return best_h, best_C5

    # Enhanced local search
    def perturb_solution(h_vec, prob=0.15):
        h_new = h_vec.copy()
        idxs = np.random.choice(n_points, size=int(prob * n_points), replace=False)
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.02), 0, 1)
        return project_feasible(h_new)
    
    # Run frequency domain optimization
    final_h, final_C5 = optimize_frequency_domain()
    
    # Local search refinement
    h_refined = final_h.copy()
    for i in range(100):  # Limited iterations for refinement
        if elapsed() > budget_s * 0.95:
            break
        h_trial = perturb_solution(h_refined)
        new_C5 = compute_C5(h_trial)
        if new_C5 < final_C5:
            final_C5 = new_C5
            final_h = h_trial.copy()
        # Occasionally check for a better solution
        h_refined = final_h.copy()
    
    return (final_h.tolist(), final_C5, len(final_h))
