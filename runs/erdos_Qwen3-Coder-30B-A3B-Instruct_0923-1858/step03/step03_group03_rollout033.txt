import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        # Create a smooth initial solution
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def objective(h_vec):
        return compute_C5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    # Direct optimization approach
    def optimize_step(h_vec, max_iter=30):
        # Create a reasonable starting point for scipy.optimize
        h_init = h_vec.copy()
        opt_result = minimize(objective, h_init, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': max_iter, 'ftol': 1e-8})
        if opt_result.success:
            h_new = project_feasible(opt_result.x)
        else:
            h_new = project_feasible(h_vec)
        return h_new

    # Fine-tune with gradient-based approach
    def refine_solution(h_vec, max_iter=20):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            h_new = optimize_step(h_new, max_iter=5)
            if elapsed() > budget_s * 0.95:
                break
        return h_new

    # Multi-initialization random search to avoid local optima
    def multi_start_search():
        nonlocal best_C5, best_h
        best_h = h.copy()
        best_C5 = compute_C5(best_h)
        
        # Try several random initializations
        for _ in range(10):
            h_init = np.random.uniform(0, 1, n_points)
            h_init = project_feasible(h_init)
            h_opt = refine_solution(h_init, max_iter=15)
            c5 = compute_C5(h_opt)
            
            if c5 < best_C5:
                best_C5 = c5
                best_h = h_opt.copy()
            
            if elapsed() > budget_s * 0.95:
                break
        
        return best_h, best_C5

    # Perform main search loop
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        h_opt = refine_solution(h.copy(), max_iter=20)
        c5 = compute_C5(h_opt)
        
        if c5 < best_C5:
            best_C5 = c5
            best_h = h_opt.copy()
        
        # Very simple random perturbation to escape local minima
        h_pert = h_opt.copy()
        idx = np.random.randint(0, n_points)
        h_pert[idx] += np.random.normal(0, 0.02)
        h_pert = project_feasible(h_pert)
        
        c5_pert = compute_C5(h_pert)
        if c5_pert < best_C5:
            best_C5 = c5_pert
            best_h = h_pert.copy()
            
        h = h_opt
        iter_count += 1

    # Final polishing with optimizer
    final_h, final_C5 = multi_start_search()
    
    return (final_h.tolist(), final_C5, len(final_h))
