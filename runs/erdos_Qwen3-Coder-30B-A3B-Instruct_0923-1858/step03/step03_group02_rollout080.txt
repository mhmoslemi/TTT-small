import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize h from parent or create a new one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create triangular-like initial guess with sharper decay
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_overlap_and_grad(h_vec, k):
        # Compute correlation for specific lag k
        h_vec = np.array(h_vec)
        lag = k
        if lag >= 0:
            # h[i] * (1 - h[i+lag]) for valid indices
            # We compute it manually for performance reasons
            corr_val = 0.0
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec) - lag):
                corr_val += h_vec[i] * (1 - h_vec[i + lag])
                grad[i] += (1 - h_vec[i + lag])
                grad[i + lag] -= h_vec[i]
        else:
            lag = -lag  # now positive
            # h[i+lag] * (1 - h[i]) for valid indices
            corr_val = 0.0
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec) - lag):
                corr_val += h_vec[i + lag] * (1 - h_vec[i])
                grad[i + lag] += (1 - h_vec[i])
                grad[i] -= h_vec[i + lag]
        return corr_val * dx, grad * dx

    def solve_qp_constraint(h_current, k_max, t_prev):
        # Solve the QP: minimize t such that t >= C_k_max and constraints hold
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= t_prev  # Ensure we do not make things worse 
        ]
        
        # Get linearized constraint for k_max
        corr_val, grad = get_overlap_and_grad(h_current, k_max)
        constraints.append(t >= corr_val + grad.T @ (h_new - h_current))
        
        # Solve QP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new_val = h_new.value
                t_val = t.value
                return h_new_val, t_val
        except:
            pass
        return h_current, t_prev

    # Main optimization loop
    eta = 0.01
    patience = 0
    max_patience = 30
    iter_count = 0
    t_current = compute_C5(h)
    
    while elapsed() < budget_s * 0.95:
        # Compute all overlaps and find max to linearize
        overlaps = []
        for k in range(1 - n_points, n_points):  # all possible shifts
            corr_val, _ = get_overlap_and_grad(h, k)
            overlaps.append((k, corr_val))
        
        # Find k_max that gives maximum overlap
        k_max = max(overlaps, key=lambda x: x[1])[0]
        max_overlap = max(overlaps, key=lambda x: x[1])[1]
        
        # Solve QP to linearize the most constraining constraint
        h_new, t_new = solve_qp_constraint(h, k_max, t_current)
        
        # Update only if the new solution is better and feasible
        h_new = project_feasible(h_new)
        t_new = compute_C5(h_new)
        
        if t_new < t_current:
            h = h_new
            t_current = t_new
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Random perturb
            for i in range(n_points):
                if np.random.random() < 0.1:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
            h = project_feasible(h)
            patience = 0

        iter_count += 1

    # Final check and projection
    final_h = project_feasible(h)
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
