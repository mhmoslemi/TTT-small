import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Convex relaxation approach to get better initialization
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve a refined relaxation: minimize sum of squares
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h.copy()

    # Improved gradient computation
    def compute_gradient(h_vec, max_k_index):
        # We compute finite differences
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Project to feasible
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_index] - corr_minus[max_k_index]) / (2 * eps)
        
        return grad

    # Trust region-like update
    def trust_region_update(h_vec, eta=0.01):
        h_new = h_vec.copy()
        corr = correlate(h_new, 1 - h_new, mode="full")
        max_k_idx = np.argmax(corr)
        grad = compute_gradient(h_new, max_k_idx)
        h_new = h_new - eta * grad
        h_new = project_feasible(h_new)
        return h_new

    # Random perturbation function with structured elements
    def perturb_solution(h_vec, strength=0.1):
        h_new = h_vec.copy()
        # Apply small Gaussian noise
        noise = np.random.normal(0, strength, len(h_vec))
        h_new += noise
        # Project back to feasibility
        h_new = project_feasible(h_new)
        return h_new

    # Warm start with convex relaxation if possible
    if elapsed() < budget_s * 0.3:
        try:
            relaxed_h = solve_convex_relaxation()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass

    # Main optimization loop with turbulence and refinement
    iter_count = 0
    last_best_update = 0
    patience = 0
    max_patience = 25
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        if patience > max_patience or (iter_count - last_best_update > 50):
            # Perturbation to escape local minima
            h = perturb_solution(h, strength=0.03)
            patience = 0
            if iter_count - last_best_update > 100:
                # Restart with convex relaxation if it's been a while
                try:
                    relaxed_h = solve_convex_relaxation()
                    if compute_C5(relaxed_h) < compute_C5(h):
                        h = relaxed_h.copy()
                except:
                    pass
        
        # Q: Use trust region refinement every 15 iterations
        if iter_count % 15 == 0 and iter_count > 0:
            # Apply several trust region steps
            eta = 0.01
            for _ in range(10):
                h = trust_region_update(h, eta)
                # Occasionally reduce learning rate
                if _ % 5 == 0:
                    eta *= 0.92
            
        # Regular random perturbation to maintain diversity
        h = perturb_solution(h, strength=0.015)
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
