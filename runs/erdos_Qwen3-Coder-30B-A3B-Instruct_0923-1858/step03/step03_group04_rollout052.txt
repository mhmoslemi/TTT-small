import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO parameters
    swarm_size = 20
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    personal_best = []
    personal_best_vals = []
    global_best = None
    global_best_val = float('inf')
    
    for _ in range(swarm_size):
        # Random feasible solution
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        swarm.append(h_init.copy())
        personal_best.append(h_init.copy())
        personal_best_vals.append(compute_C5(h_init))
        
        if personal_best_vals[-1] < global_best_val:
            global_best = h_init.copy()
            global_best_val = personal_best_vals[-1]
    
    # PSO main loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            v = (inertia * (swarm[i] - personal_best[i]) + 
                 cognitive * r1 * (personal_best[i] - swarm[i]) + 
                 social * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            v = np.clip(v, -velocity_max, velocity_max)
            
            # Update position
            new_pos = swarm[i] + v
            
            # Repair infeasibility
            new_pos = project_feasible(new_pos)
            
            # Evaluate
            new_val = compute_C5(new_pos)
            
            # Update personal best
            if new_val < personal_best_vals[i]:
                personal_best[i] = new_pos.copy()
                personal_best_vals[i] = new_val
                
                if new_val < global_best_val:
                    global_best = new_pos.copy()
                    global_best_val = new_val
                    
            swarm[i] = new_pos.copy()
    
    # Refinement using L-BFGS-B
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use the best from PSO as initial guess
    refined_h = global_best.copy()
    
    # L-BFGS-B optimization
    try:
        result = minimize(objective, refined_h, method='L-BFGS-B', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6})
        if result.success:
            refined_h = result.x
    except:
        pass
    
    # Final evaluation
    final_C5 = compute_C5(refined_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = refined_h.copy()
    
    # Ensure final solution is feasible and within bounds
    best_h = project_feasible(best_h)
    
    return (best_h.tolist(), best_C5, len(best_h))
