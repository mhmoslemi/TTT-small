import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

        # Create a triangular initial solution
        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use a trust-region like approach with bounds
    def refine_step(h_vec, max_iter=100, learning_rate=0.01):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Estimate gradient via finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-4
            current_C5 = compute_C5(h_new)
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                max_k_idx = np.argmax(corr_plus)
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - learning_rate * grad
            h_new = project_feasible(h_new)
            
            if i % 10 == 0:
                new_C5 = compute_C5(h_new)
                if new_C5 < current_C5:
                    current_C5 = new_C5
                else:
                    learning_rate *= 0.9
        return h_new

    # Hybrid local search
    iter_count = 0
    convergence_patience = 0
    max_patience = 100
    eta = 0.02
    final_refinement_steps = 5
    last_improvement = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            convergence_patience = 0
            last_improvement = iter_count
        else:
            convergence_patience += 1
            if convergence_patience > max_patience or iter_count - last_improvement > 200:
                # Perturb if no significant improvement in a while
                h = project_feasible(h + np.random.normal(0, 0.01, len(h)))
                convergence_patience = 0

        # Refinement step
        h = refine_step(h, max_iter=30, learning_rate=eta)
        iter_count += 1
        
        # Adaptively decrease learning rate
        if iter_count % 10 == 0:
            eta *= 0.97

    # Final refinement 
    for _ in range(final_refinement_steps):
        if elapsed() >= budget_s * 0.99:
            break
        h = refine_step(h, max_iter=50, learning_rate=0.005)
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
