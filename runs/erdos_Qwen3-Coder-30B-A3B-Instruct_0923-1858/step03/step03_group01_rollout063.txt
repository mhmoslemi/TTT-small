import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    h_current = project(h_current, dx)
    
    t0 = time.time()
    
    # Use gradient descent with Adam-style update and better convergence
    max_iter = 100000
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    # Adam variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    for i in range(max_iter):
        if time.time() - t0 > budget_s - 1:

        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        # Sample a subset of indices to reduce computation
        sample_indices = np.random.choice(n_points, size=min(20, n_points), replace=False)
        
        for j in sample_indices:
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (i + 1))
        v_hat = v / (1 - beta2 ** (i + 1))
        
        # Update parameters
        lr_adjusted = lr / (np.sqrt(v_hat) + epsilon)
        h_current = h_current - lr_adjusted * m_hat
        h_current = project(h_current, dx)
        
        # Track best solution
        current_C = evaluate(h_current, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_current.copy()
            
    return (best_h, best_C, n_points)
