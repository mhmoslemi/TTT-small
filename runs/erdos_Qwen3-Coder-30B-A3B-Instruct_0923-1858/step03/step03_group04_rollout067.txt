import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Add turbulence to escape local minima
    def apply_turbulence(h_vec, amplitude=0.1, frequency=1.0):
        h_new = h_vec.copy()
        # Apply sinusoidal turbulence
        x = np.linspace(0, 2*np.pi*frequency, n_points)
        noise = amplitude * np.sin(x) * np.random.normal(0, 1, n_points)
        h_new += noise
        h_new = project_feasible(h_new)
        return h_new
        
    # Gradient computation via finite differences for local refinement
    def compute_gradient(h_vec, eps=1e-4, num_samples=5):
        grad = np.zeros_like(h_vec)
        current_C5 = compute_C5(h_vec)
        
        for _ in range(num_samples):
            delta = np.random.normal(0, eps, n_points)
            h_plus = project_feasible(h_vec + delta)
            h_minus = project_feasible(h_vec - delta)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            max_k_idx = np.argmax(corr_plus)
            grad += (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * delta
        
        return grad / num_samples
        
    # Gradient descent step with projection
    def gradient_descent_step(h_vec, eta=0.01):
        grad = compute_gradient(h_vec)
        h_new = h_vec - eta * grad
        return project_feasible(h_new)

    # Main optimization loop
    iter_count = 0
    turbulence_freq = 50
    patience = 0
    max_patience = 30
    last_best_update = 0
    turbulence_amp = 0.08
    eta = 0.01

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        if patience > max_patience or (iter_count - last_best_update > 30):
            # Random restart with turbulence
            h = apply_turbulence(h, amplitude=0.15, frequency=1.0)
            patience = 0

        # Apply turbulence periodically
        if iter_count % turbulence_freq == 0 and iter_count > 0:
            h = apply_turbulence(h, amplitude=turbulence_amp, frequency=1.0)
            # Decrease turbulence amplitude over time
            turbulence_amp *= 0.98
        else:
            # Gradient descent step
            h = gradient_descent_step(h, eta=eta)
        
        # Adaptive learning rate
        if iter_count % 10 == 0 and iter_count > 0:
            eta *= 0.95

        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
