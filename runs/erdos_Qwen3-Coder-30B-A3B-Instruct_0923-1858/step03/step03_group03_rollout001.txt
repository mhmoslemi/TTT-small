        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, num_shifts=20):

        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute gradient direction through finite differences
        grad = np.zeros_like(h_vec)

        h_minus = h_vec.copy()

        for i in range(len(h_vec)):
            h_plus[i] = h_vec[i] + eps

            h_minus = project_feasible(h_minus)
