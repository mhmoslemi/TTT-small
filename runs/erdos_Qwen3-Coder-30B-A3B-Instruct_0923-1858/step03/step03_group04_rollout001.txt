import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with provided construction or create a symmetric one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Phase 1: Particle Swarm Optimization for global exploration
    def pso_optimization():
        # PSO parameters
        n_particles = 20
        max_iter = 1000
        inertia = 0.7
        cognitive = 1.5
        social = 1.5
        velocity_max = 0.1
        
        # Initialize swarm
        swarm = []
        velocities = []
        personal_best = []
        personal_best_vals = []
        global_best = None
        global_best_val = float('inf')
        
        for _ in range(n_particles):
            # Initial swarm positions with small random perturbations
            h_init = project_feasible(h + np.random.normal(0, 0.02, n_points))
            swarm.append(h_init.copy())
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
            val = compute_C5(h_init)
            personal_best.append(h_init.copy())
            personal_best_vals.append(val)
            if val < global_best_val:
                global_best = h_init.copy()
                global_best_val = val

        # PSO main loop
        for iter_count in range(max_iter):
            if elapsed() > budget_s * 0.7:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2, n_points)
                velocities[i] = (inertia * velocities[i] + 
                                cognitive * r1 * (personal_best[i] - swarm[i]) + 
                                social * r2 * (global_best - swarm[i]))
                
                # Apply max velocity
                velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
                
                # Update position
                swarm[i] += velocities[i]
                
                # Repair infeasibility
                swarm[i] = project_feasible(swarm[i])
                
                # Evaluate
                current_val = compute_C5(swarm[i])
                if current_val < personal_best_vals[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_vals[i] = current_val
                    if current_val < global_best_val:
                        global_best = swarm[i].copy()
                        global_best_val = current_val
                        
        return global_best, global_best_val

    # Phase 2: Local refinement with L-BFGS
    def lbfgsb_refinement(initial_h):
        def objective(h_vec):
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return -np.max(corr) * dx  # We want to minimize the negative (maximize max correlation)

        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2  # Need sum to be n_points/2

        # Set bounds
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_func}
        
        # Optimize
        result = minimize(objective, 
                         initial_h, 
                         method='L-BFGS-B', 
                         bounds=bounds, 
                         constraints=constraints,
                         options={'maxiter': 300, 'ftol': 1e-6})
        
        if result.success:
            refined_h = project_feasible(result.x)
        else:
            refined_h = initial_h.copy()
        
        return refined_h

    # Run PSO first
    best_h_pso, best_C5_pso = pso_optimization()
    
    # Refine with L-BFGS
    if elapsed() < budget_s * 0.95:
        refined_h = lbfgsb_refinement(best_h_pso)
        final_C5 = compute_C5(refined_h)
        if final_C5 < best_C5_pso:
            best_h = refined_h
            best_C5 = final_C5
        else:
            best_h = best_h_pso
            best_C5 = best_C5_pso
    else:
        best_h = best_h_pso
        best_C5 = best_C5_pso

    # Ensure final feasibility
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
