import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def add_turbulence(h_vec, amplitude=0.1, frequency=1):
        # Apply sinusoidal turbulence
        turbulence = amplitude * np.sin(np.pi * np.arange(n_points) * frequency / n_points) * np.random.normal(0, 1, n_points)
        h_new = h_vec + turbulence
        return project_feasible(h_new)

    def compute_gradient(h_vec, k=None):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr) if k is None else k + (len(h_vec) - 1)
        # Approximate gradient using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

    def gradient_descent_step(h_vec, learning_rate=0.01, steps=20):
        h_new = h_vec.copy()
        for _ in range(steps):
            grad = compute_gradient(h_new)
            h_new = h_new - learning_rate * grad
            h_new = project_feasible(h_new)
        return h_new

    # Main optimization loop
    iter_count = 0
    patience = 0
    max_patience = 40
    last_best_update = 0
    turbulence_amp = 0.1
    turbulence_freq = 1

    while elapsed() < budget_s * 0.95:
        # Apply turbulence and check for improvement
        h_turb = add_turbulence(h, turbulence_amp, turbulence_freq)
        current_C5 = compute_C5(h_turb)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_turb.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Periodic gradient descent
        if iter_count % 50 == 0 and iter_count > 0:
            h = gradient_descent_step(h_turb, learning_rate=0.01, steps=20)
        else:
            h = h_turb

        # Handle stagnation and restart
        if patience > max_patience or (iter_count - last_best_update > 100):
            # Reinitialize with a new random turbulence
            h = add_turbulence(h, amplitude=0.05, frequency=1)
            patience = 0
            turbulence_amp = max(0.02, turbulence_amp * 0.9)

        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
