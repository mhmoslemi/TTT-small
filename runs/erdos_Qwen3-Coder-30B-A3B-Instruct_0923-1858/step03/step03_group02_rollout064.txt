        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def local_search(h_vec, max_iter=50):
        h_current = h_vec.copy()
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            # Random small perturbations
            h_new = h_current.copy()
            idx = np.random.randint(0, n_points)
            pert = np.random.normal(0, 0.01)
            h_new[idx] = np.clip(h_new[idx] + pert, 0, 1)
            h_new = project_feasible(h_new)
            
            if compute_C5(h_new) < compute_C5(h_current):
                h_current = h_new
        return h_current

        h_current = h_vec.copy()
