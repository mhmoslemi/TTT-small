import time

def run(seed=42, budget_s=600, **kwargs):

    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    #Define the objective function for optimization
    def objective(h_vec):
        return compute_C5(h_vec)

    # Constraints for the optimization
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])

    # Combined constraint for optimization
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda x: x},  # h_i >= 0
        {'type': 'ineq', 'fun': lambda x: 1 - x}  # h_i <= 1
    ]

    def local_optimization(h_guess, max_iter=100):
        h_new = h_guess.copy()
        for i in range(max_iter):
            # Simple gradient descent with projection for bounds
            grad = np.zeros_like(h_new)
            eps = 1e-6
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_plus[j] += eps
                h_minus = h_new.copy()
                h_minus[j] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[j] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
            h_new = h_new - 0.01 * grad
            h_new = project_feasible(h_new)
        return h_new

    # Main loop: hybrid local and global search
    iter_count = 0
    patience = 0
    max_patience = 50
    eta = 0.01

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0

            patience += 1
            if patience > max_patience:
                # Random perturbation if no improvement
                h = project_feasible(h + np.random.normal(0, 0.02, len(h)))
                patience = 0

        # Perform local optimization
        h = local_optimization(h)

        # Adaptive update of learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.97

        iter_count += 1

    # Final refinement
    for _ in range(20):
        h = local_optimization(h, max_iter=20)

    final_C5 = compute_C5(h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
