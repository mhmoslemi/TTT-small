import time
from scipy.signal import correlate

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing with grid refinement
    T = 0.1
    alpha = 0.999
    iter_count = 0
    max_iter = 5000
    patience = 0
    max_patience = 500
    
    while elapsed() < budget_s * 0.95 and T > 1e-6:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Restart with a new random point near the best
                h = best_h + np.random.normal(0, 0.01, n_points)
                h = project_feasible(h)
                patience = 0
        
        # Perturb one random element
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.05)
        h_new = h.copy()
        h_new[idx] += delta
        h_new = project_feasible(h_new)
        
        new_C5 = compute_C5(h_new)
        delta_C5 = new_C5 - current_C5
        
        # Accept with simulated annealing probability
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = h_new
        
        T *= alpha
        iter_count += 1
        
        # Periodic grid refinement
        if iter_count % 1000 == 0 and n_points < 200:

            n_points *= 2

            h = np.interp(x_new, x_orig, h)
            dx = 2.0 / n_points

    # Final evaluation
    final_C5 = compute_C5(best_h)
