import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO Optimization Phase
    def pso_optimization():
        # PSO parameters
        swarm_size = 20
        max_iter = 500
        inertia = 0.7
        cognitive = 1.5
        social = 1.5
        velocity_bound = 0.1
        
        # Initialize swarm
        swarm = []
        velocities = []
        personal_best = []
        personal_best_vals = []
        
        for _ in range(swarm_size):
            h_init = project_feasible(np.random.rand(n_points) * 0.5)
            swarm.append(h_init.copy())
            velocities.append(np.random.uniform(-velocity_bound, velocity_bound, n_points))
            personal_best.append(h_init.copy())
            personal_best_vals.append(compute_C5(h_init))
            
        global_best = swarm[np.argmin(personal_best_vals)].copy()
        global_best_val = min(personal_best_vals)
        
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.45:
                break
                
            for i in range(swarm_size):
                # Update velocity
                r1, r2 = np.random.rand(2, n_points)
                velocities[i] = (inertia * velocities[i] + 
                               cognitive * r1 * (personal_best[i] - swarm[i]) + 
                               social * r2 * (global_best - swarm[i]))
                
                # Bound velocities
                velocities[i] = np.clip(velocities[i], -velocity_bound, velocity_bound)
                
                # Update position
                swarm[i] += velocities[i]
                
                # Repair feasibility
                swarm[i] = project_feasible(swarm[i])
                
                # Evaluate
                current_val = compute_C5(swarm[i])
                
                # Update personal best
                if current_val < personal_best_vals[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_vals[i] = current_val
                
                # Update global best
                if current_val < global_best_val:
                    global_best = swarm[i].copy()
                    global_best_val = current_val
                    
        return global_best, global_best_val

    # L-BFGS-B Refinement Phase
    def lbfgsb_refinement(h_start):
        def objective(h_vec):
            return compute_C5(h_vec)
            
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        # Create bounds (0 to 1) for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Bounds for sum constraint
        cons = {'type': 'eq', 'fun': constraint_func}
        
        try:
            res = minimize(objective, 
                          h_start, 
                          method='L-BFGS-B', 
                          bounds=bounds,
                          constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-6})
            
            if res.success:
                return project_feasible(res.x)
        except:
            pass
        return h_start

    # Run PSO phase
    if elapsed() < budget_s * 0.5:
        try:
            pso_h, pso_C5 = pso_optimization()
            if pso_C5 < best_C5:
                best_C5 = pso_C5
                best_h = pso_h.copy()
        except:
            pass

    # Run L-BFGS-B refinement on best solution so far
    if elapsed() < budget_s * 0.9:
        try:
            refined_h = lbfgsb_refinement(best_h.copy())
            refined_C5 = compute_C5(refined_h)
            if refined_C5 < best_C5:
                best_C5 = refined_C5
                best_h = refined_h.copy()
        except:
            pass

    # Final optimization if more time is available
    if elapsed() < budget_s * 0.95:
        # Try simple random perturbations
        perturbations = 100
        for _ in range(perturbations):
            if elapsed() > budget_s * 0.95:
                break
            h_trial = project_feasible(best_h + np.random.normal(0, 0.01, n_points))
            trial_C5 = compute_C5(h_trial)
            if trial_C5 < best_C5:
                best_C5 = trial_C5
                best_h = h_trial.copy()

    return (best_h.tolist(), best_C5, len(best_h))
