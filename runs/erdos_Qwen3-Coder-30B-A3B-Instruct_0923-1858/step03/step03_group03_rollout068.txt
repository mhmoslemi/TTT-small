import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 30
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iter_count = 0
    patience = 0
    max_patience = 1000
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        
        # Perturb one point
        delta = np.random.normal(0, 0.1)
        h_new[idx] += delta
        
        # Keep within bounds and normalize
        h_new = project_feasible(h_new)
        
        current_C5 = compute_C5(h)
        new_C5 = compute_C5(h_new)
        
        if new_C5 < current_C5:
            h = h_new
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_new.copy()
                patience = 0
        else:
            if np.random.rand() < np.exp(-(new_C5 - current_C5) / T):
                h = h_new
            patience += 1

        if patience > max_patience:
            # Refine grid
            n_points = min(n_points * 2, 1000)
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, np.linspace(0, 2, len(h)), h)
            h = project_feasible(h_new)
            patience = 0
            
        if iter_count % 50 == 0:
            T *= alpha
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
