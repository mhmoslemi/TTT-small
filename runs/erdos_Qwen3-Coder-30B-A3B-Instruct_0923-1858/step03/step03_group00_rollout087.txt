import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_time = budget_s - 2

    # Try to use the initial construction if available
    try:

        if len(initial_h) != n_points:

        initial_h = 0.3 + 0.4 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        initial_h = project(initial_h, dx)
    
    # Initialize population
    population_size = 20
    population = []
    for _ in range(population_size):
        if np.random.random() < 0.5:
            h = initial_h + np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project(h, dx)
        population.append(h)

    # Genetic algorithm
    t0 = time.time()
    generation = 0
    best_fitness = float('inf')
    best_individual = initial_h.copy()

    while time.time() - t0 < max_time:
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate(h, dx)
            fitness.append(fit)
            if fit < best_fitness:
                best_fitness = fit
                best_individual = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        # Elitism
        new_population = population[:2]

        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 3
            idx1 = np.random.choice(len(population[:10]), tournament_size, replace=False)
            parent1 = population[np.argmin([fitness[i] for i in idx1])]
            idx2 = np.random.choice(len(population[:10]), tournament_size, replace=False)
            parent2 = population[np.argmin([fitness[i] for i in idx2])]

            # Crossover
            if np.random.random() < 0.8:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()

            # Mutation
            if np.random.random() < 0.05:
                mutation_scale = max(0.001, 0.05 * (1 - generation / 1000))
                for i in range(n_points):
                    if np.random.random() < 0.05:
                        child[i] += np.random.normal(0, mutation_scale)
                child = project(child, dx)
            else:
                child = project(child, dx)

            new_population.append(child)

        population = new_population
        generation += 1

    # Return final solution
    final_h = project(best_individual, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
