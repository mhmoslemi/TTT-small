import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with existing construction or create new one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Phase 1: PSO for global exploration
    def pso_optimization():
        # PSO Parameters
        swarm_size = 20
        max_iter = 1000
        inertia_w = 0.7
        cognitive_w = 1.5
        social_w = 1.5
        vel_max = 0.1
        
        # Initialize swarm
        swarm = []
        velocities = []
        personal_best = []
        personal_best_values = []
        
        for _ in range(swarm_size):
            # Create random feasible solutions
            h_init = np.random.rand(n_points) * 0.5 + 0.25
            h_init = project_feasible(h_init)
            swarm.append(h_init.copy())
            velocities.append(np.random.uniform(-vel_max, vel_max, n_points))
            personal_best.append(h_init.copy())
            personal_best_values.append(compute_C5(h_init))
        
        # Global best tracking
        global_best_idx = np.argmin(personal_best_values)
        global_best = personal_best[global_best_idx].copy()
        global_best_value = personal_best_values[global_best_idx]
        
        # PSO iterations
        for iter_count in range(max_iter):
            if elapsed() > budget_s * 0.6:
                break
            
            for i in range(swarm_size):
                # Update velocity
                r1, r2 = np.random.rand(2, n_points)
                velocities[i] = (inertia_w * velocities[i] + 
                                cognitive_w * r1 * (personal_best[i] - swarm[i]) +
                                social_w * r2 * (global_best - swarm[i]))
                
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
                
                # Update position
                swarm[i] += velocities[i]
                
                # Repair feasibility
                swarm[i] = project_feasible(swarm[i])
                
                # Evaluate
                current_value = compute_C5(swarm[i])
                
                # Update personal best
                if current_value < personal_best_values[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_value
                    
                    # Update best solution found
                    if global_best_value < best_C5:
                        best_C5 = global_best_value
                        best_h = global_best.copy()
        
        return global_best, global_best_value

    # Phase 2: Local refinement using L-BFGS-B
    def lbfgsb_refinement(initial_h):
        if elapsed() > budget_s * 0.9:
            return initial_h
        
        def objective(h_vec):
            return compute_C5(h_vec)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, initial_h, method='L-BFGS-B', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 300, 'ftol': 1e-9})
        
        if result.success:
            refined_h = project_feasible(result.x)
            return refined_h
        else:
            return initial_h

    # Main optimization loop
    patience = 0
    max_patience = 200
    last_improvement = 0
    
    # Phase 1: PSO optimization
    if elapsed() < budget_s * 0.6:
        pso_h, pso_C5 = pso_optimization()
        if pso_C5 < best_C5:
            best_C5 = pso_C5
            best_h = pso_h.copy()
            
    # Phase 2: Refinement
    if elapsed() < budget_s * 0.95:
        refined_h = lbfgsb_refinement(best_h)
        refined_C5 = compute_C5(refined_h)
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
            
    return (best_h.tolist(), best_C5, len(best_h))
