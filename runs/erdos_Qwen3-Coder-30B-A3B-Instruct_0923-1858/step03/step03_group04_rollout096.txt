import time

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Bisection search with convex optimization
    def solve_feasibility(h_vec, C):
        # Build constraint matrix for cross-correlations
        n_shifts = min(50, n_points)
        shifts = np.linspace(-1, 1, n_shifts)  # Sample reasonable shifts
        # Note: We need a proper way to represent cross correlation constraints in convex form
        # Since this is hard, we'll use coarse relaxation involving fewer shifts
        
        # Use subset of shifts for fast feasibility check
        # Use convex relaxation using CVXPY
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # This is an approximation - for now we add some general constraints 
        # that attempt to enforce sufficient cross-correlation values 
        try:
            objective = cp.Minimize(cp.sum_squares(h_var))  # heuristic to spread out values

            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h_vec

    # Direct optimization approach
    # Target bisection search with fewer shifts for efficiency
    C_low = 0.0
    C_high = 1.0
    tolerance = 1e-4
    best_C5_found = 1.0

        C_mid = (C_low + C_high) / 2.0

        h_trial = h.copy()

        best_h = best_h.copy()
