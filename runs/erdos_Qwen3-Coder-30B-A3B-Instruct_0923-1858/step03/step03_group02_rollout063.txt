import time
from scipy.signal import correlate

        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simple optimization approach using scipy minimize
    def optimize_step(h_vec):
        # Create bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function
        def objective(h_new):
            h_new = np.array(h_new)
            corr = correlate(h_new, 1 - h_new, mode="full")
            return -np.max(corr) * dx  # negative because we minimize

        # Constraint
        def constraint_sum(h_new):
            return np.sum(h_new) - n_points / 2

        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize
        result = minimize(objective, h_vec, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 50, 'ftol': 1e-8})
        
        if result.success:
            return result.x
        return h_vec

            patience = 0

                # Random perturbation
                h = project_feasible(h + np.random.normal(0, 0.02, n_points))

        h = optimize_step(h)
