import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Coarse initial grid
        n_points = 30
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    current_h = h.copy()
    
    # Precompute correlation indices once
    full_length = 2 * n_points - 1
    full_corr_idx = np.arange(full_length)
    
    def compute_C5(h_vec):
        # Compute full correlation
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def perturb(h_vec, strength=0.1):
        h_new = h_vec.copy()
        idx = np.random.randint(0, len(h_vec))
        h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)
    
    # Simulated Annealing with grid refinement
    T = 0.1
    alpha = 0.999
    iter_count = 0
    max_iter = int(budget_s * 1000)  # Rough estimate
    
    while iter_count < max_iter and elapsed() < budget_s * 0.95:
        # Perturb
        new_h = perturb(current_h, strength=0.05)
        old_C5 = compute_C5(current_h)
        new_C5 = compute_C5(new_h)
        
        # Accept or reject
        if new_C5 < old_C5:
            current_h = new_h
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = current_h.copy()
        else:
            if np.random.rand() < np.exp(-(new_C5 - old_C5) / T):
                current_h = new_h
                
        # Temperature update
        T *= alpha
        iter_count += 1
        
        # Grid refinement: double the resolution every 1000 iterations
        if iter_count % 1000 == 0 and iter_count > 0:
            n_points *= 2
            # Interpolate to new grid
            x_old = np.linspace(0, 2, n_points // 2)
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_old, current_h)
            h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
            current_h = h_interp
            dx = 2.0 / n_points
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
