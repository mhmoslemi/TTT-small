import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from scipy.optimize import differential_evolution

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    t0 = time.time()
    best_C = float('inf')
    best_h = None

    # Use differential evolution for global optimization
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function for differential evolution
    def objective(h_vars):
        h = np.array(h_vars)
        h = project(h, dx)
        return evaluate(h, dx)
    
    # Run differential evolution
    # Use a smaller population size for speed
    result = differential_evolution(

        h_current = initial_h.copy()

        lr = 0.01
        max_iter = 20000
        for i in range(max_iter):
            if time.time() - t0 > budget_s - 1:

            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            for j in range(n_points):
                h_plus = h_current.copy()

                h_plus = project(h_plus, dx)

            h_current = h_current - lr * grad
