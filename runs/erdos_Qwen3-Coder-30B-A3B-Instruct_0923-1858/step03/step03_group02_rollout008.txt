import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a modified triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - 1.5 * abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def build_overlap_constraints(h_vec, k_max):
        h_vec = np.array(h_vec)
        # Build the linearized constraint for maximum overlap
        # For constraint t >= sum h[i] * (1 - h[i + k_max]) 
        # Linearize at current h values
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if i + k_max < len(h_vec):
                grad[i] = 1 - h_vec[i + k_max]
            else:
                grad[i] = 0
        return grad

    def solve_optimization_step(h_vec, t_current):
        h_new = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = [
            0 <= h_new,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2,
            t >= 0
        ]
        
        # Determine the most constraining shift k and build constraint
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        k_max = np.argmax(corr) - (len(h_vec) - 1)
        
        # Linearized constraint using current h
        grad = build_overlap_constraints(h_vec, k_max)
        
        c_k = np.sum(h_vec * np.concatenate([1 - h_vec[k_max:], np.zeros(k_max)]))
        
        # Add linearized constraint
        constraints.append(t >= c_k + grad @ (h_new - h_vec))
        
        # Form and solve QP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-7, reltol=1e-7)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new_sol = h_new.value
                t_sol = t.value
                return h_new_sol, t_sol
        except:
            pass
        return h_vec, t_current

    # Main optimization loop
    patience = 0
    max_patience = 30
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation 
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = project_feasible(h)
                patience = 0
            
        # Solve optimization step
        h_new, t_new = solve_optimization_step(h, current_C5)
        
        # Accept new solution if it improves the objective
        if t_new < current_C5:
            h = h_new
        else:
            h = project_feasible(h)
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    # Final projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))
