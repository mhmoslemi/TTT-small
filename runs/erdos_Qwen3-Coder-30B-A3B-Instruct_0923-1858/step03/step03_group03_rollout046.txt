def run(seed=42, budget_s=600, **kwargs):

        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def evaluate_objective(h_vec):
        h_vec = project_feasible(h_vec)
        return compute_C5(h_vec)
    
    def get_gradient(h_vec, eps=1e-4):
        h_vec = np.array(h_vec)
        current_C5 = compute_C5(h_vec)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            max_k_idx = np.argmax(corr_plus)
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Simpler gradient-based approach with momentum and adaptive learning rate
    def optimize_step(h_vec, momentum=0.9, learning_rate=0.02, max_iter=100):
        v = np.zeros_like(h_vec)
        for _ in range(max_iter):
            grad = get_gradient(h_vec)
            v = momentum * v - learning_rate * grad
            h_vec = h_vec + v
            h_vec = project_feasible(h_vec)
        return h_vec
    
    # Local perturbation with structured mutations
    def perturb_solution(h_vec, strength=0.05):
        h_new = h_vec.copy()
        # Randomly select indices to modify
        num_mutations = max(1, int(0.1 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_mutations, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    patience = 0
    max_patience = 100
    lr = 0.02

                h = perturb_solution(h, 0.03)

        h = optimize_step(h, learning_rate=lr)
        
        # Adapt learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            lr *= 0.95

        # Occasionally try a different direction
        if iter_count % 200 == 0:
            h = perturb_solution(h, 0.05)
        
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
