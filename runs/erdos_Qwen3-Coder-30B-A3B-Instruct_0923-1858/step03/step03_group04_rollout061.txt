import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def frequency_domain_update(h_vec, learning_rate=0.01):
        # Transform to frequency domain
        H = fft(h_vec)
        # Compute the conjugate of the FFT of (1-h)
        H_conj = np.conj(fft(1 - h_vec))
        # Convolution in frequency domain
        correlation_spectrum = H * H_conj
        # Inverse transform to get the correlation in time domain
        correlation = ifft(correlation_spectrum).real
        # Find the index of maximum correlation
        max_corr_idx = np.argmax(correlation)
        # Create gradient in frequency domain to reduce max correlation
        grad = np.zeros_like(H)
        # Modify filter to reduce the max entry
        grad[max_corr_idx] -= learning_rate
        # Apply gradient
        H_new = H - grad
        # Backtransform
        h_new = ifft(H_new).real
        # Project back to feasible set
        return project_feasible(h_new)
    
    # Enhanced optimization
    max_iter = 2000
    eta = 0.01
    patience = 0
    max_patience = 50
    last_progress = 0
    
    for i in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_progress = i
        else:
            patience += 1

        if patience > max_patience or i - last_progress > 100:
            # Try a more aggressive perturbation
            h_perturbed = h.copy()
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed += noise
            h = project_feasible(h_perturbed)
            patience = 0
            eta = min(0.05, eta * 1.05)

        # Frequency domain update
        h = frequency_domain_update(h, eta)
        
        if i % 50 == 0:
            eta *= 0.98
            
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
