import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)

        # Compute overlaps for all lags
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Find the lag with maximum overlap
        max_idx = np.argmax(corr)
        # The maximum overlap is the value at that index

        target_lag = max_idx - (len(corr) // 2)
        
        # Create variables
        h = cp.Variable(n_points)
        # QP to minimize this specific overlap
        # Overlap at lag k is sum h[i] * (1 - h[i + k]) for valid indices
        # This is a quadratic form
        
        # Create the quadratic term for this specific overlap
        # For overlap at lag target_lag
        # We'll compute the indices where the overlap happens
        if target_lag >= 0:
            expr = 0

                expr += h[i] * (1 - h[i + target_lag])
        else:
            expr = 0

                expr += h[i] * (1 - h[i + target_lag])

        objective = cp.Minimize(expr)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == 1.0 / dx
        ]
        
        # Create and solve QP
        prob = cp.Problem(objective, constraints)
        
        try:

                h_sol = h.value
                h_new = project(h_sol, dx)
                current_C = evaluate(h_new, dx)
                if current_C < best_C:
                    best_C = current_C
