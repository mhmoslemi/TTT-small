        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum

        h_vec = np.array(h_vec)
        # Compute frequency domain representation
        H = fft(h_vec)
        # Compute the conjugate of 1-h (the frequency domain representation is 1 - H)
        H_conj = 1 - H
        # The overlap in frequency domain corresponds to H * H_conj in frequency domain
        # We want to minimize the maximum of the inverse transform
        # This is approximated by minimizing the maximum magnitude in time domain
        
        # Compute the convolution in frequency domain
        correlation_spectrum = H * np.conj(H_conj)
        # Inverse transform to time domain
        correlation_time = np.real(ifft(correlation_spectrum))
        
        # Find the index of maximum value (this is the max correlation)
        max_idx = np.argmax(correlation_time)
        
        # Compute gradient in frequency domain (approximate)
        # We will modify H to reduce the maximum value at max_idx
        # Idea is to decrease magnitude of spectrum where correlation is large
        
        # Convert back to real signal for gradient estimation
        h_time_domain = np.real(ifft(H))
        
        # Approximate gradient with finite differences
        h_grad = np.zeros_like(H)
        eps = 1e-4

            delta = np.zeros_like(H)
            delta[i] = eps
            H_plus = H + delta
            H_minus = H - delta
            
            corr_plus = np.real(ifft(H_plus * np.conj(1 - H_plus)))
            corr_minus = np.real(ifft(H_minus * np.conj(1 - H_minus)))
            
            h_grad[i] = (corr_plus[max_idx] - corr_minus[max_idx]) / (2 * eps)

        H_new = H - learning_rate * h_grad
