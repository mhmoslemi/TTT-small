        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_gradients(h_vec, eps=1e-5):
        h_vec = np.array(h_vec)
        # Determine which lag produces the maximum overlap
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)

            # Forward difference in values
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps

            grad = compute_gradients(h_new)
            h_candidate = h_new - eta * grad
