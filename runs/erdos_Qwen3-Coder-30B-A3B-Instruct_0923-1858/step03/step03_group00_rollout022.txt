import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial guess if available

        initial_h = initial_h_values
        if len(initial_h) != n_points:

    except NameError:
        initial_h = np.ones(n_points) * 0.5
        t = np.linspace(0, 2, n_points)
        initial_h = 0.3 + 0.4 * np.sin(np.pi * t) + 0.1 * np.sin(3 * np.pi * t)
        initial_h = project(initial_h, dx)
    
    # Start with initial guess
    h_current = initial_h.copy()
    best_h = h_current.copy()
    best_C = evaluate(h_current, dx)
    
    t0 = time.time()
    max_time = budget_s - 2  # Leave some time for cleanup
    
    # Adaptive targeting QP approach
    n_qp_solves = 0
    max_qp_solves = 100
    
    while time.time() - t0 < max_time and n_qp_solves < max_qp_solves:
        # Compute all correlations to find worst lag
        corr_full = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr_full)
        max_corr_value = corr_full[max_lag_idx]
        
        # Find the corresponding lag k
        # Full correlation has length 2*n_points - 1, centered at n_points - 1
        k = max_lag_idx - (n_points - 1)
        
        # Set up QP to minimize correlation at this lag
        h = cp.Variable(n_points)
        # Objective: minimize the correlation at lag k
        # Since correlation at lag k is sum h[i] * (1 - h[i + k]) for valid i
        # This is a quadratic form
        
        # The index range for valid correlation at lag k
        if k >= 0:
            i_range = range(n_points - k)
            corr_expr = cp.sum(cp.multiply(h[i_range], 1 - h[i_range + k]))
        else:
            i_range = range(-k, n_points)
            corr_expr = cp.sum(cp.multiply(h[i_range], 1 - h[i_range - k]))
            
        objective = cp.Minimize(corr_expr)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == 1.0 / dx  # This is sum(h) * dx = 1, so sum(h) = 1/dx
        ]
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h.value
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    # Normalize to satisfy constraint
                    new_h = new_h * (1.0 / (np.sum(new_h) * dx))
                    new_C = evaluate(new_h, dx)
                    if new_C < best_C:
                        best_C = new_C
                        best_h = new_h.copy()
                    h_current = new_h
                    n_qp_solves += 1
        except Exception:
            # Fall back to projected gradient if needed
            pass
        
        # Continue with projected gradient descent for better variation
        if n_qp_solves % 5 == 0 and n_qp_solves > 0:
            # Do a quick gradient descent step
            grad = np.zeros_like(h_current)
            eps = 1e-6
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = project(h_plus, dx)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_plus = np.max(corr_plus)
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = project(h_minus, dx)
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                corr_minus = np.max(corr_minus)
                
                grad[j] = (corr_plus - corr_minus) / (2 * eps)
            
            lr = 0.01
            h_new = h_current - lr * grad
            h_new = project(h_new, dx)
            current_C = evaluate(h_new, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h_new.copy()
            h_current = h_new
            n_qp_solves += 1
    
    final_h = project(best_h, dx)
    final_C = evaluate(final_h, dx)
    return (final_h, final_C, n_points)
