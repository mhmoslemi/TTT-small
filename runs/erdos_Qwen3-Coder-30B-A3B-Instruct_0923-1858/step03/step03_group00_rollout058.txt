import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            C_plus = np.max(corr_plus) * dx
            
            h_minus = h_final.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
