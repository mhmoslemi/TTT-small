import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO Parameters
    swarm_size = 20
    max_iter_pso = 200
    inertia_weight = 0.7
    cognitive_weight = 1.5
    social_weight = 1.5
    velocity_bounds = 0.1

    # PSO Optimization Phase 1
    particles = []
    velocities = []
    personal_best = []
    personal_best_values = []
    global_best = None
    global_best_value = float('inf')
    
    for _ in range(swarm_size):
        # Initialize particle with feasible solution
        particle = project_feasible(np.random.rand(n_points) * 0.5)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))
        personal_best.append(particle.copy())
        personal_best_values.append(compute_C5(particle))
        if personal_best_values[-1] < global_best_value:
            global_best = particle.copy()
            global_best_value = personal_best_values[-1]
    
    # PSO iterations
    for iteration in range(max_iter_pso):
        if elapsed() > budget_s * 0.6:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_weight * r1 * (personal_best[i] - particles[i]) + 
                           social_weight * r2 * (global_best - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -velocity_bounds, velocity_bounds)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair feasibility
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            current_value = compute_C5(particles[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_value
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
    
    # L-BFGS-B Refinement Phase 2
    def objective(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Constraint dictionary for scipy minimize
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine using L-BFGS-B
    try:
        result = minimize(objective, global_best, method='L-BFGS-B', 
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-5},
                         callback=lambda x: None)
        
        if result.success:
            refined_h = project_feasible(result.x)
            refined_c5 = compute_C5(refined_h)
            if refined_c5 < best_C5:
                best_C5 = refined_c5
                best_h = refined_h.copy()
    except:
        pass

    # Final refinement with gradient descent for remaining time
    if elapsed() < budget_s * 0.9:
        h_refined = best_h.copy()
        alpha = 0.01
        patience = 0
        max_patience = 100
        
        for i in range(500):  # Limit iterations
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute gradients via finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_minus = h_refined.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
            
            # Gradient step
            h_refined = h_refined - alpha * grad
            h_refined = project_feasible(h_refined)
            
            # Update best
            current_c5 = compute_C5(h_refined)
            if current_c5 < best_C5:
                best_C5 = current_c5
                best_h = h_refined.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break

    return (best_h.tolist(), best_C5, len(best_h))
