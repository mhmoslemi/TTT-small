import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO parameters
    swarm_size = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            noise = np.random.normal(0, 0.05, n_points)
            particle = project_feasible(h + noise)
        swarm.append(particle)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Personal bests
    p_best = [swarm[i].copy() for i in range(swarm_size)]
    p_best_C5 = [compute_C5(swarm[i]) for i in range(swarm_size)]
    
    # Global best
    global_best_idx = np.argmin(p_best_C5)
    g_best = swarm[global_best_idx].copy()
    g_best_C5 = p_best_C5[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.6:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - swarm[i]) + 
                           c2 * r2 * (g_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            current_C5 = compute_C5(swarm[i])
            
            # Update personal best
            if current_C5 < p_best_C5[i]:
                p_best[i] = swarm[i].copy()
                p_best_C5[i] = current_C5
                
                # Update global best
                if current_C5 < g_best_C5:
                    g_best = swarm[i].copy()
                    g_best_C5 = current_C5
        
        if g_best_C5 < best_C5:
            best_C5 = g_best_C5
            best_h = g_best.copy()
    
    # L-BFGS-B refinement with the best PSO solution
    def objective(h_vec):
        return compute_C5(h_vec)
    
    def callback(xk):
        C5_val = compute_C5(xk)
        if C5_val < best_C5:
            best_C5 = C5_val
            best_h = xk.copy()
    
    # Refinement with L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Only refine if we still have time
    if elapsed() < budget_s * 0.9:
        try:
            result = minimize(
                objective, 
                g_best,
                method='L-BFGS-B',
                bounds=bounds,
                options={'ftol': 1e-5, 'maxiter': 300},
                callback=callback if 'callback' in locals() else None
            )
            if result.success:
                refined_h = project_feasible(result.x)
                refined_C5 = compute_C5(refined_h)
                if refined_C5 < best_C5:
                    best_C5 = refined_C5
                    best_h = refined_h.copy()
        except:
            pass
    
    return (best_h.tolist(), best_C5, len(best_h))
