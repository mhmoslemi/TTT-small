import numpy as np
import time
from scipy.optimize import dual_annealing, differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Function to evaluate objective with bounds
    def objective(h_vec):
        h_vec = project_to_feasible(h_vec)
        return compute_C5(h_vec)

    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=int(1000 * budget_s / 600),
        popsize=15,
        seed=seed,
        polish=True,
        atol=1e-6,
        rtol=1e-6
    )
    
    # Final optimization with L-BFGS-B
    h_opt = project_to_feasible(de_result.x)
    lbfgs_bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine with L-BFGS-B
    try:
        result = minimize(
            objective,
            h_opt,
            method='L-BFGS-B',
            bounds=lbfgs_bounds,
            options={
                'maxiter': int(300 * budget_s / 600),
                'ftol': 1e-9,
                'gtol': 1e-6
            }
        )
        h_final = project_to_feasible(result.x)
    except:
        h_final = h_opt

    final_C5 = compute_C5(h_final)
    return (h_final.tolist(), final_C5, n_points)
