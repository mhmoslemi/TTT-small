import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization parameters
    swarm_size = 50
    max_iter = 1000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_w = 2.0
    social_w = 2.0
    velocity_max = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # First particle is the initial solution
    particles.append(h.copy())
    velocities.append(np.zeros_like(h))
    
    # Remaining particles are perturbed versions
    for i in range(1, swarm_size):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        particles.append(project_feasible(perturbed))
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [compute_C5(p) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_w * r1 * (personal_best[i] - particles[i])
            social = social_w * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    # Early stopping check
                    if global_best_fitness < 0.3807:
                        break
        
        # Occasionally perturb particles if no progress
        if iteration % 30 == 0 and iteration > 0:
            for i in range(swarm_size):
                if np.random.random() < 0.5:
                    perturbation = np.random.normal(0, 0.03, n_points)
                    particles[i] = particles[i] + perturbation
                    particles[i] = project_feasible(particles[i])
    
    # Final update of best solution with current global best
    final_C5 = compute_C5(global_best)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = global_best.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
