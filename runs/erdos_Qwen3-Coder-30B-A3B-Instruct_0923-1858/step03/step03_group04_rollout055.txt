import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    swarm_size = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    if elapsed() < budget_s * 0.6:
        # Use initial solution as one particle
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(compute_C5(h))
        
        # Add noisy variants
        for _ in range(swarm_size - 1):
            perturbation = np.random.normal(0, 0.05, n_points)
            candidate = h + perturbation
            candidate = project_feasible(candidate)
            swarm.append(candidate)
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            pbest.append(candidate.copy())
            pbest_vals.append(compute_C5(candidate))
    else:
        # Initialize randomly
        for _ in range(swarm_size):
            candidate = np.random.uniform(0, 1, n_points)
            candidate = project_feasible(candidate)
            swarm.append(candidate)
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            pbest.append(candidate.copy())
            pbest_vals.append(compute_C5(candidate))

    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]

    # PSO iteration
    for iter_count in range(max_iter):
        if elapsed() >= budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - swarm[i]) +
                c2 * r2 * (gbest - swarm[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            new_val = compute_C5(swarm[i])
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = new_val
                
                # Update global best
                if new_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = new_val

        # Update best solution found so far
        if gbest_val < best_C5:
            best_C5 = gbest_val
            best_h = gbest.copy()

    # Local refinement with L-BFGS-B
    def objective(h_vec):
        return compute_C5(h_vec)

    def constraint_fun(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    if elapsed() < budget_s * 0.99:
        try:
            # Use scipy.optimize.minimize with L-BFGS-B
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_fun}
            
            result = minimize(
                objective, 
                gbest, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if result.success:
                refined_h = project_feasible(result.x)
                refined_C5 = compute_C5(refined_h)
                
                if refined_C5 < best_C5:
                    best_C5 = refined_C5
                    best_h = refined_h.copy()
        except:
            pass

    return (best_h.tolist(), best_C5, len(best_h))
