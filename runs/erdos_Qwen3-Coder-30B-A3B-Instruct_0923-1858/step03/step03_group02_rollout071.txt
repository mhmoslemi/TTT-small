import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize from initial_h_values if available
    try:
        h_base = np.array(initial_h_values)
        n_points = len(h_base)
    except NameError:
        # Create a triangular base if no initial values
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h_base = np.maximum(0, 1 - np.abs(x - 1))
        h_base = h_base * (n_points / 2) / np.sum(h_base)

    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 1000
    
    # Particle swarm parameters
    inertia = 0.9
    inertia_decay = 0.995
    cognitive = 2.0
    social = 2.0
    velocity_clamp = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    # Base particle
    particles.append(h_base.copy())
    velocities.append(np.zeros(n_points))
    p_best.append(h_base.copy())
    p_best_fitness.append(float('inf'))
    
    # Perturbed particles
    for i in range(1, swarm_size):
        h = h_base + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        p_best.append(h.copy())
        p_best_fitness.append(float('inf'))
    
    s_best = particles[0].copy()
    s_best_fitness = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    iter_count = 0
    no_improvement_count = 0
    patience = 30
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            v_new = (
                inertia * velocities[i] +
                cognitive * r1 * (p_best[i] - particles[i]) +
                social * r2 * (s_best - particles[i])
            )
            
            # Velocity clamping
            v_new = np.clip(v_new, -velocity_clamp, velocity_clamp)
            velocities[i] = v_new
            
            # Update position
            h_new = particles[i] + velocities[i]
            h_new = project_feasible(h_new)
            
            # Evaluate fitness
            fitness_new = compute_C5(h_new)
            
            # Update personal best
            if fitness_new < p_best_fitness[i]:
                p_best[i] = h_new.copy()
                p_best_fitness[i] = fitness_new
                
                # Update global best
                if fitness_new < s_best_fitness:
                    s_best = h_new.copy()
                    s_best_fitness = fitness_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            
            # Update particle position
            particles[i] = h_new
            
        # Check for stagnation and perturb
        if no_improvement_count > patience:
            for i in range(1, swarm_size):  # Skip base particle
                h = p_best[i] + np.random.normal(0, 0.03, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = h
                velocities[i] = np.random.uniform(-0.05, 0.05, n_points)
            no_improvement_count = 0
            
        # Decay inertia
        inertia *= inertia_decay
        
        iter_count += 1

    # Final safeguard: re-evaluate best solution
    final_fitness = compute_C5(s_best)
    if final_fitness < s_best_fitness:
        s_best_fitness = final_fitness

    return (s_best.tolist(), s_best_fitness, n_points)
