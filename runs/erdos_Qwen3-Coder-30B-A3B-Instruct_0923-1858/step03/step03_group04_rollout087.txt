import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Frequency domain approach
    def fft_optimize(h_vec, max_iter=200):
        h_vec = np.array(h_vec).copy()
        H = np.fft.fft(h_vec)
        learning_rate = 0.01
        prev_max = float('inf')
        patience = 0
        max_patience = 10
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute correlation in frequency domain
            H_conj = np.conj(H)
            corr_freq = H * H_conj
            corr_time = np.fft.ifft(corr_freq).real
            
            # Find max index
            max_idx = np.argmax(corr_time)
            max_val = corr_time[max_idx]
            
            # Compute gradient in frequency domain
            grad = np.zeros_like(H)
            # Approximate gradient
            eps = 1e-6
            for j in range(len(H)):
                H_plus = H.copy()
                H_plus[j] += eps
                corr_plus = np.fft.ifft(H_plus * np.conj(H_plus)).real
                grad[j] = (corr_plus[max_idx] - max_val) / eps
            
            # Update frequency domain
            H = H - learning_rate * grad
            # Project back to real domain
            h_new = np.fft.ifft(H).real
            h_new = project_feasible(h_new)
            H = np.fft.fft(h_new)
            
            # Update solution
            current_C5 = compute_C5(h_new)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Try different learning rate
                    learning_rate *= 0.9
            
            if abs(prev_max - max_val) < 1e-8:
                patience += 1
            else:
                patience = 0
            prev_max = max_val
        
        return best_h if 'best_h' in locals() else h_new

    # Try to improve with convex heuristic
    def convex_heuristic(h_vec):
        # Solve a convex problem: minimize sum(h^2) subject to constraints
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        objective = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return np.maximum(0, np.minimum(1, h_var.value))
        except:
            return h_vec

    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    max_iter = 1000

    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count

        # Try convex relaxation
        if iter_count % 10 == 0:
            relaxed_h = convex_heuristic(h)
            relaxed_C5 = compute_C5(relaxed_h)
            if relaxed_C5 < current_C5:
                h = relaxed_h.copy()
        
        # Try frequency domain optimization
        if iter_count % 5 == 0:
            h = fft_optimize(h, max_iter=10)

        # Random perturbation to escape local minima
        if iter_count % 20 == 0 and iter_count > 0:
            for i in range(n_points):
                if np.random.random() < 0.2:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.03), 0, 1)
            h = project_feasible(h)
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
