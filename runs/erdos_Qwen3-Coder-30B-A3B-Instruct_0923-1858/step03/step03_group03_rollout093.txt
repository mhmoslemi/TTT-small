import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 30
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing approach
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iteration = 0
    patience = 0
    max_patience = 1000
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        # Perturb solution
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        h_new[idx] += delta
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Compute new C5
        new_C5 = compute_C5(h_new)
        old_C5 = compute_C5(h)
        
        # Decide whether to accept
        if new_C5 < old_C5:
            h = h_new
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_C5 - old_C5) / T):
                h = h_new
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h_new.copy()
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1

        # Decrease temperature
        T *= alpha
        iteration += 1
        
        # Grid refinement every 1000 iterations
        if iteration % 1000 == 0 and iteration > 0 and n_points < 90:
            # Double the grid resolution
            n_points *= 2
            h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points//2), h)
            h_new = project_feasible(h_new)
            h = h_new
            dx = 2.0 / n_points

        # Early stopping if no improvement
        if patience > max_patience:
            break

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
