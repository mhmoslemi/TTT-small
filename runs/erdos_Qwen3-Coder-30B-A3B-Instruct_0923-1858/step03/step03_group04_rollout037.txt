import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Frequency-domain optimization approach
    def freq_optimize(h_vec, max_iter=100):
        h_vec = h_vec.copy()
        # Convert to frequency domain
        H = fft(h_vec)
        
        # Gradient descent in frequency domain
        learning_rate = 0.01
        for i in range(max_iter):
            # Compute correlation in frequency domain
            # overlap = ifft(H * conj(H)) => max overlap = max |overlap|
            H_conj = np.conj(H)
            overlap_fft = H * H_conj
            overlap = ifft(overlap_fft).real
            max_idx = np.argmax(overlap)
            max_overlap = overlap[max_idx]
            
            # Compute gradient approximation
            grad_freq = np.zeros_like(H)
            
            # Perturb each frequency component
            eps = 1e-5
            for j in range(len(H)):
                perturbed_H = H.copy()
                perturbed_H[j] += eps
                perturbed_overlap_fft = perturbed_H * np.conj(perturbed_H)
                perturbed_overlap = ifft(perturbed_overlap_fft).real
                grad_freq[j] = (perturbed_overlap[max_idx] - max_overlap) / eps
            
            # Update in frequency domain
            H -= learning_rate * grad_freq
            
            # Project back to spatial domain and apply constraints
            h_new = ifft(H).real
            h_new = project_feasible(h_new)
            
            # Convert back to frequency
            H = fft(h_new)
            
            # Early stopping
            if elapsed() > budget_s * 0.95:
                break
                
        return h_new
    
    # Main optimization loop
    iter_count = 0
    last_best_update = 0
    patience = 0
    max_patience = 30

        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1
            
        # Gradient-based refinement in frequency domain
        h_refined = freq_optimize(h, max_iter=10)
        h = h_refined.copy()
        
        # Additional perturbation to avoid local minima
        if patience > 10:
            idxs = np.random.choice(n_points, size=n_points // 8, replace=False)
            for i in idxs:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)
            patience = 0
            
        iter_count += 1
    
    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
