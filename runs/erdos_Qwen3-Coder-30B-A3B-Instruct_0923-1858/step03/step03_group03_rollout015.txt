import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with existing construction or create a triangular one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 30
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def perturb(h_vec, perturbation_strength=0.1):
        h_new = h_vec.copy()
        i = np.random.randint(len(h_vec))
        h_new[i] += np.random.normal(0, perturbation_strength)
        return project_feasible(h_new)
    
    # Simulated annealing with grid refinement
    T = 0.1
    alpha = 0.999
    iter_count = 0
    refinement_interval = 1000
    max_refinements = 3
    current_refinement = 0
    last_improvement = 0
    
    while elapsed() < budget_s * 0.95:
        iter_count += 1
        
        # Perturb solution
        h_new = perturb(h, 0.1)
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        C5_old = compute_C5(h)
        delta = C5_new - C5_old
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                last_improvement = iter_count
        
        # Temperature update
        T *= alpha
        
        # Grid refinement
        if iter_count % refinement_interval == 0 and current_refinement < max_refinements:
            # Double resolution
            n_points_new = n_points * 2
            x_new = np.linspace(0, 2, n_points_new)
            h_new_interp = np.interp(x_new, np.linspace(0, 2, n_points), h)
            h_new_interp = project_feasible(h_new_interp)
            
            h = h_new_interp
            n_points = n_points_new
            dx = 2.0 / n_points
            current_refinement += 1
            
            # Reset temperature for new resolution
            T = 0.1
            
        # Early stopping if no improvement for long
        if iter_count - last_improvement > 5000 and T < 1e-6:
            break
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
