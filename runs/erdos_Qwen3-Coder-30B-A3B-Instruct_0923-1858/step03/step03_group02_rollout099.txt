import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    swarm_size = 50
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        fitness = compute_C5(particle)
        p_best.append(particle.copy())
        p_best_fitness.append(fitness)

    s_best_idx = np.argmin(p_best_fitness)
    s_best = p_best[s_best_idx].copy()
    s_best_fitness = p_best_fitness[s_best_idx]

    inertia = 0.9
    inertia_decay = (0.9 - 0.4) / (budget_s * 0.8 / 50)
    cognitive_w = 2.0
    social_w = 2.0
    velocity_clamp = 0.1
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 30
    
    while elapsed() < budget_s * 0.95:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            new_velocity = (
                inertia * velocities[i] +
                cognitive_w * r1 * (p_best[i] - particles[i]) +
                social_w * r2 * (s_best - particles[i])
            )
            
            # Velocity clamping
            new_velocity = np.clip(new_velocity, -velocity_clamp, velocity_clamp)
            velocities[i] = new_velocity
            
            new_position = particles[i] + velocities[i]
            new_position = project_feasible(new_position)
            
            new_fitness = compute_C5(new_position)
            
            particles[i] = new_position
            
            if new_fitness < p_best_fitness[i]:
                p_best[i] = new_position.copy()
                p_best_fitness[i] = new_fitness
                
                if new_fitness < s_best_fitness:
                    s_best = new_position.copy()
                    s_best_fitness = new_fitness
                    no_improvement_count = 0
                    
        no_improvement_count += 1
        if no_improvement_count > max_no_improvement:
            # Perturb particles
            for i in range(swarm_size):
                particles[i] += np.random.normal(0, 0.03, n_points)
                particles[i] = project_feasible(particles[i])
                fitness = compute_C5(particles[i])
                if fitness < p_best_fitness[i]:
                    p_best[i] = particles[i].copy()
                    p_best_fitness[i] = fitness
                    
                    if fitness < s_best_fitness:
                        s_best = particles[i].copy()
                        s_best_fitness = fitness
            no_improvement_count = 0
            
        # Decay inertia
        inertia = max(0.4, inertia - inertia_decay)
            
        iteration += 1
        if iteration % 50 == 0:
            # Periodic check for early termination
            if elapsed() > budget_s * 0.9:
                break

    final_C5 = compute_C5(s_best)
    return (s_best.tolist(), final_C5, len(s_best))
