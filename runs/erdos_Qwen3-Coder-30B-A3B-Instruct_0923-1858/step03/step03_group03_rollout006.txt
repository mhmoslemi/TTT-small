import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral"""
    h = np.array(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
