import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def mutate_individual(h, mutation_rate, mutation_scale):
    """Mutate an individual with Gaussian noise."""
    h_mut = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, mutation_scale)
    h_mut = project(h_mut, 2.0 / n_points)
    return h_mut

def crossover(parent1, parent2):
    """One-point crossover between two parents."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

    np.random.seed(seed)

        # Create next generation
        new_population = []
