import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def local_optimize(h_vec, max_iter=100):
        """Use gradient descent with projection."""
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Compute gradient with respect to the maximum overlap
            # We do this by finite difference
            eps = 1e-5
            grad = np.zeros_like(h_new)
            base_corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(base_corr)
            
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - 0.01 * grad
            h_new = project_feasible(h_new)
            
            if i % 10 == 0:
                current_C5 = compute_C5(h_new)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h[:] = h_new[:]
        return h_new

    def perturb_solution(h_vec, mutation_strength=0.05):
        h_new = h_vec.copy()
        num_mutations = max(1, int(0.1 * len(h_vec)))
        indices = np.random.choice(len(h_vec), num_mutations, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, mutation_strength)
        return project_feasible(h_new)

    # Refined search with adaptive learning rate
    patience = 0
    max_patience = 50
    iter_count = 0
    eta = 0.02
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h[:] = h[:]
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = perturb_solution(h, 0.03)
                patience = 0

        # Gradient-based refinement
        h = local_optimize(h, max_iter=30)
        
        if iter_count % 20 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h[:] = best_h[:]

    return (best_h.tolist(), best_C5, len(best_h))
