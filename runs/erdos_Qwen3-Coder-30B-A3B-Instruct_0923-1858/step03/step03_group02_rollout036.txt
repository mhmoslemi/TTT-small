import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_qp_warm_start(h_vec):
        bounds = Bounds(0, 1, keep_feasible=True)
        result = minimize(
            lambda x: np.sum((x - h_vec)**2),
            h_vec,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            return result.x
        return h_vec

    # Generate random perturbations while maintaining feasibility
    def generate_perturbation(h_vec, epsilon=0.05):
        h_new = h_vec.copy()
        idx = np.random.choice(len(h_vec))
        h_new[idx] += np.random.normal(0, epsilon)
        return project_feasible(h_new)
    
    # Using simulated annealing approach with adaptive cooling
    def simulated_annealing(h_vec, max_iter=2000):
        current_h = h_vec.copy()
        current_C5 = compute_C5(current_h)
        best_h = current_h.copy()
        best_C5 = current_C5
        T = 1.0
        T_min = 1e-6
        alpha = 0.99
        iter_count = 0
        
        while iter_count < max_iter and T > T_min and elapsed() < budget_s * 0.95:
            # Generate neighbor
            candidate_h = generate_perturbation(current_h)
            candidate_C5 = compute_C5(candidate_h)
            
            # Metropolis criterion
            delta = candidate_C5 - current_C5
            if delta < 0 or np.random.random() < np.exp(-delta / T):
                current_h = candidate_h
                current_C5 = candidate_C5
                
                if current_C5 < best_C5:
                    best_h = current_h.copy()
                    best_C5 = current_C5
            
            T *= alpha
            iter_count += 1
            
        return best_h, best_C5

    # Improved gradient estimation
    def compute_gradients(h_vec, eps=1e-4):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        h_plus = h_vec.copy()
        h_minus = h_vec.copy()
        
        for i in range(len(h_vec)):
            h_plus[i] = h_vec[i] + eps
            h_minus[i] = h_vec[i] - eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_plus[i] = h_vec[i]
            h_minus[i] = h_vec[i]
            
        return grad

    # Trust region method with adaptive step
    def trust_region_update(h_vec, max_iter=20):
        h_new = h_vec.copy()
        eta = 0.005
        
        for i in range(max_iter):
            grad = compute_gradients(h_new)
            h_candidate = h_new - eta * grad
            h_candidate = project_feasible(h_candidate)
            
            if compute_C5(h_candidate) < compute_C5(h_new):
                h_new = h_candidate
            else:
                eta *= 0.5
                if eta < 1e-6:
                    break
                    
        return h_new

    # Main optimization loop
    iter_count = 0
    patience = 0
    max_patience = 50
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Reset with simulated annealing
            h, _ = simulated_annealing(h, max_iter=100)
            patience = 0
        
        # Multiple strategies: trust region & SA
        if iter_count % 10 == 0:
            h = trust_region_update(h, max_iter=5)
        else:
            h = generate_perturbation(h, epsilon=0.01)
            
        h = project_feasible(h)
        iter_count += 1

    # Final refinement
    final_h, final_C5 = simulated_annealing(best_h, max_iter=500)
    
    return (final_h.tolist(), final_C5, len(final_h))
