import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Specialized approach: binary search on C_5 with convex feasibility check
    def feasibility_check(h_vec, C):
        # Create constrained optimization problem
        # min 0 s.t. integrate h(x) * (1 - h(x+k)) dx <= 1 - C for all k
        # This is a more accurate reflection of what we want to bound
        
        # We just return the max correlation directly for checking feasibility of this C
        # If all correlations are <= 1 - C, then C is feasible
        # But we are maximizing correlations, so we just check if the max is <= 1 - C
        # That is, we try to make all correlations >= 1 - C which would make max <= 1 - C
        
        # Change perspective entirely: let's minimize max correlation
        # So we're trying to maximize correlation to make it smaller
        # Actually no - we're minimizing the max from the perspective of:
        # minimize C5 such that max_k int h(x) (1 - h(x+k)) dx <= C5
        return True  # Simplified check
    
    # Use binary search approach but focus on using trust region properly
    # Try larger mutation sizes, better gradients, better projection
    
    def trust_region_update(h_vec, eta=0.02, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Estimate gradient for just one direction at a time
            grad = np.zeros_like(h_vec)
            eps = 1e-4
            k_idx = np.argmax(correlate(h_new, 1 - h_new, mode="full"))
            max_k = k_idx - (len(h_new) - 1)
            
            # Finite difference: slight perturbations along all dimensions
            for i_dim in range(len(h_new)):
                delta = np.zeros_like(h_new)
                delta[i_dim] = eps
                h_plus = project_feasible(h_new + delta)
                h_minus = project_feasible(h_new - delta)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i_dim] = (corr_plus[k_idx] - corr_minus[k_idx]) / (2 * eps)
            
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    def perturb_solution(h_vec, prob=0.15):
        h_new = h_vec.copy()
        idxs = np.random.choice(n_points, size=int(prob * n_points), replace=False)
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.03), 0, 1)
        return project_feasible(h_new)
    
    # Improved handling of convex relaxation
    def solve_convex_relaxation_2():
        # simplex + quadratic objective
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        # Slightly more natural regularization
        objective = cp.Minimize(cp.sum_squares(h_var) / 2)  # Minimize squared sum
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            sol = np.maximum(0, np.minimum(1, h_var.value))
            return sol
        except:
            return h.copy()
    
    # Try convex relaxation for warm start
    if elapsed() < budget_s * 0.6:
        try:
            relaxed_h = solve_convex_relaxation_2()
            if compute_C5(relaxed_h) < compute_C5(h):
                h = relaxed_h.copy()
        except:
            pass

    # Now optimize with high iteration count and adaptive learning rate
    eta = 0.02
    patience = 0
    max_patience = 40
    iter_count = 0
    last_best_update = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_best_update = iter_count
        else:
            patience += 1

        # Check if we're stuck and should perturb or restart
        if patience > max_patience or (iter_count - last_best_update > 60):
            h = perturb_solution(h)
            patience = 0
            if iter_count - last_best_update > 100:
                # Occasional full restart to get out of local minima
                try:
                    relaxed_h = solve_convex_relaxation_2()
                    if compute_C5(relaxed_h) < compute_C5(h):
                        h = relaxed_h.copy()
                except:
                    pass
        
        # Trust-region update
        h = trust_region_update(h, eta)
        
        # Adaptive learning rate decay
        if iter_count % 25 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
