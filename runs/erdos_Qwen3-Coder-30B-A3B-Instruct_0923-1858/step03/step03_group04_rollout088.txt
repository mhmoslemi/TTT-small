import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Perform optimization in frequency domain
    H = fft(h)
    learning_rate = 0.01
    smoothing = 0.9
    old_grad = None
    
    iter_count = 0
    while elapsed() < budget_s * 0.95 and iter_count < 1000:
        # Compute current overlapping integral in frequency domain
        H_conj = np.conj(H)
        overlap_spectrum = H * H_conj
        overlap_fft = ifft(overlap_spectrum)
        overlap_values = np.real(overlap_fft)
        
        # Find the max overlap index
        max_k_idx = np.argmax(overlap_values)
        
        # Estimate gradient in frequency domain
        grad = np.zeros_like(H)
        eps = 1e-5
        
        # Finite difference approximation of gradient
        for i in range(n_points):
            H_plus = H.copy()
            H_minus = H.copy()
            H_plus[i] += eps
            H_minus[i] -= eps
            
            overlap_plus = ifft(H_plus * np.conj(H_plus))
            overlap_minus = ifft(H_minus * np.conj(H_minus))
            
            grad[i] = (np.real(overlap_plus[max_k_idx]) - np.real(overlap_minus[max_k_idx])) / (2 * eps)
        
        # Apply learning rate and smoothing
        if old_grad is not None:
            grad = smoothing * old_grad + (1 - smoothing) * grad
        old_grad = grad
        
        # Update H in frequency domain
        H = H - learning_rate * grad
        
        # Project back to spatial domain
        h_new = np.real(ifft(H))
        
        # Project to feasible set: [0,1] and sum = n_points/2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update H for next iteration
        H = fft(h_new)
        
        # Evaluate
        current_C5 = compute_C5(h_new)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_new.copy()
            
        # Adjust step size
        if iter_count % 50 == 0 and iter_count > 0:
            learning_rate *= 0.95
            
        iter_count += 1

    return (best_h.tolist(), best_C5, len(best_h))
