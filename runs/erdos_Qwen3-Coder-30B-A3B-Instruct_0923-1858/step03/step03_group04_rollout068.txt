import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Initialize from initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial symmetric triangular function
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def perturb_turbulence(h_vec, alpha=0.1):
        # Apply sine-based turbulence
        turbulence = alpha * np.sin(np.pi * np.arange(n_points) / n_points) * np.random.normal(0, 1, n_points)
        h_turb = h_vec + turbulence
        return project_feasible(h_turb)

    def gradient_step(h_vec, lr=0.01):
        # Estimating gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        base_corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(base_corr)
        
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        h_new = h_vec - lr * grad
        return project_feasible(h_new)

    # Start optimization loop
    iter_count = 0
    turbulence_frequency = 30
    patience = 0
    max_patience = 30
    last_improvement = 0
    lr = 0.01

    while elapsed() < budget_s * 0.95:
        # Apply turbulence periodically
        if iter_count % turbulence_frequency == 0:
            h = perturb_turbulence(h, alpha=0.05 + 0.05 * np.sin(iter_count / 100))

        # Gradient step occasionally
        if iter_count % 5 == 0:
            h = gradient_step(h, lr=lr)

        # Keep the best solution
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_improvement = iter_count
        else:
            patience += 1

        # Early stopping criteria
        if patience > max_patience or iter_count - last_improvement > 100:
            # Random restart
            h = perturb_turbulence(h, alpha=0.1)
            patience = 0

        # Adaptive learning rate
        if iter_count % 20 == 0:
            lr *= 0.95
        
        iter_count += 1

        if elapsed() > budget_s * 0.9:
            break

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
