import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def pso_optimization():
        # PSO parameters
        n_particles = 20
        max_iter = 500
        w = 0.7  # inertia weight
        c1 = c2 = 1.5  # cognitive/social weights
        v_max = 0.1  # maximum velocity
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_vals = []
        for _ in range(n_particles):
            particle_h = project_feasible(np.random.rand(n_points) * 0.5)
            particles.append(particle_h.copy())
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            pbest.append(particle_h.copy())
            pbest_vals.append(compute_C5(particle_h))
        
        # Global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        # PSO main loop
        for iter_count in range(max_iter):
            if elapsed() > budget_s * 0.5:
                break
            for i in range(n_particles):
                # Update velocity and position
                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                particles[i] += velocities[i]
                
                # Repair infeasible solutions
                particles[i] = project_feasible(particles[i])
                
                # Evaluate
                current_C5 = compute_C5(particles[i])
                
                # Update personal best
                if current_C5 < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = current_C5
                
                # Update global best
                if current_C5 < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_C5
                    
        return gbest, gbest_val

    def lbfgsb_refinement(initial_h):
        # L-BFGS-B optimization with constraints
        bounds = Bounds([0.0] * n_points, [1.0] * n_points)
        
        def objective(h_vec):
            return compute_C5(h_vec)
        
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        result = minimize(objective, initial_h, method='L-BFGS-B', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-6},
                         callback=lambda x: None)
        
        return result.x if result.success else initial_h

    # Phase 1: PSO for global search
    if elapsed() < budget_s * 0.7:
        try:
            pso_h, pso_C5 = pso_optimization()
            if pso_C5 < best_C5:
                best_C5 = pso_C5
                best_h = pso_h
        except:
            pass
    
    # Phase 2: Local refinement with L-BFGS-B
    if elapsed() < budget_s * 0.95:
        try:
            refined_h = lbfgsb_refinement(best_h)
            refined_C5 = compute_C5(refined_h)
            if refined_C5 < best_C5:
                best_C5 = refined_C5
                best_h = refined_h
        except:
            pass

    # Final check and return
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
