import numpy as np
import time
import math

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    t0 = time.time()
    max_time = budget_s - 2

    T_start = 0.1
    T = T_start

    cooling_interval = 500
    scale_start = 0.1
    scale = scale_start
    scale_min = 0.01
    scale_decay = 0.995
    
    iteration = 0
    while time.time() - t0 < max_time:
        if iteration % cooling_interval == 0 and T > 1e-6:
            T *= alpha
            scale *= scale_decay
            scale = max(scale, scale_min)
            
        # Generate candidate
        delta = np.random.normal(0, scale, n_points)
        h_candidate = h_current + delta
        h_candidate = project(h_candidate, dx)
        
        # Evaluate
        current_C = evaluate(h_candidate, dx)
        
        # Accept or reject
        if current_C < best_C:
            best_C = current_C

            # Accept with probability
            if T > 0:
                prob = math.exp(-(current_C - best_C) / T)
                if np.random.random() < prob:

    # Final refinement
    h_final = project(best_h, dx)
    final_C = evaluate(h_final, dx)
    
    return (h_final, final_C, n_points)
