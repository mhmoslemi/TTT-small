import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:

        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Modern optimization with bounds
    def objective(h_vec):
        return compute_C5(h_vec)

    def constraint_func(h_vec):
        return np.sum(h_vec) * dx - 1

    # Use scipy minimize for better optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}

    # Generate random perturbations
    def perturb_solution(h_vec, strength=0.05):
        h_new = h_vec.copy()
        # Randomly modify a few indices
        indices = np.random.choice(len(h_vec), max(1, int(0.1 * len(h_vec))), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, strength)
        return project_feasible(h_new)

    # Step size decay
    eta = 0.02
    max_iter = 200
    patience = 0
    max_patience = 100

    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.95:
            break

        # Optimization step
        res = minimize(objective, h, method='SLSQP', bounds=bounds, 
                      constraints=constraints, options={'maxiter': 20})
        if res.success:
            h = project_feasible(res.x)
        else:
            # Fallback
            h = perturb_solution(h, 0.05)
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = perturb_solution(h, 0.05)
                patience = 0
        
        # Adaptive learning rate
        if iteration % 50 == 0 and iteration > 0:
            eta *= 0.95
            
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
