import time

        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use scipy.optimize.minimize with constraints
    def optimize_step(h_vec, max_iter=30):
        def objective(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return np.max(corr) * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B with bounds
        result = minimize(objective, h_vec, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                         options={'maxiter': max_iter, 'gtol': 1e-6})
        if result.success:
            return project_feasible(result.x)
        else:
            return project_feasible(h_vec)

    # Mimic a local search with gradient descent steps
    def local_search(h_vec, max_iter=20, step_size=0.01):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            # Compute gradient via finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-4
            base_val = compute_C5(h_new)
            for i in range(len(h_new)):
                h_plus = h_new.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus)
                val_plus = compute_C5(h_plus)
                grad[i] = (val_plus - base_val) / eps
            h_new = h_new - step_size * grad
            h_new = project_feasible(h_new)
