import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback initial construction
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 1000
    inertia = 0.9
    inertia_decay = 0.995
    cognitive_w = 2.0
    social_w = 2.0
    velocity_clamp = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        fitness = compute_C5(particle)
        p_best.append(particle.copy())
        p_best_fitness.append(fitness)
        
    # Global best
    best_idx = np.argmin(p_best_fitness)
    s_best = p_best[best_idx].copy()
    s_best_fitness = p_best_fitness[best_idx]
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 30
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        if iter_count % 50 == 0:
            if no_improvement_count > max_no_improvement:
                # Perturb all particles
                for i in range(n_particles):
                    particles[i] += np.random.normal(0, 0.03, n_points)
                    particles[i] = project_feasible(particles[i])
                no_improvement_count = 0
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_w * r1 * (p_best[i] - particles[i])
            social = social_w * r2 * (s_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project into feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < s_best_fitness:
                    s_best = particles[i].copy()
                    s_best_fitness = fitness
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        
        inertia *= inertia_decay
        iter_count += 1
    
    # Final check of best solution
    final_fitness = compute_C5(s_best)
    if final_fitness < s_best_fitness:
        s_best_fitness = final_fitness
        
    return (s_best.tolist(), s_best_fitness, n_points)
