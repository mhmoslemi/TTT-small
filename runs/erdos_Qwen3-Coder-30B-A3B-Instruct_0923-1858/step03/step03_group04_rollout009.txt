import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization parameters
    n_particles = 20
    max_iter = 200
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    # Use initial solution as global best
    global_best = h.copy()
    global_best_score = compute_C5(global_best)
    best_C5 = global_best_score
    best_h = global_best.copy()

    # Initialize swarm
    for i in range(n_particles):
        if i == 0:
            # Use initial solution for first particle
            pos = global_best.copy()
        else:
            # Add small random noise to initial solution
            pos = project_feasible(global_best + np.random.normal(0, 0.02, n_points))
        particles.append(pos.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        p_best.append(pos.copy())
        p_best_scores.append(compute_C5(pos))

    # PSO main loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.6:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (p_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate new position
            score = compute_C5(particles[i])
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    if score < best_C5:
                        best_C5 = score
                        best_h = global_best.copy()
        
        # Early stopping based on improvement
        if iteration > 0 and iteration % 10 == 0:
            if global_best_score > best_C5 + 1e-6:
                # Reset with current best
                for i in range(n_particles):
                    particles[i] = project_feasible(best_h + np.random.normal(0, 0.02, n_points))
                    velocities[i] = np.random.uniform(-v_max, v_max, n_points)
                    p_best[i] = particles[i].copy()
                    p_best_scores[i] = compute_C5(particles[i])

    # Local refinement with L-BFGS-B
    if elapsed() < budget_s * 0.9:
        def objective(h_vec):
            return compute_C5(h_vec)
            
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint_obj = {'type': 'eq', 'fun': constraint}
        
        # Use global best as starting point for local optimization
        try:
            result = minimize(
                objective,
                global_best,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraint_obj,
                options={'maxiter': 300, 'ftol': 1e-6, 'gtol': 1e-5}
            )
            if result.success:
                refined_h = project_feasible(result.x)
                refined_C5 = compute_C5(refined_h)
                if refined_C5 < best_C5:
                    best_h = refined_h
                    best_C5 = refined_C5
        except:
            pass

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
