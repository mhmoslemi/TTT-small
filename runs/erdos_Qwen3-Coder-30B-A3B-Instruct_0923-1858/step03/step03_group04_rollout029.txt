        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def pso_optimization():
        # PSO parameters
        swarm_size = 20
        max_iter = 500
        w = 0.7
        c1 = c2 = 1.5
        v_max = 0.1
        
        # Initialize swarm
        particles = []
        velocities = []
        pbest = []
        pbest_values = []
        
        for _ in range(swarm_size):
            h_init = project_feasible(h + np.random.normal(0, 0.05, n_points))
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            pbest.append(h_init.copy())
            pbest_values.append(compute_C5(h_init))
        
        # Global best
        gbest_idx = np.argmin(pbest_values)
        gbest = pbest[gbest_idx].copy()
        gbest_value = pbest_values[gbest_idx]

                                c2 * r2 * (gbest - particles[i]))
                
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])

                current_value = compute_C5(particles[i])
