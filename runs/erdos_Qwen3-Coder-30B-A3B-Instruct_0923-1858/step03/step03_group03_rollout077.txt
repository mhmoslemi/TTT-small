        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 0.5:
                h[i] = 2 * x[i]
            elif x[i] < 1.5:
                h[i] = 1
            else:
                h[i] = 2 * (2 - x[i])
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Compute gradient more accurately using finite differences
    def compute_gradient(h_vec, k_max):
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        h_ref = h_vec.copy()
        corr_ref = correlate(h_ref, 1 - h_ref, mode="full")
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[k_max] - corr_minus[k_max]) / (2 * eps)

    def refine_solution(h_vec, max_iter=30, adaptive_lr=True):
        h_new = h_vec.copy()
        lr = 0.03 if adaptive_lr else 0.015
        for i in range(max_iter):
            # Get the shift that gives maximum overlap
            corr = correlate(h_new, 1 - h_new, mode="full")
            k_max = np.argmax(corr)

            grad = compute_gradient(h_new, k_max)
            # Update step
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
