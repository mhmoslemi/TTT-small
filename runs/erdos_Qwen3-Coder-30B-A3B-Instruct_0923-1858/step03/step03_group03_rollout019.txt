import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize with initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Start with a coarse grid
        n_points = 30
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Compute C5 using correlate
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Project to feasible region
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simulated Annealing with grid refinement
    T = 0.1
    alpha = 0.999
    min_T = 1e-6
    iteration = 0
    patience = 0
    max_patience = 200
    
    while elapsed() < budget_s * 0.95 and T > min_T:
        # Perturb one random point
        idx = np.random.randint(0, n_points)
        h_new = h.copy()
        
        # Small random perturbation
        delta = np.random.normal(0, 0.1)
        h_new[idx] += delta
        
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Compute C5 for both
        old_C5 = compute_C5(h)
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        if new_C5 < old_C5:
            h = h_new
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_C5 - old_C5) / T):
                h = h_new
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h_new.copy()
                    patience = 0
            else:
                patience += 1
        
        # Reduce temperature
        T *= alpha
        
        # Grid refinement
        if iteration % 1000 == 0 and iteration > 0:
            # Double grid and interpolate
            n_points_new = n_points * 2
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_new_interp = np.interp(x_new, x_old, h)
            h_new_interp = project_feasible(h_new_interp)
            h = h_new_interp
            n_points = n_points_new
            dx = 2.0 / n_points

        iteration += 1
        if patience > max_patience:
            # Random restart
            h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_points) - 1))
            h = project_feasible(h)
            patience = 0

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
