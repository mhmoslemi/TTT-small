import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Bisection search approach to find optimal C5
    def solve_feasibility(h_vec, C):
        # Discretize shifts
        shifts = np.arange(-n_points + 1, n_points)
        n_shifts = len(shifts)
        # Sample a subset of shifts for efficiency
        sampled_shifts = shifts[::max(1, n_shifts // 50)]
        
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        for k in sampled_shifts:
            # Formulate correlation constraint
            if k >= 0:
                corr_expr = cp.sum(cp.multiply(h_var[:n_points-k], h_var[k:]))
            else:
                corr_expr = cp.sum(cp.multiply(h_var[-k:], h_var[:n_points+k]))
            
            # We want constraint that this is at least 1-C
            constraints.append(corr_expr >= 1 - C)
        
        # Dummy objective to solve infeasible problem
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return np.maximum(0, np.minimum(1, h_var.value))
            else:
                return None
        except:
            return None

    # Binary search on the C5 value
    low, high = 0.0, 1.0
    epsilon = 1e-4
    best_C = 0.0
    
    # Initial feasible point
    h_current = h.copy()
    
    # Take several sample points to try to find a better approach
    while elapsed() < budget_s * 0.95 and high - low > epsilon:
        mid = (low + high) / 2.0
        
        # Try solving feasibility with C = mid
        try:
            h_feasible = solve_feasibility(h_current, mid)
            if h_feasible is not None:
                # Project to ensure feasibility even after solving
                h_feasible = project_feasible(h_feasible)
                C5_test = compute_C5(h_feasible)
                if C5_test < best_C5:
                    best_C5 = C5_test
                    best_h = h_feasible.copy()
                    best_C = mid
                high = mid
            else:
                low = mid
        except:
            # If infeasible or error, go to higher bound
            low = mid
            
        # Regular local good step
        if elapsed() > budget_s * 0.7:
            break

    # Final check with best h
    current_C5 = compute_C5(best_h)
    if current_C5 < best_C5:
        best_C5 = current_C5

    # Try a simple gradient-based refinement
    def gradient_descent_refinement(h_vec, num_iter=100):
        step_size = 0.001
        for i in range(num_iter):
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_vec) - 1)
            
            # Estimate gradient
            eps = 1e-4
            grad = np.zeros_like(h_vec)
            for j in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_vec = h_vec - step_size * grad
            h_vec = project_feasible(h_vec)
            
            if i % 20 == 0:
                step_size *= 0.95
                if step_size < 1e-6:
                    break
        
        return h_vec

    if elapsed() < budget_s * 0.95:
        h_refined = gradient_descent_refinement(best_h.copy())
        final_C5 = compute_C5(h_refined)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = h_refined.copy()

    return (best_h.tolist(), best_C5, len(best_h))
