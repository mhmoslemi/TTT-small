import numpy as np
import time
from scipy.signal import correlate
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 100
    inertia = 0.9
    cognitive = 2.0
    social = 2.0
    mutation_rate = 0.05
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Start with diverse random solutions
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h_init.copy())
        pbest_values.append(compute_C5(h_init))

    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    # PSO main loop
    for iteration in range(max_iter):
        if elapsed() > budget_s * 0.9:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (pbest[i] - particles[i]) + 
                           social * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate and update personal best
            current_value = compute_C5(particles[i])
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value

            # Mutation
            if np.random.rand() < mutation_rate:
                perturb_idx = np.random.randint(n_points)
                particles[i][perturb_idx] = np.clip(
                    particles[i][perturb_idx] + np.random.normal(0, 0.02), 0, 1
                )
                particles[i] = project_feasible(particles[i])

        # Update inertia
        inertia = 0.9 - (0.5 * iteration / max_iter)
    
    # Final refinement with gradient-based optimization
    h = gbest.copy()
    current_C5 = compute_C5(h)
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = h.copy()
    
    # Local refinement with optimization
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)

        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    # Trust region refinement
    eta = 0.01
    for _ in range(200):
        if elapsed() > budget_s * 0.95:
            break
        grad = get_gradient(h)
        h_new = h - eta * grad
        h_new = project_feasible(h_new)
        
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            h = h_new
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        else:
            eta *= 0.9
            if eta < 1e-6:
                break

    return (best_h.tolist(), best_C5, len(best_h))
