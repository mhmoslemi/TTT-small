import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    best_h = initial_h.copy()

        def objective_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (1.0 / (np.sum(h_vec) * dx))

                h_plus[i] += eps

                h_minus[i] -= eps
