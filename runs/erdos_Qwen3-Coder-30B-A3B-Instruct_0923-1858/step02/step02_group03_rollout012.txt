import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        # Normalize
        initial_h = initial_h / (np.sum(initial_h) * dx)
    
    # Use convex optimization with McCormick relaxations
    start_time = time.time()
    
    # Variables
    h = cp.Variable(n_points)
    # Auxiliary variables for McCormick relaxation
    y = cp.Variable((n_points, n_points))
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    constraints.append(cp.sum(h) * dx == 1.0)  # Integral constraint
    
    # McCormick relaxations for y[i,k] = h[i] * h[i+k]
    for k in range(1, n_points):
        for i in range(n_points - k):
            # y[i,k] >= h[i] + h[i+k] - 1
            constraints.append(y[i,k] >= h[i] + h[i+k] - 1)
            # y[i,k] <= h[i]
            constraints.append(y[i,k] <= h[i])
            # y[i,k] <= h[i+k]
            constraints.append(y[i,k] <= h[i+k])
            # y[i,k] >= 0
            constraints.append(y[i,k] >= 0)
    
    # Overlap constraints
    t = cp.Variable()
    
    for k in range(1, n_points):
        # S(k) = sum_{i=0}^{n_p-1-k} (h[i] - y[i,k]) * dx <= t
        expr = cp.sum(h[:n_points-k] - y[:n_points-k, k]) * dx
        constraints.append(expr <= t)
        
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Attempt to solve
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            final_h = h.value
        else:
            # Fallback to original solution if optimization fails
            final_h = initial_h.copy()
    except Exception:
        # On failure, return the initial solution
        final_h = initial_h.copy()
    
    # Ensure it's feasible
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h / (np.sum(final_h) * dx)
    
    # Evaluate final C5
    corr = correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Make sure it's within budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        return (initial_h, evaluate(initial_h, dx), n_points)
    else:
        return (final_h, final_c5, n_points)

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
