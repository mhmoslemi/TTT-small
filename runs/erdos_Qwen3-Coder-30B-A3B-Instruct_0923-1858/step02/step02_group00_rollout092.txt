import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        # Create a triangular initial solution

            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Initialize swarm for PSO
    n_particles = 50
    swarm = []
    velocities = []
    pbest = []
    pbest_fitness = []
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Create initial population
    for _ in range(n_particles):
        particle = np.random.rand(n_points) * 0.5
        particle = project_feasible(particle)
        swarm.append(particle.copy())
        velocities.append(np.zeros(n_points))
        fitness = compute_C5(particle)
        pbest.append(particle.copy())
        pbest_fitness.append(fitness)
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = len(swarm) - 1
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_iter = 100
    c1_start = 2.0
    c2_start = 2.0
    c1_end = 1.0
    c2_end = 1.0
    
    # PSO optimization
    iter_count = 0
    max_pso_iter = 500
    update_rate = max_pso_iter // 10
    
    while elapsed() < budget_s * 0.6 and iter_count < max_pso_iter:
        w = w_start - (w_start - w_end) * (iter_count / max_pso_iter)
        c1_t = c1_start - (c1_start - c1_end) * (iter_count / max_pso_iter)
        c2_t = c2_start - (c2_start - c2_end) * (iter_count / max_pso_iter)

            velocities[i] = (w * velocities[i] +
                            c1_t * r1 * (pbest[i] - swarm[i]) +
                            c2_t * r2 * (swarm[gbest_idx] - swarm[i]))
            
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Update personal best
            fitness = compute_C5(swarm[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
        
        # Update best solution found
        if gbest_fitness < best_C5:
            best_C5 = gbest_fitness
            best_h = swarm[gbest_idx].copy()
        
        iter_count += 1
    
    # Gradient refinement on the best solution from PSO
    h = best_h.copy()

    def line_search(h_vec, grad, step_size=0.01, max_iter=10):
        for _ in range(max_iter):
            h_new = h_vec - step_size * grad
            h_new = project_feasible(h_new)
            if compute_C5(h_new) < compute_C5(h_vec):
                return h_new
            step_size *= 0.5
        return h_vec

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)

        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

    eta = 0.01
    refining_iterations = 0
    
    while elapsed() < budget_s and refining_iterations < 500:
        grad = estimate_gradient(h)
        h_new = h - eta * grad
        h_new = project_feasible(h_new)
        
        if compute_C5(h_new) < compute_C5(h):
            h = h_new
            if compute_C5(h) < best_C5:
                best_C5 = compute_C5(h)
                best_h = h.copy()
        else:
            eta *= 0.8
        
        refining_iterations += 1

    # Final adjustment & evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
