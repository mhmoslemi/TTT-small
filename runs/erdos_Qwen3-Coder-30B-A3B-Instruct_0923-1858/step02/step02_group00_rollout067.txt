import time

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    n_particles = 50
    max_iter = 100
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        # Initialize each particle with a random feasible solution
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand)
        particles.append(h_rand.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(h_rand.copy())
        pbest_vals.append(compute_C5(h_rand))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]

            r1, r2 = np.random.rand(2)

            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            particles[i] = project_feasible(particles[i])
