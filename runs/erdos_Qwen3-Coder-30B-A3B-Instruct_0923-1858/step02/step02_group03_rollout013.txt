def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate_erdos_solution(h):
    # This is declared externally; use it directly
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
