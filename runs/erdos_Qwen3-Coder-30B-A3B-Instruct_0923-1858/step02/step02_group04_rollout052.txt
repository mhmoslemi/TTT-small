import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex optimization approach
    def solve_convex_optimization(h0):
        # Create variables
        h = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # For each shift k, compute the correlation (overlap) constraint
            # We approximate this with linear constraints in the convex setting
            # But we still work with full correlation for actual evaluation
            corr_sum = 0
            for i in range(n_points):
                if i + k < n_points:
                    corr_sum += h[i] * (1 - h[i + k])
            constraints.append(corr_sum <= t)
            
        # Objective
        objective = cp.Minimize(t)
        
        # Create problem
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                sol_h = h.value
                return sol_h
        except:
            pass
        return h0
    
    # Initialize out of the loop
    h_trial = h.copy()
    
    # Local refinement step
    patience = 0
    max_patience = 50
    max_iter = 1000
    iter_count = 0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.98:
            break
            
        # Convex optimization step
        h_opt = solve_convex_optimization(h_trial)
        h_opt = project_feasible(h_opt)
        
        # Evaluate new solution
        current_C5 = compute_C5(h_opt)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_opt.copy()
            patience = 0
        else:
            patience += 1
        
        # Local search step
        if patience > max_patience and iter_count > 50:
            # Add noise to break local optima
            noise = np.random.normal(0, 0.02, n_points)
            h_trial = h_opt + noise
            h_trial = project_feasible(h_trial)
            patience = 0
        else:
            h_trial = h_opt
        
        iter_count += 1
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
