import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Parameter settings
    n_points_list = [50, 100, 200]  # Multi-resolution approach
    population_size = 10
    max_patience = 50
    initial_learning_rate = 0.05
    adaptive_decay = 0.995
    
    best_h = None
    best_C5 = float('inf')
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
        # Make it feasible for the specific constraint
        h = h * (n_points / 2) / np.sum(h)
        initial_pop = [h.copy()]
        for _ in range(population_size - 1):
            h_pert = h + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            initial_pop.append(h_pert)
    except NameError:
        # Create a triangular solution
        n_points = 50
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
        initial_pop = [h.copy()]
        for _ in range(population_size - 1):
            h_pert = h + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            initial_pop.append(h_pert)
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / len(h_vec))
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (len(h_vec) / 2) / current_sum
        return h_vec

    def adaptive_gradient_descent(h_vec, lr, max_iter=50):
        h_current = h_vec.copy()
        current_C5 = compute_C5(h_current)
        
        for _ in range(max_iter):
            # Compute gradient using finite differences for the maximum correlation term
            eps = 1e-5
            max_k_idx = np.argmax(correlate(h_current, 1 - h_current, mode="full"))
            max_k = max_k_idx - (len(h_current) - 1)
            
            grad = np.zeros_like(h_current)
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] += eps
                h_minus[i] -= eps

                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_current - lr * grad
            h_new = project_feasible(h_new)
            
            new_C5 = compute_C5(h_new)
            if new_C5 < current_C5:
                h_current = h_new
                current_C5 = new_C5
                lr *= 1.05  # Increase learning rate for faster convergence
            else:
                lr *= 0.5  # Reduce learning rate for stability
            
            if lr < 1e-6:
                break
                
        return h_current, current_C5
    
    def crossover(parent1, parent2, alpha=0.5):
        child = alpha * parent1 + (1 - alpha) * parent2
        return project_feasible(child)
    
    def mutate(parent, mu=0.1, sigma=0.05):
        # Gaussian mutation
        noise = np.random.normal(0, sigma, len(parent))
        mutated = parent + mu * noise
        return project_feasible(mutated)
    
    resolution_iter = 0
    for n_points in n_points_list:
        if resolution_iter > 0:
            # Transfer best solution to new resolution
            if best_h is not None:
                # Interpolate to new grid
                old_x = np.linspace(0, 2, len(best_h))
                new_x = np.linspace(0, 2, n_points)
                h_interp = np.interp(new_x, old_x, best_h)
                # Project to meet the constraint
                h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
                initial_pop = [h_interp]
                for _ in range(population_size - 1):
                    h_pert = h_interp + np.random.normal(0, 0.03, n_points)
                    h_pert = np.clip(h_pert, 0, 1)
                    h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                    initial_pop.append(h_pert)
            else:
                initial_pop = [np.zeros(n_points)]
                for _ in range(population_size):
                    h = np.random.rand(n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    initial_pop.append(h)
        
        population = [project_feasible(h) for h in initial_pop]
        
        patience = 0
        learning_rate = initial_learning_rate
        while True:
            elapsed = time.time() - start_time
            if elapsed >= time_limit:
                break
            
            # Evaluate population
            fitness = [compute_C5(h) for h in population]
            
            # Track best
            min_fitness = min(fitness)
            if min_fitness < best_C5:
                best_C5 = min_fitness
                best_h = population[np.argmin(fitness)].copy()
                patience = 0
            else:
                patience += 1
            
            if patience > max_patience:
                # Restart with new population
                population = [project_feasible(np.random.rand(n_points) * 0.5) for _ in range(population_size)]
                patience = 0
            
            # Gradient refinement
            for i in range(population_size):
                population[i], _ = adaptive_gradient_descent(population[i], learning_rate)
            
            # Selection + Crossover + Mutation
            indices = np.argsort(fitness)
            selected = [population[indices[i]] for i in range(population_size // 2)]
            new_population = selected.copy()
            
            # Crossover and mutation for new generation
            for _ in range(population_size - len(selected)):
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_population.append(child)
            
            population = new_population
            
            learning_rate *= adaptive_decay
            
            # Early pruning if good enough
            if best_C5 <= 0.3807:
                break
                
        resolution_iter += 1
            
    if best_C5 < float('inf'):
        return (best_h.tolist(), best_C5, len(best_h))
    else:
        # Fallback
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.array([0.5]*50)
        h = project_feasible(h)
        final_C5 = compute_C5(h)
        return (h.tolist(), final_C5, len(h))
