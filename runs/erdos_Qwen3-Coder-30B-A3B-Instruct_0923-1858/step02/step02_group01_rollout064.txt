import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, dx, mutation_rate=0.1):
    """Mutate an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def tournament_selection(population, fitness, tournament_size=3):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx]

    dx = 2.0 / n_points

    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    max_gens = 100000
    
    # Initialize

        if len(initial_h) != n_points:

    else:
        for _ in range(population_size):
            population.append(generate_individual(n_points, dx))
    
    # Main GA loop
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    best_C = float('inf')
    best_h = None
    
    gen = 0
    stagnation_counter = 0
    
    while time.time() - t0 < max_time and gen < max_gens:

        fitness = [evaluate(individual, dx) for individual in population]
        
        # Track best solution

        if fitness[current_best_idx] < best_C:
            best_C = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Adapt mutation rate if needed
        if stagnation_counter > 50:

        if stagnation_counter > 100:
            # Reset with better initialization
            population = []

        while len(new_population) < population_size:

            if np.random.rand() < crossover_prob:
                child1, child2 = crossover(parent1, parent2)

            child1 = mutate(child1, dx, mutation_rate)
            child2 = mutate(child2, dx, mutation_rate)
            
            new_population.append(child1)
            if len(new_population) < population_size:
