import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def fourier_to_h(a, b, t):
    # a[0] is the DC component, a[1:] are cosine coefficients, b[1:] are sine coefficients
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n-1] * np.sin(n * np.pi * t)
    return h

    # PSO parameters
    N = 10  # Number of Fourier coefficients (half of total)
    P = 50  # Number of particles
    max_iter = 1000
    
    # Initialize particles
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N))  # 2*N for a and b coefficients
    velocities = np.random.uniform(-0.1, 0.1, (P, 2*N))
    pbest = particles.copy()
    pbest_scores = np.full(P, np.inf)
    
    # Global best
    gbest = None
    gbest_score = np.inf
    
    start_time = time.time()
    max_time = budget_s - 10
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(P):
            # Convert coefficients to h(x)
            a = particles[i, :N]
            b = particles[i, N:]
            
            t = np.linspace(0, 2, n_points)
            h = fourier_to_h(a, b, t)
            
            # Project to feasible region
            h = project(h, dx)
            
            # Evaluate
            score = evaluate(h, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
        
        # Update velocities and positions
        omega = 0.7
        phi1, phi2 = 1.5, 1.5
        
        for i in range(P):
            r1, r2 = np.random.rand(2)

                            phi1 * r1 * (pbest[i] - particles[i]) +
                            phi2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
    
    # Convert best to final h
    if gbest is not None:
        a = gbest[:N]
        b = gbest[N:]
        t = np.linspace(0, 2, n_points)
        h = fourier_to_h(a, b, t)
        h = project(h, dx)
        final_score = evaluate(h, dx)
