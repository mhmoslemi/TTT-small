import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Convex optimization approach
    dx = 2.0 / n_points
    
    # Define optimization variables
    h_vars = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        t >= 0
    ]
    
    # Add the overlap constraints
    for k in range(2 * n_points - 1):
        # Compute the convolution-like constraint manually
        start_idx = max(0, k - n_points + 1)
        end_idx = min(n_points, k + 1)
        
        # Values of h in lags [start_idx, end_idx)
        h_vals = h_vars[start_idx:end_idx]
        # Values of (1 - h) in lags [start_idx, end_idx) but shifted by k
        # This involves working with the proper indices in the full corr 
        # but we'll simulate this as an inner product of the overlapping
        # parts
        
        # We are computing sum h[i] * (1 - h[i + k]) for i = 0..n_points-1-k
        if k < n_points:
            # When k is positive (h is shifted to the right)
            overlapping_h = h_vars[:n_points-k]
            overlapping_1_minus_h = h_vars[k:]
            overlap_expr = cp.sum(cp.multiply(overlapping_h, 1 - overlapping_1_minus_h))
            constraints.append(overlap_expr <= t)
        else:
            # When k >= n_points, shifted too much - will be zeros
            # start_idx would be k - n_points + 1 so it's > 0, and end_idx = n_points
            # so we are computing sum over indices [k - n_points + 1, n_points-1]
            # then indices of h[k-np+1:n_points] * (1 - h[:n_points - (k-np+1)])  
            # This requires careful indexing
            # The hardest case to get aright is when k becomes too large such that start_idx >= end_idx
            # Let's do it more carefully
            
            # We know overlap is where we sum h[i] * (1 - h[i + k]) for valid i
            # i ranges from 0 to n_points - k -1 inclusive when k < n_points
            # and must be i+k < n_points so i < n_points - k
            # else we shift things too much, everything is 0
            if k >= n_points:
                # When k >= n_points, shift makes it 0 for all, so this constraint is redundant
                continue
            else:
                # Valid case: k < n_points, we compute  
                # sum_{i=0}^{n_points-1-k} h[i] * (1 - h[i+k])
                # It is equivalent to the above formula
                overlapping_h = h_vars[:n_points-k]
                overlapping_1_minus_h = h_vars[k:]
                overlap_expr = cp.sum(cp.multiply(overlapping_h, 1 - overlapping_1_minus_h))
                constraints.append(overlap_expr <= t)
    
    # Objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        # Solve the problem
        prob.solve(solver=cp.MOSEK, verbose=False)
        
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_sol = h_vars.value
            t_sol = t.value
        else:
            # Fallback to current solution
            h_sol = h
            t_sol = np.max(correlate(h, 1 - h, mode="full")) * dx
            
    except Exception:
        # Fallback to deterministic solution
        h_sol = h
        t_sol = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    # Ensure solution is valid
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
    
    # Final evaluation
    final_C5 = np.max(correlate(h_sol, 1 - h_sol, mode="full")) * dx
    
    return (h_sol.tolist(), final_C5, n_points)
