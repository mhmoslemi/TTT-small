import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 100
    inertia_w = 0.9
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_fitness = []
    
    for _ in range(n_particles):
        h_init = h + np.random.normal(0, 0.05, n_points)
        h_init = project_feasible(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h_init.copy())
        p_best_fitness.append(compute_C5(h_init))
    
    # Global best
    global_best_idx = np.argmin(p_best_fitness)
    global_best = p_best[global_best_idx].copy()
    global_best_fitness = p_best_fitness[global_best_idx]

    # PSO iterations
    for iter_count in range(max_iter):
        if elapsed() > budget_s * 0.8:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            fitness = compute_C5(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness

    # Refinement with gradient descent
    refined_h = global_best.copy()
    
    # Gradient estimation with finite differences
    def compute_gradient(h_vec, eps=1e-4):
        h_vec = np.array(h_vec)
        corr_base = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr_base)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        return grad

    # Trust region optimization
    def trust_region_update(h_vec, max_iter=50, eta=0.01):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            grad = compute_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    refined_h = trust_region_update(global_best)
    refined_C5 = compute_C5(refined_h)
    
    if refined_C5 < best_C5:
        best_C5 = refined_C5
        best_h = refined_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
