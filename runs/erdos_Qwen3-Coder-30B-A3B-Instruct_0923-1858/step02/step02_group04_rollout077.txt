import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create triangular
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    # Define the optimization problem with CVXPY
    dx = 2.0 / n_points
    
    # Optimization variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # CVXPY constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        t >= 0
    ]
    
    # Constraint for each shift k
    for k in range(n_points):
        # Create the overlap for shift k
        k_start = max(0, k)
        k_end = min(n_points, n_points + k)
        h_k_start = max(0, -k)
        h_k_end = min(n_points, n_points - k)
        
        shift_expr = 0
        for i in range(h_k_start, h_k_end):
            shift_expr += h_var[i] * (1 - h_var[i + k])
        
        constraints.append(shift_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-6, abstol=1e-6)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_solution = h_var.value
        else:
            h_solution = h.copy()
    except:
        h_solution = h.copy()
        
    # Post-processing to make sure the solution is feasible
    h_solution = np.clip(h_solution, 0, 1)
    h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
    
    # Compute actual C5
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    final_C5 = compute_C5(h_solution)
    
    return (h_solution.tolist(), final_C5, n_points)
