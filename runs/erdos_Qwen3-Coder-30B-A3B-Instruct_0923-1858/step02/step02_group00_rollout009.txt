import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, k_max):
        h_vec = np.array(h_vec)
        # Evaluate gradient w.r.t. all variables for the specific k_max shift
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            # Compute correlation for both
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[k_max + len(h_vec) - 1] - corr_minus[k_max + len(h_vec) - 1]) / (2 * eps)
            
        return grad

    # Enhanced main optimization loop with multiple strategies
    eta = 0.01
    patience = 0
    max_patience = 30
    iter_count = 0
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    
    # Main optimization loop
    while elapsed() < budget_s * 0.95:
        # Update the best solution so far
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
            patience = 0
        else:
            patience += 1

        # Periodically update projection and projections
        current_h = project_feasible(current_h)
        current_C5 = compute_C5(current_h)
        
        # Find which k maximizes the correlation to focus gradient
        corr = correlate(current_h, 1 - current_h, mode="full")
        k_max_idx = np.argmax(corr)
        k_max = k_max_idx - (len(current_h) - 1)

        # Compute gradient
        grad = get_gradient(current_h, k_max)
        
        # Take step
        current_h = current_h - eta * grad
        current_h = project_feasible(current_h)
        
        # Compute new objective
        current_C5 = compute_C5(current_h)
        
        # Adaptive learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.95

        # Perturb occasionally to escape local minima
        if patience > max_patience:
            idxs = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in idxs:
                current_h[i] = np.clip(current_h[i] + np.random.normal(0, 0.05), 0, 1)
            current_h = project_feasible(current_h)
            patience = 0
            eta = max(0.001, eta * 0.9)

        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
