import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    n = len(h)
    # Compute correlation manually to avoid padding effects
    max_corr = 0
    for k in range(-n+1, n):
        corr = 0
        if k >= 0:
            for i in range(n-k):
                corr += h[i] * (1 - h[i+k])
        else:
            for i in range(-k, n):
                corr += h[i] * (1 - h[i+k])
        max_corr = max(max_corr, corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    dx = 2.0 / n_points
    max_iter = 1000
    n_particles = 50
    w, c1, c2 = 0.7, 2.0, 2.0
    v_max = 0.2
    
    try:

        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)
        initial_h = project(initial_h, dx)
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for _ in range(n_particles):
        # Start from initial_h with small random perturbations
        h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate(h, dx)
        p_best.append(h.copy())
        p_best_scores.append(score)
    
    # Global best
    best_idx = np.argmin(p_best_scores)
    g_best = p_best[best_idx].copy()
    g_best_score = p_best_scores[best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2
    
    for it in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (p_best[i] - particles[i]) +
                            c2 * r2 * (g_best - particles[i]))
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            score = evaluate(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = particles[i].copy()
                
                # Update global best if needed
                if score < g_best_score:
                    g_best_score = score
                    g_best = particles[i].copy()
    
    # Final validation
    final_h = project(g_best, dx)
    final_score = evaluate(final_h, dx)
    
    return (final_h, final_score, n_points)
