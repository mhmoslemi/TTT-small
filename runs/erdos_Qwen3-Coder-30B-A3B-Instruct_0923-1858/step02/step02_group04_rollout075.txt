import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95
    
    # Initialize
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial triangular function
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Generate all possible shifts for overlap computation
    max_shift = n_points - 1
    shifts = np.arange(-max_shift, max_shift + 1)
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def solve_convex(h_in):
        # Set up the convex optimization problem
        h = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = []
        
        # h[i] in [0, 1]
        for i in range(n_points):
            constraints.append(h[i] >= 0)
            constraints.append(h[i] <= 1)
        
        # Sum constraint
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Overlap constraints
        # For each shift k, compute the corresponding overlap
        # This is approachable via the full correlation
        # But for large dimension, let's use CVXPY's warm-start options
        
        # Create matrix for overlap constraints (simplified approach)
        # We basically want max_k (sum h[i] * (1 - h[i+k])) <= t
        # This is equivalent to each overlap <= t
        # We'll compute it iteratively for constraints
        
        # Direct approach: compute overlap for each shift k manually
        # Time intensive but more accurate than approximation
        
        def create_overlap_constraints():
            constraints = []
            h_vec = h
            # Use correlation result to get overlap values for all shifts
            # But directly building a system for Cvxpy
            # We compute overlaps using equivalent expression
            for k in range(-max_shift, max_shift + 1):
                left = max(0, -k)
                right = min(n_points, n_points - k)
                if right > left:
                    # Sum over valid indices
                    if k >= 0:
                        overlap_sum = cp.sum(cp.multiply(h_vec[left:right], 1 - h_vec[left+k:right+k]))
                    else:
                        overlap_sum = cp.sum(cp.multiply(h_vec[left+k:right+k], 1 - h_vec[left:right]))
                    constraints.append(overlap_sum <= t)
            return constraints
        
        try:
            # Convex optimization approach
            constraints = create_overlap_constraints()
            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                return h.value
            else:
                # Fallback solution
                return h_in
        except:
            return h_in
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def find_optimal_solution(initial_h):
        # This is the convex optimization approach
        try:
            solution = solve_convex(initial_h)
            solution = project_feasible(solution)
            return solution
        except:
            return initial_h
    
    # Local refinement function to improve solution
    def local_refine(h_in, max_iter=20):
        h_current = h_in.copy()
        for _ in range(max_iter):
            if time.time() - start_time >= time_limit:
                break
            new_h = find_optimal_solution(h_current)
            current_C5 = compute_C5(h_current)
            new_C5 = compute_C5(new_h)
            if new_C5 < current_C5:
                h_current = new_h
            else:
                break
        return h_current
    
    # Initial search
    refined_h = local_refine(h)
    current_C5 = compute_C5(refined_h)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = refined_h.copy()
    
    # Additional refinement with perturbations
    patience = 0
    max_patience = 20
    for iteration in range(100):
        if time.time() - start_time >= time_limit:
            break
            
        # Random perturb and refine
        idx = np.random.choice(n_points, size=min(5, n_points//10), replace=False)
        pert_h = refined_h.copy()
        for i in idx:
            pert_h[i] += np.random.normal(0, 0.02)
        pert_h = project_feasible(pert_h)
        pert_h = local_refine(pert_h)
        
        new_C5 = compute_C5(pert_h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = pert_h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                break
    
    # Final optimization on solution
    final_h = local_refine(best_h)
    final_C5 = compute_C5(final_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
