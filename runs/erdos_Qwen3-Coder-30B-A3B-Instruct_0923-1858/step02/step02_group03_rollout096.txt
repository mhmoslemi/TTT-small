def evaluate_erdos_solution(h, dx):
    """Evaluate the overlap integral."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def adaptive_gradient_descent(h, dx, start_time, budget_s):
    """Improved gradient descent"""
    max_time = budget_s - 10
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h, dx)
    
    lr = 0.02
    patience = 0
    max_patience = 100
    
    for i in range(50000):
        if time.time() - start_time > max_time:
            break
        
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        h_new = h - lr * grad
        h_new = project(h_new, dx)
        
        current_C = evaluate_erdos_solution(h_new, dx)
        
        if current_C < best_C:
            h = h_new
            best_C = current_C
            best_h = h.copy()
            patience = 0
            lr = min(0.05, lr * 1.05)
        else:
            h = h_new
            patience += 1
            if patience > max_patience:
                lr = max(1e-5, lr * 0.95)
                patience = 0
                
        if i % 30 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.01, len(h))
            h = project(h + noise, dx)
            
    return (best_h, best_C)

def project(h, dx):
    """Project h onto feasible region"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h
