import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values, dx):
    # Compute the overlap integral for the given h
    n = len(h_values)
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    # Initialize h
    try:
        h = initial_h_values
        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        h = np.ones(n_points) * 0.5  # Uniform starting point

    start_time = time.time()
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h, dx)
    
    # Use cutting plane method with CVX
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Get the worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        k_worst = np.argmax(corr)
        worst_val = corr[k_worst]

        # Set up CVX problem
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(sum(h_vars) * dx == 1)
        
        # Add constraint for the worst shift
        # S(k) = 1 - \int h(x) h(x+k) dx 
        shifted = np.zeros(n_points)
        if k_worst < n_points:
            shifted[k_worst:] = h[:n_points-k_worst]
        else:
            shifted[:n_points-(k_worst-n_points)] = h[k_worst-n_points:]
        h_shifted = shifted
        
        # Just a basic approximation of the integral
        integral_approx = cp.sum(cp.multiply(h_vars, h_shifted))
        S_k = 1 - integral_approx
        constraints.append(S_k <= t)
        
        # Objective
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_vars.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (1.0 / (np.sum(h_new) * dx))
                new_C = evaluate_erdos_solution(h_new, dx)
                if new_C < best_C:
                    best_h = h_new
                    best_C = new_C
                    h = h_new
        except:
            pass
        
        # Project h to feasible region
        h = np.clip(h, 0, 1)
        h = h * (1.0 / (np.sum(h) * dx))
        
    return (best_h, best_C, n_points)
