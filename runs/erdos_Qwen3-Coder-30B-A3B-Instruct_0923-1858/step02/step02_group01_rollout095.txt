import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def perturb(h, dx):
    """Generate a neighbor solution by perturbing h."""
    perturbation = np.random.uniform(-0.01, 0.01, size=h.shape)
    h_candidate = h + perturbation
    return project(h_candidate, dx)

def simulated_annealing(h, dx, budget_s):
    """Run simulated annealing to minimize C5."""
    T_initial = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    T = T_initial
    best_h = h.copy()
    best_C = evaluate(h, dx)
    start_time = time.time()
    
    # Counter for stagnation detection
    stagnation_count = 0
    prev_best_C = best_C
    
    while T > T_min and time.time() - start_time < budget_s - 10:
        h_candidate = perturb(h, dx)
        current_C = evaluate(h_candidate, dx)
        
        delta = current_C - best_C
        
        # Metropolis criterion
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h = h_candidate.copy()
            best_C = current_C
            stagnation_count = 0  # Reset stagnation counter
            
        # Update temperature
        if current_C < best_C:
            T = T * cooling_rate
        else:
            stagnation_count += 1
            if stagnation_count > 500:
                # If we're stagnating, increase temperature slightly
                T = min(T * 1.1, T_initial)
                stagnation_count = 0
        
        # Adaptive cooling rate: speed up when improvement is slow
        if prev_best_C - best_C < 1e-6:
            cooling_rate = min(cooling_rate * 1.01, 0.999)
        else:
            cooling_rate = max(cooling_rate * 0.999, 0.95)
            
        prev_best_C = best_C
    
    return best_h, best_C
