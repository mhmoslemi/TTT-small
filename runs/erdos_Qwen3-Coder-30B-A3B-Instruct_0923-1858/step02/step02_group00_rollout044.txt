import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution with more points for better resolution
        n_points = 64
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use a hybrid approach: PSO for global search and gradient descent for local refinement
    def pso_optimization():
        # PSO parameters
        n_particles = 50
        max_iter = 100
        inertia_w = 0.9
        cognitive = 2.0
        social = 2.0
        max_velocity = 0.1
        
        # Initialize particles
        particles = []
        velocities = []
        best_positions = []
        best_values = []
        
        for _ in range(n_particles):
            # Initialize with small random perturbations around the current solution
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_feasible(particle)
            particles.append(particle.copy())
            velocities.append(np.zeros(n_points))
            best_positions.append(particle.copy())
            best_values.append(compute_C5(particle))
        
        global_best_idx = np.argmin(best_values)
        global_best = best_positions[global_best_idx].copy()
        global_best_value = best_values[global_best_idx]
        
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.9:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (inertia_w * velocities[i] + 
                                cognitive * r1 * (best_positions[i] - particles[i]) + 
                                social * r2 * (global_best - particles[i]))
                
                # Limit velocity
                velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Evaluate
                current_value = compute_C5(particles[i])
                
                # Update personal best
                if current_value < best_values[i]:
                    best_values[i] = current_value
                    best_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < global_best_value:
                    global_best_value = current_value
                    global_best = particles[i].copy()
            
            # Decrease inertia weight
            inertia_w = 0.9 - (0.5 * iteration / max_iter)
        
        return global_best, global_best_value
    
    def gradient_descent_refinement(h_start):
        h_refined = h_start.copy()
        max_iter = 1000
        step_size = 0.005
        tolerance = 1e-6
        
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.9:
                break
                
            # Compute current objective and get the maximum correlation index
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_k_idx = np.argmax(corr)
            
            # Compute finite difference gradient
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus)
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update with gradient descent
            h_refined = h_refined - step_size * grad
            h_refined = project_feasible(h_refined)
            
            # Check convergence
            if np.linalg.norm(grad) < tolerance:
                break
                
            # Adaptive step size
            current_C5 = compute_C5(h_refined)
            if iteration % 10 == 0:
                step_size *= 0.95
        
        return h_refined
    
    # Main loop
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        # PSO phase
        psobest_h, psobest_C5 = pso_optimization()
        
        # Gradient refinement phase
        refined_h = gradient_descent_refinement(psobest_h)
        refined_C5 = compute_C5(refined_h)
        
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
        
        # Occasionally reinitialize with a new random perturbation
        if iter_count % 10 == 0:
            h = np.random.rand(n_points) * 0.5
            h = project_feasible(h)
        
        iter_count += 1
        
        if elapsed() > budget_s * 0.95:
            break

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
