        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

        # Sample a subset of possible shifts
        shifts = np.arange(-max_shift, max_shift + 1)
        if len(shifts) <= n_samples:
            return shifts
        return np.random.choice(shifts, n_samples, replace=False)

    # Efficient optimization approach
    def optimize_step(h_vec, lr=0.005, n_shifts=15, max_shift=15):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute gradient by finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vec)

            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
