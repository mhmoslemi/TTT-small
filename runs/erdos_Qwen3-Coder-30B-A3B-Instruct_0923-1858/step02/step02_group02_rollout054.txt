import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        h = np.zeros(n_points)
        h[10:35] = 0.5
        
    # PSO parameters
    n_particles = 50
    max_iter = 100000
    inertia = 0.7
    cog_param = 1.5
    soc_param = 1.5
    max_velocity = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Initialize some particles using the current solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    p_best.append(h.copy())
    p_best_vals.append(evaluate(h, dx))
    
    for i in range(1, n_particles):
        if i < n_particles * 0.1:  # 10% use perturbed initial solution
            h_pert = h.copy()
            perturb_idx = np.random.randint(0, n_points, 5)
            for idx in perturb_idx:
                h_pert[idx] += np.random.normal(0, 0.05)
            h_pert = project(h_pert, dx)
        else:  # Generate random particles
            h_rand = np.random.rand(n_points)
            h_rand = project(h_rand, dx)
        particles.append(h_rand)
        velocities.append(np.random.rand(n_points) * 0.1)
        p_best.append(h_rand.copy())
        p_best_vals.append(evaluate(h_rand, dx))
    
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # PSO loop
    t0 = time.time()
    max_time = budget_s - 10
    
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cog_param * r1 * (p_best[i] - particles[i]) + 
                            soc_param * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
    
    return (g_best, g_best_val, n_points)
