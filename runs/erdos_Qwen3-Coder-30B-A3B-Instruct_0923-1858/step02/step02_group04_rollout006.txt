import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    time_limit = budget_s * 0.95
    
    # Use initial solution if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a piecewise linear initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            if x[i] <= 1:
                h[i] = x[i]
            else:
                h[i] = 2 - x[i]
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # correlation with mode='full', no circular boundary
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # We only care about the valid range of correlations (no wrapping)
        # The full correlation has length 2*n_points -1
        # The valid range is from 0 to 2*n_points-2 and we want max_k in [0, 2*n_points-2]
        # But for k = 0 to n_points-1, correlation corresponds to shifts of 0 to n_points-1
        # For k = n_points to 2*n_points-2, shifts of -n_points+1 to -1 => not in our domain
        # But since we're looking for max_k, we're fine with the entire range
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing with local perturbations
    T = 1.0
    cooling_rate = 0.99
    min_T = 1e-6
    mutation_step = 0.1
    
    iter_count = 0
    max_iter = 100000
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = current_h.copy()
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        
        iter_count += 1
        
        # Mutate solution
        new_h = current_h.copy()
        num_mutations = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=num_mutations, replace=False)
        for idx in indices:
            delta = np.random.normal(0, mutation_step)
            new_h[idx] += delta
        
        # Project to feasible region
        new_h = project_feasible(new_h)
        
        # Compute new value
        new_C5 = compute_C5(new_h)
        
        # Accept or reject
        delta = new_C5 - current_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_C5 = new_C5
            
            # Update best
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
        
        # Cooling
        T = max(min_T, T * cooling_rate)
        
        # Adjust mutation step based on temperature
        mutation_step = 0.1 * T
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
