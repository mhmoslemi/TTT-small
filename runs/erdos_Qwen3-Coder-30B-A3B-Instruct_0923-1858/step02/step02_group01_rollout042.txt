import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_kernel(sigma):
    """Create a symmetric Hamming window kernel for smoothing."""
    kernel_size = 2 * sigma + 1
    kernel = np.zeros(kernel_size)
    for i in range(-sigma, sigma+1):
        if i < 0:
            val = 0.54 - 0.46 * np.cos(np.pi * i / sigma)
        else:
            val = 0.54 + 0.46 * np.cos(np.pi * i / sigma)
        kernel[i + sigma] = val
    kernel /= kernel.sum()  # Ensure kernel sums to 1
    return kernel

def compute_smooth_gradient(h, h_smooth, dx, kernel):
    """Compute gradient using smoothed solution."""
    eps = 1e-5
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus_smooth = np.convolve(h_plus, kernel, mode='same')
        c_plus = evaluate(h_plus_smooth, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus_smooth = np.convolve(h_minus, kernel, mode='same')
        c_minus = evaluate(h_minus_smooth, dx)
        
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def refine_solution(h, dx, n_points_new):
    """Upsample and refine the solution."""

    x_new = np.linspace(0, 2, n_points_new)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, 2.0 / n_points_new)
