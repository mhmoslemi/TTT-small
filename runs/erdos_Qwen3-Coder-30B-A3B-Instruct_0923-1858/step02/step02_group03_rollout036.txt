import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    try:
        h_init = initial_h_values
        if len(h_init) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Fallback to a smooth initialization
        t = np.linspace(0, 2, n_points)
        h_init = np.exp(-((t - 1)**2) / 0.2)
        h_init = h_init / (np.sum(h_init) * dx)
    
    start_time = time.time()
    
    # Create variables
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) * dx == 1, t >= 0]
    
    # McCormick relaxation for the bilinear terms
    # For each shift k, introduce auxiliary variables y[i,k] = h[i] * h[i+k]
    # We build a matrix to hold the computations
    M = n_points
    y = Variable((M, M))  # y[i,k] = h[i] * h[i+k]
    
    # Add McCormick constraints
    for k in range(1, M):
        for i in range(M - k):
            # y[i,k] >= h[i] + h[i+k] - 1
            constraints.append(y[i, k] >= h[i] + h[i + k] - 1)
            # y[i,k] <= h[i]
            constraints.append(y[i, k] <= h[i])
            # y[i,k] <= h[i+k]
            constraints.append(y[i, k] <= h[i + k])
            # y[i,k] >= 0
            constraints.append(y[i, k] >= 0)
    
    # Now add the overlap constraints
    for k in range(1, M):
        # S(k) = sum_i h[i] - y[i,k] for valid i, and subtract the overlap
        # Actually let's reformulate:
        # S(k) = sum_i (h[i] - (h[i] * h[i+k])) * dx
        #      = sum_i h[i] * (1 - h[i+k]) * dx
        # But the way the problem is stated, it's easier to evaluate:
        # Overlap integral for shift k = sum_i h[i] * (1 - h[i+k]) * dx
        # # Add constraint that this is <= t
        # Overlap contribution for shift k
        S = 0
        for i in range(M - k):
            S += h[i] * (1 - h[i + k]) * dx
        constraints.append(S <= t)

    # Objective
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # Try to solve with available solvers
    solver = None
    solvers = [ECOS, OSQP, CVXOPT, GLPK, GUROBI, MOSEK]
    for s in solvers:
        try:
            problem.solve(solver=s, verbose=False)
            solver = s
            break
        except:
            continue
    
    if solver is None:
        # Fallback to the best we can do with reasonable solver
        try:
            problem.solve(solver=ECOS, verbose=False)
        except:
            # If we can't solve, use the initial solution
            h_values = h_init
            c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
            return (h_values, c5_bound, n_points)
    
    # Extract solution
    h_solution = h.value
    if h_solution is None:
        h_values = h_init
        c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
        return (h_values, c5_bound, n_points)
    
    # Apply final projection to ensure all constraints are tight
    h_values = np.clip(h_solution, 0, 1)
    h_values = h_values / (np.sum(h_values) * dx)
    
    # Evaluate final C5 bound
    c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
    
    # Ensure we return within time
    if time.time() - start_time > budget_s - 10:
        # If we're running out of time, return the best we had
        pass

    return (h_values, c5_bound, n_points)
