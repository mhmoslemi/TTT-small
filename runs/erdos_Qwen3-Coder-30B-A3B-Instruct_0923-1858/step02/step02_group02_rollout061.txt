import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

        segments = 10
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)
    
    # Initialize
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    # Parameters for alternating optimization
    max_iter = 10000
    lr_pos = 0.1
    lr_height = 0.1
    decay_rate = 0.99
    segment_count = 10
    patience = 100
    
    # Track the best solution
    best_eval = best_C
    no_improvement = 0
    
    # Time tracking
    t0 = time.time()
    max_time = budget_s - 10  # Leave headroom for cleanup
    
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        # Alternative 1: optimize heights with fixed positions
        if iteration % 2 == 0:
            # Segment positions
            pos_indices = np.linspace(0, n_points-1, segment_count+1, dtype=int)
            pos_indices[-1] = n_points-1  # Ensure last index
            pos = (pos_indices[1:] + pos_indices[:-1]) / 2.0
            
            # Heights optimization via CVXPY
            heights = cp.Variable(segment_count)
            # Subject to constraints
            constraints = [
                cp.sum(cp.multiply(heights, dx)) == 1,
                heights >= 0,
                heights <= 1
            ]
            
            # Assemble piecewise h from heights
            h_trial = np.zeros(n_points)
            for i in range(segment_count):
                start = pos_indices[i]
                end = pos_indices[i+1]
                h_trial[start:end] = heights[i]
            
            # Compute DFT-based correlation to get max overlap
            corr = correlate(h_trial, 1 - h_trial, mode='full')
            max_corr = cp.Max(cp.multiply(corr, dx))
            
            # Make a simple approximation of the objective
            # instead of minimizing full overlap, we minimize a heuristic:
            objective = cp.Minimize(max_corr)
            
            try:
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    final_height = heights.value
                    # Update h from optimized heights
                    h_trial_new = np.zeros(n_points)
                    for i in range(segment_count):
                        start = pos_indices[i]
                        end = pos_indices[i+1]
                        h_trial_new[start:end] = final_height[i]
                    h = project(h_trial_new, dx)
            except:
                # Fall back if CVXPY optimization fails
                pass
        
        # Alternative 2: optimize positions with fixed heights (simpler)
        else:
            pos_indices = np.linspace(0, n_points-1, segment_count+1, dtype=int)
            pos_indices[-1] = n_points-1

            for i in range(len(pos)):

            # Reconstruct hybrid h
            h_new = np.zeros(n_points)
            for i in range(segment_count):

                if end <= start:
                    end = start + 1
                h_new[start:end] = h[start:end].mean()
            h = project(h_new, dx)
        
        # Fine-tune with gradient descent updates
        if iteration % 5 == 0:
            # Estimate gradient and update
            epsilon = 1e-4
            grad = np.zeros_like(h)
            h_plus = h.copy()
            h_minus = h.copy()
            
            for i in range(len(h)):
                h_plus[i] += epsilon
                h_plus = project(h_plus, dx)
                corr_plus = correlate(h_plus, 1-h_plus, mode='full')
                corr_plus_max = np.max(corr_plus)
                
                h_minus[i] -= epsilon
                h_minus = project(h_minus, dx)
                corr_minus = correlate(h_minus, 1-h_minus, mode='full')
                corr_minus_max = np.max(corr_minus)
                
                grad[i] = (corr_plus_max - corr_minus_max) / (2 * epsilon)
                h_plus[i] -= epsilon
                h_minus[i] += epsilon
                
            # Update h
            h = h - lr_height * grad
            h = project(h, dx)
        
        # Update learning rates
        if iteration % 100 == 0:
            lr_height *= decay_rate
            lr_pos *= decay_rate
        
        # Track best solution
        current_C = evaluate(h, dx)
        if current_C < best_eval:
            best_eval = current_C
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement > patience:
            # Try different configuration
            h = project(h + 0.01 * np.random.randn(n_points), dx)
            no_improvement = 0
    
    return (best_h, best_eval, n_points)
