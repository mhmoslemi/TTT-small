import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    try:

        if len(initial_h) != n_points:

        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    # Optimization parameters
    max_time = budget_s - 10  # Leave some time for final assessment
    start_time = time.time()
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Use a convex optimization approach with cutting planes
    # Define variables
    h_vars = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints for the initial guess
    constraints = [
        cp.sum(h_vars) == 1.0/dx,
        h_vars >= 0,
        h_vars <= 1
    ]
    
    # Build constraints for current h 
    current_C = evaluate(h, dx)
    best_C = current_C
    
    # Iterative cutting plane method
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < max_time:
        # Update h in numpy
        h = project(h, dx)
        
        # Get correlation values for full mode
        corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Append the constraint that current max correlation <= t
        # Reconstruct the shift (full mode has len(h)*2 - 1 correlations)
        k = max_k - (n_points - 1)
        if abs(k) < n_points:
            # We'll add a constraint for this k
            # For shift of k, S(k) = int h(x)(1-h(x+k)) dx
            # This is equivalent to: 1 - int h(x)h(x+k) dx  
            # Create the indices for the convolution
            shift = k
            if shift >= 0:
                # h(x) and 1-h(x+k) where x+k is valid
                left_start = shift
                right_end = n_points
                right_start = 0
                left_end = n_points - shift
            else:
                # h(x) and 1-h(x+k) where x+k is valid, so x >= -k
                left_start = 0
                right_start = -shift
                left_end = n_points
                right_end = n_points + shift
            
            if left_end > left_start and right_end > right_start:
                # Sum over overlap
                indices_left = np.arange(left_start, left_end)
                indices_right = np.arange(right_start, right_end)
                # This is a bit tricky, we want to compute: sum(h[i]*h[i-shift]*dx)
                h_vals = h[indices_left]
                h_shift_vals = h[indices_right] 
                integral = np.sum(h_vals * h_shift_vals) * dx
                # S(k) = 1 - integral
                S_k = 1.0 - integral
                constraints.append(S_k <= t)
            
            # Possibly also bound this constraint in terms of current bound
            # That would require some re-optimization logic
            
        # Re-optimize with new constraints
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_vals = h_vars.value
                if h_vals is not None and np.all(h_vals >= 0) and np.all(h_vals <= 1):
                    h = np.array(h_vals)
                    h = project(h, dx)
                    current_C = evaluate(h, dx)
                    if current_C < best_C:
                        best_h = h.copy()
                        best_C = current_C
        except:
            pass
        
        # Apply small random perturbations to escape local minima
        if iter_count % 10 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project(h, dx)
            
        iter_count += 1
        
    # Return best found solution
    return (best_h, best_C, n_points)
