import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a more sophisticated initial guess with multiple segments
        h = np.zeros(n_points)
        segments = 3
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)

    # PSO parameters
    n_particles = 50
    ω = 0.7
    φ₁ = 1.5
    φ₂ = 1.5
    max_velocity = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    # Initialize with some particles from the current solution
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = project(particle, dx)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        score = evaluate(particle, dx)
        p_best.append(particle.copy())
        p_best_scores.append(score)
    
    # Global best
    best_idx = np.argmin(p_best_scores)
    g_best = p_best[best_idx].copy()
    g_best_score = p_best_scores[best_idx]
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    # PSO iterations
    for iter in range(100000):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (ω * velocities[i] + 
                            φ₁ * r1 * (p_best[i] - particles[i]) + 
                            φ₂ * r2 * (g_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            score = evaluate(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = score
                
                # Update global best if necessary
                if score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = score
    
    # Final evaluation with a more precise function
    best_C = evaluate(g_best, dx)
    best_h = g_best.copy()
    
    return (best_h, best_C, n_points)
