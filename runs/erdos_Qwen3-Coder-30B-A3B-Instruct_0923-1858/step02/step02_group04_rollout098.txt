import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Initialize population
    try:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    except NameError:
        n_points = 50
        x = np.linspace(0, 2, n_points)
        initial_h = np.maximum(0, 1 - np.abs(x - 1))
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    population_size = 10
    population = [initial_h.copy() for _ in range(population_size)]
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / len(h_vec))
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def parametrize_feasible(h_vec):
        """Convert to feasible form through logit transformation"""
        # Apply sigmoid to ensure [0,1] bounds
        h_sigmoid = 1 / (1 + np.exp(-10 * (h_vec - 0.5)))
        # Normalize
        h_sigmoid = h_sigmoid * (n_points / 2) / np.sum(h_sigmoid)
        h_sigmoid = np.clip(h_sigmoid, 0, 1)
        return h_sigmoid
    
    # Multi-resolution approach
    resolutions = [50, 75, 100, 150, 200]
    current_res_idx = 0
    
    best_solution = None
    best_C5 = float('inf')
    patience = 0
    max_patience = 50
    
    iteration = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Initialize at current resolution if needed
        if current_res_idx >= len(resolutions):
            current_res_idx = len(resolutions) - 1
        n_points = resolutions[current_res_idx]
        
        # If we need to increase resolution, interpolate from previous solution
        if iteration % 10 == 0 and current_res_idx < len(resolutions) - 1:
            # Interpolate best solution from previous resolution
            if best_solution is not None:
                old_h = best_solution
                new_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(old_h)), old_h)
                population = [new_h.copy() for _ in range(population_size)]
                current_res_idx += 1
        
        # Update population
        current_C5 = [compute_C5(h) for h in population]
        indices = np.argsort(current_C5)
        
        # Keep best and mutate rest
        best_in_pop = population[indices[0]] if indices.size > 0 else None
        if best_in_pop is not None:
            if compute_C5(best_in_pop) < best_C5:
                best_C5 = compute_C5(best_in_pop)
                best_solution = best_in_pop.copy()
                patience = 0
            else:
                patience += 1
                
        # Create new population
        new_pop = [best_in_pop.copy() if best_in_pop is not None else population[0].copy()]
        
        # Mutation and crossover
        for i in range(population_size - 1):
            p1_idx, p2_idx = np.random.choice(population_size, 2, replace=False)
            parent1 = population[p1_idx]
            parent2 = population[p2_idx]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Add noise
            noise_magnitude = 0.05
            noise = np.random.normal(0, noise_magnitude, n_points)
            child = child + noise
            
            # Project to feasible set
            child = project_feasible(child)
            
            new_pop.append(child)
        
        population = new_pop
        
        # Gradient updates via simple line search
        for i, h in enumerate(population):
            # Estimate gradient via finite differences
            eps = 1e-5
            grad = np.zeros_like(h)
            for j in range(len(h)):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[j] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
            
            # Line search for step size
            def obj(alpha):
                h_new = h - alpha * grad
                h_new = project_feasible(h_new)
                return compute_C5(h_new)
            
            # Find optimal alpha using scalar minimization
            try:
                result = minimize_scalar(obj, bounds=(0, 1), method='bounded')
                step_size = result.x
            except:
                step_size = 0.01
            
            # Update with line search result
            new_h = h - step_size * grad
            new_h = project_feasible(new_h)
            population[i] = new_h
            
        iteration += 1
        
        # Early termination if no improvement
        if patience >= max_patience:
            break
    
    final_C5 = compute_C5(best_solution)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_solution.tolist(), best_C5, len(best_solution))
