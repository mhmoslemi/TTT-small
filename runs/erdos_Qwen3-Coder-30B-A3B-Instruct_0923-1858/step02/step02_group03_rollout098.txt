import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = initial_h / (np.sum(initial_h) * dx)
    
    start_time = time.time()
    
    # Optimization using cvxpy
    h = Variable(n_points)
    C5 = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) * dx == 1, C5 >= 0]
    
    # Compute correlation using a direct MIP approach
    # For each k, we want to minimize max_k integral h(x)(1-h(x+k))dx
    
    # Build the constraint list for each possible shift
    # We'll compute approximate C5 for all shifts k manually since we can't use
    # full convolution with cvxpy on final problem.
    
    # Use a sampling-based approach, might not be tight, but try a modified
    # gradient approach
    
    h_vals = initial_h.copy()
    best_C = evaluate_erdos_solution(h_vals)
    
    # Simple gradient descent approach with momentum and early stopping
    max_iter = 100000
    lr = 0.01
    momentum = 0.9
    v = np.zeros_like(h_vals)
    
    patience = 0
    max_patience = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate gradient
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        
        for j in range(n_points):
            h_plus = h_vals.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / (np.sum(h_plus) * dx)
            
            h_minus = h_vals.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / (np.sum(h_minus) * dx)
            
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        v = momentum * v - lr * grad
        h_vals = h_vals + v
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / (np.sum(h_vals) * dx)
        
        new_C = evaluate_erdos_solution(h_vals)
        if new_C < best_C:
            best_C = new_C
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                lr = max(1e-5, lr * 0.95)
                patience = 0
                
    return (h_vals, best_C, n_points)

def evaluate_erdos_solution(h):
    # This is the same evaluation as before
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))
