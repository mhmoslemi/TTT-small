import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution with reflection properties
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def extract_top_shifts(h_vec, num_shifts=10):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_indices = np.argsort(-corr)[:num_shifts]
        shifts = max_indices - (n_points - 1)
        return shifts

    def solve_qp_projection(h_vec, shifts):
        # Formulate a QP to find the nearest feasible point
        h_var = cp.Variable(n_points)
        # Objective: Euclidean distance to h_vec
        obj = cp.Minimize(cp.sum_squares(h_var - h_vec))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_var.value
        except:
            pass
        # Fallback to projection
        return project_feasible(h_vec)

    # Gradient estimates for alternating optimization
    def estimate_gradient(h_vec, k):
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps

            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)

            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")

            grad[i] = (corr_plus[k + n_points - 1] - corr_minus[k + n_points - 1]) / (2 * eps)
        return grad

    # Step size selection methods
    def adaptive_step_size(h_vec, delta_h, current_shifts, C5_val):
        step_sizes = np.linspace(0.001, 0.05, 10)
        best_step = step_sizes[-1]
        best_improvement = 0
        for step in step_sizes:
            h_trial = h_vec - step * delta_h
            h_trial = project_feasible(h_trial)
            trial_C5 = compute_C5(h_trial)
            improvement = C5_val - trial_C5
            if improvement > best_improvement:
                best_improvement = improvement
                best_step = step
        return best_step if best_improvement > 0 else 0.01

    def optimize_shift(shift_k, h_vec, max_iter=200):
        h_opt = h_vec.copy()
        C5_val = compute_C5(h_opt)
        # Simple gradient descent for this specific shift
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
            grad = estimate_gradient(h_opt, shift_k)
            step = adaptive_step_size(h_opt, grad, [shift_k], C5_val)
            h_new = h_opt - step * grad
            h_new = project_feasible(h_new)
            new_C5 = compute_C5(h_new)
            if new_C5 < C5_val:
                h_opt = h_new
                C5_val = new_C5
            else:
                # Flatten step size
                step *= 0.9
                if step < 1e-6:
                    break
        return h_opt, C5_val

    # Deeper exploration via candidate point generation
    def local_search_refinement():
        # Use the best known solution to perturb
        h_sol = best_h.copy()
        # Refine by optimizing top contributing shifts
        top_shifts = extract_top_shifts(h_sol, num_shifts=8)
        for k in top_shifts:
            if elapsed() > budget_s * 0.95:
                break
            h_sol, C5_new = optimize_shift(k, h_sol)
            if C5_new < compute_C5(best_h):
                best_h[:] = h_sol[:]
                return True
        return False

    # Main loop
    iter_count = 0
    last_flush = 0
    max_iter = 3000  # Prevent too many loops
    
    # Progress checker
    while iter_count < max_iter and elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h[:] = h[:]
            last_flush = iter_count
        
        # Periodic updates
        if iter_count % 20 == 0:
            # Local refinement
            local_search_refinement()
            
        # Refine only top shifts every 30 iterations
        if iter_count % 30 == 0:
            top_shifts = extract_top_shifts(h, 5)
            for k in top_shifts:
                h, C5_new = optimize_shift(k, h)
                if C5_new < compute_C5(h):
                    h = h.copy()  # Ensure value isn't updated in-place incorrectly
        
        # Update with local search
        if iter_count % 50 == 0:
            h = solve_qp_projection(h, extract_top_shifts(h, 10))
        
        # Random perturbation occasionally
        if iter_count % 100 == 0 and iter_count > 0:
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.normal(0, 0.03)
            h = project_feasible(h)

        iter_count += 1

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, len(best_h))
