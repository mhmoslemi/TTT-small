import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values at points t"""
    h = a[0] * np.ones_like(t)
    for i in range(1, len(a)):
        h += a[i] * np.cos(i * np.pi * t) + b[i] * np.sin(i * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Clip h to [0,1] and normalize to integral = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # PSO parameters
    N = 10  # number of Fourier coefficients (a0, a1, ..., aN, b1, ..., bN)
    P = 50  # number of particles
    max_iter = 1000
    omega = 0.7
    phi1 = phi2 = 1.5
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N + 1))  # a0, a1..aN, b1..bN
    velocities = np.zeros_like(particles)
    personal_best = particles.copy()
    personal_best_C = np.full(P, np.inf)
    
    # Global best
    global_best_idx = 0
    global_best_C = np.inf
    
    # Time tracking
    start_time = time.time()
    max_time = budget_s - 5  # leave a bit of wiggle room
    
    # Try to load initial if available
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)
        initial_h = clip_and_normalize(initial_h, dx)
    
    # Initialize with initial point
    t = np.linspace(0, 2, n_points)
    # Convert initial_h to Fourier coefficients (approximate)
    initial_coeffs = np.zeros(2*N + 1)  # a0, a1,...,aN, b1,...,bN
    # Try fitting just a0, a1, b1 for smoother initial
    fit_a0 = np.mean(initial_h)
    fit_a1 = 2 * np.mean(initial_h * np.cos(np.pi * t)) 
    fit_b1 = 2 * np.mean(initial_h * np.sin(np.pi * t))
    initial_coeffs[0] = fit_a0
    initial_coeffs[1] = fit_a1
    initial_coeffs[N+1] = fit_b1
    
    # Use this as first particle
    particles[0] = initial_coeffs.copy()
    
    # Fill rest with random
    for i in range(1, P):
        particles[i] = np.random.uniform(-0.5, 0.5, 2*N + 1)
        
    # Start PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(P):
            # Update h from coefficients (for this particle)
            a = particles[i, 0:N+1]   # a0, a1, ..., aN
            b = particles[i, N+1:]    # b1, ..., bN
            
            h = fourier_to_h(a, b, t)
            h = clip_and_normalize(h, dx)
            
            # Evaluate C5
            current_C = evaluate(h, dx)
            
            if current_C < personal_best_C[i]:
                personal_best[i] = particles[i].copy()
                personal_best_C[i] = current_C
                
            if current_C < global_best_C:
                global_best_C = current_C
                global_best_idx = i
                
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] 
                            + phi1 * r1 * (personal_best[i] - particles[i])
                            + phi2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
    # Extract the best solution
    best_coeffs = personal_best[global_best_idx]
    a = best_coeffs[0:N+1]
    b = best_coeffs[N+1:]
    best_h = fourier_to_h(a, b, t)
    best_h = clip_and_normalize(best_h, dx)
    
    return (best_h, global_best_C, n_points)
