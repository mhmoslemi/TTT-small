import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    t = np.linspace(0, 2, n_points)
    
    # Initialize from parent solution if available
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Use a refined starting guess
        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        initial_h = project(initial_h, dx)
    
    # PSO parameters
    n_particles = 50
    c1, c2 = 2.0, 2.0
    w = 0.7
    max_vel = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best_h = []
    p_best_C = []
    
    for i in range(n_particles):
        if 'initial_h' in locals():
            h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        else:
            h = 0.5 + 0.3 * np.sin(np.pi * t) + np.random.normal(0, 0.05, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best_h.append(h.copy())
        p_best_C.append(evaluate(h, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_C)
    g_best_h = p_best_h[g_best_idx].copy()
    g_best_C = p_best_C[g_best_idx]
    
    # PSO exploration phase
    t0 = time.time()
    max_time_explore = budget_s * 0.7
    
    for iter_i in range(100000):
        if time.time() - t0 > max_time_explore:
            break
            
        for p in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[p] = (w * velocities[p] + 
                           c1 * r1 * (p_best_h[p] - particles[p]) + 
                           c2 * r2 * (g_best_h - particles[p]))
            # Clip velocity
            velocities[p] = np.clip(velocities[p], -max_vel, max_vel)
            
            # Update position
            particles[p] += velocities[p]
            # Clip to [0,1]
            particles[p] = np.clip(particles[p], 0, 1)
            # Rescale to meet constraint
            particles[p] = project(particles[p], dx)
            
            # Evaluate
            current_C = evaluate(particles[p], dx)
            
            # Update personal best
            if current_C < p_best_C[p]:
                p_best_C[p] = current_C
                p_best_h[p] = particles[p].copy()
            
            # Update global best
            if current_C < g_best_C:
                g_best_C = current_C
                g_best_h = particles[p].copy()
    
    # Gradient descent exploitation phase
    current_h = g_best_h.copy()
    lr = 0.005
    max_time_exploit = budget_s - (time.time() - t0)
    iters = 0
    
    best_h = current_h.copy()
    best_C = g_best_C
    
    while time.time() - t0 < max_time_exploit and iters < 20000:
        # Gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        current_h = current_h - lr * grad
        current_h = project(current_h, dx)
        
        # Check for improvement
        current_C = evaluate(current_h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
            
        # Adaptive learning rate decay
        iters += 1
        if iters % 1000 == 0:
            lr *= 0.98
    
    return (best_h, best_C, n_points)
