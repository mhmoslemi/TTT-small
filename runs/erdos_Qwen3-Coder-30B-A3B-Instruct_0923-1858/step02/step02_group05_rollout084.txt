import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

        h = project(h, dx)
    except NameError:
        # Create initial point based on sine wave
        t = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * t)
        h = project(h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_c5 = []
    
    for _ in range(n_particles):
        # Start with slightly perturbed version of h
        particle = h + np.random.uniform(-0.05, 0.05, n_points)
        particle = project(particle, dx)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_c5.append(evaluate(particle, dx))
    
    # Find global best
    g_best_idx = np.argmin(p_best_c5)
    g_best = p_best[g_best_idx].copy()
    g_best_c5 = p_best_c5[g_best_idx]
    
    t0 = time.time()
    iter_count = 0
    
    while iter_count < max_iter and time.time() - t0 < budget_s - 2:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_c5 = evaluate(particles[i], dx)
            
            # Update p_best
            if current_c5 < p_best_c5[i]:
                p_best[i] = particles[i].copy()
                p_best_c5[i] = current_c5
            
            # Update g_best
            if current_c5 < g_best_c5:
                g_best = particles[i].copy()
                g_best_c5 = current_c5
                
        iter_count += 1
    
    return (g_best, g_best_c5, n_points)
