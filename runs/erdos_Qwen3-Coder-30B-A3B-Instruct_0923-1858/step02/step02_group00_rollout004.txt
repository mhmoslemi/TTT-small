import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, k_target=None):
        h_vec = np.array(h_vec)
        if k_target is None:
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            k_target = max_k_idx - (len(h_vec) - 1)
        
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        h_plus = h_vec + eps
        h_minus = h_vec - eps
        h_plus = project_feasible(h_plus)
        h_minus = project_feasible(h_minus)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
        
        grad = (corr_plus[k_target + n_points - 1] - corr_minus[k_target + n_points - 1]) / (2 * eps)
        return grad

    def trust_region_update(h_vec, eta=0.1, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    def get_worst_shift(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        return max_k, corr[max_k_idx]

    # Convex relaxation approach
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        # Create the maximum overlap constraint in a convex way
        # We'll use a finite number of shifts to make it tractable
        # This is a scalable implementation
        obj = cp.Minimize(cp.sum_squares(h_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'optimal':
                return np.maximum(0, np.minimum(1, h_var.value))
        except:
            pass
        return h.copy()

    # Main optimization loop with alternating updates
    eta = 0.01
    patience = 0
    max_patience = 50
    iter_count = 0
    last_reset = 0
    reset_interval = 100

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                h = np.random.rand(n_points) * 0.5
                h = project_feasible(h)
                patience = 0

        # Get worst shift and update with gradient
        worst_k, _ = get_worst_shift(h)
        h = trust_region_update(h, eta)
        
        # Periodic learning rate decay and reset
        iter_count += 1
        if iter_count % reset_interval == 0:
            eta *= 0.95
            # Do periodic refresh with convex relaxation
            if elapsed() < budget_s * 0.7:
                h = solve_convex_relaxation()
        elif iter_count % 10 == 0:
            eta *= 0.98

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
