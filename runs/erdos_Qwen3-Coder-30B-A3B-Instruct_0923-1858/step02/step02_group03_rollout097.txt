import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_time = budget_s - 10  # Leave some time for final assessment
    start_time = time.time()
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    # Try MIP approach for better optimization
    try:
        # Quick MIP approach on a smaller scale
        h_vars = cp.Variable(n_points, lower_bound=0, upper_bound=1)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_vars) * dx == 1, C5 >= 0]
        
        # For each k, add the constraint that the correlation is bounded by C5
        # We'll do this manually for efficiency with smaller problems
        for k in range(1, n_points):
            # Sum over i: h[i] * (1 - h[i+k]) = sum(h[i]) - sum(h[i]*h[i+k])
            if k < n_points:
                expr = 0
                for i in range(n_points - k):
                    expr += h_vars[i] * (1 - h_vars[i + k])
                constraints.append(expr <= C5)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Run for a bit
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_vars.value
            if h_sol is not None:
                # Normalize to exact constraint
                h_sol = project(h_sol, dx)
                C_sol = evaluate(h_sol, dx)
                if C_sol < best_C:
                    best_h = h_sol
                    best_C = C_sol
    except Exception:
        pass
    
    # Refine with gradient-based optimization if time allows
    if time.time() - start_time < max_time:
        h = best_h.copy()
        lr = 0.02
        patience = 0
        max_patience = 100
        
        for i in range(50000):
            if time.time() - start_time > max_time:
                break
                
            # Compute gradient estimation for each variable
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = project(h_minus, dx)
                
                corr_plus = correlate(h_plus, 1-h_plus, mode='full')
                corr_minus = correlate(h_minus, 1-h_minus, mode='full')
                
                grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
            # Update step
            h_new = h - lr * grad
            h_new = project(h_new, dx)
            
            # Test the new point
            current_C = evaluate(h_new, dx)
            if current_C < best_C:
                h = h_new
                best_C = current_C
                best_h = h.copy()
                patience = 0
                lr = min(0.05, lr * 1.02)  # Adaptively increase learning rate
            else:
                h = h_new
                patience += 1
                if patience > max_patience:
                    lr = max(1e-5, lr * 0.95)  # Reduce learning rate
                    patience = 0
            
            # Occasionally add noise to escape local minima
            if i % 50 == 0 and np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.005, len(h))
                h = project(h + noise, dx)
    
    return (best_h, best_C, n_points)
