    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Exponential smoothing and barrier penality approach
    beta = 1000.0  # Temperature parameter
    gamma = 1e-4   # Barrier strength
    eta = 1e-5     # Learning rate
    patience = 0
    max_patience = 50
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0
        
        # Compute cross-correlations
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        
        # Apply exponential penalty
        if max_corr > 0:
            corr_scaled = corr / max_corr
        else:
            corr_scaled = corr
            
        # Exponential penalty
        exp_penalty = np.sum(np.exp(-beta * corr_scaled))
        
        # Barrier term

        # Compute gradient estimate
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)

            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")

        # Update
        h = h - eta * grad
        h = project_feasible(h)

            beta *= 0.99
            gamma *= 1.1

            eta *= 0.5
            
        iter_count += 1
        
        # Occasionally force projection
        if iter_count % 50 == 0:
            h = project_feasible(h)

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
