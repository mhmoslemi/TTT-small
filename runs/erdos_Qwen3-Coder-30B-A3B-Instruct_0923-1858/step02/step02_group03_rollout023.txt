import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:

    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = initial_h / (np.sum(initial_h) * dx)
    
    start_time = time.time()
    max_time = budget_s - 10
    
    # Use a hybrid approach: first try convex optimization
    best_C = float('inf')
    best_h = initial_h.copy()
    
    # Try to find a better solution with CVXPY
    try:
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) * dx == 1,
        ]
        
        # Create McCormick envelopes for bilinear terms
        # We need to compute max_k int h(x)(1 - h(x+k)) dx
        # This can be rewritten as sum over all i,j of h[i]*h[j]*dx for j=i+k
        # But we don't want to manually set all these constraints
        
        # Instead, use a smarter approach:
        # Minimize C5 subject to: integral <= C5 for all k
        # So we will compute all the overlap integrals
        
        # For each k, we want max of correlate(h, 1-h) over full convolution
        # This is computationally expensive...
        
        # Use an iterative improvement with projected gradient
        h = initial_h.copy()
        for iteration in range(5000):
            if time.time() - start_time > max_time:
                break
                
            # Gradient step using finite differences
            eps = 1e-6
            grad = np.zeros_like(h)
            
            # Estimate the gradient using finite differences
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / (np.sum(h_plus) * dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus / (np.sum(h_minus) * dx)
                
                corr_plus = np.max(correlate(h_plus, 1 - h_plus, mode='full'))
                corr_minus = np.max(correlate(h_minus, 1 - h_minus, mode='full'))
                
                grad[i] = (corr_plus - corr_minus) / (2 * eps)
            
            # Update rule with adaptive learning rate
            lr = 0.01 / (1 + iteration / 1000)
            h_new = h - lr * grad
            
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / (np.sum(h_new) * dx)
            
            # Evaluate new solution
            current_C = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
            
            if current_C < best_C:
                best_C = current_C
                best_h = h_new.copy()
                if best_C < 0.3807:  # Early stopping condition
                    break
            
            h = h_new
            
            # Occasionally add noise to escape local minima
            if iteration % 100 == 0:
                noise = np.random.normal(0, 0.005, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h / (np.sum(h) * dx)
                
    except Exception as e:
        pass
    
    # Final check to make sure constraints are satisfied
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / (np.sum(best_h) * dx)
    best_C = np.max(correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return (best_h, best_C, n_points)
