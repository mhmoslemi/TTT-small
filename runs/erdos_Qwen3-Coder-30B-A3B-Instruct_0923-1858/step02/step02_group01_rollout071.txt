import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def perturb(h, dx, magnitude=0.01):
    h_candidate = h + np.random.uniform(-magnitude, magnitude, size=h.shape)
    return project(h_candidate, dx)

def simulated_annealing(h, dx, budget_s):
    T_initial = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    T = T_initial
    best_h = h.copy()
    best_C = evaluate(h, dx)
    start_time = time.time()
    
    iterations = 0
    stagnation_count = 0
    max_stagnation = 1000
    
    while T > T_min and time.time() - start_time < budget_s - 10:
        h_candidate = perturb(h, dx)
        current_C = evaluate(h_candidate, dx)
        
        delta = current_C - best_C
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h = h_candidate
            if current_C < best_C:
                best_h = h_candidate.copy()
                best_C = current_C
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        if stagnation_count > max_stagnation:
            T = max(T_min, T * 0.9)
            stagnation_count = 0
            
        T = max(T_min, T * cooling_rate)
        iterations += 1
        
        # Occasionally increase perturbation if stuck
        if iterations % 500 == 0 and T < 0.1:
            magnitude = min(0.05, magnitude * 1.1)
            
    return best_h, best_C
