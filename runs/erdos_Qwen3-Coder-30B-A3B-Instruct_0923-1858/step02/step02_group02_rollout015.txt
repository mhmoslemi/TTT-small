import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    # Simulated Annealing parameters
    T0 = 100.0
    alpha = 0.999
    max_iter = 10000
    T = T0
    current_h = h.copy()
    current_C = evaluate(current_h, dx)
    best_h = current_h.copy()
    best_C = current_C
    
    t0 = time.time()
    max_time = budget_s - 10
    
    for i in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        # Create a proposal by perturbing some points
        proposal = current_h.copy()
        # Perturb about 5% of points
        n_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            proposal[idx] += np.random.normal(0, 0.1)
        
        # Project to feasible region
        proposal = project(proposal, dx)
        
        # Evaluate proposal
        proposal_C = evaluate(proposal, dx)
        
        # Accept or reject
        delta = proposal_C - current_C
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = proposal
            current_C = proposal_C
            if current_C < best_C:
                best_C = current_C
                best_h = current_h.copy()
        
        # Update temperature
        T = T * alpha
    
    # Final projection of best solution
    best_h = project(best_h, dx)
    final_C = evaluate(best_h, dx)
    
    return (best_h, final_C, n_points)
