        return h_vec

    def optimize_step(h_vec, learning_rate=0.01, max_iter=20):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            # Estimate gradient by finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-4
            baseline = compute_C5(h_new)

                h_plus = h_new.copy()
                h_plus[i] += eps
