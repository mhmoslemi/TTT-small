import time

        h = h * (n_points / 2) / np.sum(h)

    def solve_convex_optimization(h_init):
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            t >= 0
        ]
        
        # For all possible shifts, add constraints
        for k in range(n_points):
            # The overlap integral for shift k
            # We consider the correlation, but note that we compute it differently
            # We approximate ∫h(x)(1-h(x+k))dx using discrete sum
            if k == 0:
                overlap_expr = cp.sum(cp.multiply(h_var, 1 - h_var))
            else:
                # For k-th shift, we compute the overlap
                # h[i]* (1 - h[i+k]) when i+k < n_points
                # h[i]*(1-h[i+k]) when i + k < n_points; else 0
                # This requires a bit more care in indexing
                # Let's reorder the indices: h[i] * (1 - h[i+k])
                # So for k shift, we compare h[i] with h[i+k] (when i+k < n_points)
                expr = 0
                for i in range(n_points - k):
                    expr += h_var[i] * (1 - h_var[i + k])
                # Add constraint on this expression
                constraints.append(expr <= t)
        
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_var.value
        except:
            pass
        
        # Fallback to initial
        return h_init.copy()

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # First solve convex optimization
    h_opt = solve_convex_optimization(h)

    try:

            best_C5 = C5_opt

        # Compute gradient approximation using finite differences
        h_vec = np.array(h_vec)
        C5_val = compute_C5(h_vec)

            h_minus[i] -= eps

            C5_minus = compute_C5(h_minus)

        h_new = h_vec - eta * grad
        h_new = project_feasible(h_new)
        return h_new, grad

        else:
            patience += 1
            if patience >= max_patience:
                # Random perturbation if too many steps without improvement

                h_current[idx] = np.clip(h_current[idx] + np.random.normal(0, 0.02), 0, 1)
                h_current = project_feasible(h_current)
                patience = 0
                
        eta = max(1e-5, eta * 0.995)
        iter_count += 1
    
    # Final cleanup and evaluation

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
