import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        segments = 6
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            # Vary heights in a non-uniform way to have local variations
            base_height = 0.4 + 0.2 * np.sin(i * np.pi / 3)
            h[start:end] = base_height + 0.1 * np.random.rand(end - start)
        
        h = project(h, dx)

    population_size = 50
    population = [project(np.random.rand(n_points) * 0.5, dx) for _ in range(population_size)]

    # Execution parameters
    max_time = budget_s - 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    generations = 0
    
    t0 = time.time()
    
    while time.time() - t0 < max_time:
        # Evaluate population
        fitness = []
        for individual in population:
            C = evaluate(individual, dx)
            fitness.append(1.0 / (1e-10 + C))  # Avoid division by zero
            
        # Selection with tournament
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]
            
            new_population.append(population[winner_index].copy())
            
        # Crossover and mutation
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate and i + 1 < population_size:
                parent1 = new_population[i]
                parent2 = new_population[i + 1]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutate children
                for j in range(n_points):

        if generations % 100 == 0:
            # Adapt mutation rate
            mutation_rate = max(0.01, mutation_rate * 0.99)
