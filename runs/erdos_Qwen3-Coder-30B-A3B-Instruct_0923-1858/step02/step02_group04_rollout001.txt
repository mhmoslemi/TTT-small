import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a triangular function
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    # Create a convex optimization problem to minimize the maximum overlap
    dx = 2.0 / n_points
    h_var = cp.Variable(n_points)
    
    # Objective: minimize the maximum overlap
    t = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, t >= 0]
    
    # For each shift, add the constraint on overlap
    max_k = n_points - 1
    for k in range(max_k + 1):
        # Compute overlap for shift k
        if k == 0:
            overlap = cp.sum(cp.multiply(h_var, 1 - h_var))
        else:
            # We need to properly handle the convolution-like operation
            # h[i] * (1 - h[i+k]) for valid indices
            overlap = 0
            for i in range(n_points - k):
                overlap += h_var[i] * (1 - h_var[i + k])
            overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        constraints.append(overlap <= t)
    
    # Solve the convex optimization problem
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        # Use MOSEK solver for speed and reliability
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_cvx = h_var.value
        else:
            # Fallback to original h if optimization fails
            h_cvx = h.copy()
    except:
        # If CVXPY fails, use the starting approximation
        h_cvx = h.copy()
    
    # Clip and normalize the solution
    h_cvx = np.clip(h_cvx, 0, 1)
    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
    
    # Compute final C5
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    best_h = h_cvx.copy()
    best_C5 = compute_C5(best_h)
    
    # Early return if we are close enough to target
    if best_C5 <= 0.3807:
        return (best_h.tolist(), best_C5, n_points)
    
    # Local refining using gradient-based method
    patience = 0
    max_patience = 100
    eta = 0.01
    iter_count = 0
    time_limit = budget_s * 0.98
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        current_C5 = compute_C5(h_cvx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_cvx.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb randomly
                idx = np.random.choice(n_points)
                h_cvx[idx] = np.clip(h_cvx[idx] + np.random.normal(0, 0.03), 0, 1)
                h_cvx = project_feasible(h_cvx)
                patience = 0
        
        # Gradient-based update
        try:
            grad = get_gradient(h_cvx)
            h_cvx = h_cvx - eta * grad
            h_cvx = project_feasible(h_cvx)
            eta = max(0.001, eta * 0.995)  # decay
        except:
            # Fallback if gradient fails
            idx = np.random.choice(n_points, size=5, replace=False)
            h_cvx[idx] = np.clip(h_cvx[idx] + np.random.normal(0, 0.02, len(idx)), 0, 1)
            h_cvx = project_feasible(h_cvx)
        
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
