import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    T0 = 100.0
    alpha = 0.999
    max_iter = 10000
    T = T0
    t0 = time.time()
    
    # Define perturbation size and fraction
    perturb_fraction = 0.05
    perturb_sigma = 0.1
    num_perturb = max(1, int(perturb_fraction * n_points))
    
    for i in range(max_iter):
        if time.time() - t0 > budget_s - 10:

        # Make a perturbation
        h_proposal = h.copy()
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            h_proposal[idx] += np.random.normal(0, perturb_sigma)
        
        # Project to feasible region
        h_proposal = project(h_proposal, dx)
        
        # Evaluate
        E_proposal = evaluate(h_proposal, dx)
        E_current = evaluate(h, dx)
        
        delta_E = E_proposal - E_current
        
        # Accept or reject
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_proposal
            if E_proposal < best_C:
                best_C = E_proposal
                best_h = h.copy()
        
        # Cool down
        T = T * alpha
    
    return (best_h, best_C, n_points)
