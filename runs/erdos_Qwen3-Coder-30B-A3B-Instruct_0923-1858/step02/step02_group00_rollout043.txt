import numpy as np
import time
from scipy.signal import correlate
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Initialize swarm
    swarm_size = 50
    particles = []
    velocities = []
    
    for _ in range(swarm_size):
        # Create random particles with correct sum constraint
        h_trial = np.random.rand(n_points)
        h_trial = project_feasible(h_trial)
        particles.append(h_trial.copy())
        velocities.append(np.random.randn(n_points) * 0.01)
    
    # Keep track of personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_C5 = [compute_C5(p) for p in particles]
    
    # Keep track of global best
    global_best_idx = np.argmin(personal_best_C5)
    global_best = personal_best[global_best_idx].copy()
    global_best_C5 = personal_best_C5[global_best_idx]
    
    # PSO parameters
    c1 = 2.0
    c2 = 2.0
    inertia = 0.9
    max_iter = 100
    
    # Phase 1: PSO global search
    for iter_count in range(max_iter):
        if elapsed() > budget_s * 0.8:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_feasible(particles[i])
            
            # Evaluate and update personal best
            current_C5 = compute_C5(particles[i])
            if current_C5 < personal_best_C5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_C5[i] = current_C5
                
                # Update global best if needed
                if current_C5 < global_best_C5:
                    global_best = particles[i].copy()
                    global_best_C5 = current_C5
        
        # Decrease inertia
        inertia = 0.9 - (0.5 * iter_count / max_iter)
    
    # Update best solution found so far
    if global_best_C5 < best_C5:
        best_C5 = global_best_C5
        best_h = global_best.copy()
    
    # Phase 2: Local refinement with gradient methods
    # Start from the best solution from PSO
    
    def get_gradient(h_vec, k_opt):
        # Calculate the gradient at the optimal k
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        
        # We are computing the gradient of max_k (integrate h(x)(1-h(x+k))dx)
        # With simple finite differences
        
        eps = 1e-5
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[k_opt + len(h_vec) - 1] - 
                      corr_minus[k_opt + len(h_vec) - 1]) / (2 * eps)
        
        return grad

    # Use trust region method for fine-tuning  
    def trust_region_update(h_vec, max_iter=50, eta=0.1):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Get current maximum correlation index
            corr = correlate(h_new, 1 - h_new, mode="full")
            k_idx = np.argmax(corr)
            k_opt = k_idx - (len(h_new) - 1)
            
            grad = get_gradient(h_new, k_opt)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
            # Decrease step size
            eta *= 0.95
            
        return h_new

    # Refinement loop
    refinement_budget = budget_s * 0.2
    max_refine_iter = 500
    
    for refine_iter in range(max_refine_iter):
        if elapsed() > budget_s:
            break
            
        # Refine the current best
        h_new = trust_region_update(best_h)
        new_C5 = compute_C5(h_new)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
    
    # Final evaluation   
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
