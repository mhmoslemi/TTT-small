import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_points = 45
    dx = 2.0 / n_points

        if len(initial_h) != n_points:

        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        initial_h = project(initial_h, dx)

    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    velocity_max = 0.2

    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for i in range(n_particles):
        # Initialize particle with slight perturbations
        h = initial_h.copy()
        if i > 0:  # Only perturb non-initial particles
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            h = h + perturb
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())
        p_best_values.append(evaluate(h, dx))

    # Global best initialization
    best_idx = np.argmin(p_best_values)
    g_best = p_best[best_idx].copy()
    g_best_value = p_best_values[best_idx]

    t0 = time.time()
    for iteration in range(max_iter):
        if time.time() - t0 > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project(particles[i], dx)
            
            # Evaluate current particle
            current_value = evaluate(particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value
                
                # Update global best if needed
                if current_value < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_value

    # Return best solution found
    final_h = g_best
    final_c5 = g_best_value
    return (final_h, final_c5, n_points)
