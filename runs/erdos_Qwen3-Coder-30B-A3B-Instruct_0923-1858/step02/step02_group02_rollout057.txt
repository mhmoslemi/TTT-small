import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_random_individual(n_points, segments=10):
    """Create a random piecewise constant individual."""
    h = np.zeros(n_points)
    segment_size = n_points // segments
    for i in range(segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < segments - 1 else n_points
        h[start:end] = np.random.uniform(0, 1, end - start)
    return h

def mutate(h, mutation_rate=0.1, sigma=0.1):
    """Mutate a chromosome by adjusting segment heights."""
    h = h.copy()
    segments = 10
    segment_size = len(h) // segments
    for i in range(segments):
        if np.random.rand() < mutation_rate:
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else len(h)
            adjustment = np.random.normal(0, sigma, end - start)
            h[start:end] += adjustment
    return h

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform single-point crossover."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    point = np.random.randint(1, n_points-1)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmax([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

    except NameError:

    # Fill population to required size

        population.append(h)
    
    # Main optimization loop
    t0 = time.time()
    best_C = float('inf')
    best_h = None
    
    # Precompute some constants for efficiency
    generation = 0
    not_improved = 0
    max_not_improved = 500
    
    while generation < generations and time.time() - t0 < budget_s - 10:
        # Evaluate fitness for all individuals
        fitnesses = []
        for i, h in enumerate(population):
            C = evaluate(h, dx)
            fitnesses.append(1.0 / (C + 1e-12))  # Avoid division by zero

                best_C = C

            elif i == 0:  # Only check first individual for this iteration

            child1 = mutate(child1, mutation_rate)

            child1 = project(child1, dx)
