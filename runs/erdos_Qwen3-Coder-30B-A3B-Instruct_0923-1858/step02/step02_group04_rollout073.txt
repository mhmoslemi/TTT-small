import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.95

        n_points = 50

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def parametrize_feasible(h_vec):
        # Project onto feasible region while maintaining structure
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec, k=None):
        eps = 1e-6
        h_vec = np.array(h_vec)
        if k is None:
            # compute C5 first
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_vec) - 1)
        else:
            max_k = k
            
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = parametrize_feasible(h_plus)
            h_minus = parametrize_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k + len(h_vec) - 1] - corr_minus[max_k + len(h_vec) - 1]) / (2 * eps)
        
        return grad
    
    # Multi-resolution search
    resolutions = [50, 75, 100, 150, 200]
    current_res = 0
    
    # Population-based search
    population_size = 10
    population = [h.copy()]

        pert_h = h.copy() + np.random.normal(0, 0.02, len(h))
        population.append(parametrize_feasible(pert_h))

        fitnesses = []

            fitnesses.append(f)

        best_idx = np.argmin(fitnesses)

                last_improvement = iter_count

                idx = np.random.choice(n_points, size=min(3, n_points//4), replace=False)

                h = parametrize_feasible(h)
                patience = 0

        new_population = []
        for i in range(population_size):

            parents = np.random.choice(len(population), size=3, replace=False)

            alpha = np.random.rand()

            # Mutation
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.02, len(child))
                child = child + noise

            new_population.append(child)

        # Gradient update for best candidate
        try:
            grad = get_gradient(population[best_idx])
            current_h = population[best_idx]

            # Line search
            def line_search_step(step_size):
                trial_h = current_h - step_size * grad
                trial_h = parametrize_feasible(trial_h)
                return compute_C5(trial_h)

            try:

                alpha_opt = eta

            population[best_idx] = parametrize_feasible(population[best_idx])

        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
