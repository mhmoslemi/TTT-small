import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # PSO implementation
    def pso_optimize():
        # PSO parameters
        n_particles = 50
        max_iter = 100
        inertia_w = 0.9
        cognitive_c = 2.0
        social_c = 2.0
        mutation_rate = 0.05

        # Initialize particles
        particles = []
        pbest = []
        pbest_vals = []
        for _ in range(n_particles):
            # Create random feasible particle
            particle = np.random.rand(n_points)
            particle = project_feasible(particle)
            particles.append(particle)
            pbest.append(particle.copy())
            pbest_vals.append(compute_C5(particle))
        
        # Initialize global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        # Velocity initialization
        velocities = [np.zeros(n_points) for _ in range(n_particles)]
        
        for iter_num in range(max_iter):
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia_w * velocities[i] + 
                               cognitive_c * r1 * (pbest[i] - particles[i]) + 
                               social_c * r2 * (gbest - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply feasibility projection and mutation
                particles[i] = project_feasible(particles[i])
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(0, n_points)
                    particles[i][idx] = np.clip(particles[i][idx] + np.random.normal(0, 0.05), 0, 1)
                    particles[i] = project_feasible(particles[i])
                
                # Evaluate fitness
                new_val = compute_C5(particles[i])
                
                # Update personal best
                if new_val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = new_val
                    
                    # Update global best
                    if new_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = new_val
            
            # Decay inertia weight
            inertia_w = 0.9 - (0.5 * iter_num / max_iter)
            
            if elapsed() > budget_s * 0.9:
                break
                
        return gbest, gbest_val
    
    # Gradient descent refinement
    def refine_with_gradient(h_vec, max_iter=50):
        h_refined = h_vec.copy()
        
        for _ in range(max_iter):
            if elapsed() > budget_s * 0.9:
                break
                
            # Compute correlation
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_k_idx = np.argmax(corr)
            
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros_like(h_refined)
            h_perturbed = h_refined.copy()

                h_plus = h_refined.copy()
                h_minus = h_refined.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Gradient step
            step_size = 0.01
            h_refined = h_refined - step_size * grad
            h_refined = project_feasible(h_refined)
            
        return h_refined
    
    # Main optimization loop with alternating PSO and refinement
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        # PSO step
        pso_h, pso_C5 = pso_optimize()
        
        # Refinement step
        refined_h = refine_with_gradient(pso_h)
        refined_C5 = compute_C5(refined_h)
        
        # Update best
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h.copy()
        
        iter_count += 1
        if iter_count > 100:
            break
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
