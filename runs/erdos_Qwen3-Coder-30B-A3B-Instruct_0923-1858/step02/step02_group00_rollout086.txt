import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):

    except NameError:
        # Create a triangular initial solution
        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_subproblem_for_shift(h_vec, shift_k):
        # Solve a subproblem to adjust h_vec for shift_k
        # Minimize <h', (1-h)> over all valid shifts
        # Make change Δ = h'-h

        # Extract correlation indices
        corr_len = 2 * n_points - 1
        corr_idx = shift_k + (n_points - 1)
        all_shifts = np.arange(-n_points+1, n_points)
        
        # Create a quadratic model
        def objective(delta):
            # This is squared delta just for regularization
            return np.sum(delta**2)
        
        def constraint(delta):
            # Shift k correlation: sum over i, 0 <= i < n_points, 0 <= i+k < n_points
            if shift_k >= 0:
                start_idx = shift_k
                end_idx = n_points
                h_shift = h_vec[start_idx:end_idx]
                h_base = h_vec[:end_idx-shift_k]
            else:
                start_idx = 0
                end_idx = n_points + shift_k
                h_shift = h_vec[-shift_k:]
                h_base = h_vec[:n_points+shift_k]
            
            if len(h_base) == len(h_shift):
                corr = np.sum(h_base * (1 - h_shift))
                return corr
            else:
                return 0

        # For this approach, we'll make a simple iterative improvement:
        # We'll do gradient-based approach to adjust h such that the kth shift correlation 
        # is reduced
        
        # Gradient of correlation w.r.t. h
        def grad_corr_at_shift(h_vec, shift_k):
            g = np.zeros_like(h_vec)
            if shift_k >= 0:
                # h[i] contributes to correlation with (1-h[i+shift_k]) 
                # for all valid i's
                h_vals = h_vec[shift_k:]
                h_base = h_vec[:-shift_k]
                for i in range(len(h_base)):
                    g[i] += (1 - h_vals[i])
                    g[i+shift_k] -= h_base[i]
            else:
                # shift_k < 0: h[i+shift_k] contributes to correlation with (1-h[i])
                shift_k = -shift_k
                h_vals = h_vec[shift_k:]
                h_base = h_vec[:-shift_k]
                for i in range(len(h_base)):
                    g[i] += (1 - h_vals[i])
                    g[i+shift_k] -= h_base[i]
            return g
        
        # Adjustment via gradient descent
        adjustment = grad_corr_at_shift(h_vec, shift_k) * 0.01
        new_h = h_vec - adjustment
        new_h = project_feasible(new_h)
        
        return new_h

    # Main optimization loop with gentle updates

            h = solve_subproblem_for_shift(h, shift_idx)

            prog = np.random.rand() < 0.7  # 70% chance
            if prog:
                # Some localized change
                idx = np.random.randint(n_points)
                h[idx] += np.random.normal(0, 0.01)
                h = project_feasible(h)
            else:
                # Global restart
                h = np.zeros(n_points)
                x = np.linspace(0, 2, n_points)
                for i in range(n_points):
                    h[i] = max(0, 1 - abs(x[i] - 1))
                h = h * (n_points / 2) / np.sum(h)

        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_improved = iter_count
        else:
            patience += 1
            if patience > max_patience and iter_count - last_improved > 200:
                # Perturb randomly if no progress
                idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.1), 0, 1)
                h = project_feasible(h)
                patience = 0

        iter_count += 1

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
