import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 100
    inertia_w = 0.9
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.05
    
    # Initialize particles
    particles = [project_feasible(np.random.rand(n_points)) for _ in range(n_particles)]
    velocities = [np.random.randn(n_points) * 0.1 for _ in range(n_particles)]
    
    # Personal best
    pbest = [p.copy() for p in particles]
    pbest_C5 = [compute_C5(p) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_C5)
    gbest = pbest[gbest_idx].copy()
    gbest_C5 = pbest_C5[gbest_idx]
    
    # PSO loop
    for iter_num in range(max_iter):
        if elapsed() > budget_s * 0.9:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply mutation
            if np.random.random() < mutation_rate:
                idxs = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                for j in idxs:
                    particles[i][j] += np.random.normal(0, 0.02)
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate and update personal best
            current_C5 = compute_C5(particles[i])
            if current_C5 < pbest_C5[i]:
                pbest[i] = particles[i].copy()
                pbest_C5[i] = current_C5
                
                # Update global best if needed
                if current_C5 < gbest_C5:
                    gbest = particles[i].copy()
                    gbest_C5 = current_C5
            
        # Adaptive parameters
        inertia_w = 0.9 - (0.5 * iter_num / max_iter)
    
    # Refinement with trust region method
    h = gbest.copy()
    
    # Gradient-based refinement
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad

    def trust_region_update(h_vec, eta=0.01, max_iter=10):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            grad = get_gradient(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Refinement loop
    eta = 0.01
    patience = 0
    max_patience = 20
    
    while elapsed() < budget_s:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb solution slightly
                h = project_feasible(h + np.random.normal(0, 0.01, n_points))
                patience = 0
                eta *= 0.95  # decay learning rate
        
        # Apply trust region update
        h = trust_region_update(h, eta)
        
        # Periodic update of eta
        if patience % 5 == 0:
            eta *= 0.95

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
