import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()

            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def get_gradient(h_vec, k_idx=None):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        if k_idx is None:
            k_idx = np.argmax(corr)
        
        # Compute gradient of correlation at k_idx
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[k_idx] - corr_minus[k_idx]) / (2 * eps)
            
        return grad
    
    # More aggressive optimization with mixed strategies
    def adaptive_update(h_vec, eta=0.02, max_iter=20):
        h_new = h_vec.copy()
        for i in range(max_iter):
            # Get current max overlap and its corresponding shift
            corr = correlate(h_new, 1 - h_new, mode="full")
            max_k_idx = np.argmax(corr)
            
            # Compute gradient and update
            grad = get_gradient(h_new, max_k_idx)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
            
            # Occasionally perturb to escape local minima
            if i % 5 == 0 and np.random.rand() < 0.2:

                for idx in idxs:
                    h_new[idx] += np.random.normal(0, 0.02)
            
        return h_new
    
    # Main optimization loop
    eta = 0.02
    patience = 0
    max_patience = 100
    iter_count = 0

            best_C5 = current_C5

            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)

                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)

            eta *= 0.92
        
        iter_count += 1

    # Final evaluation

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
