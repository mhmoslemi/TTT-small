import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

            h[i] = max(0, 1 - abs(x[i] - 1))

        h = h * (n_points / 2) / np.sum(h)

        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def adaptive_line_search(h_vec, grad, eta):
        # Simple backtracking line search
        alpha = eta
        current_val = compute_C5(h_vec)
        while alpha > 1e-6:
            test_h = project_feasible(h_vec - alpha * grad)
            test_val = compute_C5(test_h)
            if test_val <= current_val:
                return alpha
            alpha *= 0.5
        return eta
    
    # Multi-resolution approach
    resolutions = [50, 80, 120, 200]
    current_n = n_points
    current_h = h.copy()
    
    # Deep search with population-based optimization
    def optimize_with_population():
        pop_size = 15
        population = []
        
        # Initialize population
        for _ in range(pop_size):
            h_candidate = current_h.copy()
            # Random perturbations with small noise
            noise = np.random.normal(0, 0.02, len(h_candidate))
            h_candidate += noise
            h_candidate = project_feasible(h_candidate)
            population.append(h_candidate)

            fitness = [compute_C5(hp) for hp in population]

            best_fitness = fitness[sorted_indices[0]]

            # Interpolate to finer resolution

            current_h = np.interp(x_new, x_old, old_h)
