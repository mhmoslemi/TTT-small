import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def parametrization(h_vec):
        # Parametrize to keep constraints satisfied
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Gradient approximation
    def get_gradient(h_vec, eps=1e-6):
        h_vec = np.array(h_vec)
        # Find the index that maximizes correlation
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)

            new_h = h_vec - alpha * grad
            new_h = project_feasible(new_h)
            return compute_C5(new_h)

        try:
            res = minimize_scalar(phi, bounds=(0, alpha_max), method='bounded')

                    population[i] = updated
                    fitness[i] = new_val
