import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)

        n_points = 45

        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Exponential smoothing with barrier method
    def optimize_with_exponential_smoothing():
        # Parameters
        beta = 1000.0  # temperature
        gamma = 1e-4   # barrier strength
        eta = 1e-5     # learning rate
        beta_decay = 0.99
        gamma_increase = 1.1
        max_iter = 10000
        patience_threshold = 100
        patience = 0
        best_obj = float('inf')
        best_h_global = h.copy()
        
        h_current = h.copy()
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute correlation
            corr = correlate(h_current, 1 - h_current, mode="full")
            max_corr = np.max(corr)
            
            if max_corr == 0:
                max_corr = 1
            
            # Rescale correlation for numerical stability
            scaled_corr = corr / max_corr
            
            # Exponential penalty
            exp_penalty = np.sum(np.exp(-beta * scaled_corr))
            
            # Barrier term
            barrier = -gamma * np.sum(np.log(np.clip(h_current, 1e-8, 1-1e-8)) + 
                                     np.log(np.clip(1 - h_current, 1e-8, 1-1e-8)))
            
            # Total objective
            obj = exp_penalty + barrier
            
            # Compute gradient using finite differences
            eps = 1e-5
            grad = np.zeros_like(h_current)
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                scaled_plus = corr_plus / max_corr
                scaled_minus = corr_minus / max_corr
                
                exp_plus = np.sum(np.exp(-beta * scaled_plus))
                exp_minus = np.sum(np.exp(-beta * scaled_minus))
                
                grad[j] = (exp_plus - exp_minus) / (2 * eps) + \
                        (-gamma / (h_current[j] + 1e-8) + gamma / (1 - h_current[j] + 1e-8))
            
            # Update
            h_new = h_current - eta * grad
            h_new = project_feasible(h_new)
            
            # Update parameters
            if i % 50 == 0:
                beta *= beta_decay
                gamma *= gamma_increase
            
            # Check for improvement
            if obj < best_obj:
                best_obj = obj
                best_h_global = h_new.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > patience_threshold:
                # Perturb a bit
                indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                for idx in indices:
                    h_new[idx] += np.random.normal(0, 0.01)
                h_new = project_feasible(h_new)
                patience = 0
            
            h_current = h_new
        
        return best_h_global, best_obj
    
    # Run the optimization
    final_h, final_obj = optimize_with_exponential_smoothing()
    
    # Final evaluation
    final_C5 = compute_C5(final_h)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
