import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to h(x)"""
    h = a[0] * np.ones_like(t)
    for i in range(1, len(a)):
        h += a[i] * np.cos(i * np.pi * t) + b[i] * np.sin(i * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Clip h to [0,1] and normalize so that integral = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Evaluate C_5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    n_points = 45

    N = 10  # Number of Fourier coefficients (excluding DC)
    P = 50  # Number of particles
    max_iter = 1000
    
    # Initialize particles
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N + 1))  # a0, a1,...,aN, b1,...,bN
    velocities = np.zeros_like(particles)

    # Evaluate initial solution
    initial_C = evaluate(initial_h, dx)
    global_best = initial_h.copy()

    w = 0.7  # Inertia weight

            # Convert coefficients to h(x)
            a = particles[i, :N+1]  # a0, a1, ..., aN
            b = particles[i, N+1:]  # b1, b2, ..., bN
            h = fourier_to_h(a, b, t)
            h = clip_and_normalize(h, dx)
            
            # Evaluate fitness
            fitness = evaluate(h, dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:

                personal_best_fitness[i] = fitness

        for i in range(P):

    return (global_best, global_best_fitness, n_points)
