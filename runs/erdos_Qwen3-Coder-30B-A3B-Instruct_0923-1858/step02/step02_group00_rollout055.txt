            h[i] = max(0, 1 - abs(x[i] - 1))

        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simple alternate optimization approach
    def optimize_step(h_vec, num_samples=15):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Update using a subset of elements
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        indices_to_sample = np.random.choice(len(h_vec), size=min(num_samples, len(h_vec)), replace=False)
        
        for i in indices_to_sample:
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)

    # Final evaluation

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
