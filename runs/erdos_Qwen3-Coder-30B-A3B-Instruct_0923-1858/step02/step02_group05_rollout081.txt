import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        initial_h = project(initial_h, dx)
    
    # PSO Parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        # Start from initial_h with small random perturbations
        h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())

    # Find initial global best
    best_idx = np.argmin(p_best_values)
    g_best = p_best[best_idx].copy()
    g_best_value = p_best_values[best_idx]
    
    # PSO iterations
    t0 = time.time()
    max_time = budget_s - 2  # Leave headroom
    
    iteration = 0
    while time.time() - t0 < max_time:
        iteration += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] +

            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            # Update position
            particles[i] = particles[i] + velocities[i]
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_value = evaluate(particles[i], dx)
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value
                # Update global best if needed
                if current_value < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_value

        if g_best_value < 0.3807:
