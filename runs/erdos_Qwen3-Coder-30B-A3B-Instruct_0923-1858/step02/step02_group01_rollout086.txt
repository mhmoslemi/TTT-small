import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Apply mutation to an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2, dx, crossover_prob=0.8):
    """Perform uniform crossover between two parents."""
    if np.random.rand() < crossover_prob:
        mask = np.random.rand(len(parent1)) > 0.5
        offspring1 = np.where(mask, parent1, parent2)
        offspring2 = np.where(mask, parent2, parent1)
        offspring1 = project(offspring1, dx)
        offspring2 = project(offspring2, dx)
        return offspring1, offspring2
    else:
        return parent1.copy(), parent2.copy()

def tournament_selection(population, fitness, tournament_size=3):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index].copy()

    dx = 2.0 / n_points

        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = project(initial_h, dx)

    # GA parameters
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    tournament_size = 3
    max_generations = 10000
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points, dx))
        
    # Evaluate initial population
    fitness = [evaluate(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C = fitness[best_idx]
    
    t0 = time.time()
    max_time = budget_s - 10
    
    stagnation_count = 0
    reset_count = 0
    
    for generation in range(max_generations):
        if time.time() - t0 > max_time:
            break
            
        # Selection, crossover, mutation
        new_population = []
        for _ in range(population_size // 2):
            # Selection
            parent1 = tournament_selection(population, fitness, tournament_size)
            parent2 = tournament_selection(population, fitness, tournament_size)
            
            # Crossover
            offspring1, offspring2 = crossover(parent1, parent2, dx, crossover_prob)
            
            # Mutation
            offspring1 = mutate(offspring1, dx, mutation_rate)
            offspring2 = mutate(offspring2, dx, mutation_rate)
            
            new_population.extend([offspring1, offspring2])
        
        # Evaluate new population
        new_fitness = [evaluate(ind, dx) for ind in new_population]
        
        # Replace worst individuals with new offspring
        population = new_population
        fitness = new_fitness
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_C:
            best_C = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adjust parameters if stagnation detected
        if stagnation_count > 50 and mutation_rate < 0.2:
            mutation_rate = 0.2
        if stagnation_count > 100:
            if reset_count < 3:
                # Reset with new random individuals

                best_h = population[best_idx].copy()
                best_C = fitness[best_idx]
                stagnation_count = 0
                reset_count += 1
            else:
                # Terminate if we've exhausted reset attempts
                break

    return (best_h, best_C, n_points)
