import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_kernel(sigma):
    """Create a smoothing kernel."""
    kernel_size = 2 * sigma + 1
    kernel = np.zeros(kernel_size)
    for i in range(-sigma, sigma+1):
        kernel[i + sigma] = 0.54 + 0.46 * np.cos(np.pi * i / sigma)
    kernel /= kernel.sum()
    return kernel

    """Compute gradient using finite differences with smoothing."""

        h_minus[j] -= eps

        c_minus = evaluate(h_minus_smooth, dx)
        
        grad[j] = (c_plus - c_minus) / (2 * eps)
    
    return grad
