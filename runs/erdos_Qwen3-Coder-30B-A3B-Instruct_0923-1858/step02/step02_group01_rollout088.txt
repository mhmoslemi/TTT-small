import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Mutate an individual with Gaussian noise."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.normal(0, 0.05, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2, dx, crossover_prob=0.8):
    """Uniform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    
    return project(child1, dx), project(child2, dx)

def tournament_selection(population, fitness, tournament_size=3):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

        # First 15 points: increasing linearly from 0 to 1
        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = project(initial_h, dx)
    
    # GA parameters
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    max_generations = 5000
    stagnation_threshold = 50
    reset_threshold = 100
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points, dx))
    
    # Evaluate initial population
    fitness = [evaluate(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C = fitness[best_idx]
    
    # Track progress
    no_improvement_count = 0
    last_best = best_C
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave headroom
    
    for gen in range(max_generations):

        # Tournament selection
        new_population = []

            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2, dx, crossover_prob)
            
            child1 = mutate(child1, dx, mutation_rate)
            child2 = mutate(child2, dx, mutation_rate)
            
            new_population.extend([child1, child2])

        new_fitness = [evaluate(ind, dx) for ind in new_population]

        all_populations = population + new_population
        all_fitness = fitness + new_fitness
        
        # Keep best individuals
        sorted_indices = np.argsort(all_fitness)
        population = [all_populations[i] for i in sorted_indices[:population_size]]

        # Update best
        current_best_idx = np.argmin(fitness)

            no_improvement_count = 0
            mutation_rate = 0.1
