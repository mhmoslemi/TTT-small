import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available, else create triangular
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Helper to project feasible
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Track best
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.99
    min_T = 1e-6
    mutation_step = 0.1
    iter_count = 0
    max_iter = 100000
    
    # Main SA loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        if iter_count >= max_iter:
            break
            
        # Make a perturbation
        h_new = h.copy()
        n_mutate = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=n_mutate, replace=False)
        for i in indices:
            delta = np.random.normal(0, mutation_step)
            h_new[i] += delta
            
        # Project to feasible region
        h_new = project_feasible(h_new)
        
        # Evaluate
        C5_new = compute_C5(h_new)
        C5_old = compute_C5(h)
        
        # Accept or reject
        delta = C5_new - C5_old
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new.copy()
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                
        # Anneal temperature
        T = max(min_T, T * cooling_rate)
        mutation_step = T
        
        iter_count += 1
    
    return (best_h.tolist(), best_C5, len(best_h))
