import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Convex relaxation approach
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # This will serve as a warm start
        try:
            prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var)), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                return np.array(h_var.value)
        except:
            pass
        return h.copy()

    # Gradient computation using finite differences 
    def get_gradient(h_vec, k_idx):
        h_vec = np.array(h_vec)
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[k_idx] - corr_minus[k_idx]) / (2 * eps)
            
        return grad

    # Optimizer using projected gradient descent with line search
    def optimize_step(h_vec, lr=0.01):
        # Identify worst shift
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        
        # Compute gradient
        grad = get_gradient(h_vec, max_k_idx)
        
        # Line search
        for alpha in [1.0, 0.5, 0.1, 0.05, 0.01]:
            h_new = h_vec - alpha * grad
            h_new = project_feasible(h_new)
            
            if compute_C5(h_new) < compute_C5(h_vec):
                return h_new, alpha
        
        return h_vec, lr

    # Simple trust region method
    def trust_region_update(h_vec, eta=0.05, max_iter=20):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new, np.argmax(correlate(h_new, 1-h_new, mode="full")))
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Initialize with convex relaxation 
    if elapsed() < budget_s * 0.3:
        h = solve_convex_relaxation()

    # Main optimization loop
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Optimization step
        h, lr = optimize_step(h)
        
        # Occasionally perturb the solution to escape local minima
        if iter_count % 20 == 0 and elapsed() < budget_s * 0.9:
            idxs = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            for i in idxs:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)
                
        iter_count += 1

    # Final refinement with simpler local search
    for _ in range(50):
        if elapsed() > budget_s * 0.95:
            break
        h = trust_region_update(h)
        
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, len(best_h))
