import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    dx = 2.0 / n_points

        particles.append(h.copy())

    c2 = 2.0

    p_best = [p.copy() for p in particles]
    p_best_scores = [evaluate(p, dx) for p in particles]
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # Main PSO loop
    for iteration in range(1000):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] +

            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            score = evaluate(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = particles[i].copy()
                
                # Update global best
                if score < g_best_score:
                    g_best_score = score
                    g_best = particles[i].copy()
    
    # Final validation
    final_h = project(g_best, dx)
    final_C = evaluate(final_h, dx)
    
    return (final_h, final_C, n_points)
