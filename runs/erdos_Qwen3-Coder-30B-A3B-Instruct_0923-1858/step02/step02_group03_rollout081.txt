import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values"""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Ensure h is in [0,1] and integrates to 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    particles = np.random.uniform(-0.5, 0.5, (P, 2*N))  # 2*N because of real/imag parts

    # Initialize personal bests
    personal_bests = particles.copy()
    personal_best_scores = np.full(P, np.inf)
    
    # Initialize global best
    global_best = particles[0].copy()
    global_best_score = np.inf
    
    start_time = time.time()

    # Precompute t values
    t = np.linspace(0, 2, n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(P):
            # Update h from coefficients
            a = particles[i, :N]
            b = particles[i, N:]
            h = fourier_to_h(a, b, t)
            h = clip_and_normalize(h, dx)
            
            # Evaluate
            score = evaluate(h, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_bests[i] = particles[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = particles[i].copy()
                
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            phi1 * r1 * (personal_bests[i] - particles[i]) + 
                            phi2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
    
    # Final evaluation with the best found
    a_best = global_best[:N]
    b_best = global_best[N:]
    h_final = fourier_to_h(a_best, b_best, t)
    h_final = clip_and_normalize(h_final, dx)
    final_C = evaluate(h_final, dx)
    
    return (h_final, final_C, n_points)
