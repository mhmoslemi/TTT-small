import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:

        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = initial_h / (np.sum(initial_h) * dx)

    try:
        # Create variables
        h = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) * dx == 1,
            t >= 0
        ]
        
        # Compute overlap bounds using McCormick relaxations
        # For each shift k, we'll create a constraint
        max_shift = n_points - 1
        
        # We'll compute the constraints for all shifts
        for k in range(1, max_shift):
            # For shift k, we need to consider h[i] * h[i+k] terms
            # Introduce auxiliary variables y[i,k] = h[i] * h[i+k] for valid i
            for i in range(n_points - k):
                # Add McCormick relaxation:
                # y[i,k] >= h[i] + h[i+k] - 1
                # y[i,k] <= h[i]
                # y[i,k] <= h[i+k]
                # y[i,k] >= 0
                y = cp.Variable()
                constraints.append(y >= h[i] + h[i+k] - 1)
                constraints.append(y <= h[i])
                constraints.append(y <= h[i+k])
                constraints.append(y >= 0)
                
                # The constraint we want to write for this shift:
                # S(k) = \sum_i (h[i] - y[i,k]) dx \leq t
                # but since y = h[i]*h[i+k], 
                # the actual overlap integrand is h[i](1 - h[i+k]) = h[i] - h[i]*h[i+k]
                # so S(k) = \sum_i (h[i] - (h[i]*h[i+k])) dx = (sum h[i] - sum h[i]*h[i+k]) dx
                # Which needs to be bounded by t
                # So: sum(h[i] - h[i]*h[i+k]) * dx <= t
                # i.e., sum(h[i] - (h[i]*h[i+k])) * dx <= t
                # We'll use the auxiliary variables:
                # sum_i h[i] * dx - sum_i y[i,k] * dx <= t
                constraints.append(-y * dx <= t)  # This is wrong
                # Actually we just need:
                # sum(h[i]*(1-h[i+k])) * dx <= t
                # This is sum of h[i] - h[i] * h[i+k] 
                # The auxiliary can be defined as: y[i,k] = h[i] * h[i+k]
                # Then sum of h[i] - y[i,k] ≤ t/dx
                # So sum(h[i]) - sum(y[i,k]) <= t/dx
                # i.e. sum(h) - sum(y) <= t/dx
                # This is:
                # Note: For compound constraints, it is simpler to recompute them

        # Introduce auxiliary variables for constraints on overlap integrals

        # We need to ensure S(k) <= t for each k

        # We want to bound all overlaps.

        max_iter = 50000

        adaptive_lr = 0.01

            if time.time() - start_time > max_time:
                break
                
            # Compute gradient (numerical)
            eps = 1e-5
            grad = np.zeros_like(h_current)
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)

                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (1.0 / (np.sum(h_minus) * dx))
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                
                grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
                
            # Update with projected gradient descent
            h_new = h_current - adaptive_lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (1.0 / (np.sum(h_new) * dx))  # Project
            
            # Calculate new overlap
            current_C = evaluate(h_new, dx)
            
            if current_C < best_C:
                h_current = h_new
                best_C = current_C
                best_h = h_new.copy()
                adaptive_lr = min(0.05, adaptive_lr * 1.05)
            else:
                h_current = h_new
                adaptive_lr = max(1e-6, adaptive_lr * 0.95)
                
            # Inject noise occasionally
            if iterations % 100 == 0 and np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.01, len(h_current))
                h_current = np.clip(h_current + noise, 0, 1)
                h_current = h_current * (1.0 / (np.sum(h_current) * dx))
            
            iterations += 1
            
        return (best_h, best_C, n_points)

        # Fall back to simple optimization
        h_current = initial_h.copy()
        best_h = h_current.copy()
        best_C = evaluate(h_current, dx)
        
        startTime = time.time()
        lr = 0.02
        patience = 0
        max_patience = 200
        
        iter_limit = 100000  # Ceiling iterations
        for i in range(iter_limit):
            if time.time() - startTime > budget_s - 10:
                break
                
            # Compute gradient estimation for each variable
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[j] += eps

                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)

                corr_plus = correlate(h_plus, 1-h_plus, mode='full')
                corr_minus = correlate(h_minus, 1-h_minus, mode='full')
                
                grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
            # Update step
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (1.0 / (np.sum(h_new) * dx))

            current_C = evaluate(h_new, dx)
            if current_C < best_C:
                h_current = h_new
                best_C = current_C
                best_h = h_new.copy()
                patience = 0
                lr = min(0.05, lr * 1.05)  # Adaptively increase learning rate
            else:
                h_current = h_new
                patience += 1
                if patience > max_patience:
                    lr = max(1e-5, lr * 0.95)  # Reduce learning rate
                    patience = 0
            
            # Occasionally add noise to escape local minima
            if i % 50 == 0 and np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.01, len(h_current))
                h_current = np.clip(h_current + noise, 0, 1)
                h_current = h_current * (1.0 / (np.sum(h_current) * dx))
    
    return (best_h, best_C, n_points)

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
