import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

        if len(h) != n_points:

    except NameError:

        h = np.zeros(n_points)
        segments = 10
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            # Use sinusoidal variation for diversity
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        h = project(h, dx)

    # Genetic Algorithm parameters
    population_size = 50
    generations = 10000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.1:  # 10% from initial solution
            individual = h + np.random.normal(0, 0.05, n_points)
        else:
            individual = np.random.rand(n_points) * 0.5 + 0.25  # Random in [0.25, 0.75]
        individual = project(individual, dx)
        population.append(individual)
    
    # Track best solution
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    # Generation loop
    for gen in range(generations):
        if time.time() - t0 > max_time:
            break
            
        # Evaluate fitness for all population
        fitness = []
        for individual in population:
            C = evaluate(individual, dx)
            fitness.append(1.0 / (1e-10 + C))  # Avoid division by zero
        
        # Sort by fitness and select elites
        indices = np.argsort(fitness)[::-1]
        elite = [population[i] for i in indices[:elite_size]]
        
        # Create new population
        new_population = elite.copy()
        
        # Tournament selection and reproduction
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]
            
            # Clone best of tournament
            parent = population[winner_index].copy()
            
            if np.random.rand() < crossover_rate and len(new_population) < population_size - 1:
                # Selection of second parent
                tournament_indices2 = np.random.choice(len(population), tournament_size)
                tournament_fitness2 = [fitness[i] for i in tournament_indices2]
                parent2_index = tournament_indices2[np.argmax(tournament_fitness2)]
                parent2 = population[parent2_index].copy()
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent[crossover_point:]])
                
                # Mutate
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child1[i] += np.random.normal(0, 0.02)
                    if np.random.rand() < mutation_rate:
                        child2[i] += np.random.normal(0, 0.02)
                
                # Project
                child1 = project(child1, dx)
                child2 = project(child2, dx)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just mutate
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        parent[i] += np.random.normal(0, 0.02)
                parent = project(parent, dx)
                new_population.append(parent)
        
        population = new_population[:population_size]
        
        # Check for improvement in this generation
        current_C = min([evaluate(individual, dx) for individual in population])
        if current_C < best_C:
            best_C = current_C
            best_h = min(population, key=lambda x: evaluate(x, dx)).copy()
    
    return (best_h, best_C, n_points)
