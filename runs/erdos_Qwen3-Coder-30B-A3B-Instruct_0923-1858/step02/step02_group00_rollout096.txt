import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    elapsed = lambda: time.time() - start_time

        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Temperature control
    beta = 1000.0
    gamma = 1e-4
    eta = 1e-5
    beta_decay = 0.99
    gamma_incr = 1.1
    patience = 0
    max_patience = 20
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)  # Keep away from boundaries
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_objective_and_grad(h_vec, beta, gamma):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_val = np.max(corr)
        if max_val > 0:
            corr = corr / max_val
        else:
            corr = corr.copy()
            
        # Exponential penalty
        penalty = np.sum(np.exp(-beta * corr))
        
        # Barrier function
        barrier = -gamma * np.sum(np.log(h_vec) + np.log(1 - h_vec))
        
        # Combine
        F = penalty + barrier
        
        # Compute gradient
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            # Scale correlation before exponent
            max_val_plus = np.max(corr_plus)
            max_val_minus = np.max(corr_minus)
            
            if max_val_plus > 0:
                corr_plus = corr_plus / max_val_plus
            if max_val_minus > 0:
                corr_minus = corr_minus / max_val_minus
            
            exp_plus = np.sum(np.exp(-beta * corr_plus))
            exp_minus = np.sum(np.exp(-beta * corr_minus))
            
            grad[i] = (exp_plus - exp_minus) / (2 * eps) - gamma * (1/h_vec[i] - 1/(1-h_vec[i]))
            
        return F, grad
    
    # Optimization loop
    last_F = float('inf')
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb randomly to escape local minima
                idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.03), 1e-6, 1 - 1e-6)
                h = project_feasible(h)
                patience = 0

        F, grad = compute_objective_and_grad(h, beta, gamma)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Update hyperparameters
        if iter_count % 50 == 0:
            beta *= beta_decay
            gamma *= gamma_incr
            if eta > 1e-7:
                eta *= 0.5
        
        # Update learning rate if no improvement
        if abs(F - last_F) < 1e-6 and iter_count % 100 == 0:
            eta *= 0.9
            
        iter_count += 1
        last_F = F
        
        # Occasionally reproject
        if iter_count % 50 == 0:
            h = project_feasible(h)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
