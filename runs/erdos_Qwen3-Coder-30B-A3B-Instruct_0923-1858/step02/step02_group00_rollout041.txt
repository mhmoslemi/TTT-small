import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Particle Swarm Optimization (PSO) approach
    def pso_optimization():
        # PSO parameters
        n_particles = 50
        max_iter = 100
        inertia_weight = 0.9
        cognitive_weight = 2.0
        social_weight = 2.0
        mutation_rate = 0.05
        
        # Initialize particles
        particles = []
        velocities = []
        personal_best = []
        personal_best_values = []
        
        for i in range(n_particles):
            h_init = h.copy()
            # Add some noise to create diversity
            noise = np.random.normal(0, 0.1, n_points)
            h_init += noise
            h_init = project_feasible(h_init)
            particles.append(h_init.copy())
            velocities.append(np.zeros(n_points))
            personal_best.append(h_init.copy())
            personal_best_values.append(compute_C5(h_init))
        
        # Global best
        best_idx = np.argmin(personal_best_values)
        global_best = personal_best[best_idx].copy()
        global_best_value = personal_best_values[best_idx]
        
        for iteration in range(max_iter):
            if elapsed() > budget_s * 0.8:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (inertia_weight * velocities[i] 
                               + cognitive_weight * r1 * (personal_best[i] - particles[i])
                               + social_weight * r2 * (global_best - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Evaluate new position
                new_value = compute_C5(particles[i])
                
                # Update personal best
                if new_value < personal_best_values[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_values[i] = new_value
                
                # Update global best
                if new_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = new_value
            
            # Optional mutation
            for i in range(n_particles):
                if np.random.random() < mutation_rate:
                    idxs = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                    for idx in idxs:
                        particles[i][idx] += np.random.normal(0, 0.05)
                    particles[i] = project_feasible(particles[i])
                    new_value = compute_C5(particles[i])
                    if new_value < personal_best_values[i]:
                        personal_best[i] = particles[i].copy()
                        personal_best_values[i] = new_value
                        if personal_best_values[i] < global_best_value:
                            global_best = personal_best[i].copy()
                            global_best_value = personal_best_values[i]
                            
        return global_best, global_best_value

    # Refinement with gradient descent and trust region
    def refine_solution(h_init):
        h_refined = h_init.copy()
        max_iter = 100
        eta = 0.01
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Compute the current maximum overlap
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_refined) - 1)
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            for j in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_minus = h_refined.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update with trust region
            h_new = h_refined - eta * grad
            h_new = project_feasible(h_new)
            
            # Accept if better
            if compute_C5(h_new) < compute_C5(h_refined):
                h_refined = h_new
            else:
                eta *= 0.9  # Reduce step size if no improvement
                
        return h_refined

    # PSO phase
    h_pso, c5_pso = pso_optimization()
    if c5_pso < best_C5:
        best_C5 = c5_pso
        best_h = h_pso.copy()

    # Refinement phase
    h_refined = refine_solution(h_pso)
    c5_refined = compute_C5(h_refined)
    if c5_refined < best_C5:
        best_C5 = c5_refined
        best_h = h_refined.copy()
    
    # Additional refinement using a local search
    for _ in range(20):
        if elapsed() > budget_s * 0.99:
            break
        # Small random perturbations
        h_pert = best_h.copy()
        idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for j in idxs:
            h_pert[j] += np.random.normal(0, 0.01)
        h_pert = project_feasible(h_pert)
        c5_pert = compute_C5(h_pert)
        
        if c5_pert < best_C5:
            best_C5 = c5_pert
            best_h = h_pert.copy()
        
        # Multiple local searches
        for _ in range(3):
            h_search = h_pert.copy()
            h_search = refine_solution(h_search)
            c5_search = compute_C5(h_search)
            if c5_search < best_C5:
                best_C5 = c5_search
                best_h = h_search.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
