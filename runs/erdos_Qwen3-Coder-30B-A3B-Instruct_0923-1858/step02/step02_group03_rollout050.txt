import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_dimensions = 20  # Number of Fourier coefficients
    max_iter = 1000
    omega = 0.7
    phi1 = phi2 = 1.5
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_dimensions) * 2 - 1  # Random coefficients in [-1,1]
    velocities = np.zeros_like(particles)
    
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, np.inf)
    
    global_best_idx = 0
    global_best_fitness = np.inf
    
    # For PSO, we need to convert Fourier coefficients to function values
    def coefficients_to_h(coeffs):
        # Create h(x) from Fourier coefficients
        t = np.linspace(0, 2, n_points)
        h = np.zeros_like(t)
        for i, coeff in enumerate(coeffs):
            if i % 2 == 0:  # Cosine terms
                h += coeff * np.cos(i * np.pi * t / 2)
            else:  # Sine terms
                h += coeff * np.sin(i * np.pi * t / 2)
        # Add constant term
        h += coeffs[0]  # This will be the constant term
        return h
    
    # Time tracking
    start_time = time.time()
    max_time = budget_s - 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(n_particles):
            # Convert coefficients to h(x)
            h = coefficients_to_h(particles[i])
            h = project(h, dx)
            
            # Evaluate fitness
            fitness = evaluate(h, dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_idx = i
        
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           phi1 * r1 * (personal_best[i] - particles[i]) +

            particles[i] = np.clip(particles[i], -2, 2)
    
    # Get best solution
    best_coeffs = personal_best[global_best_idx]
    best_h = coefficients_to_h(best_coeffs)
    best_h = project(best_h, dx)
    best_C = evaluate(best_h, dx)
