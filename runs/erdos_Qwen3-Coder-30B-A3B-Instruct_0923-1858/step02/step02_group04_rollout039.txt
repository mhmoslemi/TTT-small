    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Feasibility projection
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Memoized correlation compute
    corr_cache = {}
    def cached_correlate(h_vec):
        h_tuple = tuple(h_vec)
        if h_tuple in corr_cache:
            return corr_cache[h_tuple]
        else:
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            corr_cache[h_tuple] = corr
            return corr
    
    # Gradient approximation
    def approx_grad(h_vec, eps=1e-5):
        h_vec = np.array(h_vec)
        corr = cached_correlate(h_vec)
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = cached_correlate(h_plus)
            corr_minus = cached_correlate(h_minus)
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        return grad
    
    # Optimization loop
    patience = 0
    max_patience = 100
    eta = 0.02
    iter_count = 0
    last_improvement = 0

            last_improvement = iter_count

        try:
            grad = approx_grad(h)

            last_improvement = iter_count

    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
