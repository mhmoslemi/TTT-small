import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Initialize
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Convex relaxation to get better starting point
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Simplified relaxation using few key correlations
        max_corr = cp.Variable()
        constraints.append(max_corr >= 0)
        
        # Set up a small subset of correlations
        num_shifts = min(20, n_points // 2)
        shifts = list(range(-num_shifts, num_shifts + 1))
        
        for k in shifts:
            corr_sum = 0
            valid_indices = []
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    corr_sum += h_var[i] * (1 - h_var[j])
                    valid_indices.append((i, j))
            constraints.append(corr_sum <= max_corr)
        
        prob = cp.Problem(cp.Minimize(max_corr), constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'optimal':
                return np.maximum(0, np.minimum(1, h_var.value))
        except:
            pass
        return h.copy()

    # Better initialization
    h = solve_convex_relaxation()
    h = project_feasible(h)

    def adaptive_gradient(h_vec, num_samples=5):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        # Use a subset of points for efficient gradient estimation
        grad = np.zeros_like(h_vec)
        eps = 1e-4

        # Sample a few points for gradient estimation
        sample_indices = np.random.choice(len(h_vec), 
                                         min(num_samples, len(h_vec)), 
                                         replace=False)
        
        for i in sample_indices:
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad

    # Simpler trust region step
    def trust_region_update(h_vec, eta=0.02):
        grad = adaptive_gradient(h_vec, num_samples=10)
        h_new = h_vec - eta * grad
        h_new = project_feasible(h_new)
        return h_new

    # Main optimization using coordinate descent variant
    eta = 0.01
    patience = 0
    max_patience = 30
    max_iter = 5000
    iter_count = 0
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Slight random perturb for escape from local minima
                perturb = np.random.normal(0, 0.02, n_points)
                h = h + perturb
                h = project_feasible(h)
                patience = 0

        h = trust_region_update(h, eta)
        
        # Decrease learning rate over time
        if iter_count % 500 == 0 and iter_count > 0:
            eta *= 0.97
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
