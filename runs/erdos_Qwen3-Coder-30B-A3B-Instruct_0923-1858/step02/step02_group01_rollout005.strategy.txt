def project(h, dx):
    h = np.clip(h, 0, 1)
    current_integral = np.sum(h) * dx
    if current_integral == 0:
        return h
    scale_factor = 1.0 / current_integral
    return h * scale_factor

    eps = 1e-5
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        h_plus = project(h_plus, dx)
        h_minus = project(h_minus, dx)
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        gradient[i] = (c_plus - c_minus) / (2 * eps)
