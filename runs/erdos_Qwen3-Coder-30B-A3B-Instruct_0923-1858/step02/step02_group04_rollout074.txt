import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    time_limit = budget_s * 0.98

    def logistic(x):
        return 1 / (1 + np.exp(-x))

        # Map to [0,1] and normalize
        h = logistic(h_raw)
        h = h * (len(h) / 2) / np.sum(h)
        return np.clip(h, 0, 1)

        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / len(h_vec))

    best_C5 = float('inf')

            selected_indices = np.argsort(fitness)[:P//2]

        bounds = [(-10, 10)] * len(best_h)
