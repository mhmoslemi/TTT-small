import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def fourier_basis(x, n_coeff):
    """Generate Fourier basis functions up to n_coeff"""
    basis = np.ones((len(x), n_coeff))
    for i in range(1, n_coeff):
        basis[:, i] = np.cos(i * np.pi * x)
    return basis

def fourier_to_function(coeffs, x, n_coeff):
    """Convert Fourier coefficients to function values"""
    basis = fourier_basis(x, n_coeff)
    return np.dot(basis, coeffs)
