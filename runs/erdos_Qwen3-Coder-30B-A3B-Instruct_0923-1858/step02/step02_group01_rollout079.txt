import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

        initial_h[15:] = np.linspace(1, 0, 30)

        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx))

    def objective(h):
        return evaluate_erdos_solution(h, dx)

    # Define bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    # Define the constraint: sum(h) * dx = 1, or equivalently sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum}

    # Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (1.0 / (h_sum * dx))
        return h

    best_h = initial_h.copy()
    best_C = objective(best_h)
    max_iter = 1000
    t0 = time.time()

    # SLSQP optimization with bounds and constraints
    try:
        res = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons,

            h_opt = project(res.x)
            C_opt = objective(h_opt)

        for i in range(500):

            # Estimate gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            for j in range(len(h)):
                h_plus = h.copy()
