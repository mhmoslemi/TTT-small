import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):

        h = project(h, dx)
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    num_generations = 10000
    num_segments = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.1 and 'initial_h_values' in globals():
            # Start with perturbed initial solution
            h = initial_h_values.copy() + np.random.normal(0, 0.01, n_points)
        else:
            # Random piecewise constant function
            h = np.random.rand(n_points)
        h = project(h, dx)
        population.append(h.copy())
    
    # GA main loop
    t0 = time.time()
    last_improvement = 0
    mutation_rate_step = 0.005
    
    for gen in range(num_generations):
        if time.time() - t0 > budget_s - 10:
            break
            
        # Evaluate population
        fitnesses = []
        for ind in population:
            fit = evaluate(ind, dx)
            fitnesses.append(1.0 / (fit + 1e-10))
        
        # Track best solution
        best_idx = np.argmin([evaluate(ind, dx) for ind in population])
        if evaluate(population[best_idx], dx) < best_C:
            best_C = evaluate(population[best_idx], dx)
            best_h = population[best_idx].copy()
            last_improvement = gen
        
        # Early stopping
        if gen - last_improvement > 500:

        new_population = []

        elite_idx = np.argsort(fitnesses)[-elite_size:]
        for idx in elite_idx:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            def tournament_select():
                tournament_indices = np.random.choice(len(population), tournament_size)
                tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
                return population[winner_idx].copy()
            
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

                child1, child2 = parent1.copy(), parent2.copy()

            else:
                mutation_rate = max(0.01, mutation_rate - mutation_rate_step)
    
    return (best_h, best_C, n_points)
