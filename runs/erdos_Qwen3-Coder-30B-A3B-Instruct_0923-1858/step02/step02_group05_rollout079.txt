import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    particles = np.zeros((n_particles, n_points))

    global_best_idx = np.argmin(p_best_vals)
    g_best = p_best[global_best_idx].copy()

            particles[i] = particles[i] + velocities[i]

            particles[i] = project(particles[i], dx)

            current_val = evaluate(particles[i], dx)

            if current_val < p_best_vals[i]:
                p_best_vals[i] = current_val
                p_best[i] = particles[i].copy()
