import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx))
    
    start_time = time.time()
    best_C = float('inf')
    best_h = initial_h.copy()
    
    # Use a convex relaxation with McCormick envelopes
    # Minimize C subject to:
    # C >= sum(h[i] - h[i]*h[i+k]) * dx for all k
    # 0 <= h[i] <= 1 for all i
    # sum(h) * dx = 1
    
    # Create variables
    h_vars = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) * dx == 1, C >= 0]
    
    # For each shift k, add the overlap constraint
    # Overlap for shift k = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) * dx
    # Equivalent to sum_{i=0}^{n-k-1} (h[i] - h[i]*h[i+k]) * dx
    
    # Use McCormick envelopes
    # Let z[i,k] = h[i] * h[i+k] where we define z[i,k] >= h[i] + h[i+k] - 1
    # z[i,k] <= h[i], z[i,k] <= h[i+k], z[i,k] >= 0
    
    # We approximate with a simpler linearization
    # Minimum of sum(h[i] - z[i,k]) over all feasible z[i,k]
    
    # Compute overlap for each shift via McCormick relaxation
    for k in range(1, n_points):
        # For each i, compute overlap term
        overlap_expr = 0
        for i in range(n_points - k):
            # Add McCormick constraints for z[i,k] = h[i] * h[i+k]
            z = cp.Variable()
            constraints.append(z >= h_vars[i] + h_vars[i+k] - 1)
            constraints.append(z <= h_vars[i])
            constraints.append(z <= h_vars[i+k])
            constraints.append(z >= 0)
            overlap_expr += (h_vars[i] - z) * dx
        constraints.append(overlap_expr <= C)

    # Solve the problem
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_vars.value
            if h_sol is not None:
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol / (np.sum(h_sol) * dx)  # re-project to constraint
                C_sol = evaluate(h_sol, dx)
                if C_sol < best_C:
                    best_C = C_sol
                    best_h = h_sol.copy()
    except Exception as e:
        # If convex optimization fails, use the Markov chain method as backup
        pass
    
    # If we still have a valid best solution, return it
    # Otherwise just return the initial one
    return (best_h, best_C, n_points)
