import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:

        if len(initial_h) != n_points:

    start_time = time.time()
    
    # Initialize
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Convex optimization setup
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1]
    # Add integral constraint
    constraints.append(cp.sum(h_var) * dx == 1)
    
    # Cutting plane method
    iter_count = 0
    max_iter = 1000
    
    while time.time() - start_time < max_time and iter_count < max_iter:
        iter_count += 1
        # Compute current values
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
        
        # Compute all correlations to find most violating constraint
        corr = correlate(h, 1-h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        
        # Add constraint: S(k) <= t    (S(k) = 1 - int h(x) h(x+k) dx)
        # which means: 1 - int h(x) h(x+k) dx <= t
        # =>: int h(x) h(x+k) dx >= 1 - t
        # =>: sum(h[i] * h[i+k] * dx) >= 1 - t for shift k = max_corr_idx - (n_points-1)
        k_shift = max_corr_idx - (n_points - 1)
        # Consider k_shift as sliding, i.e., re-indexing the full convolution
        if k_shift < 0:
            # right-shift: h[i] * h[i+k] 
            # where k is negative => i in [0, n_points+k], conclude first and last indices
            # why do we need this constraint? Because S(k) = corr[n_points-1+k] / dx
            # includes the same terms as sliding. But extreme case problematic.
            # Actually corr corresponds to S(k) = int h(x) (1 - h(x+k)) dx = 1 - int h(x) h(x+k) dx.
            # The correct indexing.
            if k_shift <= -n_points:
                pass
            else:
                # 1 for overlap, 0 for no overlap.
                # We access h[i] referenced from original h starting at i k positions to the right,
                # but take only as many elements as fit.
                if k_shift < 0:  # xs' are right shifted
                    valid_indices = range(-k_shift)
                    # h[i] * h[i+k_shift]  i = 0,...,len(h)-1 ?>
                    # so h[i] corresponds to h[i+k_shift] for valid i
                    valid_i = range(len(h) + k_shift)
                    start_offset = -k_shift
                    active_i = [i for i in valid_i if i < len(h) and i+k_shift >= 0]
                    active_j = [i+k_shift for i in active_i if i+k_shift < len(h)]
                    left_idx = 0
                    right_idx = len(active_j)-1
                    # for all i in some range:
                    # compute: h[i]*h[i+k_shift]
                    # Indices from [0, len(h)-1]
                    # Can only be >= shift valid:
                    # i+k_shift >= 0, so i >= -k_shift
                    # i+k_shift < len(h), so i < len(h) - k_shift
                    # i in [ max(0, -k_shift), min(len(h), len(h) - k_shift) )
                    # = [ max(0, -k_shift), len(h) + min(0, k_shift) )
                    # = [ -k_shift, len(h) + k_shift ]  if k_shift < 0

                    # idx = 0, to n_points - 1
                    # idx + k_shift = 0, to n_points + k_shift - 1 = n_points + k_shift - 1

                    # Or we precompute S(k) for all k. 
                    # So we compute 45 elements to build one constraint for a shift
                    constraint_indices_a = [i for i in range(len(h)) if i + k_shift >= 0 and i < len(h)]
                    constraint_indices_b = [i + k_shift for i in constraint_indices_a]

                        # We essentially create a new constraint in the form:
                        # sum( h_var[a] * h_var[b] ) >= 1 - t
                        # where a, b are lists.
                        # Total inner product constraint for fixed k_shift:
                        # Inner product over k-shifted part of h used to compute the overlap.
                        term = cp.sum(cp.multiply(h_var[constraint_indices_a],
                                                 h_var[constraint_indices_b])) * dx
                        # We are imposing this. S(k) = 1 - int h(x)h(x+k) dx
                        # So S(k) <= t  =>  1 - int h(x)h(x+k) dx <= t
                        # => int h(x)h(x+k) dx >= 1 - t
                        # => sum(h[i]*h[i+k] * dx) >= 1 - t
                        constraints.append(term >= 1 - t)
                else:
                    # This is a simple case
                    pass
        else:
            # k_shift >= 0
            # h[i] * h[i-k_shift]
            # i goes up to n_points - 1, so i - k_shift in [i-k_shift, i-k_shift+44]
            # for valid i: 0 <= i <= 44, and i - k_shift >= 0 && i - k_shift < 45

            # Which means from indices [k_shift, ..., min(n_points-1, k_shift+44)]
            valid_indices = range(k_shift, min(n_points, k_shift + n_points))
            # for i in valid_indices, we have h[i]*h[i-k_shift]
            # to match to the right dimension
            valid_i = [i for i in range(len(h)) if i >= k_shift and i < n_points]
            valid_j = [i - k_shift for i in valid_i]
            if len(valid_i) > 0:
                term = cp.sum(cp.multiply(h_var[valid_i],
                                         h_var[valid_j])) * dx
                constraints.append(term >= 1 - t)
            
        # Formulate new optimization problem
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = project(h_new, dx)
                
                # Check whether any significant improvement
                current_new_C = evaluate(h_new, dx)
                if current_new_C < best_C - 1e-6:  # Prevent oscillation
                    h = h_new
                    best_C = current_new_C
                    best_h = h.copy()
                else:
                    # Keep h current and re-add constraints for stability?
                    h = h_new
            else:
                # Fallback to next iteration     # Stop optimization
                pass
        except:
            pass

    # Final evaluation
    final_C = evaluate(best_h, dx)
    return (best_h, final_C, n_points)
