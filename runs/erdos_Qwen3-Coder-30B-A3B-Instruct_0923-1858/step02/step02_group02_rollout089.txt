import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def construct_piecewise_segments(h_vals, n_segments):
    """Construct segments from continuous function."""
    n_points = len(h_vals)
    segment_width = n_points // n_segments
    segments = []
    
    for i in range(n_segments):
        start = i * segment_width
        end = (i+1) * segment_width if i < n_segments - 1 else n_points
        segments.append(np.mean(h_vals[start:end]))
    
    return np.array(segments)

    # Initialize with segments for better convergence
    n_segments = 10
    segment_positions = np.linspace(0, 2, n_segments + 1)
    h_segment = construct_piecewise_segments(h, n_segments)
    
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10
    
    # Alternating optimization loop
    for iteration in range(10000):
        if time.time() - t0 > max_time:
            break
            
        # Optimize segment heights while fixing positions
        h_segment_prev = h_segment.copy()
        try:
            # CVXPY approach for convex optimization of heights
            h_cvx = cp.Variable(n_segments)
            # This is a simplified model - exact formulation requires more complex constraints
            # We'll use a perceptual scheme to ensure bounds and constraint
            objective = cp.Minimize(cp.sum(cp.multiply(h_cvx, cp.multiply(1.0 - h_cvx, 1.0))))
            constraints = [
                cp.sum(h_cvx) == 1.0/dx,
                h_cvx >= 0,
                h_cvx <= 1
            ]
            prob = cp.Problem(objective, constraints)

                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_segment = h_cvx.value
            except:
                pass
        except:

        # Reconstruct h from segments
        h_reconstructed = np.zeros(n_points)

            end = (i+1) * segment_width if i < n_segments - 1 else n_points
            h_reconstructed[start:end] = h_segment[i]
            
        # Convex relaxation didn't improve - just project the result
        h_reconstructed = project_to_feasible(h_reconstructed, dx)

        current_C = evaluate_erdos_solution(h_reconstructed, dx)
        if current_C < best_C:
