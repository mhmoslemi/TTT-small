import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_initial_h(n_points, dx):
    """Create a symmetric initial guess for h."""
    h = np.zeros(n_points)
    peak = 1.0
    decay_points = n_points // 2
    h[:decay_points] = np.linspace(0, peak, decay_points)
    h[decay_points:] = np.linspace(peak, 0, decay_points + 1)[1:]
    h = project(h, dx)
    return h

def compute_gradient(h, dx):
    """Compute gradient via finite differences."""
    eps = 1e-5
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        h_plus = project(h_plus, dx)
        h_minus = project(h_minus, dx)
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def fire_optimize(h, dx, max_iter=10000, budget_s=600):
    """Optimize using FIRE algorithm."""
    t0 = 0.5
    t1 = 0.8
    alpha = 0.99
    beta = 0.95
    max_bad_steps = 5
    current_t = 1.0
    current_step = 1.0
    
    velocity = np.zeros_like(h)
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        gradient = compute_gradient(h, dx)
        
        # Update velocities and position
        velocity = alpha * velocity + beta * gradient
        h_trial = h + velocity
        
        # Enforce constraints
        h_trial = project(h_trial, dx)
        
        # Compute trial objective
        trial_C = evaluate(h_trial, dx)
        
        # Check trial improvement
        if trial_C < best_C:
            h = h_trial
            best_C = trial_C
            current_step = 1.0
            current_t = t0
        else:
            # Reject step
            velocity = velocity * 0.9
            current_step = min(current_step + 1, max_bad_steps)
            current_t = current_t * (1 / current_step)
        
        # Schedule temperature
        current_t = min(current_t, t1)
        velocity = velocity * current_t
        best_h = h.copy()
    
    return best_h, best_C

def upsample_h(h_low, n_points_high):
    """Upsample solution to higher resolution."""
    x_low = np.linspace(0, 2, len(h_low))
    x_high = np.linspace(0, 2, n_points_high)
    f = interp1d(x_low, h_low, kind='linear')
    h_high = f(x_high)
    dx_high = 2.0 / n_points_high
    return project(h_high, dx_high)
