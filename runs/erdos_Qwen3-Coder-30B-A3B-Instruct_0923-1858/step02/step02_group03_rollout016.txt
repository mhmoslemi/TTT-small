import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        initial_h = np.ones(n_points) * 0.5  # uniform initialization
    
    max_time = budget_s - 10
    start_time = time.time()
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Cutting-plane method approach
    # Variables
    h_vars = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Initial constraints (will be expanded)
    constraints = [
        0 <= h_vars,
        h_vars <= 1,
        cp.sum(h_vars) * dx == 1,
        t >= 0
    ]
    
    # Build initial problem and solve
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Iteratively solve and add violated constraints
    max_iter = 1000
    for it in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        # Fix current h in problem and solve
        try:
            _prob = cp.Problem(objective, constraints)
            _prob.solve(solver=cp.ECOS, verbose=False)
            
            h_new = h_vars.value
            h_new = project(h_new, dx)
            current_C = evaluate(h_new, dx)
            
            if current_C < best_C:
                best_C = current_C
                best_h = h_new.copy()
        except:
            h_new = project(h, dx)
            current_C = evaluate(h_new, dx)
        
        # Check all shifts for the current solution
        c = correlate(h_new, 1-h_new, mode='full')
        max_idx = np.argmax(c)
        max_k = max_idx - (len(h_new) - 1)
        max_val = c[max_idx]
        
        # Add constraint for this shift
        h_shift = np.roll(h_new, max_k)
        inner_product = np.sum(h_new * h_shift)
        s_val = 1 - inner_product
        
        # Add a new constraint if it's violating
        if s_val > 0.99 * t.value:  # Add some tolerance
            new_constraint = s_val <= t
            constraints.append(new_constraint)
        
        # Update h for next iteration
        h = h_new.copy()
        
    return (best_h, best_C, n_points)
