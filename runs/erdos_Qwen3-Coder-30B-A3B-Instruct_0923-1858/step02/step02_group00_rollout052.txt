import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Convex relaxation to improve initial solution
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # This is a bit of a hack to solve in a way that's close to actual problem
        # We'll construct a proxy problem that's solvable and reasonable
        obj = cp.Minimize(cp.sum(h_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return np.array(h_var.value)
        except:
            pass
        return h.copy()

    # Optimized gradient-based update using finite differences
    def compute_gradient(h_vec, max_k, eps=1e-6):
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k] - corr_minus[max_k]) / (2 * eps)
            
        return grad

    # Trust region update
    def trust_region_update(h_vec, max_k, max_iter=20):
        h_new = h_vec.copy()
        eta = 0.005
        
        for i in range(max_iter):
            grad = compute_gradient(h_new, max_k)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
            
        return h_new

    # Importance sampling to focus on worst shifts
    def find_worst_shift(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        return max_k_idx - (len(h_vec) - 1)

    # Main optimization loop
    iter_count = 0
    max_patience = 50
    patience = 0
    max_evals = 10000
    eval_count = 0

    while elapsed() < budget_s * 0.95 and eval_count < max_evals:
        if eval_count % 50 == 0:
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Perturb a few points
                    idxs = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                    for i in idxs:
                        h[i] = np.clip(h[i] + np.random.normal(0, 0.03), 0, 1)
                    h = project_feasible(h)
                    patience = 0
        
        # Find worst shift
        max_k = find_worst_shift(h)
        
        # Update using trust region
        h = trust_region_update(h, max_k, max_iter=5)
        
        eval_count += 1
        iter_count += 1

    # Final cleanup and evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
