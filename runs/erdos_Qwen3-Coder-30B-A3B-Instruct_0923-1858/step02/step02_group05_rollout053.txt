import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    # Initialize from parent construction if available
    try:

        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a different starting point
        t = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        h = project(h, dx)

    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w, c1, c2 = 0.7, 2.0, 2.0
    max_vel = 0.2

    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        # Start near initial solution with some noise
        particle = h + np.random.uniform(-0.1, 0.1, n_points)
        particle = project(particle, dx)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_vals.append(evaluate(particle, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]

    t0 = time.time()
    max_time = budget_s - 2  # Leave headroom

    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (p_best[i] - particles[i]) + 
                             c2 * r2 * (g_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project(particles[i], dx)
            
            # Evaluate current particle
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val

    return (g_best, g_best_val, n_points)
