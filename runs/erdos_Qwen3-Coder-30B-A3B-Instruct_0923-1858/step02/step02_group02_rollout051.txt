import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):

    # PSO parameters
    n_particles = 50
    max_iter = 12000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.2

    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_values = []

    # Initialize with current solution or random
    for i in range(n_particles):
        if i == 0:
            h_particle = h.copy()
        else:
            h_particle = np.random.rand(n_points) * 0.5 + 0.25
        h_particle = project(h_particle, dx)
        particles.append(h_particle)
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        value = evaluate(h_particle, dx)
        pbest.append(h_particle.copy())
        pbest_values.append(value)

    # Global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]

    t0 = time.time()
    for iteration in range(max_iter):
        if time.time() - t0 > budget_s - 5:
            break

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            value = evaluate(particles[i], dx)
            
            # Update personal best
            if value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = value
                
                # Update global best
                if value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = value

    return (gbest, gbest_value, n_points)
