import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_symmetric_initial_h(n_points, dx):
    """Create a symmetric piecewise linear initial guess."""
    h = np.zeros(n_points)
    mid = n_points // 2
    # Increasing part
    h[:mid] = np.linspace(0, 1, mid)
    # Decreasing part
    h[mid:] = np.linspace(1, 0, n_points - mid)
    return project(h, dx)

def compute_gradient(h, dx, eps=1e-5):
    """Compute gradient via finite differences."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def fire_step(h, velocity, grad, dx, current_t=1.0, alpha=0.99, beta=0.95, t0=0.5, t1=0.8):
    """Perform one FIRE step."""
    velocity = alpha * velocity + beta * grad
    h_trial = h + velocity
    h_trial = project(h_trial, dx)
    
    # If this improves the objective, accept
    trial_c = evaluate(h_trial, dx)
    if trial_c < evaluate(h, dx):
        h = h_trial
        current_t = t0
    else:
        # Reject step
        velocity = velocity * 0.9
        current_t = current_t * 0.5
        current_t = max(current_t, t1)
    
    # Apply temperature
    velocity = velocity * current_t
    return h, velocity, trial_c, current_t

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s - 10  # Leave buffer

    # Phase 1: Coarse resolution optimization
    n_points_coarse = 30
    dx_coarse = 2.0 / n_points_coarse
    h = create_symmetric_initial_h(n_points_coarse, dx_coarse)
    best_h = h.copy()
    best_C = evaluate(h, dx_coarse)
    velocity = np.zeros_like(h)
    current_t = 1.0

    max_iter_coarse = 5000
    for i in range(max_iter_coarse):
        if time.time() - start_time > max_time:
            break
        grad = compute_gradient(h, dx_coarse)
        h, velocity, c, current_t = fire_step(h, velocity, grad, dx_coarse, current_t)
        if c < best_C:
            best_C = c
            best_h = h.copy()

    # Phase 2: Refine to fine resolution
    n_points_fine = 60
    dx_fine = 2.0 / n_points_fine
    # Upsample from coarse to fine
    x_coarse = np.linspace(0, 2, n_points_coarse)
    x_fine = np.linspace(0, 2, n_points_fine)
    f = interp1d(x_coarse, best_h, kind='linear')
    h_fine = f(x_fine)
    h_fine = project(h_fine, dx_fine)
    best_h = h_fine
    best_C = evaluate(h_fine, dx_fine)
    velocity = np.zeros_like(h_fine)
    current_t = 1.0

    max_iter_fine = 5000
    for i in range(max_iter_fine):
        if time.time() - start_time > max_time:
            break
        grad = compute_gradient(h_fine, dx_fine)
        h_fine, velocity, c, current_t = fire_step(h_fine, velocity, grad, dx_fine, current_t)
        if c < best_C:
            best_C = c
            best_h = h_fine.copy()
    
    return (best_h, best_C, len(best_h))
