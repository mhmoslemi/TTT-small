import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Convex optimization approach
    def solve_convex():
        # Set up the convex optimization problem
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = []
        
        # Bounds
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints: sum h[i] * (1 - h[i + k]) <= t for all k
        for k in range(n_points):
            # Compute the overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    overlap += h_vars[i] * (1 - h_vars[j])
            constraints.append(overlap <= t)
        
        # Formulate and solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_opt = h_vars.value
                # Project to feasible region
                h_opt = np.clip(h_opt, 0, 1)
                current_sum = np.sum(h_opt)
                if current_sum > 0:
                    h_opt = h_opt * (n_points / 2) / current_sum
                return h_opt
        except:
            pass
        return h.copy()
    
    # Local search refinement
    def refine_solution(h_vec):
        h_vec = np.array(h_vec)
        dx = 2.0 / len(h_vec)
        
        def compute_C5(h_vec):
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return np.max(corr) * dx
        
        # Simple gradient descent with small steps
        step_size = 0.01
        for _ in range(100):
            current_C5 = compute_C5(h_vec)
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Estimate gradient by finite difference  
            grad = np.zeros_like(h_vec)
            eps = 1e-4
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                current_sum = np.sum(h_plus)
                if current_sum > 0:
                    h_plus = h_plus * (n_points / 2) / current_sum
                current_sum = np.sum(h_minus)
                if current_sum > 0:
                    h_minus = h_minus * (n_points / 2) / current_sum
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
            
            h_vec = h_vec - step_size * grad
            h_vec = np.clip(h_vec, 0, 1)
            current_sum = np.sum(h_vec)
            if current_sum > 0:
                h_vec = h_vec * (n_points / 2) / current_sum
            step_size *= 0.99
            
            # Early stopping if improvement is small
            new_C5 = compute_C5(h_vec)
            if new_C5 >= current_C5:
                break
        
        return h_vec
    
    # Main loop for improvements
    max_iter = 1000
    patience = 0
    max_patience = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Solve convex problem
        h_new = solve_convex()
        
        # Refine with local search
        h_new = refine_solution(h_new)
        
        # Evaluate
        current_C5 = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb solution to escape local minima
                idx = np.random.choice(n_points)
                h_new[idx] = np.clip(h_new[idx] + np.random.normal(0, 0.02), 0, 1)
                h_new = np.clip(h_new, 0, 1)
                current_sum = np.sum(h_new)
                if current_sum > 0:
                    h_new = h_new * (n_points / 2) / current_sum
                patience = 0
        
        h = h_new
        
    return (best_h.tolist(), best_C5, n_points)
