import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial construction if available
    try:
        h = initial_h_values
        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create initial guess
        t = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * t)
    
    # Normalize to sum to 1
    h = h * (1.0 / (np.sum(h) * dx))
    
    # Project to feasible region
    h = np.clip(h, 0, 1)
    h = h * (1.0 / (np.sum(h) * dx))
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # If we're not running the LP solver, we'll perform a gradient-based approach
    # But due to time constraints, let's call the sparse optimization approach
    
    # Ensure that the gradient method proposes new candidates
    # We'll try an improved local search with more sophisticated update
    
    # Define objective function to minimize C5
    def objective(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Define constraints for projection
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (1.0 / (h_sum * dx))
        return h_vec

    best_h = h.copy()
    best_C = objective(best_h)
    
    # Simple gradient descent with projection
    lr = 0.005
    tolerance = 1e-6
    
    for i in range(20000):
        if time.time() - t0 > max_time:
            break
            
        if i % 1000 == 0:
            lr *= 0.99
            
        # Compute gradient estimate via finite difference
        grad = np.zeros_like(best_h)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus)
            
            h_minus = best_h.copy()
            h_minus[j] -= eps
            h_minus = project_to_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step with projection
        new_h = best_h - lr * grad
        new_h = project_to_feasible(new_h)
        
        # Check for improvement
        new_C = objective(new_h)
        if new_C < best_C:
            best_h = new_h
            best_C = new_C
            
        # Early stopping condition for local minima
        if np.abs(new_C - best_C) < tolerance:
            break

    return (best_h, best_C, n_points)
