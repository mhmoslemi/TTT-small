import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    max_velocity = 0.2
    
    # Initialize swarm
    swarm = []
    p_best = []
    p_best_scores = []
    
    try:
        # Use initial_h if available
        h = initial_h_values.copy()
        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create initial guess
        h = np.random.rand(n_points) * 0.5 + 0.25
    
    # Initialize particles
    for i in range(n_particles):
        if i == 0:
            particle_h = h.copy()
        elif i < n_particles // 10:  # 10% start with perturbed initial
            particle_h = h + np.random.normal(0, 0.02, n_points)
        else:
            particle_h = np.random.rand(n_points) * 0.5 + 0.25
            
        particle_h = project(particle_h, dx)
        swarm.append(particle_h.copy())
        score = evaluate(particle_h, dx)
        p_best.append(particle_h.copy())
        p_best_scores.append(score)
    
    # Global best
    best_idx = np.argmin(p_best_scores)
    g_best = p_best[best_idx].copy()
    g_best_score = p_best_scores[best_idx]
    
    # PSO main loop
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    for iter in range(100000):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (omega * velocities[i] + 
                           phi1 * r1 * (p_best[i] - swarm[i]) + 
                           phi2 * r2 * (g_best - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible set
            swarm[i] = project(swarm[i], dx)
            
            # Evaluate and update personal best
            curr_score = evaluate(swarm[i], dx)
            if curr_score < p_best_scores[i]:
                p_best_scores[i] = curr_score
                p_best[i] = swarm[i].copy()
                
                # Update global best
                if curr_score < g_best_score:
                    g_best_score = curr_score
                    g_best = swarm[i].copy()
    
    return (g_best, g_best_score, n_points)
