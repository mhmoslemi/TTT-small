import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values"""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Clip values to [0,1] and normalize to integral = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Evaluate the C5 constant for given h"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    n_points = 45
    dx = 2.0 / n_points
    
    # Get initial solution if available

    max_time = budget_s - 10  # Leave time for final processing
    start_time = time.time()
    
    # Initialize PSO parameters
    N = 10  # Number of Fourier coefficients (excluding DC)
    P = 50  # Number of particles
    max_iter = 1000
    
    # Initialize particles with random coefficients
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N + 1))  # a0, a1..aN, b1..bN
    velocities = np.random.uniform(-0.1, 0.1, (P, 2*N + 1))
    pbest = particles.copy()
    pbest_vals = np.full(P, np.inf)
    
    # Global best
    gbest = particles[0].copy()
    gbest_val = np.inf
    
    # PSO parameters
    omega = 0.7
    phi1 = phi2 = 1.5
    
    t = np.linspace(0, 2, n_points)
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        # Evaluate all particles
        for i in range(P):
            # Convert coefficients to h(x)
            a = particles[i, :N+1]
            b = particles[i, N+1:]
            h = fourier_to_h(a, b, t)
            h = clip_and_normalize(h, dx)
            
            # Evaluate C5
            C5 = evaluate(h, dx)
            
            # Update personal best
            if C5 < pbest_vals[i]:
                pbest_vals[i] = C5
                pbest[i] = particles[i].copy()
                
            # Update global best
            if C5 < gbest_val:
                gbest_val = C5
                gbest = particles[i].copy()
        
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            phi1 * r1 * (pbest[i] - particles[i]) + 
                            phi2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
    
    # Final evaluation of global best
    a = gbest[:N+1]
    b = gbest[N+1:]
    final_h = fourier_to_h(a, b, t)
    final_h = clip_and_normalize(final_h, dx)
    final_C5 = evaluate(final_h, dx)
    
    # Return the best solution found
    return (final_h, final_C5, n_points)
