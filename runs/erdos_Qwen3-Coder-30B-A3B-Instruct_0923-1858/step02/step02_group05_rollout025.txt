import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:

        if len(initial_h) != n_points:

    except NameError:
        t = np.linspace(0, 2, n_points)
        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)
        initial_h = project(initial_h, dx)

    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    v_max = 0.2
    pos_range = 0.1

    # Initialize particles
    particles = []
    velocities = []
    p_best_pos = []
    p_best_val = []
    
    for _ in range(n_particles):
        h = initial_h + np.random.uniform(-pos_range, pos_range, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        p_best_pos.append(h.copy())
        p_best_val.append(evaluate(h, dx))

    # Global best
    g_best_idx = np.argmin(p_best_val)
    g_best_pos = p_best_pos[g_best_idx].copy()
    g_best_val = p_best_val[g_best_idx]

    t0 = time.time()
    max_time = budget_s - 2
    
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best_pos[i] - particles[i]) + 
                           c2 * r2 * (g_best_pos - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_val[i]:
                p_best_val[i] = current_val
                p_best_pos[i] = particles[i].copy()
                
                # Update global best
                if current_val < g_best_val:
                    g_best_val = current_val
                    g_best_pos = particles[i].copy()

    return (g_best_pos, g_best_val, n_points)
