    def compute_gradient(h, dx):
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / (np.sum(h_plus) * dx)  # Renormalize

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / (np.sum(h_minus) * dx)  # Renormalize

            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')

            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
