import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    c1, c2 = 2.0, 2.0
    w = 0.7
    max_iter_pso = 20000
    max_iter_gd = 20000
    velocity_max = 0.2
    lr = 0.005
    
    # Initialize particles
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        t = np.linspace(0, 2, n_points)
        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)
    
    particles = []
    for i in range(n_particles):
        if i == 0:
            h = initial_h.copy()
        else:
            h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
    
    # PSO phase
    p_best = [p.copy() for p in particles]
    p_best_C = [evaluate(p, dx) for p in particles]
    g_best_idx = np.argmin(p_best_C)
    g_best = p_best[g_best_idx].copy()
    g_best_C = p_best_C[g_best_idx]
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    t0 = time.time()
    max_time_pso = budget_s * 0.7
    
    for iteration in range(max_iter_pso):
        if time.time() - t0 > max_time_pso:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_C = evaluate(particles[i], dx)
            
            # Update p_best
            if current_C < p_best_C[i]:
                p_best_C[i] = current_C
                p_best[i] = particles[i].copy()
                
                # Update g_best
                if current_C < g_best_C:
                    g_best_C = current_C
                    g_best = particles[i].copy()
    
    # Gradient descent phase
    current_h = g_best.copy()
    best_h = current_h.copy()
    best_C = g_best_C
    
    max_time_gd = budget_s * 0.3
    lr_decay = 0.98
    lr_decay_interval = 1000
    iter_gd = 0
    
    while iter_gd < max_iter_gd and time.time() - t0 < budget_s:
        if iter_gd % lr_decay_interval == 0 and iter_gd > 0:
            lr *= lr_decay
            
        # Compute numerical gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        current_h = current_h - lr * grad
        current_h = project(current_h, dx)
        
        # Check for improvement
        current_C = evaluate(current_h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
            
        iter_gd += 1
        
        if time.time() - t0 > budget_s:
            break
    
    # Final validation
    best_h = project(best_h, dx)
    final_C = evaluate(best_h, dx)
    
    return (best_h, final_C, n_points)
