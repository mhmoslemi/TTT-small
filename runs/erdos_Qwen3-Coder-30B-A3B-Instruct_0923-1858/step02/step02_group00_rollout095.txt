import time
from scipy.signal import correlate

        # Create an initial solution that's piecewise linear with a peak at 1

        x = np.linspace(0, 2, n_points)
        # Initialize with a triangular shape
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Temperature schedule
    beta = 1000.0
    gamma = 1e-4
    eta = 1e-5
    beta_decay = 0.99
    gamma_growth = 1.1
    decay_every = 50
    eta_decay = 0.5
    eta_decay_every = 100
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-8, 1 - 1e-8)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    iter_count = 0
    last_improvement = 0
    patience = 0
    max_patience = 100
    
    while elapsed() < budget_s * 0.98:
        iter_count += 1
        
        # Update temperature and barrier
        if iter_count % decay_every == 0:
            beta *= beta_decay
            gamma *= gamma_growth
            
        if iter_count % eta_decay_every == 0:
            eta *= eta_decay
        
        # Compute gradients with finite differences
        h_current = h.copy()
        corr = correlate(h_current, 1 - h_current, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)
        
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        # Update using gradient ascent (we're minimizing so we go opposite)
        h -= eta * grad
        
        # Apply barrier
        barrier_grad = np.zeros(n_points)
        for i in range(n_points):
            barrier_grad[i] = -gamma / h[i] + gamma / (1 - h[i])
        
        h -= eta * barrier_grad
        h = project_feasible(h)
        
        # Check for improvement
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:

            best_h = h.copy()
            last_improvement = iter_count
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Add noise to escape local minima
                noise_magnitude = 0.02
                h += np.random.normal(0, noise_magnitude, n_points)
                h = project_feasible(h)
                patience = 0
                
        # Occasionally do a full re-computation
        if iter_count % 50 == 0:
            h = project_feasible(h)
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
