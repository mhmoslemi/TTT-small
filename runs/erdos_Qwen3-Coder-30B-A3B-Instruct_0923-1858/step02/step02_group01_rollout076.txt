import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

        # Middle section: increasing then decreasing
        mid_start = 15

        initial_h[mid_start:mid_end] = np.linspace(0, 1, mid_end - mid_start)
        initial_h[mid_end:] = np.linspace(1, 0, n_points - mid_end)
        # Normalize
        initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx))
    
    def objective(h):
        return evaluate(h, dx)
        
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    best_h = initial_h.copy()
    best_C = evaluate(initial_h, dx)
    
    t0 = time.time()
    max_time = budget_s - 5
    
    # Run optimization with SLSQP method, multiple random restarts
    for restart in range(20):
        if time.time() - t0 > max_time:
            break
            
        # Start from slightly perturbed version of best solution, or initial
        if restart == 0:
            h0 = initial_h.copy()
        else:
            # Random perturbation of best solution
            noise = np.random.normal(0, 0.02, n_points)
            h0 = best_h + noise
            h0 = np.clip(h0, 0, 1)
            # Renormalize
            h0 = h0 * (1.0 / (np.sum(h0) * dx))
            
        try:
            res = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 500, 'ftol': 1e-8})
            if res.success:
                current_C = evaluate(res.x, dx)
                if current_C < best_C:
                    best_C = current_C
                    best_h = res.x.copy()
        except:
            continue
            
    return (best_h, best_C, n_points)
