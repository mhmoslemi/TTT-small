import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = initial_h / (np.sum(initial_h) * dx)
    
    start_time = time.time()
    max_time = budget_s - 10
    
    # Use a hybrid approach combining gradient descent with random restarts
    best_C = float('inf')
    best_h = initial_h.copy()
    
    # Define model parameters
    lr = 0.01
    patience = 0
    max_patience = 100
    
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h / (np.sum(h) * dx)
    
    # Local search using projected gradient descent
    for i in range(100000):
        if time.time() - start_time > max_time:
            break
            
        # Estimate gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / (np.sum(h_plus) * dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / (np.sum(h_minus) * dx)
            
            C_plus = np.max(correlate(h_plus, 1-h_plus, mode='full'))
            C_minus = np.max(correlate(h_minus, 1-h_minus, mode='full'))
            
            grad[j] = (C_plus - C_minus) / (2 * eps)
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / (np.sum(h_new) * dx)
        
        C_new = np.max(correlate(h_new, 1-h_new, mode='full'))
        
        if C_new < best_C:
            h = h_new
            best_C = C_new
            best_h = h.copy()
            patience = 0
            lr = min(0.05, lr * 1.02)
        else:
            h = h_new
            patience += 1
            if patience > max_patience:
                lr = max(1e-6, lr * 0.9)
                patience = 0
                
        # Periodic noise injection to escape local minima
        if i % 30 == 0 and np.random.rand() < 0.15:
            noise = np.random.normal(0, 0.005, len(h))
            h = np.clip(h + noise, 0, 1)
            h = h / (np.sum(h) * dx)
    
    # Make a final check with a smaller step
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final / (np.sum(h_final) * dx)
    final_C = np.max(correlate(h_final, 1-h_final, mode='full'))
    
    return (h_final, final_C, n_points)
