import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def init_segmented_h(n_points, m_segments=10):
    """Initialize a segmented h function."""
    h = np.zeros(n_points)
    segment_size = n_points // m_segments
    for i in range(m_segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < m_segments - 1 else n_points
        h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
    return h

def get_segments(h, n_points, m_segments=10):
    """Extract segment positions and heights."""
    segment_size = n_points // m_segments
    positions = np.linspace(0, 2, m_segments + 1)
    heights = []
    for i in range(m_segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < m_segments - 1 else n_points
        heights.append(np.mean(h[start:end]))
    return positions, heights

def update_segments(h, positions, heights, n_points, m_segments=10, dx=2.0/45):
    """Update h based on positions and heights."""
    new_h = np.zeros(n_points)
    segment_size = n_points // m_segments
    for i in range(m_segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < m_segments - 1 else n_points
        new_h[start:end] = heights[i]
    return new_h

    h_vals = np.array(heights)

    h_sum = np.sum(h_vals)
    if h_sum > 0:
        h_vals = h_vals * (1.0 / (h_sum * dx))
    return list(h_vals)

def position_gradient(h, dx, positions, heights, m_segments=10):
    """Estimate gradient for positions."""
    # Simplified approximation with finite differences
    eps = 1e-4
    grad = np.zeros_like(positions[:-1])  # without last point
    
    for i in range(len(positions)-1):
        # Shift position slightly
        pos_plus = positions.copy()
        pos_plus[i] += eps
        pos_minus = positions.copy() 
        pos_minus[i] -= eps

        h_plus = update_segments(h, pos_plus, heights, len(h), m_segments, dx)
        h_minus = update_segments(h, pos_minus, heights, len(h), m_segments, dx)
        
        # Evaluate
        corr_plus = correlate(h_plus, 1-h_plus, mode='full')
        corr_minus = correlate(h_minus, 1-h_minus, mode='full')
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    
    return grad
