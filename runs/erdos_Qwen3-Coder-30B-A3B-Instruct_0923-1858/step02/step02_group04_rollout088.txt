import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric triangular initial solution
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Use a genetic algorithm approach for better exploration
    def mutate(h_vec, rate=0.03):
        h_new = h_vec.copy()
        for i in range(len(h_vec)):
            if np.random.random() < rate:
                h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.02), 0, 1)
        return project_feasible(h_new)
    
    def crossover(h1, h2):
        # Single point crossover
        point = np.random.randint(1, len(h1)-1)
        h_new = np.concatenate([h1[:point], h2[point:]])
        return project_feasible(h_new)
    
    # Population-based optimization
    population_size = 20
    population = [project_feasible(h + np.random.normal(0, 0.01, n_points)) for _ in range(population_size)]
    fitness = [compute_C5(ind) for ind in population]
    
    # Initial elite
    elite_idx = np.argmin(fitness)
    best_h = population[elite_idx].copy()
    best_C5 = fitness[elite_idx]
    
    iter_count = 0
    patience = 0
    max_patience = 100
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Selection based on fitness (tournament selection)
        tournament_size = 3
        new_population = []
        
        # Keep the best individual
        new_population.append(population[elite_idx].copy())
        
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Mutation
            mutated = mutate(population[winner_idx])
            new_population.append(mutated)
        
        # Generate offspring through crossover
        for i in range(0, len(new_population)-1, 2):
            new_population.append(crossover(new_population[i], new_population[i+1]))
        
        # Trim population
        new_population = new_population[:population_size]
        
        # Evaluate fitness
        fitness = [compute_C5(ind) for ind in new_population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C5:
            best_C5 = fitness[min_fitness_idx]
            best_h = new_population[min_fitness_idx].copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Add random individual to break local optimum
                new_population[-1] = project_feasible(h + np.random.normal(0, 0.03, n_points))
                fitness[-1] = compute_C5(new_population[-1])
                patience = 0
        
        population = new_population
        elite_idx = np.argmin(fitness)
        
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
