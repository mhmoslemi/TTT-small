import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Temperature and barrier parameters
    beta = 1000.0
    gamma = 1e-4
    eta = 1e-5
    beta_decay = 0.99
    gamma_growth = 1.1
    eta_decay = 0.5
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_gradient(h_vec, beta_val, gamma_val):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Compute the exponential penalty
        exp_corr = np.exp(-beta_val * corr)
        
        # Compute barrier gradient
        barrier_grad = -gamma_val * (1/h_vec + 1/(1-h_vec))
        
        # We need to compute the gradient of the exponential penalty
        # Approximate with finite differences for simplicity
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            # Recompute correlation
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (exp_corr[max_k_idx] - exp_corr[max_k_idx]) / (2 * eps)  # Same correlation due to symmetry
            
        # Use numerical derivative of exp(-beta*corr)
        grad_total = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad_total[i] = (np.exp(-beta_val * corr_plus[max_k_idx]) - np.exp(-beta_val * corr_minus[max_k_idx])) / (2 * eps)
            
        return grad_total + barrier_grad

    iter_count = 0
    last_improvement = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count
            
        # Update parameters
        if iter_count % 50 == 0 and iter_count > 0:
            beta *= beta_decay
            gamma *= gamma_growth
            if iter_count % 100 == 0:
                eta *= eta_decay
                
        # Compute gradient
        grad = compute_gradient(h, beta, gamma)
        
        # Update h
        h = h - eta * grad
        h = project_feasible(h)
        
        # Occasionally project onto sum constraint more explicitly
        if iter_count % 50 == 0:
            h = project_feasible(h)
            
        # Check for stagnation
        if iter_count - last_improvement > 1000:
            # Perturb a bit
            idxs = np.random.choice(n_points, size=n_points//10, replace=False)
            for i in idxs:
                h[i] += np.random.normal(0, 0.01)
            h = project_feasible(h)
            last_improvement = iter_count
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
