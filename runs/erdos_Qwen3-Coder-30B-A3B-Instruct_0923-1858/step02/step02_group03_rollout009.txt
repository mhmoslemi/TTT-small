import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def fourier_to_h(a, b, t):
    # a[0] is the constant term, a[1:], b[1:] are cosine/sine coefficients
    h = a[0] * np.ones_like(t)
    for i in range(1, len(a)):
        h += a[i] * np.cos(i * np.pi * t) + b[i] * np.sin(i * np.pi * t)
    return h

    # Use Fourier representation with PSO
    N = 10  # Number of Fourier coefficients (including constant)
    P = 50  # Number of particles
    max_iter = 1000
    
    # Initialize particles (coefficients: a, b)
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N))
    velocities = np.zeros((P, 2*N))
    pbest = particles.copy()
    pbest_scores = np.full(P, np.inf)
    
    # Global best
    gbest = None
    gbest_score = np.inf
    
    t = np.linspace(0, 2, n_points)
    
    # PSO parameters
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    
    start_time = time.time()
    max_time = budget_s - 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(P):
            # Convert coefficients to h(x)
            a = particles[i, :N]
            b = particles[i, N:]
            
            h = fourier_to_h(a, b, t)
            h = project(h, dx)
            
            # Evaluate score
            score = evaluate(h, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
            
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest = particles[i].copy()
        
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            phi1 * r1 * (pbest[i] - particles[i]) +
                            phi2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
    
    # Final evaluation with the best found coefficients
    if gbest is not None:
        a = gbest[:N]
        b = gbest[N:]
        final_h = fourier_to_h(a, b, t)
        final_h = project(final_h, dx)
        final_score = evaluate(final_h, dx)
