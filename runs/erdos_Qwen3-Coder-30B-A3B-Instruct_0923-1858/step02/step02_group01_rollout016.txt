import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_kernel(sigma):
    kernel_size = 2 * sigma + 1
    kernel = np.zeros(kernel_size)
    for i in range(-sigma, sigma+1):
        val = 0.54 + 0.46 * np.cos(np.pi * i / sigma)
        kernel[i + sigma] = val
    kernel /= kernel.sum()
    return kernel

def compute_learning_rate(gradient, iter):
    base_lr = 0.02
    decay = 0.95
    return base_lr * (decay ** (iter // 500))
