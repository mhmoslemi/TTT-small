import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Mutate an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2, dx, crossover_prob=0.8):
    """Perform uniform crossover between two parents."""
    if np.random.rand() < crossover_prob:
        mask = np.random.rand(len(parent1)) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        child1 = project(child1, dx)
        child2 = project(child2, dx)
        return child1, child2
    else:
        return parent1.copy(), parent2.copy()

def tournament_selection(population, fitness, tournament_size=3):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitness[i] for i in selected_idx])]
    return population[best_idx].copy()

        initial_h[15:] = np.linspace(1, 0, 30)

        initial_h = project(initial_h, dx)
    
    # GA parameters
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8

    # Initialize population
    population = [initial_h.copy()]

    # Evaluate initial population
    fitness = [evaluate(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)

    # Tracking for stagnation
    last_improvement = 0

        sorted_indices = np.argsort(fitness)

        fitness = [fitness[i] for i in sorted_indices]

        if generation - last_improvement > stagnation_limit:

                mutation_rate = 0.2

        new_population = [population[0].copy()]  # Elitism

            new_population.extend([child1, child2])
            
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Evaluate new population
        new_fitness = [evaluate(ind, dx) for ind in new_population]
        
        # Update best solution
