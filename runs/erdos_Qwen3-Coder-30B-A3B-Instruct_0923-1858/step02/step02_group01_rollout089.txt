import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def crossover(parent1, parent2):
    """Uniform crossover between two parents."""
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, dx, mutation_rate=0.1):
    """Mutation operator."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def tournament_selection(population, fitness, tournament_size=3):
    """Tournament selection."""
    indices = np.random.choice(len(population), size=tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

    np.random.seed(seed)

        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = project(initial_h, dx)
    
    # Genetic Algorithm
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    max_generations = 10000

    no_improvement_count = 0
    best_C = float('inf')
    best_h = None
    t0 = time.time()
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):

        h = project(h, dx)
        population.append(h)

        if time.time() - t0 > budget_s - 10:

        fitness = [evaluate(h, dx) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C:
            best_C = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:

        # Check for stagnation
        if no_improvement_count > stagnation_limit:

                population = [initial_h.copy()]

        while len(new_population) < population_size:

            parent2 = tournament_selection(population, fitness)

            child1 = mutate(child1, dx, mutation_rate)
