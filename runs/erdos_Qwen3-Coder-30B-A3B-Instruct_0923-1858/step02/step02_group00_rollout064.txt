import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution (n=45)
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Different optimization approach: direct gradient descent with curvature
    def optimize_with_curvature():
        current_h = h.copy()
        best_h = h.copy()
        best_C5 = compute_C5(h)
        
        # Decreasing learning rates and momentum
        lr = 0.02
        momentum = 0.9
        v = np.zeros_like(h)
        
        iter_count = 0
        while elapsed() < budget_s * 0.95:
            iter_count += 1
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
            
            # Compute gradient via finite differences
            grad = np.zeros_like(current_h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = current_h.copy()
                h_minus = current_h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
            
            # Update with momentum
            v = momentum * v - lr * grad
            current_h = current_h + v
            current_h = project_feasible(current_h)
            
            # Adaptive learning rate decay
            if iter_count % 10 == 0:
                lr *= 0.98
            if lr < 1e-6:
                lr = 1e-6
                
        return best_h, best_C5

    # Additional local search with perturbations
    h_opt, C5 = optimize_with_curvature()
    
    # Further improvement through random localized refinements
    def localized_refinement(h_vec, num_iter=100):
        current_h = h_vec.copy()
        current_C5 = compute_C5(current_h)
        best_h = current_h.copy()
        best_C5 = current_C5
        
        for i in range(num_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            # Randomly perturb a small subset
            perturb_count = max(1, n_points // 20)
            indices = np.random.choice(n_points, size=perturb_count, replace=False)
            new_h = current_h.copy()
            for idx in indices:
                new_h[idx] += np.random.normal(0, 0.02)
            new_h = project_feasible(new_h)
            
            new_C5 = compute_C5(new_h)
            if new_C5 < current_C5:
                current_h = new_h
                current_C5 = new_C5
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = current_h.copy()
                    
        return best_h, best_C5
    
    if elapsed() < budget_s * 0.95:
        ref_h, ref_C5 = localized_refinement(h_opt)
        if ref_C5 < C5:
            h_opt = ref_h
            C5 = ref_C5
    
    return (h_opt.tolist(), C5, n_points)
