import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

        if len(h) != n_points:

    except NameError:

        h = np.full(n_points, 0.5)
        
    # Genetic Algorithm parameters
    N = 50  # Population size
    generations = 10000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    
    # Segment-based representation
    segments = 10
    segment_size = n_points // segments
    if n_points % segments != 0:
        segment_size = n_points // segments + 1
    
    # Initialize population
    population = []
    for _ in range(N):
        if 'initial_h_values' in globals() and np.random.rand() < 0.1:
            # Seed with perturbed initial solution
            h_temp = initial_h_values.copy()
            h_temp += np.random.normal(0, 0.05, n_points)
        else:
            h_temp = np.random.rand(n_points)
        h_temp = project(h_temp, dx)
        population.append(h_temp)
    
    # Track best solution
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    # For adaptive mutation rate
    no_improvement_count = 0
    improvement_threshold = 100
    
    for gen in range(generations):
        if time.time() - t0 > max_time:
            break
            
        # Evaluate population
        fitnesses = []
        for h_temp in population:
            fitness = -evaluate(h_temp, dx)  # Negative because we minimize
            fitnesses.append(fitness)
        
        # Tournament selection
        new_population = []
        for _ in range(N):
            # Select tournament participants
            tournament_indices = np.random.choice(N, tournament_size, replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
            
            # Clone the winner
            parent = population[winner_idx].copy()
            
            # Apply crossover if chosen
            if np.random.rand() < crossover_rate:
                # Select another parent for crossover
                parent2_idx = np.random.choice(N, 1, replace=False)[0]
                parent2 = population[parent2_idx].copy()
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                # Mutate in segments, pick a random segment
                segment_idx = np.random.randint(0, segments)
                segment_start = segment_idx * segment_size
                segment_end = min((segment_idx + 1) * segment_size, n_points)
                for i in range(segment_start, segment_end):
                    if np.random.rand() < 0.1:  # 10% chance of mutation per element
                        child[i] += np.random.normal(0, 0.1)
            
            # Project to constraints
            child = project(child, dx)
            new_population.append(child)
        
        population = new_population
        
        # Check for best solution in current generation
        for h_temp in population:
            current_C = evaluate(h_temp, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = h_temp.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Adaptive mutation rate 
        if no_improvement_count > improvement_threshold:
            mutation_rate = min(0.5, mutation_rate + 0.005)
        else:
            mutation_rate = max(0.01, mutation_rate - 0.005)
    
    return (best_h, best_C, n_points)
