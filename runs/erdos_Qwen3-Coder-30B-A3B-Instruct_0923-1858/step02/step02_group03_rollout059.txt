import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values"""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Ensure h is in [0,1] and integrates to 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
