import time
from scipy.signal import correlate

        for i in range(n_points):
            h_init[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convex optimization approach
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1]
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Max overlap constraints
    dx = 2.0 / n_points
    for k in range(n_points):
        # This is a bit tricky to compute exactly, but for convex optimization
        # we linearize the constraint using the fact that we're looking at 
        # correlation partial terms.
        # For each shift k, we approximate linear constraints
        corr = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                corr += h[i] * (1 - h[j])
        constraints.append(corr <= t)
        
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        h_sol = h.value
    except:
        # Fallback to initial solution
        h_sol = h_init.copy()
    
    # Post-processing: ensure constraints
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
    
    # Final evaluation
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    final_C5 = compute_C5(h_sol)
    
    # Return best solution found
    return (h_sol.tolist(), final_C5, n_points)
