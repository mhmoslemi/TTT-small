import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        # Create a piecewise constant structure
        segments = 3
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)

    # PSO parameters
    n_particles = 50
    max_iter = 12000
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    max_velocity = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    # Initialize with existing solution or random
    for i in range(n_particles):
        if i == 0:
            # Use the initial solution for first particle
            particle_h = h.copy()
        else:
            # Initialize particle with random noise
            particle_h = np.random.rand(n_points) * 0.5 + 0.25
            particle_h = project(particle_h, dx)
            
        particles.append(particle_h)
        velocities.append(np.zeros(n_points))
        p_best.append(particle_h.copy())
        p_best_values.append(evaluate(particle_h, dx))
    
    # Find global best
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    # PSO main loop
    t0 = time.time()
    for iteration in range(max_iter):

            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           phi1 * r1 * (p_best[i] - particles[i]) + 
                           phi2 * r2 * (g_best - particles[i]))
            
            # Clamp velocities
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_value = evaluate(particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value

                if current_value < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_value
    
    # Return best solution found
    return (g_best, g_best_value, n_points)
