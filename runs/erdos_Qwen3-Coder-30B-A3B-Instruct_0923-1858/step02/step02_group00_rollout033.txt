    best_h = h.copy()

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Alternative: better gradient approximation
    def get_gradient(h_vec, num_samples=20):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Sample a small subset of shifts for gradient estimation
        shift_indices = np.random.choice(len(h_vec)*2 - 1, size=min(num_samples, len(h_vec)*2 - 1), replace=False)
        shift_indices = np.sort(shift_indices)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            # Use only relevant shift for this update
            if max_k_idx in shift_indices:
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            else:
                # Estimate gradient using the most relevant shift
                # Naively estimate gradient from shifts that show large overlap
                grad[i] = 0.0
                
        return grad

    # Adam-like update with adaptive step size 
    def adam_update(h_vec, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = h_vec - step
        return h_new, m, v

                patience = 0

        grad = get_gradient(h)
        t += 1
        h, m, v = adam_update(h, grad, m, v, t, lr=eta)

        # Periodic reevaluation and learning rate decay
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1

    # Final evaluation with reflection
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
