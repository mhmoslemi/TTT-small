        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

        # Implicitly we want to minimize max_k (h ⋆ (1-h))[k]
        
        # This approach uses a different formulation to find a good starting point
        # Formulate a problem that encourages low overlap
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        num_shifts = min(30, n_points)

        obj = cp.Minimize(cp.sum_squares(h_var))

            if prob.status == cp.OPTIMAL:
                return np.maximum(0, np.minimum(1, h_var.value))

        # Apply gradient ascent with adaptive step size
        step_size = 1e-4 * beta
        h_new = h - step_size * grad
        h_new = project_feasible(h_new)
        
        # Optimization improvement test
        new_C5 = compute_C5(h_new)

                best_h = h.copy()
            # Add slight random noise
            h += np.random.normal(0, 1e-5, n_points) 
            h = project_feasible(h)

    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
