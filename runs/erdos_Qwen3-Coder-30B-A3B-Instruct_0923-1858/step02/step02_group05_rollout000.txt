import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        h = project(h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Create initial particles with small random perturbations from h
    for _ in range(n_particles):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h_p = h + perturbation
        h_p = project(h_p, dx)
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Track personal bests
    p_best = [p.copy() for p in particles]
    p_best_values = [evaluate(p, dx) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(p_best_values)
    g_best = p_best[global_best_idx].copy()
    g_best_value = p_best_values[global_best_idx]
    
    # PSO loop
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) +
                           c2 * r2 * (g_best - particles[i]))
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            new_pos = particles[i] + velocities[i]
            
            # Project to feasible region
            new_pos = project(new_pos, dx)
            
            # Evaluate new position
            new_value = evaluate(new_pos, dx)
            
            # Update personal best
            if new_value < p_best_values[i]:
                p_best[i] = new_pos.copy()
                p_best_values[i] = new_value
                
                # Update global best if needed
                if new_value < g_best_value:
                    g_best = new_pos.copy()
                    g_best_value = new_value
            
            # Update particle position
            particles[i] = new_pos.copy()
    
    return (g_best, g_best_value, n_points)
