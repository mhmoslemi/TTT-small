import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    max_iter = 12000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_velocity = 0.2
    
    # Initialize swarm
    swarm = []

    except NameError:

    for i in range(n_particles):
        if i == 0:
            # Start with the provided solution or a random one
            h = h if 'h' in locals() and len(h) == n_points else np.random.rand(n_points) * 0.5
        else:
            # Perturb slightly
            h = np.random.rand(n_points) * 0.5
        h = project(h, dx)
        swarm.append(h.copy())
        p_best.append(h.copy())
        p_best_values.append(evaluate(h, dx))
    
    # Global best
    best_idx = np.argmin(p_best_values)
    g_best = p_best[best_idx].copy()
    g_best_value = p_best_values[best_idx]
    
    # Velocity initialization
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    t0 = time.time()
    max_time = budget_s - 10
    
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (p_best[i] - swarm[i]) + 
                            social * r2 * (g_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project(swarm[i], dx)
            
            # Evaluate new solution
            current_value = evaluate(swarm[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = swarm[i].copy()
                p_best_values[i] = current_value
                
                # Update global best
                if current_value < g_best_value:
                    g_best = swarm[i].copy()
                    g_best_value = current_value
    
    return (g_best, g_best_value, n_points)
