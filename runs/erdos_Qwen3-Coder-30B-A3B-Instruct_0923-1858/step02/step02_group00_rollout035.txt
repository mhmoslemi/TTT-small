import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()

    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_subproblem(shift_k):
        # Solve a linear program to find good adjustments for a specific shift
        # We use a simplified model: try to reduce the overlap at shift k
        h_var = cp.Variable(n_points)
        
        # Objective: minimize the contribution of overlap at shift k
        # This is a heuristic: try to make h[i] such that h[i]*h[i+shift_k] is minimized
        # for i in range(n_points - abs(shift_k))
        # The overlap term: int h(x) * (1 - h(x + k)) dx ~ sum h[i] * (1 - h[i+k])
        
        # Simplified approach: try to create a structurally better h
        # For now, just find a minimal "bad" overlap and target it
        # This is still a subproblem that we will solve heuristically
        
        # For shift k, we focus on sum_{i: i+k in bounds} h[i] * (1 - h[i+k]) = h[i] - h[i]*h[i+k]
        # The goal is to minimize this over all such pairings
        
        # We will solve a simpler LP that attempts to create h that makes this minimized
        # But more directly:
        # The constraints are just h in [0,1] and sum = n_points/2,
        # The objective will be to reduce single largest overlap term
        
        # That's what we want to get:
        # In simpler really approach:
        # We solve a simple linear program just to project the answer (as regular convex constraint)
        # (or do a line search)
        
        # For heuristic, try immediate augmentation attempt
        return h_var.value if h_var.value is not None else h.copy()

    def identify_critical_shifts(h_vec, num=10):
        """Identify the shifts that create the highest overlaps"""
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        shifts = np.arange(-len(h_vec)+1, len(h_vec))
        # Get indices of largest correlations
        idxs = np.argsort(corr)[::-1][:num]
        critical_shifts = shifts[idxs]
        return critical_shifts

    # Use more targeted gradient descent to avoid local minima
    def stochastic_descent(h_vec, eta=0.01, n_iter=100):
        h_current = h_vec.copy()
        for i in range(n_iter):
            if elapsed() > budget_s * 0.95:
                break
            # Find the index of maximum correlation
            corr = correlate(h_current, 1 - h_current, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_current) - 1)

            # Estimate finite difference gradient
            grad = np.zeros_like(h_current)
            eps = 1e-4
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += eps
                h_minus[j] -= eps

                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)

                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")

                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)

            # Update
            h_current = h_current - eta * grad
            h_current = project_feasible(h_current)
            
            # Adaptive step-size based on improvement
            if i % 10 == 0:
                eta *= 0.995
                
        return h_current

    # Heavy augmenting around critical shifts
    def iterative_surrounding_adjustment(h_vec, n_cycles=5, alpha=0.02):
        h_current = h_vec.copy()
        for cycle in range(n_cycles):
            if elapsed() > budget_s * 0.95:
                break
                
            # Re-evaluate currents critical shifts every cycle
            critical_shifts = identify_critical_shifts(h_current)
            for k in critical_shifts[:5]:  # Top few
                # For shift k, this contributes to overlap at 
                # sum over valid i where h[i] * (1 - h[i+k]) is considered
                # Perturb a lot of locations to lower that term
                if k >= 0:
                    # h[i] * (1 - h[i+k]) 
                    for i in range(len(h_current) - k):
                        if h_current[i] > 0 and h_current[i + k] < 1:
                            h_current[i] += alpha * (1 - h_current[i+k])
                            h_current[i+k] -= alpha * h_current[i]
                else:
                    k = -k
                    for i in range(len(h_current) - k):
                        if h_current[i] > 0 and h_current[i + k] < 1:
                            h_current[i] += alpha * (1 - h_current[i+k])
                            h_current[i+k] -= alpha * h_current[i]
                h_current = project_feasible(h_current)

        return h_current

    # Main optimization loop
    iter_count = 0
    patience = 0
    max_patience = 100
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # Try basic descent update
        if iter_count % 5 == 0:
            h = stochastic_descent(h, eta=0.01, n_iter=50)
        h = project_feasible(h)
            
        # Try targeted adjustments
        h = iterative_surrounding_adjustment(h, n_cycles=1, alpha=0.01)
        h = project_feasible(h)
        
        # Occasional random jump
        if iter_count % 20 == 0 and np.random.rand() < 0.15:
            # Make small random adjustments
            idxs = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in idxs:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)

        # Smart pruning of exhaust
        if patience > max_patience:
            # Adjust to diverse choices
            h = project_feasible(np.random.rand(n_points) * 0.5 + 0.25)
            patience = 0

        iter_count += 1

    # Final cleanup
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
