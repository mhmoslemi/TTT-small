import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values"""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Ensure h is in [0,1] and integrates to 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    N = 10  # Number of Fourier coefficients
    P = 50  # Number of particles
    max_iter = 1000
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        initial_h = np.ones(n_points) * 0.5
    
    # Create time points
    t = np.linspace(0, 2, n_points)
    
    # Use initial h as a reference
    if len(initial_h) == n_points:
        ref_h = initial_h
    else:
        ref_h = np.ones(n_points) * 0.5
    
    # Initialize particles
    particles = []
    for i in range(P):
        a = np.random.uniform(-0.5, 0.5, N)
        b = np.random.uniform(-0.5, 0.5, N)
        particles.append((a, b))
    
    # Particle velocities
    velocities = [(np.zeros(N), np.zeros(N)) for _ in range(P)]
    
    # Initialize bests
    best_particle = None
    best_C = float('inf')
    
    # PSO parameters
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        for i, (a, b) in enumerate(particles):
            # Compute function values
            h = fourier_to_h(a, b, t)
            h = clip_and_normalize(h, dx)
            
            # Evaluate current particle
            C = evaluate(h, dx)
            
            # Update personal best
            if C < best_C:
                best_C = C
                best_particle = (a.copy(), b.copy())
            
            # Update velocity and position for this particle
            if best_particle is not None:
                a_best, b_best = best_particle
                r1, r2 = np.random.rand(N), np.random.rand(N)
                for n in range(N):
                    velocities[i][0][n] = (
                        omega * velocities[i][0][n] +
                        phi1 * r1[n] * (a_best[n] - a[n]) +
                        phi2 * r2[n] * (a_best[n] - a[n])
                    )
                    velocities[i][1][n] = (
                        omega * velocities[i][1][n] +
                        phi1 * r1[n] * (b_best[n] - b[n]) +
                        phi2 * r2[n] * (b_best[n] - b[n])
                    )
                    
                    a[n] += velocities[i][0][n]
                    b[n] += velocities[i][1][n]
    
    # Extract best solution
    if best_particle is not None:
        a_best, b_best = best_particle
        best_h = fourier_to_h(a_best, b_best, t)
        best_h = clip_and_normalize(best_h, dx)
        best_C = evaluate(best_h, dx)
    else:
        best_h = ref_h
        best_C = evaluate(best_h, dx)
    
    return (best_h, best_C, n_points)
