import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    start_time = time.time()
    max_time = budget_s - 10
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Use CVXPY for robust optimization
    for iteration in range(1000):
        if time.time() - start_time > max_time:
            break
            
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) * dx == 1,
            t >= 0
        ]
        
        # Add constraints for all shifts
        max_shift = n_points - 1
        for k in range(max_shift + 1):
            # Build the correlation constraint efficiently
            shifted_h = np.roll(h_var, k)
            # Compute the inner product manually for constraints
            inner_product = cp.sum(cp.multiply(h_var, shifted_h))
            # Correlation is 1 - inner_product
            # We want 1 - inner_product <= t
            constraints.append(1 - inner_product <= t)
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                new_h = h_var.value
                new_h = project(new_h, dx)
                current_C = evaluate(new_h, dx)
                if current_C < best_C:
                    best_h = new_h.copy()
                    best_C = current_C
                    h = new_h.copy()
        except:
            # Fallback to gradient-based method if optimization fails
            pass
        
        # Local search fallback
        lr = 0.01
        if iteration % 10 == 0:
            grad = np.zeros_like(h)
            eps = 1e-5
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = project(h_minus, dx)
                
                corr_plus = correlate(h_plus, 1-h_plus, mode='full')
                corr_minus = correlate(h_minus, 1-h_minus, mode='full')
                
                grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
            h_new = h - lr * grad
            h_new = project(h_new, dx)
            current_C = evaluate(h_new, dx)
            if current_C < best_C:
                h = h_new
                best_C = current_C
                best_h = h.copy()
    
    return (best_h, best_C, n_points)
