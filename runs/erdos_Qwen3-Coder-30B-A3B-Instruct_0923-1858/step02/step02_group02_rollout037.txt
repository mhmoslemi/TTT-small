import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def compute_segmentation(h, n_segments=10):
    """Compute segment positions and heights from piecewise constant h."""
    n_points = len(h)
    segment_size = n_points // n_segments
    positions = [0]
    heights = []
    for i in range(n_segments):
        start = i * segment_size
        end = (i+1) * segment_size if i < n_segments - 1 else n_points
        avg_h = np.mean(h[start:end])
        heights.append(avg_h)
        positions.append(end)
    return np.array(positions) * 2.0 / n_points, np.array(heights)

def reconstruct_h(positions, heights, n_points):
    """Reconstruct h from segment positions and heights."""
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i, (pos_start, pos_end) in enumerate(zip(positions[:-1], positions[1:])):
        start_idx = int(pos_start / dx)
        end_idx = int(pos_end / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = heights[i]
    return h

def optimize_heights(positions, dx, heights, n_points):
    """Optimize heights given fixed positions using convex relaxation."""
    # Create a convex optimization problem for minimizing max correlation

    # obtained from the current segment setup through DFT
    try:
        h = reconstruct_h(positions, heights, n_points)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    except:
        return np.inf
