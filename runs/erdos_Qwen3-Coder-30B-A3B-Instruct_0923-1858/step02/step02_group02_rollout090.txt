import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        # Create a piecewise constant structure
        segments = 3
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)

    # Simulated Annealing parameters
    T0 = 100.0
    alpha = 0.999
    max_iter = 10000
    T = T0
    current_h = h.copy()
    current_C = evaluate(current_h, dx)
    best_h = current_h.copy()
    best_C = current_C
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    for i in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        # Perturb current solution
        proposal_h = current_h.copy()
        num_perturbed = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, num_perturbed, replace=False)
        for idx in indices:
            proposal_h[idx] += np.random.normal(0, 0.1)
        
        # Project proposal to feasible region
        proposal_h = project(proposal_h, dx)
        
        # Evaluate proposal
        proposal_C = evaluate(proposal_h, dx)
        
        # Accept or reject
        delta_C = proposal_C - current_C
        if delta_C < 0 or np.random.rand() < np.exp(-delta_C / T):
            current_h = proposal_h
            current_C = proposal_C
            if current_C < best_C:
                best_C = current_C
                best_h = current_h.copy()
        
        # Update temperature
        T = T0 * (alpha ** i)
    
    # Final evaluation of best solution
    best_C = evaluate(best_h, dx)
    return (best_h, best_C, n_points)
