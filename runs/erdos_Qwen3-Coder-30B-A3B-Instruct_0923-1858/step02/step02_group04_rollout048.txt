import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.95
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 50
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_gradients(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    def line_search(h_vec, grad, max_iter=10):
        alpha = 1.0
        step_size = 0.01
        current_C5 = compute_C5(h_vec)
        
        for _ in range(max_iter):
            new_h = h_vec - alpha * grad
            new_h = project_feasible(new_h)
            new_C5 = compute_C5(new_h)
            
            if new_C5 < current_C5:
                return new_h, alpha
            alpha *= 0.5
            
        return h_vec, step_size
    
    # Multi-resolution optimization
    resolutions = [50, 75, 100, 125, 150, 175, 200]
    current_res_idx = 0
    
    def get_initial_h(res):
        try:
            h = np.array(initial_h_values)
            if len(h) == res:
                return h
        except NameError:
            pass
        
        h = np.zeros(res)
        x = np.linspace(0, 2, res)
        for i in range(res):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        h = h * (res / 2) / np.sum(h)
        return h
    
    # Start from largest resolution 
    n_points = 200
    h = get_initial_h(n_points)
    
    # Population-based search with mutations
    population_size = 10
    population = [project_feasible(h + np.random.normal(0, 0.01, n_points)) for _ in range(population_size)]
    
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        
        # Evaluate population
        fitness = [compute_C5(p) for p in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_C5:
            best_C5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Select top 5 for reproduction
        top_indices = np.argsort(fitness)[:5]
        top_individuals = [population[i] for i in top_indices]
        
        # Generate offspring via mutation
        new_population = []
        for i in range(population_size):
            parent = top_individuals[i % len(top_individuals)]
            
            # Crossover with random parent
            if np.random.rand() < 0.5:
                other_parent = top_individuals[np.random.randint(len(top_individuals))]
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent[:crossover_point], other_parent[crossover_point:]])
            else:
                child = parent.copy()
            
            # Add mutation
            if np.random.rand() < 0.8:
                mutation_idx = np.random.randint(0, n_points)
                child[mutation_idx] = np.clip(child[mutation_idx] + np.random.normal(0, 0.02), 0, 1)
            
            new_population.append(project_feasible(child))
        
        population = new_population
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count % 10 == 0:
            # Reoptimize with gradient steps
            current_best = population[best_idx]
            grad = compute_gradients(current_best)
            new_h, step_size = line_search(current_best, grad)
            population[0] = new_h
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
