import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    particles = []
    velocities = []
    p_best = []
    p_best_values = []

            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to valid range and reproject to sum constraint
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            fitness = evaluate(particles[i], dx)
            
            # Update personal best
            if fitness < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = fitness
                
                # Update global best
                if fitness < g_best_value:

                    g_best_value = fitness
                    
    return (g_best, g_best_value, n_points)
