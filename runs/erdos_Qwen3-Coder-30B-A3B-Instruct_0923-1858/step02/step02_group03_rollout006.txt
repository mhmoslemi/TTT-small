import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def fourier_to_h(a, b, t):
    # a: [a0, a1, ..., a_N]
    # b: [b1, ..., b_N]
    # t: array of points in [0, 2]
    N = len(a) - 1
    h = a[0] * np.ones_like(t)
    for n in range(1, N + 1):
        h += a[n] * np.cos(n * np.pi * t) + b[n-1] * np.sin(n * np.pi * t)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize using available construction if possible
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:

    except NameError:

        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    start_time = time.time()
    max_time = budget_s - 10
    
    # PSO parameters
    N = 10  # Number of Fourier coefficients (excluding a0)
    P = 50  # Number of particles
    max_iter = 1000
    omega = 0.7
    phi1 = phi2 = 1.5
    
    # Initialize particles (coefficients of Fourier expansion)
    particles = np.random.uniform(-0.5, 0.5, (P, 2 * N)) # [a1,...,aN, b1,...,bN]
    velocities = np.zeros_like(particles)
    
    # Personal best positions and fitness
    personal_best = particles.copy()
    personal_best_fitness = np.full(P, np.inf)
    
    # Global best position and fitness
    global_best_idx = 0
    global_best_fitness = np.inf
    
    # Create initial h function from particles for use with optimize
    h = project(initial_h, dx)
    current_C = evaluate(h, dx)
    best_C = current_C
    best_h = h.copy()
    
    # Convert initial to Fourier coefficients
    t = np.linspace(0, 2, n_points)
    
    # Start PSO iterations
    for iter_idx in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for p in range(P):
            # Get coefficients for particle p
            a_coeffs = np.zeros(N + 1)
            a_coeffs[1:] = particles[p, :N]
            b_coeffs = particles[p, N:]
            
            # Convert to h(x)
            current_h = fourier_to_h(a_coeffs, b_coeffs, t)
            current_h = project(current_h, dx)
            
            # Compute C_5
            current_C = evaluate(current_h, dx)
            
            # Update personal best
            if current_C < personal_best_fitness[p]:
                personal_best_fitness[p] = current_C
                personal_best[p] = particles[p].copy()
            
            # Update global best
            if current_C < global_best_fitness:
                global_best_fitness = current_C
                global_best_idx = p
                best_C = current_C
                best_h = current_h.copy()
            
            # PSO update
            r1, r2 = np.random.rand(2)
            velocities[p] = (omega * velocities[p] +
                             phi1 * r1 * (personal_best[p] - particles[p]) +
                             phi2 * r2 * (personal_best[global_best_idx] - particles[p]))
            
            particles[p] += velocities[p]
        
        # Time check inside loop
        if time.time() - start_time > max_time:
            break
    
    return (best_h, best_C, n_points)
