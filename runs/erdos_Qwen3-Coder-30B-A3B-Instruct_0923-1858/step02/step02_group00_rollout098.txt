import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Exponential smoothing with barrier method
    def exp_smooth_objective(h_vec, beta, gamma):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Exponential penalty
        penalty = np.sum(np.exp(-beta * corr))
        # Barrier penalty
        barrier = -gamma * np.sum(np.log(h_vec + 1e-10) + np.log(1 - h_vec + 1e-10))
        return penalty + barrier

    def compute_gradient(h_vec, beta, gamma, eps=1e-5):
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure feasibility
            h_plus = np.clip(h_plus, 1e-10, 1 - 1e-10)
            h_minus = np.clip(h_minus, 1e-10, 1 - 1e-10)
            
            obj_plus = exp_smooth_objective(h_plus, beta, gamma)
            obj_minus = exp_smooth_objective(h_minus, beta, gamma)
            
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad

    # Main optimization loop
    beta = 1000.0  # Temperature for exponential smoothing
    gamma = 1e-4   # Barrier strength
    eta = 1e-5     # Learning rate
    patience = 0
    max_patience = 100
    last_improvement = 0

    while elapsed() < budget_s * 0.95:
        # Current objective and C5
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = elapsed()

        # Optimization step
        grad = compute_gradient(h, beta, gamma)
        
        # Gradient descent update
        h = h - eta * grad
        
        # Project to feasible bounds
        h = np.clip(h, 0, 1)
        
        # Project to sum constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        
        # Update parameters
        if elapsed() > last_improvement + 60:  # If no improvement in 60s
            beta *= 0.99
            gamma *= 1.1
            eta *= 0.95
            last_improvement = elapsed()
        
        # Cooling schedule
        if int(elapsed() / 50) > int((elapsed() - 50) / 50):
            beta *= 0.99
            gamma *= 1.1
            
        # Decay learning rate
        if int(elapsed() / 100) > int((elapsed() - 100) / 100):
            eta *= 0.95
            
        # Early stopping
        if patience > max_patience:
            # Random restart
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            
        patience += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
