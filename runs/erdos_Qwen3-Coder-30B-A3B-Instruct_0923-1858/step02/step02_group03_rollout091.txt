import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_points = 45
    dx = 2.0 / n_points

    max_time = budget_s - 10
    start_time = time.time()
    
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    # Initialize with cutting-plane method
    h_current = initial_h.copy()
    t = cp.Variable()
    h_vars = cp.Variable(n_points)
    
    # Constraints for all shifts
    constraints = []
    
    # Create all possible shifts
    shifts = list(range(n_points))
    
    # Initial optimization
    for k in shifts:
        # For k = 0, we compute integral of h(x) * (1 - h(x+k)) over [0, 2]
        # This translates to, for discrete case:
        # S(k) = sum(h[i] * (1 - h[(i+k)%n_points]) * dx)
        # So S(k) = sum(h[i] * dx) - sum(h[i] * h[(i+k)%n_points] * dx)
        # = 1 - sum(h[i] * h[(i+k)%n_points] * dx)
        if k == 0:
            continue  # This is a special case; we'll compute it after
        
        # Add shifting constraints
        constraints.append(
            cp.sum(cp.multiply(h_vars, cp.roll(h_vars, k))) <= (1 - t)  # S(k) = 1 - integral(h[i]*h[i+k]) <= t -> integral(h[i]*h[i+k]) >= 1-t
        )
    
    # Add normalization constraint
    constraints.append(cp.sum(cp.multiply(h_vars, dx)) == 1)
    constraints.append(cp.sum(h_vars) == 1/dx)  # since sum(h[i]*dx) = 1
    constraints.append(h_vars >= 0)
    constraints.append(h_vars <= 1)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Optimization loop
    best_C = evaluate(h_current, dx)
    best_h = h_current.copy()
    
    for iteration in range(1000):
        if time.time() - start_time > max_time:
            break
            
        # Optimize with current constraints
        try:
            t_current = evaluate(h_current, dx)

            max_s_violation = -1
            best_shift = -1

                # Compute correlation

    final_C = evaluate(best_h, dx)
    
    return (best_h, final_C, n_points)
