import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        initial_h = project(initial_h, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        # Start from initial_h with small random perturbations
        h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())
        p_best_vals.append(evaluate(h, dx))
    
    # Global best
    global_best_idx = np.argmin(p_best_vals)
    g_best = p_best[global_best_idx].copy()
    g_best_val = p_best_vals[global_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best if needed
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
            
            # Check if we should continue the search or if we're stuck
            if iteration % 100 == 0 and iteration > 0:
                # Occasionally reinitialize particles to avoid stagnation
                if np.random.random() < 0.1:
                    for j in range(n_particles):
                        particles[j] = initial_h + np.random.uniform(-0.05, 0.05, n_points)
                        particles[j] = project(particles[j], dx)
    
    # Final evaluation with best solution
    best_h = g_best
    final_C = evaluate(best_h, dx)
