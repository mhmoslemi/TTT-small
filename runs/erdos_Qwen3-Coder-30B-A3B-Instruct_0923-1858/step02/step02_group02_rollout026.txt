import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    max_iter = 12000
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    max_velocity = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    
    try:

    except NameError:

    # Create initial swarm
    for i in range(n_particles):
        if i < n_particles * 0.1:  # 10% start from initial
            h = h_init.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
        else:
            h = np.random.uniform(0, 1, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Evaluate initial particles
    particle_best = [evaluate(p, dx) for p in particles]
    particle_best_h = [p.copy() for p in particles]
    
    # Find global best
    global_best_idx = np.argmin(particle_best)
    global_best = particle_best[global_best_idx]
    global_best_h = particle_best_h[global_best_idx].copy()
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave headroom
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           phi1 * r1 * (particle_best_h[i] - particles[i]) +
                           phi2 * r2 * (global_best_h - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_c = evaluate(particles[i], dx)
            
            # Update personal best
            if current_c < particle_best[i]:
                particle_best[i] = current_c
                particle_best_h[i] = particles[i].copy()
            
            # Update global best
            if current_c < global_best:
                global_best = current_c
                global_best_h = particles[i].copy()
    
    # Return best solution found
    return (global_best_h, global_best, n_points)
