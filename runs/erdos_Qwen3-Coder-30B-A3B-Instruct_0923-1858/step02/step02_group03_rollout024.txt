import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = initial_h_values
        n_points = len(h)
    except NameError:
        # Default construction
        n_points = 45
        t = np.linspace(0, 2, n_points)
        h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        h = h / (np.sum(h) * 2.0 / n_points)  # Normalize to sum = 1
    
    dx = 2.0 / n_points
    
    # Better starting point: refined version
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    
    start_time = time.time()
    elapsed = 0
    
    # Use a gradient-based method but start from a better initial guess
    def evaluate(h_vec, dx):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project(h_vec, dx):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (1.0 / (h_sum * dx))
        return h_vec
    
    def compute_gradient(h_vec, dx):
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_corr = correlate(h_vec, 1 - h_vec, mode='full')
        base_max = np.max(base_corr)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
        return grad
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    lr = 0.01
    patience = 0
    max_patience = 100
    
    while elapsed < budget_s - 2:
        current_time = time.time()
        elapsed = current_time - start_time
        
        grad = compute_gradient(h, dx)
        
        h_new = h - lr * grad
        h_new = project(h_new, dx)
        
        new_C = evaluate(h_new, dx)
        
        if new_C < best_C:
            h = h_new
            best_C = new_C
            best_h = h.copy()
            patience = 0
            lr = min(0.1, lr * 1.02)
        else:
            h = h_new
            patience += 1
            if patience > max_patience:
                lr = max(1e-5, lr * 0.9)
                patience = 0
                
        # Add noise to avoid local minima
        if elapsed % 10 < 1 and np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.01, len(h))
            h = project(h + noise, dx)
    
    # Return best solution
    return (best_h, best_C, n_points)
