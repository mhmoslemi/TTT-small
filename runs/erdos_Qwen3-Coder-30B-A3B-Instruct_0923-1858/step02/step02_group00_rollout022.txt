import time
from scipy.signal import correlate

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Exponential smoothing approach with temperature control
    beta = 1000.0  # Temperature parameter
    gamma = 1e-4   # Barrier strength
    eta = 1e-5     # Learning rate
    patience = 0
    max_patience = 100
    last_improvement = 0

    def compute_exp_penalty(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Normalize correlation to avoid overflow
        max_corr = np.max(corr)
        if max_corr > 0:
            corr = corr / max_corr
        penalty = np.sum(np.exp(-beta * corr))
        return penalty

    def compute_barrier(h_vec):
        # Ensure h stays in feasible region with log barrier
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)
        barrier = -gamma * np.sum(np.log(h_vec) + np.log(1 - h_vec))
        return barrier

    def compute_gradient(h_vec, eps=1e-6):
        h_vec = np.array(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            penalty_plus = compute_exp_penalty(h_plus)
            penalty_minus = compute_exp_penalty(h_minus)
            
            grad[i] = (penalty_plus - penalty_minus) / (2 * eps)
        
        # Add barrier gradient
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)
        barrier_grad = gamma * (1/h_vec - 1/(1-h_vec))
        grad = grad + barrier_grad
        return grad

    # Optimization loop
    iter_count = 0
    while elapsed() < budget_s * 0.95:
        # Compute current value
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_improvement = iter_count
        else:
            patience += 1

        # Temperature decay and barrier adjustment
        if iter_count % 50 == 0 and iter_count > 0:
            beta *= 0.99
            gamma *= 1.1
            
        # Update learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.5

        # Gradient step
        grad = compute_gradient(h)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Random perturbation if no improvement for too long
        if patience > max_patience:
            for i in range(n_points):
                if np.random.random() < 0.05:

            h = project_feasible(h)
            patience = 0
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
