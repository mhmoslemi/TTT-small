import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def fourier_to_h(a, b, t):
    # Convert Fourier coefficients to function values
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    # Clamp to [0,1] and normalize to integral = 1
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Try to use initial construction
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = clip_and_normalize(initial_h, dx)
    
    # This is an optimization problem over Fourier coefficients
    # Let's try differential evolution on the coefficient space
    max_time = budget_s - 10
    start_time = time.time()
    
    best_C = evaluate(initial_h, dx)
    best_h = initial_h.copy()
    
    # Number of Fourier coefficients to optimize
    N = 10  # Half the number since we'll do real and imaginary separately
    
    # Objective function that takes coefficient vector and returns C5
    def objective(x):
        # x = [a0, a1, ..., aN, b1, ..., bN]
        a = np.zeros(N + 1)
        b = np.zeros(N + 1)
        a[0] = x[0]
        a[1:N+1] = x[1:N+1]
        b[1:N+1] = x[N+1:2*N+1]
        
        t = np.linspace(0, 2, n_points)
        h = fourier_to_h(a, b, t)
        h = clip_and_normalize(h, dx)
        
        C = evaluate(h, dx)
        return C
    
    # Generate bounds for all coefficients
    # a0: [-1, 1], others: [-0.5, 0.5], b coefficients: [-0.5, 0.5]
    bounds = [(-1, 1)]  # a0
    bounds += [(-0.5, 0.5)] * N  # a1 to aN
    bounds += [(-0.5, 0.5)] * N  # b1 to bN
    
    # Run optimization
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=500, 
        popsize=15, 
        seed=seed,
        polling_interval=50,
        disp=False
    )
    
    if time.time() - start_time > max_time:
        return (best_h, best_C, n_points)
    
    # Extract best solution
    x_opt = result.x
    a = np.zeros(N + 1)
    b = np.zeros(N + 1)
    a[0] = x_opt[0]
    a[1:N+1] = x_opt[1:N+1]
    b[1:N+1] = x_opt[N+1:2*N+1]
    
    t = np.linspace(0, 2, n_points)
    h_opt = fourier_to_h(a, b, t)
    h_opt = clip_and_normalize(h_opt, dx)
    
    C_opt = evaluate(h_opt, dx)
    
    if C_opt < best_C:
        best_h = h_opt
        best_C = C_opt
    
    # Try fixed point iteration on the better solution to refine
    lr = 0.01
    patience = 0
    max_patience = 50
    
    for i in range(1000):
        if time.time() - start_time > max_time:
            break
            
        # Estimate gradient using finite differences
        grad = np.zeros(len(x_opt))
        eps = 1e-5
        
        for j in range(len(x_opt)):
            x_plus = x_opt.copy()
            x_plus[j] += eps
            x_minus = x_opt.copy()
            x_minus[j] -= eps
            
            corr_plus = objective(x_plus)
            corr_minus = objective(x_minus)
            grad[j] = (corr_plus - corr_minus) / (2 * eps)
        
        # Update step
        x_new = x_opt - lr * grad
        x_new = np.clip(x_new, [b[0] for b in bounds], [b[1] for b in bounds])
        
        h_new = fourier_to_h(
            np.concatenate([[x_new[0]], x_new[1:N+1]]), 
            np.concatenate([[0], x_new[N+1:2*N+1]]), 
            t)
        h_new = clip_and_normalize(h_new, dx)
        C_new = evaluate(h_new, dx)
        
        if C_new < best_C:
            best_h = h_new
            best_C = C_new
            x_opt = x_new.copy()
            patience = 0
            lr = min(0.05, lr * 1.05)
        else:
            x_opt = x_new
            patience += 1
            if patience > max_patience:
                lr = max(1e-5, lr * 0.95)
                patience = 0
    
    return (best_h, best_C, n_points)
