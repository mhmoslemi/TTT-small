import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Apply mutation to an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2, dx, crossover_prob=0.8):
    """Uniform crossover between two parents."""
    if np.random.rand() < crossover_prob:
        mask = np.random.rand(len(parent1)) > 0.5
        offspring1 = np.where(mask, parent1, parent2)
        offspring2 = np.where(mask, parent2, parent1)
        return project(offspring1, dx), project(offspring2, dx)
    else:
        return parent1.copy(), parent2.copy()

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialization parameters
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    tournament_size = 3
    stagnation_limit = 50
    reset_limit = 100
    
    # Generate initial population
    population = []
    
    # Try to use initial construction if available

        if len(initial_h) != n_points:

    except NameError:

        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        initial_h = project(initial_h, dx)
    
    population.append(initial_h)
    
    # Fill remaining population with random individuals
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points, dx))
    
    # Evaluate initial population
    fitnesses = [evaluate(individual, dx) for individual in population]
    
    best_fitness = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    stagnation_count = 0
    
    generation = 0
    while time.time() - t0 < max_time:
        generation += 1
        
        # Create new population
        new_population = [best_individual.copy()]  # Elitism
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            offspring1, offspring2 = crossover(parent1, parent2, dx, crossover_prob)
            
            offspring1 = mutate(offspring1, dx, mutation_rate)
            offspring2 = mutate(offspring2, dx, mutation_rate)
            
            new_population.extend([offspring1, offspring2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Evaluate new population
        new_fitnesses = [evaluate(individual, dx) for individual in new_population]
        
        # Update best individual
        current_best_idx = np.argmin(new_fitnesses)
        current_best_fitness = new_fitnesses[current_best_idx]
        
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness

            stagnation_count = 0
        else:

        # Adjust parameters if stagnation detected
        if stagnation_count > stagnation_limit:
            mutation_rate = 0.2
        if stagnation_count > reset_limit:
            # Reinitialize population
            population = [best_individual.copy()]
            for _ in range(population_size - 1):
                population.append(generate_individual(n_points, dx))
            stagnation_count = 0
            mutation_rate = 0.1
