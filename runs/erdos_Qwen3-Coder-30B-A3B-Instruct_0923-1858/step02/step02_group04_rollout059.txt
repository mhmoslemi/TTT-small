import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        # Create a more refined initial solution
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        # Create a piecewise linear function with smooth transitions
        for i in range(n_points):
            if x[i] < 0.5:
                h[i] = 2 * x[i]
            elif x[i] < 1.5:
                h[i] = 1
            else:
                h[i] = 2 - 2 * x[i]
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.99
    min_T = 1e-6
    mutation_step = 0.1
    
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def mutate(h_vec, step_size):
        h_new = h_vec.copy()
        # Select random points to perturb
        indices = np.random.choice(n_points, size=max(1, int(0.05 * n_points)), replace=False)
        for idx in indices:
            delta = np.random.uniform(-step_size, step_size)
            h_new[idx] += delta
        return project_feasible(h_new)
    
    # Main optimization loop
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Mutate the current solution
        h_new = mutate(h, mutation_step)
        
        # Compute C5 values
        current_C5 = compute_C5(h)
        new_C5 = compute_C5(h_new)
        
        # Accept or reject the new solution
        delta = new_C5 - current_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        
        # Update temperature and mutation step
        T *= cooling_rate
        mutation_step = 0.1 * (T / 1.0)  # Scale mutation by temperature
        
        if T < min_T:
            T = min_T
            
        iteration += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
