import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    T0 = 100.0
    alpha = 0.999
    max_iter = 10000
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    # Initialize
    h_current = project(h, dx)
    E_current = evaluate(h_current, dx)
    best_h = h_current.copy()
    best_E = E_current
    T = T0
    
    # Annealing loop
    for i in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        # Generate proposal by perturbing some elements
        h_proposal = h_current.copy()
        
        # Perturb 5% of the points
        num_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        for idx in indices:
            h_proposal[idx] += np.random.normal(0, 0.1)
            
        # Project to feasibility
        h_proposal = project(h_proposal, dx)
        
        # Evaluate proposal
        E_proposal = evaluate(h_proposal, dx)
        
        # Accept or reject
        delta_E = E_proposal - E_current
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h_current = h_proposal
            E_current = E_proposal
            
            if E_current < best_E:
                best_E = E_current
                best_h = h_current.copy()
        
        # Update temperature
        T = T * alpha

    return (best_h, best_E, n_points)
