import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    max_iter = 12000
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    max_velocity = 0.2
    
    # Initialize swarm
    particles = []

    # Initialize the swarm
    for i in range(n_particles):
        if i == 0:
            particle_h = h.copy()
        else:
            # Start with random perturbation of the best known
            particle_h = h + np.random.normal(0, 0.05, n_points)
        particle_h = project(particle_h, dx)
        particles.append(particle_h.copy())
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))
    
    # Track personal bests
    p_best = [p.copy() for p in particles]
    p_best_scores = [evaluate(p, dx) for p in particles]
    
    # Global best
    best_idx = np.argmin(p_best_scores)
    g_best = p_best[best_idx].copy()
    g_best_score = p_best_scores[best_idx]
    
    t0 = time.time()
    max_time = budget_s - 10
    
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (omega * velocities[i] + 
                            phi1 * r1 * (p_best[i] - particles[i]) + 
                            phi2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            score = evaluate(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = particles[i].copy()
                
                # Update global best
                if score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = score
    
    # Final evaluation and return
    final_score = evaluate(g_best, dx)
    
    return (g_best, final_score, n_points)
