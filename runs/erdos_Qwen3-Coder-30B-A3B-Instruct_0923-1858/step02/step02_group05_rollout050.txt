import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    for iteration in range(1000):

            r1, r2 = np.random.random(2)

            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)

            particles[i] = particles[i] + velocities[i]

            particles[i] = project(particles[i], dx)

            current_C = evaluate(particles[i], dx)
            
            # Update personal best
            if current_C < p_best_C[i]:
                p_best_C[i] = current_C
                p_best[i] = particles[i].copy()
                
            # Update global best
            if current_C < g_best_C:
                g_best_C = current_C
                g_best = particles[i].copy()
    
    return (g_best, g_best_C, n_points)
