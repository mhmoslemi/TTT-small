import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_kernel(sigma):
    """Create a symmetric Hamming window kernel."""
    kernel_size = 2 * sigma + 1
    kernel = np.zeros(kernel_size)
    for i in range(-sigma, sigma+1):
        val = 0.54 + 0.46 * np.cos(np.pi * i / sigma)
        kernel[i + sigma] = val
    kernel /= kernel.sum()
    return kernel

def compute_smooth_gradient(h, h_smooth, kernel, dx, eps=1e-5):
    """Compute gradient on smoothed approximation."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus_smooth = np.convolve(h_plus, kernel, mode='same')
        c_plus = evaluate(h_plus_smooth, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus_smooth = np.convolve(h_minus, kernel, mode='same')
        c_minus = evaluate(h_minus_smooth, dx)
        
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def compute_learning_rate(gradient, iter):
    """Adaptive learning rate based on gradient magnitude."""
    grad_norm = np.linalg.norm(gradient)
    base_lr = 0.02
    if grad_norm > 0:
        lr = base_lr / (1 + 0.001 * grad_norm)
    else:
        lr = base_lr
    # Decay schedule
    if iter % 500 == 0 and iter > 0:
        lr *= 0.95
    return max(lr, 1e-6)

def time_expired(budget_s, start_time):
    """Check if time budget has expired."""
    return time.time() - start_time > budget_s - 10

    h_refined = np.interp(x_new, x_old, h)
    return project(h_refined, 2.0 / n_new_points)
