import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx))

    def objective(h):
        return evaluate_erdos_solution(h, dx)

    bounds = [(0, 1) for _ in range(n_points)]

    t0 = time.time()
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,

    )
