import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def perturb(h, dx):
    """Perturb h with small random changes - maintains feasibility."""
    perturbation = np.random.uniform(-0.01, 0.01, size=h.shape)
    h_candidate = h + perturbation
    return project(h_candidate, dx)

def simulated_annealing(h, dx, budget_s):
    """Simulated Annealing optimization for C5 minimization."""
    T_initial = 0.5
    T_min = 1e-4
    cooling_rate = 0.99
    T = T_initial
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    start_time = time.time()
    
    while T > T_min and time.time() - start_time < budget_s - 10:
        # Generate neighbor solution
        h_candidate = perturb(h, dx)
        current_C = evaluate(h_candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        delta = current_C - best_C
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h = h_candidate.copy()
            if current_C < best_C:
                best_C = current_C
                best_h = h.copy()
        
        # Cool temperature
        T *= cooling_rate
        # Adaptive cooling based on progress
        if delta >= 0:
            cooling_rate = min(cooling_rate * 1.01, 0.999)
        else:
            cooling_rate = max(cooling_rate * 0.999, 0.95)
    
    return best_h, best_C
