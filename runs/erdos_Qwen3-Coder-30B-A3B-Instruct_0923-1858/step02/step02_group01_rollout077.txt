import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

    np.random.seed(seed)

    dx = 2.0 / n_points

        initial_h[:15] = np.linspace(0, 1, 15)
        # Last 30 points: decreasing from 1 to 0
        initial_h[15:] = np.linspace(1, 0, 30)
        # Project to constraints
        initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx))
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the constraint for integral
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # You need to define both the objective function and constraints for minimize
    def objective(h):
        return evaluate(h, dx)
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Create a plausible random initialization with slightly different values
    def generate_random_init(n):

        h = h * (1.0 / (np.sum(h) * dx))  # Rescale to meet constraint
        return h

    max_time = budget_s - 5  # Leave some headroom

    opt_runs = 0
    while time.time() - t0 < max_time:
        # Run optimization
        try:
            result = minimize(objective,

                              method='SLSQP',

                              constraints=cons,
                              options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9})

                # If optimization failed, try with a random initialization
                current_h = generate_random_init(n_points)
                current_C = evaluate(current_h, dx)
                if current_C < best_C:
                    best_C = current_C
                    best_h = current_h.copy()
                    
        except Exception:
            # Fallback to random search if any error happens
            current_h = generate_random_init(n_points)
            current_C = evaluate(current_h, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = current_h.copy()
        
        opt_runs += 1

            # Occasionally try a new starting point to avoid local minima
            new_start = generate_random_init(n_points)

            if current_C < best_C:
