import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    max_time = budget_s - 10  # Leave some time for final assessment
    start_time = time.time()
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Convex optimization approach using cutting-plane method
    # We model the problem as:
    # minimize t
    # subject to S(k) <= t for all k
    # where S(k) = int h(x)(1-h(x+k)) dx = 1 - int h(x)h(x+k) dx
    
    # To create a reasonable starting solution, we do some iterative refinement
    # using simple gradient descent with projection
    lr = 0.01
    patience = 0
    max_patience = 100
    max_iter = 5000
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        # Simple gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-5
        h_eval = project(h, dx)  # Evaluate with current projection
        corr_base = correlate(h_eval, 1-h_eval, mode='full')
        base_max = np.max(corr_base)
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)

            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')

        h_new = h - lr * grad
        h_new = project(h_new, dx)
        
        # Test the new point
        current_C = evaluate(h_new, dx)

                            best_C = new_C

    except:
        pass
    
    return (best_h, best_C, n_points)
