import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def create_initial_h(n_points, dx):
    """Create a symmetric initial h with piecewise linear profile."""
    h = np.zeros(n_points)
    # Create symmetric piecewise linear function
    half = n_points // 2
    # Rising part
    h[:half] = np.linspace(0, 1, half)
    # Falling part
    h[half:] = np.linspace(1, 0, n_points - half + 1)[1:]
    # Project to sum = 1
    h = project(h, dx)
    return h

def compute_gradient(h, dx, eps=1e-5):
    """Compute numerical gradient for FIRE algorithm."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def upsample_h(h_low, n_points_high):
    """Upsample h from low to high resolution with interpolation."""
    x_low = np.linspace(0, 2, len(h_low))
    f = interp1d(x_low, h_low, kind='linear', fill_value='extrapolate')
    x_high = np.linspace(0, 2, n_points_high)
    h_high = f(x_high)
    return project(h_high, 2.0 / n_points_high)
