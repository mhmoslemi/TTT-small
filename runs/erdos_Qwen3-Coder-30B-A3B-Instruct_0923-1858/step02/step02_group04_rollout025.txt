        h_vars = cp.Variable(n_points)

        constraints += [h_vars[i] >= 0 for i in range(n_points)]

        constraints += [cp.sum(h_vars) == n_points / 2]

                    overlap += h_vars[i] * (1 - h_vars[j])
            constraints += [overlap <= t]

        objective = cp.Minimize(t)
