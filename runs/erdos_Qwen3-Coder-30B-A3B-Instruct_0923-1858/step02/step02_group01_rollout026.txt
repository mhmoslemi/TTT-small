import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

        initial_h[15:] = np.linspace(1, 0, 30)
        
    best_h = initial_h.copy()
    best_C = evaluate(initial_h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10
    
    def objective(h):
        return evaluate(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Run optimization
    for i in range(100):  # Limit number of restarts
        if time.time() - t0 > max_time:
            break
            
        # Random perturbation for restarts
        if i > 0:
            h_start = initial_h + np.random.normal(0, 0.05, n_points)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (1.0 / (np.sum(h_start) * dx))
        else:
            h_start = initial_h.copy()
            
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 1000, 'ftol': 1e-8})
        
        if res.success:
            current_C = evaluate(res.x, dx)
            if current_C < best_C:
                best_C = current_C
                best_h = res.x.copy()
                
    return (best_h, best_C, n_points)
