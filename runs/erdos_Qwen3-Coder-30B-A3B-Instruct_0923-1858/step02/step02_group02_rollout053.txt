import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available

        if len(h) != n_points:

    except NameError:
        # Create a more sophisticated initial guess
        h = np.zeros(n_points)
        # Create a piecewise constant structure
        segments = 4
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.2 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)

    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.2
    
    # Initialize swarm
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Create initial particles
    for i in range(n_particles):
        if i == 0:  # First particle uses the initial solution
            particle = h.copy()
        else:
            # Random perturbation of initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
            # Occasionally use completely random particles
            if np.random.rand() < 0.1:
                particle = np.random.rand(n_points)
        particle = project(particle, dx)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        
        # Evaluate and store personal best
        val = evaluate(particle, dx)
        p_best.append(particle.copy())
        p_best_vals.append(val)
    
    # Find initial global best
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()
    g_best_val = p_best_vals[best_idx]
    
    # PSO loop
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    for iter in range(100000):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (p_best[i] - particle) + 
                   c2 * r2 * (g_best - particle))
            
            # Clamp velocity
            vel = np.clip(vel, -max_velocity, max_velocity)
            
            # Update position
            particle = particle + vel
            
            # Project to feasible region
            particle = project(particle, dx)
            
            # Store updated particle
            particles[i] = particle
            
            # Evaluate
            val = evaluate(particle, dx)
            
            # Update personal best
            if val < p_best_vals[i]:
                p_best[i] = particle.copy()
                p_best_vals[i] = val
                
                # Update global best if needed
                if val < g_best_val:
                    g_best = particle.copy()
                    g_best_val = val
            
            # Store the final position (may be re-projected later)
            particles[i] = particle

    return (g_best, g_best_val, n_points)
