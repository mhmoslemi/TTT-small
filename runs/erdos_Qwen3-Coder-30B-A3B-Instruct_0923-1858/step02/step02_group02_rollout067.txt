import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    T0 = 100.0

    best_h = current_h.copy()

    max_time = budget_s - 10  # Leave some headroom
    
    # Annealing loop
    for i in range(max_iter):

        # Create proposal by perturbing 5% of points
        proposal_h = current_h.copy()
        num_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)

            proposal_h[idx] += np.random.normal(0, 0.1)
        
        # Project proposal to feasible region
        proposal_h = project(proposal_h, dx)
        
        # Evaluate new solution
        proposal_C = evaluate(proposal_h, dx)
        
        # Accept or reject
        delta_E = proposal_C - current_C
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            current_h = proposal_h.copy()
            current_C = proposal_C
        
        # Update best solution
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
            
        # Update temperature
        T = T * alpha
