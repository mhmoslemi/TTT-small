import time

def run(seed=42, budget_s=600, **kwargs):

        # Create a triangular initial solution with more points for better resolution
        n_points = 101

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Using a high-quality coordinate descent approach
    patience = 0
    max_patience = 100
    eta = 0.01
    iter_count = 0
    
    def objective(h_vec):
        val = compute_C5(h_vec)
        return val
    
    # Frank-Wolfe step helper
    def frank_wolfe_step(h_vec):
        # Solve the linearized optimization problem
        # ∇f(x) · (y - x) where f(x) is the maximum overlap
        # We want to find the direction of steepest ascent for the max overlap
        # This is approximate but efficient to compute
        
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # For k-th lag, compute gradient w.r.t. h[i]:
        # ∂C_k/∂h[i] = (1 - h[i+k]) - h[i+k]
        # but we need to adjust for circular indexing and use finite difference for stable gradient
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        return grad
    
    # Coordinate descent with Frank-Wolfe steps
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Compute current value
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Apply small random perturbation
                idx = np.random.choice(n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.02), 0, 1)
                h = project_feasible(h)
                patience = 0
                
        # Sample a random order for coordinates
        coord_order = np.random.permutation(n_points)
        
        # Update each coordinate one by one
        for coord in coord_order:
            # Evaluate gradient using finite difference for accuracy
            eps = 1e-6
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[coord] += eps
            h_minus[coord] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            # Gradient direction for objective wrt h[coord]
            grad = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
            
            # Update coordinate
            h[coord] = h[coord] - eta * grad
            
            # Project to unit interval
            h[coord] = np.clip(h[coord], 0, 1)
        
        # Apply feasibility projection to maintain constraints
        h = project_feasible(h)
        
        # Decay learning rate
        eta = max(0.001, eta * 0.995)
        
        # Occasionally perform a Frank-Wolfe step
        if iter_count % 50 == 0 and iter_count > 0:
            try:
                fw_grad = frank_wolfe_step(h)
                h = h - eta * fw_grad
                h = project_feasible(h)
            except:
                pass  # Fallback to standard update if FW step fails
        
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
