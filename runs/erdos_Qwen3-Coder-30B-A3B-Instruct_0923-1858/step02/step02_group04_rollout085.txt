        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Use 50 control points for smoother exploration
    n_control = 50  
    n_basis = n_control
    # Create B-spline basis matrix
    t = np.linspace(0, 2, n_basis + 4)
    T = np.array([t] * n_points)

    # Use B-spline basis functions
    def bspline_basis(x, i, k):
        if k == 0:
            return 1.0 if t[i] <= x < t[i+1] else 0.0
        if t[i+k] == t[i]:
            c1 = 0
        else:
            c1 = (x - t[i]) / (t[i+k] - t[i]) * bspline_basis(x, i, k-1)
        if t[i+k+1] == t[i+1]:
            c2 = 0
        else:
            c2 = (t[i+k+1] - x) / (t[i+k+1] - t[i+1]) * bspline_basis(x, i+1, k-1)
        return c1 + c2

    basis = np.zeros((n_points, n_basis))

        x = X[i]
        for j in range(n_basis):
            basis[i, j] = bspline_basis(x, j, 3)

    patience = 0
    max_patience = 50
    iter_count = 0
    
    # Start with current solution
    control_points = np.linalg.lstsq(basis, h, rcond=None)[0]

    step_size = 0.001
    max_iter = 10000

        h_current = basis @ control_points

                patience = 0

        eps = 1e-4
        gradient = np.zeros(n_control)

            control_plus = control_points.copy()

            control_plus[i] += eps
            control_minus[i] -= eps

            h_minus = basis @ control_minus

            h_minus = project_feasible(h_minus)
            
            c_plus = compute_C5(h_plus)
            c_minus = compute_C5(h_minus)
            gradient[i] = (c_plus - c_minus) / (2 * eps)
        
        # Gradient descent step
        control_points = control_points - step_size * gradient
        control_points = np.clip(control_points, 0, 1)
        step_size *= 0.9999

        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
