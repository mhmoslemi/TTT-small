import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    pbest = particles.copy()
    pbest_scores = np.array([evaluate(p, dx) for p in particles])
    
    # Find global best
    gb_idx = np.argmin(pbest_scores)
    gbest = pbest[gb_idx].copy()

    best_h = gbest.copy()

    # PSO iteration loop
    for iteration in range(max_iter):

            r1, r2 = np.random.rand(2)

            particles[i] = project(particles[i], dx)

            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest = particles[i].copy()
                    best_C = gbest_score

    return (best_h, best_C, n_points)
