import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx))
    
    # Use a more advanced optimization approach
    start_time = time.time()
    max_time = budget_s - 10
    
    # Try multiple configurations for robustness
    best_C = float('inf')
    best_h = initial_h.copy()
    
    # Use a combination of Nelder-Mead and gradient descent
    h = initial_h.copy()
    
    # Simple but effective optimization
    for iteration in range(5000):
        if time.time() - start_time > max_time:
            break
            
        # Estimate gradient by finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (1.0 / (np.sum(h_plus) * dx))  # Renormalize
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (1.0 / (np.sum(h_minus) * dx))  # Renormalize
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update with a more adaptive learning rate
        lr = 0.01 / (1 + iteration / 1000)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (1.0 / (np.sum(h_new) * dx))  # Renormalize
        
        # Evaluate new point
        current_C = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
        if current_C < best_C:
            h = h_new
            best_C = current_C
            best_h = h.copy()
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and np.random.rand() < 0.1:
            perturb = np.random.normal(0, 0.01, len(h))
            h = np.clip(h + perturb, 0, 1)
            h = h * (1.0 / (np.sum(h) * dx))
    
    # Final evaluation
    final_C = np.max(correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_C, n_points)
