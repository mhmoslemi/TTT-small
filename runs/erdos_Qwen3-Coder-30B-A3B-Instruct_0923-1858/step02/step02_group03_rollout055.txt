import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values at points t."""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Clip h to [0,1] and normalize so that integral is 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    N = 10  # number of Fourier coefficients

    particles = np.random.uniform(-0.5, 0.5, (P, 2*N))
    velocities = np.zeros_like(particles)
    
    # Initialize best positions
    personal_best = particles.copy()
    personal_best_fitness = np.full(P, np.inf)
    global_best_idx = 0
    global_best_fitness = np.inf
    
    t = np.linspace(0, 2, n_points)
    start_time = time.time()
    iter_count = 0
    
    # PSO Parameters
    omega = 0.7
    phi1 = phi2 = 1.5

    while iter_count < max_iter and (time.time() - start_time) < budget_s - 10:
        for i in range(P):
            # Translate coefficients to function
            a = np.zeros(N)
            b = np.zeros(N)
            a[0] = particles[i, 0]
            for n in range(1, N):
                a[n] = particles[i, 2*n - 1]
                b[n] = particles[i, 2*n]
                
            h = fourier_to_h(a, b, t)
            h = clip_and_normalize(h, dx)
            
            # Evaluate function
            fitness = evaluate(h, dx)
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_idx = i
                
        # Update particles
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] +
                             phi1 * r1 * (personal_best[i] - particles[i]) +
                             phi2 * r2 * (personal_best[global_best_idx] - particles[i]))
            particles[i] += velocities[i]
            
        iter_count += 1

    # Get best solution
    a = np.zeros(N)
    b = np.zeros(N)

    for n in range(1, N):
        a[n] = personal_best[global_best_idx, 2*n - 1]
        b[n] = personal_best[global_best_idx, 2*n]
        
    h_opt = fourier_to_h(a, b, t)
    h_opt = clip_and_normalize(h_opt, dx)
    final_c5 = evaluate(h_opt, dx)
    
    return (h_opt, final_c5, n_points)
