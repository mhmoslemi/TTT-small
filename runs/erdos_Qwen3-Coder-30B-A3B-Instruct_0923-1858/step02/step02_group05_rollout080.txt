import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    velocity_limit = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        # Start near the initial solution with small perturbations
        h_init = h + np.random.uniform(-0.1, 0.1, n_points)
        h_init = project(h_init, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h_init.copy())
        p_best_values.append(evaluate(h_init, dx))
    
    # Find initial global best
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    t0 = time.time()
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - t0 > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_value = evaluate(particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value
                
                # Update global best if needed
                if current_value < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_value
        
    return (g_best, g_best_value, n_points)
