import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_symmetric_initial_h(n_points, dx):
    """Create a symmetric piecewise linear initial guess."""
    h = np.zeros(n_points)
    mid = n_points // 2
    # Increasing part
    h[:mid] = np.linspace(0, 1, mid)
    # Decreasing part
    h[mid:] = np.linspace(1, 0, n_points - mid)
    h = project(h, dx)
    return h

def compute_gradient(h, dx, eps=1e-5):
    """Compute gradient using finite differences."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def fire_optimize(h, dx, max_iter=10000, budget_s=600):
    """Optimize using FIRE algorithm."""

    alpha = 0.99

            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Update velocities and position

        # Enforce constraints
        h_trial = project(h_trial, dx)
        
        # Compute trial objective
        trial_C = evaluate(h_trial, dx)
        
        # Accept or reject step
        if trial_C < best_C:
            h = h_trial
            best_C = trial_C
            current_t = t0
        else:
            velocity = velocity * 0.9
            current_t = current_t * 0.5
            
        # Schedule temperature
        current_t = min(current_t, t1)
        velocity = velocity * current_t

        if iteration % 200 == 0 and np.random.rand() < 0.1:
            h += np.random.normal(0, 0.005, size=len(h))
            h = project(h, dx)
            
        if trial_C < best_C:
            best_h = h.copy()
            best_C = trial_C
            
    return best_h, best_C

def upsample_h(h_low, n_points_high):
    """Upsample from lower resolution to higher."""
    x_low = np.linspace(0, 2, len(h_low))
    x_high = np.linspace(0, 2, n_points_high)
    f = interp1d(x_low, h_low, kind='linear', fill_value='extrapolate')
    h_high = f(x_high)
    return project(h_high, 2.0 / n_points_high)
