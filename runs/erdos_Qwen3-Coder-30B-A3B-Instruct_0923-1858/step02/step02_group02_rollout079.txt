import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    max_iter = 12000
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    max_velocity = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Use initial_h if available

        if len(init_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        init_h = np.ones(n_points) * 0.5
    
    # Create initial population
    for i in range(n_particles):
        if i < int(0.1 * n_particles):  # 10% use initial
            h = init_h + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))
        p_best.append(h.copy())
        p_best_vals.append(evaluate(h, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 10
    
    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (omega * velocities[i] + 
                           phi1 * r1 * (p_best[i] - particles[i]) + 
                           phi2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_val = evaluate(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best if needed
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
    
    return (g_best, g_best_val, n_points)
