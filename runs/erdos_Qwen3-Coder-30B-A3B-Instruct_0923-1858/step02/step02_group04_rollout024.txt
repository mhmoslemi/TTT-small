import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 50
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
        
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
        
    # Multi-resolution strategy
    resolutions = [50, 100, 200]
    current_resolution = 0
    
    # Population-based optimization
    population_size = 10
    population = []
    population_scores = []
    
    # Initialize population
    for _ in range(population_size):
        if current_resolution == 0:
            # Start with the given solution
            h_candidate = h.copy()
        else:
            # Interpolate from previous resolution
            h_candidate = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, len(h)), h)
            
        # Add noise to diversify
        h_candidate = h_candidate + np.random.normal(0, 0.01, n_points)
        h_candidate = project_feasible(h_candidate)
        population.append(h_candidate)
        population_scores.append(compute_C5(h_candidate))
    
    # Evolutionary strategy with local refinement
    patience = 0
    max_patience = 50
    eta = 0.02
    max_iter = 5000
    
    for iter_count in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        
        # Update best solution
        current_best_idx = np.argmin(population_scores)
        if population_scores[current_best_idx] < best_C5:

            best_h = population[current_best_idx].copy()
            patience = 0

            patience += 1

                # Restart with random perturbations
                for i in range(population_size):
                    idx = np.random.choice(n_points)
                    population[i][idx] = np.clip(population[i][idx] + np.random.normal(0, 0.05), 0, 1)
                    population[i] = project_feasible(population[i])
                patience = 0
        
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament_size = 3
            candidates = np.random.choice(population_size, tournament_size, replace=False)

            parent1 = selected[np.random.randint(population_size)]
            parent2 = selected[np.random.randint(population_size)]

            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2

            if np.random.random() < 0.3:  # 30% mutation rate

    # Final refinement
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))

def local_refine(h_vec, eta):
    """Apply a local gradient update with line search"""
    dx = 2.0 / len(h_vec)
    
    def compute_C5(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def compute_gradient(h, idx):
        eps = 1e-6
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[idx] += eps
        h_minus[idx] -= eps
        h_plus = np.clip(h_plus, 0, 1)
        h_minus = np.clip(h_minus, 0, 1)
        
        # Normalize
        h_plus = h_plus * (len(h_plus) / 2) / np.sum(h_plus)
        h_minus = h_minus * (len(h_minus) / 2) / np.sum(h_minus)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
        max_k_idx = np.argmax(corr_plus)
        
        return (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
    # Gradient ascent direction (we minimize so need negative gradient)
    grad = np.array([compute_gradient(h_vec, i) for i in range(len(h_vec))])
    new_h = h_vec - eta * grad
    new_h = np.clip(new_h, 0, 1)
    
    # Normalize
    new_h = new_h * (len(new_h) / 2) / np.sum(new_h)
    
    return new_h
