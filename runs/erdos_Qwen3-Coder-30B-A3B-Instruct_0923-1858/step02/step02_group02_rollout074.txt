import numpy as np
import time
from scipy.signal import correlate
from scipy import signal

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        segments = 4

        for i in range(segments):

            if i % 2 == 0:

    def estimate_gradient(h, dx, epsilon=1e-4):

            h_plus[i] += epsilon
            h_plus = project(h_plus, dx)

            h_minus[i] -= epsilon

            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)
