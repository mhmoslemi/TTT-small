import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_kernel(sigma):
    """Create a symmetric Hamming window kernel."""
    kernel_size = 2 * sigma + 1
    kernel = np.zeros(kernel_size)
    for i in range(-sigma, sigma+1):
        if i < 0:
            val = 0.54 - 0.46 * np.cos(np.pi * i / sigma)
        else:
            val = 0.54 + 0.46 * np.cos(np.pi * i / sigma)
        kernel[i + sigma] = val
    kernel /= kernel.sum()  # Ensure kernel sums to 1
    return kernel

def compute_smooth_gradient(h, h_smooth, dx, kernel, eps=1e-5):
    """Compute gradient using smooth approximation."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus_smooth = np.convolve(h_plus, kernel, mode='same')
        c_plus = evaluate(h_plus_smooth, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus_smooth = np.convolve(h_minus, kernel, mode='same')
        c_minus = evaluate(h_minus_smooth, dx)
        
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def time_expired(start_time, budget_s):
    """Check if time budget is exceeded."""
    return time.time() - start_time > budget_s - 10

            break
            
        # Smooth current solution        
        h_smooth = np.convolve(h, kernel, mode='same')
        
        # Compute gradient on smooth approximation
        gradient = compute_smooth_gradient(h, h_smooth, dx, kernel)
        
        # Adaptive learning rate
        if iter % decay_interval == 0 and iter > 0:
            lr *= lr_decay
            
        # Update step
        h = h - lr * gradient
        
        # Enforce constraints
        h = project(h, dx)
        
        # Evaluate and update best solution
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_h = h.copy()
            best_C = current_C
            
        # Periodic restart with higher sigma
        if iter % 500 == 0 and iter > 0:
            sigma += 2
            if sigma > 10:
                sigma = 5
            kernel = create_kernel(sigma)
        
    return best_h, best_C

def refine_solution(h, dx, n_points):
    """Refine solution by upsampling."""
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_points)
    h_refined = np.interp(x_new, x_old, h)
    return project(h_refined, 2.0 / n_points)
