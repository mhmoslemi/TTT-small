import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    max_iter = 12000
    inertia = 0.7
    cognitivism = 1.5
    social = 1.5
    max_velocity = 0.2

    # Create initial swarm
    swarm = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    # Initialize with some particles using base
    for i in range(n_particles):
        if i < n_particles // 10:  # 10% get perturbed base
            h = h_base + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points) * 0.5 + 0.25  # Random init in [0.25, 0.75]
        h = project(h, dx)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
        
        score = evaluate(h, dx)
        p_best.append(h.copy())
        p_best_scores.append(score)
    
    # Global best tracking
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    # PSO loop
    for iter_count in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitivism * r1 * (p_best[i] - swarm[i]) +
                            social * r2 * (g_best - swarm[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project(swarm[i], dx)
            
            # Evaluate
            new_score = evaluate(swarm[i], dx)
            
            # Update personal best
            if new_score < p_best_scores[i]:
                p_best_scores[i] = new_score
                p_best[i] = swarm[i].copy()
                
                # Update global best
                if new_score < g_best_score:
                    g_best_score = new_score
                    g_best = swarm[i].copy()
    
    return (g_best, g_best_score, n_points)
