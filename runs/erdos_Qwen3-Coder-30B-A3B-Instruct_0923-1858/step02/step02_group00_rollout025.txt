import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_relaxation():
        # Use a much simpler convex approach to get a good starting point
        # We want to minimize max_k integral(h(x)(1-h(x+k)) dx)
        # This is equivalent to minimizing max_k sum_i h[i] * (1 - h[i+k])
        # A relaxation is to fix the worst shift k and minimize that term
        # We instead do a symmetric relaxation: try to distribute h as evenly as possible
        # Simple approach: set h to all 1/2, then project
        h_cvx = np.full(n_points, 0.5)
        h_cvx = project_feasible(h_cvx)
        return h_cvx

    # Compute the gradient via finite differences with careful attention to feasibility
    def compute_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        # Normalize gradient slightly, not too much
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        return grad

    def trust_region_update(h_vec, lr=0.01, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = compute_gradient(h_new)
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
        return h_new

    # Smaller n_points for faster computation
    # Work with the smallest possible structure that still gives good solutions
    
    iteration = 0
    eta = 0.01
    patience = 0
    max_patience = 50
    
    # Initial convex relaxation to warm-start
    h = solve_convex_relaxation()
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation with mixing as noise
                idxs = np.random.choice(n_points, size=min(5, n_points//4), replace=False)
                h[idxs] += np.random.normal(0, 0.01, len(idxs))
                h = project_feasible(h)
                patience = 0

        # Apply trust region update
        h = trust_region_update(h, eta)
        
        # Periodic adaptation
        if iteration % 50 == 0 and iteration > 0:
            eta *= 0.95
            
        iteration += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
