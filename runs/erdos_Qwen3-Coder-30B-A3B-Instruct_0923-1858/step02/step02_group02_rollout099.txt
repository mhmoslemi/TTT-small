import numpy as np
import time
from scipy.signal import correlate
from scipy import signal

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        h = project(h, dx)

    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    # Precompute the window for smoother updates
    window = signal.windows.hann(n_points)
    
    lr = 0.01
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    patience = 0
    max_patience = 200

        if time.time() - t0 > max_time:

        # Generate gradient estimates using finite differencing
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)
        
        # Apply gradient step with smoothing
        h = h - lr * grad * window
        
        # Project to constraints
        h = project(h, dx)
        
        # Occasionally perturb to escape local minima
        if i % 50 == 0 and np.random.rand() < 0.3:
            for _ in range(5):
                idx = np.random.randint(0, len(h))
                h[idx] += np.random.normal(0, 0.005)
            h = project(h, dx)
        
        # Occasionally smooth the solution to prevent jumpy behavior
        if i % 100 == 0:
            h = np.convolve(h, np.ones(5)/5, mode='same')
            h = project(h, dx)
        
        # Check if current solution is better
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Decay learning rate
        if i % 100 == 0:
            lr *= 0.99
            
        # Early stopping if no improvement
        if patience > max_patience:
            break

    return (best_h, best_C, n_points)
