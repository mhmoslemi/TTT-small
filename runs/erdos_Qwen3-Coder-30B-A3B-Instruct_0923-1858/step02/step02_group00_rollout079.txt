def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Compute C5 value
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Project to feasible region
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Simple convex relaxation
    def solve_convex_relaxation():
        # Simple approach: due to time constraints, just return a
        # scaled version of a known good construction if it exists
        
        # Construct a better starting point using a smooth approximation
        # using a smooth triangular base
        x = np.linspace(0, 2, n_points)
        h_init = np.zeros_like(x)
        for i in range(n_points):
            if x[i] <= 1:
                h_init[i] = x[i]
            else:
                h_init[i] = 2 - x[i]
        # Normalize
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        return h_init

    # Try better initialization first
    h = solve_convex_relaxation()
    
    # Pathological case handling
    def safe_correlate(h_vec):
        # Avoid the edge case errors
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return corr

    def estimate_gradient(h_vec, k=None):
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        
        if k is None:
            # Find worst k
            corr = safe_correlate(h_vec)
            max_k_idx = np.argmax(corr)
            k = max_k_idx - (len(h_vec) - 1)
            
        # Estimate gradient by finite differences
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = safe_correlate(h_plus)
            corr_minus = safe_correlate(h_minus)
            
            grad[i] = (corr_plus[k + n_points - 1] - corr_minus[k + n_points - 1]) / (2 * eps)
            
        return grad

    # Trust region update
    def trust_region_update(h_vec, lr=0.01, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = estimate_gradient(h_new)
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
        return h_new

    # Avoid local minima with dynamic exploration
    def explore_shifts(h_vec, num_samples=10):
        # Focus on key shifts in the correlation matrix
        corr = safe_correlate(h_vec)
        top_indices = np.argsort(corr)[-num_samples:]
        shifts = [idx - (n_points - 1) for idx in top_indices]
        return shifts

    # Main optimization loop
    eta = 0.01
    patience = 0
    max_patience = 50
    iter_count = 0
    last_improvement = 0
    step_size_decay = 0.99
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
            last_improvement = iter_count
        else:
            patience += 1
            if patience > max_patience and iter_count - last_improvement > 50:
                # Perturb solution
                noise = np.random.normal(0, 0.01, n_points)
                h = project_feasible(h + noise)
                patience = 0

        h = trust_region_update(h, eta)
        # Reduce learning rate over time
        eta *= step_size_decay
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
