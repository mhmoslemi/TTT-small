            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Improve with a hybrid approach combining gradient and local search
    patience = 0
    max_patience = 50
    eta = 0.01
    iter_count = 0
    max_iter = 5000
    
    def objective(h_vec):
        val = compute_C5(h_vec)
        return val
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Compute current value
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0

            patience += 1

                # Perturb randomly
                idx = np.random.choice(n_points)

                h = project_feasible(h)
                patience = 0

        try:

            eps = 1e-5

                h_minus[i] -= eps

    if final_C5 < best_C5:
