import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    # Initialize
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_quadratic_project(h_vec):
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        objective = cp.Minimize(cp.sum_squares(h_var - h_vec))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h_var.value
        except:
            pass
        return h_vec

    # Main loop
    max_iter = 10000
    iter_count = 0
    eta = 0.05
    min_eta = 1e-6
    patience = 0
    max_patience = 50
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                h = project_feasible(h + np.random.normal(0, 0.01, n_points))
                patience = 0

        # Gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-4
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(n_points):
            h_plus[i] = h[i] + eps
            h_minus[i] = h[i] - eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
            h_plus[i] = h[i]
            h_minus[i] = h[i]

        # Update step
        if np.linalg.norm(grad) > 1e-8:
            h = h - eta * grad
        h = project_feasible(h)

        if iter_count % 100 == 0 and eta > min_eta:
            eta *= 0.98

        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
