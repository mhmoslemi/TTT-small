import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)

        x = np.linspace(0, 2, n_points)

        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def trust_region_update(h_vec, eta=0.01, max_iter=10):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = np.zeros_like(h_new)
            eps = 1e-5
            
            for j in range(len(h_new)):
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                max_k_idx = np.argmax(corr_plus)
                grad[j] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    def local_optimization(h_vec, max_iter=50):
        h_current = h_vec.copy()
        best_local = h_current.copy()
        best_local_C5 = compute_C5(h_current)
        patience = 0
        max_patience = 20
        
        for i in range(max_iter):
            if elapsed() > budget_s * 0.98:
                break
                
            h_new = trust_region_update(h_current, 0.005)
            C5_new = compute_C5(h_new)
            
            if C5_new < best_local_C5:
                best_local = h_new.copy()
                best_local_C5 = C5_new
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
                    
            h_current = h_new
            
        return best_local, best_local_C5

    # Start optimization
    iter_count = 0
    eta = 0.01
    patience = 0
    max_patience = 30
    current_h = h.copy()
    
    while elapsed() < budget_s * 0.95:
        if iter_count % 10 == 0:
            current_h, current_C5 = local_optimization(current_h, max_iter=30)
        
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=max(1, n_points // 15), replace=False)
                for i in idxs:
                    current_h[i] = np.clip(current_h[i] + np.random.normal(0, 0.03), 0, 1)
                current_h = project_feasible(current_h)
                patience = 0

        # Perturb
        for i in range(n_points):
            if np.random.random() < 0.05:
                current_h[i] = np.clip(current_h[i] + np.random.normal(0, 0.02), 0, 1)
        
        current_h = project_feasible(current_h)
        
        # Adaptive step size
        if iter_count % 20 == 0 and iter_count > 0:
            eta *= 0.95
            
        iter_count += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, len(best_h))
