import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create an initial pyramid-like solution
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def identify_critical_shifts(h_vec, num_shifts=10):
        """Identify shifts that contribute most to the maximum overlap."""
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Get indices of top shifts
        top_indices = np.argsort(corr)[-num_shifts:][::-1]
        # Convert to shift values
        shifts = top_indices - (n_points - 1)
        return shifts

    def solve_subproblem(h_vec, k):
        """Solve a linear subproblem to adjust h to reduce overlap for shift k."""
        # Set up a simple LP to minimize h[k] * (1 - h[0]) subject to constraints
        # But we want to reduce the correlation for shift k
        # Let's minimize the correlation at shift k through careful adjustment
        
        # Parameters for subproblem
        n = len(h_vec)
        
        # Create a simple adjustment vector based on the top contributing shift
        # We reduce the contribution by adjusting the h values on both sides
        adjustment = np.zeros(n)
        if k >= 0:
            # Right shift
            adjustment[k:] += 1e-2
            adjustment[:n-k] -= 1e-2
        else:
            # Left shift
            adjustment[:k] += 1e-2
            adjustment[-k:] -= 1e-2
            
        # Better approach: weight h near points that contribute to this shift
        adjustment = np.zeros(n)
        # For shift k, high correlation when h[i]*h[i+k] is large
        # So reduce h[i] if h[i+k] is large, and vice versa
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                adjustment[i] += 1e-4 * h_vec[j]
                adjustment[j] += 1e-4 * h_vec[i]
        
        # Normalize 
        if np.max(np.abs(adjustment)) > 0:
            adjustment = adjustment / np.max(np.abs(adjustment))
        
        return adjustment

    # Main optimization loop with dual approach
    patience = 0
    max_patience = 50
    last_improvement = 0
    iter_count = 0
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count
        
        if iter_count - last_improvement > max_patience:
            # Shuffle to escape local minima
            idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for i in idxs:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)
            last_improvement = iter_count
            
        # Incrementally adjust h using information from large correlation spikes
        critical_shifts = identify_critical_shifts(h, num_shifts=5)
        step_size = 0.005 / (1.0 + iter_count / 200.0)
        
        for k in critical_shifts:
            adjustment = solve_subproblem(h, k)
            h = h - step_size * adjustment
            # Project back
            h = project_feasible(h)
            
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Validate result
    if np.isnan(best_C5) or best_C5 <= 0:
        # Return a fallback if validation fails
        try:
            h = np.array(initial_h_values)
        except NameError:
            x = np.linspace(0, 2, n_points)
            h = np.maximum(0, 1 - np.abs(x - 1))
            h = h * (n_points / 2) / np.sum(h)
            
        final_C5 = compute_C5(h)
        
    return (best_h.tolist(), final_C5, len(best_h))
