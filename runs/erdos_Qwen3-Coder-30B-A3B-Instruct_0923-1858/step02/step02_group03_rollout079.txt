import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    particles = []
    velocities = []
    
    for _ in range(n_particles):
        # Random Fourier coefficients
        coeffs = np.random.normal(0, 0.1, 2*n_coeffs + 1)  # a0, a1, b1, a2, b2, ...
        particles.append(coeffs)
        velocities.append(np.random.normal(0, 0.01, 2*n_coeffs + 1))
    
    best_global = None
    best_global_C = float('inf')
    inertia = 0.7
    c1 = c2 = 1.5
    
    # Generate x grid once for performance
    x = np.linspace(0, 2, n_points)
    
    # Helper function to reconstruct h from coefficients
    def reconstruct_h(coeffs):
        h = coeffs[0] * np.ones(n_points)  # a0
        for i in range(1, n_coeffs + 1):
            h += coeffs[2*i-1] * np.cos(i * np.pi * x) + coeffs[2*i] * np.sin(i * np.pi * x)
        return h
    
    iter_count = 0
    while time.time() - start_time < max_time:
        iter_count += 1 
        for i in range(n_particles):
            # Reconstruct h from coefficients
            h = reconstruct_h(particles[i])
            h = project(h, dx)
            
            # Evaluate
            C = evaluate(h, dx)
            
            # Update personal best
            if C < best_global_C or best_global is None:
                best_global = particles[i].copy()
                best_global_C = C
            
            # Update velocities and positions
            r1, r2 = np.random.rand(2)
            for j in range(len(particles[i])):
                velocities[i][j] = (
                    inertia * velocities[i][j] +
                    c1 * r1 * (particles[i][j] - particles[i][j]) +
                    c2 * r2 * (best_global[j] - particles[i][j])
                )
                particles[i][j] += velocities[i][j]
                
        # Occasionally add random particles to avoid local minima
        if iter_count % 10 == 0:
