import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_kernel(sigma):
    """Create a symmetric Hamming window kernel."""
    kernel_size = 2 * sigma + 1
    kernel = np.zeros(kernel_size)
    for i in range(-sigma, sigma+1):
        val = 0.54 + 0.46 * np.cos(np.pi * i / sigma)
        kernel[i + sigma] = val
    kernel /= kernel.sum()  # Normalize
    return kernel

def compute_learning_rate(gradient, iteration):
    """Compute adaptive learning rate."""
    grad_norm = np.linalg.norm(gradient)
    if grad_norm == 0:
        return 0.01
    return min(0.02 / (1.0 + iteration / 1000), 0.01 / (grad_norm + 1e-8))

def compute_smooth_gradient(h, h_smooth, dx, kernel):
    """Compute gradient using smooth approximation."""
    eps = 1e-5
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus_smooth = np.convolve(h_plus, kernel, mode='same')
        c_plus = evaluate(h_plus_smooth, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus_smooth = np.convolve(h_minus, kernel, mode='same')
        c_minus = evaluate(h_minus_smooth, dx)
        
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def refine_solution(h, dx, n_new_points):
    """Refine solution by upsampling."""
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new_points)
    h_refined = np.interp(x_new, x_old, h)
    return project(h_refined, 2.0 / n_new_points)
