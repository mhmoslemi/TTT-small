import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_initial_h(n_points, dx):
    """Create a symmetric initial guess."""
    h = np.zeros(n_points)
    # Linear increase to peak and then decrease
    mid = n_points // 2
    h[:mid] = np.linspace(0, 1, mid)
    h[mid:] = np.linspace(1, 0, n_points - mid)
    h = project(h, dx)
    return h

def compute_gradient(h, dx, eps=1e-5):
    """Compute finite difference gradient."""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        grad[i] = (c_plus - c_minus) / (2 * eps)
    return grad

def upsample_h(h_low, n_points_high):
    """Upsample low resolution h to higher resolution."""
    x_low = np.linspace(0, 2, len(h_low))
    f = interp1d(x_low, h_low, kind='linear')
    x_high = np.linspace(0, 2, n_points_high)
    h_high = f(x_high)
    return project(h_high, 2.0 / n_points_high)
