import time
from scipy.signal import correlate

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Optimization parameters
    beta = 1000.0  # Temperature for exponential penalty
    gamma = 1e-4   # Barrier strength
    eta = 1e-5     # Learning rate
    decay_beta = 0.99
    decay_gamma = 1.1
    decay_eta = 0.5
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def compute_objective_and_grad(h_vec, beta, gamma):
        h_vec = np.array(h_vec)
        # Compute correlation
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_corr = np.max(corr)
        if max_corr <= 0:
            max_corr = 1.0  # Avoid division by zero
            
        # Normalize correlation
        norm_corr = corr / max_corr
        
        # Exponential penalty
        exp_penalty = np.sum(np.exp(-beta * norm_corr))
        
        # Barrier term
        barrier = -gamma * np.sum(np.log(h_vec) + np.log(1 - h_vec))
        
        # Total objective
        F = exp_penalty + barrier
        
        # Compute gradient numerically
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            # Normalize
            norm_plus = corr_plus / max_corr
            norm_minus = corr_minus / max_corr
            
            # Exponential penalty
            exp_plus = np.sum(np.exp(-beta * norm_plus))
            exp_minus = np.sum(np.exp(-beta * norm_minus))
            
            # Barrier term
            barr_plus = -gamma * np.sum(np.log(h_plus) + np.log(1 - h_plus))
            barr_minus = -gamma * np.sum(np.log(h_minus) + np.log(1 - h_minus))
            
            grad[i] = (exp_plus + barr_plus - exp_minus - barr_minus) / (2 * eps)
        
        return F, grad

    iteration = 0
    last_improvement = 0
    no_improve_count = 0
    
    while elapsed() < budget_s * 0.95:

        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iteration
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 200:
                # Random perturbation
                idxs = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 1e-6, 1 - 1e-6)
                h = project_feasible(h)
                no_improve_count = 0

        F, grad = compute_objective_and_grad(h, beta, gamma)
        
        # Update step

        h = project_feasible(h)
        
        # Update parameters
        if iteration % 50 == 0:
            beta *= decay_beta
            gamma *= decay_gamma
            if iteration % 200 == 0 and eta > 1e-8:
                eta *= decay_eta
                
        # Check for convergence
        if iteration - last_improvement > 300:
            # Temperature too low, increase
            beta = min(beta * 1.1, 10000.0)
            
        iteration += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
