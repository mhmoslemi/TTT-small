import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        t = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * t)
        h = project(h, dx)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.2
    inertia_decay = 0.99
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for _ in range(n_particles):
        # Start with perturbed version of h
        particle = h + np.random.normal(0, 0.02, n_points)
        particle = project(particle, dx)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate(particle, dx)
        p_best.append(particle.copy())
        p_best_scores.append(score)
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2
    
    # PSO iterations
    for iteration in range(10000):
        if time.time() - t0 > max_time:
            break

            w *= inertia_decay
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)

            # Clip velocity
            vel_update = np.clip(vel_update, -max_vel, max_vel)
            velocities[i] = vel_update
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project(particles[i], dx)
            
            # Evaluate new position
            current_score = evaluate(particles[i], dx)
            
            # Update personal best
            if current_score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = current_score
