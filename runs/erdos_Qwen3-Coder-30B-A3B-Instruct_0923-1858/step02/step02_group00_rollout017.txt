import numpy as np
import time
from scipy.signal import correlate
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # PSO parameters
    n_particles = 50
    max_iter = 100
    inertia_w = 0.9
    cognitive_c = 2.0
    social_c = 2.0
    mutation_rate = 0.05
    
    # Initialize particles
    particles = []
    particle_fitness = []
    personal_best = []
    personal_best_fitness = []
    
    for _ in range(n_particles):
        h_p = np.random.uniform(0, 1, n_points)
        h_p = project_feasible(h_p)
        particles.append(h_p)
        fitness = compute_C5(h_p)
        particle_fitness.append(fitness)
        personal_best.append(h_p.copy())
        personal_best_fitness.append(fitness)
    
    # Global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if elapsed() > budget_s * 0.9:
            break
            
        for i in range(n_particles):
            h_p = particles[i]
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_c * r1 * (personal_best[i] - h_p)
            social = social_c * r2 * (global_best - h_p)
            vel = inertia_w * (h_p - np.mean(particles, axis=0)) + cognitive + social
            
            new_h = h_p + vel
            new_h = project_feasible(new_h)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        new_h[j] = np.clip(new_h[j] + np.random.normal(0, 0.05), 0, 1)
                new_h = project_feasible(new_h)
            
            # Evaluate
            fitness = compute_C5(new_h)
            particles[i] = new_h
            particle_fitness[i] = fitness
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = new_h.copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = new_h.copy()
                    global_best_fitness = fitness
        
        # Reduce inertia weight
        inertia_w = 0.9 - (0.5 * iter_count / max_iter)
    
    # Refinement using gradient descent
    h_ref = global_best.copy()
    best_ref_h = h_ref.copy()
    best_ref_C5 = global_best_fitness
    
    # Gradient descent parameters
    lr = 0.01
    max_evals = 2000
    eval_count = 0
    
    def compute_penalty(h_vec, k):
        # For large k, we'll evaluate whether overlaps are relevant
        # But we optimize for max over all valid k
        # Here just compute correlation for a specific shift
        # We do not use this but for reference
        return np.sum(h_vec * np.roll(1 - h_vec, k)) * dx

    # Compute gradients using finite differences
    def compute_gradient(h_vec):
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        base_corr = correlate(h_vec, 1 - h_vec, mode="full")
        base_max_k = np.argmax(base_corr) - (len(h_vec) - 1)
        base_max_val = np.max(base_corr)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[base_max_k + len(h_vec) - 1] - corr_minus[base_max_k + len(h_vec) - 1]) / (2 * eps)
        return grad
    
    while elapsed() < budget_s * 0.95 and eval_count < max_evals:
        current_C5 = compute_C5(h_ref)
        if current_C5 < best_ref_C5:
            best_ref_C5 = current_C5
            best_ref_h = h_ref.copy()
        
        # Compute gradient
        grad = compute_gradient(h_ref)
        
        # Gradient step
        h_ref = h_ref - lr * grad
        h_ref = project_feasible(h_ref)
        
        eval_count += 1
        
        # Reduce learning rate
        if eval_count % 100 == 0:
            lr *= 0.95
    
    # Final check
    final_C5 = compute_C5(best_ref_h)
    if final_C5 < best_ref_C5:
        best_ref_C5 = final_C5
        best_ref_h = best_ref_h.copy()

    return (best_ref_h.tolist(), best_ref_C5, n_points)
