def evaluate_erdos_solution(h_values, dx):
    corr = correlate(h_values, 1 - h_values, mode='full')
    return np.max(corr) * dx

def gradient_based_optimization(h, dx, n_points, budget_s):
    best_h = h.copy()
    best_C = evaluate_erdos_solution(h, dx)
    t0 = time.time()
    max_time = budget_s - 2
    lr = 0.001

    for i in range(1000000):  # Large enough for optimization
        if time.time() - t0 > max_time:
            break
        # Compute gradient numerically
        grad = np.zeros_like(h)
        eps = 1e-6

        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / (2 * np.sum(h_plus)))
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / (2 * np.sum(h_minus)))

            corr_plus = evaluate_erdos_solution(h_plus, dx)
            corr_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[j] = (corr_plus - corr_minus) / (2 * eps)

        # Update step with projection
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / (2 * np.sum(h_new)))
        
        current_C = evaluate_erdos_solution(h_new, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h_new.copy()
            
        h = h_new
        lr *= 0.9999

    return best_h, best_C
