import time
from scipy.signal import correlate

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Temperature and barrier parameters
    beta = 1000.0
    gamma = 1e-4
    eta = 1e-5
    decay_factor = 0.99
    barrier_increase = 1.1
    beta_decay_interval = 50
    barrier_increase_interval = 50
    update_interval = 50
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 1e-6, 1 - 1e-6)  # Avoid log singularities
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def compute_gradient(h_vec, beta, gamma):
        h_vec = np.array(h_vec)
        # Cross-correlation with 1-h
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Gradients for exponential penalty
        exp_corr = np.exp(-beta * corr)
        # Normalize exp_corr
        exp_corr_sum = np.sum(exp_corr)
        if exp_corr_sum > 0:
            exp_corr = exp_corr / exp_corr_sum
        
        # Backpropagation through convolution
        # Using the backward pass idea: for each element
        # dF/dh[i] comes from derivative of exp(-beta * correlation) 
        grad = np.zeros_like(h_vec)
        
        # Manual gradient computation
        for i in range(len(h_vec)):
            # Approximate with finite differences
            eps = 1e-6
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
            grad[i] = (np.exp(-beta * np.max(corr_plus)) - np.exp(-beta * np.max(corr_minus))) / (2 * eps)
        
        # Add barrier gradient 
        barrier_grad = -gamma * (1/h_vec - 1/(1-h_vec))
        grad += barrier_grad
        
        return grad

    iter_count = 0
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

        # Compute gradient and update
        grad = compute_gradient(h, beta, gamma)
        h = h - eta * grad
        h = project_feasible(h)
        
        # Update parameters
        iter_count += 1
        
        if iter_count % beta_decay_interval == 0:
            beta *= decay_factor
            
        if iter_count % barrier_increase_interval == 0:
            gamma *= barrier_increase
            
        if iter_count % update_interval == 0:
            # Adaptive learning rate
            if iter_count > 100:
                eta *= 0.5

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
