import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    phi1 = 1.5
    phi2 = 1.5
    max_velocity = 0.2

    # Initialize swarm
    particles = []
    velocities = []
    
    # Use initial_h values for some particles
    num_initial = max(1, n_particles // 10)
    for i in range(num_initial):
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Initialize random particles
    for i in range(num_initial, n_particles):
        h_rand = np.random.rand(n_points) * 0.5 + 0.25
        h_rand = project(h_rand, dx)
        particles.append(h_rand)
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)

    # Track personal best
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    t0 = time.time()
    max_time = budget_s - 10

    for iteration in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (omega * velocities[i] + 
                 phi1 * r1 * (personal_best[i] - particles[i]) + 
                 phi2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            new_pos = particles[i] + v
            
            # Project to feasible region
            new_pos = project(new_pos, dx)
            
            # Evaluate
            new_score = evaluate(new_pos, dx)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best[i] = new_pos.copy()
                personal_best_scores[i] = new_score
                
                # Update global best if needed
                if new_score < global_best_score:
                    global_best = new_pos.copy()
                    global_best_score = new_score
            
            # Update particle
            particles[i] = new_pos
            velocities[i] = v

    return (global_best, global_best_score, n_points)
