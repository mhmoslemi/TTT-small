import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    # Initialize with best known construction if available
    h = initial_h.copy()
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    max_time = budget_s - 10
    start_time = time.time()
    
    # MIP approach with McCormick envelopes
    # Build the problem
    h_vars = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Constraints: 0 <= h <= 1
    constraints = [h_vars >= 0, h_vars <= 1]
    
    # Constraint: Integral = 1
    constraints.append(cp.sum(h_vars) * dx == 1)
    
    # Add the constraint that max overlap is at most C5
    # For each shift k = 1 to n_points-1, compute S(k) and constrain it <= C5
    # This requires term h[i] * h[i+k]
    
    # Introduce auxiliary variables y[i][j] = h[i] * h[j]
    # Create negative correlations using the fact that 1-h
    y_vars = {}
    for i in range(n_points):
        for j in range(i, n_points):
            if j - i > 0:  # Only make pairs for shifts
                y_vars[(i, j)] = cp.Variable(name=f"y_{i}_{j}")
    
    # Add McCormick constraints for bilinear terms
    for i in range(n_points):
        for j in range(i, n_points):
            if j - i > 0:
                y = y_vars[(i, j)]
                h_i = h_vars[i]
                h_j = h_vars[j]
                constraints.extend([
                    y >= 0,
                    y <= h_i,
                    y <= h_j,
                    y >= h_i + h_j - 1
                ])
    
    # Compute each S(k) term for k >= 1
    for shift in range(1, n_points):
        # Sum_{i=0}^{n_points-shift-1} h[i] - h[i]*h[i+shift]
        # This is: sum(h) - sum(h[i] * h[i+shift]) for i = 0 to n_points-shift-1
        s_expr = 0
        for i in range(n_points - shift):
            s_expr += h_vars[i] - y_vars[(i, i + shift)]
        
        # Add constraint S(k) <= C5
        constraints.append(s_expr <= C5)
    
    # Objective
    problem = cp.Problem(cp.Minimize(C5), constraints)
    
    # Solve the problem
    try:
        problem.solve(solver=cp.ECOS, time_limit=max_time - (time.time() - start_time))
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_vars.value
            c5_sol = C5.value
            # Make sure solution is valid
            if h_sol is not None and c5_sol is not None:
                # Project to [0,1]
                h_sol = np.clip(h_sol, 0, 1)
                # Renormalize to sum = 1
                h_sol = project(h_sol, dx)
                
                # Finally evaluate
                final_C = evaluate(h_sol, dx)
                if final_C < best_C:
                    best_C = final_C
                    best_h = h_sol
        
    except Exception:
        pass  # Fall back to gradient descent if MIP doesn't work
    
    # If still not good enough, try gradient method
    if best_C > 0.3808:
        h = initial_h.copy()
        h = project(h, dx)
        best_C = evaluate(h, dx)
        best_h = h.copy()
        
        # More aggressive/adapted gradient descent
        lr = 0.02
        patience = 0
        max_patience = 50
        
        for i in range(100000):
            if time.time() - start_time > max_time:
                break
                
            # Compute gradient estimation for each variable (with adaptive learning rate)
            grad = np.zeros_like(h)
            eps = 1e-5
            
            # Vectorized gradient estimation
            h_plus = h + eps
            h_plus = project(h_plus, dx)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            max_corr_plus = np.max(corr_plus)
            
            h_minus = h - eps
            h_minus = project(h_minus, dx)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            max_corr_minus = np.max(corr_minus)
            
            grad = (max_corr_plus - max_corr_minus) / (2 * eps)
            
            # Simple gradient descent step
            h_new = h - lr * grad * dx  # Adjusted by dx to prevent overstepping
            h_new = project(h_new, dx)
            
            current_C = evaluate(h_new, dx)
            
            if current_C < best_C:
                h = h_new
                best_C = current_C
                best_h = h.copy()
                patience = 0
                lr = min(0.1, lr * 1.02)
            else:
                h = h_new
                patience += 1
                if patience > max_patience:
                    lr = max(1e-6, lr * 0.95)
                    patience = 0
            
            if i % 40 == 0 and np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.005, len(h))
                h = project(h + noise, dx)
    
    return (best_h, best_C, n_points)
