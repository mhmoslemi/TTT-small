import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible random individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Mutate an individual with probability mutation_rate."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    mask = np.random.rand(len(parent1)) < 0.5
    offspring1 = np.where(mask, parent1, parent2)
    offspring2 = np.where(mask, parent2, parent1)
    return offspring1, offspring2

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

        initial_h[15:] = np.linspace(1, 0, 30)
        initial_h = project(initial_h, dx)
    
    # Genetic Algorithm parameters
    population_size = 25
    mutation_rate = 0.1
    crossover_prob = 0.8
    tournament_size = 3
    max_generations = 1000
    stagnation_threshold = 50
    reset_threshold = 100

    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points, dx))
    
    best_h = initial_h.copy()
    best_C = evaluate(best_h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10

    stagnant_count = 0
    prev_best = best_C
    
    while time.time() - t0 < max_time and generation < max_generations:
        # Evaluate current population
        fitnesses = [evaluate(ind, dx) for ind in population]
        
        # Track best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_C:
            best_C = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnant_count = 0

            mutation_rate = 0.2

            # Mutation
            offspring1 = mutate(offspring1, dx, mutation_rate)

            new_population.extend([offspring1, offspring2])
        
        # Trim to exact size
        population = new_population[:population_size]
