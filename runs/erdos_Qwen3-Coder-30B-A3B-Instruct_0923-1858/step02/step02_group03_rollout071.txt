import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    # Optimization parameters
    max_time = budget_s - 10  # Leave some time for final assessment
    start_time = time.time()
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # Use a more sophisticated optimization algorithm: Particle Swarm Optimization (PSO) 
    # with inertia weight and adaptive parameters
    
    # Initialize particles
    n_particles = 30
    max_iter = 20000
    inertia_weight = 0.7
    cognitive_weight = 1.5
    social_weight = 1.5
    
    # Particle positions and velocities
    positions = np.random.rand(n_particles, n_points)
    velocities = np.zeros((n_particles, n_points))
    
    # Scale positions to [0,1]
    positions = np.clip(positions, 0, 1)
    
    # Normalize
    for i in range(n_particles):
        positions[i] = project(positions[i], dx)
    
    # Personal best positions and values
    personal_best_positions = positions.copy()
    personal_best_values = np.array([evaluate(p, dx) for p in positions])
    
    # Global best
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                             cognitive_weight * r1 * (personal_best_positions[i] - positions[i]) + 
                             social_weight * r2 * (global_best_position - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Keep in bounds
            positions[i] = np.clip(positions[i], 0, 1)
            
            # Normalize
            positions[i] = project(positions[i], dx)
            
            # Evaluate
            current_value = evaluate(positions[i], dx)
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best_values[i] = current_value
                personal_best_positions[i] = positions[i].copy()
                
                # Update global best if needed
                if current_value < global_best_value:
                    global_best_value = current_value
                    global_best_position = positions[i].copy()
                    best_h = positions[i].copy()
                    best_C = current_value
        
        # Adjust parameters adaptively
        if iter_num % 500 == 0 and iter_num > 0:
            inertia_weight = max(0.4, inertia_weight * 0.95)
            cognitive_weight = max(1.2, cognitive_weight * 0.99)
            social_weight = max(1.2, social_weight * 0.99)
    
    # Final evaluation
    final_C = global_best_value
    final_h = global_best_position
    
    return (final_h, final_C, n_points)
