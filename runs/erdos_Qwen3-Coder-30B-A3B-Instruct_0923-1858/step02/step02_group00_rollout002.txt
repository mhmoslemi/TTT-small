        return np.max(corr) * dx

        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Convex relaxation to get initial better solution
    def solve_convex_relaxation():
        # Only use a subset of shifts to make it tractable
        num_shifts = min(15, n_points)
        shift_indices = list(range(-n_points//2, n_points//2 + 1))[:num_shifts]
        
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Helper to compute correlation with shift k
        # Only a few shift samples to aid convergence
        for k in shift_indices:
            if k != 0:  # Main case, others are handled by symmetry/linearization
                # Create a rough proxy for correlation constraints
                pass  # Continue with trust region approach
        
        # Simple warm start from the current solution
        return h.copy()

    # Gradient estimation with focus on dominant shifts
    def get_gradient(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Simultaneous perturbation gradient estimator
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        sp_direction = np.random.uniform(-1, 1, len(h_vec))
        
        h_plus = project_feasible(h_vec + eps * sp_direction)
        h_minus = project_feasible(h_vec - eps * sp_direction)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
        
        grad = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps) * sp_direction
        return grad

    # Trust region update with adaptive step size
    def trust_region_update(h_vec, eta=0.01, max_iter=20):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_gradient(h_new)
            
            # Adaptive step size
            if np.linalg.norm(grad) > 1e-8:
                step = -eta * grad / np.linalg.norm(grad)
            else:
                step = -eta * grad
                
            h_candidate = h_new + step
            h_new = project_feasible(h_candidate)
            
        return h_new

        max_k_idx = np.argmax(corr)
        return max_k_idx - (len(h_vec) - 1)

    # Main optimization loop
    eta = 0.015

    perturb_prob = 0.08
    iter_count = 0
    recent_improvement = 0

    while elapsed() < budget_s * 0.95:

            h = solve_convex_relaxation()
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            recent_improvement = 0
        else:
            recent_improvement += 1
            # More aggressive perturbations
            if recent_improvement > max_patience:
                idxs = np.random.choice(n_points, size=int(perturb_prob * n_points), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                recent_improvement = 0

        # Trust region update
        h = trust_region_update(h, eta)
        
        # Periodic reevaluation and learning rate decay
        if iter_count % 30 == 0 and iter_count > 0:
            eta *= 0.97
            if eta < 1e-5:
                eta = 1e-5
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
