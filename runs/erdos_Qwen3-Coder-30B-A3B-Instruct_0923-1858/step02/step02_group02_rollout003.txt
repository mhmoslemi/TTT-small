import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    max_velocity = 0.2
    
    # Initialize swarm
    swarm = []
    personal_bests = []
    personal_best_scores = []
    
    # Use initial_h if available

        if len(h0) != n_points:

    except NameError:

        h0 = np.zeros(n_points)
        segments = 3
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h0[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h0 = project(h0, dx)

    # Initialize particles
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            # Small random perturbation around initial solution
            h = h0 + np.random.normal(0, 0.02, n_points)
        
        h = project(h, dx)
        swarm.append(h.copy())
        score = evaluate(h, dx)
        personal_bests.append(h.copy())
        personal_best_scores.append(score)
        
    # Global best tracking
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Velocity initialization
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    iter_count = 0
    while time.time() - t0 < max_time:
        iter_count += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            phi1 * r1 * (personal_bests[i] - swarm[i]) + 
                            phi2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project(swarm[i], dx)
            
            # Evaluate
            score = evaluate(swarm[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_bests[i] = swarm[i].copy()
                personal_best_scores[i] = score
                
                # Update global best if needed
                if score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = score
    
    return (global_best, global_best_score, n_points)
