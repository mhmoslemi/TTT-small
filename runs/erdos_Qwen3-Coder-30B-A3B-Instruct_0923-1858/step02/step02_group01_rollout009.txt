import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def create_initial_h(n_points, dx):
    """Create a symmetric piecewise linear initial guess."""
    h = np.zeros(n_points)
    peak = 1.0
    decay_points = n_points // 2
    h[:decay_points] = np.linspace(0, peak, decay_points)
    h[decay_points:] = np.linspace(peak, 0, decay_points + 1)[1:]
    h = project(h, dx)
    return h

def compute_gradient(h, dx, eps=1e-5):
    """Compute finite difference gradient."""
    gradient = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        
        gradient[i] = (c_plus - c_minus) / (2 * eps)
    return gradient

def fire_optimize(h, dx, max_iter=10000, budget_s=600):
    """Optimize with FIRE algorithm."""
    start_time = time.time()
    
    # FIRE parameters
    alpha = 0.99
    beta = 0.95
    t0 = 0.5
    t1 = 0.8
    max_bad_steps = 5
    
    # Initialize
    velocity = np.zeros_like(h)
    current_t = 1.0
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # Update velocity with FIRE
        velocity = alpha * velocity + beta * grad
        h_trial = h + velocity
        
        # Project to feasible set
        h_trial = project(h_trial, dx)
        
        # Check improvement
        trial_C = evaluate(h_trial, dx)
        
        if trial_C < best_C:
            h = h_trial
            best_C = trial_C
            best_h = h.copy()
            current_t = t0
        else:
            velocity = velocity * 0.9
            current_t = current_t * 0.95
        
        # Reduce temperature
        if current_t < t1:
            current_t = t1
            
        velocity = velocity * current_t
    
    return best_h, best_C

def upsample_h(h_low, n_points_high):
    """Upsample solution from low to high resolution."""
    x_low = np.linspace(0, 2, len(h_low))
    x_high = np.linspace(0, 2, n_points_high)
    f = interp1d(x_low, h_low, kind='linear', fill_value='extrapolate')
    h_high = f(x_high)
    return project(h_high, 2.0 / n_points_high)
