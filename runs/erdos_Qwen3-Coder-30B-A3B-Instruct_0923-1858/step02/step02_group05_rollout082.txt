import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_iterations = 1000

        if len(initial_h) != n_points:

        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)
        initial_h = project(initial_h, dx)
    
    # Initialize PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Convert to float array
    h_particles = np.array([initial_h.copy() for _ in range(n_particles)])
    
    # Initialize velocity
    velocities = np.zeros_like(h_particles)
    
    # Initialize personal bests
    p_best_positions = h_particles.copy()
    p_best_values = np.array([evaluate(p, dx) for p in h_particles])
    
    # Find global best
    best_idx = np.argmin(p_best_values)
    g_best_position = p_best_positions[best_idx].copy()
    g_best_value = p_best_values[best_idx]
    
    t0 = time.time()
    max_time = budget_s - 2  # Leave headroom
    
    for iteration in range(max_iterations):
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best_positions[i] - h_particles[i]) + 
                           c2 * r2 * (g_best_position - h_particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            h_particles[i] += velocities[i]
            
            # Project to feasible set
            h_particles[i] = project(h_particles[i], dx)
            
            # Evaluate new position
            current_value = evaluate(h_particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best_values[i] = current_value
                p_best_positions[i] = h_particles[i].copy()
                
                # Update global best
                if current_value < g_best_value:
                    g_best_value = current_value
                    g_best_position = h_particles[i].copy()
    
    return (g_best_position, g_best_value, n_points)
