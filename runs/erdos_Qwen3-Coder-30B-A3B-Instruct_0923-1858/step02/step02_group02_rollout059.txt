import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

        # Create a piecewise constant structure
        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end - start))
        
        h = project(h, dx)

    # Genetic Algorithm parameters
    population_size = 50
    generations = 10000
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'h' in locals() and np.random.rand() < 0.1:
            # Perturb initial solution
            new_h = h + np.random.normal(0, 0.05, n_points)
            new_h = project(new_h, dx)
        else:
            # Random construction
            new_h = np.random.rand(n_points)
            new_h = project(new_h, dx)
        population.append(new_h)
    
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    for gen in range(generations):
        if time.time() - t0 > max_time:
            break
            
        # Evaluate fitness (inverse of C5)
        fitness_scores = []
        for ind in population:
            C = evaluate(ind, dx)
            fitness_scores.append(1.0 / C if C > 0 else float('inf'))
        
        # Track best solution
        best_idx = np.argmin([evaluate(ind, dx) for ind in population])
        current_C = evaluate(population[best_idx], dx)
        if current_C < best_C:
            best_C = current_C
            best_h = population[best_idx].copy()
        
        # Selection and reproduction
        new_population = []
        
        # Elitism: keep best solutions
        sorted_indices = np.argsort(fitness_scores)[::-1][:elite_size]
        for i in sorted_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmax(tournament_fitness)]

                child2 = parent2.copy()

                    child2[i] += np.random.normal(0, 0.05)
            
            # Project to constraints
            child1 = project(child1, dx)
            child2 = project(child2, dx)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:population_size]
    
    return (best_h, best_C, n_points)
