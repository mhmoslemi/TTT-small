import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing
    T = 1.0
    cooling_rate = 0.99
    mutation_step = 0.1
    iter_count = 0
    max_iter = 100000
    patience = 0
    max_patience = 500
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = current_h.copy()
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Create candidate solution
        candidate_h = current_h.copy()
        
        # Mutate: randomly select about 5% points and perturb
        num_mutations = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=num_mutations, replace=False)
        for idx in indices:
            perturbation = np.random.normal(0, mutation_step)
            candidate_h[idx] += perturbation
            
        # Project to feasible region
        candidate_h = project_feasible(candidate_h)
        
        # Evaluate
        candidate_C5 = compute_C5(candidate_h)
        
        # Accept or reject
        delta = candidate_C5 - current_C5
        if delta < 0 or np.exp(-delta / T) > np.random.random():
            current_h = candidate_h
            current_C5 = candidate_C5
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Cooling and adaptation
        if iter_count % 100 == 0:
            T *= cooling_rate
            mutation_step = max(0.001, mutation_step * 0.995)
            
        iter_count += 1
        
        if patience > max_patience:
            # Random restart
            h = np.zeros(n_points)
            x = np.linspace(0, 2, n_points)
            for i in range(n_points):
                h[i] = max(0, 1 - abs(x[i] - 1))
            h = h * (n_points / 2) / np.sum(h)
            current_h = h.copy()
            current_C5 = compute_C5(current_h)
            patience = 0
            T = 1.0  # Reset temperature
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
