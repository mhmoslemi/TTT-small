import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def create_symmetric_initial_h(n_points, dx):
    """Create a symmetric initial guess for h."""
    h = np.zeros(n_points)
    mid = n_points // 2
    # Increasing part
    h[:mid] = np.linspace(0, 1, mid)
    # Decreasing part
    h[mid:] = np.linspace(1, 0, n_points - mid)
    return project(h, dx)

def compute_gradient(h, dx, eps=1e-5):
    """Compute gradient using finite differences."""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project(h_minus, dx)
        
        c_plus = evaluate(h_plus, dx)
        c_minus = evaluate(h_minus, dx)
        
        grad[i] = (c_plus - c_minus) / (2 * eps)
    return grad

def fire_optimize(h, dx, budget_s, max_iter=None):
    """Optimize using FIRE algorithm."""
    if max_iter is None:
        max_iter = 10000
    
    t0 = 0.5

    beta = 0.95
    max_bad_steps = 5

    current_step = 1.0
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        gradient = compute_gradient(h, dx)
        
        # Update velocities and position
        velocity = alpha * velocity + beta * gradient
        h_trial = h + velocity
        
        # Enforce constraints
        h_trial = project(h_trial, dx)
        
        # Compute trial objective
        trial_C = evaluate(h_trial, dx)
        
        # Check trial improvement
        if trial_C < best_C:
            h = h_trial
            best_C = trial_C
            best_h = h.copy()
            current_step = 1.0
            current_t = t0
        else:
            # Reject step
            velocity = velocity * 0.9
            current_step = min(current_step + 1, max_bad_steps)
            current_t = current_t * (1 / current_step)
        
        # Schedule temperature
        current_t = min(current_t, t1)
        velocity = velocity * current_t
    
    return best_h, best_C
