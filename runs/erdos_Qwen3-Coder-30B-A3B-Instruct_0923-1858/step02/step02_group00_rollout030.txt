import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_relaxation():
        # Try a cleaner convex relaxation
        # We'll instead use a straightforward approach:
        # Start with a good solution and apply projected gradient descent
        
        # Build more reliable starting point
        try:
            h_start = np.array(initial_h_values)
        except NameError:
            # Create basic pyramid
            h_start = np.zeros(n_points)
            x = np.linspace(0, 2, n_points)
            for i in range(n_points):
                h_start[i] = max(0, 1 - abs(x[i] - 1))
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        return h_start

    def compute_exact_grad(h_vec, k_max):
        """Compute analytical gradient for the k-th correlation offset"""
        h_vec = np.array(h_vec)
        n = len(h_vec)
        grad = np.zeros_like(h_vec)
        
        # For correlation at shift k_max, we have:
        # corr[k_max] = sum_{i=0}^{n-1} h[i] * (1 - h[i + k_max])
        # This is valid when i + k_max < n   (i.e., i < n - k_max) 
        # or when i + k_max >= 0  (i >= -k_max)
        
        # In full mode, correlations can be shifted by any k from first to last
        # src [h_0 h_1 h_2 ... h_{n-1}]
        # for shift k, we align as
        #     [h_0 h_1 h_2 ... h_{n-1}]
        # [h_0 h_1 h_2 ... h_{n-1}]
        # But with offset k, the whole shift bring alignment changes
        
        # Simplified loop works for all cases
        # Let's calculate the j^{th} term of gradient w.r.t. h[i]
        
        for i in range(n):
            # how the gradient of the correlation with respect to h[i]
            j = i - k_max  # This is how i connects to a term in (1-h[i+k_max])
            if j >= 0 and j < n:
                # The common term: h[i] * (1 - h[i+k_max]) contributes
                # we differentiate f = h[i] * (1 - h[i+k_max])
                # df/dh[i] = 1 - h[i+k_max]
                # df/dh[i+k_max] = -h[i]
                if j < n and j >= 0:  
                    # h[j] multiplied by 1 - h[i] in the correlation
                    grad[i] = 1 - h_vec[j]
                if i + k_max < n and i + k_max >= 0:
                    # h[i] multiplied by 1 - h[i+k_max] with opposite sign due to minus
                    # This introduces h[i+k_max] contribution
                    grad[i] = grad[i] - h_vec[i + k_max] if i+k_max >= 0 else 0
            else:
                # Just compute contribution from h[i], needed for accounting though.
                grad[i] = 1 - h_vec[i + k_max] if 0 <= i + k_max < n else 0

        return grad

    def sample_grd(h_vec, num_samples=10):
        # Sample significant k values according to some heuristic
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_idx = np.argmax(corr)
        max_k = max_idx - (len(h_vec) - 1)  # Find corresponding k
        
        # Also get a few more to explore
        indices = list(np.argsort(corr)[-num_samples:])
        k_values = [i - (len(h_vec) - 1) for i in indices]  # Convert indices to shifts
        
        grad_sum = np.zeros_like(h_vec)
        grad_sum.fill(0.0)
        for k in k_values:
            # Take small sample of actual shifts
            if -n_points < k < n_points:
                grad = compute_exact_grad(h_vec, k)
                grad_sum += grad
        
        return grad_sum / len(k_values) if len(k_values) > 0 else grad_sum

    def trust_region_update(h_vec, eta=0.01, max_iter=50):
        h_new = h_vec.copy()
        for _ in range(max_iter):
            grad = sample_grd(h_new)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Main optimization loop
    eta = 0.01
    patience = 0
    max_patience = 50
    perturb_prob = 0.1
    iter_count = 0

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random perturbation
                idxs = np.random.choice(n_points, size=int(perturb_prob * n_points), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Trust region update
        h = trust_region_update(h, eta)
        
        # Periodic reevaluation and learning rate decay
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
