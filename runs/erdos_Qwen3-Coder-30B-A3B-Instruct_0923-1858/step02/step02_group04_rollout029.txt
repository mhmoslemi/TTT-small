import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Solve using convex optimization
    dx = 2.0 / n_points
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, t >= 0]
    
    # Add overlap constraints
    for k in range(1, n_points):
        # Compute overlap for shift k
        overlap = 0
        for i in range(n_points - k):
            overlap += h_var[i] * (1 - h_var[i + k])
        constraints.append(overlap <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_opt = h_var.value
        else:
            h_opt = h
    except:
        h_opt = h
    
    # Project to valid range and normalize
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = np.sum(h_opt)
    if current_sum > 0:
        h_opt = h_opt * (n_points / 2) / current_sum
    
    # Final evaluation
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    final_C5 = compute_C5(h_opt)
    
    return (h_opt.tolist(), final_C5, n_points)
