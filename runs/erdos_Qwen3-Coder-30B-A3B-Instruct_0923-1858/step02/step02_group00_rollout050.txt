import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Improved initialization with convex relaxation
    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Simple relaxation by having a few key constraints
        # This will solve a tractable version to get a good starting point
        prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            # Return the relaxed solution, but make sure it's projected
            result = np.array(h_var.value).flatten()
            return project_feasible(result)
        else:
            # Fallback
            return h.copy()

    # Get a good initialization
    h = solve_convex_relaxation()
    
    # Focus on optimizing for the most critical shifts
    def compute_gradients(h_vec, num_shifts=20):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Focus gradient estimation on regions where overlap is larger
        # Sample shift indices that matter most
        shift_indices = list(range(-n_points//2, n_points//2 + 1))
        shift_indices = sorted(shift_indices, key=lambda k: -abs(k - max_k))
        used_shifts = shift_indices[:num_shifts]
        
        # Compute gradients w.r.t. to selected shifts
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            # Accumulate gradient across important shifts
            for k in used_shifts:
                idx = k + n_points - 1
                if 0 <= idx < len(corr_plus):
                    grad[i] += (corr_plus[idx] - corr_minus[idx]) / (2 * eps) * 0.1
        
        return grad

    # Use stochastic gradient decent with adaptive learning rate
    def sgd_update(h_vec, lr=0.01, num_updates=100):
        h_new = h_vec.copy()
        for _ in range(num_updates):
            grad = compute_gradients(h_new)
            h_new = h_new - lr * grad
            h_new = project_feasible(h_new)
        return h_new

    # Refinement using many updates and shifting focus
    iter_count = 0
    start_optimization = time.time()
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

        # Adaptive learning rate
        lr = max(1e-5, 0.01 * (1 - elapsed() / (budget_s * 0.9)))
        h = sgd_update(h, lr=lr, num_updates=50)

        iter_count += 1
        
        # Occasionally explore with random purturbation to avoid getting stuck
        if iter_count % 10 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.02, size=n_points)
            h = project_feasible(h + noise)

    # Final refinement check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
