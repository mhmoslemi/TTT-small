import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values"""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Clip h to [0,1] and normalize to integral = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Compute C5 value for given h"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    particles = np.random.uniform(-0.5, 0.5, (P, 2*N))  # a0 to aN, b1 to bN
    velocities = np.zeros_like(particles)
    
    # For tracking personal bests
    personal_best = particles.copy()
    personal_best_vals = np.full(P, np.inf)
    
    # Global best tracking
    gbest_idx = 0
    gbest_val = np.inf
    
    start_time = time.time()

    for iter_num in range(max_iter):
        if time.time() - start_time > max_time:

            a = np.zeros(N)

            # Evaluate
            current_val = evaluate(h_func, dx)
            
            # Update personal best
            if current_val < personal_best_vals[i]:
                personal_best_vals[i] = current_val
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if current_val < gbest_val:
                gbest_val = current_val
                gbest_idx = i
        
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                omega * velocities[i] +
                phi1 * r1 * (personal_best[i] - particles[i]) +
                phi2 * r2 * (personal_best[gbest_idx] - particles[i])

            best_h = clip_and_normalize(best_h, dx)
    
    # Final evaluation of best solution
    final_C = evaluate(best_h, dx)
    return (best_h, final_C, n_points)
