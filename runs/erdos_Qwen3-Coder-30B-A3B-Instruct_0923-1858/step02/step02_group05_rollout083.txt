import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Particle positions and velocities
    positions = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    p_best = np.zeros((n_particles, n_points))
    p_best_values = np.full(n_particles, np.inf)
    
    # Initialize particles
    for i in range(n_particles):
        if i == 0:
            # Use the h as first particle
            positions[i] = h.copy()
        else:
            # Add small random perturbations
            perturb = np.random.normal(0, 0.05, n_points)
            positions[i] = h + perturb
            positions[i] = project(positions[i], dx)
        
        # Evaluate initial position
        C_val = evaluate(positions[i], dx)
        p_best_values[i] = C_val
        p_best[i] = positions[i].copy()

    # Find global best
    gb_idx = np.argmin(p_best_values)
    g_best = p_best[gb_idx].copy()

    for iter in range(10000):

            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +

                            c2 * r2 * (g_best - positions[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            positions[i] += velocities[i]
            positions[i] = project(positions[i], dx)

            C_val = evaluate(positions[i], dx)
            
            # Update personal best
            if C_val < p_best_values[i]:
