import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:

        n_points = len(h)
    except NameError:
        # Create a good initial solution using a smoothed step function
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a piecewise constant approximation to a step function
        h[x < 0.5] = 1.0
        h[(x >= 0.5) & (x < 1.5)] = 0.5
        h[x >= 1.5] = 0.0
        # Normalize to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Try different local search strategies if we have time
    max_pso_iter = 300
    max_grad_iter = 300
    pso_iter = 0
    grad_iter = 0
    
    # PSO parameters
    n_particles = 50
    inertia = 0.9
    c1 = c2 = 2.0
    max_vel = 0.2
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest_idx = 0
    gbest_val = float('inf')
    
    # Create initial swarm
    for _ in range(n_particles):
        # Random perturbation around initial solution
        h_pert = h + np.random.normal(0, 0.05, n_points)
        particles.append(project_feasible(h_pert))
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        pbest.append(particles[-1].copy())
        val = compute_C5(particles[-1])
        pbest_vals.append(val)
        if val < gbest_val:
            gbest_val = val
            gbest_idx = len(particles) - 1
    
    # Main optimization loop
    while elapsed() < budget_s * 0.95:
        # PSO update
        if pso_iter < max_pso_iter:
            for i in range(n_particles):
                r1, r2 = np.random.rand(2)
                velocities[i] = (
                    inertia * velocities[i] +
                    c1 * r1 * (pbest[i] - particles[i]) +
                    c2 * r2 * (gbest_val * np.ones(n_points) - particles[i])
                )
                velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
                particles[i] += velocities[i]
                particles[i] = project_feasible(particles[i])
                
                # Update personal best
                new_val = compute_C5(particles[i])
                if new_val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = new_val
                    if new_val < gbest_val:
                        gbest_val = new_val
                        gbest_idx = i
                        
            pso_iter += 1
            continue
            
        # Try gradient descent refinement
        if grad_iter < max_grad_iter and elapsed() < budget_s * 0.9:
            # Get current gbest
            h_new = particles[gbest_idx].copy()
            current_C5 = pbest_vals[gbest_idx]
            
            # Simple gradient descent steps
            for _ in range(5):
                if elapsed() >= budget_s * 0.9:
                    break
                    
                grad = np.zeros_like(h_new)
                corr = correlate(h_new, 1 - h_new, mode="full")
                max_k = np.argmax(corr) - (len(h_new) - 1)
                
                # Compute finite differences for gradient
                eps = 1e-5
                for j in range(len(h_new)):
                    h_plus = h_new.copy()
                    h_minus = h_new.copy()
                    h_plus[j] += eps
                    h_minus[j] -= eps
                    h_plus = project_feasible(h_plus)
                    h_minus = project_feasible(h_minus)
                    
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    grad[j] = (corr_plus[max_k + len(h_new) - 1] - 
                              corr_minus[max_k + len(h_new) - 1]) / (2 * eps)
                
                # Update
                h_new = h_new - 0.005 * grad
                h_new = project_feasible(h_new)
                
                # Check improvement
                new_C5 = compute_C5(h_new)
                if new_C5 < current_C5:
                    current_C5 = new_C5
                    particles[gbest_idx] = h_new.copy()
                    pbest_vals[gbest_idx] = current_C5
                    if current_C5 < gbest_val:
                        gbest_val = current_C5
                        gbest_idx = i
                else:
                    break
                    
            grad_iter += 1
            
        # Check if we've improved the global best
        trial_C5 = compute_C5(particles[gbest_idx])
        if trial_C5 < best_C5:
            best_C5 = trial_C5
            best_h = particles[gbest_idx].copy()
            
        # Occasionally random initialization
        if np.random.rand() < 0.05:
            h_pert = h + np.random.normal(0, 0.05, n_points)
            particles[np.random.randint(0, n_particles)] = project_feasible(h_pert)
            val = compute_C5(particles[-1])
            if val < gbest_val:
                gbest_val = val
                gbest_idx = len(particles) - 1
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, len(best_h))
