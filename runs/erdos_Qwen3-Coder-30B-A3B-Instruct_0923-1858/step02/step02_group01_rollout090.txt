import numpy as np
import time
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def create_kernel(sigma):
    kernel_size = 2 * sigma + 1
    kernel = np.zeros(kernel_size)
    for i in range(-sigma, sigma+1):
        val = 0.54 + 0.46 * np.cos(np.pi * i / sigma)
        kernel[i + sigma] = val
    kernel /= kernel.sum()
    return kernel

def compute_gradient(h, dx, kernel, eps=1e-5):
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus_smooth = np.convolve(h_plus, kernel, mode='same')
        c_plus = evaluate(h_plus_smooth, dx)

        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus_smooth = np.convolve(h_minus, kernel, mode='same')
        c_minus = evaluate(h_minus_smooth, dx)

        grad[i] = (c_plus - c_minus) / (2 * eps)
    return grad

def time_expired(start_time, budget_s):
    return time.time() - start_time > budget_s - 5

    h_refined = np.interp(x_new, x_old, h)
    return project(h_refined, 2.0 / n_new_points)
