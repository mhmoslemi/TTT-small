import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h if available
    try:
        h = initial_h_values.copy()
        if len(h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create an initial piecewise constant structure with multiple segments
        h = np.zeros(n_points)
        segments = 10
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i+1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.2 * np.sin(np.linspace(0, np.pi, end - start))
        h = project(h, dx)

    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    # Segment-based alternating optimization
    m_segments = 10  # Number of segments
    positions = np.linspace(0, 2, m_segments + 1)
    segment_sizes = np.diff(positions)
    
    # Create segment boundaries
    def get_segment_bounds():
        return np.linspace(0, 2, m_segments + 1)
    
    def update_h_from_segments(heights, positions):
        """Convert segment heights to full h vector"""
        h = np.zeros(n_points)
        for i in range(m_segments):
            start_idx = int(positions[i] / 2.0 * n_points)
            end_idx = int(positions[i+1] / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = heights[i]
        return project(h, dx)
    
    def update_segments_from_h(h_full):
        """Convert h vector to segment heights"""
        heights = np.zeros(m_segments)
        bounds = get_segment_bounds()
        for i in range(m_segments):
            start_idx = int(bounds[i] / 2.0 * n_points)
            end_idx = int(bounds[i+1] / 2.0 * n_points)
            if end_idx > start_idx:
                heights[i] = np.mean(h_full[start_idx:end_idx])
        return heights
    
    # Alternating optimization
    lr_pos = 0.1
    lr_height = 0.1
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    no_improvement_count = 0
    
    # Initial segment heights and positions
    current_heights = update_segments_from_h(h)
    current_positions = np.linspace(0, 2, m_segments + 1)
    
    def compute_overlap_gradient(current_h, dx):
        """Compute approximate gradient of overlap with respect to h"""
        # We compute finite differences for gradient estimation
        epsilon = 1e-4
        grad = np.zeros_like(current_h)
        for i in range(len(current_h)):
            h_plus = current_h.copy()
            h_plus[i] += epsilon
            h_plus = project(h_plus, dx)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_plus_max = np.max(corr_plus)
            
            h_minus = current_h.copy()
            h_minus[i] -= epsilon
            h_minus = project(h_minus, dx)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            corr_minus_max = np.max(corr_minus)
            
            grad[i] = (corr_plus_max - corr_minus_max) / (2 * epsilon)
        return grad
    
    for iteration in range(100000):
        if time.time() - t0 > max_time:
            break
            
        # Optimize heights (convex formulation)
        heights = update_segments_from_h(h)
        bounds = get_segment_bounds()
        
        # Create convex optimization problem
        h_vars = cp.Variable(m_segments)
        objective = cp.Minimize(cp.sum(cp.multiply(h_vars, 1-h_vars)))
        constraints = [
            cp.sum(cp.multiply(h_vars, segment_sizes)) == 1.0,  # Integral constraint
            h_vars >= 0,
            h_vars <= 1
        ]
        prob = cp.Problem(objective, constraints)
        
        try:

                heights = h_vars.value

                delta = 0.01

                positions_minus[i] -= delta
                h_minus = update_h_from_segments(heights, positions_minus)
                overlap_minus = evaluate(h_minus, dx)
                
                # Gradient of overlap with respect to position
                grad_pos = (overlap_plus - overlap_minus) / (2 * delta)
                new_positions[i] -= lr_pos * grad_pos
                
        # Ensure valid bounds for positions
        new_positions = np.clip(new_positions, 0, 2)
        # Ensure strict monotonicity
        new_positions = np.sort(new_positions)
        # Set bounds
        new_positions[0] = 0
        new_positions[-1] = 2
        
        positions = new_positions
        
        # Update h from segments
        h = update_h_from_segments(heights, positions)
        
        # Project to ensure feasibility
        h = project(h, dx)
        
        # Evaluate current solution
        current_C = evaluate(h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update learning rates
        if iteration % 100 == 0:
            lr_pos *= 0.99
            lr_height *= 0.99
            
        if no_improvement_count > 100:
            break

    # Final polishing with local gradient descent on full h
    lr = 0.01
    for i in range(1000):
        if time.time() - t0 > max_time:
            break
        grad = compute_overlap_gradient(best_h, dx)
        best_h = best_h - lr * grad
        best_h = project(best_h, dx)
        current_C = evaluate(best_h, dx)
        if current_C < best_C:
            best_C = current_C
        lr *= 0.99

    return (best_h, best_C, n_points)
