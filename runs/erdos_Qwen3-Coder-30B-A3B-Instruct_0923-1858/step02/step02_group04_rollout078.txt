import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a triangular function
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 101
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    # Convert to CVX variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Add overlap constraints
    dx = 2.0 / n_points
    for k in range(2 * n_points - 1):
        # Compute the correlation for shift k
        if k < n_points:
            # Forward overlap
            corr = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            # Backward overlap
            corr = cp.sum(cp.multiply(h_var[k-n_points+1:], 1 - h_var[:n_points-(k-n_points+1)]))
        constraints.append(corr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with MOSEK
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        h_opt = h_var.value
        if h_opt is not None:
            h_opt = np.clip(h_opt, 0, 1)
            # Normalize properly
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        else:
            h_opt = h.copy()
    except:
        h_opt = h.copy()
    
    # Final evaluation
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    best_C5 = compute_C5(h_opt)
    return (h_opt.tolist(), best_C5, n_points)
