import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    # Simulated Annealing parameters
    T0 = 100.0
    alpha = 0.999
    max_iter = 10000
    perturb_frac = 0.05
    perturb_sigma = 0.1
    
    # Initial temperature and current solution
    T = T0
    h_current = h.copy()
    h_current = project(h_current, dx)
    E_current = evaluate(h_current, dx)
    
    # Track best solution
    h_best = h_current.copy()
    E_best = E_current
    
    t0 = time.time()
    max_time = budget_s - 10  # Leave some headroom
    
    for i in range(max_iter):
        if time.time() - t0 > max_time:
            break
            
        # Create proposal by perturbing some points
        h_proposal = h_current.copy()
        n_perturb = max(1, int(perturb_frac * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            h_proposal[idx] += np.random.normal(0, perturb_sigma)
            
        # Project to feasible region
        h_proposal = project(h_proposal, dx)
        
        # Evaluate proposal
        E_proposal = evaluate(h_proposal, dx)
        
        # Accept or reject based on Metropolis criterion
        delta_E = E_proposal - E_current
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h_current = h_proposal
            E_current = E_proposal
            
            # Update best solution if improved
            if E_current < E_best:
                E_best = E_current
                h_best = h_current.copy()
                
        # Update temperature
        T *= alpha
        
    return (h_best, E_best, n_points)
