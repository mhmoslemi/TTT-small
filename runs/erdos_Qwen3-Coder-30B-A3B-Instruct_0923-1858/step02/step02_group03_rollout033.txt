import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def clip_and_normalize(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def fourier_to_h(a, b, t):
    N = len(a) - 1
    h = a[0] + np.sum([a[n+1] * np.cos(np.pi * (n+1) * t) + b[n] * np.sin(np.pi * (n+1) * t) for n in range(N)], axis=0)
    return h

    # Initialize particles
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N + 1))  # a_0, a_1..a_N, b_1..b_N
    velocities = np.random.uniform(-0.1, 0.1, (P, 2*N + 1))
    pbest = particles.copy()
    pbest_C = np.full(P, np.inf)
    
    # Global best
    gbest = particles[0].copy()
    gbest_C = np.inf
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        for i in range(P):
            # Decode coefficients
            a = pbest[i, 0]
            a_coeffs = pbest[i, 1:N+1]
            b_coeffs = pbest[i, N+1:]
            
            # Generate function
            t = np.linspace(0, 2, n_points)
            h = fourier_to_h(np.concatenate([[a], a_coeffs]), b_coeffs, t)

            C = evaluate(h, dx)
            
            # Update personal best
            if C < pbest_C[i]:
                pbest_C[i] = C
                pbest[i] = particles[i].copy()
            
            # Update global best
            if C < gbest_C:
                gbest_C = C
                gbest = particles[i].copy()
        
        # Update velocities and positions
        for i in range(P):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
    
    # Generate final h from gbest
    a = gbest[0]
    a_coeffs = gbest[1:N+1]
    b_coeffs = gbest[N+1:]
    t = np.linspace(0, 2, n_points)
