import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Main optimization loop with tailored algorithm
    max_iter = 1000
    patience = 0
    max_patience = 50
    iter_count = 0
    eta = 0.01
    
    # Sample modification parameters
    perturb_prob = 0.15
    local_search_radius = 0.1
    crossover_prob = 0.05
    
    while elapsed() < budget_s * 0.95 and iter_count < max_iter:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Introduce random perturbation
                idxs = np.random.choice(n_points, size=int(perturb_prob * n_points), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.03), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Stochastic gradient descent with projection
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # Estimate gradient using finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)

        # Update
        h = h - eta * grad
        h = project_feasible(h)
        
        # Occasionally add Gaussian noise to exploration
        if np.random.random() < 0.05:
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
            h = project_feasible(h)
            
        # Occasionally perform crossover with best
        if np.random.random() < crossover_prob:
            for i in range(n_points):
                if np.random.random() < 0.5:
                    h[i] = best_h[i]
            h = project_feasible(h)
            
        # Adjust learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            eta *= 0.97
            
        iter_count += 1

    # Final re-evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
