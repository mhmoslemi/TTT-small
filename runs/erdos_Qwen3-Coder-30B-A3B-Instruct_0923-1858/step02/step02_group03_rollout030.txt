import numpy as np
import time
from scipy.signal import correlate

def fourier_to_h(a, b, t):
    """Convert Fourier coefficients to function values"""
    h = a[0] * np.ones_like(t)
    for n in range(1, len(a)):
        h += a[n] * np.cos(n * np.pi * t) + b[n] * np.sin(n * np.pi * t)
    return h

def clip_and_normalize(h, dx):
    """Clip function to [0,1] and normalize to integral = 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum
    return h

def evaluate(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set parameters
    n_points = 45
    dx = 2.0 / n_points
    N = 10  # Number of Fourier coefficients (a0, a1, ..., an, b1, ..., bn)
    P = 50  # Number of particles
    max_iter = 1000
    
    # Initialize particles
    particles = np.random.uniform(-0.5, 0.5, (P, 2*N))  # a coefficients + b coefficients
    velocities = np.zeros_like(particles)
    
    # Initialize personal best and global best
    pbest = particles.copy()
    pbest_C = np.full(P, np.inf)
    
    # Use initial_h_values if available
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smoother initial point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2) + 0.2 * np.sin(3 * np.pi * t)
        initial_h = clip_and_normalize(initial_h, dx)
    
    # Convert initial point to Fourier coefficients
    start_time = time.time()
    max_time = budget_s - 10
    
    # Get starting point
    t = np.linspace(0, 2, n_points)
    h = initial_h.copy()
    h_fourier = np.fft.fft(h)
    a = np.real(h_fourier[:N])
    b = -np.imag(h_fourier[:N])
    particles[0] = np.concatenate([a, b])
    
    # PSO parameters
    omega = 0.7
    phi1 = 1.5
    phi2 = 1.5
    
    gbest_idx = 0
    gbest_C = np.inf
    best_h = None
    
    for iter in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        # Evaluate all particles
        for i in range(P):
            # Convert coefficients to function values
            a_coeffs = particles[i, :N]
            b_coeffs = particles[i, N:]
            h_func = fourier_to_h(a_coeffs, b_coeffs, t)
            h_func = clip_and_normalize(h_func, dx)
            
            # Evaluate
            C = evaluate(h_func, dx)
            
            # Update personal best
            if C < pbest_C[i]:
                pbest[i] = particles[i].copy()
                pbest_C[i] = C
                
                # Update global best
                if C < gbest_C:
                    gbest_C = C
                    gbest_idx = i
                    best_h = h_func.copy()
        
        # Update velocities and positions
        r1, r2 = np.random.rand(P, 1), np.random.rand(P, 1)
        gbest = particles[gbest_idx]
        
        for i in range(P):
            velocities[i] = (
                omega * velocities[i] +
                phi1 * r1[i] * (pbest[i] - particles[i]) +
                phi2 * r2[i] * (gbest - particles[i])
            )
            particles[i] += velocities[i]
    
    # Final check with best found
    if best_h is None:
        # Fallback to initial function if nothing found
        best_h = initial_h.copy()
        gbest_C = evaluate(best_h, dx)
    
    return (best_h, gbest_C, n_points)
