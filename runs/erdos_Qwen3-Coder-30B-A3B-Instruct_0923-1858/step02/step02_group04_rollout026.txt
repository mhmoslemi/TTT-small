import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    
    # Try convex optimization approach
    def solve_convex_optimization(h_initial):
        # Variables
        h_vars = cp.Variable(n_points)
        t = cp.Variable()

        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2, t >= 0]

        # Maximum overlap constraint for all shifts k
        for k in range(n_points):
            # Compute overlap for shift k
            # This is equivalent to  sum h[i] * (1 - h[i + k]) over all valid i
            overlap = 0
            for i in range(n_points):
                if i + k < n_points:
                    overlap += h_vars[i] * (1 - h_vars[i + k])
                else:
                    # This happens in correlation full mode
                    break
            constraints.append(overlap <= t)

        # Objective
        objective = cp.Minimize(t)

        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_vars.value
        except Exception:
            pass
        # Fallback to initial solution if optimization fails
        return h_initial.copy()

    # Initial solve
    best_h = solve_convex_optimization(h)

    # Refine with local search if time permits
    time_limit = budget_s * 0.95
    elapsed = time.time() - start_time
    if elapsed < time_limit:
        def compute_C5(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return np.max(corr) * dx

        def project_feasible(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            current_sum = np.sum(h_vec)
            if current_sum > 0:
                h_vec = h_vec * (n_points / 2) / current_sum
            return h_vec

        best_C5 = compute_C5(best_h)
        patience = 0
        max_patience = 100
        eta = 0.01

        def get_gradient(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_vec) - 1)
            
            eps = 1e-6
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            return grad

        # Local search refinement
        iter_count = 0
        while True:
            elapsed = time.time() - start_time
            if elapsed >= time_limit:
                break
                
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Perturb randomly
                    idx = np.random.choice(n_points)
                    best_h[idx] = np.clip(best_h[idx] + np.random.normal(0, 0.03), 0, 1)
                    best_h = project_feasible(best_h)
                    patience = 0
            
            try:
                grad = get_gradient(best_h)
                best_h = best_h - eta * grad
                best_h = project_feasible(best_h)
                eta = max(0.001, eta * 0.995)
            except:
                # Fallback
                idx = np.random.choice(n_points, size=5, replace=False)
                best_h[idx] = np.clip(best_h[idx] + np.random.normal(0, 0.02, len(idx)), 0, 1)
                best_h = project_feasible(best_h)
            
            iter_count += 1

        final_C5 = compute_C5(best_h)
        if final_C5 < best_C5:
            best_C5 = final_C5

    # Final computation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
