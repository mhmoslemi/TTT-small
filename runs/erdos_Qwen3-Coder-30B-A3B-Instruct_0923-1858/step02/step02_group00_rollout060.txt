import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial pyramid function
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Improved optimization approach using alternating projections
    def optimize_with_alternating_projection():
        h_current = h.copy()
        max_iter = 1000
        patience = 0
        max_patience = 100
        step_size = 0.01
        
        for iter_num in range(max_iter):
            if elapsed() > budget_s * 0.95:
                break
                
            current_C5 = compute_C5(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Random perturbation
                    idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                    for i in idxs:
                        h_current[i] = np.clip(h_current[i] + np.random.normal(0, 0.05), 0, 1)
                    h_current = project_feasible(h_current)
                    patience = 0

            # Gradient estimate with finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-4
            corr_base = correlate(h_current, 1 - h_current, mode="full")
            max_k_idx = np.argmax(corr_base)
            
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            # Update step
            h_current = h_current - step_size * grad
            h_current = project_feasible(h_current)
            
            # Adaptive step size
            if iter_num % 50 == 0 and iter_num > 0:
                step_size *= 0.9
            
        return best_h, best_C5

    # More aggressive enhancement using local search with mashup of methods
    def enhanced_local_search():
        h_current = h.copy()
        best_C5 = compute_C5(h_current)
        best_solution = h_current.copy()
        
        # Track number of iterations for adaptive behavior
        iteration_count = 0
        
        # Basic local search with random perturbations
        for _ in range(1000):
            if elapsed() > budget_s * 0.95:
                break
                
            iteration_count += 1

            # Try random perturbations
            if np.random.random() < 0.3:
                idxs = np.random.choice(n_points, size=max(1, n_points // 15), replace=False)
                for i in idxs:
                    h_current[i] = np.clip(h_current[i] + np.random.normal(0, 0.02), 0, 1)
                h_current = project_feasible(h_current)
                
            # Gradient descent step with projection
            else:
                corr_base = correlate(h_current, 1 - h_current, mode="full")
                max_k_idx = np.argmax(corr_base)
                eps = 1e-5
                
                grad = np.zeros_like(h_current)
                for i in range(n_points):
                    h_plus = h_current.copy()
                    h_minus = h_current.copy()
                    h_plus[i] += eps
                    h_minus[i] -= eps
                    h_plus = project_feasible(h_plus)
                    h_minus = project_feasible(h_minus)
                    
                    corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
                
                step_size = 0.01 * (1.0 - iteration_count / 1000.0)
                h_current = h_current - step_size * grad
                h_current = project_feasible(h_current)
                
            current_C5 = compute_C5(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_solution = h_current.copy()

        return best_solution, best_C5

    # Perform optimization
    best_h, best_C5 = enhanced_local_search()
    
    # Final validation check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
