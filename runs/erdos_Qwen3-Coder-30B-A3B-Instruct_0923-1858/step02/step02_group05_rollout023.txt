import numpy as np
import time
from scipy.signal import correlate

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize from current construction if available
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create initial h with slight variations
        t = np.linspace(0, 2, n_points)
        initial_h = 0.5 + 0.3 * np.sin(np.pi * t)  # Sine-based waveform
        initial_h = project(initial_h, dx)
    
    # PSO parameters
    n_particles = 50
    c1, c2 = 2.0, 2.0
    w = 0.7
    vmax = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for i in range(n_particles):
        if 'initial_h' in locals():
            # Start near initial solution with noise
            h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        else:
            # Random setup
            h = np.random.random(n_points) * 0.5 + 0.25
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())
        p_best_values.append(evaluate(h, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    t0 = time.time()
    max_time_pso = budget_s * 0.7
    max_time_gd = budget_s * 0.3
    
    # PSO Exploration Phase
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 5000  # Prevent infinite loop
    
    while time.time() - t0 < max_time_pso and no_improvement_count < max_no_improvement:
        iteration += 1
        if iteration % 1000 == 0:
            print(f"PSO Iteration {iteration}; Best C5: {g_best_value:.6f}")
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (p_best[i] - particles[i]) + 
                             c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vmax, vmax)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            current_value = evaluate(particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value
                if current_value < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_value
                    no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        if iteration % 500 == 0:
            # Reduce inertia slightly
            w = max(0.3, w * 0.995)
    
    # Gradient Descent Exploitation Phase
    current_h = g_best.copy()
    best_h = g_best.copy()
    best_C = g_best_value
    
    # Gradient descent with adaptive learning rate
    lr = 0.005
    lr_decay = 0.98
    decay_interval = 1000
    
    for i in range(20000):
        if time.time() - t0 > max_time_pso + max_time_gd:
            break
            
        if i % decay_interval == 0:
            lr *= lr_decay
            
        # Numerical gradient with finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        current_h = current_h - lr * grad
        current_h = project(current_h, dx)
        
        # Check for improvement
        current_C = evaluate(current_h, dx)
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
    
    return (best_h, best_C, n_points)
