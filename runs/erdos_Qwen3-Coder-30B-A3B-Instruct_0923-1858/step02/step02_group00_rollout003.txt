import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 45
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    def solve_convex_relaxation():
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Create a simpler relaxation: maximize a single worst-case correlation
        # We'll use a single dominant shift constraint for tractability
        k_max_shift = min(20, n_points)  # focus on near shifts
        shift_indices = list(range(-k_max_shift, k_max_shift + 1))
        
        # Use a counter-based constraint instead
        # More robust approach: sample a few key shifts
        obj = cp.Variable()
        constraints.append(obj >= cp.sum(cp.multiply(h_var, 1 - h_var[0:n_points])))

        # Just try a simple convex relaxation pass
        # A better approach is to use a cutting plane or SMB method
        # Let's not overcomplicate - let's just try with current structure
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                return np.array(h_var.value)
        except:
            pass
        return h.copy()  # fallback

    # Gradient estimation with batching
    def get_batched_gradient(h_vec, batch_size=10):
        h_vec = np.array(h_vec)
        corr_original = correlate(h_vec, 1 - h_vec, mode="full")
        max_k_idx = np.argmax(corr_original)
        max_k = max_k_idx - (len(h_vec) - 1)
        
        # Sample gradient directions
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        
        sample_indices = np.random.choice(n_points, size=min(batch_size, n_points), replace=False)
        for i in sample_indices:
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
        # Normalize gradient
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
            
        return grad

    # Take trust region step
    def trust_region_update(h_vec, eta=0.01, max_iter=3):
        h_new = h_vec.copy()
        for i in range(max_iter):
            grad = get_batched_gradient(h_new, batch_size=15)
            h_new = h_new - eta * grad
            h_new = project_feasible(h_new)
        return h_new

    # Main optimization loop with heuristic restart
    eta = 0.03
    patience = 0
    max_patience = 20
    iter_count = 0
    restart_count = 0
    max_restarts = 5

    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Add jitter and encourage exploration
                h = project_feasible(h + np.random.normal(0, 0.02, n_points))
                if restart_count < max_restarts:
                    h = solve_convex_relaxation()
                    restart_count += 1
                patience = 0

        # Update step using trust region
        h = trust_region_update(h, eta)

        # Decay learning rate
        if iter_count % 50 == 0 and iter_count > 0:
            eta *= 0.95
        
        iter_count += 1

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, len(best_h))
