import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    except NameError:

    # PSO parameters
    n_particles = 50
    c1, c2, w = 2.0, 2.0, 0.7
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for i in range(n_particles):
        if 'initial_h' in locals():
            h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        else:
            h = np.random.uniform(0.3, 0.7, n_points)
        h = project(h, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        score = evaluate(h, dx)
        p_best.append(h.copy())
        p_best_scores.append(score)
    
    # Global best
    best_idx = np.argmin(p_best_scores)
    g_best = p_best[best_idx].copy()
    g_best_score = p_best_scores[best_idx]
    
    t0 = time.time()
    max_time = budget_s * 0.7  # 70% for PSO
    
    # PSO loop
    for it in range(20000):  # Maximum iterations
        if time.time() - t0 > max_time:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] +

                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], dx)
            
            # Evaluate
            score = evaluate(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = score
                
                # Update global best
                if score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = score
    
    # Gradient descent refinement with early stopping
    max_time = budget_s * 0.3  # Remaining budget for gradient descent
    current_h = g_best.copy()

        # Update step
        current_h = current_h - lr * grad
        current_h = project(current_h, dx)
        
        # Check for improvement
        current_C = evaluate(current_h, dx)

            best_C = current_C
            best_h = current_h.copy()
    
    return (best_h, best_C, n_points)
