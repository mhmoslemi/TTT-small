import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    # Cast to float
    h = h.astype(float)
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    # Define the convex optimization problem
    def solve_convex_optimization(h_init):
        n = len(h_init)
        h = cp.Variable(n)
        
        # Create a variable for the maximum overlap
        t = cp.Variable()
        
        # Distribute the shift indices properly
        constraints = []
        for k in range(2 * n - 1):
            # Determine the shift indices based on k
            start_idx = max(0, k - n + 1)
            end_idx = min(n, k + 1)
            if end_idx <= start_idx:
                continue
                
            if k < n:
                # Shift to the right
                left_slice = slice(start_idx, end_idx)
                right_slice = slice(k, k + (end_idx - start_idx))
            else:
                # Shift to the left
                left_slice = slice(k - n + 1, k)
                right_slice = slice(0, k - n + 1)
                # We still need to make sure the slices are valid:
                if left_slice.stop <= left_slice.start:
                    continue
                if right_slice.stop <= right_slice.start:
                    continue
            
            # Correct way: proper indexing
            temp = h[left_slice] * (1 - h[right_slice])
            # sum over the overlapping parts
            overlap = cp.sum(temp)
            constraints.append(overlap <= t)
            
        # Add constraints for bounds and sum
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == n / 2)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Create problem and solve
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_solution = h.value
                return h_solution
            else:
                return h_init.copy()
        except Exception:
            return h_init.copy()

    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Attempt convex optimization with initial solution
    try:
        improved_h = solve_convex_optimization(h)
        if improved_h is not None:

            if C5_improved < best_C5:
                best_h = improved_h
                best_C5 = C5_improved
    except Exception:
        pass

    patience = 0
    max_patience = 50
    eta = 0.01
    iter_count = 0

        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum

        if elapsed >= time_limit:

        if current_C5 < best_C5:

                h = project_feasible(h)
                patience = 0

        try:

            grad = np.zeros_like(h)

                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                max_k_idx = np.argmax(corr_plus)
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            h = h - eta * grad
            h = project_feasible(h)
            eta = max(0.001, eta * 0.995)
        except:
            idx = np.random.choice(n_points, size=3, replace=False)
            h[idx] = np.clip(h[idx] + np.random.normal(0, 0.02, len(idx)), 0, 1)
            h = project_feasible(h)
        
        iter_count += 1
    
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
