import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    # Use initial_h_values if available

        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 50
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec

    # Initialize population with multiple diversified samples
    population_size = 12
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        population.append(project_feasible(perturbed))
    
    # Multi-resolution approach
    resolution_levels = [50, 70, 100, 150, 200]
    current_level_idx = 0
    
    def solve_single(h_vec, n_points):
        dx = 2.0 / n_points
        
        # Objective with gradient
        def objective(h_vec):
            h_vec = np.array(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            return -np.max(corr) * dx  # Negative for minimization

        def gradient(h_vec):
            h_vec = np.array(h_vec)
            # Compute the k that maximizes the correlation
            corr = correlate(h_vec, 1 - h_vec, mode="full")
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_vec) - 1)
            
            eps = 1e-6
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
            
            return grad
        
        # Gradient descent with adaptive step size and line search
        eta = 0.01
        max_iter = 200
        tolerance = 1e-6
        
        for _ in range(max_iter):
            elapsed = time.time() - start_time
            if elapsed >= time_limit:
                break
            current_C5 = compute_C5(h_vec)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_vec.copy()
            
            grad = gradient(h_vec)
            # Line search
            def line_search_obj(alpha):
                h_test = project_feasible(h_vec - alpha * grad)
                return compute_C5(h_test)
            
            # Find best alpha
            try:
                res = minimize_scalar(line_search_obj, bounds=(0, 2*eta), method='bounded')
                alpha = res.x
            except:
                alpha = eta
            
            h_vec = project_feasible(h_vec - alpha * grad)
            if np.linalg.norm(grad) < tolerance:
                break
            
            eta *= 0.995
            eta = max(0.0001, eta)
        
        return h_vec
    
    # Main optimization loop
    max_iters = 1500
    iter_count = 0
    patience = 0
    max_patience = 200
    
    for _ in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        iter_count += 1
        # Evolve population
        new_population = []
        for p in population:
            # Mutation with small random perturbation
            if np.random.rand() < 0.7:
                mutation_strength = 0.05
                mutated = p + np.random.normal(0, mutation_strength, n_points)
                mutated = project_feasible(mutated)
            else:
                mutated = p.copy()
            
            # Reopt local solution
            refined = solve_single(mutated, n_points)
            new_population.append(refined)
        
        # Select best from population
        best_in_pop = min(new_population, key=compute_C5)
        current_C5 = compute_C5(best_in_pop)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = best_in_pop.copy()
            patience = 0
        else:
            patience += 1
        
        # If no improvement for a while, restart from best and introduce more diversity
        if patience > max_patience:
            population = [best_h.copy()]
            for _ in range(population_size - 1):
                perturbed = best_h + np.random.normal(0, 0.08, n_points)
                population.append(project_feasible(perturbed))
            patience = 0
        else:
            population = new_population
            
        # Occasionally handle resolution increase
        if iter_count % 100 == 0 and current_level_idx < len(resolution_levels) - 1:
            current_level_idx += 1
            n_points_new = resolution_levels[current_level_idx]
            if n_points_new > n_points:
                # Interpolate to new resolution
                x_old = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, n_points_new)
                h_new = np.interp(x_new, x_old, best_h)
                h_new = project_feasible(h_new)
                n_points = n_points_new
                best_h = h_new.copy()
                best_C5 = compute_C5(best_h)
                # Reset population with new resolution
                population = [best_h.copy()]
                for _ in range(population_size - 1):
                    perturbed = best_h + np.random.normal(0, 0.08, n_points_new)
                    population.append(project_feasible(perturbed))
    
    # Final refinement of best solution
    final_h = solve_single(best_h, n_points)
    final_C5 = compute_C5(final_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, len(best_h))
