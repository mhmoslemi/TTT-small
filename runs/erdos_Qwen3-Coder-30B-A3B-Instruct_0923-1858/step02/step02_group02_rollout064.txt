import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def project_h(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def segment_to_full(h_segments, positions, n_points, dx):
    """Convert piecewise segments to full function."""
    h = np.zeros(n_points)
    for i in range(len(h_segments)):
        start = int(positions[i] / dx)
        end = int(positions[i+1] / dx)
        if end > start:
            h[start:end] = h_segments[i]
    return h

def full_to_segments(h, positions, n_points, dx):
    """Convert full function to piecewise segments."""
    h_segments = np.zeros(len(positions) - 1)
    for i in range(len(h_segments)):
        start = int(positions[i] / dx)
        end = int(positions[i+1] / dx)
        if end > start:
            h_segments[i] = np.mean(h[start:end])
    return h_segments
