import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):

        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def local_search(h_vec, max_iter=100):
        h_vec = np.array(h_vec)
        current_C5 = compute_C5(h_vec)
        best_h = h_vec.copy()
        best_C5 = current_C5
        
        for _ in range(max_iter):
            grad = np.zeros_like(h_vec)
            for i in range(len(h_vec)):
                eps = 1e-4
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = project_feasible(h_plus)
                h_minus = project_feasible(h_minus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
                corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                corr_max_plus = np.max(corr_plus)
                corr_max_minus = np.max(corr_minus)
                
                grad[i] = (corr_max_plus - corr_max_minus) / (2 * eps)
            
            # Adaptive step size
            def obj_step(alpha):
                new_h = h_vec - alpha * grad
                new_h = project_feasible(new_h)
                return compute_C5(new_h)
            
            # Line search
            result = minimize_scalar(obj_step, bounds=(0, 1), method='bounded')
            alpha = result.x
            
            h_vec = h_vec - alpha * grad
            h_vec = project_feasible(h_vec)
            
            new_C5 = compute_C5(h_vec)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_vec.copy()
        
        return best_h, best_C5
    
    def mutate(h_vec, rate=0.1):
        h_new = h_vec.copy()
        for i in range(len(h_vec)):
            if np.random.random() < rate:
                h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.05), 0, 1)
        return project_feasible(h_new)

    resolutions = [50, 75, 100, 150, 200]
    best_solution = None
    best_C5_val = float('inf')

            break
            
        n_points = res

        population = []
        for _ in range(population_size):
            h_test = mutate(best_h if best_h is not None else h, 0.2)
            population.append(h_test)

                break
                
            # Local search on each individual
            for i in range(len(population)):
                population[i], c5 = local_search(population[i], 50)
                if c5 < best_C5_val:
                    best_C5_val = c5
                    best_solution = population[i].copy()
            
            # Selection and reproduction
            population.sort(key=compute_C5)
            new_population = population[:5]  # Keep top 5
            
            # Reproduction
            while len(new_population) < population_size:
                parent1 = new_population[np.random.randint(0, 5)]
                parent2 = new_population[np.random.randint(0, 5)]
                
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                child = mutate(child, 0.1)
                new_population.append(child)
                
            population = new_population
    
    # Final refinement
    if best_solution is not None:
        refined_solution, final_C5 = local_search(best_solution, 100)
        if final_C5 < best_C5_val:
            best_C5_val = final_C5
            best_solution = refined_solution
    
    return (best_solution.tolist(), best_C5_val, len(best_solution))
