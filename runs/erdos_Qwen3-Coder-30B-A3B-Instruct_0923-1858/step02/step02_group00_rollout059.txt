import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        n_points = 45

        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def subsample_shifts(n, max_shifts=20):
        # Sample a few shifts that are representative for gradient estimation
        shifts = list(range(-n//2 + 1, n//2))
        if len(shifts) <= max_shifts:
            return shifts
        return np.random.choice(shifts, max_shifts, replace=False).tolist()
    
    # Use adaptive learning rate
    def adaptive_eta(iteration, base_eta=0.01, decay=0.98):
        return base_eta * (decay ** (iteration // 10))
    
    # Main optimization loop
    iter_count = 0
    max_patience = 50
    patience = 0
    perturb_prob = 0.1
    
    while elapsed() < budget_s * 0.95:
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Perturb a fraction of values
                idxs = np.random.choice(n_points, size=int(perturb_prob * n_points), replace=False)
                for i in idxs:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.03), 0, 1)
                h = project_feasible(h)
                patience = 0

        # Compute gradient estimation using finite differences
        eps = 1e-4
        h_grad = np.zeros_like(h)
        shifts = subsample_shifts(n_points, 15)
        
        # Get the shift that maximizes overlap for this h
        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(h) - 1)
        
        # Estimate gradient for few shifts
        # For efficiency, we use the one shift that gave the max at this step
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_feasible(h_plus)
            h_minus = project_feasible(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            h_grad[i] = (corr_plus[max_k_idx] - corr_minus[max_k_idx]) / (2 * eps)
        
        # Update with gradient descent
        eta = adaptive_eta(iter_count)
        h = h - eta * h_grad
        h = project_feasible(h)
        
        iter_count += 1
        
        # Occasionally relocate to avoid local minima
        if iter_count % 200 == 0:
            # Pick a random point and slightly shift it
            i = np.random.randint(0, n_points)
            h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
            h = project_feasible(h)
            
    # Final evaluation
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, len(best_h))
