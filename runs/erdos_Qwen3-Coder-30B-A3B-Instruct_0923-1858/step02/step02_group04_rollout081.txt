import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    try:

        n_points = len(h)
    except NameError:
        # Create a triangular initial solution
        n_points = 101
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.99
    min_T = 1e-6
    mutation_step = 0.1
    max_iters = 100000
    
    start_time = time.time()
    time_limit = budget_s * 0.98
    
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
            
        # Generate new candidate by perturbing current solution
        h_new = h.copy()
        num_perturb = max(1, int(0.05 * n_points))  # 5% of points
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            delta = np.random.normal(0, mutation_step)
            h_new[idx] += delta
        
        # Project new solution to feasible region
        h_new = project_feasible(h_new)
        
        # Compute objective values
        current_C5 = compute_C5(h)
        new_C5 = compute_C5(h_new)
        
        # Simulated Annealing acceptance criterion
        if new_C5 < current_C5 or np.random.random() < np.exp(-(new_C5 - current_C5) / T):
            h = h_new.copy()
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        
        # Update temperature and mutation step
        T = max(min_T, T * cooling_rate)
        mutation_step = 0.1 * (T / 1.0)  # Scale mutation with temperature
        
        iter_count += 1
        if iter_count >= max_iters:
            break
    
    return (best_h.tolist(), best_C5, len(best_h))
