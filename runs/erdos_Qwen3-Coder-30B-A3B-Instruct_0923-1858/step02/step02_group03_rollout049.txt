import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = initial_h / (np.sum(initial_h) * dx)
    
    start_time = time.time()
    max_time = budget_s - 10
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = float('inf')
    
    # Try several warm-starts with different initializations
    for attempt in range(5):
        if time.time() - start_time > max_time:
            break
            
        # Experiment with different initializations
        if attempt == 0:
            h = initial_h.copy()
        elif attempt == 1:
            # Step function
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
            h = h / (np.sum(h) * dx)
        elif attempt == 2:
            # Cosine-based
            t = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.cos(np.pi * t)
            h = h / (np.sum(h) * dx)
        elif attempt == 3:
            # Sin-based
            t = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(np.pi * t)
            h = h / (np.sum(h) * dx)
        else:  # attempt == 4
            # Random with constraints
            h = np.random.rand(n_points)
            h = h / (np.sum(h) * dx)
        
        # Local optimization with coordinate descent-like approach
        for iteration in range(1000):
            if time.time() - start_time > max_time:
                break
                
            # Use a more robust gradient estimator
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus / (np.sum(h_plus) * dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus / (np.sum(h_minus) * dx)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                
                grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
            # Adaptive step size
            lr = 0.005 / (1.0 + iteration/100.0)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / (np.sum(h_new) * dx)
            
            new_C = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
            
            if new_C < best_C:
                h = h_new
                best_C = new_C
                best_h = h.copy()
            else:
                # Occasionally perturb to escape local minima
                if iteration % 50 == 0:
                    noise = np.random.normal(0, 0.005, len(h))
                    h_perturbed = h + noise
                    h_perturbed = np.clip(h_perturbed, 0, 1)
                    h_perturbed = h_perturbed / (np.sum(h_perturbed) * dx)
                    perturbed_C = np.max(correlate(h_perturbed, 1 - h_perturbed, mode='full')) * dx
                    if perturbed_C < best_C:
                        h = h_perturbed
                        best_C = perturbed_C
                        best_h = h.copy()
    
    return (best_h, best_C, n_points)
