import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_time = budget_s - 10
    start_time = time.time()
    
    try:

        if len(h) != n_points:

    except NameError:
        h = np.ones(n_points) * 0.5  # uniform initial guess
        h = project(h, dx)
    
    # Track best solution
    best_h = h.copy()
    best_C = evaluate(h, dx)
    
    # Collect all shifts for constraint building
    shifts = []
    
    # Cutting plane loop
    iter_count = 0
    max_iter = 1000
    
    while time.time() - start_time < max_time and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate which shift is most violating
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr_val = corr[max_idx]
        
        if max_corr_val <= best_C:
            break
            
        # Add constraint: S(k) <= t
        # S(k) = 1 - \int h(x) h(x+k) dx = 1 - dot(h, roll(h, k))
        # So: 1 - dot(h, roll(h, k)) <= t => dot(h, roll(h, k)) >= 1 - t
        
        k = max_idx - (len(h) - 1)  # Convert full convolution index to shift k
        shifts.append(k)
        
        # Now you need to form the constraint in optimization framework
        # Instead, do a simple adaptive step with refinement
        # Modify h slightly along a direction that reduces the max correl
        h_new = h.copy()
        h_plus = h.copy()
        h_minus = h.copy()
        eps = 1e-4
        
        # Find gradients for each point
        grad = np.zeros_like(h)
        for j in range(len(h)):
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
            
            h_plus[j] -= eps
            h_minus[j] += eps
        
        # Simple gradient descent
        lr = 0.005
        h_new = h - lr * grad
        h_new = project(h_new, dx)
        
        # Update if better
        new_C = evaluate(h_new, dx)
        if new_C < best_C:
            h = h_new
            best_C = new_C
            best_h = h.copy()
        else:
            # Try to escape by slightly modifying the h
            noise = np.random.normal(0, 0.005, len(h))
            h = project(h + noise, dx)
    
    # Final check
    final_C = evaluate(best_h, dx)
    
    return (best_h, final_C, n_points)
