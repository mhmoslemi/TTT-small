import numpy as np
import time

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    n_points = len(h)
    # Compute correlation using manual implementation for clarity
    corr = np.zeros(2 * n_points - 1)
    for i in range(n_points):
        for j in range(n_points):
            lag = j - i
            if 0 <= lag + n_points - 1 < len(corr):
                corr[lag + n_points - 1] += h[i] * (1 - h[j])
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h
