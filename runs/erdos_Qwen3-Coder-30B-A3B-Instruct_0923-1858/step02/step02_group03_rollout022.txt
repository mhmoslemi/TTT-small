import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    try:
        initial_h = initial_h_values
        if len(initial_h) != n_points:
            raise ValueError("Wrong length")
    except NameError:
        # Create a smooth, feasible starting point
        t = np.linspace(0, 2, n_points)
        initial_h = np.exp(-((t - 1)**2) / 0.2)  # Gaussian peak at center
        initial_h = project(initial_h, dx)
    
    # Use L-BFGS optimization with projected gradient
    max_time = budget_s - 10
    start_time = time.time()
    
    h = initial_h.copy()
    best_h = initial_h.copy()
    best_C = evaluate(h, dx)
    
    # For initial phase, faster gradient estimate
    lr = 0.05
    
    for i in range(50000):
        if time.time() - start_time > max_time:
            break
            
        # Compute gradient: finite difference estimation
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus, dx)
            
            corr_plus = correlate(h_plus, 1-h_plus, mode='full')
            corr_minus = correlate(h_minus, 1-h_minus, mode='full')
            
            grad[j] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        
        # Update step
        h_new = h - lr * grad
        h_new = project(h_new, dx)
        
        # Test
        current_C = evaluate(h_new, dx)
        if current_C < best_C:
            h = h_new
            best_C = current_C
            best_h = h.copy()
            lr = min(0.1, lr * 1.01)
        else:
            h = h_new
            lr = max(1e-6, lr * 0.99)
            
        # Occasionally add noise
        if i % 100 == 0 and np.random.rand() < 0.05:
            noise = np.random.uniform(-0.01, 0.01, len(h))
            h = project(h + noise, dx)
    
    return (best_h, best_C, n_points)
