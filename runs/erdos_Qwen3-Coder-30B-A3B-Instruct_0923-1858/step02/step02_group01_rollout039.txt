import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    corr = correlate(h, 1 - h, mode='full')
    # The maximum is at the center of the full correlation
    max_corr = np.max(corr)
    return max_corr * dx

def project(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

def generate_individual(n_points, dx):
    """Generate a feasible individual."""
    h = np.random.uniform(0, 1, n_points)
    return project(h, dx)

def mutate(h, dx, mutation_rate=0.1):
    """Apply mutation to an individual."""
    mutated = h.copy()
    if np.random.rand() < mutation_rate:
        perturbation = np.random.uniform(low=-0.1, high=0.1, size=h.shape)
        mutated += perturbation
        mutated = np.clip(mutated, 0, 1)
        mutated = project(mutated, dx)
    return mutated

def crossover(parent1, parent2, dx, crossover_prob=0.8):
    """Perform uniform crossover between two parents."""
    if np.random.rand() < crossover_prob:
        # Uniform crossover
        mask = np.random.rand(len(parent1)) > 0.5
        offspring1 = np.where(mask, parent1, parent2)
        offspring2 = np.where(mask, parent2, parent1)
        # Project to ensure feasibility
        offspring1 = project(offspring1, dx)
        offspring2 = project(offspring2, dx)
        return offspring1, offspring2
    else:
        return parent1.copy(), parent2.copy()

def tournament_selection(population, fitness, tournament_size=3):
    """Select one individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index].copy()
